    2·n·max(conv(a,a)) / (∑ a)²

The optimiser works in four stages

1️⃣  **Seed generation** – a large, diverse collection of normalised
    candidates (uniform, geometric, power‑law, beta‑like, half‑sine,
    exponential, logistic, random Dirichlet, sparse patterns, and the
    previous best solution if available).

2️⃣  **Adam on a smooth max** – each seed is refined by Adam applied to a
    log‑sum‑exp (soft‑max) approximation of the `max` operator.
    The smoothing parameter β is increased slowly so that the optimiser
    gradually focuses on the true max.

3️⃣  **Exact sub‑gradient mirror descent** – from the best Adam result we
    perform exponentiated‑gradient steps using the true sub‑gradient of
    `max(conv)`.  The step‑size is η=0.7/√t.

4️⃣  **High‑frequency simulated‑annealing** – a cheap incremental update
    of the full convolution makes it possible to perform millions of
    cheap moves.  Two types of moves are used:

        • random Dirichlet re‑allocation over 2‑4 randomly chosen indices,
        • *targeted* re‑allocation that preferentially touches indices
          participating in the current maximal convolution entry.

    The temperature follows a geometric cooling schedule.
    The SA phase is the most time‑consuming and therefore dominates the
    budget.

All phases respect the global time budget (default 1000 s) and the best
sequence found is returned as a plain ``list[float]``.  The routine works
with any length up to a few ten‑thousands; the default set of lengths is
[2000, 4000, 6000, 8000] but the search automatically aborts when the
deadline is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a simplex‑normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i  (0<r<1)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_extended(n: int, r: float) -> np.ndarray:
    """Same as _geom, but r may be very close to 1 (e.g. 0.99)."""
    return _geom(n, r)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    """a[i] = exp(-i/sigma) (slow exponential decay)."""
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    """a[i] = 1/(1+exp((i-offset)/scale))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet draw (conc < 1 → very sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""

    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families (coarse)
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))

    # geometric families – values close to 1 (very gentle decay)
    for r in np.linspace(0.90, 0.99, 5):
        seeds.append(_geom_extended(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    # beta‑like shapes (asymmetric)
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays with several time‑constants
    for sigma in (n / 20, n / 10, n / 5, n / 2):
        seeds.append(_exp_decay(n, sigma))

    # logistic shapes – a fast drop near the beginning
    for scale in (n / 20, n / 10, n / 5):
        offset = scale * 2   # shift a bit to the left
        seeds.append(_logistic_shape(n, offset, scale))

    # random Dirichlet draws (different concentrations)
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse previous solution if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 15000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp (soft‑max) approximation of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)                     # parameters of a softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness, grows geometric
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:d} it={it:d} best={cur_val:.6f}")

        # ---- smooth max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # max for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum Σ w_k·conv_k  w.r.t. a
        # conv = a * a  ⇒  ∂/∂a = 2·(rev(a) * w)
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping to keep updates stable
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:d} it={it:d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)                # step‑size schedule
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with incremental convolution updates.
    Two move types are used:
        * random Dirichlet re‑allocation on 2‑4 positions,
        * targeted re‑allocation that touches indices contributing to the
          current maximal convolution entry.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)       # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initial temperature proportional to current objective
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, min_temp)

        # decide which move type to use
        if rng.random() < 0.5:
            # ----------- random move (Dirichlet on 2‑4 indices) -----------
            k = rng.integers(2, 5)                     # 2‑4 indices
            idx = rng.choice(n, size=k, replace=False)
        else:
            # ----------- targeted move ---------------------------------
            # indices that take part in the current maximal convolution entry
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            # weight each index by its contribution a[i]·a[k_star‑i]
            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                # pick one index proportionally to its contribution
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)

            # pick a second index uniformly from the complement
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # new allocation for the selected positions
        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            # a*Δ + Δ*a  (commutative)
            conv_new[p : p + n] += 2.0 * delta * cur
            # Δ*Δ term (self)
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i·Δ_j for i≠j (appear twice)
        for i in range(idx.size):
            for j in range(i + 1, idx.size):
                pos = idx[i] + idx[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:d} it={it:d} best={new_obj:.6f}")

    return best_seq, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Grab the previous construction (if any) to seed the search.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Lengths to try – larger lengths allow finer discretisation but cost more.
    lengths = [2000, 4000, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Phase 0 – quick evaluation of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on smooth max (only the most promising seeds)
        # ------------------------------------------------------------------
        # keep only top‑k seeds (by seed objective) for Adam to save time
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda t: t[1])
        top_k = min(4, len(seed_vals))
        for s, _ in seed_vals[:top_k]:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=25000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=500000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – High‑frequency simulated‑annealing (longest phase)
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            # allocate a generous number of SA iterations but respect the deadline
            remaining = deadline - time.time()
            # rough estimate: ~0.00008 s per SA iteration for n≈4000
            # we conservatively target up to 2 million moves if time permits.
            target_iters = int(min(2_000_000, remaining * 12_000))
            seq, val = _sa_incremental_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=target_iters,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Optional verification with the provided evaluate_sequence
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
