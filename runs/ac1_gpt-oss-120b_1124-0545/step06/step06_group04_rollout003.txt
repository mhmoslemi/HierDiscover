"""
Propose a step‑function that improves the self‑convolution infinity‑norm.
The search combines several complementary heuristics:

1.  A very diverse seed pool (uniform, geometric, power‑law, β‑distribution,
    Gaussian, sine windows, Dirichlet draws, and an up‑scaled previous best).
2.  Adam optimisation on a smooth “log‑sum‑exp” surrogate of the max‑convolution.
    The smoothing parameter β is increased to a large value (≈ 500) so that the
    surrogate becomes very close to the true max.
3.  Exact sub‑gradient mirror descent on the true max‑convolution entry.
4.  A short alternating linear‑programming refinement:
    for a fixed sequence f we solve  
        minimise   t
        s.t.       (f * g)_k ≤ t   (for all k)
                   Σ g_i = 1
                   g_i ≥ 0
    The solution g is mixed with f (step‑size γ) which strictly reduces the peak.
    This exploits the linear program used in the literature but runs only a few
    iterations (the LP size is O(n²) but for n≤4000 it is fast with SciPy‑HiGHS).
5.  A cheap simulated‑annealing fine‑tuner that swaps tiny probability fragments.
6.  Final symmetrisation (averaging with the reversed sequence) – the optimal
    solution is known to be symmetric.

The routine tries the lengths 2000 → 4000 → 8000 (if time permits).  All
intermediate steps are aborted as soon as the global deadline (≈ 1000 s)
is reached.  The best sequence found is cached globally as ``height_sequence_1``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for many calls)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max converting logits to a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sin2_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _hanning(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)
    a[a < 0] = 0.0
    return a / a.sum()


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving total mass."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a large, diverse set of normalised start vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))
    seeds.append(_sine_window(n))
    seeds.append(_sin2_window(n))
    seeds.append(_hanning(n))
    for sigma in [0.12, 0.18, 0.25]:
        seeds.append(_gaussian(n, sigma_factor=sigma))
    seeds.append(_tent(n))

    # beta families – a dense grid
    for a in np.linspace(0.2, 2.5, 7):
        for b in np.linspace(0.2, 2.5, 7):
            seeds.append(_beta_seed(n, a, b))

    # Dirichlet draws
    for alpha in [0.05, 0.1, 0.3, 0.6, 1.0, 2.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrored copies (mirroring can help when the optimum is symmetric)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # upscale previous best if available
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.shape != (n,):
                up = _upscale(prev, n)
                seeds.append(up)
            else:
                seeds.append(prev / prev.sum())
        except Exception:
            pass

    # ensure non‑empty
    if not seeds:
        seeds.append(_uniform(n))

    return seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000,
                   beta0: float = 2.0,
                   beta_target: float = 500.0,
                   beta_mul: float = 1.001,
                   lr0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth max loss L = (1/β)·log Σ exp(β·conv).
    Returns best sequence (array) and its true objective.
    """
    v = np.log(seed + 1e-30)                # logits -> simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # point on simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ramp β up to tighten the surrogate
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # weights for smoothed max (log‑sum‑exp)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                     # under‑flow protection
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient w.r.t. a (via reversed convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through the soft‑max
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # Adam updates
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)               # simple decay
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step on the simplex
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               gamma: float = 0.3,
               max_lp_iters: int = 5) -> tuple[np.ndarray, float]:
    """
    Alternating linear‑programming refinement.
    For the current distribution f we solve:
        minimise   t
        s.t.       (f * g)_k ≤ t   (∀k)
                   Σ g_i = 1
                   g_i ≥ 0
    and then mix f ← (1‑γ)·f + γ·g.
    """
    if not _SCIPY_AVAILABLE:
        # SciPy missing – skip this stage
        return f, _objective(f)

    best_seq = f.copy()
    best_val = _objective(f)

    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        # Build the Toeplitz convolution matrix for f:
        # row k, column j contains f_{k‑j} if 0≤k‑j<n else 0
        A = diags(f, offsets=range(n), shape=(2 * n - 1, n), format='csr')

        # last column for variable t (negative coefficient because we write
        #   A·g - t ≤ 0)
        neg_t_col = csr_matrix((-np.ones(2 * n - 1),
                                (np.arange(2 * n - 1), np.zeros(2 * n - 1))),
                               shape=(2 * n - 1, 1))
        A_ub = hstack([A, neg_t_col], format='csr')
        b_ub = np.zeros(2 * n - 1)

        # equality Σ g_i = 1 (t does not appear)
        A_eq = csr_matrix(np.hstack([np.ones(n), np.array([0.0])]))
        b_eq = np.array([1.0])

        # variable bounds: g_i ≥ 0, t ≥ 0
        bounds = [(0.0, None)] * n + [(0.0, None)]

        # objective: minimise t
        c = np.zeros(n + 1)
        c[-1] = 1.0

        res = linprog(c,
                      A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            print(f"[LP] iteration {lp_it} failed: {res.message}")
            break

        g = res.x[:n]
        t_val = res.x[-1]

        # convex combination
        f = (1.0 - gamma) * f + gamma * g
        f = np.clip(f, 0.0, None)
        f /= f.sum()

        cur_val = _objective(f)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
            print(f"[LP] n={n:5d} iter={lp_it} obj={cur_val:.6f}")

        # very tiny improvement → stop early
        if best_val - cur_val < 1e-7:
            break

    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Lightweight Metropolis‑Hastings simulated annealing that moves tiny
    probability fragments between positions.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature proportional to the current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a small fraction of the source mass
        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n:5d} it={it:6d} best={new_val:.6f}")

        # occasional global jitter (helps escape plateaus)
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n:5d} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1️⃣  Build seed pool and keep a few promising starts
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # Quick evaluation of raw seeds
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(8, len(seed_vals))]]

        # Record the best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣  Phase‑1 : Adam optimisation from each top seed
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=12000,
                beta0=2.0,
                beta_target=500.0,
                beta_mul=1.001,
                lr0=0.8,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # 3️⃣  Sub‑gradient refinement (exact max)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=150000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # 4️⃣  LP‑based alternating refinement (only for modest n)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n and n <= 4000:
            seq, val = _lp_refine(global_best_seq, n, deadline,
                                  gamma=0.3, max_lp_iters=6)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall LP] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # 5️⃣  Simulated‑annealing fine‑tuner
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(
                global_best_seq, n, deadline, max_iters=None
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # 6️⃣  Symmetrisation (the optimum is known to be symmetric)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[symmetrisation] n={n} improved to {sym_val:.6f}")

        # Propagate the current best forward (for up‑scaling in the next length)
        previous = global_best_seq

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(2000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Cache for possible later calls
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
