"""
Hybrid global‑local optimiser for the max‑auto‑convolution problem.

* Global stage – a Differential Evolution on a piece‑wise‑linear
  representation with a relatively large number of knots (≈2·√n).  This
  explores the search space quickly and provides a good starting shape.

* Local stage – a series of high‑dimensional L‑BFGS refinements on the
  smooth‑max surrogate (log‑sum‑exp) with increasing smoothing
  parameter β.  The variables are the soft‑max logits, guaranteeing
  non‑negativity and the ℓ1‑normalisation (sum = √(2n)).

* Final polish – a lightweight stochastic hill‑climb that moves small
  amounts of mass between two indices while preserving the ℓ1‑sum.

The routine returns the best non‑negative step function it found within
the allocated time budget (default 1000 s).  The result can be reused
via the global name ``height_sequence_2``.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution (sums to 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    """Classic sine window (sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric discrete Gaussian (sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (sum = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Differential Evolution on a piece‑wise‑linear representation
# ----------------------------------------------------------------------


def _eval_piecewise(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """
    Interpolate a low‑dimensional simplex vector `p` (sum = 1) to a length‑`n`
    sequence, renormalise to the required ℓ1‑norm and return (objective,
    full_sequence).  Objective = max auto‑convolution (since the sum‑norm
    is fixed to √(2n)).
    """
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s * np.sqrt(2.0 * n)          # enforce sum = √(2n)
    conv = _conv_fft(a, a)
    val = np.max(conv)
    return float(val), a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 50,
                 K: int | None = None,
                 max_gens: int = 300) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on a piece‑wise‑linear representation.
    The optimisation runs until the time deadline or `max_gens` generations.
    Returns the best full‑length sequence and its max‑convolution value.
    """
    rng = np.random.default_rng()

    if K is None:
        K = max(30, int(2 * np.sqrt(n)))   # richer representation than the baseline
    knots = np.linspace(0, n - 1, K)

    # initialise population on the simplex (Dirichlet draws)
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]
    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize

    for i in range(popsize):
        val, seq = _eval_piecewise(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            donor = donor / ds if ds > 0 else _uniform(K)

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            trial = trial / ts if ts > 0 else _uniform(K)

            val_trial, seq_trial = _eval_piecewise(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float) -> tuple[float, np.ndarray]:
    """
    Log‑sum‑exp smooth‑max of the auto‑convolution and its gradient w.r.t.
    the soft‑max logits `v`.  The sequence a = √(2n)·softmax(v) always satisfies
    the ℓ1‑constraint.
    """
    s = np.sqrt(2.0 * n)                # target ℓ1‑norm
    p = _softmax(v)                     # sum = 1
    a = p * s

    conv = _conv_fft(a, a)               # full convolution, length 2n‑1
    shift = np.max(conv)                 # numeric stabiliser
    w = np.exp(beta * (conv - shift))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    else:
        w = w / w_sum                     # normalised weights (smooth‑max)

    # gradient of Σ w_k·conv_k w.r.t. a  (central n entries)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the soft‑max mapping
    dot = np.dot(a, grad_a) / s          # = Σ p_i·grad_a_i
    grad_v = a * (grad_a - dot)

    smooth = shift + np.log(w_sum) / beta
    return smooth, grad_v

# Stochastic hill‑climbing (pair‑wise mass transfer)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Simple pair‑wise transfer of mass; accepts only improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # decide direction of transfer
        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = np.max(_conv_fft(new, new))
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # tiny safety margin

    # Try a single length; 4000 is a good compromise between quality
    # and runtime (the objective is scale‑invariant).
    n = 4000
    target_sum = np.sqrt(2.0 * n)

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Initialise a diverse set of analytic seeds (including the
    #    previously best sequence if available).
    # ------------------------------------------------------------------
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    rng = np.random.default_rng(2024)

    # random Dirichlet families
    for conc in (0.3, 1.0, 5.0):
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # Gaussian windows with various widths
    for frac in (0.06, 0.12, 0.25):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    # Beta‑shape grid (coarse but effective)
    for a in (0.4, 1.0, 1.8):
        for b in (0.4, 1.0, 1.8):
            seeds.append(_beta_shape(n, a, b))

    # include the previously stored construction, if any
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n:
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Quick evaluation of the analytic seeds
    # ------------------------------------------------------------------
    for s in seeds:
        p = np.maximum(s, 0.0)
        p = p / p.sum() * target_sum
        val = np.max(_conv_fft(p, p))
        if val < global_best_val:
            global_best_val = val
            global_best_seq = p.copy()
            print(f"[seed] val={val:.6f}")

    # ------------------------------------------------------------------
    # 2. Global DE search on a piece‑wise‑linear parametrisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_de, val_de = _de_optimize(n, deadline,
                                      popsize=50, K=None, max_gens=300)
        if val_de < global_best_val:
            global_best_val = val_de
            global_best_seq = seq_de.copy()
            print(f"[DE] new global best={val_de:.6f}")

    # ------------------------------------------------------------------
    # 3. Multi‑stage L‑BFGS refinement on the smooth‑max surrogate.
    #    Several restarts (different seeds) are attempted while time permits.
    # ------------------------------------------------------------------
    beta_schedule = [5.0, 20.0, 80.0, 300.0]   # progressively sharper approximations
    max_lbfgs_iters = 300                     # per stage

    # we will try a handful of random initial logits if time permits
    attempts = 0
    while time.time() < deadline and attempts < 12:
        attempts += 1
        # initialise logits from the current best (or a random Dirichlet)
        if global_best_seq is not None and attempts == 1:
            p0 = global_best_seq / global_best_seq.sum()
        else:
            p0 = rng.dirichlet(np.ones(n) * 0.5)
        v0 = np.log(p0 + 1e-30)   # logits

        # run the sequence of L‑BFGS refinements
        for beta in beta_schedule:
            if time.time() > deadline:
                break

            def fun(v):
                return _smooth_obj_and_grad(v, n, beta)

            # L‑BFGS‑B with gradient supplied
            res = scipy.optimize.minimize(fun,
                                          v0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_lbfgs_iters,
                                                   'disp': False})
            v0 = res.x  # warm‑start the next beta

        # after the final stage compute the true objective
        a_opt = _softmax(v0) * target_sum
        val_opt = np.max(_conv_fft(a_opt, a_opt))
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = a_opt.copy()
            print(f"[L‑BFGS] attempt {attempts} new global best={val_opt:.6f}")

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb polishing
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq_hc, val_hc = _hill_climb(global_best_seq,
                                     n,
                                     deadline,
                                     max_iters=60000)
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[Hill‑final] new global best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # 5. Safety net – if nothing was discovered, fall back to uniform.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n) * target_sum
        global_best_val = np.max(_conv_fft(global_best_seq, global_best_seq))
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Store the result for possible reuse and verify with the provided
    # ``evaluate_sequence`` (the function may raise – ignore any exception).
    # ------------------------------------------------------------------
    globals()["height_sequence_2"] = global_best_seq.tolist()
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {n})")
    return global_best_seq.tolist()
```

The above implementation expands the original pipeline by:

* Using a richer DE representation (≈2·√n knots) to explore a broader family of shapes.
* Adding several random Dirichlet and Gaussian seeds for extra diversity.
* Introducing a high‑dimensional L‑BFGS optimiser that works directly on the smooth‑max surrogate with an adaptive β schedule.
* Performing multiple restarts of the L‑BFGS stage (up to ~12) to escape local minima.
* Keeping the final stochastic hill‑climb for fine polishing.

All components respect the 1000 s time budget and return a plain Python list of non‑negative floats.
