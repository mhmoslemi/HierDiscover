"""
Search for a non‑negative step function that reduces the
``evaluate_sequence`` objective.

The algorithm works in three stages:

1. **Deterministic seeds** – a flat “Catalan” shape derived from the
   generating function (1‑z)^‑½ (known to give a near‑constant
   convolution) and a uniform vector are evaluated.

2. **Massive random Dirichlet sampling** – while time permits we draw
   many independent Dirichlet(α=1) vectors of several lengths (2000,
   4000, 8000, 16000).  For each sample we compute  
   ``val = 2·n·max(conv(a,a))`` (the same quantity used inside
   ``evaluate_sequence`` when the vector is normalised).  The best
   sample so far is retained.

3. **Exact sub‑gradient mirror‑descent** – the current best vector is
   refined on the simplex using the true sub‑gradient of the
   ``max(conv)`` functional.  The update is an exponentiated‑gradient
   step (mirror descent with the KL‑divergence) and runs until the
   remaining time is exhausted.

The routine is time‑aware: it never exceeds the supplied budget
(`budget_s`, default 1000 s) and returns the best sequence it has
found as a plain ``list`` of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for many calls)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Returns the quantity that ``evaluate_sequence`` would compute
    when ``seq`` is already normalised to sum = 1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2i,i)/4**i (the series of (1‑z)^‑½).
    After truncation they give a near‑constant auto‑convolution.
    The vector is normalised to sum = 1.
    """
    r = np.empty(n, dtype=np.float64)
    c = 1.0  # i = 0 term
    r[0] = c
    for i in range(1, n):
        # recurrence: b_{i} = b_{i-1} * ((2i)*(2i-1)) / (4*i*i)
        # with i‑1 in the previous step
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        r[i] = c
    total = r.sum()
    if total == 0.0:
        return _uniform(n)
    return r / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling (α = 1)
# ----------------------------------------------------------------------


def _random_dirichlet_one(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(1,…,1) vector (i.e. normalised exponentials)."""
    # Using the gamma method: Gamma(1,1) == Exp(1)
    x = rng.exponential(1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Sub‑gradient mirror descent on the simplex
# ----------------------------------------------------------------------


def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the simplex.
    Stops when ``time.time()`` exceeds ``deadline``.
    Returns the best vector seen and its objective.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step with diminishing step size
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        # normalise in log‑space for numerical stability
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

Search for a non‑negative step function that minimises the
``evaluate_sequence`` objective.

The algorithm proceeds in three time‑aware stages:

1. **Deterministic seeds** – a flat “Catalan” shape (coefficients of
   (1‑z)^‑½ truncated to the chosen length) and a uniform vector are
   evaluated.  If a previous solution is available it is also inspected.

2. **Massive random Dirichlet sampling** – while time allows, many
   independent Dirichlet(α=1) vectors of several lengths (2000, 4000,
   8000, 16000) are drawn.  For each sample we compute the quantity

        val = 2·n·max(conv(a,a))

   (the same quantity that ``evaluate_sequence`` would return for a
   normalised vector).  The best sample seen so far is kept.

3. **Exact sub‑gradient mirror descent** – the current best vector is
   refined on the simplex using the true sub‑gradient of the ``max(conv)``
   functional.  The update is an exponentiated‑gradient (mirror descent
   with KL‑divergence) and runs until the remaining time budget is
   exhausted.

The routine is fully time‑aware, never exceeding the supplied budget
(`budget_s`, default 1000 s).  It returns the best sequence found as a

# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

    ``seq`` is already normalised to sum = 1:
