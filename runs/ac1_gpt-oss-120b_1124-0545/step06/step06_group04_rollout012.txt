    seeds = _make_seeds(n, previous)
    # add Dirichlet draws
    rng = np.random.default_rng(12345)  # Use fixed for reproducibility
    for alpha in [0.01, 0.03, 0.1, 0.2, 0.5]:
        for _ in range(5):
            seeds.append(_dirichlet_seed(n, alpha, rng))

    # add sparse seeds
    for k in [5, 10, 15]:
        for _ in range(5):
            indices = rng.choice(n, size=k, replace=False)
            seq = np.zeros(n)
            seq[indices] = 1.0/k
            seeds.append(seq)

    # evaluate seeds
    seed_vals = []
    for s in seeds:
        if time.time() > deadline:
            break
        seed_vals.append((s, _objective(s)))
    if not seed_vals:
        return None, None

    seed_vals.sort(key=lambda x: x[1])
    top_seeds = [s for s,_ in seed_vals[:min(8, len(seed_vals))]]

    best_seq = seed_vals[0][0]
    best_val = seed_vals[0][1]

    # Adam on top seeds
    for idx, sv in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(sv, n, deadline,
                max_iters=8000,
                beta0=2.0,
                beta_target=300.0,
                beta_mul=1.001,
                lr0=0.5)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq

    # subgradient
    if time.time() < deadline:
        seq, val = _subgradient_refine(best_seq, n, deadline,
                                       max_iters=50000)

            best_seq = seq

    # anneal

        sym /= sym.sum()

            best_seq = sym

"""
Propose a step‑function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear convolution.  The algorithm works in three
stages:

1.  **Coarse search** (small n≈200) – many diverse random seeds are
    generated (uniform, geometric, power‑law, Dirichlet draws,
    sparse spikes, Beta‑type windows).  Each seed is refined briefly with
    an Adam optimiser on a smooth max‐surrogate.  The best few
    candidates are kept.

2.  **Medium refinement** – the surviving candidates are up‑scaled to a
    medium size (≈800) and run through a longer exponentiated‑gradient
    sub‑gradient descent on the true `max` (the “mirror‑descent”
    method).  A lightweight simulated‑annealing phase follows.

3.  **Final scaling** – the best medium candidate is up‑scaled to the
    target length (4000 or 8000 depending on remaining time) and refined
    again with the same sub‑gradient + annealing pipeline.  A final
    symmetrisation (averaging with its mirror) often reduces the peak
    further.

All steps respect the global 1000 s budget.  The best sequence found
is cached globally as ``height_sequence_1`` for reuse in subsequent
runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sin2_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _hanning(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)
    a[a < 0] = 0.0
    return a / a.sum()


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) on (0,1) – interior points only."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _sparse_seed(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k uniformly‑chosen spikes with equal mass."""
    idx = rng.choice(n, size=min(k, n), replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / idx.size
    return a


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a diverse pool of start vectors (already normalised)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r > 0.0:
            for p in np.linspace(0.0, 1.8, 5):
                seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))
    seeds.append(_sine_window(n))
    seeds.append(_sin2_window(n))
    seeds.append(_hanning(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_tent(n))

    # beta families
    for a in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        seeds.append(_beta_seed(n, a, a))
        seeds.append(_beta_seed(n, a, a * 1.5))
        seeds.append(_beta_seed(n, a * 1.5, a))

    # Dirichlet draws (various concentrations)
    for alpha in [0.01, 0.03, 0.1, 0.2, 0.5, 0.8, 1.5, 3.0]:
        for _ in range(2):
            seeds.append(_dirichlet_seed(n, alpha, rng))

    # sparse random spikes
    for k in [5, 10, 15, 20]:
        for _ in range(2):
            seeds.append(_sparse_seed(n, k, rng))

    # mirrored copies
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # upscale previous best if provided
    if previous is not None:
        if previous.shape != (n,):
            try:
                up = _upscale(previous, n)
                seeds.append(up)
            except Exception:
                pass
        else:
            seeds.append(previous / previous.sum())

    if not seeds:
        seeds.append(_uniform(n))

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 6000,
    beta0: float = 2.0,
    beta_target: float = 500.0,
    beta_mul: float = 1.001,
    lr0: float = 0.5,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the smoothed max loss L = (1/β)·log Σ exp(β·conv).
    Returns the best sequence encountered and its true objective.
    """
    v = np.log(seed + 1e-30)                # logits → simplex via soft‑max
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # point on simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # increase β gradually, never exceed the target
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # smooth‑max weights (log‑sum‑exp trick)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                     # under‑flow guard
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient wrt a (via reversed convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 100000,
) -> tuple[np.ndarray | None, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the genuine sub‑gradient
    of max(conv).  A diminishing step‑size 1/√t is employed.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        # stabilise numerically
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray | None, float]:
    """
    Lightweight simulated‑annealing local search that swaps tiny fragments
    of mass between two positions.  The temperature decays geometrically.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # starting temperature – a small fraction of the current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # ≤2.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global jitter to escape shallow wells
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    Returns the best sequence (as a Python ``list``) found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 999)  # independent stream for seeds
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # reuse previous best if the variable already exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Stage specifications – increasingly larger problems.
    # Each tuple: (target length, fraction of remaining time we would like to spend)
    stages = [(200, 0.25), (800, 0.35), (4000, 0.40)]

    global_best_seq = None
    global_best_val = float("inf")

    for stage_idx, (n, time_frac) in enumerate(stages):
        now = time.time()
        if now >= deadline:
            break

        # allocate a slice of the remaining budget
        remaining = deadline - now
        stage_deadline = now + min(remaining, budget_s * time_frac)

        print(f"\n=== Stage {stage_idx + 1}: length {n}, time budget {stage_deadline - now:.1f}s ===")

        # --------------------------------------------------------------
        # Build a rich pool of starting points
        # --------------------------------------------------------------
        seeds = _make_seeds(n, previous, rng)

        # --------------------------------------------------------------
        # Quick evaluation of raw seeds; keep the top few (≤12)
        # --------------------------------------------------------------
        seed_vals = []
        for s in seeds:
            if time.time() >= stage_deadline:
                break
            seed_vals.append((s, _objective(s)))
        if not seed_vals:
            continue
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(12, len(seed_vals))]]

        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] new global best={global_best_val:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – Adam optimisation from each top seed
        # --------------------------------------------------------------
        for idx, sv in enumerate(top_seeds):
            if time.time() >= stage_deadline:
                break
            seq, val = _adam_optimize(
                sv,
                n,
                stage_deadline,
                max_iters=12000,
                beta0=2.0,
                beta_target=800.0,
                beta_mul=1.001,
                lr0=0.5,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] stage {stage_idx+1} seed {idx} best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – sub‑gradient refinement (longer for the current best)
        # --------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
            and time.time() < stage_deadline
        ):
            seq, val = _subgradient_refine(
                global_best_seq,
                n,
                stage_deadline,
                max_iters=120000,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] stage {stage_idx+1} best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – simulated‑annealing fine tuning
        # --------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
            and time.time() < stage_deadline
        ):
            seq, val = _anneal_optimize(
                global_best_seq,
                n,
                stage_deadline,
                max_iters=None,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] stage {stage_idx+1} best={val:.6f}")

        # --------------------------------------------------------------
        # Symmetrisation – averaging with the mirror often reduces the peak
        # --------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
            and time.time() < stage_deadline
        ):
            sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[symmetrisation] stage {stage_idx+1} improved to {sym_val:.6f}")

        # Prepare for the next stage (up‑scale the current best)
        previous = global_best_seq

        # Stop early if we have already beaten the target
        if global_best_val < 1.503:
            print(f"Target reached early (objective {global_best_val:.6f}); aborting further stages.")
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Cache the result globally for any subsequent calls
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
