def _smooth_max_and_grad(v: np.ndarray, n: int, beta: float):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    # avoid overflow with large beta: shift conv - shift may be negative; exponentiate
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    # gradient wrt a:
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    # gradient wrt logits due to softmax:
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    # smooth max value:
    smooth_val = shift + np.log(sum_exp) / beta
    return smooth_val, grad_v

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_analytic_seeds(
    n: int, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Collect a diverse set of normalised analytic start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # beta‑like shapes – finer grid than the previous run
    alphas = np.linspace(0.1, 3.0, 13)   # extended range
    betas = np.linspace(0.1, 3.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet families (different concentrations)
    rng = np.random.default_rng(2023)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # softmax of Gaussian logits (non‑parametric)
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Random piece‑wise‑linear seeds
# ----------------------------------------------------------------------


def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random piecewise‑linear distribution via Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Fast objective (sum is assumed to be 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def _smooth_max_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (smooth_max, grad) where smooth_max = (1/β)·log Σ exp(β·conv)
    and grad is the gradient w.r.t. the logits v (soft‑max parametrisation).
    """
    a = _softmax(v)                     # stays on the simplex
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    # safely compute the soft‑max weights
    w = np.exp(beta * (conv - shift))
    w_sum = w.sum()
    w /= w_sum

    # gradient w.r.t. a: 2·conv(w, rev_a) central part
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]

    # convert to gradient w.r.t. logits (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    # smooth‑max value
    smooth_val = shift + np.log(w_sum) / beta
    return smooth_val, grad_v


def _smooth_obj(v, n, beta):
    """Wrapper required by scipy.optimize – returns (f, grad)."""
    return _smooth_max_and_grad(v, n, beta)


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _lbfgs_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float],
    max_iter_each: int = 300,
) -> tuple[np.ndarray | None, float]:
    """
    Runs a sequence of L‑BFGS optimisations with increasing smoothing β.
    Returns the best sequence and its true objective value.
    """
    v = np.log(seed + 1e-30)  # logits
    best_seq = None
    best_val = float("inf")

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        res = scipy.optimize.minimize(
            _smooth_obj,
            v,
            args=(n, beta),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter_each, "disp": False},
        )
        v = res.x
        a = _softmax(v)
        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.2f} it={res.nit} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 30000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a, n)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent step in log‑space (exponentiated‑gradient)
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # numerical stabilisation
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# L²‑flattening descent (drives a*a towards a constant)
# ----------------------------------------------------------------------


def _l2_flatten_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on ½‖a*a – μ·1‖², where μ = 1/(2n‑1).
    The update is performed in log‑space to stay on the simplex.
    """
    a = start_seq.copy()
    a = a / a.sum()
    mu = 1.0 / (2 * n - 1)          # target convolution value
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_fft(residual, rev_a)   # length 3n‑2
        grad = 4.0 * grad_full[n - 1 : n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L2Flat] it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted stochastic hill‑climb
# ----------------------------------------------------------------------


def _targeted_hill_climb(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 30000,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers, but the donor is sampled proportional to its
    contribution to the currently maximal convolution coefficient.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total <= 0.0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[HillClimb] it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Upsample helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_len` and renormalise."""
    old_len = len(seq)
    if old_len == new_len:
        return seq.copy()
    old_x = np.arange(old_len, dtype=np.float64)
    new_x = np.linspace(0, old_len - 1, new_len, dtype=np.float64)
    up = np.interp(new_x, old_x, seq)
    up = up / up.sum() if up.sum() != 0 else _uniform(new_len)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # warm‑start from a previously found solution, if any
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Multi‑scale optimisation – start small, then enlarge
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]   # increasing resolution

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # ----------------------------------------------------------------
        # 1. Seed generation
        # ----------------------------------------------------------------
        analytic_seeds = _make_analytic_seeds(n, previous)
        rng = np.random.default_rng()
        random_seeds = [_random_piecewise_linear(n, rng) for _ in range(300)]

        all_seeds = analytic_seeds + random_seeds
        top_k = 6
        top_candidates: list[tuple[float, np.ndarray]] = []

        for s in all_seeds:
            val = _objective(s, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")
            top_candidates.append((val, s))

        # keep the best few seeds for intensive polishing
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:top_k]

        # ----------------------------------------------------------------
        # 2. Polishing each candidate
        # ----------------------------------------------------------------
        for cid, (seed_val, seed_seq) in enumerate(top_candidates, 1):
            if time.time() > deadline:
                break
            print(f"\n--- Polishing candidate #{cid}/{len(top_candidates)} (n={n}) ---")

            # (a) L‑BFGS on smooth‑max surrogate
            beta_sched = [0.5, 2.0, 8.0, 30.0, 100.0, 300.0]
            seq_opt, val_opt = _lbfgs_optimize(
                seed_seq,
                n,
                deadline,
                beta_schedule=beta_sched,
                max_iter_each=400,
            )
            if seq_opt is not None and val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"[polish‑LBFGS] n={n} best={val_opt:.6f}")

            # (b) Exact sub‑gradient descent
            seq_sg, val_sg = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=25000
            )
            if val_sg < global_best_val:
                global_best_val = val_sg
                global_best_seq = seq_sg.copy()
                print(f"[polish‑SubGrad] n={n} best={val_sg:.6f}")

            # (c) L²‑flattening descent
            seq_l2, val_l2 = _l2_flatten_refine(
                global_best_seq, n, deadline, max_iters=20000
            )
            if val_l2 < global_best_val:
                global_best_val = val_l2
                global_best_seq = seq_l2.copy()
                print(f"[polish‑L2Flat] n={n} best={val_l2:.6f}")

            # (d) Targeted hill‑climb
            seq_hc, val_hc = _targeted_hill_climb(
                global_best_seq, n, deadline, max_iters=25000
            )
            if val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[polish‑Hill] n={n} best={val_hc:.6f}")

        # ----------------------------------------------------------------
        # 3. Quick extra L‑BFGS pass on the current global best (if time permits)
        # ----------------------------------------------------------------
        if time.time() < deadline - 5.0 and global_best_seq is not None:
            extra_seq, extra_val = _lbfgs_optimize(
                global_best_seq,
                n,
                deadline,
                beta_schedule=[500.0, 1000.0],
                max_iter_each=200,
            )
            if extra_seq is not None and extra_val < global_best_val:
                global_best_val = extra_val
                global_best_seq = extra_seq.copy()
                print(f"[extra‑LBFGS] n={n} best={extra_val:.6f}")

        # upscale the current best for the next (larger) length
        previous = _upsample(global_best_seq, n)

    # ----------------------------------------------------------------------
    # Final verification (fallback to uniform if something went wrong)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)

    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[verification] sequence rejected – falling back to uniform")
        global_best_seq = _uniform(len(global_best_seq))
        global_best_val = _objective(global_best_seq, len(global_best_seq))

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
