"""
Proposes a non‑negative step function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three stages:

1. **Broad analytic sampling** – a dense grid of Beta‑shaped families
   (including the arcsine distribution) and a few simple geometric families
   is evaluated on a chosen length (8000, falling back to 4000).  The best
   few seeds are retained.

2. **Deterministic polishing** – each retained seed is refined by a
   sequence of optimiser passes:
        * Adam on a smooth‑max surrogate (log‑softmax parametrisation).
        * Exact sub‑gradient descent (mirror‑descent) on the true max.
        * L₂‑flattening descent that pushes the auto‑convolution towards a
          constant.
        * Targeted stochastic hill‑climb that moves mass away from the
          current maximal convolution entry.

   All passes respect the remaining time budget and constantly update the
   global best sequence.

3. **Verification** – the final candidate is passed to the provided
   `evaluate_sequence` routine; if it is rejected we fall back to the
   uniform distribution.

The routine adapts automatically to the 1000 s time budget, keeps the
best sequence in the global variable `height_sequence_1` for warm‑starts,
and returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on a simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric progression a_i ∝ r**i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric sequence (mirrored)."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw."""
    a = rng.dirichlet(np.full(n, conc))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Fast objective (assumes Σ a = 1)
# ----------------------------------------------------------------------


def _objective_normed(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a already ℓ₁‑normalised."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation with a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing coefficient – start modest, grow exponentially
    beta = 2.0
    beta_mul = 1.025

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # increase smoothness
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent (mirror‑descent) on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_normed(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated‑gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)    # stabilise
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

    return best_seq, best_val


def _l2_flatten_refine(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on ½‖a*a – μ·1‖² where μ = 1/(2n‑1) (uniform target).
    The update is performed in log‑space to stay on the simplex.
    """
    a = start_seq.copy()
    a = a / a.sum()
    mu = 1.0 / (2 * n - 1)

    best_seq = a.copy()
    best_val = _objective_normed(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        residual = conv - mu                # r = b - μ
        rev_a = a[::-1]
        grad_full = _conv_fft(residual, rev_a)   # length 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective_normed(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L2Flat] n={n:5d} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


def _targeted_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Random mass transfers. The donor is sampled proportionally to its
    contribution to the currently maximal convolution coefficient.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_normed(cur)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02   # relative size of a single transfer

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions to the maximal coefficient
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total <= 0.0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective_normed(new)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        # geometric decay of step size
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

def propose_candidate(budget_s: int = 1000, seed: int = 123) -> list[float]:
    """
    Search for a step‑function that minimises the given evaluation metric.
    The routine mixes a dense analytic grid search with several deterministic
    refinements while respecting the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for final verification
    deadline = start_time + budget_s - 2.0

    # try a large length first – larger n gives a lower theoretical bound
    lengths = [8000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    # reuse the sequence from a previous run if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.ndim != 1:
                previous = None
        except Exception:
            previous = None

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        top_candidates = []   # (value, sequence) tuples

        # ------------------------------------------------------------------
        # 1. Analytic families – dense grid
        # ------------------------------------------------------------------
        # uniform (baseline)
        seq = _uniform(n)
        val = _objective_normed(seq)
        top_candidates.append((val, seq.copy()))
        if val < global_best_val:
            global_best_val, global_best_seq = val, seq.copy()
            print(f"[seed] uniform new global best={val:.6f}")

        # arcsine (Beta(0.5,0.5))
        seq = _beta_shape(n, 0.5, 0.5)
        val = _objective_normed(seq)
        top_candidates.append((val, seq.copy()))
        if val < global_best_val:
            global_best_val, global_best_seq = val, seq.copy()
            print(f"[seed] arcsine new global best={val:.6f}")

        # ---- fine Beta grid ------------------------------------------------
        alphas = np.linspace(0.25, 1.75, 31)   # 31 values → 961 combos
        betas = alphas
        for a in alphas:
            for b in betas:
                if time.time() > deadline - 5.0:
                    break
                seq = _beta_shape(n, a, b)
                val = _objective_normed(seq)
                top_candidates.append((val, seq.copy()))
                if val < global_best_val:
                    global_best_val, global_best_seq = val, seq.copy()
                    print(f"[beta] α={a:.3f} β={b:.3f} new global best={val:.6f}")

        # ---- symmetric / plain geometric families -------------------------
        for r in np.linspace(0.90, 0.9999, 25):
            if time.time() > deadline - 5.0:
                break
            seq = _geom(n, r)
            val = _objective_normed(seq)
            top_candidates.append((val, seq.copy()))
            if val < global_best_val:
                global_best_val, global_best_seq = val, seq.copy()
                print(f"[geom] r={r:.5f} new global best={val:.6f}")

            seq = _sym_geom(n, r)
            val = _objective_normed(seq)
            top_candidates.append((val, seq.copy()))
            if val < global_best_val:
                global_best_val, global_best_seq = val, seq.copy()
                print(f"[sym_geom] r={r:.5f} new global best={val:.6f}")

        # ---- random Dirichlet draws ----------------------------------------
        rng = np.random.default_rng(seed + 999)
        for conc in (0.3, 0.7, 1.0, 2.0, 5.0):
            for _ in range(20):
                if time.time() > deadline - 5.0:
                    break
                seq = _random_dirichlet(n, conc, rng)
                val = _objective_normed(seq)
                top_candidates.append((val, seq.copy()))
                if val < global_best_val:
                    global_best_val, global_best_seq = val, seq.copy()
                    print(f"[dirich] conc={conc} new global best={val:.6f}")

        # include previous sequence if compatible
        if previous is not None and previous.shape == (n,):
            seq = previous / previous.sum()
            val = _objective_normed(seq)
            top_candidates.append((val, seq.copy()))
            if val < global_best_val:
                global_best_val, global_best_seq = val, seq.copy()
                print(f"[prev] new global best={val:.6f}")

        # keep only the best few seeds for polishing
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:10]   # at most ten seeds

        # ------------------------------------------------------------------
        # 2. Deterministic polishing of the best seeds
        # ------------------------------------------------------------------
        for idx, (seed_val, seed_seq) in enumerate(top_candidates):
            if time.time() > deadline:
                break
            print(f"\n--- Polishing seed #{idx + 1} (n={n}) ---")

            # (a) Adam on smooth‑max
            seq_adam, val_adam = _adam_optimize(seed_seq, n, deadline,
                                                max_iters=20000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val, global_best_seq = val_adam, seq_adam.copy()
                print(f"[polish‑Adam] new global best={val_adam:.6f}")

            # (b) Exact sub‑gradient descent
            seq_sg, val_sg = _subgradient_refine(global_best_seq, n, deadline,
                                                 max_iters=50000)
            if val_sg < global_best_val:
                global_best_val, global_best_seq = val_sg, seq_sg.copy()
                print(f"[polish‑SubGrad] new global best={val_sg:.6f}")

            # (c) L₂‑flattening descent
            seq_l2, val_l2 = _l2_flatten_refine(global_best_seq, n, deadline,
                                                max_iters=30000)
            if val_l2 < global_best_val:
                global_best_val, global_best_seq = val_l2, seq_l2.copy()
                print(f"[polish‑L2Flat] new global best={val_l2:.6f}")

            # (d) Targeted stochastic hill‑climb
            seq_hc, val_hc = _targeted_hill_climb(global_best_seq, n, deadline,
                                                  max_iters=30000)
            if val_hc < global_best_val:
                global_best_val, global_best_seq = val_hc, seq_hc.copy()
                print(f"[polish‑Hill] new global best={val_hc:.6f}")

        # ------------------------------------------------------------------
        # 3. Quick extra Adam pass on the current best (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0 and global_best_seq is not None:
            extra_seq, extra_val = _adam_optimize(global_best_seq, n, deadline,
                                                  max_iters=5000)
            if extra_seq is not None and extra_val < global_best_val:
                global_best_val, global_best_seq = extra_val, extra_seq.copy()
                print(f"[extra‑Adam] new global best={extra_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback if nothing was found (should never happen)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n_fallback = 4000
        global_best_seq = _uniform(n_fallback)
        global_best_val = _objective_normed(global_best_seq)
        print("[fallback] using uniform distribution")

    # ----------------------------------------------------------------------
    # Verification with the provided evaluate_sequence()
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[verification] sequence rejected – falling back to uniform")
        global_best_seq = _uniform(len(global_best_seq))
        global_best_val = _objective_normed(global_best_seq)

    # store for potential warm‑starts in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
