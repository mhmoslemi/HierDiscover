def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x≥0, Σx=s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)).  ``a`` is assumed non‑negative."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


def _normalise_on_simplex(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to non‑negative and project onto the ℓ₁‑simplex of a given sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, target_sum / a.shape[0])
    else:
        a = a * (target_sum / s)
    return a


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _softmax_random(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of normalised start vectors."""
    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # power law
    for p in np.linspace(0.0, 3.0, 9):
        seeds.append(_power(n, p))

    # beta shapes – a relatively fine grid, extended beyond earlier range
    for a in np.linspace(0.1, 5.0, 11):
        for b in np.linspace(0.1, 5.0, 11):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows – from very tight to fairly broad
    for frac in np.linspace(0.02, 0.5, 10):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    # Dirichlet draws
    rng = np.random.default_rng(2023)
    for conc in (0.1, 0.3, 1.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random softmax vectors
    for _ in range(10):
        seeds.append(_softmax_random(n, rng))

    if previous is not None and previous.shape == (n,):
        # normalise previous to ℓ₁‑simplex (the objective is scale‑invariant)
        prev = previous.copy()
        prev = prev / max(prev.sum(), 1e-12)
        seeds.append(prev)

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray, n: int, deadline: float,
                   max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑softmax parametrisation, using an exponentially
    increasing smooth‑max (log‑sum‑exp) surrogate.
    """
    v = np.log(seed + 1e-30)          # initialise log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 5.0            # initial smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # print only occasional updates to avoid clutter
            if it % 2000 == 0 or it == 1:
                print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = 0.8 / np.sqrt(it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    # Return the best full sequence (already normalised on the simplex)
    if best_seq is None:
        best_seq = _softmax(v)
        best_val = _objective(best_seq, n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (mirror‑descent style)
# ----------------------------------------------------------------------


def _proj_subgradient_refine(start_seq: np.ndarray, n: int,
                             deadline: float,
                             max_iters: int = 250000,
                             target_sum: float | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent onto the ℓ₁‑simplex.
    """
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)

    a = _project_simplex(start_seq, target_sum)

    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[ProjSub] it={it:5d} best={cur_val:.6f}")

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        lr = 1.0 / np.sqrt(it)

        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, target_sum)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple (1+1) Evolution Strategy on log‑softmax parameters
# ----------------------------------------------------------------------


def _es_logsoftmax(start_seq: np.ndarray, n: int,
                   deadline: float,
                   max_iters: int = 200000,
                   sigma_start: float = 1.0,
                   sigma_decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    (1+1) evolution strategy on the log‑softmax parametrisation.
    Proposes v' = v + σ·N(0,I) and accepts if the objective improves.
    σ decays multiplicatively.
    """
    v = np.log(start_seq + 1e-30)   # stay on the simplex via softmax
    cur_seq = _softmax(v)
    cur_val = _objective(cur_seq, n)

    best_seq = cur_seq.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    sigma = sigma_start

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        v_candidate = v + sigma * rng.standard_normal(n)
        cand_seq = _softmax(v_candidate)
        cand_val = _objective(cand_seq, n)

        if cand_val < cur_val:
            v = v_candidate
            cur_seq = cand_seq
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand_seq.copy()
                if it % 5000 == 0:
                    print(f"[ES] it={it:5d} new best={cand_val:.6f}")

        # sigma schedule (slowly contract)
        sigma *= sigma_decay
        # ensure sigma never collapses to zero
        if sigma < 1e-6:
            sigma = 1e-6

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (pairwise mass transfer)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray, n: int,
                deadline: float,
                max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02  # initial relative transfer size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a small amount from i→j or j→i
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:5d} new best={val:.6f}")

        # slowly tighten the step size
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size and normalisation
    # ------------------------------------------------------------------
    n = 4000
    target_sum = np.sqrt(2.0 * n)   # same normalisation used in the literature

    # ------------------------------------------------------------------
    # Retrieve previous best (if any) from the global namespace
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1. Analytic seed evaluation (fast)
    # ------------------------------------------------------------------
    print("\n=== Analytic seeding ===")
    seeds = _make_seeds(n, previous)
    global_best_seq = None
    global_best_val = float("inf")

    for s in seeds:
        if time.time() > deadline:
            break
        s = _normalise_on_simplex(s, target_sum)
        val = _objective(s, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = s.copy()
            print(f"[seed] new global best = {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Global stochastic search – (1+1) Evolution Strategy on log‑softmax
    # ------------------------------------------------------------------
    print("\n=== Evolution Strategy (log‑softmax) ===")
    # Run a few independent ES runs (different random seeds) as long as we have time
    es_runs = 0
    while time.time() < deadline and es_runs < 8:
        es_runs += 1
        seq_es, val_es = _es_logsoftmax(global_best_seq, n, deadline,
                                        max_iters=120000,
                                        sigma_start=1.0,
                                        sigma_decay=0.9996)
        if val_es < global_best_val:
            global_best_val = val_es
            global_best_seq = seq_es.copy()
            print(f"[ES] run={es_runs} improved global best = {val_es:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam refinement (smooth‑max surrogate)
    # ------------------------------------------------------------------
    print("\n=== Adam refinement ===")
    if global_best_seq is not None:
        seq_adam, val_adam = _adam_optimize(global_best_seq, n, deadline,
                                            max_iters=20000)
        if val_adam < global_best_val:
            global_best_val = val_adam
            global_best_seq = seq_adam.copy()
            print(f"[Adam] improved global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient refinement (long run)
    # ------------------------------------------------------------------
    print("\n=== Projected sub‑gradient refinement ===")
    if global_best_seq is not None:
        seq_psg, val_psg = _proj_subgradient_refine(global_best_seq, n,
                                                   deadline,
                                                   max_iters=250000,
                                                   target_sum=target_sum)
        if val_psg < global_best_val:
            global_best_val = val_psg
            global_best_seq = seq_psg.copy()
            print(f"[ProjSub] improved global best = {val_psg:.6f}")

    # ------------------------------------------------------------------
    # 5. Light stochastic hill‑climb (final polishing)
    # ------------------------------------------------------------------
    print("\n=== Hill climbing (final polish) ===")
    if global_best_seq is not None:
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline,
                                    max_iters=50000)
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[Hill] improved global best = {val_hc:.6f}")

    # ------------------------------------------------------------------
    # 6. Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        print("[fallback] using uniform distribution")
        global_best_seq = _uniform(n)
        global_best_seq = _normalise_on_simplex(global_best_seq, target_sum)
        global_best_val = _objective(global_best_seq, n)

    # ------------------------------------------------------------------
    # 7. Final verification (optional)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    elapsed = time.time() - start_time
    print(f"\n=== Finished (t={elapsed:.1f}s) ===")
    print(f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
