Propose a step‑function sequence with a very low
`evaluate_sequence` value.

Algorithm Overview
------------------
1. **Seed generation** – a handful of analytic families (uniform,
   sine‑window, geometric, beta‑shapes, random Dirichlet and
   random piece‑wise‑linear) are produced and ℓ₁‑normalised.

2. **Local smooth‑max optimisation** – each seed is refined with
   a fast Adam optimiser that works on a log‑softmax parametrisation
   and a smooth‑max surrogate of the convolution maximum.
   Optionally a limited L‑BFGS run is performed for higher precision.

3. **Linear‑programming polishing** – given a current distribution `f`,
   we solve the LP  

   ```
   maximise   Σ g_i
   subject to  (f * g)_k ≤ M   ∀k    (M = max(f*f))
               g_i ≥ 0
   ```

   which is linear because `f` is fixed.  The LP supplies a promising
   direction `g`; we move a fraction `t` towards it and iterate a few
   times, each time recomputing `M`.

4. **Best‑of‑all** – the best sequence found for any length is kept,
   verified with the provided `evaluate_sequence`, stored in the global
   variable `height_sequence_1` and returned as a plain list.

The routine respects a 1000 s time budget, periodically checks the
deadline and prints progress messages.

    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_analytic_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of ℓ₁‑normalised analytic start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    rng = np.random.default_rng(2023)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Random piecewise‑linear seeds
# ----------------------------------------------------------------------


def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑linear shape via a Dirichlet on knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective (fast version, assumes ℓ₁‑norm = 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimiser on smooth max (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _lbfgs_optimize(seed: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 1200,
                    beta: float = 20.0) -> tuple[np.ndarray, float]:
    """Limited‑memory BFGS on a smooth‑max surrogate."""
    # initialise logits
    v = np.log(seed + 1e-30)

    def fun_and_grad(v_in: np.ndarray):
        a = _softmax(v_in)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        wexp = np.exp(beta * (conv - shift))
        w_sum = wexp.sum()
        smooth_val = shift + np.log(w_sum) / beta
        w = wexp / w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return smooth_val, grad_v

    options = {"maxiter": max_iters, "disp": False}
    res = scipy.optimize.minimize(fun_and_grad, v, method="L-BFGS-B",
                                  jac=True, options=options)
    v_opt = res.x
    a_opt = _softmax(v_opt)
    best_val = 2.0 * n * np.max(_conv_fft(a_opt, a_opt))
    return a_opt, best_val


# ----------------------------------------------------------------------
# Linear‑programming polishing step
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve:
        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k
                    g_i ≥ 0
    Returns the optimal g (ℓ₁‑matched to f) or None if the LP fails.
    """
    n = len(f)
    m_rows = 2 * n - 1

    # Build the convolution matrix A where (A @ g)_k = (f * g)_k
    # A has constant value f[i] on the diagonal with offset i.
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = list(range(n))
    A = sp.diags(data, offsets, shape=(m_rows, n), format="csr")

    # Linear program: maximise cᵀ g  ⇒  minimise -cᵀ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    b_ub = np.full(m_rows, M, dtype=np.float64)

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=bounds, method="highs")
    if not res.success:
        return None

    g = res.x
    sum_f = f.sum()
    sum_g = g.sum()
    if sum_g == 0.0:
        return None
    # Rescale to keep ℓ₁‑norm equal to that of f
    g = g * (sum_f / sum_g)
    return g


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_steps: int = 8,
               blend: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Repeated LP polishing.  At each step we solve the LP for the current
    distribution `f`, blend a fraction `blend` towards the LP solution
    and keep the best sequence seen.
    """
    best = f.copy()
    best_val = 2.0 * n * np.max(_conv_fft(best, best))

    steps = 0
    while steps < max_steps and time.time() < deadline:
        M = np.max(_conv_fft(f, f))

        g = _lp_step(f, M)
        if g is None:
            break

        # convex combination
        f_new = (1.0 - blend) * f + blend * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s == 0.0:
            f_new = _uniform(n)
        else:
            f_new = f_new / s

        cur_val = 2.0 * n * np.max(_conv_fft(f_new, f_new))
        if cur_val < best_val:
            best_val = cur_val
            best = f_new.copy()
            print(f"[LP] step={steps+1:2d} best={cur_val:.6f}")

        f = f_new
        steps += 1
        # slowly decrease the blend factor to stabilise
        blend = max(blend * 0.85, 0.05)

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine mixes broad random seeding,
    smooth‑max Adam/L‑BFGS refinement and a dedicated linear‑programming
    polishing stage.  The best sequence (as a plain list of floats) is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    # previously found sequence (if any) for warm‑start
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length.  4000 is fast enough for LP; 8000 is tried if time permits.
    # ------------------------------------------------------------------
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper for maintaining a small set of top candidates
    # ------------------------------------------------------------------
    def _update_top(top_list, val, seq, max_keep=6):
        top_list.append((val, seq))
        top_list.sort(key=lambda x: x[0])
        if len(top_list) > max_keep:
            top_list.pop()

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        top_candidates: list[tuple[float, np.ndarray]] = []

        # ------------------------------------------------------------------
        # 1. Seed generation (analytic + a few random piecewise‑linear)
        # ------------------------------------------------------------------
        analytic = _make_analytic_seeds(n, previous)
        for s in analytic:
            val = _objective(s, n)
            _update_top(top_candidates, val, s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed‑analytic] n={n:5d} new global best={val:.6f}")

        rng = np.random.default_rng(seed + 999)
        for _ in range(800):  # limited random sampling
            cand = _random_piecewise_linear(n, rng)
            val = _objective(cand, n)
            _update_top(top_candidates, val, cand)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = cand.copy()
                print(f"[seed‑randPL] n={n:5d} new global best={val:.6f}")

        # Keep only the elite few for intensive polishing
        top_candidates = top_candidates[:3]

        # ------------------------------------------------------------------
        # 2. Intensive polishing of each elite seed
        # ------------------------------------------------------------------
        for idx, (init_val, seq) in enumerate(top_candidates):
            if time.time() > deadline:
                break
            print(f"\n--- Polishing seed #{idx+1} (n={n}) ---")

            # (a) Adam on smooth‑max
            seq_adam, val_adam = _adam_optimize(seq, n, deadline, max_iters=12000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[polish‑Adam] n={n:5d} best={val_adam:.6f}")

            # (b) L‑BFGS for higher precision (start from Adam output)
            seq_lbfgs, val_lbfgs = _lbfgs_optimize(seq_adam or seq, n, deadline,
                                                   max_iters=1500, beta=30.0)
            if val_lbfgs < global_best_val:
                global_best_val = val_lbfgs
                global_best_seq = seq_lbfgs.copy()
                print(f"[polish‑L‑BFGS] n={n:5d} best={val_lbfgs:.6f}")

            # (c) Linear‑programming polishing
            seq_lp, val_lp = _lp_refine(seq_lbfgs or seq_adam or seq, n, deadline,
                                         max_steps=7, blend=0.30)
            if val_lp < global_best_val:
                global_best_val = val_lp
                global_best_seq = seq_lp.copy()
                print(f"[polish‑LP] n={n:5d} best={val_lp:.6f}")

        # ------------------------------------------------------------------
        # 3. Quick post‑processing: a final short Adam run on the current best
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0 and global_best_seq is not None:
            extra_seq, extra_val = _adam_optimize(global_best_seq, n, deadline,
                                                  max_iters=3000)
            if extra_seq is not None and extra_val < global_best_val:
                global_best_val = extra_val
                global_best_seq = extra_seq.copy()
                print(f"[final‑Adam] n={n:5d} best={extra_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)
        print("[fallback] using uniform distribution")

    # ----------------------------------------------------------------------
    # Verify the sequence with the provided evaluate_sequence()
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        # If the verifier rejects, fall back to uniform (should never happen)
        print("[verification] sequence rejected – falling back to uniform")
        global_best_seq = _uniform(len(global_best_seq))
        global_best_val = _objective(global_best_seq, len(global_best_seq))

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    # Store for a possible next run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
