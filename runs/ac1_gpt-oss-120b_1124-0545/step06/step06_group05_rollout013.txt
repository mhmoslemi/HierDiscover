def _greedy_refine(a, n, deadline, max_iters=200000, move_frac=0.01):
    a = a.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # compute g_i
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idxs = np.arange(i_start, i_end+1)
        rev_idx = k - idxs
        g = np.zeros_like(a)
        g[idxs] = a[rev_idx]  # contributions for each i (half)
        # Actually conv_max = sum_i a_i a_{k-i} = sum_i g_i a_i
        # So gradient for a_i = 2 * a_{k-i}
        g = 2 * g
        # find i_max where g is largest, i_min where g is smallest among indices with positive a?
        # but we also need a_i>0 to move.
        # We look at indices i where a[i]>1e-12
        valid_i = np.where(a > 1e-12)[0]
        if len(valid_i) == 0:
            break
        # Among those, pick i where g[i] is max.
        i_max = valid_i[np.argmax(g[valid_i])]
        # Among all indices (including zeros), find j where g[j] minimal; if multiple, choose with small a maybe.
        # We'll pick j among all indices; if a[j] is zero, we can add mass.
        j_min = np.argmin(g)  # even if a[j] =0 fine
        if i_max == j_min:
            # try find second best.
            # we can break if no improvement possible.
            break

        # amount to move: delta = min(move_frac * a[i_max], a[i_max])
        delta = move_frac * a[i_max]
        # ensure not too large; limit <= a[i_max] maybe.
        if delta <= 0:
            break
        a[i_max] -= delta
        a[j_min] += delta
        # optionally renormalize to sum=1 for numeric drift (should be exact but due to floating errors)
        # can renorm rarely every 1000 iters.
        if it % 1000 == 0:
            s = a.sum()
            if s != 0:
                a /= s

        # compute new objective occasionally
        if it % 1000 == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Greedy] n={n} it={it} best={val:.6f}")

    return best_seq, best_val

def _project_simplex(v):
    # projects vector v onto simplex {x>=0, sum=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _greedy_refine(a, n, deadline, max_iters=20000, move_frac=0.01):
    a = a.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # compute gradient contributions
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1)
        rev_idx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[rev_idx]  # gradient w.r.t a_i

        # pick i_max among indices where a_i>0
        positive_idx = np.where(a > 1e-12)[0]
        if len(positive_idx) == 0:
            break
        i_max = positive_idx[np.argmax(g[positive_idx])]
        # j_min: index with smallest g (including possibly same as i_max)
        j_min = np.argmin(g)

        if i_max == j_min:
            # try alternative: pick second best.
            sorted_idx = np.argsort(-g[positive_idx])
            if len(sorted_idx) > 1:
                i_max = positive_idx[sorted_idx[0]]
                j_min = positive_idx[sorted_idx[1]]
            else:
                break

        delta = move_frac * a[i_max]
        if delta <= 0:
            continue

        a[i_max] -= delta
        a[j_min] += delta

        # renorm after each move maybe not needed because we kept sum unchanged
        # But due to floating errors, renorm occasionally
        if it % 1000 == 0:

            if a_sum > 0:
                a /= a_sum

        # Evaluate new objective occasionally
        if it % 500 == 0:
            new_val = _objective(a)
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()

def _two_spikes(n, alpha):
    # allocate total mass alpha to spikes (split half each)
    # remaining 1-alpha uniform over interior (n-2)
    if n < 2:
        return _uniform(n)
    interior = max(n-2, 1)
    a = np.full(n, (1-alpha) / interior, dtype=np.float64)
    a[0] = alpha / 2.0
    a[-1] = alpha / 2.0
    # If interior zero (n=2) then just assign spikes
    if interior <= 0:
        a[0] = a[1] = 0.5
    # ensure sum=1
    a = a / a.sum()
    return a

"""
Improved search for a non‑negative step function minimising
evaluate_sequence.  The algorithm combines:

* a very diverse set of analytic seeds (uniform, geometric, power,
  symmetric geometric, sine window, Gaussian, Dirichlet draws,
  arcsine family, two‑spike, triangular and random convex mixes);
* a few passes of L‑BFGS optimisation on a smooth surrogate of the
  ∞‑norm (soft‑max of the self‑convolution) with an L2‑variance
  regulariser;
* an exact sub‑gradient refinement (mirror‑descent on the simplex);
* a cheap greedy “mass‑transfer’’ refinement that moves a tiny
  fraction of probability mass from indices contributing most to the
  current maximal convolution entry to those contributing least;
* a final Metropolis‑Hastings annealing stage.

Time is split between the stages; every phase checks the global
deadline (default 1000 s) and aborts early if necessary.
The best sequence found is stored in the global variable
`height_sequence_1` for possible warm‑starts in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (L‑BFGS optimisation)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    _HAS_SCIPY = True
except Exception:                     # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))², seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse or dense Dirichlet draw (Gamma(α,1) normalised)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """
    Discrete analogue of the continuous arcsine law.
    a_i ∝ ((i+0.5)*(n‑0.5‑i))**exponent  with exponent < 0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide="ignore", invalid="ignore"):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        return _uniform(n)
    return a / s


def _two_spikes(n: int, alpha: float) -> np.ndarray:
    """
    Heavy probability mass at both ends, the remainder uniformly spread.
    alpha∈[0,1] is the total mass allocated to the two spikes.
    """
    if n < 2:
        return _uniform(n)
    interior = max(n - 2, 1)
    a = np.full(n, (1.0 - alpha) / interior, dtype=np.float64)
    a[0] = a[-1] = alpha / 2.0
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangle (peak in the centre, zero at the extremes)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a rich pool of start vectors (already normalised to sum = 1).
    In addition to the families used in the reference code we add a few
    extra shapes and a batch of random convex mixes.
    """
    rng = np.random.default_rng(12345 + n)   # deterministic but n‑dependent
    seeds: list[np.ndarray] = []

    # Basic families
    seeds.append(_uniform(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # Dirichlet draws – cover sparse to dense regimes
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # Arcsine‑type seeds (negative exponents)
    for exp in [-1.5, -1.0, -0.75, -0.5, -0.25]:
        seeds.append(_arcseed(n, exp))

    # Two‑spike family
    for a in [0.2, 0.4, 0.6, 0.8]:
        seeds.append(_two_spikes(n, a))

    # Simple triangle
    seeds.append(_triangular(n))

    # random convex mixes of existing seeds (adds diversity)
    for _ in range(30):
        s1, s2 = rng.choice(seeds, size=2, replace=False)
        w = rng.random()
        mix = w * s1 + (1.0 - w) * s2
        if mix.sum() == 0.0:
            continue
        mix = mix / mix.sum()
        seeds.append(mix)

    # reuse previous best construction if it matches the length
    if previous is not None and previous.shape == (n,):
        prev = previous / max(previous.sum(), 1e-12)
        seeds.append(prev)

    # guarantee non‑empty
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# L‑BFGS optimisation on a smooth surrogate of the ∞‑norm
# ----------------------------------------------------------------------


def _lbfgs_fun(v: np.ndarray, n: int, beta: float, lam: float):
    """
    Surrogate loss:
        L(v) = (1/β)·log∑ₖexp(β·convₖ(a)) + λ·∑ₖconvₖ(a)²,
    where a = softmax(v).  Returns (loss, gradient wrt v).
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)                    # length 2n‑1
    vmax = conv.max()
    exp_terms = np.exp(beta * (conv - vmax))
    sum_exp = exp_terms.sum()
    # smooth max term
    loss = (np.log(sum_exp) + beta * vmax) / beta + lam * np.sum(conv ** 2)

    # weights for the smooth‑max
    w = exp_terms / sum_exp                     # shape 2n‑1, non‑negative, sums to 1

    # gradient wrt a
    rev_a = a[::-1]
    full_max = _conv_fft(rev_a, w)            # length 2n‑1
    grad_max = 2.0 * full_max[n - 1:n - 1 + n]

    full_sq = _conv_fft(rev_a, conv)          # length 2n‑1
    grad_sq = 2.0 * full_sq[n - 1:n - 1 + n]

    grad_a = grad_max + lam * grad_sq

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return loss, grad_v.astype(np.float64)


def _lbfgs_optimize_from_logits(v0: np.ndarray,
                                n: int,
                                deadline: float,
                                max_iter: int = 200,
                                beta: float = 10.0,
                                lam: float = 1.0):
    """Wraps scipy.optimize.minimize for a single β/λ pair."""
    # Abort early if time is already gone
    if time.time() > deadline:
        a = _softmax(v0)
        return a, _objective(a)

    if not _HAS_SCIPY:                # pragma: no cover
        a = _softmax(v0)
        return a, _objective(a)

    try:
        res = _opt.minimize(_lbfgs_fun,
                            v0,
                            args=(n, beta, lam),
                            method="L-BFGS-B",
                            jac=True,
                            options={"maxiter": max_iter, "disp": False})
    except Exception:                 # pragma: no cover
        a = _softmax(v0)
        return a, _objective(a)

    a_opt = _softmax(res.x)
    val = _objective(a_opt)
    return a_opt, val


def _lbfgs_multi_phase(seed: np.ndarray,
                       n: int,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Run a short L‑BFGS cascade with increasing β (tighter surrogate).
    Starts from the supplied seed and returns the best sequence seen.
    """
    v = np.log(seed + 1e-30)
    best_seq = seed.copy()
    best_val = _objective(seed)

    # a modest schedule – few β steps, modest max‑iter per step
    for beta in (5.0, 15.0, 40.0, 80.0):
        if time.time() > deadline:
            break
        seq, val = _lbfgs_optimize_from_logits(v,
                                               n,
                                               deadline,
                                               max_iter=200,
                                               beta=beta,
                                               lam=1.0)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
        # re‑initialise logits for the next (tighter) phase
        v = np.log(seq + 1e-30)

    return best_seq, best_val

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient
    of max(conv).  Returns the best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # protect from overflow
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Greedy “mass‑transfer’’ refinement
# ----------------------------------------------------------------------


def _greedy_refine(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 40000,
                   move_frac: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small fraction of probability mass from the index
    that contributes most to the current maximal convolution entry to the
    index that contributes least.  Accepts every move (the objective is
    monitored periodically) and keeps the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the max entry w.r.t. a (ignoring the factor 2 for ordering)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        grad = np.zeros_like(a)
        grad[idx] = a[rev_idx]               # proportional to contribution

        # pick source (largest contribution among positive entries)
        pos = np.where(a > 1e-12)[0]
        if pos.size == 0:
            break
        src = pos[np.argmax(grad[pos])]

        # pick destination (smallest contribution – may be zero)
        dst = int(np.argmin(grad))

        if src == dst:
            # nothing to move
            continue

        delta = move_frac * a[src]
        if delta <= 0.0:
            continue

        a[src] -= delta
        a[dst] += delta

        # occasional renormalisation to avoid drift
        if it % 1000 == 0:
            s = a.sum()
            if s > 0.0:
                a /= s

        # monitor improvement every few hundred steps
        if it % 500 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Greedy] n={n} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (unchanged)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis‑Hastings walk that moves a tiny fraction of mass
    between two random positions, with exponential cooling.
    Occasionally performs a global Gaussian jitter.
    Returns the best sequence discovered.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)       # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a tiny fraction of mass from i to j
        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # periodic global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0:
                previous = previous / max(previous.sum(), 1e-12)
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start modest, enlarge if time permits
    # ------------------------------------------------------------------
    lengths = [4000, 8000]          # 12000 can be added if time remains
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1) Build a rich collection of seeds (including random mixes)
        # --------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # --------------------------------------------------------------
        # 2) Evaluate raw seeds and keep the most promising ones
        # --------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_k = min(20, len(seed_vals))
        top_seeds = [s for s, _ in seed_vals[:top_k]]

        # Record best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – L‑BFGS (smooth max + variance regulariser)
        # ------------------------------------------------------------------
        max_lbfgs = min(5, len(top_seeds))
        for idx, seed_vec in enumerate(top_seeds[:max_lbfgs]):
            if time.time() > deadline:
                break
            seq, val = _lbfgs_multi_phase(seed_vec, n, deadline)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[LBFGS] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=60000)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Greedy mass‑transfer refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _greedy_refine(global_best_seq, n, deadline,
                                      max_iters=40000, move_frac=0.01)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Greedy] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Anneal] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    # Save for potential warm‑starts in subsequent runs
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
