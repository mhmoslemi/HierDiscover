Project onto simplex algorithm:

def _project_onto_simplex(v):
    """Projection of v onto probability simplex (non-negative entries sum to 1)."""
    # Standard algorithm: sort v descending, compute cumulative sum - 1,
    # find rho, compute theta.
    # Implementation from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) with a very low
self‑convolution ∞‑norm.

The method consists of three stages:

1.  **Exploratory low‑dimensional families** (Beta, power‑law,
    arcsine, uniform, Dirichlet) on a modest size  n_small≈2000.
    The best few seeds are kept.

2.  **Projected sub‑gradient descent** on the true max‑convolution.
    The simplex constraint is handled by an Euclidean projection.
    A decreasing step size  η_t = η₀/√t is used; the routine runs until
    the global time budget is exhausted (up to ≈60 k iterations on the
    final length).

3.  **Light simulated‑annealing perturbations** to escape tiny local
    minima, followed by an optional Gaussian smoothing pass.

All work is done inside the allowed 1000 s; the best sequence (as a
plain list of floats) is returned.  The code does not rely on any
external files or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq))²."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive after shift – return a vertex of the simplex
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Simple analytic families (all return a normalised non‑negative vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.ones(n, dtype=np.float64) / n


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """
    Discrete arcsine‑type seed: a_i ∝ ((i+0.5)*(n‑0.5‑i))**exponent,
    exponent < 0 produces a U‑shape.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide="ignore", invalid="ignore"):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the discrete grid {1,…,n} (normalised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p  (i starts at 0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a symmetric Dirichlet(α) and normalise."""
    vec = rng.gamma(alpha, 1.0, size=n)
    total = vec.sum()
    if total <= 0:
        return _uniform(n)
    return vec / total

def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of initial vectors (already normalised)."""
    rng = np.random.default_rng(987654321)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))

    # arcsine family – many negative exponents
    for exp in np.linspace(-1.5, -0.1, 15):
        seeds.append(_arcseed(n, exp))

    # power‑law tails
    for p in [0.5, 0.8, 1.0, 1.3, 1.5, 2.0]:
        seeds.append(_powerlaw(n, p))

    # Beta distributions (symmetric)
    for a in [0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5]:
        seeds.append(_beta_shape(n, a, a))

    # mixture of arcsine and uniform (helps breaking symmetry)
    arc = _arcseed(n, -0.5)
    uni = _uniform(n)
    for w in [0.2, 0.4, 0.6, 0.8]:
        mix = w * arc + (1.0 - w) * uni
        mix /= mix.sum()
        seeds.append(mix)

    # Dirichlet draws – cover sparse vs dense regimes
    for a in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, a, rng))

    # reuse a previous construction if supplied
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / max(previous.sum(), 1e-12))

    # safety – never return an empty list
    if not seeds:
        seeds.append(_uniform(n))

    return seeds

def _proj_subgrad_refine(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 60000,
                         init_lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected (Euclidean) sub‑gradient descent on the true objective.
    Returns the best sequence seen and its objective value.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # compute sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size schedule
        lr = init_lr / np.sqrt(it)

        # gradient step + projection onto the simplex
        a = a - lr * g
        a = _project_simplex(a)

        # evaluate
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # occasional early‑stop check
        if it % 5000 == 0:
            # if no improvement in the last block we may exit early
            pass

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis‑Hastings walk that moves a tiny fraction of mass between
    two random positions, with exponential cooling and occasional global jitter.
    """
    rng = np.random.default_rng()
    n = start_seq.size
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-8, cur_val * 0.04)   # initial temperature proportional to the current value
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a tiny fraction of mass from i to j
        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # periodic global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            jittered = cur + noise
            jittered = np.clip(jittered, 0.0, None)
            if jittered.sum() == 0.0:
                jittered = _uniform(n)
            else:
                jittered /= jittered.sum()
            jitter_val = _objective(jittered)
            if jitter_val < cur_val or rng.random() < np.exp(-(jitter_val - cur_val) / (T + 1e-12)):
                cur = jittered
                cur_val = jitter_val
                if jitter_val < best_val:
                    best_val = jitter_val
                    best = jittered.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={jitter_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Small smoothing pass (optional – can improve a few extra digits)
# ----------------------------------------------------------------------


def _smooth_pass(seq: np.ndarray, sigma: float = 0.7) -> np.ndarray:
    """
    Convolve with a tiny Gaussian kernel and re‑normalise.
    """
    n = seq.size
    radius = int(np.ceil(3 * sigma))
    idx = np.arange(-radius, radius + 1, dtype=np.float64)
    kernel = np.exp(-0.5 * (idx / sigma) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(seq, kernel, mode="same")
    smoothed = np.maximum(smoothed, 0.0)
    total = smoothed.sum()
    if total <= 0:
        return seq
    return smoothed / total


# ----------------------------------------------------------------------
# Upsampling / interpolation
# ----------------------------------------------------------------------


def _upsample(seq_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation from size len(seq_small) to n_large,
    keeping positivity and normalising to sum = 1.
    """
    n_small = seq_small.size
    x_small = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    x_large = (np.arange(n_large, dtype=np.float64) + 0.5) / n_large
    seq_large = np.interp(x_large, x_small, seq_small, left=0.0, right=0.0)
    seq_large = np.maximum(seq_large, 0.0)
    total = seq_large.sum()
    if total <= 0:
        return _uniform(n_large)
    return seq_large / total


# ----------------------------------------------------------------------
# Main entry point – the function called by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the provided time budget (≈1000 s) and returns the
    best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 2.0

    print("\n=== STARTING SEARCH ===")

    # ------------------------------------------------------------------
    # Grab a previously discovered construction if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0:
                previous = previous / max(previous.sum(), 1e-12)
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – cheap exploration on a small problem size
    # ------------------------------------------------------------------
    n_small = 2000
    print(f"\n[Stage 1] Generating seeds (n={n_small})")
    seeds_small = _make_seeds(n_small, previous)

    # evaluate all seeds
    seed_vals = [(s, _objective(s)) for s in seeds_small]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(7, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    best_seq_small = top_seeds[0].copy()
    best_val_small = _objective(best_seq_small)
    print(f"[Stage 1] Best raw seed objective ≈ {best_val_small:.6f}")

    # refine each top seed with projected sub‑gradient descent (short run)
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n[Stage 1‑Refine] seed {idx+1}/{len(top_seeds)}")
        seq, val = _proj_subgrad_refine(
            seed_vec,
            deadline,
            max_iters=12000,
            init_lr=0.7,
        )
        if val < best_val_small:
            best_val_small = val
            best_seq_small = seq.copy()
            print(f"[Stage 1‑Refine] NEW best ≈ {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upsample to a larger length and perform intensive optimisation
    # ------------------------------------------------------------------
    n_large = 8000
    print(f"\n[Stage 2] Upsampling best seed to n={n_large}")
    seq_large = _upsample(best_seq_small, n_large)

    # a quick sanity evaluation
    cur_val = _objective(seq_large)
    print(f"[Stage 2] Initial upsampled objective ≈ {cur_val:.6f}")

    # heavy projected sub‑gradient descent on the large problem
    print("\n[Stage 2‑Grad] Starting projected sub‑gradient refinement")
    seq_large, val_large = _proj_subgrad_refine(
        seq_large,
        deadline,
        max_iters=80000,
        init_lr=0.5,
    )
    print(f"[Stage 2‑Grad] After refinement objective ≈ {val_large:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – light simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Stage 3] Simulated‑annealing fine‑tuning")
        seq_large, val_large = _anneal_optimize(
            seq_large,
            deadline,
            max_iters=None,
        )
        print(f"[Stage 3] After annealing objective ≈ {val_large:.6f}")

    # ------------------------------------------------------------------
    # Optional tiny Gaussian smoothing (may give a modest extra drop)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        smoothed = _smooth_pass(seq_large, sigma=0.6)
        sm_val = _objective(smoothed)
        print(f"[Smooth] Objective after smoothing ≈ {sm_val:.6f}")
        if sm_val < val_large:
            seq_large = smoothed
            val_large = sm_val
            print("[Smooth] Accepted smoothed version")

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    print(
        f"\n=== SEARCH FINISHED ===\nBest objective ≈ {val_large:.6f} "
        f"with length {n_large}"
    )
    # Store globally for potential reuse in subsequent calls
    globals()["height_sequence_1"] = seq_large.copy()

    return seq_large.tolist()
