"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm works in three stages inside the 1000 s budget:

1️⃣ **Seed generation** – a modest set of analytically‑motivated families
   (uniform, sine‑window, geometric, power‑law, beta‑like) together with
   a few random Dirichlet draws.  The best seed (by the exact objective)
   is taken forward.

2️⃣ **Targeted simulated‑annealing** – a fast incremental update of the
   full convolution is used.  Two kinds of moves are mixed:
        * a Dirichlet re‑allocation of mass among 2‑4 random positions,
        * a *targeted* transfer of a fraction of mass from an index that
          strongly contributes to the current maximal convolution entry
          to a random other index.
   The temperature follows a geometric cooling schedule and the best
   solution seen is retained.  Every 50 000 iterations the convolution is
   recomputed from scratch to avoid numerical drift.

3️⃣ **Final exponentiated‑gradient polishing** – a few hundred thousand
   exact sub‑gradient mirror‑descent steps are applied to the best vector
   obtained from SA.

All operations respect the simplex (∑a = 1); the objective is invariant
to scaling, so the final vector is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) using real FFT."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming ∑a == 1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (always normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _make_seeds(n: int, rng: np.random.Generator,
                previous: np.ndarray | None = None) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # deterministic baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # beta‑like (asymmetric) families
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # a few random Dirichlet draws
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, conc)))
    # a few sparse draws
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                if k >= n:
                    seeds.append(rng.dirichlet(np.full(n, conc)))
                    continue
                idx = rng.choice(n, size=k, replace=False)
                vals = rng.dirichlet(np.full(k, conc))
                a = np.zeros(n, dtype=np.float64)
                a[idx] = vals
                seeds.append(a / a.sum())

    # reuse previous solution if it matches length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

# Targeted simulated‑annealing (incremental convolution update)
# ----------------------------------------------------------------------


def _sa_targeted_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          init_temp_factor: float = 0.02,
                          min_temp: float = 1e-12,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    High‑frequency SA with two move types:
        * Dirichlet re‑allocation on 2‑4 random indices,
        * Targeted transfer from an index that contributes heavily to the
          current maximum convolution entry.
    Uses an incremental update of the full convolution.
    Returns the best sequence seen and its exact objective value.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)                # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # temperature schedule
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    # periodic full recompute to keep numerical errors bounded
    recompute_every = 50_000
    steps_since_recomp = 0

    # probability of taking a Dirichlet move
    p_dirichlet = 0.80

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, min_temp)

        # ----- select move type -----
        if rng.random() < p_dirichlet:
            # ----- Dirichlet move (2‑4 indices) -----
            k = rng.integers(2, 5)                     # 2‑4 positions
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            # draw concentration parameter at random (encourages diversity)
            conc = rng.choice([0.3, 0.5, 1.0, 2.0])
            new_alloc = rng.dirichlet(np.full(k, conc)) * total
            deltas = new_alloc - cur[idx]
            # idx and deltas are ready
        else:
            # ----- Targeted transfer move (2 indices) -----
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            contrib = cur[idx_range] * cur[k_star - idx_range]
            s_contrib = contrib.sum()
            if s_contrib == 0.0:
                continue
            probs = contrib / s_contrib
            i = rng.choice(idx_range, p=probs)   # source index (high impact)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)               # ensure j ≠ i
            # transfer a fraction of the source mass (max 30 %)
            max_frac = 0.30
            delta = rng.random() * cur[i] * max_frac
            if delta <= 0.0:
                continue
            idx = np.array([i, j], dtype=int)
            deltas = np.array([-delta, delta], dtype=np.float64)

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        # a*Δ + Δ*a contributions
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur   # length‑n slice
            conv_new[2 * p] += delta * delta        # Δ*Δ self term
        # cross terms Δ_i*Δ_j (i ≠ j) – appear twice in Δ*Δ
        if len(idx) >= 2:
            for ii in range(len(idx)):
                for jj in range(ii + 1, len(idx)):
                    pos = idx[ii] + idx[jj]
                    conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

        new_obj = 2.0 * n * np.max(conv_new)

        # ----- Metropolis acceptance -----
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept
            cur[idx] = cur[idx] + deltas
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA‑Imp] n={n:5d} it={it:7d} best={new_obj:.6f}")

        # occasional full recompute to avoid drift
        steps_since_recomp += 1
        if steps_since_recomp >= recompute_every:
            cur_conv = _conv_fft(cur, cur)
            cur_obj = 2.0 * n * np.max(cur_conv)
            steps_since_recomp = 0

    return best_seq, best_obj

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient steps using the true sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)                     # step‑size schedule
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                     # numeric stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val

            max_iters=5_000_000,
        )
        if val_sa < global_best_val:
            global_best_val = val_sa
            global_best_seq = seq_sa.copy()
            print(f"[SA‑overall] n={best_n:5d} new global best={val_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final mirror‑descent polishing (if time permits)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline - 5.0:
        print("\n=== Final sub‑gradient polishing ===")
        seq_pg, val_pg = _subgradient_refine(
            global_best_seq,
            best_n,
            deadline,
            max_iters=300_000,
        )
        if val_pg < global_best_val:
            global_best_val = val_pg
            global_best_seq = seq_pg.copy()
            print(f"[Polish] n={best_n:5d} new global best={val_pg:.6f}")

    # ------------------------------------------------------------------
    # Normalise to unit sum (just in case of tiny numerical drift)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # fallback – uniform distribution
        best_n = 4000
        global_best_seq = _uniform(best_n)
        global_best_val = _objective(global_best_seq)

    global_best_seq = np.maximum(0.0, global_best_seq)
    s = global_best_seq.sum()
    if s == 0.0:
        global_best_seq = _uniform(best_n)
    else:
        global_best_seq /= s

    # Optional sanity check with the provided evaluate_sequence
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {best_n})")
    return global_best_seq.tolist()
