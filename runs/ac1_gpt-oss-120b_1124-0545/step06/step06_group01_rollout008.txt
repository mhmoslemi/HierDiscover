def smooth_obj_and_grad(v, n, beta):
    a = softmax(v)
    conv = conv_fft(a, a)
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))  # scaled to avoid overflow
    sumw = w.sum()
    if sumw == 0:
        w_norm = np.full_like(w, 1/len(w))
    else:
        w_norm = w / sumw
    # objective = 2*n * (maxc + np.log(sumw)/beta)
    obj = 2.0 * n * (maxc + np.log(sumw) / beta)
    # gradient on a
    rev_a = a[::-1]
    full = conv_fft(rev_a, w_norm) # length 3n-2
    grad_a = 4.0 * n * full[n - 1:n - 1 + n]  # central n entries
    # convert to gradient wrt v via softmax: grad_v = a * (grad_a - np.dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

Optimises a non‑negative step‑function for the max‑auto‑convolution
objective using two stages:

1. **Broad analytical seeding** – a diverse set of normalised analytic
   families (uniform, sine‑window, geometric, power‑law, beta‑like and
   a dedicated arcsine shape) together with a few random Dirichlet
   draws are generated.  The best few seeds are kept.

2. **Smooth‑max L‑BFGS refinement** – each kept seed is optimised on a
   smooth approximation of the max‑convolution
   `f_β(a)=2·n·log‑sum‑exp_β(conv(a,a))/β`.  A log‑softmax
   parametrisation guarantees the ℓ₁‑norm constraint and non‑negativity.
   The optimizer follows a continuation scheme with increasing β values
   (coarse → fine smoothing) and a limited‑memory BFGS method.
   The refined candidates are finally polished by a short stochastic
   hill‑climb that shifts a tiny amount of mass from “offending’’ indices
   to random recipients.

The routine respects a time budget (default 1000 s) and returns the
best sequence it discovered as a plain list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a_sum = a.sum()
    return a / a_sum if a_sum != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a_sum = a.sum()
    return a / a_sum if a_sum != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a_sum = a.sum()
    return a / a_sum if a_sum != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    a_sum = a.sum()
    return a / a_sum if a_sum != 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine (beta(0.5,0.5)) density."""
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    a_sum = a.sum()
    return a / a_sum if a_sum != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_analytic_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of normalised analytic start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes – including extremes near 0.5
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # dedicated arcsine shape (beta(0.5,0.5))
    seeds.append(_arcsine_shape(n))

    # random Dirichlet families
    rng = np.random.default_rng(2024)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # softmax of random logits (non‑parametric)
    for _ in range(4):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Fast evaluation (sum is assumed to be 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser
# ----------------------------------------------------------------------


def _smooth_max_obj_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (f, g) where f = 2·n·log‑sum‑exp_β(conv(a,a))/β
    and g = ∇_v f for a = softmax(v).
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)

    # max‑shift for numerical stability
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    sumw = w.sum()
    if sumw == 0.0:
        # degenerate – fall back to the hard max
        f = 2.0 * n * maxc
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        f = 2.0 * n * (maxc + np.log(sumw) / beta)
        w_norm = w / sumw

    # gradient wrt a: 4·n·conv_rev(a, w_norm)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w_norm)                # length = 3n‑2
    grad_a = 4.0 * n * full[n - 1:n - 1 + n]

    # convert to gradient wrt logits (softmax Jacobian)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    return f, grad_v


def _lbfgs_smooth_max(start_a: np.ndarray,
                      n: int,
                      deadline: float,
                      betas: list[float],
                      iters_per_beta: int = 120) -> tuple[np.ndarray, float]:
    """
    Runs a continuation of L‑BFGS on the smooth‑max surrogate.
    Returns the best exact (hard‑max) sequence encountered.
    """
    v = np.log(start_a + 1e-30)          # initialise logits
    best_seq = start_a.copy()
    best_val = _objective(best_seq, n)

    for beta in betas:
        if time.time() > deadline:
            break

        # L‑BFGS on the current β
        def fun_and_grad(v_flat):
            return _smooth_max_obj_grad(v_flat, n, beta)

        # run limited‑memory BFGS; maxiter ≈ number of curvature updates
        res = opt.minimize(fun_and_grad,
                           v,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': iters_per_beta,
                                    'disp': False,
                                    'gtol': 1e-12})
        v = res.x
        a = _softmax(v)
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS β={beta:.2f}] n={n:5d} new best={val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted stochastic hill‑climb (fine polishing)
# ----------------------------------------------------------------------


def _targeted_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 20000,
                         init_step: float = 0.02) -> tuple[np.ndarray, float]:
    """Move a tiny amount of mass from a “heavy’’ donor to a random receiver."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = init_step

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        # contribution of each index to the current maximal coefficient
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()
        if total <= 0.0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor],
                           step_scale * cur[donor] + 1e-14)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} new best={val:.6f}")

        # decay step size slowly
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine uses analytic seeding,
    continuation L‑BFGS on a smooth‑max surrogate and a final hill‑climb.
    The best sequence (as a plain list of floats) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from previous run (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Choose a length – larger lengths give a tighter theoretical bound.
    # We try 8000 first; if time is already low we fall back to 4000.
    # ------------------------------------------------------------------
    lengths = [8000, 4000]
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1. Create analytic seeds (full‑length, ℓ₁‑normalised)
        # --------------------------------------------------------------
        seeds = _make_analytic_seeds(n, previous)

        # --------------------------------------------------------------
        # 2. Quick evaluation of the raw seeds
        # --------------------------------------------------------------
        seed_vals = []
        for s in seeds:
            val = _objective(s, n)
            seed_vals.append((val, s))
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # keep the most promising few for intensive polishing
        seed_vals.sort(key=lambda x: x[0])
        top_k = 6
        top_candidates = [seq for _, seq in seed_vals[:top_k]]

        # --------------------------------------------------------------
        # 3. Continuation L‑BFGS on the smooth‑max surrogate
        # --------------------------------------------------------------
        # β schedule (progressively sharper)
        beta_sched = [1.0, 3.0, 9.0, 27.0, 81.0]
        iters_each = 150  # L‑BFGS iterations for each β

        for idx, cand in enumerate(top_candidates):
            if time.time() > deadline:
                break
            print(f"\n--- Polishing candidate #{idx + 1} (n={n}) ---")
            refined_seq, refined_val = _lbfgs_smooth_max(
                start_a=cand,
                n=n,
                deadline=deadline,
                betas=beta_sched,
                iters_per_beta=iters_each,
            )
            if refined_val < global_best_val:
                global_best_val = refined_val
                global_best_seq = refined_seq.copy()
                print(f"[polish‑L‑BFGS] n={n:5d} best={refined_val:.6f}")

        # --------------------------------------------------------------
        # 4. Final stochastic hill‑climb on the current global best
        # --------------------------------------------------------------
        if global_best_seq is not None:
            hill_seq, hill_val = _targeted_hill_climb(
                start_seq=global_best_seq,
                n=n,
                deadline=deadline,
                max_iters=20000,
                init_step=0.02,
            )
            if hill_val < global_best_val:
                global_best_val = hill_val
                global_best_seq = hill_seq.copy()
                print(f"[final‑Hill] n={n:5d} best={hill_val:.6f}")

        # ------------------------------------------------------------------
        # Early‑exit sanity run (tiny Adam pass) if time permits
        # ------------------------------------------------------------------
        if time.time() < deadline - 2.0 and global_best_seq is not None:
            # a very short Adam run on the exact max (helps with tiny drift)
            extra_seq, extra_val = _lbfgs_smooth_max(
                start_a=global_best_seq,
                n=n,
                deadline=deadline,
                betas=[162.0],
                iters_per_beta=60,
            )
            if extra_val < global_best_val:
                global_best_val = extra_val
                global_best_seq = extra_seq.copy()
                print(f"[extra‑L‑BFGS] n={n:5d} best={extra_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)
        print("[fallback] uniform distribution used")

    # ----------------------------------------------------------------------
    # Verify with the provided verify function (protect against accidental
    # NaNs or infinities after rounding).  If the verification fails we
    # fall back to a uniform distribution.
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[verification] sequence rejected – falling back to uniform")
        n = len(global_best_seq)
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
