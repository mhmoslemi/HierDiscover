"""
Optimise a non‑negative step function for the objective

    2·n·max(conv(a, a)) / (∑a)²

The search proceeds in several time‑aware stages:

1. **Diverse seed generation** – analytic families (uniform, triangular,
   decreasing linear ramp, geometric, power‑law, beta, sine‑window,
   sine^γ) together with random Dirichlet / sparse draws.

2. **Adam on a smooth max** – the max is approximated by a log‑sum‑exp
   with a slowly increasing temperature.  Parameters are kept on the
   probability simplex via a log‑softmax; Adam updates are performed
   until the deadline.

3. **Long sub‑gradient mirror descent** – exact sub‑gradient of the true
   max‑convolution is used with an exponentiated‑gradient step.  The
   iteration budget is substantially larger than in the reference
   implementation.

4. **Intensive simulated‑annealing** – neighbourhood moves consist of
   redistributing the mass of a random subset of coordinates via a
   Dirichlet draw.  The temperature decay is slower, and larger subsets
   (up to 6 indices) are explored.

5. **Fine‑grained coordinate descent** – a simple pairwise mass‑swap
   hill‑climber that makes tiny deterministic moves (ε·a_i) to polish
   the solution.

All stages respect a global time budget (default 1000 s).  The best
sequence found is returned as a plain ``list`` of ``float`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for repeated use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming ∑seq == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Peak in the centre, linearly decreasing to the edges."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _decreasing_linear(n: int) -> np.ndarray:
    """Monotone decreasing ramp from left to right."""
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    """Flat top of relative width `width_fraction`, linear ramps on each side."""
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(np.round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    # rising ramp
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    # flat part
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Right‑skewed geometric decay a_i ∝ r^i."""
    if r <= 0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometries: r**i + r**(n‑1‑i)."""
    if r <= 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^‑p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_gamma(n: int, gamma: float) -> np.ndarray:
    """Sine window raised to a power γ ≥ 0."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with scalar concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse vector with non‑zeros sampled from a Dirichlet."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """
    Assemble a diverse set of normalised candidate vectors for length ``n``.
    """
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_decreasing_linear(n))
    seeds.append(_trapezoidal(n, width_fraction=0.2))
    seeds.append(_sine_window(n))

    # Geometric families
    for r in np.linspace(0.2, 0.99, 14):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family
    for p in np.linspace(0.0, 3.0, 15):
        seeds.append(_power(n, p))

    # Beta‑family grid
    alphas = np.linspace(0.05, 2.5, 12)
    betas = np.linspace(0.05, 2.5, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Sine^γ families
    for gamma in np.linspace(0.2, 5.0, 12):
        seeds.append(_sine_gamma(n, gamma))

    # Random Dirichlet draws
    for conc in (0.05, 0.15, 0.5, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns
    for k in (3, 7, 12):
        for conc in (0.1, 0.4):
            seeds.append(_random_sparse(n, k, conc, rng))

    # Previous global best (if compatible length)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on a smoothed version of ``max(conv)``.
    The vector lives on the simplex via a log‑softmax parametrisation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5            # initial smooth‑max temperature
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # increase smoothing (cap at beta_max)
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp over convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.9
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent using exponentiated‑gradient updates.
    Learning rate decays as 1/√t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max at lag max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-10,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill climbing with a simple annealing schedule.
    A random subset of 2‑6 positions reshuffles its total mass via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling (slow)
        T = max(T * 0.9997, min_temp)

        k = rng.integers(2, 7)          # 2…6 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _coordinate_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200000,
    eps_frac: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Tiny pairwise mass‑swap hill climbing.
    Moves ε·a_i from a source to a destination (ε decays with iteration).
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # decreasing step size
        eps = eps_frac / np.sqrt(it)

        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] < eps:
            continue
        new = cur.copy()
        delta = eps * cur[i]          # amount moved from i to j
        new[i] -= delta
        new[j] += delta

        new_val = _objective(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[CD] n={n:5d} it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """

    previous = globals().get("height_sequence_1")

            previous = None
    else:

    # Lengths explored – broader lengths give richer search space

    seq, val = _adam_optimize(

        global_best_val = val
        print(f"[overall‑CD] improved global best to {val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

    except Exception as exc:

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
