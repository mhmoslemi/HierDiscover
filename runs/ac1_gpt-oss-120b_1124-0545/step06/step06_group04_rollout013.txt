def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 0.5

    best_seq = None
    best_val = float('inf')

    # lengths to try, prioritize larger ones after initial quick seeds
    lengths = [8000, 4000, 12000]  # try large first
    for n in lengths:
        if time.time() > deadline: break
        print(f"=== Optimizing length {n} ===")
        # generate seeds (half-length). Use a simplified seed set focusing on a few families.
        seeds = generate_seeds(n, previous=best_seq)
        # evaluate seeds quickly using _objective (fast)
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        # keep top seeds (e.g., 4)
        top = [s for s,_ in seed_vals[:4]]
        for idx, seed in enumerate(top):
            if time.time() > deadline: break
            # run symmetric Adam
            seq, val = symmetric_adam_optimize(seed, n, deadline)
            if seq is not None and val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[global Adam] n={n} seed={idx} new best={best_val:.6f}")
        # subgradient refinement on current best with this n
        if best_seq is not None and len(best_seq)==n and time.time() < deadline:
            seq, val = symmetric_subgrad_refine(best_seq, n, deadline)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[global SubGrad] n={n} new best={best_val:.6f}")
        # annealing fine tune
        if best_seq is not None and len(best_seq)==n and time.time() < deadline:
            seq, val = symmetric_anneal_optimize(best_seq, n, deadline)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[global Anneal] n={n} new best={best_val:.6f}")

        # possibly store best for next length (upscale)
        previous = best_seq

    # fallback uniform if none
    if best_seq is None:
        best_seq = _uniform(4000)

    # Save in global
    globals()["height_sequence_1"] = best_seq.tolist()
    print(f"Finished: best objective ≈ {best_val:.6f} length {len(best_seq)}")

def _sym_adam_optimize(seed, n, deadline, max_iters=12000, beta0=2.0, beta_target=200.0, beta_mul=1.001, lr0=0.5):
    v = np.log(seed + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_seq = None
    best_val = float('inf')
    beta = beta0
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        # softmax
        maxv = np.max(v)
        expv = np.exp(v - maxv)
        p = expv / expv.sum()
        # symmetrize
        a = (p + p[::-1]) * 0.5
        # compute conv
        conv = _conv_fft(a, a)
        cur_val = 2 * n * conv.max() # sum a =1
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SymAdam] n={n} it={it} best={cur_val:.6f}")
        # increase beta
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)
        # compute smoothed weights
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # gradient wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]

        # gradient transformation
        sym_g = 0.5 * (grad_a + grad_a[::-1])
        dot = np.dot(grad_a, a)  # a dot grad_a
        # gradient wrt logits v
        grad_v = p * (sym_g - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)
        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)
    return best_seq, best_val

def _sym_anneal_optimize(start_seq, n, deadline, max_iters=None):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)
        # random move preserving sum & symmetry
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while i == j:
            j = rng.integers(0, n)
        # compute delta move <=2% of cur[i]
        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # also move symmetric counterparts
        si = n-1-i
        sj = n-1-j
        if si != i:
            new_seq[si] -= delta
        else:
            # i is central, already sub delta once
            pass
        if sj != j:
            new_seq[sj] += delta
        # ensure non-negativity
        new_seq = np.clip(new_seq, 0.0, None)
        # renormalize (should stay same sum but sym changes may violate sum)
        new_seq_sum = new_seq.sum()
        if new_seq_sum == 0:
            continue
        new_seq = new_seq / new_seq_sum
        # enforce symmetry again
        new_seq = (new_seq + new_seq[::-1]) * 0.5
        new_seq = new_seq / new_seq.sum()
        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[SymAnneal] n={n} it={it} best={new_val:.6f}")
        if it % 1000 == 0:
            # occasional global jitter
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0:
                continue
            new_seq = new_seq / new_seq.sum()
            new_seq = (new_seq + new_seq[::-1]) * 0.5
            new_seq = new_seq / new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):

                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[SymAnneal] n={n} it={it} (jump) best={new_val:.6f}")
    return best, best_val

def _make_sym_seeds(n, previous):

    sym_seeds = []
    for s in seeds:
        s_sym = (s + s[::-1]) * 0.5
        s_sym = s_sym / s_sym.sum()
        sym_seeds.append(s_sym)
    # add previous if exists
    if previous is not None:
        prev_arr = np.asarray(previous, dtype=np.float64)

            up = _upscale(prev_arr, n)

            arr = arr / arr.sum()
            sym_seeds.append(arr)

    if not sym_seeds:

"""
Symmetric multi‑stage optimizer for the self‑convolution infinity norm.

The search works on a single hidden vector (logits) that is turned into a
probability vector by a soft‑max, then mirrored to enforce f[i]=f[n‑1‑i].
The objective to minimise is 2·n·max(conv(f,f)) (the sum is normalised to 1).
Three phases are performed for each target length:

  1.  Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution.
     The surrogate becomes tighter as β grows, and the updates respect the
     simplex constraints via the soft‑max parametrisation.

  2.  Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
     max‑convolution entry – a cheap way to tighten the bound after Adam.

  3.  Lightweight simulated‑annealing that moves tiny amounts of mass while
     preserving symmetry and the simplex constraint.

The algorithm starts with a modest length (8000) – long enough to expose a
good constant‑factor but short enough to allow many iterations inside the
1000 s budget – and up‑scales the current best solution to larger lengths
if time permits.  The best sequence found is cached in the global variable
``height_sequence_1`` for reuse across runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for many calls)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (sum is 1 for our normalised vectors)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families (normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) on (0,1) – interior points avoid infinities."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _tent(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)
    a[a < 0] = 0.0
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hanning(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry and renormalise to a simplex."""
    s = (v + v[::-1]) * 0.5
    total = s.sum()
    if total == 0.0:
        return _uniform(v.size)
    return s / total


def _make_sym_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest, diverse pool of *symmetric* seeds (already normalised)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families (already symmetric after the call to _symmetrize)
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_hanning(n))
    seeds.append(_tent(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # geometric and power‑law families
    for r in np.linspace(0.6, 0.98, 5):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 5):
        seeds.append(_power(n, p))

    # beta draws (symmetric if α=β)
    for a in [0.5, 1.0, 1.5, 2.0]:
        seeds.append(_beta_seed(n, a, a))
        seeds.append(_beta_seed(n, a, a * 1.2))

    # Dirichlet draws
    for alpha in [0.1, 0.3, 0.5, 0.8]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # include up‑scaled previous best if available
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            up = _upscale(prev_arr, n)
            seeds.append(up)
        except Exception:
            pass

    # enforce symmetry and normalisation
    sym_seeds = [_symmetrize(s) for s in seeds]
    # guarantee at least one seed
    if not sym_seeds:
        sym_seeds.append(_uniform(n))
    return sym_seeds


# ----------------------------------------------------------------------
# Symmetric Adam optimiser (smooth max surrogate)
# ----------------------------------------------------------------------


def _sym_adam_optimize(seed: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 12000,
                       beta0: float = 2.0,
                       beta_target: float = 300.0,
                       beta_mul: float = 1.001,
                       lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam on Lβ = (1/β)·log Σ exp(β·conv) with symmetry enforced.
    Returns the best symmetric sequence seen and its *exact* objective.
    """
    v = np.log(seed + 1e-30)                # logits → simplex via soft‑max
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex point and enforced symmetry
        p = _softmax(v)
        a = (p + p[::-1]) * 0.5                # symmetric sequence, sum(p)=1 ⇒ sum(a)=1

        # exact objective (fast)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max()        # sum(a)=1, objective = 2·n·max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SymAdam] n={n} it={it:5d} best={cur_val:.6f}")

        # increase β gradually
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # smooth‑max weights
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                     # under‑flow guard
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of the smooth‑max with respect to a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # length‑n gradient

        # propagate through symmetry and soft‑max
        sym_g = 0.5 * (grad_a + grad_a[::-1])  # S·grad_a
        dot = np.dot(grad_a, a)                # a·grad_a == p·sym_g
        grad_v = p * (sym_g - dot)             # chain rule for soft‑max

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Symmetric sub‑gradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _sym_subgradient_refine(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    The gradient is the sub‑gradient of the active convolution entry.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # enforce symmetry on the gradient (otherwise symmetry would be broken)
        g = 0.5 * (g + g[::-1])

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        a = np.exp(log_a)

        # re‑impose symmetry and renormalise to the simplex
        a = (a + a[::-1]) * 0.5
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SymSubGrad] n={n} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val

# Symmetric simulated‑annealing fine‑tuner
# ----------------------------------------------------------------------


def _sym_anneal_optimize(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Lightweight Metropolis‑Hastings on the simplex with symmetry preservation."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # temperature starts as a few percent of the current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two (possibly distinct) positions
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while i == j:
            j = rng.integers(0, n)

        # move a tiny fraction of mass from i to j (mirrored as well)
        delta = rng.random() * 0.025 * cur[i]   # ≤2.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        si, sj = n - 1 - i, n - 1 - j
        if si != i:
            new_seq[si] -= delta
        if sj != j:
            new_seq[sj] += delta

        new_seq = np.clip(new_seq, 0.0, None)
        total = new_seq.sum()
        if total == 0.0:
            continue
        new_seq /= total
        # enforce exact symmetry again
        new_seq = (new_seq + new_seq[::-1]) * 0.5
        new_seq /= new_seq.sum()

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[SymAnneal] n={n} it={it:6d} best={new_val:.6f}")

        # every 1000 steps try a global jitter to escape plateaus
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            trial = cur + noise
            trial = np.clip(trial, 0.0, None)
            if trial.sum() == 0.0:
                continue
            trial /= trial.sum()
            trial = (trial + trial[::-1]) * 0.5
            trial /= trial.sum()
            trial_val = _objective(trial)
            if trial_val < cur_val or rng.random() < np.exp(-(trial_val - cur_val) / (T + 1e-12)):
                cur = trial
                cur_val = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best = trial.copy()
                    print(f"[SymAnneal] n={n} it={it:6d} (jump) best={trial_val:.6f}")

    return best, best_val

    """

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # reuse a previously found construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # try a large length first; if time permits, fall back to a smaller one
    lengths = [8000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # Seed generation (symmetric)
        # ------------------------------------------------------------------
        seeds = _make_sym_seeds(n, previous)

        # ------------------------------------------------------------------
        # Quick evaluation of raw seeds, keep the top few
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(6, len(seed_vals))]]

        # keep the best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation from each top seed
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _sym_adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=12000,
                beta0=2.0,
                beta_target=300.0,
                beta_mul=1.001,
                lr0=0.5,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Sub‑gradient refinement on the current best
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _sym_subgradient_refine(
                global_best_seq, n, deadline, max_iters=80000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _sym_anneal_optimize(
                global_best_seq, n, deadline, max_iters=None
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # Propagate the current best forward (up‑scale for the next length)
        previous = global_best_seq

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Cache for later calls
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
