"""
Hybrid global seeding + multi‑stage L‑BFGS refinement.

* A diverse set of analytic seeds (uniform, sine, Gaussian, Beta,
  Dirichlet, previous best, random softmax) is generated.
* Each seed is optimised on the *smooth‑max* surrogate
  ϕ_β(a)=max_β(a*a) = max_i a*a[i] + log‑sum‑exp(β·(a*a−max))/β.
  The surrogate becomes the true max as β→∞.
* For a fixed β we minimise 2·n·ϕ_β(a) with respect to the
  log‑softmax parameters v (a = softmax(v)).  The gradient is
  computed analytically via FFT‑based convolution, and SciPy’s
  L‑BFGS‑B handles the line‑search and curvature information.
* β is increased geometrically (β←β·5) – a continuation method
  that sharpens the surrogate and lets L‑BFGS descend to a better
  local minimum.
* The best exact objective 2·n·max(a*a) encountered over all seeds
  and all β‑stages is returned as the final step‑function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always positive, sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of normalised start vectors."""
    rng = np.random.default_rng(2023)
    seeds: list[np.ndarray] = [
        _uniform(n),
        _sine_window(n),
    ]

    # Gaussian windows – σ from 5 % to 30 % of n
    for frac in np.linspace(0.05, 0.30, 6):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    # A grid of Beta shapes (α,β ∈ [0.3,2.0])
    for a in np.linspace(0.3, 2.0, 7):
        for b in np.linspace(0.3, 2.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws with various concentrations
    for conc in (0.2, 0.8, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random softmax vectors
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        # Normalise previous best to the ℓ₁‑simplex
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())
        else:
            seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient w.r.t. log‑softmax params
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns the smooth‑max surrogate 2·n·ϕ_β(a) and its gradient w.r.t. v,
    where a = softmax(v) (so ∑a = 1).
    """
    a = _softmax(v)                       # shape (n,)
    conv = _conv_fft(a, a)                # length 2·n‑1

    # smooth‑max of conv
    shift = np.max(conv)
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    smooth_val = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_val

    # weights for the smooth‑max (∂ϕ_β/∂conv)
    w = exp_beta / sum_exp                # length 2·n‑1

    # gradient of Σ w_k·conv_k w.r.t. a (via convolution with reversed a)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length 3·n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

    # chain rule through softmax
    grad_J_a = 2.0 * n * grad_a
    dot = np.dot(grad_J_a, a)
    grad_v = a * (grad_J_a - dot)

    return float(obj), grad_v.astype(np.float64)


def _exact_objective(a: np.ndarray) -> float:
    """
    Exact evaluation 2·n·max(a*a) with ∑a = 1.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Multi‑stage L‑BFGS optimisation for a single seed
# ----------------------------------------------------------------------


def _lbfgs_multi_stage(seed: np.ndarray,
                       n: int,
                       deadline: float,
                       beta_start: float = 1.0,
                       beta_factor: float = 5.0,
                       beta_max: float = 200.0,
                       maxiter_per_stage: int = 400) -> tuple[np.ndarray, float]:
    """
    Runs a continuation of L‑BFGS‑B optimisations on progressively
    sharper smooth‑max surrogates.
    Returns the best exact objective found and the corresponding sequence.
    """
    # initialise log‑softmax parameters
    v = np.log(seed + 1e-30)   # shape (n,)

    best_seq = None
    best_val = float("inf")
    beta = beta_start

    while beta <= beta_max and time.time() < deadline:
        # SciPy minimise expects a callable returning (f, grad)
        def fun(x):
            return _smooth_obj_and_grad(x, n, beta)

        res = minimize(fun,
                       v,
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "disp": False,
                                "gtol": 1e-12,
                                "ftol": 1e-12})

        v = res.x  # continue from the solution of this stage
        a = _softmax(v)
        val = _exact_objective(a)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS] n={n:5d} β={beta:6.2f} val={val:.6f}")

        beta *= beta_factor

    return best_seq, best_val

# Light stochastic hill‑climbing (optional final polish)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Simple pair‑wise mass transfer hill‑climb."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _exact_objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor, receiver = (i, j) if rng.random() < 0.5 else (j, i)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _exact_objective(new)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 3000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val
