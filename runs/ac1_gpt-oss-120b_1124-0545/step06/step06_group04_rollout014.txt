"""
A fast, high‑precision optimiser for the self‑convolution infinity‑norm.

The method proceeds in three stages:

1.  **Massive parametric seed sweep** –  a few hundred analytically
    defined families (geometric, symmetric geometric, power‑law,
    geometric×power, Beta, Gaussian, arcsine‑like and random Dirichlet
    draws) are instantiated on the target length.  Each candidate is
    normalised to total mass 1 and evaluated with an exact FFT‑based
    convolution.  The ten best seeds are retained.

2.  **High‑β Adam optimisation** –  each of the retained seeds is
    refined by Adam on the smooth surrogate   
    `Lβ(a)=β⁻¹·log Σ exp(β·(a*a)ₖ)`.  
    The inverse temperature β is increased from 2 up to ≈ 500,
    allowing the optimiser to see the true max‑convolution while still
    receiving a useful gradient.  The Adam run is limited only by the
    remaining time budget.

3.  **Exponentiated‑gradient sub‑gradient descent + annealing** –  the
    best sequence from step 2 is further polished by a mirror‑descent
    on the exact max‑convolution (step‑size ∝ 1/√t) and a cheap
    simulated‑annealing “mass‑swap’’ finetuner.  Finally a
    symmetrisation (average with its mirror) is tried because the
    objective is invariant under reversal.

All steps respect the 1000 s global deadline; the algorithm returns
the best sequence (list of floats) discovered before the timeout.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, soft‑max, objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for long vectors)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum².  seq must be a 1‑D numpy array."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))

# Seed families (already normalised to sum 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    left = r ** np.arange(n, dtype=np.float64)
    right = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a = left + right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_seed(n: int, a: float, b: float) -> np.ndarray:
    """Symmetric Beta(α,β) on (0,1) – interior points avoid infinities."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a_vals = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    a_vals = np.nan_to_num(a_vals, nan=0.0, posinf=0.0, neginf=0.0)
    s = a_vals.sum()
    return a_vals / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_like(n: int, power: float = 1.0) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-0.5-i))^{-power/2} – a discrete arcsine‑like shape."""
    i = np.arange(n, dtype=np.float64)
    base = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = base ** power
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length – preserves total mass."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq

    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                # point on the simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # raise β gradually, but never exceed the target
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # smooth‑max weights (log‑sum‑exp trick)
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                # under‑flow – fall back to hard max
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of the smoothed loss w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # chain‑rule through the soft‑max parametrisation
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # Adam update on the logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200000,
                        patience: int = 2000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient of
    the max‑convolution entry.  Stops early if no improvement for `patience`
    iterations.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    stale = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            stale = 0
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")
        else:
            stale += 1
            if stale >= patience:
                break

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Light‑weight simulated‑annealing that swaps tiny fragments of mass.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # ≤2.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:5d} best={new_val:.6f}")

        if it % 1000 == 0:
            # occasional global jitter
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Candidate generation utilities
# ----------------------------------------------------------------------


def _generate_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of normalised seed vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # baseline
    seeds.append(_uniform(n))

    # geometric – fine grid on c = (1‑r)·n
    for c in np.linspace(0.1, 5.5, 240):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
            seeds.append(_sym_geom(n, r))

    # power‑law
    for p in np.linspace(0.01, 2.5, 180):
        seeds.append(_power(n, p))

    # geometric × power‑law (two‑parameter family)
    for c in np.linspace(0.3, 3.0, 30):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 30):
            seeds.append(_geom_power(n, r, p))

    # Beta family (symmetric and a bit skewed)
    for a in np.linspace(0.2, 3.0, 30):
        seeds.append(_beta_seed(n, a, a))
        seeds.append(_beta_seed(n, a, a * 1.5))
        seeds.append(_beta_seed(n, a * 1.5, a))

    # arcsine‑like (parameterised singularity strength)
    for power in np.linspace(0.3, 4.0, 30):
        seeds.append(_arcsine_like(n, power))

    # Gaussian windows with varying width
    for sigma in np.linspace(0.12, 0.35, 15):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws (different concentration parameters)
    for alpha in [0.03, 0.05, 0.08, 0.12, 0.2, 0.4, 0.7, 1.2, 2.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # upscale previous best if available
    if previous is not None:
        try:
            if previous.shape != (n,):
                seeds.append(_upscale(previous, n))
            else:
                seeds.append(previous / previous.sum())
        except Exception:
            pass

    return seeds

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The function respects a global 1000 s time budget and returns the best
    sequence (list of floats) found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    # reuse a previously discovered construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Choose the working length – 4000 gives a good trade‑off between
    # expressiveness and speed for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation + quick evaluation
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, previous)

    best_seq = None
    best_val = float("inf")
    evaluated = []  # (value, seq)

    print(f"\n=== Seed evaluation – length {n} ({len(seeds)} seeds) ===")
    for idx, seq in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective(seq)
        evaluated.append((val, seq))
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[seed {idx:03d}] new global best = {val:.6f}")

    # keep a few most promising seeds for the heavy optimisation stages
    evaluated.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in evaluated[:min(8, len(evaluated))]]

    # ------------------------------------------------------------------
    # 2️⃣  Adam refinement on the top seeds
    # ------------------------------------------------------------------
    print("\n=== Adam refinement stage ===")
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        # allocate a modest number of iterations – the exact amount is
        # limited by the remaining time.
        seq_opt, val_opt = _adam_optimize(
            seed_vec,
            n,
            deadline,
            max_iters=15000,
            beta0=2.0,
            beta_target=500.0,
            beta_mul=1.002,
            lr0=0.6,
        )
        if seq_opt is not None and val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"[Adam overall] seed {idx} improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Sub‑gradient refinement (long run on the current best)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n=== Sub‑gradient refinement stage ===")
        seq_ref, val_ref = _subgradient_refine(
            best_seq,
            n,
            deadline,
            max_iters=200000,
            patience=3000,
        )
        if seq_ref is not None and val_ref < best_val:
            best_val = val_ref
            best_seq = seq_ref.copy()
            print(f"[SubGrad overall] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n=== Simulated‑annealing fine‑tuning ===")
        seq_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            deadline,
            max_iters=None,  # run until time out
        )
        if seq_ann is not None and val_ann < best_val:
            best_val = val_ann
            best_seq = seq_ann.copy()
            print(f"[Anneal overall] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Symmetrisation (averaging with the reversed sequence)
    # ------------------------------------------------------------------
    if best_seq is not None:
        sym = (best_seq + best_seq[::-1]) / 2.0
        sym /= sym.sum()
        sym_val = _objective(sym)
        if sym_val < best_val:
            best_val = sym_val
            best_seq = sym.copy()
            print(f"[symmetrisation] improved to {sym_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Finalisation
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    # cache the result for possible subsequent calls
    globals()["height_sequence_1"] = best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
