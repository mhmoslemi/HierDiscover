"""
Hybrid optimisation for the max‑auto‑convolution problem (enhanced).

The routine follows the same multi‑stage pipeline as the reference
implementation but with larger populations, more iterations and an
additional stochastic refinement phase.  The idea is simple:

* generate many analytic seeds (uniform, sine, geometric, beta, Gaussian,
  Dirichlet, softmax);
* run a stronger Differential‑Evolution (DE) search on a piece‑wise‑linear
  parametrisation;
* polish the DE result with Adam on a smooth‑max surrogate (more iterations);
* further polish with exact sub‑gradient (mirror descent) and a Euclidean
  projection variant;
* finish with a light stochastic hill‑climber and a random perturbation
  optimiser.

All stages respect the global time budget and keep the best sequence
found so far.  The returned sequence is a plain list of non‑negative
floats (on the simplex – scaling does not affect the objective).
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional – not used in the core pipeline


# ----------------------------------------------------------------------
# Low‑level numerical helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian window."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of normalised start vectors."""
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Beta shapes – dense grid
    for a in np.linspace(0.2, 2.0, 9):
        for b in np.linspace(0.2, 2.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows – σ from 5 % to 40 % of n
    for frac in np.linspace(0.05, 0.4, 8):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective (sum(a) is assumed to be 1)."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 25000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑softmax parametrisation using a smooth‑max
    surrogate (log‑sum‑exp).  Returns the best sequence encountered.
    """
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # initial smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 120000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent (mirror‑descent style) on the true max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated‑gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _proj_subgradient_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 250000,
                             target_sum: float | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (Euclidean projection onto the ℓ₁‑simplex).
    """
    if target_sum is None:
        target_sum = 1.0  # we keep the sequence normalised to sum = 1

    a = _project_simplex(start_seq, target_sum)
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size and Euclidean projection
        lr = 1.0 / np.sqrt(it)
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, target_sum)

    return best_seq, best_val


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 60000) -> tuple[np.ndarray, float]:
    """Simple random pair‑wise mass transfer hill‑climber."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02  # initial step scale

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:6d} best={val:.6f}")

        # slowly tighten the step size
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


def _random_refine(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000,
                   sigma0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbation + simplex projection.
    Accepts any improvement; otherwise keeps the current point.
    """
    rng = np.random.default_rng()
    a = _project_simplex(start_seq, 1.0)  # ensure normalised
    best_seq = a.copy()
    best_val = _objective(a, n)

    sigma = sigma0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        cand = a + rng.normal(scale=sigma, size=n)
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand, 1.0)

        val = _objective(cand, n)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[RandRef] n={n:5d} it={it:6d} best={val:.6f}")

        if val < _objective(a, n):
            a = cand

        # slowly shrink the noise scale
        if it % 5000 == 0:
            sigma *= 0.6
            sigma = max(sigma, 1e-4)

    return best_seq, best_val

def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """Interpolate a low‑dimensional simplex vector `p` to a full sequence."""
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    val = _objective(a, n)
    return val, a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 80,
                 K: int | None = None,
                 max_gens: int = 500) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on a piece‑wise‑linear representation.
    """
    rng = np.random.default_rng()

    if K is None:
        K = max(30, int(np.sqrt(n) * 1.5))
    knots = np.linspace(0, n - 1, K)

    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]

    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_param(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            donor = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            donor = donor / ds if ds > 0 else _uniform(K)

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            trial = trial / ts if ts > 0 else _uniform(K)

            val_trial, seq_trial = _eval_param(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:4d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine combines a global DE stage,
    extensive analytic seeding and several deterministic & stochastic
    refinements.  Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 3.0

    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Choose a few problem sizes (larger sizes give more freedom but cost
    # more time).  We start with 6000, fall back to 4000 if we run out of
    # time.
    # ------------------------------------------------------------------
    lengths = [6000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1️⃣ Analytic seeds
        # --------------------------------------------------------------
        seeds = _make_seeds(n, previous)
        for s in seeds:
            val = _objective(s, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # --------------------------------------------------------------
        # 2️⃣ Global Differential Evolution
        # --------------------------------------------------------------
        seq_de, val_de = _de_optimize(n, deadline,
                                      popsize=80,
                                      K=None,
                                      max_gens=500)
        if val_de < global_best_val:
            global_best_val = val_de
            global_best_seq = seq_de.copy()
            print(f"[DE] n={n:5d} new global best={val_de:.6f}")

        # --------------------------------------------------------------
        # 3️⃣ Adam on smooth‑max surrogate (more iterations)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_adam, val_adam = _adam_optimize(global_best_seq, n,
                                                deadline, max_iters=30000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val_adam:.6f}")

        # --------------------------------------------------------------
        # 4️⃣ Exact sub‑gradient refinement (more iterations)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_sg, val_sg = _subgradient_refine(global_best_seq, n,
                                                 deadline, max_iters=150000)
            if val_sg < global_best_val:
                global_best_val = val_sg
                global_best_seq = seq_sg.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val_sg:.6f}")

        # --------------------------------------------------------------
        # 5️⃣ Projected sub‑gradient refinement (new)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_psg, val_psg = _proj_subgradient_refine(global_best_seq, n,
                                                       deadline,
                                                       max_iters=300000)
            if val_psg < global_best_val:
                global_best_val = val_psg
                global_best_seq = seq_psg.copy()
                print(f"[overall‑ProjSub] n={n:5d} new global best={val_psg:.6f}")

        # --------------------------------------------------------------
        # 6️⃣ Light stochastic hill‑climbing (final polishing)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_hc, val_hc = _hill_climb(global_best_seq, n,
                                         deadline, max_iters=80000)
            if val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[overall‑Hill] n={n:5d} new global best={val_hc:.6f}")

        # --------------------------------------------------------------
        # 7️⃣ Random Gaussian refinement (to escape possible local minima)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_rr, val_rr = _random_refine(global_best_seq, n,
                                            deadline, max_iters=25000)
            if val_rr < global_best_val:
                global_best_val = val_rr
                global_best_seq = seq_rr.copy()
                print(f"[overall‑RandRef] n={n:5d} new global best={val_rr:.6f}")

        # keep a copy for the next length (if any)
        previous = global_best_seq

    # ------------------------------------------------------------------
    # Fallback in the very unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Verify with the provided evaluate_sequence() – this also forces any
    # final clipping performed by the evaluator.
    # --------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    # store for potential reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
