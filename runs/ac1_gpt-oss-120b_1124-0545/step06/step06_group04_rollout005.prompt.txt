def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sin2_window(n: int) -> np.ndarray:
    """sin²‑window (Hann‑type)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2

def _hanning(n: int) -> np.ndarray:
    """Classical Hanning window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()

def _tent(n: int) -> np.ndarray:
    """Symmetric tent (linear rise‑fall)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)

def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) on (0,1) – uses interior points to avoid infinities."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:

def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 6000,
                   beta0: float = 2.0,
                   beta_target: float = 120.0,
                   beta_mul: float = 1.001,
                   lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed max loss L = (1/β)·log Σ exp(β·conv).
    Only the max‑term is kept (no variance regulariser) – this
    drives the actual objective more directly.
    Returns the best sequence and its true objective.
    """
    v = np.log(seed + 1e-30)                # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # current point on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # increase β gradually, but never exceed beta_target
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # smoothed max weights (log‑sum‑exp)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                     # under‑flow – fall back to hard max
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient wrt a (via reversed convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 100000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the genuine sub‑gradient
    of max(conv).  A slower step‑size (1/√t) is employed for stability.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)               # a slightly larger prefactor than before
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,

    """Lightweight local search that swaps tiny mass fragments."""

    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # starting temperature is a modest fraction of the current obj.
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # move ≤2.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global jitter

            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:

            else:

            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
