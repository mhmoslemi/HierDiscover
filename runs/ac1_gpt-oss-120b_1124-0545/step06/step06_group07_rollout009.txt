def _gd_monotone(seed, n, deadline, max_iters=20000):
    a = seed.copy()
    target_sum = np.sqrt(2.0 * n)
    a = _project_simplex(a, target_sum)
    # Ensure monotonic decreasing: we can sort descending and then isotonic regression keep decreasing.
    a = _project_monotone_nonincreasing(a)
    a = _project_simplex(a, target_sum)  # renormalize
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)
    beta = 5.0
    beta_mul = 1.02
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)

        # clip nonnegative

        a_tmp = _project_simplex(a_tmp, target_sum)

                best_seq = a.copy()
                print(f"[GD-Mono] n={n:5d} it={it:5d} best={cur_val:.6f}")
        beta *= beta_mul
    return best_seq, best_val

"""
Monotone‐gradient optimiser for the max‑auto‑convolution problem.

The routine builds a large library of analytic start vectors (uniform,
sine, geometric, exponential, linear, Gaussian, Beta‑shapes, Dirichlet
samples, etc.) – many of them monotone decreasing.  Each seed is put on
the ℓ₁‑simplex (∑a = √(2n)) and then refined by a simple projected‑gradient
descent that minimises a smoothed max (log‑sum‑exp) of the auto‑convolution.
After each gradient step the vector is projected onto the non‑negative
simplex **and** onto the cone of non‑increasing sequences (isotonic
regression).  The best sequence found is finally polished by the existing
sub‑gradient and stochastic hill‑climb refiners.

The algorithm respects the 1000 s budget, never exceeds 2 CPU threads and
returns a plain list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _project_monotone_nonincreasing(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the cone of non‑increasing vectors
    (isotonic regression for a decreasing sequence).
    """
    n = len(v)
    # Initialise blocks
    level = v.astype(float).copy()
    weight = np.ones(n, dtype=int)

    i = 0
    # PAVA loop
    while i < len(level) - 1:
        if level[i] < level[i + 1] - 1e-12:  # violation of decreasing order
            # Merge blocks i and i+1
            tot_w = weight[i] + weight[i + 1]
            avg = (weight[i] * level[i] + weight[i + 1] * level[i + 1]) / tot_w
            level[i] = avg
            weight[i] = tot_w
            # Delete merged entry
            level = np.delete(level, i + 1)
            weight = np.delete(weight, i + 1)
            # Step back to check previous block
            if i > 0:
                i -= 1
        else:
            i += 1

    # Expand to original length
    res = np.empty(n, dtype=float)
    pos = 0
    for lvl, w in zip(level, weight):
        res[pos:pos + w] = lvl
        pos += w
    return res


# ----------------------------------------------------------------------
# Analytic seed families (all are ℓ₁‑normalised later)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i (r∈(0,1])."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-rate * i)."""
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _linear_decay(n: int) -> np.ndarray:
    """Linear decrease a_i ∝ (n‑i). Ends at 0."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of ℓ₁‑normalised start vectors."""
    seeds: list[np.ndarray] = []

    # Classic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_linear_decay(n))

    # Geometric – a broad sweep of decay rates
    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_geom(n, r))

    # Exponential – various rates
    for rate in np.linspace(0.0005, 0.01, 7):
        seeds.append(_exp_decay(n, rate))

    # Power law – exponents 0..3
    for p in np.linspace(0.0, 3.0, 7):
        seeds.append(_power(n, p))

    # Beta shapes – a dense grid (including asymmetric extremes)
    for a in np.linspace(0.1, 5.0, 9):
        for b in np.linspace(0.1, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows – σ from 5 % to 40 % of n
    for frac in np.linspace(0.05, 0.4, 8):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    # Random Dirichlet draws
    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Softmax of random logits (dense)
    for _ in range(6):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # Include previous best if available
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Gradient descent on a smoothed max (with monotone projection)
# ----------------------------------------------------------------------


def _gd_monotone(seed: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the log‑sum‑exp surrogate of the
    max‑convolution, with Euclidean projection onto the ℓ₁‑simplex and
    the cone of non‑increasing vectors after every step.
    Returns the best sequence seen and its exact objective value.
    """
    target_sum = np.sqrt(2.0 * n)          # normalisation used in the paper
    a = seed.copy()
    a = _project_simplex(a, target_sum)
    a = _project_monotone_nonincreasing(a)
    a = _project_simplex(a, target_sum)

    best_seq = a.copy()
    cur_conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(cur_conv)

    beta = 5.0
    beta_mul = 1.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # smoothed‑max weights
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            beta *= beta_mul
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a:
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]

        # diminishing step‑size
        lr = 0.5 / np.sqrt(it)

        # Euclidean step, non‑negativity, monotone, simplex
        a_tmp = a - lr * grad
        a_tmp = np.maximum(a_tmp, 0.0)
        a_tmp = _project_monotone_nonincreasing(a_tmp)
        a_tmp = _project_simplex(a_tmp, target_sum)

        a = a_tmp

        # Refresh surrogate smoothness
        beta *= beta_mul

        # Occasionally check the true objective
        if it % 500 == 0 or it == max_iters:
            conv_cur = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv_cur)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[GD‑Mono] n={n:5d} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Existing deterministic refinements (re‑used)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    target_sum = np.sqrt(2.0 * n)
    a = _project_simplex(a, target_sum)

    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        a = a / s if s != 0 else _uniform(n)

    return best_seq, best_val


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a single problem size (larger n gives finer granularity but costs time)
    # ------------------------------------------------------------------
    n = 4000
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Load previous best if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Generate a rich collection of analytic seeds
    # ------------------------------------------------------------------
    print(f"\n=== Generating seeds for length {n} ===")
    raw_seeds = _make_seeds(n, previous)

    # Normalise every seed onto the simplex (sum = √(2n)) and keep the best few
    seed_info = []  # list of (value, seed)
    global_best_seq = None
    global_best_val = float("inf")

    for idx, s in enumerate(raw_seeds):
        s_norm = _project_simplex(s, target_sum)
        # optional monotone projection to keep the seed sensible
        s_norm = _project_monotone_nonincreasing(s_norm)
        s_norm = _project_simplex(s_norm, target_sum)

        conv = _conv_fft(s_norm, s_norm)
        val = 2.0 * n * np.max(conv)

        seed_info.append((val, s_norm))

        if val < global_best_val:
            global_best_val = val
            global_best_seq = s_norm.copy()
            print(f"[seed] idx={idx:02d} new global best={val:.6f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Keep the top K seeds for intensive refinement
    # ------------------------------------------------------------------
    seed_info.sort(key=lambda x: x[0])
    top_k = min(6, len(seed_info))
    top_seeds = [s for _, s in seed_info[:top_k]]

    # ------------------------------------------------------------------
    # Monotone projected gradient descent on each top seed
    # ------------------------------------------------------------------
    for i, seed in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n[Refine] GD‑Mono on top seed #{i}")
        seq, val = _gd_monotone(seed, n, deadline, max_iters=25000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Refine] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic polishing (sub‑gradient + hill‑climb)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline - 10:
        # Sub‑gradient refinement (shorter run – most time already spent)
        remaining = deadline - time.time()
        max_it = int(min(120000, max(20000, remaining * 200)))  # crude scaling
        seq_sg, val_sg = _subgradient_refine(global_best_seq, n, deadline, max_iters=max_it)
        if val_sg < global_best_val:
            global_best_val = val_sg
            global_best_seq = seq_sg.copy()
            print(f"[SubGrad] new global best={val_sg:.6f}")

    if global_best_seq is not None and time.time() < deadline - 5:
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, max_iters=50000)
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[Hill] new global best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        global_best_val = 2.0 * n * np.max(_conv_fft(global_best_seq, global_best_seq))
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Optional verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        check = evaluate_sequence(global_best_seq.tolist())
        print(f"Verification via evaluate_sequence(): {check}")
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
