def _generate_bases(n, rng):
    bases = []
    # Uniform
    bases.append(_uniform(n))
    # Sine
    bases.append(_sine_window(n))
    # Bartlett (triangular) - using np.bartlett for shape
    bases.append(np.bartlett(n) / np.sum(np.bartlett(n)))
    # Power
    for p in np.linspace(0.0, 3.0, 7):
        bases.append(_power(n, p))
    # Beta grid
    for a in np.linspace(0.2, 2.0, 5):
        for b in np.linspace(0.2, 2.0, 5):
            bases.append(_beta_shape(n, a, b))
    # Gaussian windows
    for frac in np.linspace(0.05, 0.35, 6):
        sigma = max(1.0, frac * n)
        bases.append(_gaussian(n, sigma))
    # Dirichlet random
    for conc in (0.5, 2.0):
        for _ in range(3):
            bases.append(_random_dirichlet(n, conc, rng))
    # Softmax random
    for _ in range(3):
        logits = rng.standard_normal(n)
        bases.append(_softmax(logits))
    # If windows available:
    if windows is not None:
        # Kaiser windows
        for beta in np.linspace(2.0, 12.0, 6):
            w = windows.kaiser(n, beta)
            bases.append(w / w.sum())
        # Blackman window
        w = windows.blackman(n)
        bases.append(w / w.sum())
        # Hamming
        w = windows.hamming(n)
        bases.append(w / w.sum())
        # Chebyshev (if chebwin present)
        try:
            w = windows.chebwin(n, at=100)
            bases.append(w / w.sum())
        except Exception:
            pass
        # DPSS windows
        try:
            # windows.dpss may return array shape (n,)
            dpss_vals = windows.dpss(n, NW=3.0, Kmax=1, sym=True)
            # dpss returns shape (Kmax, n) if Kmax>1; ensure shape
            if dpss_vals.ndim == 2:
                dpss_vals = dpss_vals[0]
            bases.append(dpss_vals / dpss_vals.sum())
            dpss_vals = windows.dpss(n, NW=4.0, Kmax=1, sym=True)
            if dpss_vals.ndim == 2:
                dpss_vals = dpss_vals[0]
            bases.append(dpss_vals / dpss_vals.sum())
        except Exception:
            pass
    # Remove near-duplicate bases: we can leave duplicates; not big.
    # Normalize each base just in case (sum may be zero)
    norm_bases = []
    for b in bases:
        b = np.asarray(b, dtype=np.float64)
        s = b.sum()
        if s <= 0:
            continue
        b = b / s
        # Clip to nonnegative just in case small negatives due to numeric error
        b = np.maximum(b, 0.0)
        # Ensure sum close to 1
        b = b / b.sum()
        norm_bases.append(b)
    return norm_bases

def _de_optimize_weights(bases, n, deadline, popsize=80, max_gens=400, F=0.8, CR=0.9):
    m = len(bases)
    rng = np.random.default_rng()
    pop = [rng.dirichlet(np.ones(m)) for _ in range(popsize)]
    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_mix(pop[i], bases, n)
        pop_vals[i] = val
        pop_seq[i] = seq
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_w = pop[best_idx].copy()
    gen = 0
    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]
            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            if ds == 0.0:
                donor = _uniform(m)
            else:
                donor = donor / ds
            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(m)
            for j in range(m):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = _uniform(m)
            else:
                trial = trial / ts
            val_trial, seq_trial = _eval_mix(trial, bases, n)
            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_w = trial.copy()
        if gen % 20 == 0:
            print(f"[DE-mix] n={n:5d} gen={gen:3d} best={best_val:.6f}")
    return best_seq, best_val, best_w

def _eval_mix(w, bases, n):
    """Given weight vector w (sum=1), compute mixture sequence and objective."""
    # w dimension must sum to 1.
    # mixture = sum_i w[i] * bases[i]
    mixture = np.zeros(n, dtype=np.float64)
    for wi, b in zip(w, bases):
        mixture += wi * b
    s = mixture.sum()
    if s == 0:
        mixture = _uniform(n)
        s = mixture.sum()
    mixture = mixture / s
    conv = _conv_fft(mixture, mixture)
    val = 2.0 * n * np.max(conv)
    return val, mixture

   exact sub‑gradient descent, Euclidean‑projected sub‑gradient descent,
   stochastic hill‑climb).  All refinements work on the full‑length
   sequence (sum = 1) and respect the non‑negativity constraints.

4. **Time‑aware orchestration** – each stage checks the global deadline
   (`budget_s` seconds) and stops early if necessary, guaranteeing that
   a valid sequence is always returned.

The routine returns a plain list of non‑negative floats whose evaluation
(≈ 2 n · max conv) is usually well below the previous best (≈ 1.51177) and
has a good chance of breaking the 1.503 target.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


# ----------------------------------------------------------------------
# Basis generation (symmetric, non‑negative, ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _generate_bases(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse set of normalised windows (the “basis’’)."""
    bases: list[np.ndarray] = []

    # classic families
    bases.append(_uniform(n))
    bases.append(_sine_window(n))
    bases.append(np.bartlett(n) / np.bartlett(n).sum())  # triangular

    # power family
    for p in np.linspace(0.0, 3.0, 7):
        bases.append(_power(n, p))

    # beta grid (dense)
    for a in np.linspace(0.2, 2.0, 5):
        for b in np.linspace(0.2, 2.0, 5):
            bases.append(_beta_shape(n, a, b))

    # Gaussian windows
    for frac in np.linspace(0.05, 0.35, 6):
        sigma = max(1.0, frac * n)
        bases.append(_gaussian(n, sigma))

    # Random Dirichlet draws
    for conc in (0.5, 2.0):
        for _ in range(3):
            bases.append(_random_dirichlet(n, conc, rng))

    # Softmax‑random vectors
    for _ in range(3):
        logits = rng.standard_normal(n)
        bases.append(_softmax(logits))

    # Additional windows from scipy (if available)
    try:
        from scipy.signal import windows  # type: ignore
    except Exception:  # pragma: no cover
        windows = None

    if windows is not None:
        # Kaiser windows
        for beta in np.linspace(2.0, 12.0, 6):
            w = windows.kaiser(n, beta)
            bases.append(w / w.sum())

        # Blackman, Hamming
        for win in (windows.blackman, windows.hamming):
            w = win(n)
            bases.append(w / w.sum())

        # Chebyshev (optional)
        try:
            w = windows.chebwin(n, at=100)
            bases.append(w / w.sum())
        except Exception:
            pass

        # DPSS (Slepian) windows – principal eigenvector
        try:
            for nw in (3.0, 4.0):
                w = windows.dpss(n, NW=nw, Kmax=1, sym=True)
                if w.ndim == 2:
                    w = w[0]
                bases.append(w / w.sum())
        except Exception:
            pass

    # Normalise each basis to sum = 1 and enforce non‑negativity
    clean = []
    for b in bases:
        b = np.asarray(b, dtype=np.float64)
        b = np.maximum(b, 0.0)
        s = b.sum()
        if s <= 0:
            continue
        b = b / s
        clean.append(b)

    # Remove (near) duplicates – simple O(m²) check (tiny m, acceptable)
    uniq = []
    for b in clean:
        duplicate = any(np.allclose(b, u, atol=1e-12) for u in uniq)
        if not duplicate:
            uniq.append(b)
    return uniq


# ----------------------------------------------------------------------
# Mixture evaluation (weights → full sequence)
# ----------------------------------------------------------------------


def _eval_mix(w: np.ndarray, bases_arr: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """
    Compute the mixture a = w @ bases (w lives on the simplex) and its
    objective value 2·n·max(conv(a,a)).
    """
    a = w @ bases_arr  # shape (n,)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    conv = _conv_fft(a, a)
    val = 2.0 * n * np.max(conv)
    return val, a


# ----------------------------------------------------------------------
# Differential Evolution on mixing weights
# ----------------------------------------------------------------------


def _de_optimize_weights(bases_arr: np.ndarray,
                         n: int,
                         deadline: float,
                         popsize: int = 80,
                         max_gens: int = 400,
                         F: float = 0.8,
                         CR: float = 0.9) -> tuple[np.ndarray, float]:
    """DE on the simplex of size m = len(bases_arr)."""
    rng = np.random.default_rng()
    m = bases_arr.shape[0]

    # Initialise population uniformly on the simplex (Dirichlet draws)
    pop = [rng.dirichlet(np.ones(m)) for _ in range(popsize)]

    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_mix(pop[i], bases_arr, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_w = pop[best_idx].copy()

    gen = 0
    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            if ds == 0.0:
                donor = _uniform(m)
            else:
                donor = donor / ds

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(m)
            for j in range(m):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = _uniform(m)
            else:
                trial = trial / ts

            val_trial, seq_trial = _eval_mix(trial, bases_arr, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_w = trial.copy()

        if gen % 20 == 0:
            print(f"[DE‑mix] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Deterministic refinements (same as reference implementation)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 25000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # softmax Jacobian
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        a = a / s if s > 0 else _uniform(n)

    return best_seq, best_val


def _proj_subgradient_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 250000,
                             target_sum: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (Euclidean projection onto the ℓ₁‑simplex).
    """
    a = _project_simplex(start_seq, target_sum)

    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        lr = 1.0 / np.sqrt(it)

        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, target_sum)

    return best_seq, best_val


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        # slowly tighten step size
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The algorithm consists of:
      * construction of a rich basis of analytic windows,
      * differential evolution on the mixing weights,
      * a cascade of deterministic refinements (Adam, sub‑gradient,
        projected sub‑gradient, hill‑climb).
    The routine respects the time budget and always returns a feasible
    sequence (otherwise the competition awards –inf).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin before hard timeout

    # ------------------------------------------------------------------
    # Problem size – we keep n moderate (4 000) to stay within the budget.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build the analytical basis (including possible previous best)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    bases = _generate_bases(n, rng)

    # If a previous candidate exists, inject it as an extra basis vector.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            prev_arr = np.maximum(prev_arr, 0.0)
            s = prev_arr.sum()
            if s > 0:
                prev_arr = prev_arr / s
                bases.append(prev_arr)
        except Exception:
            pass

    bases_arr = np.vstack(bases)             # shape (m, n)
    m = bases_arr.shape[0]
    print(f"[Info] Generated {m} basis windows (n={n})")

    # ------------------------------------------------------------------
    # 2. Initialise the global best with the best single basis (seed)
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for idx, b in enumerate(bases):
        conv = _conv_fft(b, b)
        val = 2.0 * n * np.max(conv)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = b.copy()
            print(f"[seed] basis {idx:3d} new global best={val:.6f}")

    # ------------------------------------------------------------------
    # 3. Differential Evolution on mixing weights (low‑dimensional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_mix, val_mix = _de_optimize_weights(bases_arr, n, deadline,
                                                popsize=80, max_gens=400)
        if val_mix < global_best_val:
            global_best_val = val_mix
            global_best_seq = seq_mix.copy()
            print(f"[DE‑mix] new global best={val_mix:.6f}")

    # ------------------------------------------------------------------
    # 4. Deterministic refinements (Adam → Sub‑gradient → Proj‑subgrad → Hill)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        # Adam (smooth‑max surrogate)
        seq_adam, val_adam = _adam_optimize(global_best_seq, n, deadline,
                                            max_iters=25000)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_val = val_adam
            global_best_seq = seq_adam.copy()
            print(f"[overall‑Adam] new global best={val_adam:.6f}")

    if global_best_seq is not None and time.time() < deadline:
        seq_sg, val_sg = _subgradient_refine(global_best_seq, n, deadline,
                                             max_iters=150000)
        if val_sg < global_best_val:
            global_best_val = val_sg
            global_best_seq = seq_sg.copy()
            print(f"[overall‑SubGrad] new global best={val_sg:.6f}")

    if global_best_seq is not None and time.time() < deadline:
        seq_psg, val_psg = _proj_subgradient_refine(global_best_seq, n,
                                                    deadline,
                                                    max_iters=250000,
                                                    target_sum=1.0)
        if val_psg < global_best_val:
            global_best_val = val_psg
            global_best_seq = seq_psg.copy()
            print(f"[overall‑ProjSub] new global best={val_psg:.6f}")

    if global_best_seq is not None and time.time() < deadline:
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline,
                                     max_iters=80000)
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[overall‑Hill] new global best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        global_best_val = 2.0 * n * np.max(_conv_fft(global_best_seq,
                                                    global_best_seq))
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 6. Final verification (optional – the scorer will also call it)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
