"""
propose_candidate.py

A refined optimiser for the step‑function objective
    2·n·max(conv(a,a)) / (sum(a)²)

The optimiser builds a large and diverse collection of seed vectors
including symmetric and asymmetric beta‑pdf windows, geometric,
power‑law and classic spectral windows.  After a quick evaluation it
keeps the most promising seeds and runs three optimisation phases:

* **Phase A – Adam on a smooth‑max surrogate** – a log‑sum‑exp
  approximation of the ℓ∞‑norm of the convolution is minimised on the
  simplex (log‑softmax parametrisation).

* **Phase B – Exact sub‑gradient mirror‑descent** – the true non‑smooth
  objective is handled with an exponentiated‑gradient update.  The loop
  runs as long as the time budget permits.

* **Phase C – Simulated‑annealing fine‑tuning** – a Metropolis walk that
  transfers a small amount of mass between two random positions and
  occasionally adds a global Gaussian perturbation.

All vectors are kept non‑negative and normalised to sum = 1 (the
objective is scale‑invariant).  The routine respects the supplied time
budget (default 1000 s) and returns the best sequence found as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The vector is assumed to be non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families of probability vectors (all sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1) discretised at mid‑points.
    α∈(0,1] → edge‑heavy shape, α=1 → uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_asym_window(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Asymmetric Beta(α,β) density on (0,1).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, beta - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Seed generation – a rich portfolio of start vectors
# ----------------------------------------------------------------------


def _make_seeds(
    n: int, previous: np.ndarray | None, rng: np.random.Generator
) -> list[np.ndarray]:
    """
    Assemble a large and diverse set of start vectors,
    all normalised to sum = 1.
    """
    seeds: list[np.ndarray] = []

    # baseline families -------------------------------------------------
    seeds.append(_uniform(n))

    # geometric families (r near 1 gives slow decay, r small gives fast)
    for r in [0.3, 0.5, 0.7, 0.85, 0.9, 0.95, 0.98, 0.99]:
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families ------------------------------------------------
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        seeds.append(_power(n, p))

    # combined geometric‑power families --------------------------------
    for r in [0.9, 0.95, 0.98]:
        for p in [0.5, 1.0, 1.5, 2.0]:
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric (mirror) --------------------------------------
    for r in [0.9, 0.95, 0.98, 0.99]:
        seeds.append(_sym_geom(n, r))

    # classic spectral windows -----------------------------------------
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in [0.2, 0.4, 0.6, 0.8]:
        seeds.append(_tukey_window(n, alpha))

    # Gaussian family – fine sweep of sigma factor
    for sigma_factor in np.linspace(0.06, 0.24, 9):
        seeds.append(_gaussian(n, sigma_factor))

    # symmetric Beta pdf – finer grid
    for alpha in np.linspace(0.5, 1.0, 11):
        seeds.append(_beta_window(n, alpha))

    # asymmetric Beta pdf – exploratory grid
    alphas = [0.3, 0.5, 0.7, 0.9]
    betas = [0.3, 0.5, 0.7, 0.9]
    for a in alphas:
        for b in betas:
            if a == 1.0 and b == 1.0:
                continue
            seeds.append(_beta_asym_window(n, a, b))

    # random soft‑max vectors
    for _ in range(15):
        seeds.append(_random_softmax(n, rng))

    # carry over a previous construction (if sized correctly)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final sanitisation – clip, renormalise, discard empties
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0.0:
            cleaned.append(s / total)
    return cleaned

# Simple utilities

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 12000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser applied to the log‑sum‑exp approximation of
    max(conv(a,a)).  Returns the best vector seen and its exact
    objective value.
    """
    # initialise logits (log‑softmax parametrisation)
    v = np.log(seed + 1e-30)

    # Adam moments
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    # hyper‑parameters (slightly tuned)
    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # temperature schedule for the smooth‑max
    beta = 2.0
    beta_mul = 1.003  # grow a bit slower than in the reference code

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k · (a*a)_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase B – Exact sub‑gradient mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true non‑smooth objective.
    Returns the best vector encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where the max is (almost) attained
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient – average over all active constraints
        if len(active) == 0:
            g = np.zeros_like(a)
        else:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1 : n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.95 / np.sqrt(it)  # slightly larger than the reference
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)   # keep numbers stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase C – Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Metropolis‑Hastings walk that moves a small amount of mass between
    random positions and occasionally adds a global Gaussian perturbation.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a few percent of the current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9994  # a slightly slower decay to allow longer exploration

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move up to 5 % of the source mass (larger jumps than the reference)
        delta = rng.random() * 0.05 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # every 1000 steps add a small Gaussian perturbation to stay out of plateaux
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget and
    returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the grader
    deadline = start_time + budget_s - 2.0

    # previous result (allows chaining across runs)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Length of the sequence – 4000 is a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # 1) Build a large and diverse set of seeds
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # 2) Quick evaluation – keep the most promising seeds only
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")

    # record the best raw seed (might already be very good)
    if seed_vals[0][1] < global_best_val:
        global_best_val = seed_vals[0][1]
        global_best_seq = seed_vals[0][0].copy()
        print(f"[seed] initial best={global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation on the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=12000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient mirror‑descent (main workhorse)
    # ------------------------------------------------------------------
    if global_best_seq is not None and global_best_seq.shape[0] == n:
        # allocate as much time as we have left, but cap at 350 000 iterations
        remaining = deadline - time.time()
        max_iters = 350_000
        seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=max_iters)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Simulated‑annealing fine‑tuning (uses all remaining time)
    # ------------------------------------------------------------------
    if global_best_seq is not None and global_best_seq.shape[0] == n:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely) – uniform distribution
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # store for potential subsequent runs
    globals()["height_sequence_1"] = global_best_seq.copy()
    # the grader expects a plain Python list
    return global_best_seq.tolist()
