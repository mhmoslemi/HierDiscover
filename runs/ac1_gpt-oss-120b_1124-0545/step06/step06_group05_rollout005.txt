    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n

    if s == 0:

    return a / s

def propose_candidate(seed=42, budget_s=1000):

        seeds = []

        # arcsine base seeds for many p
        for p in np.linspace(-2.0, -0.05, 40):
            seed = _arcseed(n, p)
            seeds.append(seed)

        # arcsine+uniform mixture
        uniform = _uniform(n)
        for p in np.linspace(-2.0, -0.05, 30):
            arc = _arcseed(n, p)
            for w in np.linspace(0.0, 0.5, 6):

                seed = seed / seed.sum()

        # include previous if exists

        seed_vals = [(seq, _objective(seq)) for seq in seeds]

        for seq,val in seed_vals[:1]:
            if val < best_obj:

        # For each top seed run refined optimization (Adam + subgradient)

    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n

    if s == 0:

"""
Propose a step‑function sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm:
1.  For each chosen length (4000, then 8000 if time permits) generate a
    large, diverse set of *analytic* seeds:
       – arcsine family   a[i] ∝ ((i+½)(n‑½‑i))**p   (p < 0)
       – mixture of arcsine and uniform
       – beta‑distribution family  a[i] ∝ (i+½)^{α‑1}(n‑i‑½)^{β‑1}
       – Dirichlet random draws (sparse to dense)
2.  Evaluate the true objective on every seed and keep the best few.
3.  For each of those seeds run a smooth‑max Adam optimiser
    (log‑softmax parametrisation).  The loss is
        L = (1/β)·log∑exp(β·conv)  +  λ·‖conv‖₂² .
    This quickly drives the sequence toward a low ‖conv‖∞.
4.  Refine the result with a mirror‑descent (exponentiated‑gradient)
    sub‑gradient method that works on the *exact* max‑convolution term.
5.  (If time remains) a cheap local hill‑climbing / annealing stage
    that shuffles tiny amounts of mass between random indices.
6.  Return the best sequence found (normalised to sum = 1).  The
    global best is also stored in ``height_sequence_1`` for possible
    reuse in a later run.

The whole search respects the supplied time budget (default 1000 s)
and prints progress reports.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large vectors)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the target objective.
    The sequence must be non‑negative; the function is scale invariant.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    return np.ones(n, dtype=np.float64) / n


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """
    Discrete analogue of the arcsine law.
    a[i] ∝ ((i+½)*(n‑½‑i))**exponent,   exponent < 0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide="ignore", invalid="ignore"):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete beta‑distribution on the set {0,…,n‑1}:
        a[i] ∝ (i+½)^{α‑1}·(n‑½‑i)^{β‑1}.
    Both parameters must be >0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n                       # in (0,1)
    with np.errstate(divide="ignore", invalid="ignore"):
        a = np.exp((alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x))
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration α (Gamma(α,1) normalised)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    s = vec.sum()
    if s <= 0.0:
        return _uniform(n)
    return vec / s


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000,
                   beta0: float = 0.5,
                   beta_mul: float = 1.02,
                   lr0: float = 1.0,
                   lambda0: float = 5.0,
                   lambda_decay: float = 0.996) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        L(a) = (1/β)·log∑exp(β·conv(a,a)) + λ·‖conv(a,a)‖₂² .
    The iterate is kept on the simplex via a log‑softmax representation.
    Returns the best sequence (as a numpy array) and its true objective.
    """
    # Initialise logits (log‑softmax parameters)
    v = np.log(seed + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0
    lam = lambda0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                # point on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max gradient (log‑sum‑exp) -----
        beta *= beta_mul                # increase smoothing gradually
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of the smoothed max term
        rev_a = a[::-1]
        full_max = _conv_fft(rev_a, w)
        grad_max = 2.0 * full_max[n - 1:n - 1 + n]

        # gradient of the variance regulariser λ·‖conv‖₂²
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n - 1:n - 1 + n]

        grad_a = grad_max + lam * grad_sq

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam updates on logits (bias‑corrected)
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        # decay the variance regulariser slowly
        lam *= lambda_decay

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 60000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true L∞ term.
    Works directly on the simplex; returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _local_anneal(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_steps: int = 20000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing / Metropolis step that moves a tiny fraction
    of mass between two random indices.  Used only as a final polish.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    step = 0
    while step < max_steps and time.time() < deadline:
        step += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.random() * 0.02 * cur[i]   # move ≤2 % of the source mass
        delta = min(delta, cur[i])
        if delta <= 0.0:
            continue
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_val = _objective(new_seq)

        # accept if better; otherwise accept with a tiny probability
        if new_val < cur_val or rng.random() < 0.001:
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} step={step:5d} best={new_val:.6f}")

    return best, best_val

                prev = prev / max(prev.sum(), 1e-12)
            else:
                prev = None
        except Exception:
            prev = None
    else:
        prev = None

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        seeds = []

        for p in np.linspace(-2.0, -0.05, 36):

                seeds.append(mix)

        top_seeds = [s for s, _ in seed_vals[:top_k]]

            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) Adam optimisation on the top seeds
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq_adam, val_adam = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=20000,
                beta0=0.4,
                beta_mul=1.02,
                lr0=1.0,
                lambda0=4.0,
                lambda_decay=0.997,
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val_adam:.6f}")

        # ------------------------------------------------------------------
        # 4) Exact sub‑gradient refinement (mirrored‑gradient)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_sub, val_sub = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=80000
            )
            if seq_sub is not None and val_sub < global_best_val:
                global_best_val = val_sub
                global_best_seq = seq_sub.copy()
                print(f"[overall SubGrad] n={n} best={val_sub:.6f}")

        # ------------------------------------------------------------------
        # 5) Final cheap annealing / hill‑climbing (if time left)

                global_best_seq, n, deadline, max_steps=30000

            if seq_ann is not None and val_ann < global_best_val:
                global_best_val = val_ann
                global_best_seq = seq_ann.copy()
                print(f"[overall Anneal] n={n} best={val_ann:.6f}")

    # ------------------------------------------------------------------
    # 6) Fallback – should never be needed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    # Normalise (protect against accidental zero‑sum)
    global_best_seq = np.maximum(global_best_seq, 0.0)
    s = global_best_seq.sum()
    if s < 1e-12:
        global_best_seq = _uniform(global_best_seq.size)
    else:
        global_best_seq = global_best_seq / s

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for possible reuse by a later invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
