"""
Propose a non‑negative step‑function that minimizes

        2·n·max(conv(a,a)) / (∑a)²

The routine follows three optimisation phases:

1️⃣  **Diverse seeding** – a large collection of analytically‑motivated families
    (uniform, sine, geometric, power‑law, beta‑like, Gaussian, Dirichlet,
    sparse) is generated and normalised to sum = 1.  If a previous construction
    exists it is interpolated to the current length.

2️⃣  **Mirror‑descent on the exact max‑convolution** – an exponentiated‑gradient
    (mirror descent on the simplex) updates the current best seed using the
    exact sub‑gradient of `max(conv)`.  The schedule `η = 0.9/√t` is a bit more
    aggressive than the reference implementation.

3️⃣  **Heavy‑weight simulated‑annealing** – starting from the refined vector,
    a fast incremental SA moves a small random subset (2‑4) of entries via a
    Dirichlet redistribution.  Only the affected entries of the full
    convolution are updated (O(n·k) per move).  Temperature starts at a
    noticeable fraction of the current objective and decays geometrically.
    The move count is limited only by the remaining time budget.

At the end a tiny deterministic “gradient‑swap” polishing loop is applied.
The function respects the 1000 s wall‑time limit and returns the best sequence
found as a plain ``list[float]``.
"""

import time
import numpy as np

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Objective for a vector that is already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (all return vectors that sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_factor: float) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet draw (conc < 1 → sparser)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on a log‑sum‑exp smooth approximation of ``max(conv)``.
    The returned pair is (best_sequence, best_objective) – the latter is the
    exact objective computed with ``_objective`` (no smoothing).
    """
    eps = 1e-30
    v = np.log(seed + eps)                     # optimisation variables (log‑softmax)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # smooth‑max concentration
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # stable shift
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t. a -----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # ----- convert to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping to avoid exploding steps
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated gradient) on the true
    ``max(conv)``.  Step size η = 0.9/√t (slightly larger than the reference
    implementation) gives a faster descent.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent on simplex)
        lr = 0.9 / np.sqrt(it)                # more aggressive than 0.7
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # avoid overflow
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             init_temp_factor: float = 0.02,
                             min_temp: float = 1e-9,
                             max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    High‑frequency simulated‑annealing with an incremental convolution update.
    The move set reallocates mass among 2‑4 randomly chosen indices via a
    Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # Initial temperature proportional to the current objective
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Cooling schedule
        T = max(T * decay, min_temp)

        # Pick a small random subset of positions
        k = rng.integers(2, 5)                     # 2‑4 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # New allocation for the selected positions (Dirichlet)
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        deltas = new_alloc - cur[idx]

        # Quick skip if no change
        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental update of the full convolution -----
        conv_new = cur_conv.copy()

        # a * Δ + Δ * a terms
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i * Δ_j (i ≠ j) appear twice in Δ*Δ
        for i in range(k):
            for j in range(i + 1, k):
                pos = idx[i] + idx[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj

        alpha = rng.exponential(scale=1.0) + 0.1
        beta = rng.exponential(scale=1.0) + 0.1
        seeds.append(_beta_shape(n, alpha, beta))

    # random Dirichlet draws (different concentrations)
    for conc in (0.1, 0.3, 0.7, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8):
        for conc in (0.1, 0.3):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # Gaussian‑like bumps of various widths
    for sigma_factor in (0.07, 0.12, 0.20):
        seeds.append(_gaussian(n, sigma_factor))

    # ------------------------------------------------------------------
    # Re‑use a previous construction (if present) via linear interpolation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 2:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                interp = np.interp(x_new, x_old, prev)
                interp[interp < 0.0] = 0.0
                if interp.sum() > 0.0:
                    seeds.append(interp / interp.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Quick scan of raw seeds (exact objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"[seed] initial best objective ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Exact sub‑gradient mirror descent (fast, no smoothing)
    # ------------------------------------------------------------------
    seq, val = _subgradient_refine(best_seq, n, deadline,
                                   max_iters=200_000)
    if val < best_val:
        best_val, best_seq = val, seq.copy()
        print(f"[overall‑SubGrad] new global best ≈ {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on a smooth approximation of the max operator
    # ------------------------------------------------------------------
    seq, val = _adam_optimize(best_seq, n, deadline,
                               max_iters=8000)
    if seq is not None and val < best_val:
        best_val, best_seq = val, seq.copy()
        print(f"[overall‑Adam] new global best ≈ {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Heavy simulated‑annealing with incremental updates
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    print(f"Remaining time for SA ≈ {remaining:.1f}s")
    seq, val = _sa_incremental_optimize(best_seq, n, deadline,
                                        max_iters=2_000_000,
                                        init_temp_factor=0.02,
                                        min_temp=1e-12)
    if val < best_val:
        best_val, best_seq = val, seq.copy()
        print(f"[overall‑SA] new global best ≈ {val:.6f}")

    # ------------------------------------------------------------------
    # Small deterministic polishing (gradient‑swap)
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    for it in range(5_000):
        conv = _conv_fft(cur, cur)
        k_star = int(np.argmax(conv))
        # sub‑gradient for the active index
        g = np.zeros_like(cur)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * cur[rev_idx]

        # pick a high‑gradient entry (to lose mass) and a low‑gradient entry (to gain)
        i_high = np.argmax(g)
        i_low = np.argmin(g)
        if cur[i_high] <= 1e-12:
            continue
        eps_move = min(cur[i_high] * 0.05, 1e-4)
        cur[i_high] -= eps_move
        cur[i_low] += eps_move
        cur /= cur.sum()

        new_val = _objective(cur)
        if new_val < best_val - 1e-9:
            best_val, best_seq = new_val, cur.copy()
            print(f"[polish] it={it:5d} new best ≈ {new_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence (optional)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
