def _project_simplex(v):
    """Project onto simplex {x>=0, sum(x)=1}."""
    # shift to make numeric stable
    v = np.asarray(v, dtype=np.float64)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # fallback to uniform
        return np.full_like(v, 1.0/v.shape[0])
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid smooth‑max Adam + projected sub‑gradient + annealing.

The routine builds a diverse collection of analytic and random seeds,
keeps the few most promising ones and refines each seed with Adam on a
log‑sum‑exp surrogate of the max‑convolution (the smoothing parameter β
is increased gradually).  The best result is then improved by a
projected sub‑gradient descent on the true max‑convolution (Euclidean
projection onto the simplex) and finally a cheap Metropolis–Hastings
annealing that swaps tiny amounts of mass between random positions.
All steps respect the given time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective of the challenge (scale‑invariant).

    2·n·max(conv(seq,seq)) / (sum(seq))²   with seq ≥ 0.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcseed(n: int, exponent: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide='ignore', invalid='ignore'):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    s = vec.sum()
    return vec / s if s > 0 else _uniform(n)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of start‑vectors, already normalised to sum = 1.
    """
    rng = np.random.default_rng(123456)
    seeds: list[np.ndarray] = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_geom(n, r=0.999))
    seeds.append(_geom(n, r=0.995))
    seeds.append(_power(n, p=0.5))
    seeds.append(_power(n, p=1.0))
    seeds.append(_power(n, p=1.5))

    for exp in [-1.2, -1.0, -0.8, -0.6, -0.5, -0.4]:
        seeds.append(_arcseed(n, exp))

    # a few Dirichlet draws (sparse to dense)
    for a in [0.05, 0.1, 0.3, 0.8, 1.5]:
        seeds.append(_dirichlet_seed(n, a, rng))

    # mirrored versions (helps when optimum is asymmetric)
    seeds.extend([s[::-1].copy() for s in seeds])

    # reuse previous best if available and compatible
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / max(previous.sum(), 1e-12))

    # guarantee non‑empty list
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Phase 1 – Adam on smooth max (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 25000,
                   beta0: float = 0.5,
                   beta_mul: float = 1.02,
                   beta_max: float = 200.0,
                   lr0: float = 0.8,
                   eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on

        L(a) = (1/β)·log Σ_k exp(β·conv_k(a))

    using a log‑softmax parametrisation.
    Returns the best sequence encountered and its true objective.
    """
    v = np.log(seed + 1e-30)        # logits for the simplex
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = np.inf
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        shift = conv.max()                # numerical stabilisation
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # degenerate β – just continue with current β
            pass
        else:
            w /= w_sum

        # ----- gradient of smooth max -----
        rev_a = a[::-1]
        grad = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

        # ----- chain rule through softmax -----
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # ----- Adam update -----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it + 1.0)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

        # increase smoothing, respecting an upper bound
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------


def _pg_subgradient_optimize(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 40000,
                             eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent using the exact sub‑gradient
    of max(conv).  Operates directly on the simplex.
    Returns the best sequence seen and its true objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[PG] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size decaying like 1/√t
        eta = eta0 / np.sqrt(it + 1.0)
        a = _project_simplex(a - eta * g)

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Very cheap annealing that moves a tiny fraction of probability mass
    between two random coordinates and occasionally adds a small global
    Gaussian jitter.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a small amount from i to j
        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # periodic global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            jittered = cur + noise
            jittered = np.clip(jittered, 0.0, None)
            if jittered.sum() == 0.0:
                jittered = _uniform(n)
            else:
                jittered /= jittered.sum()
            jitter_val = _objective(jittered)
            if jitter_val < cur_val or rng.random() < np.exp(-(jitter_val - cur_val) / (T + 1e-12)):
                cur = jittered
                cur_val = jitter_val
                if jitter_val < best_val:
                    best_val = jitter_val
                    best = jittered.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={jitter_val:.6f}")

    return best, best_val

        print("[fallback] using uniform distribution")

        f"\n=== Finished ===\nBest objective ≈ {best_overall_val:.6f} "
        f"(length {len(best_overall_seq)})"
    )
    return best_overall_seq.tolist()
