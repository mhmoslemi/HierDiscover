"""
Improvements over the previous baseline
=======================================

The objective to minimise is

    2·n·max(conv(a,a)) / (∑a)² ,

and because the expression is homogeneous we may normalise the sequence so
that ``∑a = 1``; the denominator then becomes constant and the problem reduces
to minimising ``max(conv(a,a))`` for a probability vector.

The new search consists of three stages that are executed for several
candidate lengths (2000, 4000, 6000, 8000).  All stages respect the global
time‑budget (default 1000 s).

1️⃣  **Diverse seeds** – deterministic families (uniform, linear, quadratic,
    logistic, beta‑like) together with a few random Dirichlet draws are
    normalised to sum = 1.

2️⃣  **Exponential‑gradient (mirror) descent** – a sub‑gradient of the
    non‑smooth ``max(conv)`` is used.  The update is performed in the
    log‑space (the classic “exponentiated‑gradient” step) and the iterate is
    projected back onto the simplex.  A larger step‑size constant (``η₀=0.9``)
    and a higher iteration limit (up to 600 000 per seed) give a finer
    descent than the previous code.

3️⃣  **Targeted simulated‑annealing** – starting from the best result of the
    previous phase we repeatedly move a small amount of mass from a point
    with the *largest* sub‑gradient (i.e. the point that contributes most to
    the current maximal convolution) to a point with the *smallest*
    sub‑gradient.  The move is accepted if it improves the objective or
    with Metropolis probability.  Incremental convolution updates keep the
    cost of a move at ``O(n·k)`` (``k≤4``).  A slower cooling schedule
    (``T←max(T·0.9999,1e‑9)``) and up to two million moves allow the
    algorithm to squeeze out extra improvement.

All helper functions are top‑level, there are no closures or lambdas, and
the routine finishes by returning the best sequence that was found (as a
plain ``list[float]``).  The returned sequence is safe for the provided
``evaluate_sequence`` function (non‑negative, finite, sum ≥ 0.01).

def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming ∑a == 1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project *v* onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Analytic seed families (always normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_desc(n: int) -> np.ndarray:
    """a[i] ∝ (n‑i).  Linear decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    return a / a.sum()


def _quadratic_desc(n: int) -> np.ndarray:
    """a[i] ∝ (n‑i)²."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** 2
    return a / a.sum()


def _cubic_desc(n: int) -> np.ndarray:
    """a[i] ∝ (n‑i)³."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** 3
    return a / a.sum()


def _logistic_desc(n: int, steep: float = 12.0) -> np.ndarray:
    """Logistic (sigmoid) decreasing shape."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - 0.5)))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex."""
    return rng.dirichlet(np.full(n, conc))


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_linear_desc(n))
    seeds.append(_quadratic_desc(n))
    seeds.append(_cubic_desc(n))
    seeds.append(_logistic_desc(n, steep=8.0))
    seeds.append(_logistic_desc(n, steep=16.0))

    # a few beta‑like asymmetrics
    for a in (0.3, 0.7, 1.0):
        for b in (0.3, 0.7, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (different concentrations)
    for conc in (0.4, 1.0, 2.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # reuse a previous solution if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Exponential‑gradient (mirror) descent on max(conv)
# ----------------------------------------------------------------------


def _exp_gradient_refine(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Mirror descent with KL‑divergence (exponentiated‑gradient) using the exact
    sub‑gradient of ``max(conv)``.  The step size is ``η = η0 / sqrt(t)`` with
    ``η0 = 0.9`` – a slightly larger constant than in the previous code.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    eta0 = 0.9

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and its arg‑max
        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        grad = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        grad[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent step in log‑space
        lr = eta0 / math.sqrt(it)
        log_a = np.log(a + eps) - lr * grad
        # stability: shift by max to avoid overflow
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted simulated annealing with incremental convolution updates
# ----------------------------------------------------------------------


def _incremental_sa(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    High‑frequency SA that moves mass from the index with the largest
    sub‑gradient to the index with the smallest one.  The convolution is
    updated incrementally (O(n·k)) so that millions of moves are cheap.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initial temperature – a modest fraction of the current objective
    T = max(0.02 * cur_obj, 1e-12)
    decay = 0.9999

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, 1e-9)

        # locate the current maximal shift and its sub‑gradient
        k_star = int(np.argmax(cur_conv))
        grad = np.zeros_like(cur)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        grad[idx_i] = 2.0 * cur[rev_idx]

        # indices to transfer mass between
        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        # nothing to move?
        if cur[i_high] <= 1e-15:
            continue

        # step size – a decreasing fraction of the source mass
        eps = min(cur[i_high] * 0.3, 0.005 / math.sqrt(it))
        if eps <= 0.0:
            continue

        # propose new allocation
        deltas = np.zeros(2, dtype=np.float64)
        idx = np.array([i_high, i_low], dtype=int)
        deltas[0] = -eps
        deltas[1] = +eps

        # incremental update of the convolution
        conv_new = cur_conv.copy()
        # contributions a*Δ + Δ*a
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur          # a is the *old* vector
            conv_new[2 * p] += delta * delta                # Δ*Δ term (self)

        # cross‑terms Δ_i * Δ_j for i≠j (appear twice)
        if len(idx) == 2:
            pos = idx[0] + idx[1]
            conv_new[pos] += 2.0 * deltas[0] * deltas[1]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / T):
            # accept
            cur[idx] = cur[idx] + deltas
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects a time budget and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # reuse the previous best if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Lengths to try – larger lengths give a finer discretisation but cost more
    lengths = [2000, 4000, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Phase 0 – quick evaluation of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – exponential‑gradient refinement
        # ------------------------------------------------------------------
        for s in seeds:
            if time.time() > deadline:
                break
            seq, val = _exp_gradient_refine(s, n, deadline,
                                            max_iters=600_000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑ExpGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – targeted simulated‑annealing
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _incremental_sa(global_best_seq, n, deadline,
                                       max_iters=2_000_000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Verify with the supplied ``evaluate_sequence`` (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
