def _local_search(start_seq, n, deadline, max_moves=200000):
    a = start_seq.copy()
    best = a.copy()
    best_val = _objective(a)
    rng = np.random.default_rng()
    while max_moves > 0 and time.time() < deadline:
        # compute convolution once
        conv = _conv_fft(a,a)
        max_val = conv.max()
        # find indices where conv near max
        # get one such shift
        candidates = np.where(conv >= max_val - 1e-9)[0]
        k = rng.choice(candidates)
        # choose i from overlapping indices
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        i = rng.integers(i_min, i_max + 1)
        j = k - i
        # random small delta transfer from i and j to random l,m
        delta = min(a[i], a[j]) * rng.random() * 0.02  # up to 2% of min
        if delta <= 0:
            max_moves -= 1
            continue
        # choose new indices to receive mass
        l = rng.integers(n)
        m = rng.integers(n)
        # ensure l,m not equal i,j? optionally allow
        new_seq = a.copy()
        new_seq[i] -= delta
        new_seq[j] -= delta
        new_seq[l] += delta/2
        new_seq[m] += delta/2
        # fix any negative due to roundoff
        if new_seq[i] < 0 or new_seq[j] < 0:
            max_moves -= 1
            continue
        new_val = _objective(new_seq)
        if new_val < best_val:
            best_val = new_val
            best = new_seq.copy()
            a = new_seq  # accept improvement
        else:
            # accept with small probability if improvement or small acceptance
            if rng.random() < 0.01:
                a = new_seq
        max_moves -= 1
    return best, best_val

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the grader:
        2·n·max(conv(seq,seq)) / (sum(seq)²)
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_increasing(n: int, r: float) -> np.ndarray:
    """Geometric growth from left to right (r>1)."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a[::-1]               # make it increasing
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_increasing(n: int, p: float) -> np.ndarray:
    """Increasing power law: a_i ∝ i^p (i starts at 0)."""
    idx = np.arange(n, dtype=np.float64)
    a = (idx + 1) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window, β ≥ 0."""
    if not SCIPY_AVAILABLE or beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    nminus1 = n - 1
    x = (2.0 * i / nminus1) - 1.0
    a = i0(beta * np.sqrt(1.0 - x * x)) / i0(beta)
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α∈(0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _dirichlet_seed(n: int, rng: np.random.Generator, alpha: float = 0.3) -> np.ndarray:
    """Very sparse Dirichlet‑type seed (small α -> many near‑zeros)."""
    a = rng.dirichlet([alpha] * n)
    return a

    seeds.append(_hann_window(n))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 10000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp (smooth‑max) surrogate.
    Returns (best_vector, best_objective_value).
    """
    v = np.log(seed + 1e-30)          # initialise logits
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    # hyper‑parameters (slightly tuned from the baseline)
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing schedule for the soft‑max surrogate
    beta = 1.5
    beta_mul = 1.006

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:4d} it={it:5d} best={cur_val:.6f}")

        # ---------- smooth‑max surrogate ----------
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k · (a⋆a)_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moment update
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 500000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror‑descent on the true non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient: average over the active constraints
        g = np.zeros_like(a)
        if len(active):
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stabilise numerics
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:4d} it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny amount of mass
    between two random positions and occasionally adds a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a small fraction of the source mass
        delta = rng.random() * 0.02 * cur[i]   # ≤ 2 % of source
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n:4d} it={it:6d} best={new_val:.6f}")

        # periodic global Gaussian perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n:4d} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

def _local_refine(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_moves: int = 200000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing that repeatedly transfers a tiny amount of mass
    between a pair of indices that directly participate in the currently
    active (max‑valued) convolution entry.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best = a.copy()
    best_val = _objective(a)

    while max_moves > 0 and time.time() < deadline:
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # pick one of the active shifts uniformly
        active_shifts = np.where(conv >= max_val - 1e-12)[0]
        k = rng.choice(active_shifts)

        # indices (i, j) that contribute to conv[k] (i+j = k)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i = rng.integers(i_low, i_high + 1)
        j = k - i

        # move a tiny proportion of the smaller of the two masses
        delta = min(a[i], a[j]) * rng.random() * 0.01
        if delta <= 0.0:
            max_moves -= 1
            continue

        # destination indices are chosen away from the current shift
        dest_i = rng.integers(0, n)
        dest_j = rng.integers(0, n)
        new_seq = a.copy()
        new_seq[i] -= delta
        new_seq[j] -= delta
        # split the transferred mass equally
        new_seq[dest_i] += delta * 0.5
        new_seq[dest_j] += delta * 0.5

        # safeguard against negativity caused by rounding
        if new_seq[i] < 0 or new_seq[j] < 0:
            max_moves -= 1
            continue

        new_val = _objective(new_seq)
        if new_val < best_val:
            best_val = new_val
            best = new_seq.copy()
            a = new_seq  # accept improvement immediately
            print(f"[Local] n={n:4d} move={max_moves:6d} best={new_val:.6f}")
        else:
            # occasional downhill move to escape shallow plateaus
            if rng.random() < 0.001:
                a = new_seq

        max_moves -= 1

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget and
    returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a safety margin

    # possible previous result (allows chaining across runs)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Length(s) to explore – we keep the proven 4000 for now.
    # ------------------------------------------------------------------
    lengths = [4000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1) Build a very diverse seed set
        # --------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # --------------------------------------------------------------
        # 2) Quick evaluation – keep the best few raw seeds
        # --------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_k = 15
        top_seeds = [s for s, _ in seed_vals[:top_k]]

        # record the best raw seed
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # --------------------------------------------------------------
        # Phase A – Adam optimisation (smooth‑max) on each top seed
        # --------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            # limit the number of iterations adaptively
            remaining = deadline - time.time()
            max_iters = min(12000, int(remaining / 0.001))  # crude budget
            seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # --------------------------------------------------------------
        # Phase B – Exact sub‑gradient mirror‑descent (exponentiated‑grad)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            remaining = deadline - time.time()
            if remaining > 10:
                max_iters = int(2e5 + 1e5 * (remaining / (budget_s - 2)))
                seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                               max_iters=max_iters)
                if seq is not None and val < global_best_val:
                    global_best_val = val
                    global_best_seq = seq.copy()
                    print(f"[overall SubGrad] n={n} best={val:.6f}")

        # --------------------------------------------------------------
        # Phase C – Simulated‑annealing global walk
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # --------------------------------------------------------------
        # Phase D – Targeted local refinement (cheap mass‑transfer)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            remaining = deadline - time.time()
            if remaining > 5:
                max_moves = int(1e5 * (remaining / (budget_s - 2)))
                seq, val = _local_refine(global_best_seq, n, deadline,
                                         max_moves=max_moves)
                if seq is not None and val < global_best_val:
                    global_best_val = val
                    global_best_seq = seq.copy()
                    print(f"[overall Local] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    # store for potential subsequent runs
    globals()["height_sequence_1"] = global_best_seq.copy()
    return global_best_seq.tolist()
