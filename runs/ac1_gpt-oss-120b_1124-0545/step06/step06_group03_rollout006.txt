def _subgradient_refine_random(start_seq, n, deadline, max_iters=300000, tol=1e-6):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a, a)
        M = conv.max()
        cur_val = 2.0*n*M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGradRand] n={n} it={it} best={cur_val:.6f}")
        # mask for near-peak indices
        # Choose tolerance relative
        threshold = M - max(tol, M*0.001)  # 0.1% tolerance
        idxs = np.where(conv >= threshold)[0]
        if len(idxs)==0:
            k_star = int(np.argmax(conv))
        else:
            k_star = int(rng.choice(idxs))
        # compute subgradient for this k_star
        g = np.zeros_like(a)
        i_start = max(0, k_star-(n-1))
        i_end = min(n-1, k_star)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a+eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s==0:
            a = _uniform(n)
        else:
            a/=s

    return best_seq,best_val

"""
Propose a near‑optimal non‑negative step function.

The algorithm works in three stages for a few candidate lengths (4000 and 8000
elements).  For each length it

1. Generates a diverse pool of *seed* vectors (uniform, sine‑window,
   beta‑shapes, geometric, power‑law and many random parametric families).
   All seeds are normalised to sum = 1.

2. Keeps the best few seeds (by the smooth objective) and refines them
   with an Adam optimiser that minimises a log‑sum‑exp approximation of
   the peak of the aperiodic auto‑convolution.

3. Starting from the best Adam result a long mirror‑descent (exponentiated‑
   gradient) is performed on the *exact* max‑convolution.  The active
   index is chosen randomly among all near‑maximal entries, which helps
   escape shallow local minima.

4. A high‑frequency simulated‑annealing pass with incremental convolution
   updates further drives the peak down.  The temperature decays slowly,
   and moves that re‑allocate mass among 2‑5 positions are accepted with
   the usual Metropolis probability.

The routine respects a 1000 s wall‑time budget, falls back to a uniform
distribution if nothing is found, and finally returns the best sequence
as a plain `list[float]`.  The returned sequence can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using a real FFT."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_power(n: int, p: float) -> np.ndarray:
    """sin‑window raised to a power."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    return a / a.sum()


def _mix_beta(n: int, a1, b1, a2, b2, w: float) -> np.ndarray:
    """Convex combination of two beta‑shapes."""
    a = w * _beta_shape(n, a1, b1) + (1.0 - w) * _beta_shape(n, a2, b2)
    return a / a.sum()


def _block_constant(n: int, blocks: int, rng: np.random.Generator) -> np.ndarray:
    """Piece‑wise constant – `blocks` contiguous sections."""
    cuts = np.sort(rng.choice(n - 1, size=blocks - 1, replace=False) + 1)
    idx = np.split(np.arange(n), cuts)
    values = rng.dirichlet(np.full(blocks, 1.0))
    a = np.empty(n, dtype=np.float64)
    for seg, v in zip(idx, values):
        a[seg] = v
    return a / a.sum()


def _uprate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised vector to a larger length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, seq)


def _make_base_seeds(n: int,
                     previous: np.ndarray | None,
                     rng: np.random.Generator) -> list[np.ndarray]:
    """Deterministic families + optional uplift of a previous solution."""
    seeds: list[np.ndarray] = [
        _uniform(n),
        _sine_window(n),
        _geom(n, 0.5),
        _geom(n, 0.8),
        _power(n, 0.5),
        _power(n, 1.5),
        _beta_shape(n, 0.5, 0.5),
        _beta_shape(n, 1.0, 2.0),
        _beta_shape(n, 2.0, 1.0),
        _sine_power(n, 1.0),
        _sine_power(n, 2.0),
    ]

    # use a previous construction if it exists and can be interpolated
    if previous is not None:
        if previous.shape == (n,):
            seeds.append(previous / previous.sum())
        else:
            seeds.append(_uprate(previous, n))

    return seeds


def _make_random_parametric_seeds(n: int,
                                  rng: np.random.Generator,
                                  count: int = 40) -> list[np.ndarray]:
    """A handful of random shapes drawn from a variety of parametrised families."""
    seeds: list[np.ndarray] = []
    for _ in range(count):
        typ = rng.integers(0, 6)
        if typ == 0:          # beta shape
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            seeds.append(_beta_shape(n, a, b))
        elif typ == 1:        # geometric
            r = rng.uniform(0.2, 0.95)
            seeds.append(_geom(n, r))
        elif typ == 2:        # power‑law
            p = rng.uniform(0.0, 2.5)
            seeds.append(_power(n, p))
        elif typ == 3:        # sine raised to a power
            p = rng.uniform(0.5, 4.0)
            seeds.append(_sine_power(n, p))
        elif typ == 4:        # mix two betas
            a1 = rng.uniform(0.2, 5.0)
            b1 = rng.uniform(0.2, 5.0)
            a2 = rng.uniform(0.2, 5.0)
            b2 = rng.uniform(0.2, 5.0)
            w = rng.random()
            seeds.append(_mix_beta(n, a1, b1, a2, b2, w))
        else:                 # block‑constant
            blocks = rng.integers(2, 9)
            seeds.append(_block_constant(n, blocks, rng))
    return seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 15000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑sum‑exp (smooth max) surrogate.
    Uses a softmax parametrisation (log‑probabilities) to keep the vector
    on the probability simplex.
    """
    eps = 1e-30
    v = np.log(seed + eps)                     # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness
    beta_mul = 1.02          # geometric increase each iteration

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t. a (conv = a*a) -----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)     # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # ----- convert to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping to keep Adam stable
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine_random(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               max_iters: int = 300000,
                               rel_tol: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    At each iteration the active index is chosen uniformly from all indices
    whose convolution value lies within `rel_tol` of the current maximum.
    This stochastic choice helps to escape plateaus.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = np.max(conv)
        cur_val = 2.0 * n * M

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # choose a (near‑)maximal index at random
        thresh = M - max(rel_tol * M, 1e-12)
        cand = np.where(conv >= thresh)[0]
        k_star = int(rng.choice(cand)) if cand.size else int(np.argmax(conv))

        # construct sub‑gradient for that index
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror) step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)            # numeric stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             init_temp_factor: float = 0.02,
                             min_temp: float = 1e-9,
                             max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with cheap O(n·k) convolution updates.
    A move picks 2‑5 positions, redistributes their total mass via a Dirichlet
    draw and updates the full convolution incrementally.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * decay, min_temp)

        # pick a random subset of indices (size 2‑4)
        k = rng.integers(2, 5)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # new allocation for the chosen positions
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        deltas = new_alloc - cur[idx]
        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        # a*Δ + Δ*a term (linear)
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur        # cur is the *old* vector
            conv_new[2 * p] += delta * delta               # Δ*Δ for the diagonal

        # cross‑terms Δ_i * Δ_j, i≠j (appear twice in Δ*Δ)
        for i in range(k):
            for j in range(i + 1, k):
                pos = idx[i] + idx[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the 1000 s time budget.

    Returns:
        list[float] – the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Previous construction (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – larger length generally yields a lower bound.
    # ------------------------------------------------------------------
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # ----- assemble a generous but bounded seed set -----------------
        seeds = _make_base_seeds(n, previous, rng)
        seeds.extend(_make_random_parametric_seeds(n, rng, count=30))

        # ----- quick scan – keep only the top‑k raw seeds -------------
        vals = [_objective(s) for s in seeds]
        top_k = 5
        top_indices = np.argsort(vals)[:top_k]
        seed_pool = [seeds[i] for i in top_indices]

        # ----- Phase 1 – Adam on a smooth max -------------------------
        best_seq_phase1 = None
        best_val_phase1 = float("inf")
        for s in seed_pool:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline,
                                      max_iters=18000 if n == 8000 else 15000)
            if seq is not None and val < best_val_phase1:
                best_val_phase1 = val
                best_seq_phase1 = seq.copy()
                print(f"[Phase‑Adam] n={n:5d} new phase‑best={val:.6f}")

        # If Adam never produced a candidate, fall back to the best raw seed
        if best_seq_phase1 is None:
            best_seq_phase1 = seed_pool[0].copy()
            best_val_phase1 = _objective(best_seq_phase1)

        # ----- Phase 2 – exact sub‑gradient (randomised active set) ----
        seq2, val2 = _subgradient_refine_random(best_seq_phase1, n, deadline,
                                                max_iters=250000 if n == 8000 else 300000)
        if val2 < best_val_phase1:
            best_seq_phase1 = seq2.copy()
            best_val_phase1 = val2

        # ----- Phase 3 – high‑frequency simulated annealing -----------
        seq3, val3 = _sa_incremental_optimize(best_seq_phase1, n, deadline,
                                              max_iters=800000 if n == 4000 else 1_200_000)
        # polish once more with a short sub‑gradient pass after SA
        seq4, val4 = _subgradient_refine_random(seq3, n, deadline,
                                                max_iters=100000)

        # choose the best among the three refinements
        candidates = [(best_seq_phase1, best_val_phase1),
                      (seq2, val2),
                      (seq3, val3),
                      (seq4, val4)]
        for seq, val in candidates:
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑best] length={n} new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case we found nothing
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Verify that the produced sequence is accepted by the provided evaluator
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
