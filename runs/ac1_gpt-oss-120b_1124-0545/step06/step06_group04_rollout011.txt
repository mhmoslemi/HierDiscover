def _subgradient_refine(start_seq, n, deadline, max_iters=20000):
   a = start_seq.copy()
   best_seq = a.copy()
   best_val = _objective(a)
   it = 0
   while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        # gradient of active term
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it} best={cur_val:.6f}")

"""
Propose a step‑function (non‑negative vector) that minimises
    2·n·max(conv(f,f)) / (sum(f)^2)

The algorithm combines:
  * a rich pool of parametric seeds,
  * L‑BFGS optimisation of a smooth “log‑sum‑exp” surrogate of the
    max‑convolution (the variables are logits → softmax, i.e. a simplex),
  * a cheap exponentiated‑gradient sub‑gradient refinement on the true
    max‑convolution,
  * a lightweight simulated‑annealing fine‑tuner,
  * optional symmetrisation and up‑scaling to larger lengths.

The search respects the 1000 s budget and caches the best sequence in the
global name ``height_sequence_1`` for reuse across calls.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real inputs)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, beta: float, n: int):
    """
    Returns (loss, grad) where loss = (1/β)·log Σ exp(β·conv)
    and grad is the gradient w.r.t. the logits v (so that a = softmax(v)).
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)

    shift = conv.max()
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    loss = shift + np.log(sum_exp) / beta

    w = exp_beta / sum_exp                     # soft max weights
    rev_a = a[::-1]
    # gradient of weighted sum of conv entries w.r.t. a:
    full = _conv_fft(rev_a, w)                # length = n + (2n-1) -1 = 3n-2
    grad_a = 2.0 * full[n - 1:n - 1 + n]      # keep the n relevant components

    # back‑prop through soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return loss, grad_v


# ----------------------------------------------------------------------
# Seed families (all already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sin2_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _hanning(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)
    a[a < 0] = 0.0
    return a / a.sum()


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length while preserving total mass."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Construct a diverse pool of normalised start vectors.
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_sin2_window(n))
    seeds.append(_hanning(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_tent(n))

    # beta families
    for a in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        seeds.append(_beta_seed(n, a, a))
        seeds.append(_beta_seed(n, a, a * 1.5))
        seeds.append(_beta_seed(n, a * 1.5, a))

    # Dirichlet draws (different alphas)
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrored copies (helps if optimum is symmetric)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # up‑scaled previous best, if available
    if previous is not None:
        try:
            up = _upscale(previous, n)
            seeds.append(up)
        except Exception:
            pass

    # guarantee at least one seed
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Optimisation helpers
# ----------------------------------------------------------------------


def _optimize_seed(seed: np.ndarray, n: int, deadline: float):
    """
    L‑BFGS optimisation of the smooth surrogate for a single seed.
    Returns the best exact objective found and the corresponding sequence.
    """
    v = np.log(seed + 1e-30)  # logits → simplex via soft‑max
    best_seq = None
    best_val = float('inf')
    # Progressive smoothing schedule
    betas = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]

    for beta in betas:
        if time.time() > deadline:
            break

        # L‑BFGS on the smooth loss;  the callable returns (loss, grad)
        def fun(vv):
            return _smooth_obj_and_grad(vv, beta, n)

        try:
            res = scipy.optimize.minimize(
                fun,
                v,
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': 30, 'disp': False},
            )
        except Exception:   # a very rare numerical crash – ignore
            res = None

        if res is not None and res.success:
            v = res.x  # continue optimisation from the new point

        # evaluate the exact objective
        seq = _softmax(v)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[L‑BFGS] n={n} β={beta:.1f} → {val:.6f}")

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 20000):
    """
    Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient
    of max(conv).  Runs up to ``max_iters`` or until the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None):
    """
    Very cheap Metropolis–Hastings local search that swaps tiny fragments.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # modest starting temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # move ≤2.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            if s == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= s
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  Returns the best sequence (list of floats) found
    before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # retrieve previously discovered construction, if any
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # lengths – start modest and grow if time permits
    lengths = [4000, 8000, 12000, 20000, 30000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # Build a diverse seed pool
        # --------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Quick evaluation of raw seeds – keep the top few
        # ------------------------------------------------------------------
        seed_objs = [(s, _objective(s)) for s in seeds]
        seed_objs.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_objs[:min(6, len(seed_objs))]]

        # keep best raw seed globally
        if seed_objs[0][1] < global_best_val:
            global_best_val = seed_objs[0][1]
            global_best_seq = seed_objs[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – L‑BFGS optimisation from each top seed
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _optimize_seed(seed_vec, n, deadline)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall L‑BFGS] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – sub‑gradient refinement (lightweight)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n and time.time() < deadline:
            max_iters = 20000 if n <= 12000 else 8000
            seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=max_iters)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n and time.time() < deadline:
            seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Optional symmetrisation (often reduces the peak)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[symmetrisation] n={n} improved to {sym_val:.6f}")

        # propagate the current best forward (for up‑scaling in the next length)
        previous = global_best_seq

    # ----------------------------------------------------------------------
    # Fallback – should never be needed but kept for safety
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # cache for the next call
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
