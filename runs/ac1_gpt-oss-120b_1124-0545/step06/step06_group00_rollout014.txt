"""
propose_candidate.py

A time‑bounded optimiser for the step‑function evaluation
    2·n·max(conv(a,a)) / (sum(a)²)

The search consists of three stages:

1️⃣  Seed generation – a large and diverse set of windows (uniform,
    geometric, power‑law, Beta‑pdf, Gaussian, Kaiser, Chebyshev,
    DPSS, random Dirichlet/soft‑max).  All seeds are normalised to sum = 1.

2️⃣  Smooth‑max Adam optimisation – each of the top‑k seeds is refined
    using Adam on a log‑sum‑exp surrogate of the max‑convolution.
    The surrogate temperature grows slowly so the optimiser sees a
    gradually sharpening objective.

3️⃣  Exact mirror‑descent + long simulated annealing – starting from the
    best Adam result we perform exponentiated‑gradient (mirror‑descent)
    on the true non‑smooth objective, followed by a prolonged annealing
    walk that moves tiny amounts of mass between random positions.
    All updates stay on the probability simplex.

The routine respects a user‑supplied time‑budget (default 1000 s) and
returns the best sequence found (plain ``list`` of floats).  No I/O,
no closures – everything is top‑level to satisfy the competition rules.
"""

import time
import math
import numpy as np

# Optional SciPy imports (used for a few classic windows)

try:

    from scipy.special import i0

except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# Low‑level convolution utilities (FFT based, real‑valued)

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation used by the optimiser:
    2·n·max(conv(seq,seq)) / (sum(seq)²)
    The input must be non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

# Simple families – all return a normalised probability vector
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i (r∈(0,1))."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric‑power a_i ∝ r**i / (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on (0,1), discretised on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _cheb_window(n: int, atten: float) -> np.ndarray:
    """Dolph‑Chebyshev window (if SciPy is available)."""
    if not SCIPY_AVAILABLE:
        return _uniform(n)
    a = signal.chebwin(n, at=atten)
    a = np.clip(a, 0.0, None)  # Chebyshev windows are non‑negative anyway
    return a / a.sum()


def _dpss_window(n: int, NW: float) -> np.ndarray:
    """First DPSS (Slepian) vector – maximises energy concentration."""
    if not SCIPY_AVAILABLE:
        return _uniform(n)
    a = signal.windows.dpss(n, NW, Kmax=1, return_ratios=False)[0]
    a = np.abs(a)  # DPSS may contain tiny negative rounding artefacts
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _logistic(n: int, k: float, t0: float) -> np.ndarray:
    """Logistic (sigmoid) decay a_i ∝ 1/(1+exp(k*(i-t0)))"""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(k * (i - t0)))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse seed pool (all normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic deterministic families
    seeds.append(_uniform(n))

    # geometric families with a fine sweep of r
    for r in np.linspace(0.998, 0.99999, 12):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # combined geometric‑power families
    for r in np.linspace(0.998, 0.9999, 6):
        for p in np.linspace(0.0, 2.5, 7):
            seeds.append(_geom_power(n, r, p))

    # beta‑pdf windows – denser sweep
    for alpha in np.linspace(0.55, 0.98, 22):
        seeds.append(_beta_window(n, alpha))

    # Gaussian windows – finer sigma sweep
    for sigma_factor in np.linspace(0.07, 0.30, 12):
        seeds.append(_gaussian(n, sigma_factor))

    # classical symmetric windows
    seeds.append(_hann_window(n))
    seeds.append(_tukey_window(n, 0.4))
    seeds.append(_tukey_window(n, 0.6))

    # Chebyshev windows with varied attenuation (if available)
    if SCIPY_AVAILABLE:
        for att in [40, 60, 80, 100]:
            seeds.append(_cheb_window(n, att))

    # DPSS windows with several time–bandwidth products
    if SCIPY_AVAILABLE:
        for NW in [2.0, 3.0, 4.0, 5.0]:
            seeds.append(_dpss_window(n, NW))

    # Logistic family (asymmetric, may beat symmetric ones)
    for k in [0.001, 0.005, 0.02]:
        for t0 in [0.25 * n, 0.5 * n, 0.75 * n]:
            seeds.append(_logistic(n, k, t0))

    # Random soft‑max vectors and Dirichlet‑like draws
    for _ in range(20):
        seeds.append(_random_softmax(n, rng))

    # previous result (if any) – a cheap way to continue optimisation
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final safety: clip negatives, renormalise and discard zero vectors
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned

# Euclidean projection onto the probability simplex (used by anneal)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Adam optimisation on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp (smooth‑max) approximation of max(conv(a,a)).
    Returns the best vector and its true objective.
    """
    v = np.log(seed + 1e-30)        # logits on the simplex

    # Adam moments
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    # hyper‑parameters
    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing schedule – start low, increase gently
    beta = 1.0
    beta_mul = 1.003

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # light logging – printed only occasionally
            if it % 200 == 0:
                print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights (soft‑max over the convolution entries) -----
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k · conv_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moment update
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val

# Exact sub‑gradient mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 600000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact objective.
    Returns the best vector encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient averaged over active indices
        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient update
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numerics safe
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] n={n} it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that transfers a tiny mass between
    two indices and occasionally adds a global Gaussian perturbation.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature at a few percent of the current value
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9997

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # --- elementary mass transfer between two random positions ---
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # transfer up to 5 % of the source mass (more aggressive than before)
        delta = rng.random() * 0.05 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                if it % 5000 == 0:
                    print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        # --- occasional global Gaussian jump (helps escape plateaus) ---
        if it % 2000 == 0:
            noise = rng.normal(scale=0.0015, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42,

    start_time = time.time()

    # Length to optimise – 4000 is a good trade‑off between quality and time

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    # store the final result for a possible subsequent run
    globals()["height_sequence_1"] = global_best_seq.copy()
    # return as plain Python list
    return global_best_seq.tolist()
