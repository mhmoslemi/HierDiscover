def _apply_deltas(conv, cur, idx, deltas):
    new_conv = conv.copy()
    for p, delta in zip(idx, deltas):
        if delta == 0: continue
        new_conv[p:p+len(cur)] += 2.0 * delta * cur
        new_conv[2 * p] += delta * delta
    # cross terms
    k = len(idx)
    for i in range(k):
        for j in range(i+1, k):
            pos = idx[i] + idx[j]
            new_conv[pos] += 2.0 * deltas[i] * deltas[j]
    return new_conv

def _guided_sa_optimize(start_seq, n, deadline,
                        init_temp_factor=0.02, min_temp=1e-9,
                        max_iters=500000, target_prob=0.3, epsilon=1e-3):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)
    best_seq = cur.copy()
    best_obj = cur_obj
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995
    
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        T = max(T*decay, min_temp)
        if rng.random() < target_prob:
            # targeted move: shift epsilon mass from high contributing indices
            conv = cur_conv
            kstar = int(np.argmax(conv))
            i_start = max(0, kstar - (n - 1))
            i_end = min(n-1, kstar)
            indices = np.arange(i_start, i_end+1)
            # compute contributions a[i]*a[kstar-i]
            contrib = cur[indices] * cur[kstar - indices]
            total_contrib = contrib.sum()
            if total_contrib <= 0:
                continue
            # pick source index weighted by contrib
            src = rng.choice(indices, p=contrib/total_contrib)
            src2 = kstar - src
            # if src and src2 same, treat singly (self-pair)
            srcs = [src] if src == src2 else [src, src2]
            # Determine mass to take: epsilon fraction of source's mass
            # We'll take min(epsilon, cur[src]) each
            deltas = []
            idx = []
            for s in srcs:
                take = min(epsilon, cur[s]/2)  # split epsilon between both if two
                if take > 0:
                    idx.append(s)
                    deltas.append(-take)
                else:
                    # can't take, fallback random move
                    idx = []
                    deltas = []
                    break
            # choose destination indices: random distinct others
            if not idx:
                # fallback to random move
                pass
            else:
                # allocate mass taken to random other indices (could be 1 or 2)
                total_taken = -sum(deltas)
                # choose dest count 1-3
                k_dest = rng.integers(1, 4)
                dest_indices = rng.choice(n, size=k_dest, replace=False)
                # ensure not in source indices
                while np.isin(dest_indices, idx).any():
                    dest_indices = rng.choice(n, size=k_dest, replace=False)
                # allocate via Dirichlet
                alloc = rng.dirichlet(np.full(k_dest, 1.0)) * total_taken
                for d, val in zip(dest_indices, alloc):
                    idx.append(d)
                    deltas.append(val)
                # apply deltas
                conv_new = _apply_deltas(cur_conv, cur, idx, deltas)
                new_obj = 2.0 * n * np.max(conv_new)
                delta_obj = new_obj - cur_obj
                if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / T):
                    # accept
                    cur[idx] += deltas
                    cur_conv = conv_new
                    cur_obj = new_obj
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_seq = cur.copy()
                        print(f"[SA-target] n={n} it={it} best={new_obj:.6f}")
                continue  # next iteration
        # random move (as earlier)
        k = rng.integers(2,5)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        deltas = new_alloc - cur[idx]
        if np.allclose(deltas, 0):
            continue
        conv_new = _apply_deltas(cur_conv, cur, idx, deltas)
        new_obj = 2.0 * n * np.max(conv_new)
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n} it={it} best={new_obj:.6f}")
    return best_seq, best_obj

"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑ a)²

The algorithm combines several ideas beyond the baseline:

1. **Richer seeds** – uniform, half‑sine, geometric, power‑law, beta‑like,
   random Dirichlet draws and a few sparse patterns.

2. **Two‑phase smooth optimisation** – a long Adam run on a
   log‑sum‑exp smooth approximation of the max (more iterations and a
   more aggressive smoothness schedule).

3. **Exact sub‑gradient mirror descent** – exponentiated‑gradient steps
   on the true max‑operator.

4. **Guided simulated‑annealing** – most moves are cheap random
   Dirichlet reallocations, but a fraction of the steps explicitly take
   mass away from the indices that currently form the maximal
   convolution entry (the active set of the max).  The convolution is
   updated incrementally, so each move is O(k²) with k≤8.

5. **Final template mixing** – the best sequence found is line‑searched
   against a half‑sine and uniform template, which often yields a small
   extra drop.

All steps respect the global time budget (default 1000 s) and keep the
sequence on the simplex (∑a = 1).  The returned list can be fed directly
to ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax that yields a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming ∑a == 1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Positive half‑sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float,
                   rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries, drawn from a Dirichlet (conc < 1 → sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(n: int,
                previous: Optional[np.ndarray],
                rng: np.random.Generator) -> List[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: List[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # beta‑like shapes
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse a previous solution if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 30000) -> Tuple[Optional[np.ndarray], float]:
    """Adam on a log‑sum‑exp smooth max of the convolution."""
    eps = 1e-30
    v = np.log(seed + eps)                     # parameters in log‑space

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # correct shift
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t. a (conv = a*a) -----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # ----- convert to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200000) -> Tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror‑descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _apply_deltas(old_conv: np.ndarray,
                  cur_vec: np.ndarray,
                  idx: List[int],
                  deltas: List[float]) -> np.ndarray:
    """
    Incrementally update a convolution when a few entries of the vector are
    changed by the given deltas.  Uses the exact formula:
        new = old + Σ 2·Δ·a (shifted) + Σ Δ_i·Δ_j (cross‑terms).
    """
    n = len(cur_vec)
    conv = old_conv.copy()
    for p, d in zip(idx, deltas):
        if d == 0.0:
            continue
        conv[p:p + n] += 2.0 * d * cur_vec
        conv[2 * p] += d * d

    # cross‑terms Δ_i·Δ_j for i≠j (appear twice in Δ*Δ)
    k = len(idx)
    for i in range(k):
        for j in range(i + 1, k):
            pos = idx[i] + idx[j]
            conv[pos] += 2.0 * deltas[i] * deltas[j]

    return conv


def _guided_sa_optimize(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        init_temp_factor: float = 0.02,
                        min_temp: float = 1e-9,
                        max_iters: int = 500000,
                        target_prob: float = 0.3,
                        epsilon: float = 0.001) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing with two kinds of moves:
    * random Dirichlet reallocations (as before);
    * *guided* moves that take a small amount of mass from the active set
      (the indices that currently build the maximal convolution entry) and
      redistribute it elsewhere.
    The convolution is updated incrementally, so each iteration is cheap.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * decay, min_temp)

        # ------------------------------------------------------------------
        # Guided move (probability target_prob)
        # ------------------------------------------------------------------
        if rng.random() < target_prob:
            conv = cur_conv
            k_star = int(np.argmax(conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            idx_set = np.arange(i_start, i_end + 1, dtype=int)
            # contributions a[i] * a[k_star-i] for the max entry
            contrib = cur[idx_set] * cur[k_star - idx_set]
            total_c = contrib.sum()
            if total_c > 0:
                # pick a source index from the active set,
                # weighted by its contribution to the max entry.
                src = rng.choice(idx_set, p=contrib / total_c)
                src2 = k_star - src
                srcs = [src] if src == src2 else [src, src2]

                # total mass we are willing to move away from the pair
                pair_mass = cur[srcs].sum()
                take = min(epsilon, pair_mass)
                if take > 0:
                    # proportional removal from the two source indices
                    rem = take * (cur[srcs] / pair_mass)
                    deltas_src = -rem
                    idx_move = srcs.copy()
                    deltas_move = deltas_src.tolist()

                    # destination indices (1‑4 random positions, avoiding sources)
                    k_dest = rng.integers(1, 5)
                    dest = rng.choice(n, size=k_dest, replace=False)
                    # ensure no overlap with source indices
                    while np.isin(dest, idx_move).any():
                        dest = rng.choice(n, size=k_dest, replace=False)
                    alloc = rng.dirichlet(np.full(k_dest, 1.0)) * take
                    idx_move.extend(dest.tolist())
                    deltas_move.extend(alloc.tolist())

                    # incremental convolution update
                    conv_new = _apply_deltas(cur_conv, cur, idx_move, deltas_move)
                    new_obj = 2.0 * n * np.max(conv_new)
                    delta_obj = new_obj - cur_obj
                    if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / T):
                        # accept
                        cur[idx_move] = cur[idx_move] + np.array(deltas_move)
                        cur_conv = conv_new
                        cur_obj = new_obj
                        if new_obj < best_obj:
                            best_obj = new_obj
                            best_seq = cur.copy()
                            print(f"[SA‑guided] n={n:5d} it={it:7d} best={new_obj:.6f}")
                        continue   # go to next iteration

        # ------------------------------------------------------------------
        # Random Dirichlet move (fallback)
        # ------------------------------------------------------------------
        k = rng.integers(2, 5)          # 2‑4 positions
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        deltas = (new_alloc - cur[idx]).tolist()
        if all(abs(d) < 1e-12 for d in deltas):
            continue

        conv_new = _apply_deltas(cur_conv, cur, idx.tolist(), deltas)
        new_obj = 2.0 * n * np.max(conv_new)
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj


def _mix_with_templates(best_seq: np.ndarray,
                        n: int,
                        deadline: float) -> Tuple[np.ndarray, float]:
    """
    Line‑search mixing the current best with a few simple templates
    (uniform and half‑sine).  The mixing coefficient α∈[0,1] is tried on a
    uniform grid; the best mixed vector (re‑normalised) is returned.
    """
    templates = [_uniform(n), _sine_window(n)]
    cur_best = best_seq.copy()
    cur_val = _objective(cur_best)

    for tmpl in templates:
        for alpha in np.linspace(0.0, 1.0, 21):
            mix = alpha * cur_best + (1.0 - alpha) * tmpl
            mix /= mix.sum()
            val = _objective(mix)
            if val < cur_val:
                cur_val = val
                cur_best = mix
                print(f"[mix] n={n:5d} alpha={alpha:.2f} tmpl={templates.index(tmpl)} val={val:.6f}")

    return cur_best, cur_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` (or equivalently the analytical objective).
    The routine respects the time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Leave a small safety margin for the final evaluation call.
    deadline = start_time + budget_s - 2.0

    # Retrieve any existing solution that might be stored globally.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # A modest set of lengths – larger lengths give finer discretisation but
    # also cost more time.  We will stop early if the deadline is near.
    lengths = [2000, 3000, 4000, 6000, 8000]

    global_best_seq: Optional[np.ndarray] = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Quick evaluation of raw seeds and keep the top few
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        # keep the best 8 distinct seeds for further processing
        top_seeds = [sv[0] for sv in seed_vals[:8]]

        # Update global best from raw seeds
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n:5d} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation (smooth max) on the top seeds
        # ------------------------------------------------------------------
        for s in top_seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=25000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient mirror‑descent
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=200000)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Guided simulated‑annealing
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _guided_sa_optimize(global_best_seq, n, deadline,
                                           max_iters=800000,
                                           target_prob=0.35,
                                           epsilon=0.0015)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – Template mixing (cheap) – improves the final constant a bit
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _mix_with_templates(global_best_seq, n, deadline)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑mix] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a valid return.
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # ----------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence (optional)
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
