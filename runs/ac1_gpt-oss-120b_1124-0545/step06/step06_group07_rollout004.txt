def _greedy_flatten(a, n, deadline, max_iters=50000):
    best_val = 2*n*np.max(_conv_fft(a,a))
    best_a = a.copy()
    for it in range(max_iters):
        if time.time() > deadline: break
        conv = _conv_fft(a,a)
        max_idx = int(np.argmax(conv))
        # compute gradient only for active max idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idxs = np.arange(i_start, i_end + 1)
        rev_idxs = max_idx - idxs
        g[idxs] = 2.0 * a[rev_idxs]
        # find i (donor) with max g, j (receiver) with min g (excluding zeros maybe)
        donor = idxs[np.argmax(g[idxs])]
        receiver = idxs[np.argmin(g[idxs])]
        if donor == receiver: break
        # transfer fraction of donor's mass
        delta = min(a[donor] * 0.3, a[donor]) # maybe up to 30% of donor mass
        if delta <= 0: break
        a[donor] -= delta
        a[receiver] += delta
        # renormalize sum to 1
        s = a.sum()
        a = a / s
        val = 2*n*np.max(_conv_fft(a,a))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Greedy] it={it} best={val}")
        # maybe break if improvement small
    return best_a, best_val
```

"""
Proposes a non‑negative step‑function that minimises ``evaluate_sequence``.
The search combines a rich library of analytic windows (including DPSS,
Kaiser, Tukey, Beta, Gaussian, etc.), a fast evaluation of the objective,
and several deterministic refinements:

1.  Exhaustive evaluation of all analytic seeds (≈200 candidates) and keep
    the best few.
2.  For the top seeds run a projected‑subgradient descent (mirror‑descent)
    that directly works on the true max‑convolution.
3.  Apply a simple greedy “mass‑transfer’’ step that repeatedly moves mass
    from indices that contribute most to the current peak towards those
    that contribute least.
4.  Finish with a light stochastic hill‑climb to polish the solution.

All phases respect the global 1000 s budget and periodically check the
deadline.  The returned sequence is a plain list of floats (sum≈1, all ≥0).
"""

import time
import numpy as np
from scipy.signal import windows as win

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian window."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse collection of ℓ₁‑normalised start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Beta grid – denser than the reference
    for a in np.linspace(0.2, 2.0, 9):
        for b in np.linspace(0.2, 2.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian family
    for frac in np.linspace(0.05, 0.4, 8):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _objective(a: np.ndarray) -> float:
    """Exact objective for a normalised (ℓ₁‑sum = 1) sequence."""
    conv = _conv_fft(a, a)
    return float(2.0 * len(a) * np.max(conv))


# ----------------------------------------------------------------------
# Projected sub‑gradient refinement (mirror descent style)
# ----------------------------------------------------------------------


def _proj_subgradient_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv) with ℓ₁‑simplex projection.
    """
    a = start_seq.copy()
    a = _project_simplex(a, 1.0)

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient only for the active maximum entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size 1/√it
        lr = 1.0 / np.sqrt(it)
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, 1.0)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer refinement (new)
# ----------------------------------------------------------------------


def _greedy_flatten(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 50000,
                    step_factor: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy algorithm that repeatedly moves mass from the
    “most responsible’’ index (largest gradient) to the ``least responsible''
    one for the current peak of the auto‑convolution.
    """
    a = start_seq.copy()
    a = _project_simplex(a, 1.0)

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # compute the sub‑gradient for this peak
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        donor = idx_i[np.argmax(g[idx_i])]
        receiver = idx_i[np.argmin(g[idx_i])]

        if donor == receiver:
            break

        # move a fraction of the donor's mass towards the receiver
        delta = min(a[donor] * step_factor, a[donor])
        if delta <= 0.0:
            break

        a[donor] -= delta
        a[receiver] += delta
        # (sum stays equal to 1)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Greedy] it={it:6d} best={cur_val:.6f}")

        # optional cooling of step size
        if it % 5000 == 0:
            step_factor = max(step_factor * 0.9, 0.01)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (light polishing)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, accepting only improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # ensure we always move from a larger to a smaller value
        donor = i if cur[i] > cur[j] else j
        receiver = j if donor == i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s time budget and returns a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of a few seconds for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Length of the sequence (4000 is a good compromise between quality
    # and runtime; feel free to change it).
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve a previously found construction (if any) to use as a seed.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1. Build a large collection of analytical seeds.
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)

    # Extra families that were not in the original reference.
    for NW in np.arange(2.0, 6.5, 0.5):
        seeds.append(_dpss_window(n, NW))

    for beta in np.linspace(0.0, 15.0, 16):
        seeds.append(_kaiser_window(n, beta))

    for alpha in np.linspace(0.0, 1.0, 11):
        seeds.append(_tukey_window(n, alpha))

    seeds.append(_blackman_window(n))

    # ------------------------------------------------------------------
    # 2. Evaluate all seeds (fast, O(n log n) each) and keep the best few.
    # ------------------------------------------------------------------
    seed_vals = []
    for s in seeds:
        if time.time() > deadline:
            break
        s = np.maximum(s, 0.0)
        s_sum = s.sum()
        if s_sum == 0.0:
            continue
        s = s / s_sum
        val = _objective(s)
        seed_vals.append((val, s))
        if val < 1.7:  # cheap progress print for promising candidates
            print(f"[Seed] val={val:.6f}")

    if not seed_vals:
        # fallback to uniform if everything failed (should never happen)
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
    else:
        seed_vals.sort(key=lambda x: x[0])
        best_val, best_seq = seed_vals[0]
        print(f"[BestSeed] initial best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinements – run several deterministic passes on the
    #    top K seeds (here K=5) to explore nearby basins.
    # ------------------------------------------------------------------
    top_K = min(5, len(seed_vals))
    for rank in range(top_K):
        if time.time() > deadline:
            break
        seq_start = seed_vals[rank][1]
        print(f"\n--- Refinement from seed #{rank+1} (val={seed_vals[rank][0]:.6f}) ---")

        # 3.1 Projected sub‑gradient descent
        seq_sg, val_sg = _proj_subgradient_refine(seq_start, n, deadline, max_iters=150000)
        if val_sg < best_val:
            best_val, best_seq = val_sg, seq_sg
            print(f"[Refine‑ProjSub] new global best={val_sg:.6f}")

        # 3.2 Greedy mass‑transfer flattening
        seq_gr, val_gr = _greedy_flatten(best_seq, n, deadline, max_iters=80000)
        if val_gr < best_val:
            best_val, best_seq = val_gr, seq_gr
            print(f"[Refine‑Greedy] new global best={val_gr:.6f}")

        # 3.3 Light hill‑climbing
        seq_hc, val_hc = _hill_climb(best_seq, n, deadline, max_iters=40000)
        if val_hc < best_val:
            best_val, best_seq = val_hc, seq_hc
            print(f"[Refine‑Hill] new global best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # 4. One final hill‑climb on the globally best sequence (if time left).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_final, val_final = _hill_climb(best_seq, n, deadline, max_iters=60000)
        if val_final < best_val:
            best_val, best_seq = val_final, seq_final
            print(f"[Final‑Hill] new global best={val_final:.6f}")

    # ------------------------------------------------------------------
    # 5. Sanity check with the provided evaluate_sequence() (optional).
    # ------------------------------------------------------------------
    try:
        # evaluate_sequence expects a list and handles normalisation internally.
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")

    # expose for potential reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
```

This implementation augments the previous hybrid optimisation with additional analytic families (DPSS, Kaiser, Tukey, Blackman), a greedy mass‑transfer refinement, and a more exhaustive local search on the top seeds. It stays within the 1000 s budget, continuously checks the deadline, and returns the best non‑negative sequence found.
