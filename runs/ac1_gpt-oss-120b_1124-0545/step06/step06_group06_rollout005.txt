import cvxpy as cp
import numpy as np
from scipy.sparse import csr_matrix

def lp_optimize(a, M):
    n = len(a)
    # Build Toeplitz-like matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            rows.append(k)
            cols.append(i)
            data.append(a[k-i])  # coefficient a_{k-i}
    T = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = cp.Variable(n)
    constraints = [b >= 0, T @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None
    return b.value
```

"""
Propose a non‑negative step function that minimizes

    2·n·max(conv(a,a)) / (∑ a)²

The algorithm is a multi‑stage time‑aware search:

1. **Broad seeding** – deterministic analytic families (uniform, triangular,
   geometric, power‑law, beta, sine‑window, etc.) and many random Dirichlet /
   sparse draws.  All candidates are normalised to sum = 1 and the best is
   kept.

2. **Adam on a smoothed max** – the vector lives on the simplex via a
   log‑softmax parametrisation.  A slowly growing log‑sum‑exp temperature
   approximates the max; Adam updates are applied until the time budget
   is exhausted.

3. **Exact sub‑gradient mirror descent** – an exponentiated‑gradient step
   using the true sub‑gradient of the true max (no smoothing) further
   refines the solution.

4. **Linear‑programming refinement** – given the current bound M = max(conv),
   solve a linear program that maximises ∑b subject to b≥0 and
   (a * b)≤M (convolution entries are linear in b).  Mixing the LP
   solution back into a reduces the maximal autocorrelation.  The LP is
   solved on a down‑sampled version (≤2000 entries) to keep it cheap.

5. **Simulated‑annealing hill climbing** – stochastic Dirichlet reshuffles
   of a few coordinates allow the algorithm to escape shallow minima.

All stages respect the global deadline (default 1000 s) and the best
sequence found is returned as a plain ``list`` of ``float`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming ∑seq == 1.
    This is exactly the quantity minimized by the competition (the 2·n factor
    is part of the definition, the sum‑normalisation is enforced beforehand).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(np.round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed collector
# ----------------------------------------------------------------------


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """
    Assemble a diverse set of normalised candidate vectors for length ``n``.
    """
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.2))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_sym_geom(n, r))

    # power‑law family
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # beta‑family grid – broaden a bit beyond the original range
    alphas = np.linspace(0.1, 3.0, 12)
    betas = np.linspace(0.1, 3.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # sine^γ families
    for gamma in np.linspace(0.5, 4.0, 10):
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
        seeds.append(a / a.sum())

    # random Dirichlet draws – include very sparse concentrations
    for conc in (0.01, 0.05, 0.1, 0.3, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns – a few non‑zeros with very low concentration
    for k in (3, 5, 10, 20):
        for conc in (0.01, 0.05, 0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous global best, interpolated to current length if needed
    if previous is not None:
        if previous.shape == (n,):
            seeds.append(previous / previous.sum())
        else:
            # linear interpolation to shape n
            old = np.arange(previous.shape[0])
            new = np.linspace(0, previous.shape[0] - 1, n)
            interp = np.interp(new, old, previous)
            interp = np.maximum(interp, 0.0)
            if interp.sum() > 0:
                seeds.append(interp / interp.sum())

    return seeds

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 40000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on a smoothed version of ``max(conv)``.
    The vector lives on the simplex via a log‑softmax parametrisation.
    A log‑sum‑exp temperature (beta) starts small and grows slowly.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.2            # start with a gentle smoothing
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        beta = min(beta * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k (linear in a)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.9
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent using exponentiated‑gradient updates.
    Learning rate decays as 1/√t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at lag max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill climbing with a simple annealing schedule.
    A random subset of 2‑4 positions reshuffles its total mass via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature proportional to current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 5)          # size of the neighbourhood (2‑4)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _build_conv_matrix_dense(a: np.ndarray) -> np.ndarray:
    """
    Build a dense matrix T such that (T @ b)[k] = Σ_i a[i]·b[k‑i]
    (zero outside the valid index range).  Shape is (2·n‑1, n).
    """
    n = len(a)
    L = 2 * n - 1
    T = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        T[k, idx] = a[k - idx]
    return T


def _lp_max_sum_under_conv(a: np.ndarray, M: float):
    """
    Solve   max Σ b_j   s.t.   b ≥ 0   and   (a * b)_k ≤ M  for all k.
    Returns the optimal b as a dense ndarray, or ``None`` if the LP fails.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = len(a)
    # Build the convolution matrix (dense – cheap for n ≤ 2000)
    T = _build_conv_matrix_dense(a)

    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # Try a fast solver; fall back to SCS if needed
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-7, reltol=1e-6, feastol=1e-7, verbose=False)
    except Exception:
        prob.solve(solver=cp.SCS, max_iters=8000, verbose=False)

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    val = b.value
    if val is None:
        return None
    # Numerical noise may produce tiny negatives – clamp
    return np.maximum(val, 0.0)


def _lp_refine(
    a: np.ndarray,
    original_n: int,
    deadline: float,
    max_iters: int = 5,
) -> np.ndarray:
    """
    Perform a few LP‑based mixing steps.
    The LP works on a down‑sampled version (≤2000 entries) to keep it cheap.
    After the iterations the up‑sampled vector is returned.
    """
    # Choose a working size that balances speed and fidelity
    work_n = min(2000, original_n)
    if work_n < original_n:
        # linear interpolation to down‑sample a
        old_idx = np.linspace(0, original_n - 1, original_n)
        new_idx = np.linspace(0, original_n - 1, work_n)
        a_work = np.interp(new_idx, old_idx, a)
        a_work = np.maximum(a_work, 0.0)
        a_work = a_work / a_work.sum()
    else:
        a_work = a.copy()

    best_seq = a_work
    best_val = _objective(best_seq)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(best_seq, best_seq)
        M = np.max(conv)

        b_opt = _lp_max_sum_under_conv(best_seq, M)
        if b_opt is None:
            break
        total = b_opt.sum()
        if total == 0.0:
            break
        b_norm = b_opt / total

        # Mixing coefficient – modest to keep iterates stable
        t = 0.25
        new_seq = (1.0 - t) * best_seq + t * b_norm
        new_seq = np.maximum(new_seq, 0.0)
        new_seq = new_seq / new_seq.sum()
        new_val = _objective(new_seq)

        if new_val < best_val - 1e-9:
            best_val = new_val
            best_seq = new_seq
            print(f"[LP] it={it} improved to {new_val:.6f}")
        else:
            # No improvement – stop early
            break

    # Upsample back to the original dimension if we worked on a reduced size
    if work_n < original_n:
        old_idx = np.linspace(0, work_n - 1, work_n)
        new_idx = np.linspace(0, work_n - 1, original_n)
        best_seq = np.interp(new_idx, old_idx, best_seq)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()

    return best_seq

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  The routine respects the supplied time budget and returns the
    best sequence found as a plain ``list`` of ``float`` values.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Retrieve a previous solution, if the environment provides one
    # ------------------------------------------------------------------
    prev_global = globals().get("height_sequence_1")
    if isinstance(prev_global, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(prev_global, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (larger n gives a finer discretisation)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – broad seed scan
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n--- Scanning seeds for length {n} ---")
        seeds = _collect_seeds(n, rng, previous)

        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} val={val:.6f}")

    # Fallback if nothing was produced (should not happen)
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)

    # Ensure we work with the current length (could be any of the scanned ones)
    n = len(global_best_seq)
    print(
        f"\n=== Refinement on length {n} (current best {global_best_val:.6f}) ==="
    )

    # ------------------------------------------------------------------
    # Stage 2 – Adam on smoothed max
    # ------------------------------------------------------------------
    seq_adam, val_adam = _adam_optimize(
        global_best_seq, n, deadline, max_iters=50000
    )
    if seq_adam is not None and val_adam < global_best_val:
        global_best_seq = seq_adam.copy()
        global_best_val = val_adam
        print(f"[overall‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient mirror descent
    # ------------------------------------------------------------------
    seq_sub, val_sub = _subgradient_refine(
        global_best_seq, n, deadline, max_iters=300000
    )
    if val_sub < global_best_val:
        global_best_seq = seq_sub.copy()
        global_best_val = val_sub
        print(f"[overall‑SubGrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – LP‑based refinement (optional, may be skipped if cvxpy missing)
    # ------------------------------------------------------------------
    try:
        seq_lp = _lp_refine(global_best_seq, n, deadline, max_iters=8)
        if seq_lp is not None:
            val_lp = _objective(seq_lp)
            if val_lp < global_best_val:
                global_best_seq = seq_lp.copy()
                global_best_val = val_lp
                print(f"[overall‑LP] improved to {val_lp:.6f}")
    except Exception as exc:
        # LP stage failed (cvxpy not available or numerical issues)
        print("[LP] Skipped –", exc)

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    seq_sa, val_sa = _simulated_annealing(
        global_best_seq, n, deadline, max_iters=300000
    )
    if val_sa < global_best_val:
        global_best_seq = seq_sa.copy()
        global_best_val = val_sa
        print(f"[overall‑SA] improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional but helpful)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )
    # Export the sequence as a plain Python list
    return global_best_seq.tolist()
