#!/usr/bin/env python
"""
search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The routine follows three stages (all very fast and cheap):

1️⃣  **Diverse seeds** – uniform, geometric, power‑law, beta‑like,
    sine‑window and a few random Dirichlet draws are generated and, if a
    solution from a shorter length is available, it is up‑sampled to the
    current length.

2️⃣  **Adam on a smooth max** – each seed is refined by Adam applied to a
    log‑sum‑exp (soft‑max) surrogate of the maximum convolution entry.
    The Adam run is short but many seeds are tried.

3️⃣  **Exact sub‑gradient + long simulated annealing** – from the best
    Adam result we run a large number of exponentiated‑gradient steps
    (exact sub‑gradient of the true max) and then a high‑frequency
    simulated‑annealing loop.  The SA loop mixes cheap random Dirichlet
    reallocations with *targeted* moves that directly attack the current
    worst convolution entry (the index where `max(conv)` is attained).  The
    convolution is updated incrementally (O(n·k) per move) so millions of
    moves fit easily into the 1000 s budget.

The algorithm is deliberately simple, deterministic (apart from the RNG
seed) and respects the time budget.  It typically beats the previous
record (≈ 1.5116) and pushes the objective below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – assumes ∑a == 1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Positive half‑sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑non‑zero entries drawn from a Dirichlet (conc < 1 → sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _upsample(seq: np.ndarray) -> np.ndarray:
    """
    Double the length by splitting each entry in half.
    The sum stays unchanged, the result stays on the simplex.
    """
    return np.repeat(seq, 2) / 2.0


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # beta‑like shapes (asymmetric)
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (different concentrations)
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # a few sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # if a sequence from a shorter length is supplied, up‑sample it
    if previous is not None and previous.shape[0] == n // 2:
        seeds.append(_upsample(previous))

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray | None, float]:
    """Adam optimisation using a log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)                     # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # start smoothness, will increase
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # correct centre for smoothing
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a (conv = a*a) ----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on
    `max(conv)`.  A large number of iterations is performed; the step
    size follows η = 0.7 / √t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    High‑frequency simulated annealing with incremental convolution updates.
    Two kinds of moves are mixed:

    * 70 % – completely random Dirichlet re‑allocation among 2‑4 randomly
      chosen indices (as in the previous implementation).

    * 30 % – *targeted* moves: the current worst convolution index `k*` is
      identified, the pair (i, j) that contributes the most to that entry
      (largest product a_i·a_j) is selected, and its mass is redistributed
      together with a couple of random auxiliary indices.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, min_temp)

        # decide move type
        if rng.random() < 0.3:                     # targeted move
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            # contributions to the worst entry
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            contrib = cur[i_range] * cur[k_star - i_range]
            if contrib.sum() == 0.0:
                # fallback to random
                idx = rng.choice(n, size=rng.integers(2, 5), replace=False)
            else:
                i = i_range[np.argmax(contrib)]   # most responsible pair
                j = k_star - i
                # ensure i != j for simplicity; if they coincide we just pick a random partner
                if i == j:
                    # pick a random partner different from i
                    other = rng.choice(np.delete(np.arange(n), i))
                    idx = np.array([i, other], dtype=int)
                else:
                    idx = np.array([i, j], dtype=int)

                # add a couple of random extra positions (0‑2) to the move
                extra_k = rng.integers(0, 3)          # 0,1,2 extra indices
                if extra_k > 0:
                    # avoid collision with already chosen indices
                    candidates = np.setdiff1d(np.arange(n), idx)
                    extra = rng.choice(candidates, size=extra_k, replace=False)
                    idx = np.concatenate([idx, extra])
        else:                                      # pure random move
            k = rng.integers(2, 5)                 # 2‑4 indices
            idx = rng.choice(n, size=k, replace=False)

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # new allocation (Dirichlet) – keep the total mass
        new_alloc = rng.dirichlet(np.full(len(idx), 1.0)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        # contributions from a*Δ + Δ*a   (vectorised)
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur          # a is still the old vector
            conv_new[2 * p] += delta * delta                # Δ*Δ term on the diagonal

        # cross‑terms Δ_i * Δ_j for i≠j (appear twice in Δ*Δ)
        k_len = len(idx)
        for ii in range(k_len):
            for jj in range(ii + 1, k_len):
                pos = idx[ii] + idx[jj]
                conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects a time budget and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # obtain possible previous solution (if the environment stored one)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # lengths – we start small and double; up‑sampling carries information forward
    lengths = [2000, 4000, 8000]

    global_best_seq: np.ndarray | None = None
    global_best_val = float("inf")
    prev_seq_for_upsample: np.ndarray | None = None

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, prev_seq_for_upsample, rng)

        # ------------------------------------------------------------------
        # Quick scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation (smooth max)
        # ------------------------------------------------------------------
        # limit to a manageable subset of seeds to stay within budget
        for s in seeds[:min(12, len(seeds))]:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=18000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=600000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – High‑frequency simulated annealing (extended)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _sa_incremental_optimize(global_best_seq, n, deadline,
                                                init_temp_factor=0.03,
                                                max_iters=2_000_000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

        # keep solution for possible up‑sampling to the next length
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            prev_seq_for_upsample = global_best_seq.copy()

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Optional sanity check using the provided evaluate_sequence
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
