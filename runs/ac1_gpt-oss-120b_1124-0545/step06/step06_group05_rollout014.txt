"""
Improved search for a low‑value step‑function (non‑negative sequence).

The optimizer works in three stages for each problem size n:

1. **Rich seed generation** – analytic families (uniform, geometric, power‑law,
   Beta‑law, arcsine, symmetric geometric, sine‑window, Gaussian) plus random
   Dirichlet draws.  All seeds are normalised to L1‑norm 1 and stored on the
   simplex.

2. **Smooth Adam optimisation** – we minimise a smooth surrogate of the true
   objective:
       L(a) = log‑sum‑exp(β·conv(a,a)) / β + λ·‖conv(a,a)‖₂² .
   The sequence is kept on the simplex via a log‑softmax parametrisation.
   For each length we run a modest number of Adam steps on the best few seeds.

3. **Exact sub‑gradient refinement + cheap annealing** – an exponentiated‑gradient
   (mirror‑descent) step using the true sub‑gradient of `max(conv(a,a))`,
   followed by a short Metropolis‑Hastings anneal that swaps tiny amounts of
   mass.  Both respect the global time budget.

The routine explores increasing lengths (2000 → 4000 → 8000 → 12000) and
re‑uses the best sequence seen so far as a seed for the next size.  All
computations use FFT‑based convolution and keep the total running time below
the 1000 s budget.  The returned list is a plain Python list of non‑negative
floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))², seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    return a / a.sum()


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _beta_dist(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) discretised on the grid { (i+0.5)/n }."""
    i = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (i ** (alpha - 1)) * ((1.0 - i) ** (beta - 1))
    a = np.where(np.isfinite(a), a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse or dense Dirichlet draw (Gamma (α,1) normalised)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """
    Discrete analogue of the continuous arcsine family.
    a_i ∝ ((i+0.5)*(n‑0.5‑i))**exponent  (exponent < 0).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide="ignore", invalid="ignore"):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    if a.sum() <= 0.0:
        return _uniform(n)
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of start‑vectors (already normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # Fundamental families
    seeds.append(_uniform(n))

    # Geometric families (smoothly decreasing)
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # Power‑law families (heavy head)
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # Combined geometric‑power
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # Symmetric geometric (U‑shaped)
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Smooth windows
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # Dirichlet draws – spanning sparse to dense regimes
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # Beta families – explore both U‑shaped and bell‑shaped densities
    for α in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        for β in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            seeds.append(_beta_dist(n, α, β))

    # Arcsine‑type seeds (negative exponents)
    for exp in [-1.2, -1.0, -0.75, -0.5, -0.25]:
        try:
            seeds.append(_arcseed(n, exp))
        except Exception:
            pass

    # Mirror each seed (helps with asymmetric optima)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # Re‑use previous best if shape matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / max(previous.sum(), 1e-12))

    # Guarantee non‑empty list
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimiser (smooth max + variance regulariser)
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 12000,
    beta0: float = 1.0,
    beta_mul: float = 1.02,
    lr0: float = 1.0,
    lambda0: float = 3.0,
    lambda_decay: float = 0.997,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate
        L(a) = (1/β)·log∑ₖexp(β·convₖ(a)) + λ·‖conv(a,a)‖₂² .
    Returns best sequence encountered and its *true* objective.
    """
    v = np.log(seed + 1e-30)               # logits for softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0
    lam = lambda0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # point on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # smooth max term
        beta *= beta_mul
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of smoothed max
        rev_a = a[::-1]
        full_max = _conv_fft(rev_a, w)
        grad_max = 2.0 * full_max[n - 1 : n - 1 + n]

        # gradient of variance regulariser
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n - 1 : n - 1 + n]

        grad_a = grad_max + lam * grad_sq

        # back‑prop through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moments
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        lam *= lambda_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) using the true sub‑gradient of
    max(conv).  Returns best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # keep numbers stable
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Cheap simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Metropolis‑Hastings walk that transfers a tiny fraction of mass between two
    random sites, with exponential cooling.  Occasionally adds a global jitter.
    Returns the best sequence discovered.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-8, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found
    as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0               # safety margin

    # ------------------------------------------------------------------
    # Grab a previously discovered construction (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0:
                previous = previous / max(previous.sum(), 1e-12)
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start modest, enlarge if time permits
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]   # larger lengths explored only if time permits

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ---- 1. Seed creation -------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ---- 2. Evaluate raw seeds, keep a few promising ones -----------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_k = min(8, len(seed_vals))
        top_seeds = [s for s, _ in seed_vals[:top_k]]

        # record best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ---- 3. Adam optimisation (smooth surrogate) -------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            # give a little more room for larger n
            max_it = 18000 if n >= 8000 else 12000
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=max_it,
                beta0=0.8,
                beta_mul=1.03,
                lr0=1.2,
                lambda0=4.0,
                lambda_decay=0.996,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ---- 4. Exact sub‑gradient refinement ---------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            max_it = 50000 if n <= 4000 else 30000
            seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=max_it)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ---- 5. Simulated‑annealing fine‑tuning -------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # Carry the best sequence forward as a seed for the next size
        previous = global_best_seq

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(2000)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
