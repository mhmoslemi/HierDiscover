def _adam_param_optimize(start_seq, n, deadline, degree=5, max_iters=200000):
    # compute feature matrix X
    i = np.arange(n, dtype=np.float64)
    t = np.abs(2 * i / (n - 1) - 1)   # distance from center
    X = np.vstack([t**k for k in range(degree)]).T   # shape (n, degree)
    # initial theta: fit to log of start_seq
    w0 = np.log(start_seq + 1e-30)
    theta, *_ = np.linalg.lstsq(X, w0, rcond=None)
    # optimizer state
    m = np.zeros_like(theta)
    v = np.zeros_like(theta)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # step
    best_theta = theta.copy()
    best_val = float('inf')
    for t_iter in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute a from theta
        w = X @ theta
        a = np.exp(w)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0/n)
        else:
            a /= a_sum

        # conv / objective
        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_theta = theta.copy()
            # maybe print progress

        # compute gradient g_a = 2 * n * 2 * a_rev at overlapping positions
        # Actually compute g_a = a_rev * 2: We'll use factor 1 (scaling ignored)
        # For gradient of objective, we need factor of 2*n*2. Use factor = 1 for now.
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        # g_a_i = 2 * a[rev_idx]
        g_a = np.zeros(n, dtype=np.float64)
        g_a[idx_i] = 2.0 * a[rev_idx]   # ignoring factor n

        # chain rule: g_w = a * (g_a - dot(a,g_a))
        dot_ag = np.dot(a, g_a)
        g_w = a * (g_a - dot_ag)

        # gradient wrt theta
        g_theta = X.T @ g_w   # shape (degree,)

        # Adam update
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta ** 2)

        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)

        lr = 0.8 / np.sqrt(t_iter)   # similar to before
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    # after loop, compute best sequence from best_theta
    w_best = X @ best_theta
    a_best = np.exp(w_best)
    a_best /= a_best.sum()
    return a_best, best_val

"""
Optimisation of a non‑negative step function minimising

    2·n·max(conv(a,a)) / (∑ a)²

The routine works inside the given time budget (default 1000 s) and
returns the best sequence (list of floats) found.

Algorithmic outline
-------------------
0. **Seed scan** – a fast sweep through many analytically motivated families
   (uniform, triangular, beta‑shapes, sine‑window, random Dirichlet, sparse …)
   to obtain an initial candidate for each examined length.

1. **Low‑dimensional parametric Adam** – the vector is forced to be symmetric
   and written as

        a_i = exp( Σ_{k=0}^{d‑1} θ_k·t_i^k )   ,  t_i = |2·i/(n‑1)‑1| ∈ [0,1]

   (a smooth “bell” shape).  Adam updates are performed directly on the
   parameters θ using the exact sub‑gradient of the max‑convolution.
   Several polynomial degrees are tried; the best result is kept.

2. **Adam on a smoothed max** – the same Adam optimiser as in the reference
   implementation but now started from the sequence obtained in step 1.
   A log‑sum‑exp with a slowly increasing temperature approximates the max.

3. **Exact sub‑gradient mirror descent** – exponentiated‑gradient (mirror)
   descent on the full vector refines the solution further.

4. **Simulated‑annealing hill climbing** – stochastic reshuffling of a few
   coordinates (Dirichlet draws) helps escape shallow minima.

All stages terminate early if the global deadline is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming ∑seq == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(np.round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a diverse set of normalised candidate vectors for length ``n``."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.2))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # power‑law family
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # beta‑family grid
    alphas = np.linspace(0.2, 2.0, 10)
    betas = np.linspace(0.2, 2.0, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # sine^γ families
    for gamma in np.linspace(0.5, 4.0, 8):
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
        seeds.append(a / a.sum())

    # random Dirichlet draws
    for conc in (0.1, 0.3, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 30000,
) -> tuple[np.ndarray | None, float]:
    """Adam optimisation on a smoothed version of ``max(conv)``."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5            # initial smooth‑max temperature
    beta_mul = 1.001
    beta_max = 200.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing (cap at beta_max)
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp over convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent using exponentiated‑gradient updates."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max at lag max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """Stochastic hill climbing with a simple annealing schedule."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 5)          # 2‑4 positions
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _param_adam_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    degree: int = 5,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a symmetric polynomial parametrisation:

        a_i = exp( Σ_{k=0}^{degree‑1} θ_k·t_i^k ) / Z,
        t_i = |2·i/(n‑1) – 1|   ∈ [0,1].

    The gradient of the exact max‑convolution is used.
    """
    # symmetric distance vector
    i = np.arange(n, dtype=np.float64)
    t = np.abs(2.0 * i / (n - 1) - 1.0)  # 0 at centre, 1 at edges

    # feature matrix X_{i,k} = t_i^k
    X = np.vstack([t ** k for k in range(degree)]).T  # shape (n, degree)

    # initialise θ by a least‑squares fit to log(start_seq)
    eps = 1e-30
    log_a = np.log(start_seq + eps)
    theta, *_ = np.linalg.lstsq(X, log_a, rcond=None)

    m = np.zeros(degree)
    v_hat = np.zeros(degree)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_theta = theta.copy()
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # build a from current θ
        w = X @ theta
        w -= np.max(w)                 # stabilise exponentials
        a = np.exp(w)
        a /= a.sum()

        # exact max‑convolution and its index
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx]

        if cur_val < best_val:
            best_val = cur_val
            best_theta = theta.copy()
            print(f"[ParamAdam] deg={degree} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the max element (ignoring constant factor)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g_a = np.zeros(n, dtype=np.float64)
        g_a[idx_i] = 2.0 * a[rev_idx]

        # chain rule through the softmax‑like mapping
        dot = np.dot(a, g_a)
        g_w = a * (g_a - dot)          # ∂L/∂w
        g_theta = X.T @ g_w            # ∂L/∂θ

        # Adam update on θ
        m = beta1 * m + (1 - beta1) * g_theta
        v_hat = beta2 * v_hat + (1 - beta2) * (g_theta ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_corr = v_hat / (1 - beta2 ** it)

        lr = 0.8 / np.sqrt(it)
        theta = theta - lr * m_hat / (np.sqrt(v_corr) + eps_adam)

    # construct the best sequence
    w_best = X @ best_theta
    w_best -= np.max(w_best)
    a_best = np.exp(w_best)
    a_best /= a_best.sum()
    best_val = _objective(a_best)
    return a_best, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a plain ``list``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    # previous global best (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # lengths to explore – larger n gives finer discretisation
    lengths = [2000, 4000, 6000, 8000, 12000, 16000]

    global_best_seq = None
    global_best_val = float("inf")
    global_best_n = None

    # ------------------------------------------------------------------
    # Stage 0 – broad seed scan (fast)
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Seed scan for length {n} ---")
        seeds = _collect_seeds(n, rng, previous)
        for s in seeds:
            if time.time() > deadline:
                break
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                global_best_n = n
                print(f"[seed] n={n:5d} val={val:.6f}")

    # fallback in the unlikely event that no seed succeeded
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_n = n
        global_best_val = _objective(global_best_seq)

    print(
        f"\n=== Starting refinement (length {global_best_n}) "
        f"current best {global_best_val:.6f} ==="
    )

    # ------------------------------------------------------------------
    # Stage 1 – low‑dimensional parametric Adam (symmetrical polynomial)
    # ------------------------------------------------------------------
    for deg in [3, 4, 5, 6]:
        if time.time() > deadline:
            break
        seq, val = _param_adam_optimize(
            global_best_seq,
            global_best_n,
            deadline,
            degree=deg,
            max_iters=25000,
        )
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall‑Param] degree={deg} improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam on smoothed max (refine further)
    # ------------------------------------------------------------------
    seq, val = _adam_optimize(
        global_best_seq,
        global_best_n,
        deadline,
        max_iters=30000,
    )
    if seq is not None and val < global_best_val:
        global_best_seq = seq.copy()
        global_best_val = val
        print(f"[overall‑Adam] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient mirror descent
    # ------------------------------------------------------------------
    seq, val = _subgradient_refine(
        global_best_seq,
        global_best_n,
        deadline,
        max_iters=200000,
    )
    if val < global_best_val:
        global_best_seq = seq.copy()
        global_best_val = val
        print(f"[overall‑SubGrad] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    seq, val = _simulated_annealing(
        global_best_seq,
        global_best_n,
        deadline,
        max_iters=300000,
    )
    if val < global_best_val:
        global_best_seq = seq.copy()
        global_best_val = val
        print(f"[overall‑SA] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # Ensure the output is a plain list of Python floats
    return [float(x) for x in global_best_seq.tolist()]
