def _lbfgs_optimize(seed, n, deadline, beta_schedule=[1.0,5.0,20.0,80.0,320.0], max_iter_per_beta=200):
    v = np.log(seed + 1e-30)
    best_seq = None
    best_val = float('inf')
    for beta in beta_schedule:
        # define closure for loss and grad that uses current beta
        def func(v):
            a = _softmax(v)
            conv = _conv_fft(a, a)
            # smoothed loss L = logsumexp(beta*conv)/beta
            shift = conv.max()
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = exp_term.sum()
            loss = (np.log(sum_exp) + beta * shift) / beta   # equals logsumexp(beta*conv)/beta
            # compute gradient
            w = exp_term / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]
            # chain rule to logits
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return loss, grad_v
        # use scipy minimize
        res = scipy.optimize.minimize(
            fun=lambda v: func(v),
            x0=v,
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_beta, 'disp': False}
        )
        v = res.x  # use final as starting for next beta
        # evaluate exact objective at current a
        a = _softmax(v)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L-BFGS] n={n} beta={beta:.1f} val={val:.6f}")
        # early exit if deadline near etc.
        if time.time() > deadline - 0.5:
            break
    return best_seq, best_val
```

"""
Propose a step‐function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

by a multi‑stage continuation + quasi‑Newton optimisation.
The algorithm:

1. Build a rich pool of seed vectors (uniform, geometric, power‑law,
   sinusoidal windows, beta‑type, Dirichlet draws and up‑scaled
   previous best).
2. For each promising seed run a continuation on the smoothed
   “log‑sum‑exp” surrogate of the max‑convolution.
   The surrogate is

        Φ_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k),

   whose gradient is inexpensive (two FFTs).  A schedule of β values
   (starting from 1 and growing to ~300) turns the problem from a
   smooth landscape into the original non‑smooth one.
   Each β‑stage is handled by L‑BFGS‑B (SciPy) on the soft‑max
   parametrisation a_i = softmax(v)_i, which guarantees a≥0 and Σa=1.
3. The best vector from the L‑BFGS passes through a short exact
   sub‑gradient mirror‑descent (exponentiated‑gradient) and a cheap
   simulated‑annealing fine‑tuner – the same refinements used in the
   reference code.
4. The global best ever found (across several lengths) is cached in
   the global variable ``height_sequence_1`` so that subsequent runs
   can start from an up‑scaled version.
The whole routine respects the 1000 s budget; it stops early if the
deadline is reached.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for many calls)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families (all produce vectors already normalised to sum‑1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sin2_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _hanning(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)
    a[a < 0] = 0.0
    return a / a.sum()


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse pool of normalised start vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_sin2_window(n))
    seeds.append(_hanning(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_tent(n))

    # beta families
    for a in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        seeds.append(_beta_seed(n, a, a))
        seeds.append(_beta_seed(n, a, a * 1.5))
        seeds.append(_beta_seed(n, a * 1.5, a))

    # Dirichlet draws
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrored copies
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # upscale previous best if it exists and length differs
    if previous is not None:
        try:
            if previous.shape != (n,):
                seeds.append(_upscale(previous, n))
            else:
                seeds.append(previous / previous.sum())
        except Exception:
            pass

    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _lbfgs_optimize(seed: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule: list[float] | None = None,
                    max_iter_per_beta: int = 200) -> tuple[np.ndarray, float]:
    """
    Continuation L‑BFGS optimizer.
    Starts from `seed` (already normalised) and walks through a
    sequence of smoothness parameters `beta` (low → high) using
    SciPy's L‑BFGS‑B on the logits parametrisation.
    Returns the best sequence seen (exact objective) and its value.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 4.0, 12.0, 36.0, 108.0, 324.0]

    # initialise logits (soft‑max parametrisation)
    v = np.log(seed + 1e-30)

    best_seq = None
    best_val = float("inf")

    for beta in beta_schedule:
        if time.time() > deadline - 0.5:
            break

        # objective + gradient for given beta
        def fun_and_grad(vv: np.ndarray):
            a = _softmax(vv)
            conv = _conv_fft(a, a)

            # smoothed max: log‑sum‑exp
            shift = conv.max()
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = exp_term.sum()
            loss = (np.log(sum_exp) + beta * shift) / beta   # = log‑sum‑exp / β

            # gradient of loss w.r.t a
            w = exp_term / sum_exp          # softmax weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            return loss, grad_v

        # SciPy L‑BFGS‑B (accepts a function returning (f,grad) when jac=True)
        res = scipy.optimize.minimize(
            fun=lambda vv: fun_and_grad(vv),
            x0=v,
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_beta, 'disp': False}
        )
        v = res.x

        # evaluate exact objective at the current point
        a = _softmax(v)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS] n={n} β={beta:.1f} exact={val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror descent (same as in the reference code)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 100000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (lightweight)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # starting temperature as a modest fraction of current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # move ≤2.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        if it % 1000 == 0:
            # occasional global jitter
            noise = rng.normal(scale=0.001, size=n)
            cand = cur + noise
            cand = np.clip(cand, 0.0, None)
            if cand.sum() == 0.0:
                cand = _uniform(n)
            else:
                cand /= cand.sum()
            cand_val = _objective(cand)
            if cand_val < cur_val or rng.random() < np.exp(-(cand_val - cur_val) / (T + 1e-12)):
                cur = cand
                cur_val = cand_val
                if cand_val < best_val:
                    best_val = cand_val
                    best = cand.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={cand_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point – respects the 1000 s budget
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the grader's
    objective.  Returns a list of floats (the best sequence found
    before the deadline).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # reuse a previously discovered construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # lengths to explore – start modest, enlarge if time permits
    lengths = [2000, 4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # Build a rich seed pool
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Quick evaluation of raw seeds, keep the top few
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(6, len(seed_vals))]]

        # keep best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – L‑BFGS continuation from each top seed
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _lbfgs_optimize(
                seed_vec,
                n,
                deadline,
                beta_schedule=[1.0, 4.0, 12.0, 36.0, 108.0, 324.0],
                max_iter_per_beta=180,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall L‑BFGS] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – sub‑gradient refinement (exact max)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=80000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(
                global_best_seq, n, deadline, max_iters=None
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Symmetrisation (averaging with its mirror) – cheap improvement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[symmetrisation] n={n} improved to {sym_val:.6f}")

        # propagate the current best forward (for up‑scaling in the next length)
        previous = global_best_seq

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(2000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # cache for the next call
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
