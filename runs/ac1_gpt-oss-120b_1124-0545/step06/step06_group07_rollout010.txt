"""Optimisation via smooth‑max L‑BFGS and adaptive projected sub‑gradient.

The algorithm:
1. Build a diverse set of analytic seed sequences (uniform, sine,
   geometric, beta, Gaussian, Dirichlet, random softmax, and a sinc‑type).
2. Keep the best few seeds according to the exact objective.
3. For each seed run a few stages of L‑BFGS‑B on a smooth‑max
   surrogate (log‑sum‑exp) with an increasing temperature β.
   Parameters are the logits of a soft‑max, guaranteeing non‑negativity
   and ℓ₁‑norm ‑ 1.
4. Finish with a short projected sub‑gradient descent on the true
   max‑convolution.
5. Return the best sequence found.
"""

import time, math, numpy as np
from scipy.optimize import minimize

# helpers
def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _project_simplex(v, s=1.0):
    # Euclidean projection onto the simplex {x≥0, Σx=s}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# analytic families
def _uniform(n):
    return np.ones(n, dtype=float) / n

def _sine_window(n):
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()

def _geom(n, r):
    a = r ** np.arange(n, dtype=float)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _sym_geom(n, r):
    a = r ** np.arange(n, dtype=float)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=float)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _power(n, p):
    idx = np.arange(1, n + 1, dtype=float)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _beta_shape(n, alpha, beta):
    i = np.arange(1, n + 1, dtype=float)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _gaussian(n, sigma):
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _random_dirichlet(n, conc, rng):
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()

def _sinc_shape(n, sigma):
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    x = (i - mu) / sigma
    a = np.sinc(x)   # sin(pi x)/(pi x)
    a = np.clip(a, 0.0, None)  # keep only non‑negative part
    s = a.sum()
    if s == 0:
        return _uniform(n)
    return a / s

def _make_seeds(n, previous):
    rng = np.random.default_rng(2023)
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_sym_geom(n, r))
    for a in np.linspace(0.2, 2.0, 9):
        for b in np.linspace(0.2, 2.0, 9):
            seeds.append(_beta_shape(n, a, b))
    for frac in np.linspace(0.05, 0.4, 8):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))
    # new family
    for sigma_frac in np.linspace(0.03, 0.15, 5):
        sigma = max(1.0, sigma_frac * n)
        seeds.append(_sinc_shape(n, sigma))
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())
    return seeds

# smooth max objective and gradient
def _smooth_obj_grad(v, n, beta):
    a = _softmax(v)                     # ℓ1‑normalised
    conv = _conv_fft(a, a)              # length 2n‑1
    # smooth max = (log sum exp(beta*conv))/beta
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_term)
    sm = (np.log(sum_exp) + beta * shift) / beta
    obj = 2.0 * n * sm

    # gradient of sm w.r.t conv: w_i = exp(beta*conv_i)/sum_exp
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length 3n‑2
    grad_a = 2.0 * full[n-1:n-1+n]              # central n entries
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)                 # Jacobian of softmax
    return obj, grad_v

def _lbfgs_refine(seed, n, deadline,
                  beta_start=5.0, beta_end=80.0,
                  stages=4, max_iter_per_stage=600):
    """Refine a seed using L‑BFGS‑B on a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)
    best_seq = None
    best_val = float('inf')
    for stage in range(stages):
        if time.time() > deadline:
            break
        beta = beta_start * (beta_end / beta_start) ** (stage / (stages - 1))
        # local function closure capturing beta
        def fun(vv):
            return _smooth_obj_grad(vv, n, beta)
        # run L‑BFGS‑B
        res = minimize(fun, v, method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter_per_stage,
                                'disp': False})
        v = res.x
        a = _softmax(v)
        conv = _conv_fft(a, a)
        val = 2.0 * n * np.max(conv)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS] stage={stage+1}/{stages} beta={beta:.2f} "
                  f"iter={res.nit} best={val:.6f}")
    return best_seq, best_val

def _proj_subgradient_refine(start_seq, n, deadline,
                             max_iters=200000, target_sum=None):
    """Projected sub‑gradient descent on the true max‑convolution."""
    if target_sum is None:
        target_sum = 1.0  # we keep ℓ1‑norm = 1
    a = _project_simplex(start_seq, target_sum)
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] it={it} best={cur_val:.6f}")
        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = 1.0 / math.sqrt(it + 0.1)
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, target_sum)
    return best_seq, best_val

    The routine builds many analytic seeds, selects the most promising,
    refines them with an L‑BFGS optimiser on a smooth‑max surrogate
    (gradients obtained analytically via FFT) and finishes with a short
    projected sub‑gradient polishing.  The best sequence found before the
    deadline is returned as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    # Use the length that performed best so far – 4000 is a good compromise.
    n = 4000
    # retrieve previous best if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=float)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1. Generate a large, diverse pool of analytic seeds.
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)

    # ------------------------------------------------------------------
    # 2. Evaluate all seeds quickly and keep the top few.
    # ------------------------------------------------------------------
    seed_vals = []
    for s in seeds:
        conv = _conv_fft(s, s)
        val = 2.0 * n * np.max(conv)   # exact objective (sum(s)=1)
        seed_vals.append((val, s))
    seed_vals.sort(key=lambda x: x[0])
    # keep at most top 7 distinct seeds
    top_seeds = [s for _, s in seed_vals[:7]]
    print(f"[Info] best seed value = {seed_vals[0][0]:.6f}")

    global_best_seq = None
    global_best_val = float('inf')

    # ------------------------------------------------------------------
    # 3. L‑BFGS refinement of the selected seeds.
    # ------------------------------------------------------------------
    for idx, seed_seq in enumerate(top_seeds, 1):
        if time.time() > deadline:
            break
        print(f"[Info] L‑BFGS refinement of seed #{idx}")
        seq_ref, val_ref = _lbfgs_refine(seed_seq, n, deadline,
                                         beta_start=5.0, beta_end=80.0,
                                         stages=4, max_iter_per_stage=600)
        if seq_ref is not None and val_ref < global_best_val:
            global_best_val = val_ref
            global_best_seq = seq_ref.copy()
            print(f"[Info] new global best after L‑BFGS = {val_ref:.6f}")

    # ------------------------------------------------------------------
    # 4. Light projected sub‑gradient polishing (if time permits).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("[Info] final projected sub‑gradient polishing")
        seq_fin, val_fin = _proj_subgradient_refine(global_best_seq, n,
                                                    deadline,
                                                    max_iters=200000)
        if val_fin < global_best_val:
            global_best_val = val_fin
            global_best_seq = seq_fin.copy()

"""
Optimisation via smooth‑max L‑BFGS and adaptive projected sub‑gradient.

The algorithm works in three stages:

1. **Analytic seeding** – a large pool of normalised shapes (uniform,
   sine, geometric, power, beta, Gaussian, sinc, random Dirichlet,
   random soft‑max, plus the previous best) is generated.

2. **L‑BFGS refinement** – the best few seeds are further optimised.
   The parameters are the logits of a soft‑max (hence the sequence always
   stays on the ℓ1‑simplex and non‑negative).  The true max‑convolution
   is replaced by a smooth approximation
   `log‑sum‑exp(β·conv)/β`.  Gradients are obtained analytically via FFT.
   Several stages with an increasing smoothing coefficient β are run.

3. **Projected sub‑gradient polishing** – a short Euclidean‑projection
   sub‑gradient descent on the genuine max‑convolution fine‑tunes the
   result.

The best sequence found before the deadline is returned as a plain
list of floats, ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑convolution helpers (FFT, soft‑max, simplex projection)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ1‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=float) / n


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=float)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=float)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=float)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _sinc_shape(n: int, sigma: float) -> np.ndarray:
    """Symmetric, non‑negative sinc‑type window."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    x = (i - mu) / sigma
    a = np.sinc(x)                # sin(πx)/(πx)
    a = np.clip(a, 0.0, None)     # discard the negative lobes
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of normalised start vectors."""
    rng = np.random.default_rng(2023)
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Beta shapes – a denser grid than before
    for a in np.linspace(0.2, 2.0, 9):
        for b in np.linspace(0.2, 2.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows – sweep σ from 5 % to 40 % of n
    for frac in np.linspace(0.05, 0.4, 8):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    # New: sinc‑type windows (smoother, long‑range tails)
    for frac in np.linspace(0.03, 0.15, 5):
        sigma = max(1.0, frac * n)
        seeds.append(_sinc_shape(n, sigma))

    # Random Dirichlet draws
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random soft‑max draws
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # Inject the previously‑found best sequence, if available
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Smooth‑max surrogate + analytical gradient (required for L‑BFGS)
# ----------------------------------------------------------------------


def _smooth_obj_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smooth‑max surrogate
    2·n·log‑sum‑exp(β·conv)/β, where conv = a*a and a = softmax(v).
    """
    a = _softmax(v)                     # stays on the ℓ1‑simplex
    conv = _conv_fft(a, a)              # length 2·n‑1

    # smooth max = (log Σ exp(β·conv))/β   (numerically stabilised)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_term)
    sm = (np.log(sum_exp) + beta * shift) / beta
    obj = 2.0 * n * sm

    # gradient of sm w.r.t. conv is the soft‑max of β·conv
    w = exp_term / sum_exp
    # sub‑gradient of each conv_k w.r.t. a is 2·a_{k‑i}
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length 3·n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]        # central n entries

    # convert to gradient w.r.t. logits v (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_refine(seed: np.ndarray,
                  n: int,
                  deadline: float,
                  beta_start: float = 5.0,
                  beta_end: float = 80.0,
                  stages: int = 4,
                  max_iter_per_stage: int = 600):
    """
    Refine a seed with L‑BFGS‑B on a smooth‑max surrogate.
    The smoothing parameter β is increased geometrically over several stages.
    """
    v = np.log(seed + 1e-30)       # initialise logits
    best_seq = None
    best_val = float('inf')

    for stage in range(stages):
        if time.time() > deadline:
            break
        # geometric interpolation of β between start and end
        beta = beta_start * (beta_end / beta_start) ** (stage / (stages - 1))

        # minimise (obj, grad) returned by _smooth_obj_grad
        def fun(vv):
            return _smooth_obj_grad(vv, n, beta)

        res = minimize(fun, v, method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter_per_stage,
                                'disp': False})
        v = res.x
        a = _softmax(v)

        # evaluate the exact (non‑smooth) objective
        conv = _conv_fft(a, a)
        val = 2.0 * n * np.max(conv)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS] stage {stage + 1}/{stages}  β={beta:.2f}  "
                  f"iters={res.nit}  best={val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max‑convolution)
# ----------------------------------------------------------------------


def _proj_subgradient_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 200000,
                             target_sum: float | None = None):
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution.
    Keeps the iterates on the ℓ1‑simplex (sum = target_sum, default = 1).
    """
    if target_sum is None:
        target_sum = 1.0
    a = _project_simplex(start_seq, target_sum)

    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] it={it}  best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 1.0 / math.sqrt(it + 0.1)      # diminishing step‑size
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, target_sum)

    return best_seq, best_val
