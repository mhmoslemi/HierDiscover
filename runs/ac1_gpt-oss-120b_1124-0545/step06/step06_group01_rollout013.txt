def _conv_full(x, y):
    L = len(x) + len(y) - 1
    N = 1 << ((L - 1).bit_length())
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def generate_seeds(n, rng, previous):
    seeds = []
    seeds.append(np.full(n, 1.0 / n))
    # sine
    i = np.arange(n)
    seeds.append(np.sin(np.pi * (i+1) / (n+1)))
    seeds[-1] /= seeds[-1].sum()
    # geometric
    for r in np.linspace(0.2, 0.95, 10):
        a = r ** np.arange(n)
        a /= a.sum()
        seeds.append(a)
    # power law
    for p in np.linspace(0.0, 4.0, 9):
        a = 1.0 / ((np.arange(1,n+1)) ** p)
        a /= a.sum()
        seeds.append(a)
    # beta shape
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            i = np.arange(1, n+1)
            a = (i ** (alpha-1)) * ((n - i + 1) ** (beta-1))
            a /= a.sum()
            seeds.append(a)
    # Dirichlet
    for conc in [0.1, 0.3, 1.0, 3.0, 10.0]:
        for _ in range(5):
            a = rng.dirichlet(np.full(n, conc))
            a /= a.sum()
            seeds.append(a)
    # Gaussian logits softmax
    for _ in range(5):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        seeds.append(a)
    # random piecewise linear (like before)
    for _ in range(5):
        K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n-1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        a /= a.sum()
        seeds.append(a)
    # log-normal shaped
    for _ in range(5):
        mu = rng.uniform(-2, 2)
        sigma = rng.uniform(0.5, 2.0)
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - mu)/sigma)**2)
        a += 1e-12
        a /= a.sum()
        seeds.append(a)
    if previous is not None and len(previous) == n:
        a = np.asarray(previous, dtype=np.float64)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a /= s
            seeds.append(a)
    return seeds

def adam_optimize(a0, n, deadline, max_iters=200000):
    # a0 assumed normalized sum=1
    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1=0.9; beta2=0.999; eps=1e-12
    lr0=0.5
    beta=5.0
    beta_mul=1.02
    best_seq = a0.copy()
    best_val = objective_of(a0,n)
    t=0
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        t+=1
        a = np.exp(v - np.max(v))  # compute softmax manually
        a = a / a.sum()
        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100 == 0:
                print(f"[Adam] iter={it} best={cur_val:.6f}")
        # update smooth max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)
        grad = 2.0 * grad_full[n-1:n-1+n]
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1**t)
        v2_hat = v2 / (1.0 - beta2**t)
        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
    return best_seq, best_val

def hill_climb_optimize(a0, n, deadline, max_iters=200000):
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = objective_of(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))
        # contributions
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)
        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)
        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        val = objective_of(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] iter={it} best={val:.6f}")
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)
    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a, a)) / (sum(a)^2)

where `conv` is the linear convolution.  The algorithm

* builds a diverse collection of analytical and random seeds,
* keeps the best seed (global best on the simplex),
* then performs a long, time‑aware local optimisation consisting of:
    – Adam on a smooth‑max surrogate (log‑softmax parametrisation),
    – exact sub‑gradient descent on the true max,
    – L2‑flattening (driving the auto‑convolution towards a constant),
    – a final targeted hill‑climb.
All phases respect the global time budget (≈1000 s).  The final
vector is returned as a plain list of Python floats and stored in the
global variable `height_sequence_1` for warm‑starting future calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)); assumes sum(a)=1."""
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed generation (diverse analytic and random families)
# ----------------------------------------------------------------------


def _make_seeds(n: int, rng: np.random.Generator,
                warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # uniform & half‑sine window
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(s / s.sum())

    # geometric families
    for r in np.linspace(0.2, 0.95, 10):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # power‑law families
    for p in np.linspace(0.0, 4.0, 9):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # beta‑like shapes
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # Dirichlet draws (different concentrations)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(5):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # Softmax of Gaussian logits
    for _ in range(5):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # Random piecewise‑linear shapes
    for _ in range(5):
        K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(a / a.sum())

    # Log‑normal (Gaussian bell) shapes
    for _ in range(5):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # warm‑start from the previous global best, if compatible
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate.
    Parameterisation: a = softmax(v).  The smooth‑max uses a temperature β
    that is increased multiplicatively.
    """
    v = np.log(a0 + 1e-30)                     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 5.0            # initial smoothness
    beta_mul = 1.02

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # softmax → simplex
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:          # numerical safeguard
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]  # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max (exponentiated‑gradient)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent)
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a*a – μ·1‖² (μ = 1/(2n‑1)) using exponentiated‑gradient.
    Drives the auto‑convolution towards a uniform vector.
    """
    a = a0.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic hill‑climb: move a tiny amount of mass away
    from the contributors to the current maximal convolution coefficient.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123,

    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found (as a plain list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Choose a length (larger n gives a lower theoretical bound)
    # ------------------------------------------------------------------
    n = 4000          # works comfortably within the time budget

    # ------------------------------------------------------------------
    # Build a diverse set of seeds
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds, keep the global best
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Allocate time for a deep refinement of the current global best
    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max
    remaining = deadline - time.time()

    seq, val = _adam_optimize(global_best_seq, n, phase_deadline,

    if val < global_best_val:

        print(f"[Phase‑Adam] best={val:.6f}")

    # Phase 2 – Exact sub‑gradient descent
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.30
    seq, val = _subgradient_optimize(global_best_seq, n, phase_deadline,

    if val < global_best_val:

        print(f"[Phase‑SubGrad] best={val:.6f}")

    # Phase 3 – L2‑flattening descent

    phase_deadline = time.time() + remaining * 0.20
    seq, val = _l2_flatten_optimize(global_best_seq, n, phase_deadline,
                                    max_iters=200_000)
    if val < global_best_val:

    # Phase 4 – Targeted hill‑climb (use whatever time is left)

                                        max_iters=200_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Hill] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:

    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
