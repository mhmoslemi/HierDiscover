1. Adam on a smoothed log‑sum‑exp surrogate of the max‑convolution.
   The temperature β is increased up to 1000, giving a tight approximation
   to the true ℓ∞‑norm.
2. Exponentiated‑gradient (mirror) descent using the exact sub‑gradient
   of the active convolution entry, with a slightly larger step size.
3. Lightweight simulated‑annealing / hill‑climbing that swaps small
   mass fragments.  A final “mass‑balancing” sweep moves mass away from
   indices that dominate the peak convolution.

All stages respect the global 1000 s budget and keep the best sequence
found globally (across different lengths).  The best result is cached
in the global variable ``height_sequence_1`` for reuse by subsequent
calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT‑based, O(N log N))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for many calls)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric progression a_i = r**i, normalised."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric × power‑law."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both sides)."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int, width_factor: float = 0.8) -> np.ndarray:
    """Cosine window with support ≈ width_factor·n (zero elsewhere)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    width = width_factor * n / 2.0
    d = np.abs(i - centre)
    a = np.where(d <= width,
                 0.5 * (1.0 + np.cos(np.pi * d / width)),
                 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inv_sqrt_triangle(n: int) -> np.ndarray:
    """a_i proportional to 1/√(1 - |i-centre|/(n/2)).  Zero at the edges."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    d = np.abs(i - centre) / (n / 2.0)
    # avoid division by zero at centre – set a finite value
    a = 1.0 / np.sqrt(np.clip(1.0 - d, 1e-12, 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) on interior points (avoid infinities)."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(alpha,…,alpha)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse pool of start vectors (already normalised)."""
    rng = np.random.default_rng(12345)
    seeds = []

    # base families
    seeds.append(_uniform(n))
    for r in np.linspace(0.2, 0.99, 7):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))
    for r in np.linspace(0.2, 0.9, 5):
        for p in np.linspace(0.0, 1.5, 5):
            seeds.append(_geom_power(n, r, p))
    for r in np.linspace(0.2, 0.9, 5):
        seeds.append(_sym_geom(n, r))
    seeds.append(_gaussian(n, sigma_factor=0.12))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_gaussian(n, sigma_factor=0.25))
    seeds.append(_raised_cosine(n, width_factor=0.5))
    seeds.append(_raised_cosine(n, width_factor=0.8))
    seeds.append(_raised_cosine(n, width_factor=0.95))
    seeds.append(_inv_sqrt_triangle(n))
    seeds.append(_tent(n := n))   # tent window (linear rise‑fall)
    seeds.append(_sine_window := lambda n: np.sin(np.pi * (np.arange(n)+1)/(n+1)))  # not normalised yet
    # normalise sine window
    # we'll compute it separately to keep code simple:
    # (the lambda above is just a placeholder; replace below)
    # Actually we include the explicit sine window:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a = a / a.sum()
    seeds.append(a)

    # beta families
    for a in [0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        seeds.append(_beta_seed(n, a, a))
        seeds.append(_beta_seed(n, a, a * 1.5))
        seeds.append(_beta_seed(n, a * 1.5, a))

    # Dirichlet draws with various alphas
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrored copies (helps symmetry)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # upscale previous best if provided
    if previous is not None:
        if previous.shape != (n,):
            try:
                seeds.append(_upscale(previous, n))
            except Exception:
                pass
        else:
            seeds.append(previous / previous.sum())

    # final safeguard
    if not seeds:
        seeds.append(_uniform(n))

    # normalise all seeds (some constructions may have been added out‑of‑order)
    normalized = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        s = np.clip(s, 0.0, None)
        if s.sum() == 0.0:
            continue
        s = s / s.sum()
        normalized.append(s)
    return normalized


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed max (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000,
                   beta0: float = 2.0,
                   beta_target: float = 1000.0,
                   beta_mul: float = 1.001,
                   lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """Adam on L_β = (1/β)·log Σ exp(β·conv)."""
    # logits (softmax parametrisation)
    v = np.log(seed + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = np.exp(v - np.max(v))
        a = a / a.sum()                # softmax → simplex

        conv = _conv_fft(a, a)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # increase temperature β gradually
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # log‑sum‑exp weights (smooth max)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # gradient w.r.t. a (via reversed convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # Adam moments
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (exponentiated gradient / mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Mirror‑descent using the exact sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_i = max_idx - i_range
        g[i_range] = 2.0 * a[rev_i]

        # exponentiated‑gradient step with a slightly larger prefactor
        lr = 1.3 / np.sqrt(it)      # 1.3 is a tuned constant
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)      # numerical stability
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climbing fine tuner
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Lightweight Metropolis‑Hastings that swaps a tiny fragment of mass."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.04 * cur[i]   # move ≤4 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global jitter to escape plateaus
        if it % 1200 == 0:
            noise = rng.normal(scale=0.001, size=n)
            candidate = cur + noise
            candidate = np.clip(candidate, 0.0, None)
            if candidate.sum() == 0.0:
                candidate = _uniform(n)
            else:
                candidate /= candidate.sum()
            cand_val = _objective(candidate)
            if cand_val < cur_val or rng.random() < np.exp(-(cand_val - cur_val) / (T + 1e-12)):
                cur = candidate
                cur_val = cand_val
                if cand_val < best_val:
                    best_val = cand_val
                    best = candidate.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={cand_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Final mass‑balancing sweep
# ----------------------------------------------------------------------


def _balance_mass(seq: np.ndarray, n: int, max_passes: int = 3) -> np.ndarray:
    """
    Simple deterministic sweep that moves a tiny fraction of mass from
    indices that contribute most to the current peak towards indices
    that contribute least.
    """
    a = seq.copy()
    for _ in range(max_passes):
        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        # contributions to the peak entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - idx
        contrib = a[idx] * a[rev]

        # pick the top‑contributing index (largest a_i*a_j)
        worst_i = idx[np.argmax(contrib)]
        # pick a low‑mass index (preferably far from centre)
        low_candidates = np.where(a < a.mean())[0]
        if low_candidates.size == 0:
            break
        best_j = low_candidates[np.argmin(np.abs(low_candidates - (n // 2)))]

        # move a tiny fragment
        delta = 0.02 * a[worst_i]
        delta = min(delta, a[worst_i])
        a[worst_i] -= delta
        a[best_j] += delta
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point – respects the 1000 s total budget
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the evaluation
    function.  Returns the best sequence (list of floats) found before
    the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # reuse a previously discovered construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # try a few lengths – larger lengths give a larger search space
    lengths = [4000, 8000, 12000, 16000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # Build a diverse seed pool
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Quick evaluation of raw seeds, keep the top few
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(10, len(seed_vals))]]

        # keep best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation from each top seed
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=12000,
                beta0=2.0,
                beta_target=1000.0,
                beta_mul=1.001,
                lr0=0.6,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=80000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            seq, val = _anneal_optimize(
                global_best_seq, n, deadline, max_iters=None
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – deterministic mass‑balancing sweep
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            balanced = _balance_mass(global_best_seq, n, max_passes=4)
            bal_val = _objective(balanced)
            if bal_val < global_best_val:
                global_best_val = bal_val
                global_best_seq = balanced.copy()
                print(f"[balance] n={n} improved to {bal_val:.6f}")

        # ------------------------------------------------------------------
        # Symmetrisation (averaging with its mirror often reduces the peak)
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[symmetrisation] n={n} improved to {sym_val:.6f}")

        # propagate best forward for the next length
        previous = global_best_seq

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")
        global_best_val = _objective(global_best_seq)

    # normalise one final time (scaling does not affect the objective)
    global_best_seq = np.clip(global_best_seq, 0.0, None)
    if global_best_seq.sum() == 0.0:
        global_best_seq = _uniform(len(global_best_seq))
    else:
        global_best_seq = global_best_seq / global_best_seq.sum()

    # cache for next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
