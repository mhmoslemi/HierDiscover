def _simulated_annealing(start_seq, n, deadline, max_iters=200000, init_temp=None, final_temp=1e-6):

    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0*n*np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    # compute init temp if not given: maybe set to (cur_val - maybe lower bound) * 0.1

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        # temperature schedule
        if init_temp is None:
            # set init based on cur_val; e.g., 0.01*cur_val
            init_temp = cur_val * 0.02
        # geometric cooling
        T = init_temp * (final_temp/init_temp) ** (it / max_iters)

        # propose random multiplicative perturbation on log-scale
        log_cur = np.log(cur + 1e-30)
        sigma = 0.5 * (1 - it / max_iters)  # start 0.5, decay linearly to 0
        new_log = log_cur + sigma * rng.standard_normal(n)
        new_a = np.exp(new_log)
        new_a /= new_a.sum()
        # ensure nonnegative
        conv_new = _conv_fft(new_a, new_a)
        new_val = 2.0*n*np.max(conv_new)

        # acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_a.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={new_val:.6f}")

    return best_seq, best_val

"""
Hybrid optimisation for the “max‑auto‑convolution’’ problem (enhanced).

* Global search – Differential Evolution on a piece‑wise‑linear description
  with more knots than before (≈2·√n).  Larger population and a few extra
  generations give a richer family of shapes.

* Analytic seeding – many classic windows (uniform, sine, geometric, power,
  beta, Gaussian) **plus** Hann, Hamming and Blackman windows.  The seeds are
  normalised to the ℓ₁‑simplex.

* Deterministic polishing – three refinements:
    1) Adam on a smooth‑max surrogate (twice as many iterations).
    2) Exact sub‑gradient descent (mirror‑descent style) with a longer budget.
    3) Projected sub‑gradient descent (Euclidean projection onto the simplex).

* Final stochastic phase – a simulated‑annealing move that perturbs the
  log‑weights and accepts uphill moves with a decreasing temperature schedule.
  This often escapes the shallow local minima that remain after the deterministic
  steps.

All phases respect the global time budget (default 1000 s).  The function
returns the best non‑negative sequence found, ready for ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic seed families (ℓ₁‑normalised)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian window, centred at (n‑1)/2."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    """Hamming window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (0.42 -
         0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _softmax_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    e = np.exp(logits - np.max(logits))
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of ℓ₁‑normalised start vectors."""
    rng = np.random.default_rng(2023)
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Beta shapes – finer grid
    for a in np.linspace(0.2, 2.0, 9):
        for b in np.linspace(0.2, 2.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for frac in np.linspace(0.05, 0.4, 8):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    # Classic windows
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # Random Dirichlet draws
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Softmax random
    for _ in range(7):
        seeds.append(_softmax_seed(n, rng))

    # Previous best (if supplied)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

# Adam optimisation on a smoothed‑max surrogate

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 30000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)  # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                # stays on the simplex (sum=1)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)    # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val

# Projected sub‑gradient refinement (Euclidean projection)

def _proj_subgradient_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300000,
                             target_sum: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (projection onto the ℓ₁‑simplex of
    specified sum, default 1.0).
    """
    a = _project_simplex(start_seq, target_sum)

    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        lr = 1.0 / np.sqrt(it)

        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, target_sum)

    return best_seq, best_val

def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor == i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

# Simulated annealing (log‑space perturbations)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200000,
                         init_temp_factor: float = 0.02,
                         final_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Stochastic annealing: multiplicative log‑perturbations with a geometric
    temperature schedule.  Accepts uphill moves with probability exp(‑Δ/ T).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    init_temp = max(cur_val * init_temp_factor, 1e-8)

    log_cur = np.log(cur + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = init_temp * (final_temp / init_temp) ** (it / max_iters)

        # perturb log‑weights; variance decays linearly
        sigma = 0.5 * (1.0 - it / max_iters) + 1e-6
        cand_log = log_cur + sigma * rng.standard_normal(n)
        cand = np.exp(cand_log)
        cand /= cand.sum()

        cand_val = 2.0 * n * np.max(_conv_fft(cand, cand))

        if cand_val < cur_val or rng.random() < np.exp(-(cand_val - cur_val) / max(T, 1e-12)):
            cur = cand
            cur_val = cand_val
            log_cur = cand_log
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={cand_val:.6f}")

    return best_seq, best_val

def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """Interpolate a low‑dimensional simplex vector onto full length‑n, normalise."""
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    conv = _conv_fft(a, a)
    val = 2.0 * n * np.max(conv)
    return val, a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 60,
                 K: int | None = None,
                 max_gens: int = 300) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on a piece‑wise‑linear representation.
    Uses ≈2·√n knots (more expressive than the reference).
    """
    rng = np.random.default_rng()

    if K is None:
        K = max(30, int(2 * np.sqrt(n)))   # richer knot set
    knots = np.linspace(0, n - 1, K)

    # initialise population with Dirichlet draws (already on the simplex)
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]

    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_param(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            if ds == 0.0:
                donor = _uniform(K)
            else:
                donor = donor / ds

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = _uniform(K)
            else:
                trial = trial / ts

            val_trial, seq_trial = _eval_param(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines a richer global DE, many analytic seeds (including
    Hann/Hamming/Blackman windows), longer deterministic refinements and a
    final simulated‑annealing polishing phase.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin for final I/O

    # retrieve previous best, if any
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # We will try a single, fairly large length; larger lengths tend to
    # asymptotically improve the constant factor.
    lengths = [4000]  # feel free to extend, e.g. [4000, 6000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1. Analytic seeds
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)
        for s in seeds:
            conv = _conv_fft(s, s)
            val = 2.0 * n * np.max(conv)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 2. Global DE search (enhanced)
        # ------------------------------------------------------------------
        seq_de, val_de = _de_optimize(n, deadline, popsize=60, K=None,
                                      max_gens=300)
        if val_de < global_best_val:
            global_best_val = val_de
            global_best_seq = seq_de.copy()
            print(f"[DE] n={n:5d} new global best={val_de:.6f}")

        # ------------------------------------------------------------------
        # 3. Adam refinement (more iterations)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_adam, val_adam = _adam_optimize(global_best_seq, n,
                                                deadline, max_iters=30000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val_adam:.6f}")

        # ------------------------------------------------------------------
        # 4. Exact sub‑gradient refinement (extended)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_sg, val_sg = _subgradient_refine(global_best_seq, n,
                                                 deadline, max_iters=150000)
            if seq_sg is not None and val_sg < global_best_val:
                global_best_val = val_sg
                global_best_seq = seq_sg.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val_sg:.6f}")

        # ------------------------------------------------------------------
        # 5. Projected sub‑gradient refinement (enhanced)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_psg, val_psg = _proj_subgradient_refine(global_best_seq, n,
                                                       deadline,
                                                       max_iters=400000,
                                                       target_sum=1.0)
            if seq_psg is not None and val_psg < global_best_val:
                global_best_val = val_psg
                global_best_seq = seq_psg.copy()
                print(f"[overall‑ProjSub] n={n:5d} new global best={val_psg:.6f}")

        # ------------------------------------------------------------------
        # 6. Light stochastic hill‑climb (final polishing)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_hc, val_hc = _hill_climb(global_best_seq, n,
                                         deadline, max_iters=100000)
            if seq_hc is not None and val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[overall‑Hill] n={n:5d} new global best={val_hc:.6f}")

        # ------------------------------------------------------------------
        # 7. Simulated annealing (escape shallow minima)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_sa, val_sa = _simulated_annealing(global_best_seq, n,
                                                  deadline,
                                                  max_iters=500000)
            if seq_sa is not None and val_sa < global_best_val:
                global_best_val = val_sa
                global_best_seq = seq_sa.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val_sa:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be required)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = 2.0 * n * np.max(_conv_fft(global_best_seq,
                                                    global_best_seq))
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Verify with the provided evaluate_sequence() (optional)
    # --------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    # expose the best sequence for a possible next call
    globals()["height_sequence_1"] = global_best_seq.tolist()

    return global_best_seq.tolist()
