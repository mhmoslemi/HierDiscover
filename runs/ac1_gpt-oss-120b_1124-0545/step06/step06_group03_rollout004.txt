"""
search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm is a three‑stage pipeline:

1️⃣  **Diverse analytical seeds** – Uniform, sine, cosine, geometric,
    power‑law, beta‑like, random Dirichlet, sparse patterns and a
    warm‑start from the previous best solution (if any).

2️⃣  **Continuous refinement** – Adam optimiser on a log‑sum‑exp smooth
    approximation of the max‑operator, followed by an exact
    sub‑gradient mirror‑descent (exponentiated‑gradient) on the true
    max.  Both stages respect the simplex (∑a = 1) and are stopped
    automatically when the global time budget expires.

3️⃣  **Heavy‑duty simulated annealing** – An incremental convolution
    update makes a move O(k·n) cheap.  Three move families are used:
        * small Dirichlet reallocations on 2‑4 random positions,
        * medium redistributions on 10‑20 positions,
        * a “global blend” that mixes in a fresh Dirichlet sample.
    A slowly cooling temperature and Metropolis acceptance allow the
    search to escape local minima.  The SA phase is run on the best
    sequence found in stage 2 for the remaining time.

The routine respects a 1000 s wall‑clock budget and returns the best
list of heights discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Normalised objective 2·n·max(conv(a,a)) assuming ∑a == 1.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i + 0.5) / n)  # positive half‑cosine
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries, drawn from a Dirichlet (conc < 1 → sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(
    n: int, previous: np.ndarray | None, rng: np.random.Generator
) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_cosine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # beta‑like shapes (asymmetric)
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (various concentrations)
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # a few sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse a previous solution if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a smooth log‑sum‑exp approximation of the max.
    The iterate stays on the simplex via a softmax parametrisation.
    """
    eps = 1e-30
    v = np.log(seed + eps)                     # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # fixed bug: use max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a  (conv = a⋆a)
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the true
    max‑convolution value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_enhanced_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2000000,
) -> tuple[np.ndarray, float]:
    """
    High‑frequency simulated annealing with three move families:
        • tiny Dirichlet updates on 2‑4 indices,
        • medium redistributions on 10‑20 indices,
        • a global blend that mixes a fresh Dirichlet sample.
    An incremental convolution update makes each move O(k·n) cheap.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)               # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # temperature proportional to current objective, slow cooling
    T = max(0.05 * cur_obj, 1e-12)
    decay = 0.9999

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # temperature schedule
        T = max(T * decay, 1e-12)

        move_type = rng.random()

        # --------------------------------------------------------------
        # tiny move (most frequent)
        # --------------------------------------------------------------
        if move_type < 0.80:
            k = rng.integers(2, 5)                     # 2‑4 positions
            idx = rng.choice(n, size=k, replace=False)

            total = cur[idx].sum()
            if total <= 0.0:
                continue

            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            deltas = new_alloc - cur[idx]

            # ----- incremental convolution update -----
            conv_new = cur_conv.copy()
            # a*Δ + Δ*a contribution
            for p, delta in zip(idx, deltas):
                if delta == 0.0:
                    continue
                conv_new[p : p + n] += 2.0 * delta * cur
                conv_new[2 * p] += delta * delta     # Δ*Δ diagonal term

            # cross‑terms Δ_i * Δ_j for i≠j (appear twice in Δ*Δ)
            for i in range(k):
                for j in range(i + 1, k):
                    pos = idx[i] + idx[j]
                    conv_new[pos] += 2.0 * deltas[i] * deltas[j]

            new_obj = 2.0 * n * np.max(conv_new)

            # Metropolis acceptance
            delta_obj = new_obj - cur_obj
            if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
                cur[idx] = new_alloc
                cur_conv = conv_new
                cur_obj = new_obj
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_seq = cur.copy()
                    print(f"[SA] n={n:5d} it={it:7d} best={new_obj:.6f}")

        # --------------------------------------------------------------
        # medium move (10‑20 indices) – still cheap enough
        # --------------------------------------------------------------
        elif move_type < 0.95:
            k = rng.integers(10, 20)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            deltas = new_alloc - cur[idx]

            conv_new = cur_conv.copy()
            for p, delta in zip(idx, deltas):
                if delta == 0.0:
                    continue
                conv_new[p : p + n] += 2.0 * delta * cur
                conv_new[2 * p] += delta * delta
            for i in range(k):
                for j in range(i + 1, k):
                    pos = idx[i] + idx[j]
                    conv_new[pos] += 2.0 * deltas[i] * deltas[j]

            new_obj = 2.0 * n * np.max(conv_new)

            delta_obj = new_obj - cur_obj
            if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
                cur[idx] = new_alloc
                cur_conv = conv_new
                cur_obj = new_obj
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_seq = cur.copy()
                    print(f"[SA‑mid] n={n:5d} it={it:7d} best={new_obj:.6f}")

        # --------------------------------------------------------------
        # global blend – occasional large jump
        # --------------------------------------------------------------
        else:
            # fresh Dirichlet draw over the whole simplex and blend a little
            alpha = 0.02  # mixing coefficient
            fresh = rng.dirichlet(np.full(n, 1.0))
            cur = (1 - alpha) * cur + alpha * fresh
            cur = cur / cur.sum()
            cur_conv = _conv_fft(cur, cur)
            cur_obj = 2.0 * n * np.max(cur_conv)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = cur.copy()
                print(f"[SA‑global] n={n:5d} it={it:7d} best={cur_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the given evaluation
    function within the supplied time budget.  Returns the best sequence
    (list of floats) found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for cleanup

    # retrieve a possible warm‑start from previous runs
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # candidate lengths – we try a few discrete resolutions
    lengths = [2000, 4000, 6000]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – explore all lengths, run Adam + sub‑gradient
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # quick evaluation of raw seeds
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # Adam optimisation on each seed (bounded by remaining time)
        for s in seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=12000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # Sub‑gradient refinement on the best sequence of this length
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=250000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive SA on the overall best (remaining time)
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        n_best = global_best_seq.shape[0]
        seq, val = _sa_enhanced_optimize(global_best_seq, n_best, deadline,
                                         max_iters=2_000_000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall‑SA] n={n_best:5d} new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Optional sanity check with the provided evaluate_sequence
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
