"""
Improved search for a low‑∞‑norm self‑convolution step function.

The previous state‑of‑the‑art used only strictly positive coefficients
(soft‑max / exponentiated‑gradient) which prevents highly sparse
constructions.  Theory shows that the optimal value can be close to 1
(the lower bound 2 n / (2 n‑1) ≃ 1).  To get there we use three ideas:

1.  **Diverse seeds** – dense (uniform, geometric, power‑law) and
    *sparse* seeds: a small random subset of positions with equal mass.
2.  **Projected sub‑gradient descent** – the true sub‑gradient of the
    infinity‑norm of the convolution is cheap (it only depends on the
    maximising shift).  After a gradient step we project back onto the
    probability simplex (Euclidean projection) which naturally creates
    zeros.
3.  **Targeted annealing** – a cheap Metropolis step that always moves
    mass from an index participating in the current maximal convolution
    to an index that does not, which directly attacks the bottleneck.

The routine runs for the whole time budget (default 1000 s), trying a
handful of different lengths (4000 → 8000 → 12000) and keeping the
globally best sequence found.  All heavy lifting is done with NumPy
FFT (O(n log n)) and a few thousand simple iterations, well within the
time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution via FFT (full length)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (exactly the same as evaluate_sequence up to clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))², seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    # Duchi et al. (2008) – O(n log n) due to sorting
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:            # all entries are zero – fallback uniform
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed factories (dense and sparse)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _sparse_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k uniformly‑weighted positions, the rest zero."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a varied set of start vectors (already normalised)."""
    seeds: list[np.ndarray] = []

    # dense analytic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    for r in np.linspace(0.5, 0.95, 7):
        seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.5, 7):
        seeds.append(_power(n, p))

    # Dirichlet draws (sparse ↔ dense)
    for alpha in [0.05, 0.2, 0.5, 1.0, 2.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # sparse uniform subsets
    for k in [30, 50, 80, 110, 150]:
        if k < n:
            seeds.append(_sparse_uniform(n, k, rng))

    # reuse previous best if it matches the size
    if previous is not None and previous.shape == (n,):
        seeds.append(previous.copy())

    return seeds


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_projected(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(conv) with Euclidean projection onto the simplex.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # step‑size schedule
    step0 = 1.0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # compute sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        # gradient = 2 * a[rev_idx]   (except possible double‑counting on the diagonal)
        g[idx_i] = 2.0 * a[rev_idx]

        # step size
        lr = step0 / np.sqrt(it)

        # gradient step followed by projection onto simplex
        a = _proj_simplex(a - lr * g)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSG] n={n} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted simulated‑annealing (mass moved away from the active sum)
# ----------------------------------------------------------------------


def _targeted_anneal(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis move that always removes a small amount of mass from an
    index participating in the current maximal convolution entry and
    adds it to an index that does **not**.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # temperature schedule (cooling)
    T0 = max(1e-6, cur_val * 0.05)
    decay = 0.9993

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        active = np.arange(i_start, i_end + 1, dtype=int)
        # complement – indices that do NOT belong to the active sum
        all_idx = np.arange(n, dtype=int)
        complement = np.setdiff1d(all_idx, active, assume_unique=True)

        if active.size == 0 or complement.size == 0:
            # fallback to a random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        else:
            # pick i proportionally to its current mass (focus on large contributors)
            probs = cur[active] / cur[active].sum()
            i = rng.choice(active, p=probs)
            j = rng.choice(complement)

        # transfer a tiny fraction of mass from i to j
        delta = rng.random() * 0.03 * cur[i]   # up to 3 % of the source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[TargetAnneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global jitter (helps escape shallow traps)
        if it % 1200 == 0:
            noise = rng.normal(scale=0.001, size=n)
            jitter = cur + noise
            jitter = np.clip(jitter, 0.0, None)
            if jitter.sum() == 0.0:
                jitter = _uniform(n)
            else:
                jitter /= jitter.sum()
            jval = _objective(jitter)
            if jval < cur_val or rng.random() < np.exp(-(jval - cur_val) / (T + 1e-12)):
                cur = jitter
                cur_val = jval
                if jval < best_val:
                    best_val = jval
                    best = jitter.copy()
                    print(f"[TargetAnneal] n={n} it={it:5d} (jitter) best={jval:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget, mixes dense & sparse seeds,
    runs a projected sub‑gradient descent and a targeted annealing phase.
    The best sequence encountered is returned as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Re‑use a previously discovered construction if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0:
                previous = previous / max(previous.sum(), 1e-12)
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start modest, enlarge if time permits
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 12000]   # larger lengths are tried only if time remains
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ----- Build a rich collection of seeds ---------------------------------
        seeds = _make_seeds(n, previous, rng)

        # ----- Evaluate raw seeds and keep the most promising ones ---------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_k = min(8, len(seed_vals))
        top_seeds = [s for s, _ in seed_vals[:top_k]]

        # Record best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ----- Phase 1 – Projected sub‑gradient descent (exact max) ---------------
        for si, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _subgradient_projected(seed_vec, n, deadline, max_iters=25000)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall ProjSG] n={n} seed={si} best={val:.6f}")

        # ----- Phase 2 – Targeted annealing --------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _targeted_anneal(global_best_seq, n, deadline, max_iters=None)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall TargetAnneal] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")
        global_best_val = _objective(global_best_seq)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
