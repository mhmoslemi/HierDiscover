"""
Optimisation of a non‑negative step function to minimise
    2·n·max(conv(a,a))/(∑a)²

The algorithm proceeds in 4 stages:

1. **Rich seed generation** – systematic scan of several analytic families
   (beta‑shapes with dense grids, two‑sided power laws, triangular,
   sinusoidal, uniform, random Dirichlet). All seeds are normalised to
   ∑a=1 and the best among them is kept.

2. **Continuation Adam** – smooth‑max optimisation on the best seed.
   The smoothness parameter β is increased exponentially (starting from 5
   and multiplied by 1.02 each iteration) so that the surrogate converges
   to the true max. The optimiser works on the log‑softmax parameters and
   respects the simplex constraint automatically.

3. **Exact sub‑gradient mirror descent** – after Adam we run a large number
   of exponentiated‑gradient steps using the true sub‑gradient of
   max(conv). This stage reduces the objective further.

4. **Incremental simulated‑annealing** – cheap moves that re‑allocate the
   total mass of a handful of positions (2–4) are performed. The convolution
   is updated incrementally which makes each move O(n·k). An exponential
   cooling schedule is used and the best solution found is retained.

All stages respect the global time budget (default 1000 s).  The function
`propose_candidate` returns the best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective value assuming ∑a = 1 (i.e. 2·n·max(conv))."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return vectors normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _triangle(n: int) -> np.ndarray:
    """Symmetric triangle: a[i] = min(i+1, n-i)."""
    i = np.arange(n, dtype=np.float64) + 1
    a = np.minimum(i, n - i + 1)
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _edge_power(n: int, p: float) -> np.ndarray:
    """Two‑sided power law a[i] ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) + ((n - i + 1) ** (-p))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _cosine_power(n: int, p: float) -> np.ndarray:
    """Cosine‑based shape a[i] ∝ (sin(π·(i+0.5)/n) + ε)^p, ε avoids zero."""
    eps = 1e-8
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (np.sin(np.pi * i / n) + eps) ** p
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _make_rich_seeds(
    n: int, previous: np.ndarray | None, rng: np.random.Generator
) -> list[np.ndarray]:
    """Return a list of diverse normalised seeds."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_triangle(n))

    # beta shapes – dense grid, emphasising edge‑heavy cases
    alphas = np.concatenate(
        (np.linspace(0.05, 0.2, 4), np.linspace(0.2, 0.6, 5), np.linspace(0.6, 0.9, 4))
    )
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # two‑sided power law
    for p in np.linspace(0.2, 4.0, 9):
        seeds.append(_edge_power(n, p))

    # cosine‑power (both positive and negative exponents)
    for p in np.linspace(-2.0, 2.0, 9):
        seeds.append(_cosine_power(n, p))

    # random Dirichlet draws with several concentrations
    for conc in (0.1, 0.3, 0.7, 1.5, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # reuse previous solution if length matches
    if previous is not None and previous.shape == (n,):
        # normalise to sum = 1 (the optimiser works on the simplex)
        prev = previous.astype(np.float64)
        s = prev.sum()
        if s > 0:
            seeds.append(prev / s)

    return seeds

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray | None, float]:
    """Adam on log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-30
    # initialise log‑softmax parameters
    v = np.log(seed + eps)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # starting smoothness for log‑sum‑exp
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # current point on simplex
        a = _softmax(v)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # increase smoothness
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a  (conv = a*a)
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) descent on the true max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # active sub‑gradient: only indices that contribute to the maximal sum
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent on simplex (exponentiated‑gradient)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # for numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
    max_iters: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """High‑frequency simulated annealing with O(n·k) updates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)            # full convolution (length 2n‑1)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, min_temp)

        # pick a small random subset of indices (2‑4)
        k = rng.integers(2, 5)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # incremental update of convolution
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            # a * Δ + Δ * a term (both ordered)
            conv_new[p : p + n] += 2.0 * delta * cur
            # Δ * Δ term (self‑product)
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i * Δ_j for i≠j (appear twice)
        for i in range(k):
            for j in range(i + 1, k):
                pos = idx[i] + idx[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the given time budget and returns the
    best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # retrieve a previously‑found construction if we are running repeatedly
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # lengths to explore – we start small (fast) and finish with the largest
    lengths = [2000, 4000, 6000, 8000, 10000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_rich_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Quick scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam (smooth max)
        # ------------------------------------------------------------------
        for s in seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=15000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=250000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------

"""
Optimisation of a non‑negative step‑function for the objective

    2·n·max(conv(a,a)) / (∑a)²

The programme follows four successive phases while respecting a global
1000 s time budget:

1️⃣ **Rich seed generation** – a dense grid of analytic families
   (beta‑shapes, two‑sided power laws, cosine‑powers, triangle,
   sine window, uniform and random Dirichlet draws) is normalised to
   ∑a = 1.  The best raw seed is kept.

2️⃣ **Continuation Adam** – a smooth (log‑sum‑exp) surrogate of the
   max‑operator is minimised with Adam on the log‑soft‑max parameters.
   The smoothing factor β is multiplied by 1.02 each iteration so that
   the surrogate converges to the true max.

3️⃣ **Exact sub‑gradient mirror‑descent** – the optimum is refined
   with exponentiated‑gradient steps using the true sub‑gradient of
   max(conv).  The step‑size follows η = 0.7/√t.

4️⃣ **Incremental simulated‑annealing** – cheap moves that re‑allocate
   the total mass of 2‑4 randomly chosen positions are performed.
   The full convolution is updated incrementally (O(n·k)) and an
   exponential cooling schedule (T←T·0.9995) decides acceptance.
   The best solution seen during the SA run is retained.

The function ``propose_candidate`` returns the best sequence found as a
plain ``list[float]`` and works with the provided ``evaluate_sequence``
routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective value assuming ∑a = 1 (i.e. 2·n·max(conv))."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return vectors normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _triangle(n: int) -> np.ndarray:
    """Symmetric triangle: a[i] = min(i+1, n-i)."""
    i = np.arange(n, dtype=np.float64) + 1
    a = np.minimum(i, n - i + 1)
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _edge_power(n: int, p: float) -> np.ndarray:
    """Two‑sided power law a[i] ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) + ((n - i + 1) ** (-p))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _cosine_power(n: int, p: float) -> np.ndarray:
    """Cosine‑based shape a[i] ∝ (sin(π·(i+0.5)/n)+ε)^{p}."""
    eps = 1e-8
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (np.sin(np.pi * i / n) + eps) ** p
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _make_rich_seeds(
    n: int, previous: np.ndarray | None, rng: np.random.Generator
) -> list[np.ndarray]:
    """Return a list of diverse normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_triangle(n))

    # beta shapes – dense grid, emphasising edge‑heavy configurations
    alphas = np.concatenate(
        (np.linspace(0.05, 0.2, 4), np.linspace(0.2, 0.6, 5), np.linspace(0.6, 0.9, 4))
    )
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # two‑sided power law
    for p in np.linspace(0.2, 4.0, 9):
        seeds.append(_edge_power(n, p))

    # cosine‑power (both side‑heavy and centre‑heavy)
    for p in np.linspace(-2.0, 2.0, 9):
        seeds.append(_cosine_power(n, p))

    # random Dirichlet draws with various concentrations
    for conc in (0.1, 0.3, 0.7, 1.5, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # reuse a previous solution if it matches the length
    if previous is not None and previous.shape == (n,):
        prev = previous.astype(np.float64)
        s = prev.sum()
        if s > 0:
            seeds.append(prev / s)

    return seeds

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)                     # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness for the log‑sum‑exp
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # increase smoothness
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (conv = a*a)
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient mirror descent on the exact max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent on simplex (exponentiated‑gradient)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
    max_iters: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """High‑frequency simulated annealing with incremental convolution updates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)               # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initial temperature proportional to current objective
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * decay, min_temp)

        # pick 2‑4 random positions and re‑allocate their total mass
        k = rng.integers(2, 5)                     # 2,3,4
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental update of the convolution -----
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            # contributions from a*Δ + Δ*a (ordered pairs)
            conv_new[p : p + n] += 2.0 * delta * cur
            # Δ*Δ self term
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i*Δ_j for i≠j (appear twice)
        for i in range(k):
            for j in range(i + 1, k):
                pos = idx[i] + idx[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj

            previous = None
    else:
        previous = None

    # Lengths to explore – start with modest sizes and finish with the
