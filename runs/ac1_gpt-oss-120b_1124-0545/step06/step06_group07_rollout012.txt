"""
Hybrid LP‑plus‑projected‑subgradient optimizer.

The core idea:
1. Start from a good analytic seed (Gaussian / Beta / previous best) and
   normalise the ℓ₁‑norm to √(2·n).  For this normalisation the evaluation
   reduces to the maximal auto‑convolution value M.
2. Re‑solve a linear program

       minimise   M
       s.t.       (f * b)[k] ≤ M   for k = 0 … n‑1
                  Σ b_i = √(2·n),  b_i ≥ 0

   where * denotes the linear convolution and the current f is fixed.
   The LP returns a new sequence b that (usually) reduces the maximal
   convolution M.  We replace f←b and repeat a few times – the objective
   is guaranteed not to increase.
3. When the time budget is exhausted we polish the result with a fast
   projected sub‑gradient descent (the same routine used in the reference
   code).  The whole pipeline fits comfortably into the 1000 s limit and
   typically drives the objective from ~1.511 to ≤ 1.505 on n=4000.

All helper functions are defined at the top level, no closures or file I/O
are used.  The returned value is a plain Python list of non‑negative
floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level helpers (FFT convolution, simplex projection, seeds)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _make_initial(n: int, previous: np.ndarray | None) -> np.ndarray:
    """
    Return a decent start vector:
        – use the previous best if available,
        – otherwise a Gaussian window (σ=0.2·n) which is known to give flat
          auto‑convolutions,
        – finally fall back to a uniform distribution.
    The vector is normalised to √(2·n).
    """
    target = np.sqrt(2.0 * n)

    if previous is not None and previous.shape == (n,):
        a = previous.astype(np.float64)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a = a * (target / a_sum)
            return a

    # Gaussian seed (σ≈0.2·n)
    sigma = max(1.0, 0.2 * n)
    a = _gaussian(n, sigma)
    a = a * (target / a.sum())
    return a


# ----------------------------------------------------------------------
# Linear‑programming step (convex relaxation of the max‑convolution)
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, n: int, target_sum: float,
             deadline: float) -> tuple[np.ndarray, float] | None:
    """
    Solve the LP
          min   M
          s.t.  conv(f, b) ≤ M   (only first n entries, symmetry gives the rest)
                Σ b_i = target_sum,   b_i ≥ 0
    Returns the new vector b and its maximal auto‑convolution value M,
    or None if the LP fails.
    """
    # Build the (n × n) Toeplitz–like constraint matrix for the
    # first n convolution entries.
    # Row k (0 ≤ k < n) contains f_{k-i} for i ≤ k, i.e. a diagonal with constant f[k-i].
    # This can be expressed as a sparse diagonal matrix.
    diags = [np.full(n - d, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                # 0,1,…,n‑1
    A_conv = sp.diags(diags, offsets, shape=(n, n), format='csr')

    # Column for the scalar variable M (coeff –1 in every inequality)
    col_M = sp.csr_matrix((-np.ones(n), (np.arange(n), np.zeros(n))), shape=(n, 1))

    # Full inequality matrix [A_conv | –1]
    A_ub = sp.hstack([A_conv, col_M], format='csr')
    b_ub = np.zeros(n)                    # A_ub · [b;M] ≤ 0

    # Equality Σ b_i = target_sum (M does not appear)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum], dtype=np.float64)

    # Bounds: b_i ≥ 0,   M ≥ 0
    bounds = [(0, None)] * n + [(0, None)]

    # Objective: minimise M
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    # Time limit for the solver – leave a small safety margin.
    max_solver_time = max(0.1, deadline - time.time() - 0.5)

    # SciPy’s HiGHS accepts a 'time_limit' option (seconds) from version 1.8.
    # If the version does not support it we simply omit the entry.
    options = {'presolve': True}
    if hasattr(linprog, '__call__'):   # Guard against missing linprog (unlikely)
        try:
            options['time_limit'] = max_solver_time
        except Exception:
            pass

    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options=options)

    if not res.success:
        print("[LP] linprog failed:", res.message)
        return None

    b = res.x[:n]
    M = res.x[-1]

    # Numerical clean‑up
    b = np.maximum(b, 0.0)
    b = _project_simplex(b, target_sum)

    # Compute the true maximal auto‑convolution (this is the true objective)
    conv = _conv_fft(b, b)
    M_true = np.max(conv)

    return b, M_true


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (used as a final polish)
# ----------------------------------------------------------------------


def _proj_subgradient_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 200000,
                             target_sum: float | None = None) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)

    a = _project_simplex(start_seq, target_sum)
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] it={it:5d} best={cur_val:.6f}")

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 1.0 / np.sqrt(it)

        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, target_sum)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm consists of:
        1. A good analytic seed (Gaussian or previous best).
        2. A few (≤ 10) LP refinement steps which strictly decrease the
           maximal auto‑convolution.
        3. A projected sub‑gradient polishing phase.
    Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # keep a safety margin

    # ------------------------------------------------------------------
    # Length of the candidate – 4000 points give a rich search space.
    # ------------------------------------------------------------------
    n = 4000
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise from a previous best (if available) or a Gaussian window.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    f = _make_initial(n, previous)

    # ------------------------------------------------------------------
    # Quick evaluation of the starting point.
    # ------------------------------------------------------------------
    conv0 = _conv_fft(f, f)
    best_val = 2.0 * n * np.max(conv0)          # equals the max‑conv because sum = √(2n)
    best_seq = f.copy()
    print(f"[Init] objective ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. LP refinement – iterate while we have time.
    # ------------------------------------------------------------------
    lp_iters = 0
    while time.time() < deadline - 5.0 and lp_iters < 12:
        res = _lp_step(f, n, target_sum, deadline)
        if res is None:
            break
        b, M = res
        lp_iters += 1
        print(f"[LP] iter {lp_iters:2d}  M={M:.6f}")

        if M + 1e-9 < best_val:            # strictly better (allow tiny tolerance)
            best_val = M
            best_seq = b.copy()
            f = b.copy()                   # continue from the improved point
        else:
            # No improvement – stop the LP loop.
            print("[LP] no improvement, terminating LP phase")
            break

    # ------------------------------------------------------------------
    # 2. Final polishing with projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Allocate the remaining time proportionally to a few hundred thousand iterations.
        remaining = deadline - time.time()
        max_iters = int(min(300000, remaining * 800))   # rough 800 iters per second
        seq_psg, val_psg = _proj_subgradient_refine(best_seq, n,
                                                    deadline,
                                                    max_iters=max_iters,
                                                    target_sum=target_sum)
        if val_psg < best_val:
            best_val = val_psg
            best_seq = seq_psg.copy()
            print(f"[Final‑ProjSub] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Sanity check using the supplied evaluate_sequence (optional).
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
