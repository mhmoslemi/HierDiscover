def solve_lp(f, M, n, time_limit):
    A = np.empty((2*n-1, n), dtype=np.float64)
    A[:] = 0.0
    for j in range(n):
        A[j:j+n, j] = f
    # A shape (2n-1, n)
    bounds = [(0, None) for _ in range(n)]
    # maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # inequality constraints: A @ g <= M (vector of length 2n-1)
    b = np.full(2*n-1, M, dtype=np.float64)
    # use linprog
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'time_limit': time_limit})
    if not res.success:
        return None
    return res.x
```

Propose a step‑function that minimises the “max‑auto‑convolution” objective.

The search proceeds in three stages:

1️⃣  Broad seeding – analytic families (uniform, geometric, beta‑shapes,
    sine‑window, random Dirichlet) and a few random piecewise‑linear
    candidates are normalised (ℓ₁‑norm = 1) and evaluated with a fast FFT
    based objective 2·n·max(conv).

2️⃣  LP‑mix refinement – starting from the best seed we repeatedly solve a
    linear program
        maximise   Σ g_j
        subject to   g_j ≥ 0 ,   (f * g)_k ≤ M   (for all k)
    where f is the current sequence and M = max(f*f).  The LP returns a
    dense vector g; after scaling it to a simplex we mix it with f using
    the optimal mixing weight t = M/(M+M_g) (M_g = max(g*g)).  This step
    provably reduces the bound (M ← M·M_g/(M+M_g)) and is cheap for
    n≈2000 (≈64 MiB dense matrix).

3️⃣  Simple sub‑gradient polishing – a few iterations of exact sub‑gradient
    descent on the true max‑convolution keep the iterates on the simplex.

The whole routine respects the 1000 s wall‑time budget, prints progress,
stores the best sequence in the global variable ``height_sequence_1`` and
returns it as a plain ``list[float]``.

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _normalize(a: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum = 1, non‑negative)."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / a.shape[0])
    return a / s


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return _normalize(a)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    return _normalize(a)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return _normalize(a)


def _make_analytic_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a modest list of diversified, ℓ₁‑normalised seeds."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    alphas = np.linspace(0.2, 2.0, 7)
    betas = np.linspace(0.2, 2.0, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    return seeds


# ----------------------------------------------------------------------
# Random piecewise‑linear seeds
# ----------------------------------------------------------------------


def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a piecewise‑linear simplex draw via Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    return _normalize(a)


# ----------------------------------------------------------------------
# Fast objective (assumes ℓ₁‑norm = 1)
# ----------------------------------------------------------------------


def _fast_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 15000) -> np.ndarray:
    """Mirror‑descent (exponentiated‑gradient) on max(conv(a,a))."""
    a = start_seq.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _fast_obj(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)           # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best


# ----------------------------------------------------------------------
# Linear‑programming mixing step
# ----------------------------------------------------------------------


def _lp_mix_step(f: np.ndarray,
                 M: float,
                 deadline: float,
                 max_iters: int = 5) -> tuple[np.ndarray | None, float]:
    """
    Perform up to ``max_iters`` LP‑mix iterations.
    Returns the improved sequence (or None) and its objective.
    """
    n = f.shape[0]
    best_seq = f.copy()
    best_val = _fast_obj(f)

    it = 0
    while it < max_iters and time.time() < deadline - 0.5:
        it += 1
        # ------------------------------------------------------------------
        # Build the (2n‑1) × n constraint matrix A  (A·g ≤ M)
        # ------------------------------------------------------------------
        A = np.empty((2 * n - 1, n), dtype=np.float64)
        A[:] = 0.0
        # column‑wise fill: each column j carries a copy of f shifted down by j
        for j in range(n):
            A[j:j + n, j] = f

        # LP: maximise Σ g   <=>   minimise -Σ g
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        time_left = deadline - time.time()
        # give the LP at most a few seconds; HiGHS respects the ``time_limit`` key
        opt = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": max(0.1, min(5.0, time_left * 0.5))},
        )
        if not opt.success:
            print("[LP] iteration failed – stopping LP refinements")
            break

        g0 = opt.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # Scale to the simplex (ℓ₁‑norm = 1)
        g = g0 / sum_g0

        # Optional symmetrisation – often helpful
        g = (g + g[::-1]) * 0.5
        g = _normalize(g)

        # Compute max‑convolution of the new candidate
        M_g = np.max(_conv_fft(g, g))
        # Mix with the optimal weight t = M / (M + M_g)
        if M_g <= 0:
            break
        t = M / (M + M_g)
        new_f = (1.0 - t) * f + t * g
        new_f = _normalize(new_f)   # safeguard against numeric drift
        new_val = _fast_obj(new_f)

        if new_val < best_val:
            best_val = new_val
            best_seq = new_f.copy()
            f = new_f               # continue from the improved point
            M = np.max(_conv_fft(f, f))
            print(f"[LPMix] it={it:3d} new best={new_val:.6f}")
        else:
            # even if not better we still continue with the new point
            f = new_f
            M = np.max(_conv_fft(f, f))

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.
    The routine combines cheap analytic seeding, a few random piecewise samples,
    a dedicated linear‑programming “mix” refinement and a short sub‑gradient
    polishing.  The best sequence found before the time budget expires is
    returned as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    # ------------------------------------------------------------------
    # Choose a working length (2000 gives a good speed‑memory trade‑off)
    # ------------------------------------------------------------------
    n = 2000
    # If a previous construction exists and its length is modest, reuse it
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) >= 1500:
        n = min(len(previous), 2500)   # keep it reasonably small for LP

    print(f"=== Starting optimisation, n = {n} ===")

    # ------------------------------------------------------------------
    # 1. Generate a set of seed candidates
    # ------------------------------------------------------------------
    top_candidates: list[tuple[float, np.ndarray]] = []

    analytic_seeds = _make_analytic_seeds(n, rng)
    for s in analytic_seeds:
        val = _fast_obj(s)
        _update = (val, s)
        top_candidates.append(_update)

    # add a few random piecewise‑linear candidates (quick exploration)
    for _ in range(500):
        cand = _random_piecewise_linear(n, rng)
        val = _fast_obj(cand)
        top_candidates.append((val, cand))

    # keep only the best few for polishing
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:5]

    # initialise global best
    global_best_seq, global_best_val = top_candidates[0][1], top_candidates[0][0]

    # ------------------------------------------------------------------
    # 2. LP‑mix refinement (the core of the new approach)
    # ------------------------------------------------------------------
    for idx, (val, seq) in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(f"\n--- LP‑mix polishing #{idx+1} (seed value {val:.6f}) ---")
        seq = _normalize(seq)
        # ensure sum = 1
        seq = seq / seq.sum()
        refined_seq, refined_val = _lp_mix_step(seq,
                                                np.max(_conv_fft(seq, seq)),
                                                deadline,
                                                max_iters=8)
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[LP‑mix] improved global best to {refined_val:.6f}")

        # final brief sub‑gradient polishing
        if time.time() < deadline - 5.0:
            sub_seq = _subgradient_descent(global_best_seq,
                                           deadline,
                                           max_iters=8000)
            sub_val = _fast_obj(sub_seq)
            if sub_val < global_best_val:
                global_best_val = sub_val
                global_best_seq = sub_seq.copy()
                print(f"[SubGrad] improved global best to {sub_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final verification with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        # Fallback – should practically never happen
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = _uniform(n)
        global_best_val = _fast_obj(global_best_seq)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {n})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

def _update_top(top_list, cand_val, cand_seq, max_keep=8):
    """Insert a candidate into a bounded sorted list (ascending by value)."""
    top_list.append((cand_val, cand_seq))
    top_list.sort(key=lambda x: x[0])
    if len(top_list) > max_keep:
        top_list.pop()
