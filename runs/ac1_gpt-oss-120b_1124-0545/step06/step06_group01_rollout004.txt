"""
Propose a step‑function (non‑negative vector) that minimizes the
`evaluate_sequence` objective.

The algorithm works in three stages:

1. **Diverse seeding** – a collection of analytic families (uniform,
   geometric, beta‑like, sine‑window, Dirichlet draws, soft‑max of
   Gaussian logits) together with a few random piecewise‑linear
   candidates are generated and ℓ₁‑normalised.

2. **Incremental hill‑climbing** – starting from the best few seeds we run
   a large‑scale stochastic local search that moves a tiny amount of mass
   from a “donor” (chosen proportionally to its contribution to the
   currently maximal auto‑convolution entry) to a random “receiver”.
   The convolution is updated *incrementally* (O(n) per move) using the
   exact formula for a rank‑2 update, and the move is accepted only if
   it lowers the objective.  The step‑size decays slowly, allowing many
   millions of cheap moves within the 1000 s budget.

3. **Fine polishing** – the best candidate is finally refined with a few
   steps of Adam on a smooth‑max surrogate (log‑softmax parametrisation)
   to squeeze out the last fraction of improvement.

All steps respect the time limit; the best sequence found is returned
as a plain list of floats and cached in the global variable
`height_sequence_1` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return _conv_fft(a, a)


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv) (sum(a) is assumed to be 1)."""
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet on a random set of knots, linear interpolation to length n."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    probs = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, probs)
    s = a.sum()
    return a / s if s else _uniform(n)


def _make_analytic_seeds(n: int,
                         previous: np.ndarray | None,
                         rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse pool of ℓ₁‑normalised analytic start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws with several concentrations
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Softmax of Gaussian logits
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # warm‑start from previous run (interpolated if needed)
    if previous is not None:
        if previous.shape != (n,):
            # linear interpolation to the new length
            x_old = np.linspace(0, 1, previous.size)
            x_new = np.linspace(0, 1, n)
            prev_interp = np.interp(x_new, x_old, previous)
            prev_interp = np.maximum(prev_interp, 0.0)
            if prev_interp.sum() > 0:
                seeds.append(prev_interp / prev_interp.sum())
        else:
            seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 5000) -> tuple[np.ndarray | None, float]:
    """Adam on a smoothed max using a softmax parametrisation."""
    v = np.log(seed + 1e-30)            # logits for softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 5.0                # initial smoothing
    beta_mul = 1.04

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on the simplex
        conv = _conv_full(a)
        cur_val = _objective_from_conv(conv, n)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Incremental hill‑climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climb(a: np.ndarray,
                n: int,
                deadline: float,
                init_val: float,
                init_seq: np.ndarray,
                max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Large‑scale stochastic hill‑climbing that moves a tiny mass from a donor
    (chosen proportionally to its contribution to the current maximal
    convolution entry) to a uniformly sampled receiver.  Convolution is
    updated exactly in O(n) per move.
    """
    rng = np.random.default_rng()
    conv = _conv_full(a)
    cur_val = init_val
    best_seq = a.copy()
    best_val = cur_val

    # start with a reasonably large step‑scale; it will decay gradually
    step_scale = 0.04

    it = 0
    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break
        it += 1

        # index where convolution attains its maximum
        max_idx = int(np.argmax(conv))

        # contributions of individual a_i to that maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        contrib = a[idx] * a[rev_idx]
        total = contrib.sum()
        if total > 0.0:
            donor = rng.choice(idx, p=contrib / total)
        else:
            donor = rng.integers(0, n)

        # pick a receiver distinct from donor
        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        # maximal transferable amount from donor
        max_transfer = min(a[donor], step_scale * a[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        # ----- incremental convolution update (exact) -----
        new_conv = conv.copy()
        # linear part: 2·delta·(e_receiver ⊛ a) - 2·delta·(e_donor ⊛ a)
        new_conv[receiver:receiver + n] += 2.0 * delta * a
        new_conv[donor:donor + n] -= 2.0 * delta * a
        # quadratic part
        idx2r = 2 * receiver
        if idx2r < new_conv.size:
            new_conv[idx2r] += delta * delta
        idx2d = 2 * donor
        if idx2d < new_conv.size:
            new_conv[idx2d] += delta * delta
        idxdr = donor + receiver
        if idxdr < new_conv.size:
            new_conv[idxdr] -= 2.0 * delta * delta

        new_val = _objective_from_conv(new_conv, n)

        # accept only improving moves (strictly better)
        if new_val < cur_val - 1e-15:
            # apply move
            a[donor] -= delta
            a[receiver] += delta
            conv = new_conv
            cur_val = new_val

            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[HillClimb] n={n:5d} it={it:6d} best={best_val:.6f}")

        # decay step scale very slowly
        step_scale = max(step_scale * 0.9999, 1e-7)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine mixes a very broad initial sampling
    with a massive incremental hill‑climb and a short Adam polishing
    phase, respecting the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve warm‑start from previous runs (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to try – larger n gives more freedom, but also costs more.
    # ------------------------------------------------------------------
    lengths = [8000, 4000, 2000]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to keep the best K candidates (sorted list)
    # ------------------------------------------------------------------
    def _update_top(top, cand_val, cand_seq, keep=5):
        top.append((cand_val, cand_seq))
        top.sort(key=lambda x: x[0])
        if len(top) > keep:
            top.pop()

    rng_global = np.random.default_rng(seed + 9999)

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        top_candidates = []

        # --------------------------------------------------------------
        # 1. Analytic & random seeds
        # --------------------------------------------------------------
        analytic = _make_analytic_seeds(n, previous, rng_global)
        for s in analytic:
            s = np.maximum(s, 0.0)
            s /= s.sum()
            conv = _conv_full(s)
            val = _objective_from_conv(conv, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")
            _update_top(top_candidates, val, s, keep=8)

        # Add a handful of random piecewise‑linear seeds
        while time.time() < deadline - 30.0 and len(top_candidates) < 30:
            cand = _random_piecewise_linear(n, rng_global)
            conv = _conv_full(cand)
            val = _objective_from_conv(conv, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = cand.copy()
                print(f"[rand‑PL] n={n:5d} new global best={val:.6f}")
            _update_top(top_candidates, val, cand, keep=8)

        # --------------------------------------------------------------
        # 2. Incremental hill‑climb on the best few seeds
        # --------------------------------------------------------------
        top_candidates = top_candidates[:5]   # keep only the very best
        for idx, (seed_val, seed_seq) in enumerate(top_candidates):
            if time.time() > deadline:
                break
            print(f"\n--- Hill‑climb start #{idx+1} (init val={seed_val:.6f}) ---")
            a = seed_seq.copy()
            cur_conv = _conv_full(a)
            cur_val = _objective_from_conv(cur_conv, n)

            # allocate a slice of the remaining time for this candidate
            slice_deadline = deadline - (len(top_candidates) - idx - 1) * 5.0
            a_opt, val_opt = _hill_climb(a, n, slice_deadline, cur_val, a)

            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = a_opt.copy()
                print(f"[Hill‑climb] n={n:5d} improved global best={val_opt:.6f}")

            # ----------------------------------------------------------
            # 3. Short Adam polishing (smooth‑max surrogate)
            # ----------------------------------------------------------
            if time.time() < deadline - 5.0:
                seq_adam, val_adam = _adam_optimize(
                    a_opt, n, deadline, max_iters=3000
                )
                if seq_adam is not None and val_adam < global_best_val:
                    global_best_val = val_adam
                    global_best_seq = seq_adam.copy()
                    print(f"[Adam polish] n={n:5d} new global best={val_adam:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that everything failed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective_from_conv(_conv_full(global_best_seq), n)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[verification] sequence rejected – falling back to uniform")
        n = len(global_best_seq)
        global_best_seq = _uniform(n)
        global_best_val = _objective_from_conv(_conv_full(global_best_seq), n)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f}"
        f" (length {len(global_best_seq)})"
    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
