"""
Propose a non‑negative step function (vector of heights) that minimizes

        2·n·max(conv(a,a)) / (∑a)²

The search is a multi‑stage, time‑aware pipeline:

1️⃣  Generate a wide collection of analytic and random seeds (uniform,
    triangular, beta‑shapes, geometric, Dirichlet, sparse …) and keep
    the best one (objective evaluated with a fast FFT‑based convolution).

2️⃣  Deterministic‑annealing Adam on a smoothed version of the max‑convolution.
    The smoothing factor β starts small and is slowly increased while the
    Adam learning‑rate decays.

3️⃣  Refine with a L‑BFGS‑B optimisation of the same smoothed objective.
    Several β‑values are used (0.5 → 200) so that the algorithm moves from a
    convexified problem to the true L∞‑norm.

4️⃣  Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the
    genuine max‑convolution.

5️⃣  Simulated‑annealing hill‑climbing to escape shallow local minima.

6️⃣  A few random “restarts’’ (local Dirichlet reshuffles) followed by a
    short sub‑gradient phase.

All steps respect the supplied time‑budget (default 1000 s).  The best
sequence found is returned as a plain ``list`` of floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast for many calls)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ onto the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) assuming ∑seq == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(np.round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a diverse set of normalised candidate vectors for length ``n``."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.2))
    seeds.append(_sine_window(n))

    # Geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_sym_geom(n, r))

    # Power‑law family
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # Beta‑family grid
    alphas = np.linspace(0.2, 2.0, 10)
    betas = np.linspace(0.2, 2.0, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Sine^γ families
    for gamma in np.linspace(0.5, 4.0, 8):
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
        seeds.append(a / a.sum())

    # Random Dirichlet draws
    for conc in (0.1, 0.3, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns
    for k in (3, 5, 10):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # Previous global best (if compatible length)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (deterministic annealing)
# ----------------------------------------------------------------------


def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation with an ever‑growing smooth‑max."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits (initialising on simplex)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5            # initial smooth‑max temperature
    beta_mul = 1.001
    beta_max = 200.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing (cap at beta_max)
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp over convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k   (linear w.r.t. a)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth objective (for L‑BFGS) and its gradient
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad) where
        f = 2·n·smooth_max(conv(a,a))    with a = softmax(v)
        grad = ∇_v f
    smooth_max is the log‑sum‑exp with temperature beta.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)                     # exact max, serves as numerical stabiliser
    exp_z = np.exp(beta * (conv - shift))    # weights before normalisation
    w_sum = exp_z.sum()
    smoothed_max = shift + np.log(w_sum) / beta
    f = 2.0 * n * smoothed_max

    # normalised weights (probability distribution over the convolution entries)
    w = exp_z / w_sum

    # gradient of weighted sum of convolutions w.r.t. a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax parametrisation
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return f, grad_v


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution; returns best sequence seen."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # Sub‑gradient of the active quadratic form
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # keep numbers stable
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill climbing
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """Stochastic neighbourhood moves with a geometric cooling schedule."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 5)          # 2‑4 coordinates
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    objective.  The routine respects the supplied time budget and returns
    the best sequence found (list of floats).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered best sequence (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – broad seed scan on several lengths
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000]
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Seed scan for length {n} ---")
        seeds = _collect_seeds(n, rng, previous)
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} val={val:.6f}")

    # Fallback (should never happen)
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)

    n = len(global_best_seq)
    print(f"\n=== Beginning refinement (length {n}, init best {global_best_val:.6f}) ===")

    # ------------------------------------------------------------------
    # Stage 2 – Adam with deterministic annealing (more iterations)
    # ------------------------------------------------------------------
    seq_adam, val_adam = _adam_optimize(global_best_seq, n, deadline, max_iters=200000)
    if seq_adam is not None and val_adam < global_best_val:
        global_best_seq = seq_adam.copy()
        global_best_val = val_adam
        print(f"[overall‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – L‑BFGS on the smoothed max (multiple β values)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import minimize
    except Exception as exc:
        print("Scipy not available – skipping L‑BFGS stage.")
        minimize = None

    if minimize is not None:
        eps = 1e-30
        v = np.log(global_best_seq + eps)

        beta_schedule = [0.5, 2.0, 10.0, 50.0, 200.0]
        for beta in beta_schedule:
            if time.time() > deadline:
                break
            # allocate a modest number of iterations; each iteration is cheap
            remaining = deadline - time.time()
            max_iter = int(min(1500, max(10, remaining * 30)))   # heuristic
            if max_iter <= 0:
                break
            print(f"[L‑BFGS] beta={beta:.1f}, max_iter={max_iter}")
            try:
                res = minimize(
                    _smooth_obj_and_grad,
                    v,
                    args=(n, beta),
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": max_iter, "disp": False},
                )
            except Exception as exc2:
                print(f"L‑BFGS failed at beta={beta}: {exc2}")
                continue
            v = res.x
            a = _softmax(v)
            val = _objective(a)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = a.copy()
                print(f"[L‑BFGS] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – exact sub‑gradient mirror descent
    # ------------------------------------------------------------------
    seq_sub, val_sub = _subgradient_refine(global_best_seq, n, deadline, max_iters=300000)
    if val_sub < global_best_val:
        global_best_seq = seq_sub.copy()
        global_best_val = val_sub
        print(f"[overall‑SubGrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline, max_iters=300000)
    if val_sa < global_best_val:
        global_best_seq = seq_sa.copy()
        global_best_val = val_sa
        print(f"[overall‑SA] improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – a few cheap random restarts to escape any remaining trap
    # ------------------------------------------------------------------
    for restart in range(3):
        if time.time() > deadline:
            break
        cur = global_best_seq.copy()
        rng_local = np.random.default_rng(seed + restart + 9999)
        k = max(2, int(0.05 * n))
        idx = rng_local.choice(n, size=k, replace=False)
        tot = cur[idx].sum()
        if tot > 0.0:
            cur[idx] = rng_local.dirichlet(np.full(k, 1.0)) * tot
        seq_r, val_r = _subgradient_refine(cur, n, deadline, max_iters=150000)
        if val_r < global_best_val:
            global_best_seq = seq_r.copy()
            global_best_val = val_r
            print(f"[restart {restart}] improved to {val_r:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
