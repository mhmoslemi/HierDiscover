"""
Propose a step‑function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm builds on the best known sequence (available as the global
variable ``height_sequence_1``).  It then iteratively reduces the peak of
the autocorrelation by solving tiny linear programmes that maximise the
total mass while keeping the cross‑correlation with the current iterate
bounded by the current peak.  After each LP a convex combination with the
new direction is attempted; the step size is back‑tracked until the peak
does not increase.  When the LP‑based phase stalls we fall back to a
few thousand steps of exponentiated‑gradient (mirror‑descent) refinement.
All work respects the supplied time budget and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP + sparse matrices)
# ----------------------------------------------------------------------
try:
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# LP utilities – build the convolution constraints for a set of indices
# ----------------------------------------------------------------------


def _build_active_matrix(f: np.ndarray, active_idx: np.ndarray) -> csr_matrix:
    """
    Construct a sparse matrix A such that (A @ b)_k = (f * b)[active_idx[k]].
    Each row corresponds to one convolution index from ``active_idx``.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    for row, k in enumerate(active_idx):
        # indices i where 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals                 # column indices for b
        rows.extend([row] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())

    return csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))


def _solve_lp_max_sum(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve  max  Σ b_j   s.t.   b ≥ 0  and  (f * b)[active_idx] ≤ M.
    Returns the optimal vector b (may be None if the solver fails).
    """
    if not SCIPY_AVAILABLE:
        return None

    A = _build_active_matrix(f, active_idx)
    c = -np.ones_like(f)                     # minimise -Σb  ⇔  maximise Σb
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    bounds = [(0, None) for _ in range(f.shape[0])]

    # Highs works with sparse matrices and is fast for this size
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    b = np.asarray(res.x, dtype=np.float64)
    # numerical safety
    b = np.maximum(b, 0.0)
    return b


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror‑descent (exponentiated‑gradient) – fine tune
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray, deadline: float,
                        max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    n = start_seq.size
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val
