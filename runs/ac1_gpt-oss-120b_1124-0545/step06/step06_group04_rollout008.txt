"""
Step‑function optimizer with a dedicated “flatten‑convolution” stage.

The search proceeds in several phases for each chosen length n:

1. **Seed generation** – a diverse pool (uniform, geometric, power‑law,
   Beta, Dirichlet, mirrored copies, up‑scaled previous best, plus a few
   extreme‑parameter families).

2. **Flattening stage** – a simple exponentiated‑gradient loop that
   minimises the L₂ distance between the self‑convolution and the
   constant value 1/(2n‑1).  This drives the convolution towards a flat
   profile, reducing the ratio max(conv)/mean(conv) and therefore the
   objective 2·n·max(conv).

3. **Smoothed‑max Adam** – a short Adam run on the log‑sum‑exp surrogate
   of the max‑convolution (the same routine used in the reference
   solution, but with a higher β‑target and more iterations).

4. **Exact sub‑gradient refinement** – exponentiated‑gradient mirror
   descent on the true max‑convolution entry.

5. **Simulated‑annealing fine‑tuning** – cheap Metropolis moves that
   swap tiny mass fragments.

6. **Symmetrisation** – averaging the sequence with its mirror often
   reduces the peak further.

All phases respect the overall 1000 s budget; the best sequence found
is cached globally as ``height_sequence_1`` for future calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for many iterations)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sin2_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _hanning(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)
    a[a < 0] = 0.0
    return a / a.sum()


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) on (0,1) – interior points avoid infinities."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse pool of start vectors (already normalised)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_sin2_window(n))
    seeds.append(_hanning(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_tent(n))

    # beta families (symmetric & slightly asymmetric)
    for a in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        seeds.append(_beta_seed(n, a, a))
        seeds.append(_beta_seed(n, a, a * 1.5))
        seeds.append(_beta_seed(n, a * 1.5, a))

    # Dirichlet draws
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrored copies
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # upscale previous best if it exists and length differs
    if previous is not None and previous.shape != (n,):
        try:
            up = _upscale(previous, n)
            seeds.append(up)
        except Exception:
            pass
    elif previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # ensure we have at least one seed
    if not seeds:
        seeds.append(_uniform(n))

    return seeds

                   beta_target: float = 120.0,
                   beta_mul: float = 1.001,
                   lr0: float = 0.5) -> tuple[np.ndarray, float]:
    v = np.log(seed + 1e-30)                # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        shift = _conv_fft(a, a).max()
        w = np.exp(beta * (_conv_fft(a, a) - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(w)
            w[_conv_fft(a, a).argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 100000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (cheap Metropolis‑Hastings)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # move ≤2.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# New stage: flattening the convolution (L2 toward constant)
# ----------------------------------------------------------------------


def _flatten_optimize_and_evaluate(seed: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   max_iters: int = 800,
                                   eta0: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient loop that drives a*a toward a flat profile.
    Returns the final sequence and its true objective.
    """
    a = seed.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    const = 1.0 / (2 * n - 1)          # target value of each convolution entry

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)               # current self‑convolution
        err = conv - const                    # deviation from flatness
        # gradient of L2(err) w.r.t. a : 4 * (err ⋆ a)  (⋆ = correlation)
        grad_full = _conv_fft(err, a[::-1])
        grad = 4.0 * grad_full[n - 1:2 * n - 1]

        eta = eta0 / np.sqrt(it)             # decreasing step
        log_a = np.log(a + 1e-30) - eta * grad
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    val = _objective(a)
    return a, val


# ----------------------------------------------------------------------
# Main entry point – respects the 1000 s budget
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence (list of floats) found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # reuse a previously discovered construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # lengths to try – start modest, enlarge if time permits
    lengths = [4000, 8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # Build a rich seed pool (including a few extreme families)
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # add a few extra extreme Beta and Dirichlet draws
        for a in [0.1, 0.05, 0.02]:
            seeds.append(_beta_seed(n, a, a))
        rng = np.random.default_rng(999)
        for alpha in [0.02, 0.05]:
            seeds.append(_dirichlet_seed(n, alpha, rng))

        # ------------------------------------------------------------------
        # Quick evaluation of raw seeds, keep the top few
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:6]]

        # keep best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 0 – flattening stage (L2‑to‑constant)
        # ------------------------------------------------------------------
        for idx, s in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _flatten_optimize_and_evaluate(
                s, n, deadline, max_iters=800, eta0=2.0
            )
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[flatten] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – short Adam optimisation from the current best
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _adam_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=8000,
                beta0=2.0,
                beta_target=300.0,
                beta_mul=1.001,
                lr0=0.7,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – exact sub‑gradient refinement (longer)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=120000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(
                global_best_seq, n, deadline, max_iters=None
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Symmetrisation – averaging with its mirror often reduces peak
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[symmetrisation] n={n} improved to {sym_val:.6f}")

        # propagate the current best forward (for up‑scaling in the next length)
        previous = global_best_seq

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # cache for the next call
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
