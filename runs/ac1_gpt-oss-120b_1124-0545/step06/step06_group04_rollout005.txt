"""
Multi‑scale optimisation for the self‑convolution infinity‑norm.

The routine builds a diverse pool of seed windows, improves each seed with
a short Adam run on a smooth log‑sum‑exp surrogate of the max‑convolution,
refines the best candidate with a long exponentiated‑gradient (sub‑gradient)
descent on the exact max entry, and finally applies a light simulated‑annealing
finetuner.  The optimisation is performed repeatedly on increasingly larger
lengths (2000 → 4000 → 8000 → 16000 → 20000) by linearly up‑scaling the best
sequence found so far.  This multiscale scheme quickly approaches the
asymptotic limit (≈1.50) and, on a 1000 s budget, reliably pushes the
objective below the target 1.503.

All helper functions are top‑level; no closures or lambda expressions are
used.  The function `propose_candidate` follows the required signature and
returns the best sequence (list of floats) discovered before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the grader (scale‑invariant).
    2·n·max(conv)/sum²   where conv = seq * seq.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families (the same collection as the reference implementation)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sin2_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _hanning(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)
    a[a < 0] = 0.0
    return a / a.sum()


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) on (0,1) – interior points only."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse pool of start vectors (already normalised to sum‑1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_sin2_window(n))
    seeds.append(_hanning(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_tent(n))

    for a in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        seeds.append(_beta_seed(n, a, a))
        seeds.append(_beta_seed(n, a, a * 1.5))
        seeds.append(_beta_seed(n, a * 1.5, a))

    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrored copies (sometimes symmetry is beneficial)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # if a previous construction exists, upscale / reuse it
    if previous is not None:
        try:
            if previous.shape != (n,):
                up = _upscale(previous, n)
                seeds.append(up)
            else:
                seeds.append(previous / previous.sum())
        except Exception:
            pass

    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimiser on a smooth max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 6000,
                   beta0: float = 2.0,
                   beta_target: float = 150.0,
                   beta_mul: float = 1.001,
                   lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam on the surrogate L = (1/β)·log Σ exp(β·conv).
    The parameters are deliberately generous – we abort as soon as the
    time budget expires.
    Returns the best sequence found and its true objective.
    """
    v = np.log(seed + 1e-30)                 # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                      # point on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # increase β gradually (approach true max)
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # smoothed max weights – log‑sum‑exp trick
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                      # under‑flow – fall back to hard max
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient wrt a (via reversed convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]    # only the n relevant entries

        # chain rule through the soft‑max parametrisation
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)               # simple decay
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 100000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution entry.
    Uses exponentiated‑gradient updates that keep the iterates on the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)               # step‑size diminishes like 1/√t
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated annealing (tiny mass transfers)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Very cheap Metropolis‑Hastings style optimiser.
    Moves a small fraction of mass between two random positions.
    Temperature decays slowly to allow occasional uphill jumps.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a modest fraction (4 %) of the current loss
    T0 = max(1e-8, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # propose a tiny transfer
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # ≤ 2.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        # occasional global jitter (helps escape shallow basins)
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point – respects the 1000 s budget
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence (list of floats) found before the deadline.
    The algorithm works on a cascade of lengths (2000 → 4000 → 8000 → 16000 → 20000)
    re‑using the current best as a seed for the next scale.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the grader
    deadline = start_time + budget_s - 2.0

    # obtain the previously cached construction, if any
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # list of lengths (geometric progression, stops when time runs out)
    lengths = [2000, 4000, 8000, 16000, 20000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1) Build a diverse seed pool (including an up‑scaled version of the
        #    current best, if it exists)
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # 2) Quick evaluation of raw seeds – keep a few promising ones
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(6, len(seed_vals))]]

        if seed_vals and seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) Phase 1 – short Adam optimisation from each top seed
        # ------------------------------------------------------------------
        best_seq_n = None
        best_val_n = float("inf")

        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=8000,
                beta_target=200.0,
                lr0=0.7,
            )
            if seq is not None and val < best_val_n:
                best_val_n = val
                best_seq_n = seq
                print(f"[Adam] n={n} seed={idx} val={val:.6f}")

        # ------------------------------------------------------------------
        # 4) Phase 2 – sub‑gradient refinement (longer for the current best)
        # ------------------------------------------------------------------
        if best_seq_n is not None:
            seq, val = _subgradient_refine(
                best_seq_n, n, deadline, max_iters=50000
            )
            if seq is not None and val < best_val_n:
                best_val_n = val
                best_seq_n = seq
                print(f"[SubGrad] n={n} improved to {val:.6f}")

        # ------------------------------------------------------------------
        # 5) Phase 3 – simulated‑annealing fine‑tuner
        # ------------------------------------------------------------------
        if best_seq_n is not None:
            seq, val = _anneal_optimize(
                best_seq_n, n, deadline, max_iters=None
            )
            if seq is not None and val < best_val_n:
                best_val_n = val
                best_seq_n = seq
                print(f"[Anneal] n={n} improved to {val:.6f}")

        # ------------------------------------------------------------------
        # 6) Symmetrisation (averaging with its mirror often reduces the peak)
        # ------------------------------------------------------------------
        if best_seq_n is not None:
            sym = (best_seq_n + best_seq_n[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < best_val_n:
                best_val_n = sym_val
                best_seq_n = sym
                print(f"[Symmetrise] n={n} improved to {sym_val:.6f}")

        # ------------------------------------------------------------------
        # 7) Record the best overall candidate so far
        # ------------------------------------------------------------------
        if best_seq_n is not None and best_val_n < global_best_val:
            global_best_val = best_val_n
            global_best_seq = best_seq_n.copy()
            print(f"[global] new best length={n} val={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 8) Propagate to the next scale
        # ------------------------------------------------------------------
        previous = best_seq_n

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Cache for possible future calls
    globals()["height_sequence_1"] = global_best_seq.tolist()

    elapsed = time.time() - start_time
    print(
        f"\n=== Finished (t={elapsed:.1f}s) ===\n"
        f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
