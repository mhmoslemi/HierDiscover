      * uniform,
      * arcsine family a_i ∝ ((i+½)*(n‑½‑i))^p   (p ∈ {‑2,…,‑0.5}),
      * beta‑type family a_i ∝ (i+1)^α (n‑i)^β   (α,β ∈ {‑2,‑1.5,‑1,‑0.5,0}),
      * symmetric Gaussian and sine‑window,
      * several Dirichlet draws (different sparsity levels),
      * the previous best construction (if available).
   All seeds are normalised to L1‑norm = 1 and both the seed and its mirror
   are stored.
3. Evaluate all seeds with the exact objective and keep the `k_best`
   (default 5) for further optimisation.
4. **Adam on a smooth surrogate** – minimise  

        L(a) = (1/β)·log ∑ₖ exp(β·convₖ(a)) + λ·∑ₖ convₖ(a)²  

   where `conv(a,a)` denotes the linear convolution. The log‑sum‑exp term
   approximates the max‑convolution while the quadratic term penalises
   variance, encouraging a flatter convolution.  The vector `a` is kept on
   the probability simplex via a log‑softmax parametrisation.
   Smoothing (`β`) starts small (0.5) and is multiplied by 1.04 each step,
   and the variance weight (`λ`) starts large (5.0) and decays slowly.
5. The best result of the Adam runs is refined by **exact sub‑gradient
   mirror‑descent** on the true objective (max‑convolution).  The step
   size follows the classic 1/√t schedule.
6. A short **Metropolis–Hastings anneal** moves tiny fractions of mass
   between random positions and occasionally adds a global jitter.  This
   can escape shallow local minima.
7. The overall best sequence seen is returned as a plain list of floats.
   All numbers are non‑negative and summed to 1, which is scale‑invariant
   for the evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full length).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq))²
    The sequence is assumed non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """
    Discrete arcsine family:
        a_i ∝ ((i+0.5)*(n-0.5-i))**exponent   (exponent < 0)
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide="ignore", invalid="ignore"):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _betaseed(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑type family:
        a_i ∝ (i+1)^α * (n-i)^β
    Allows asymmetric shapes.  α,β may be negative.
    """
    i = np.arange(1, n + 1, dtype=np.float64)      # i = 1..n
    j = np.arange(n, 0, -1, dtype=np.float64)     # j = n..1
    with np.errstate(divide="ignore", invalid="ignore"):
        a = (i ** alpha) * (j ** beta)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.20) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Gamma(α,1) draws normalised – gives sparse (α<1) to dense (α>1)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    s = vec.sum()
    if s == 0.0:
        return _uniform(n)
    return vec / s


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Assemble a diverse collection of start‑vectors, all normalised to sum = 1.
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic analytic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # arcsine exponents (more extreme than the previous code)
    for exp in [-2.0, -1.75, -1.5, -1.25, -1.0, -0.75, -0.5]:
        seeds.append(_arcseed(n, exp))

    # beta‑type family – both symmetric and asymmetric
    alphas = [-2.0, -1.5, -1.0, -0.5, 0.0]
    betas = [-2.0, -1.5, -1.0, -0.5, 0.0]
    for a in alphas:
        for b in betas:
            seeds.append(_betaseed(n, a, b))

    # Dirichlet draws – from very sparse to very dense
    for alpha in [0.05, 0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # Mirror each seed (helps with asymmetric optima)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # Re‑use the previous best construction, if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / max(previous.sum(), 1e-12))

    # guarantee non‑empty list
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation (smooth max + variance regulariser)
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
    beta0: float = 0.5,
    beta_mul: float = 1.04,
    lr0: float = 1.0,
    lambda0: float = 5.0,
    lambda_decay: float = 0.997,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the loss
        L(a) = (1/β)·log∑ₖ exp(β·convₖ(a)) + λ·∑ₖ convₖ(a)² .
    The vector `a` lives on the probability simplex via a log‑softmax.
    Returns the best sequence encountered and its true objective.
    """
    v = np.log(seed + 1e-30)        # logits for softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = np.inf
    beta = beta0
    lam = lambda0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # point on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # update smoothing and variance weight
        beta *= beta_mul
        lam *= lambda_decay

        # smoothed max term (log‑sum‑exp)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of smoothed max term
        rev_a = a[::-1]
        full_max = _conv_fft(rev_a, w)
        grad_max = 2.0 * full_max[n - 1 : n - 1 + n]

        # gradient of variance regulariser (‖conv‖²)
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n - 1 : n - 1 + n]

        grad_a = grad_max + lam * grad_sq

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 120000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient of
    max(conv).  Returns the best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update (mirror descent on the simplex)
        lr = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)            # prevent overflow
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis–Hastings walk that moves a tiny fraction of mass
    between two random positions, with exponential cooling.
    Occasionally performs a global Gaussian jitter.
    Returns the best sequence discovered.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-8, cur_val * 0.04)   # initial temperature
    decay = 0.9994

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # choose two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a tiny fraction of mass from i to j
        delta = rng.random() * 0.02 * cur[i]   # ≤2 % of the source
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # periodic global jitter (every 1500 steps)
        if it % 1500 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found
    as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Use a previously discovered construction if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0:
                previous = previous / max(previous.sum(), 1e-12)
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length that fits the remaining budget.
    # 20 000 works well on most platforms; if we run out of time we fall back.
    # ------------------------------------------------------------------
    candidate_lengths = [20000, 16000, 12000, 8000, 4000]
    global_best_seq = None
    global_best_val = float("inf")

    for n in candidate_lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # 1) Assemble seeds
        seeds = _make_seeds(n, previous)

        # 2) Quick evaluation of seeds; keep the most promising
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_k = min(6, len(seed_vals))
        top_seeds = [s for s, _ in seed_vals[:top_k]]

        # Record best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation (smooth max + variance)
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=25000,
                beta0=0.5,
                beta_mul=1.04,
                lr0=1.0,
                lambda0=5.0,
                lambda_decay=0.997,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=120000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # Keep the best construction for possible reuse at a different length
        previous = global_best_seq

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
