3. **Fine tuning** –  
   For each length we run a few Adam iterations on a smooth surrogate,
   an exact sub‑gradient mirror‑descent, and a cheap Metropolis‑Hastings
   annealing.  All steps respect the global time budget (≈1000 s).

The routine returns the best sequence found (as a plain list of floats) and
guarantees non‑negative entries.  The objective is scale‑invariant, therefore
all vectors are kept normalised to unit sum.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))², seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families (used for the initial small‑scale optimisation)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """
    Discrete analogue of the continuous arcsine law.
    a_i ∝ ((i+0.5)*(n‑0.5‑i))**exponent  (exponent < 0).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide='ignore', invalid='ignore'):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        return _uniform(n)
    return a / s


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse set of start‑vectors (already normalised to sum = 1)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n))

    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    for exp in [-1.0, -0.75, -0.5, -0.25]:
        try:
            seeds.append(_arcseed(n, exp))
        except Exception:
            pass

    # mirrored copies (helps when optimum is asymmetric)
    seeds.extend([s[::-1].copy() for s in seeds])

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / max(previous.sum(), 1e-12))

    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smooth surrogate of max‑convolution
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000,
                   beta0: float = 2.0,
                   beta_mul: float = 1.04,
                   lr0: float = 0.9,
                   lambda0: float = 0.4,
                   lambda_decay: float = 0.995) -> tuple[np.ndarray, float]:
    """Adam on L(a)=log‑sum‑exp(conv(a,a)) + λ‖conv‖² (softmax re‑param)."""
    v = np.log(seed + 1e-30)          # logits for the simplex
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0
    lam = lambda0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                # point on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # increase smoothing
        beta *= beta_mul
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of smoothed‑max term
        rev_a = a[::-1]
        full_max = _conv_fft(rev_a, w)
        grad_max = 2.0 * full_max[n - 1:n - 1 + n]

        # gradient of variance regulariser
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n - 1:n - 1 + n]

        grad_a = grad_max + lam * grad_sq

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        lam *= lambda_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent using the true sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# cheap Metropolis‑Hastings annealing
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Simple MH walk that swaps tiny masses between random positions."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a tiny fraction of mass from i to j
        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Linear‑programming boost (max‑sum under max‑convolution bound)
# ----------------------------------------------------------------------


def _lp_boost(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g   s.t.   conv(f,g) ≤ M,   g ≥ 0.
    Returns the optimised vector g (unnormalised).  If the LP fails,
    returns None.
    """
    import scipy.optimize as opt

    n = f.size
    L = 2 * n - 1

    # Build dense convolution matrix A where A @ g = conv(f,g)
    # A[i, j] = f[i-j]  if 0 ≤ i-j < n else 0
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]

    c = -np.ones(n)                 # maximise Σ g  →  minimise -Σ g
    b = np.full(L, M)
    bounds = [(0.0, None)] * n

    try:
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception as e:
        print("[LP] linprog raised:", e)
        return None

    if not res.success:
        # The LP can become unbounded if f contains exact zeros;
        # this should not happen because we keep a tiny epsilon in f.
        print("[LP] failed:", res.message)
        return None
    return res.x


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the grader
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Optional reuse of a previously discovered construction
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0:
                previous = previous / max(previous.sum(), 1e-12)
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – small‑scale LP boost (n≈2000)
    # ------------------------------------------------------------------
    n_small = 2000
    print(f"\n=== Stage 1: LP boost on length {n_small} ===")
    seeds_small = _make_seeds(n_small, previous)
    # quick screening – keep a few promising seeds
    seed_vals = [(s, _objective(s)) for s in seeds_small]
    seed_vals.sort(key=lambda x: x[1])
    top_seeds = [s for s, _ in seed_vals[:5]]

    best_seq = None
    best_val = float("inf")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(
            seed_vec,
            n_small,
            deadline,
            max_iters=5000,
            beta0=0.5,
            beta_mul=1.03,
            lr0=1.0,
            lambda0=1.0,
            lambda_decay=0.995,
        )
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Stage 1‑Adam] seed={idx} best={val:.6f}")

    if best_seq is None:
        # fallback to uniform
        best_seq = _uniform(n_small)
        best_val = _objective(best_seq)
        print("[Stage 1] fallback to uniform")

    # Ensure numerical positivity (avoid LP unboundedness)
    eps = 1e-12
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()

    # LP iterative refinement
    f = best_seq.copy()
    for it in range(6):                         # a handful of LP passes
        if time.time() > deadline:
            break
        conv_ff = _conv_fft(f, f)
        M = conv_ff.max()
        g = _lp_boost(f, M)
        if g is None:
            break
        # normalise the LP solution to the simplex
        g = np.maximum(g, eps)
        g /= g.sum()
        # replace f (full step) – this is provably non‑increasing for M
        f = g
        val = _objective(f)
        if val < best_val:
            best_val = val
            best_seq = f.copy()
            print(f"[LP‑boost] it={it} best={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – hierarchical up‑sampling and fine‑tuning
    # ------------------------------------------------------------------
    current_seq = best_seq
    lengths = [4000, 8000, 12000]   # increase only if time permits
    for n_target in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Stage 2: up‑sample to length {n_target} ===")
        # up‑sample by integer repeat where possible, else linear interpolation
        factor = n_target // current_seq.size
        if factor > 1 and current_seq.size * factor == n_target:
            up_seq = np.repeat(current_seq, factor)
        else:
            x_old = np.linspace(0.0, 1.0, current_seq.size)
            x_new = np.linspace(0.0, 1.0, n_target)
            up_seq = np.interp(x_new, x_old, current_seq)

        up_seq = np.maximum(up_seq, eps)
        up_seq /= up_seq.sum()

        # Phase A – Adam on smooth surrogate
        up_seq, val = _adam_optimize(
            up_seq,
            n_target,
            deadline,
            max_iters=8000,
            beta0=0.5,
            beta_mul=1.02,
            lr0=1.0,
            lambda0=2.0,
            lambda_decay=0.997,
        )
        if up_seq is not None and val < best_val:
            best_val = val
            current_seq = up_seq.copy()
            print(f"[Stage 2‑Adam] n={n_target} best={val:.6f}")

        # Phase B – exact sub‑gradient mirror descent
        up_seq, val = _subgradient_refine(current_seq, n_target, deadline, max_iters=40000)
        if up_seq is not None and val < best_val:
            best_val = val
            current_seq = up_seq.copy()
            print(f"[Stage 2‑SubGrad] n={n_target} best={val:.6f}")

        # Phase C – cheap annealing
        up_seq, val = _anneal_optimize(current_seq, n_target, deadline, max_iters=20000)
        if up_seq is not None and val < best_val:
            best_val = val
            current_seq = up_seq.copy()
            print(f"[Stage 2‑Anneal] n={n_target} best={val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check / fallback
    # ------------------------------------------------------------------
    if current_seq is None:
        current_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(current_seq)})"
    )
    # Return a plain Python list as required by the grader
    return current_seq.tolist()
