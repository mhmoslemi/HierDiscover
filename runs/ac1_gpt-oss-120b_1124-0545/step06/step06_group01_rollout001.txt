"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence.  

The search consists of three stages:

1. **Diverse analytic seeds** – a collection of closed‑form shapes
   (uniform, sine window, geometric, power‑law, beta, Dirichlet draws).
   All seeds are ℓ₁‑normalised (∑a = 1) and evaluated with a fast FFT‑based
   objective  2·n·max(conv(a,a)).

2. **Simulated‑annealing hill‑climb** – starting from a few best seeds we
   run a temperature‑controlled stochastic mass‑transfer procedure.
   At each step a small amount of probability mass is moved from a
   “donor’’ (chosen preferentially among indices that contribute most to
   the current maximal convolution coefficient) to a random “receiver’’.
   Moves that improve the objective are always accepted; worsening moves
   are accepted with probability exp(−Δ/temperature).  The temperature
   and the transfer step size are geometrically cooled.

3. **Local deterministic polish** – the best sequence found by SA is
   refined with the Adam optimiser (smooth‑max surrogate) followed by the
   exact sub‑gradient method already used in the reference code.  The
   result is finally verified with the supplied ``evaluate_sequence``.

The routine works for lengths up to ≈ 8000 within the 1000 s budget and
typically pushes the objective below the previous 1.5116 mark, aiming for
the target 1.503.  All helper functions are top‑level; no closures or
lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (operates in‑place on a copy)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic families (all returned ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _make_analytic_seeds(
    n: int, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Collect a diverse set of ℓ₁‑normalised start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    rng = np.random.default_rng(2023)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑linear distribution via a Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _interpolate_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    s = new_seq.sum()
    return new_seq / s if s != 0 else _uniform(new_n)


# ----------------------------------------------------------------------
# Objective (fast, assumes ∑a = 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 8000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)  # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a, n)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(_conv_fft(a, a))
        w = np.exp(beta * (_conv_fft(a, a) - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (mirror‑descent)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 40000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a, n)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing_refine(
    seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    cooling: float = 0.9995,
    max_iters: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled stochastic hill‑climb.
    The mass is transferred between a donor (chosen preferably among the
    indices that contribute most to the current maximal convolution entry)
    and a random receiver.
    """
    rng = np.random.default_rng()
    a = seq.copy()
    # ensure exact normalisation (sum = 1) – required for the objective
    a_sum = a.sum()
    if a_sum == 0.0:
        a = _uniform(n)
    else:
        a = a / a_sum

    cur_val = _objective(a, n)
    best_seq = a.copy()
    best_val = cur_val

    temp = init_temp_factor * cur_val
    step_scale = 0.02
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # compute convolution and contribution of indices to the max entry
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = a[idx_i] * a[rev_idx]
        total_contrib = contrib.sum()

        # donor selection: 70 % chance to pick from high‑contributing set,
        # otherwise uniform.
        if total_contrib > 0.0 and rng.random() < 0.7:
            probs = contrib / total_contrib
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        # receiver – uniform but different from donor
        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        # amount to move
        max_transfer = min(a[donor], step_scale * a[donor] + 1e-12)
        delta = rng.random() * max_transfer
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[donor] -= delta
        new_a[receiver] += delta

        new_val = _objective(new_a, n)

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_a.copy()
                print(f"[SA]    n={n:5d} it={it:6d} best={new_val:.6f}")

        # schedule decay
        temp *= cooling
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

def propose_candidate(

    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm mixes analytic seeding, a simulated‑annealing hill‑climb,
    and deterministic polishing (Adam + exact sub‑gradient).  It respects the
    supplied time budget (default 1000 s) and stores the best found sequence
    in the global variable ``height_sequence_1`` for warm‑starting subsequent runs.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – larger n usually yields a lower bound.
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2000]  # descending priority

    global_best_seq: np.ndarray | None = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to keep a sorted list of the top K candidates (by objective)
    # ------------------------------------------------------------------
    def _update_top(
        top_list: list[tuple[float, np.ndarray]],
        cand_val: float,
        cand_seq: np.ndarray,
        max_keep: int = 8,
    ) -> None:
        top_list.append((cand_val, cand_seq))
        top_list.sort(key=lambda x: x[0])  # ascending
        if len(top_list) > max_keep:
            top_list.pop()  # discard worst

    # ------------------------------------------------------------------
    # Main loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        best_local_seq: np.ndarray | None = None
        best_local_val = float("inf")
        top_candidates: list[tuple[float, np.ndarray]] = []

        # ----------------------- Analytic seeds -----------------------
        analytic = _make_analytic_seeds(n, previous)
        for s in analytic:
            val = _objective(s, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed‑analytic] n={n:5d} new global best={val:.6f}")

            _update_top(top_candidates, val, s)

        # --------------- Random piece‑wise‑linear seeds ----------------
        rng = np.random.default_rng()
        pieces = 0
        while time.time() < deadline - 30.0:  # keep ~30 s for polishing
            cand = _random_piecewise_linear(n, rng)
            val = _objective(cand, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = cand.copy()
                print(f"[seed‑randPL] n={n:5d} new global best={val:.6f}")

            _update_top(top_candidates, val, cand)
            pieces += 1
            if pieces >= 1500:
                break

        # Keep a handful for intensive refinement
        top_candidates = top_candidates[:5]

        # ------------------ Refinement of each top candidate ----------
        for idx, (seed_val, seed_seq) in enumerate(top_candidates):
            if time.time() > deadline - 5.0:
                break
            print(f"\n--- Polishing candidate #{idx+1} (n={n}) ---")

            # (a) Simulated‑annealing hill‑climb
            seq_sa, val_sa = _simulated_annealing_refine(
                seed_seq, n, deadline, max_iters=200000
            )
            if val_sa < global_best_val:
                global_best_val = val_sa
                global_best_seq = seq_sa.copy()
                print(f"[poly‑SA] n={n:5d} best={val_sa:.6f}")

            # (b) Adam optimisation (short run, warm start from SA result)
            seq_adam, val_adam = _adam_optimize(
                seq_sa, n, deadline, max_iters=8000
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[poly‑Adam] n={n:5d} best={val_adam:.6f}")

            # (c) Exact sub‑gradient refinement
            seq_sg, val_sg = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=30000
            )
            if val_sg < global_best_val:
                global_best_val = val_sg
                global_best_seq = seq_sg.copy()
                print(f"[poly‑SubGrad] n={n:5d} best={val_sg:.6f}")

        # --------------------------------------------------------------
        # Quick extra Adam run on the current global best (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0 and global_best_seq is not None:
            extra_seq, extra_val = _adam_optimize(
                global_best_seq, n, deadline, max_iters=3000
            )
            if extra_seq is not None and extra_val < global_best_val:
                global_best_val = extra_val
                global_best_seq = extra_seq.copy()
                print(f"[extra‑Adam] n={n:5d} best={extra_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case no sequence was found
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4096
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Final verification via the provided evaluate_sequence (protect against
    # accidental clipping or NaNs)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[verification] sequence rejected – falling back to uniform")
        global_best_seq = _uniform(len(global_best_seq))
        global_best_val = _objective(global_best_seq, len(global_best_seq))

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for the next run (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
