"""
Optimiser for the self‑convolution infinity‑norm.

The algorithm proceeds in stages of increasing resolution.
A modest initial length (≈2000) is explored with a rich pool of
parameterised seed families (including low‑α Beta‑distributions).
The best seed is refined by Adam on a smooth max‑surrogate,
then by exponentiated‑gradient sub‑gradient descent on the true
∞‑norm.  The obtained distribution is then up‑scaled (linear
interpolation) to larger lengths (4000 → 8000 → 12000) and the
refinement steps are repeated.  Finally a brief simulated‑annealing
pass and a symmetry‑averaging step are applied.  The whole search
honours the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the grader (scale‑invariant).
    The sequence is assumed to be non‑negative and already normalised
    to sum = 1.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    n = seq.size
    return 2.0 * n * conv.max() / (s * s)

# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sin2_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _hanning(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)
    a[a < 0] = 0.0
    return a / a.sum()


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta(α,β) on (0,1)."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse pool of start vectors (already normalised)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))
    seeds.append(_sin2_window(n))
    seeds.append(_hanning(n))
    seeds.append(_gaussian(n, sigma_factor=0.15))
    seeds.append(_tent(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Beta families – include very “U‑shaped” distributions
    alphas = [0.1, 0.15, 0.2, 0.25, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    for a in alphas:
        seeds.append(_beta_seed(n, a, a))                # symmetric
        seeds.append(_beta_seed(n, a, a * 1.5))          # a little skewed
        seeds.append(_beta_seed(n, a * 1.5, a))

    # Dirichlet draws (including very spiky ones)
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrored copies (ensures symmetry exploration)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # upscale previous best if provided
    if previous is not None:
        try:
            up = _upscale(previous, n)
            seeds.append(up)
        except Exception:
            pass

    # ensure at least one seed
    if not seeds:
        seeds.append(_uniform(n))

    return seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 4000,
                   beta0: float = 2.0,
                   beta_target: float = 800.0,
                   beta_mul: float = 1.001,
                   lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed max loss L = (1/β)·log Σ exp(β·conv).
    Returns the best sequence (as a probability vector) and its true objective.
    """
    v = np.log(seed + 1e-30)            # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # point on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # increase β slowly, never exceeding the target
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # smoothed‑max weights via log‑sum‑exp trick
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of L wrt a (via reversed convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # Adam updates on the logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact ‖conv‖∞.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Lightweight simulated‑annealing fine‑tuner."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # ≤2.5 % move
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            cand = cur + noise
            cand = np.clip(cand, 0.0, None)
            if cand.sum() == 0.0:
                cand = _uniform(n)
            else:
                cand /= cand.sum()
            cand_val = _objective(cand)
            if cand_val < cur_val or rng.random() < np.exp(-(cand_val - cur_val) / (T + 1e-12)):
                cur = cand
                cur_val = cand_val
                if cand_val < best_val:
                    best_val = cand_val
                    best = cand.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={cand_val:.6f}")

    return best, best_val


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Average with its mirror, re‑normalise."""
    sym = (seq + seq[::-1]) * 0.5
    if sym.sum() == 0:
        return seq
    sym /= sym.sum()
    return sym

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence (list of floats) found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # Load previously discovered construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            previous = previous / previous.sum()
        except Exception:
            previous = None

    # --------------------------------------------------------------
    # Stage schedule – start small, then upscale
    # --------------------------------------------------------------
    # lengths are chosen such that the total work fits comfortably into the budget.
    stage_lengths = [2000, 4000, 8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in stage_lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Stage length {n} ===")

        # ------------------------------------------------------------------
        # Build start vectors
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Quick evaluation of raw seeds – keep a few promising ones
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(6, len(seed_vals))]]

        # Record the best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam refinement from each top seed
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=4000,
                beta0=2.0,
                beta_target=800.0,
                beta_mul=1.001,
                lr0=0.6,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – sub‑gradient refinement (longer for the current best)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=50000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(
                global_best_seq, n, deadline, max_iters=None
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – enforce symmetry (often reduces the peak)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            sym = _symmetrize(global_best_seq)
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[symmetrisation] n={n} improved to {sym_val:.6f}")

        # ------------------------------------------------------------------
        # Propagate the current best forward (for up‑scaling)
        # ------------------------------------------------------------------
        previous = global_best_seq

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(2000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Cache for a potential subsequent call
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
