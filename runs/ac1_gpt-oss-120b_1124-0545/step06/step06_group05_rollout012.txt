def solve_lp_for_b(f, M):
    n = len(f)
    N = 2*n - 1
    # Build constraint matrix A shape (N, n)
    rows = []
    cols = []
    data = []
    for k in range(N):
        # i from max(0, k-(n-1)) to min(k, n-1)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # for each i (i is index for f)
        # b index j = k - i
        # So for each i, add coefficient f[i] to variable b[j]
        # Not necessary to loop all k and i: total O(N*n) = O(n^2). Could be heavy but okay for n=4000.
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(N, n))
    b_ub = np.full(N, M)
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    # bounds: b >=0, no upper bound
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    else:
        return None

def improve_by_lp(f, deadline, max_outer_iters=5, alpha=0.5):
    n = len(f)
    best_f = f.copy()
    best_val = _objective(best_f)
    for it in range(max_outer_iters):
        if time.time() > deadline: break
        conv = _conv_fft(best_f, best_f)
        M = conv.max()
        # Build LP matrix
        A = build_conv_matrix(best_f)
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # no improvement possible
            print(f"[LP] iteration {it} no sum improvement sum_b={sum_b:.6f}")
            break
        b_norm = b / sum_b
        # update best_f with convex combo
        best_f = (1 - alpha) * best_f + alpha * b_norm
        # ensure nonneg and sum=1 (should hold). Minor numeric drift -> renorm
        best_f = np.maximum(best_f, 0)
        best_f /= best_f.sum()
        new_val = _objective(best_f)
        print(f"[LP] iter {it} sum_b={sum_b:.6f}, val={new_val:.6f}")
        if new_val < best_val:
            best_val = new_val
            # keep best_f as is
    return best_f, best_val

"""
Propose a step‐function minimizing the self‑convolution ∞‑norm.

Algorithm
---------
1. Create a diverse set of analytic seeds (uniform, geometric,
   power‑law, Gaussian, arcsine families, random Dirichlet draws,
   plus the previously best construction if available) and normalise
   them to sum = 1.

2. Keep the most promising seeds (lowest true objective) and for each
   of them run a *linear‑programming improvement*:
      – Let f be the current sequence, M = max(conv(f,f)).
      – Solve   max 1ᵀb   s.t.   conv(f,b) ≤ M  (element‑wise),   b ≥ 0.
        This LP is linear because conv(f,b) is linear in b.
      – If Σb > 1, replace the current f by a convex combination
        (or fully) with the normalised LP solution b̂ = b/Σb.
        The bound on the convolution drops to M/Σb, so the objective
        strictly improves.  Iterate a few times.

3. Refine the resulting sequence with a mirror‑descent (exponentiated‑gradient)
   sub‑gradient method that directly minimises the true max‑convolution.
   This is fast because each iteration only needs one FFT.

4. Finish with a cheap Metropolis–Hastings annealing to escape any tiny
   remaining local minima.

All stages respect a global deadline (default 1000 s).  The best
sequence found (a plain list of non‑negative floats) is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real) – fast for our sizes."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective of the competition:
        2·n·max(conv(seq,seq)) / (sum(seq))² .
    (seq assumed non‑negative; the objective is scale‑invariant.)
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """
    Discrete analogue of the arcsine law:
        a_i ∝ ((i+0.5)*(n-0.5-i))**exponent   (exponent < 0)
    Puts more mass near the ends.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide="ignore", invalid="ignore"):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        return _uniform(n)
    return a / s


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a rich pool of start vectors (already normalised)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # Basic analytic families
    seeds.append(_uniform(n))

    # Geometric series
    for r in np.linspace(0.2, 0.95, 6):
        seeds.append(_geom(n, r))

    # Power‑law families
    for p in np.linspace(0.0, 2.2, 7):
        seeds.append(_power(n, p))

    # Mixed geometric‑power
    for r in np.linspace(0.2, 0.9, 4):
        for p in np.linspace(0.0, 1.5, 4):
            seeds.append(_geom_power(n, r, p))

    # Symmetric geometric
    for r in np.linspace(0.2, 0.9, 5):
        seeds.append(_sym_geom(n, r))

    # Sine window and Gaussian
    seeds.append(_sine_window(n))
    for sf in [0.12, 0.18, 0.25, 0.33]:
        seeds.append(_gaussian(n, sigma_factor=sf))

    # Random Dirichlet draws (sparse ↔ dense regimes)
    for alpha in [0.05, 0.1, 0.3, 0.8, 1.5, 3.0]:
        for _ in range(2):
            seeds.append(_dirichlet_seed(n, alpha, rng))

    # Arcsine‑type seeds (negative exponents)
    for exp in np.linspace(-2.0, -0.1, 12):
        seeds.append(_arcseed(n, exp))

    # Mirror each seed (helps with asymmetric optima)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # Re‑use the previously best construction if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / max(previous.sum(), 1e-12))

    # Guarantee non‑empty list
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Linear‑programming improvement (convex combination with LP solution)
# ----------------------------------------------------------------------


def _lp_improve(f: np.ndarray,
                deadline: float,
                max_outer: int = 5,
                alpha: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve  max Σb   s.t.   conv(f,b) ≤ M,  b ≥ 0,
    where M = max(conv(f,f)).  Replace f by a convex mixture with the
    normalised LP solution.  Returns the improved sequence and its
    true objective.
    """
    n = f.size
    best_f = f.copy()
    best_val = _objective(best_f)

    for it in range(1, max_outer + 1):
        if time.time() > deadline:
            break

        conv_ff = _conv_fft(best_f, best_f)
        M = conv_ff.max()

        # Build the linear constraint matrix A (size (2n‑1) × n)
        # Each diagonal offset d (0 … n‑1) has the constant value best_f[d].
        diag_vals = [np.full(n, best_f[d]) for d in range(n)]
        offsets = np.arange(n)  # 0,1,…,n‑1
        A = diags(diag_vals, offsets, shape=(2 * n - 1, n), format="csr")
        b_ub = np.full(2 * n - 1, M)

        # Linear programme: maximise sum(b)  → minimise -sum(b)
        c = -np.ones(n)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method="highs")
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b_vec = res.x
        sum_b = b_vec.sum()

        # No improvement possible if Σb ≤ 1
        if sum_b <= 1.0 + 1e-12:
            print(f"[LP] iteration {it} no sum‑gain (Σb={sum_b:.6f})")
            break

        # Normalise and mix
        b_norm = b_vec / sum_b
        best_f = (1.0 - alpha) * best_f + alpha * b_norm
        # Numerical cleanup: stay on the simplex
        best_f = np.maximum(best_f, 0.0)
        best_f /= best_f.sum()

        cur_val = _objective(best_f)
        print(f"[LP] it={it:2d} Σb={sum_b:.6f}  val={cur_val:.6f}")

        if cur_val < best_val:
            best_val = cur_val

    return best_f, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient
    of the max‑convolution entry.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

        # Sub‑gradient for the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient step
        lr = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (cheap Metropolis–Hastings)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Small Metropolis–Hastings walk that swaps tiny mass between two
    random positions, with an exponentially decaying temperature, and
    occasional global Gaussian jitter.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)     # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a tiny fraction of mass from i to j
        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:5d} best={new_val:.6f}")

        # periodic global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            jittered = cur + noise
            jittered = np.clip(jittered, 0.0, None)
            if jittered.sum() == 0.0:
                jittered = _uniform(n)
            else:
                jittered /= jittered.sum()
            jitter_val = _objective(jittered)
            accept = jitter_val < cur_val or rng.random() < np.exp(-(jitter_val - cur_val) / (T + 1e-12))
            if accept:
                cur = jittered
                cur_val = jitter_val
                if jitter_val < best_val:
                    best_val = jitter_val
                    best = jittered.copy()
                    print(f"[Anneal] it={it:5d} (jump) best={jitter_val:.6f}")

    return best, best_val
