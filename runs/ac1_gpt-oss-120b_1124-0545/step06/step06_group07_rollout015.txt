def propose_candidate(seed: int=123, budget_s: int=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # Determine length: 4000

    # Setup target sum for simplex: 1.0 (since seeds normalized)

    # get previous best if any
    previous = globals().get("height_sequence_1")

            prev_arr = np.asarray(previous, dtype=np.float64)
            # Normalize to sum 1
            prev_arr = prev_arr / (prev_arr.sum() if prev_arr.sum() !=0 else 1.0)
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # Generate seeds
    seeds = _make_seeds(n, prev_arr)
    # add random permutations of prev if available
    if prev_arr is not None:
        rng = np.random.default_rng()
        for _ in range(10):
            perm = rng.permutation(n)
            seeds.append(prev_arr[perm])

    # Evaluate seeds and keep top few
    best_val = float('inf')
    best_seq = None

    for s in seeds:
        val = _evaluate(s)
        seed_vals.append((val, s))
        if val < best_val:

            best_seq = s.copy()
    # keep top K
    seed_vals.sort(key=lambda x: x[0])
    top_k = 5
    top_seeds = [seq for _, seq in seed_vals[:top_k]]

    # sequential refinement
    # 1. Adam smoothing
    for i, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break
        # Adam expects probabilities (sum=1)
        a0 = seed_seq / (seed_seq.sum() if seed_seq.sum()!=0 else 1.0)
        seq_adam, val_adam = _adam_optimize(a0, n, deadline, max_iters=60000)
        if seq_adam is not None and val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam.copy()
            print(f"[Adam refinement] new best={val_adam:.6f} from seed {i}")

    # 2. Projected subgradient refinement long
    if best_seq is not None:
        # Ensure sum=target_sum=1
        a0 = best_seq / best_seq.sum()
        seq_psg, val_psg = _proj_subgradient_refine(a0, n, deadline,

def _simulated_annealing(seq, n, deadline, max_iters=200000):
    rng = np.random.default_rng()
    cur = seq.copy()
    cur_val = _evaluate(cur)
    best_seq = cur.copy()
    best_val = cur_val
    temp0 = 0.1
    temp_end = 1e-6
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # temperature schedule linear decay
        t = temp0 * (1 - it / max_iters) + temp_end * (it / max_iters)
        # choose random pair i,j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # decide transfer fraction
        delta = rng.random() * cur[i] * 0.1  # up to 10% of donor
        donor, receiver = i, j
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        delta = rng.random() * cur[donor] * 0.05
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        val = _evaluate(new)
        if val < cur_val or rng.random() < np.exp(-(val - cur_val)/max(t,1e-12)):
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Anneal] it={it} new best={val:.6f}")
    return best_seq, best_val

1. **Seed generation** – a rich collection of analytically‑derived,
   L1‑normalised candidates (uniform, sine, geometric, power‑law,
   Beta‑shapes, Gaussians, Dirichlet draws).  The parameter ranges are
   much wider than in the reference implementation (β‑shape exponents up
   to 5, geometric ratios down to 0.1, etc.).  The previously best known
   sequence (if available as ``height_sequence_1``) and a few of its
   random permutations are also added.

2. **Seed screening** – all seeds are evaluated cheaply.  The five
   best candidates are kept for further optimisation.

3. **Adam on a smooth‑max surrogate** – each of the top seeds is refined
   with Adam on a log‑softmax parametrisation and a gradually sharpening
   smooth‑max (up to 60 000 iterations or until the time budget expires).

4. **Projected sub‑gradient descent** – the current best candidate is
   then refined for a long period (up to 400 000 iterations) using
   Euclidean projection onto the ℓ₁‑simplex (sum = 1).  This targets the
   exact max‑convolution directly.

5. **Simulated annealing (fine polishing)** – a cheap stochastic hill‑
   climb with a temperature schedule is run for a few hundred thousand
   cheap mass‑transfer moves to escape any remaining local minima.

All heavy loops check the deadline and stop early if time runs out.
The routine returns the best sequence (as a list of floats) found
within the allotted time.  The sequence is guaranteed to be non‑negative
and to have a positive ℓ₁ norm, so ``evaluate_sequence`` will not reject
it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution / simplex helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – uniform on the simplex
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian window, centred at (n‑1)/2."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """One‑sided exponential decay a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect an extensive set of normalised start vectors."""
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families – wider sweep
    for r in np.linspace(0.1, 0.96, 13):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.5, 0.99, 9):
        seeds.append(_sym_geom(n, r))

    # power‑law family
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # Beta shapes – denser grid and larger exponents
    for a in np.linspace(0.2, 5.0, 9):
        for b in np.linspace(0.2, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows – σ from 3 % to 35 % of n
    for frac in np.linspace(0.03, 0.35, 9):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    # Exponential decay – λ from 0.001 to 0.01 (slow) and up to 0.2 (fast)
    for lam in np.concatenate([np.linspace(0.001, 0.01, 5),
                               np.linspace(0.02, 0.2, 5)]):
        seeds.append(_exp_decay(n, lam))

    # Random Dirichlet draws
    rng = np.random.default_rng(2023)
    for conc in (0.2, 0.6, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random softmax vectors
    for _ in range(8):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # Include the previous best (if supplied)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Objective evaluation (fast, no clipping)
# ----------------------------------------------------------------------


def _evaluate(seq: np.ndarray) -> float:
    """Return the exact value of evaluate_sequence for a normalised L1‑1 vector."""
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    conv = _conv_fft(seq, seq)
    maxb = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * maxb / (sum_a ** 2)


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 60000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 2.0            # initial smoothing coefficient
    beta_mul = 1.025

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex (sum = 1)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)        # exact objective for sum=1
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a of Σ w_k·conv_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _proj_subgradient_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 400000,
                             target_sum: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (Euclidean projection onto the ℓ₁‑simplex).
    The sequence is kept on the simplex Σ x = target_sum.
    """
    a = _project_simplex(start_seq, target_sum)

    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        lr = 1.0 / np.sqrt(it)

        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, target_sum)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing (tiny random mass moves)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Light stochastic hill‑climb with a temperature schedule.
    Keeps the ℓ₁‑norm constant by moving a small amount of mass
    from one index to another.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _evaluate(cur)

    best_seq = cur.copy()
    best_val = cur_val

    # linear temperature decay
    T0 = 0.1
    T_end = 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # temperature
        T = T0 * (1 - it / max_iters) + T_end * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # pick donor/receiver randomly
        donor, recv = (i, j) if rng.random() < 0.5 else (j, i)

        if cur[donor] <= 0:
            continue
        # transfer at most 5 % of donor mass
        delta = rng.random() * 0.05 * cur[donor]
        new = cur.copy()
        new[donor] -= delta
        new[recv] += delta

        val = _evaluate(new)
        improve = val < cur_val
        accept = improve or (rng.random() < np.exp(-(val - cur_val) / max(T, 1e-12)))
        if accept:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Anneal] it={it:6d} best={val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine concentrates on intensive
    local optimisation of a very rich seed set and uses the full time
    budget (≈1000 s) to drive the objective down.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – 4000 works well with the time budget.
    # ------------------------------------------------------------------
    n = 4000
    target_sum = 1.0                       # all seeds are L1‑normalised

    # ------------------------------------------------------------------
    # Load the previously best known sequence (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n:
                # normalise to L1 = 1
                s = prev_arr.sum()
                if s > 0:
                    prev_arr = prev_arr / s
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Assemble a very diverse seed pool
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, prev_arr)

    # add a few random permutations of the previous best (if available)
    if prev_arr is not None:
        rng = np.random.default_rng()
        for _ in range(10):
            perm = rng.permutation(n)
            seeds.append(prev_arr[perm])

    print(f"Generated {len(seeds)} seeds (length {n}).")

    # ------------------------------------------------------------------
    # 2. Quick evaluation – keep the top K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for s in seeds:
        val = _evaluate(s)
        evaluated.append((val, s))
    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    top_seeds = [seq for _, seq in evaluated[:top_k]]
    best_val = evaluated[0][0]
    best_seq = evaluated[0][1].copy()
    print(f"Best initial seed value: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam refinement on the smooth‑max surrogate
    # ------------------------------------------------------------------
    for idx, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break
        # Adam works with a probability vector (sum = 1)
        a0 = seed_seq / (seed_seq.sum() if seed_seq.sum() != 0 else 1.0)
        seq_adam, val_adam = _adam_optimize(a0, n, deadline, max_iters=60000)
        if seq_adam is not None and val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam.copy()
            print(f"[Phase Adam] seed {idx} improved best to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent (long run)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a0 = best_seq / best_seq.sum()
        seq_psg, val_psg = _proj_subgradient_refine(a0, n, deadline,
                                                    max_iters=400000,
                                                    target_sum=target_sum)
        if val_psg < best_val:
            best_val = val_psg
            best_seq = seq_psg.copy()
            print(f"[Phase ProjSub] improved best to {val_psg:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated annealing (final polishing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq_ann, val_ann = _simulated_annealing(best_seq, n, deadline,
                                                max_iters=200000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = seq_ann.copy()
            print(f"[Phase Anneal] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # 6. Fallback – just in case nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _evaluate(best_seq)
        print("[Fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Optional verification with the provided evaluate_sequence()
    # --------------------------------------------------------------
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
