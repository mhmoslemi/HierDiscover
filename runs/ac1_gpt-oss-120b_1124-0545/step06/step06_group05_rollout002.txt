def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start=time.time()
    deadline = start + budget_s - 2.0

    # choose length: start with 8000 (if time). If we have previous var, we might also try to a longer length of previous? We'll ignore previous.

    # In fact we could try use previous variable if it exists (height_sequence_1). If it's length close to 4000 perhaps we can upsample it to longer length by interpolation.

    # Let's define length = 8000.

    n = 8000

    # We will generate candidates via param families:
    candidates = []
    # unify n.
    # Uniform
    candidates.append(_uniform(n))
    # arcsine exponents
    for exp in np.linspace(-1.5, -0.1, 30):
        candidates.append(_arcseed(n, exp))
    # beta shape alpha
    for alpha in np.linspace(0.15, 0.95, 30):
        seq = _beta_shape(n, alpha)
        candidates.append(seq)
    # mixture uniform+arc
    for exp in np.linspace(-1.2, -0.2, 15):
        arc = _arcseed(n, exp)
        for w in np.linspace(0.2, 0.8, 5):
            mix = w * _uniform(n) + (1-w) * arc
            mix /= mix.sum()
            candidates.append(mix)
    # geom symmetric
    for r in np.linspace(0.2, 0.98, 20):
        sym = _sym_geom(n, r)
        candidates.append(sym)

    # Evaluate each, keep top K (maybe top 20)
    cand_vals = [(seq, _objective(seq)) for seq in candidates]
    cand_vals.sort(key=lambda x: x[1])
    top_k = min(20, len(cand_vals))
    top_seeds = [seq for seq,_ in cand_vals[:top_k]]

    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)

    # Now subgradient refinement on each top seed with limited iterations (maybe 20000 each).
    for idx, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _subgradient_refine(seed_seq, n, deadline, max_iters=40000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Improved] seed {idx} val {val}")

        # optionally small anneal
        if time.time() > deadline:
            break
        seq2, val2 = _anneal_optimize(global_best_seq, n, deadline, max_iters=12000)
        if val2 < global_best_val:
            global_best_val = val2
            global_best_seq = seq2.copy()
            print(f"[Anneal improved] val {val2}")

    # Return best sequence
    # Ensure nonnegative and sum>0
    best = global_best_seq
    # Clip small values
    best = np.clip(best, 0.0, None)
    if best.sum() < 1e-12:
        best = _uniform(n)
    else:
        best = best / best.sum()
    return best.tolist()

"""
Optimisation of a non‑negative step function for the self‑convolution ∞‑norm.

The algorithm proceeds in three stages:

1. **Parametric seed generation** – a dense sweep over several analytic
   families (arcsine‑type, symmetric Beta, dual power‑law, geometric,
   mixed uniform+arc) for a chosen length (8000 or 12000).  All candidates
   are normalised to sum = 1 and the best few are retained.

2. **Exact sub‑gradient mirror‑descent** – starting from the most promising
   seed we run exponentiated‑gradient descent on the true objective
   (max‑self‑convolution).  The step size follows 0.75/√t and the loop
   stops as soon as the global time budget (≈ 1000 s) is exhausted.

3. **Lightweight simulated annealing** – a final Metropolis‑Hastings walk
   that shuffles tiny amounts of mass between random positions and
   occasionally adds a global Gaussian jitter.

The procedure respects the 1000 s limit, uses only NumPy (no external
LP solvers) and returns the best sequence found as a plain list of floats.
"""

from __future__ import annotations

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq))²."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Analytic seed families (all return a probability vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """Discrete analogue of the arcsine law: a_i ∝ ((i+½)(n‑½‑i))^exponent."""
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide="ignore", invalid="ignore"):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on a discrete grid.
    a_i ∝ (i)^{α‑1} (n‑i+1)^{α‑1}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore"):
        a = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (alpha - 1.0))
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _dual_power(n: int, p: float) -> np.ndarray:
    """Two‑sided power law: a_i ∝ (i+1)^‑p + (n‑i)^‑p."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    with np.errstate(divide="ignore", invalid="ignore"):
        a = idx ** (-p) + (n + 1 - idx) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _geom_sym(n: int, r: float) -> np.ndarray:
    """Symmetric geometric law: a_i ∝ r^i + r^{n‑1‑i}."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_uniform_arc(n: int, exponent: float, w: float) -> np.ndarray:
    """Mixture w·uniform + (1‑w)·arcsine‑type."""
    arc = _arcseed(n, exponent)
    uni = _uniform(n)
    mix = w * uni + (1.0 - w) * arc
    mix = np.clip(mix, 0.0, None)
    s = mix.sum()
    if s <= 0.0:
        return _uniform(n)
    return mix / s


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    if new_seq.sum() < 1e-12:
        return _uniform(new_n)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray, n: int, deadline: float, max_iters: int = 100_000
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror‑descent) on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # Sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated gradient step
        lr = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)  # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Cheap simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """Simple Metropolis‑Hastings walk with occasional global jitter."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)

    best = cur.copy()
    best_val = cur_val

    # initial temperature proportional to current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # move a tiny fraction of mass
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # global Gaussian jitter every 1000 steps
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            jittered = cur + noise
            jittered = np.clip(jittered, 0.0, None)
            if jittered.sum() == 0.0:
                jittered = _uniform(n)
            else:
                jittered /= jittered.sum()
            jitter_val = _objective(jittered)
            accept = jitter_val < cur_val or rng.random() < np.exp(-(jitter_val - cur_val) / (T + 1e-12))
            if accept:
                cur = jittered
                cur_val = jitter_val
                if jitter_val < best_val:
                    best_val = jitter_val
                    best = jittered.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={jitter_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found
    as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Re‑use a previously discovered construction if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0:
                previous = previous / max(previous.sum(), 1e-12)
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start modest, enlarge if time permits
    # ------------------------------------------------------------------
    lengths = [8000, 12_000]  # longer lengths give a better asymptotic bound
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1) Build a rich set of analytic seeds (already normalised)
        # ------------------------------------------------------------------
        seeds = []

        # Uniform baseline
        seeds.append(_uniform(n))

        # Arcsine‑type family (dense sweep of the exponent)
        for exp in np.linspace(-1.5, -0.05, 30):
            seeds.append(_arcseed(n, exp))

        # Symmetric Beta family (α in (0.2,1))
        for alpha in np.linspace(0.20, 0.95, 30):
            seeds.append(_beta_shape(n, alpha))

        # Dual power‑law (p controls tail heaviness)
        for p in np.linspace(0.2, 2.0, 20):
            seeds.append(_dual_power(n, p))

        # Symmetric geometric
        for r in np.linspace(0.30, 0.98, 15):
            seeds.append(_geom_sym(n, r))

        # Mixtures of uniform and arcsine
        for exp in np.linspace(-1.0, -0.2, 9):
            for w in np.linspace(0.2, 0.8, 5):
                seeds.append(_mix_uniform_arc(n, exp, w))

        # Include previous construction (interpolated if needed)
        if previous is not None:
            if previous.shape[0] != n:
                seeds.append(_interpolate(previous, n))
            else:
                seeds.append(previous.copy())

        # ------------------------------------------------------------------
        # 2) Evaluate all seeds, keep the most promising ones
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_k = min(15, len(seed_vals))
        top_seeds = [s for s, _ in seed_vals[:top_k]]

        # Record the best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) Exact sub‑gradient refinement on the single best seed
        # ------------------------------------------------------------------
        best_seed = top_seeds[0]
        refined_seq, refined_val = _subgradient_refine(
            best_seed, n, deadline, max_iters=100_000
        )
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[overall SubGrad] n={n} best={refined_val:.6f}")

        # ------------------------------------------------------------------
        # 4) Simulated‑annealing fine‑tuning (short run – we stay within budget)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            annealed_seq, annealed_val = _anneal_optimize(
                global_best_seq, n, deadline, max_iters=None
            )
            if annealed_val < global_best_val:
                global_best_val = annealed_val
                global_best_seq = annealed_seq.copy()
                print(f"[overall Anneal] n={n} best={annealed_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(8000)
        print("[fallback] using uniform distribution")

    # Ensure feasibility before returning
    best = np.clip(global_best_seq, 0.0, None)
    if best.sum() < 1e-12:
        best = _uniform(best.shape[0])
    else:
        best = best / best.sum()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(best)})"
    )
    return best.tolist()
