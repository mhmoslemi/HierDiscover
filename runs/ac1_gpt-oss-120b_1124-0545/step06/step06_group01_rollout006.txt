def smooth_max_surrogate(a, beta):
   conv = conv_fft(a,a)
   shift = np.max(conv)
   w = np.exp(beta*(conv - shift))
   w_sum = np.sum(w)
   smooth_max = shift + np.log(w_sum)/beta
   return smooth_max, w

We will compute gradient:

def grad_smooth_max(a, w):
   rev_a = a[::-1]
   full = conv_fft(rev_a, w)
   grad_a = 2 * full[n-1:n-1+n]  # central part
   return grad_a

Now for variable v (logits), we update using gradient on v: grad_v = a * (grad_a - np.dot(grad_a, a))

def l_bfgs_optimize(logits, n, beta, max_iter):
    def fun(v):
        a = softmax(v)
        smooth_val, w = smooth_max_surrogate(a, beta)
        # compute gradient w.r.t a: grad_a = Σ w_i * d conv_i/da = 2 * conv_fft(rev_a, w)[n-1:...]
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        # convert to gradient w.r.t v
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return smooth_val, grad_v.astype(np.float64)

"""
Propose a step‑function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the linear convolution.  The algorithm combines
* a rich set of analytic and random seeds,
* a L‑BFGS optimiser on a smooth‑max surrogate (the only place where SciPy
  is used – a pure‑Python Adam fallback works if SciPy is unavailable),
* a few cheap mirror‑descent and hill‑climbing refinements,
* and a simple basin‑hopping loop that repeatedly perturbs the current
  best candidate and re‑optimises it.

All heavy work is done with FFT‑based convolutions, so the routine easily
fits into the 1000 s budget for lengths up to ≈10 000.  The final sequence
is returned as a plain list of floats; `evaluate_sequence` will check
validity.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Global state for the SciPy L‑BFGS callback (no closures, no lambdas)
# ----------------------------------------------------------------------
_lbfgs_n = None          # type: int | None
_lbfgs_beta = None       # type: float | None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_analytic_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of ℓ₁‑normalised analytic start vectors."""
    seeds: list[np.ndarray] = []
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet families
    rng = np.random.default_rng(2023)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # softmax of Gaussian logits
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


def _resample(seq: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence to a new length n."""
    old_n = len(seq)
    if old_n == n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n, dtype=np.float64)
    x_new = np.linspace(0, 1, n, dtype=np.float64)
    a = np.interp(x_new, x_old, seq)
    a = a / a.sum() if a.sum() != 0 else _uniform(n)
    return a

def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random piecewise‑linear distribution via Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray | None, float]:
    """Adam on a smooth‑max surrogate (log‑softmax parametrisation)."""
    v = np.log(seed + 1e-30)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # initial smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution (mirror descent)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _targeted_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """Random mass transfers focusing on the maximal convolution entry."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total <= 0.0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[TargetHill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate (requires SciPy)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_obj(v: np.ndarray) -> tuple[float, np.ndarray]:
    """Objective for SciPy L‑BFGS: smooth‑max of the convolution."""
    global _lbfgs_n, _lbfgs_beta
    n = _lbfgs_n
    beta = _lbfgs_beta
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)

    # smoothed max
    scaled = beta * (conv - shift)            # ≤ 0
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    smooth_val = shift + np.log(w_sum) / beta

    # gradient wrt a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # gradient wrt logits (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    return float(smooth_val), grad_v.astype(np.float64)


def _lbfgs_optimize(seed: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule: list[float] = (5.0, 15.0, 40.0),
                    iters_schedule: list[int] = (200, 200, 200)) -> tuple[np.ndarray, float]:
    """Run a brief L‑BFGS cascade. Returns (best_seq, best_val)."""
    global _lbfgs_n, _lbfgs_beta
    _lbfgs_n = n

    # start from logits
    v = np.log(seed + 1e-30)

    best_seq = None
    best_val = float("inf")

    for beta, max_it in zip(beta_schedule, iters_schedule):
        if time.time() > deadline:
            break
        _lbfgs_beta = beta
        if _opt is None:
            # fall back to Adam if SciPy missing
            seq, val = _adam_optimize(_softmax(v), n, deadline, max_iters=max_it)
            if seq is None:
                break
            v = np.log(seq + 1e-30)
            best_seq, best_val = seq, val
            continue

        res = _opt.minimize(_lbfgs_obj,
                             v,
                             method='L-BFGS-B',
                             jac=True,
                             options={'maxiter': max_it, 'disp': False})
        v = res.x
        a = _softmax(v)
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS] n={n:5d} beta={beta:.1f} it={max_it} best={val:.6f}")

    if best_seq is None:
        # Should never happen, but keep a safe fallback
        best_seq = seed.copy()
        best_val = _objective(best_seq, n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple basin‑hopping perturbation + quick re‑optimise
# ----------------------------------------------------------------------


def _perturb_and_refine(a: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """Apply a few random mass transfers then a short L‑BFGS run."""
    rng = np.random.default_rng()
    cur = a.copy()
    # perform 3–5 random transfers
    for _ in range(rng.integers(3, 7)):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = rng.random() * 0.05 * cur[i]      # up to 5 % of donor
        cur[i] -= delta
        cur[j] += delta
    cur = np.maximum(cur, 0.0)
    cur /= cur.sum() if cur.sum() != 0 else _uniform(n)

    # a short L‑BFGS pass (small iteration budget)
    seq, val = _lbfgs_optimize(cur, n, deadline,
                               beta_schedule=(5.0, 20.0),
                               iters_schedule=(100, 100))
    return seq, val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine mixes a wide family of analytic
    and random seeds, L‑BFGS on a smooth‑max surrogate, mirror‑descent,
    targeted hill‑climbing and a final basin‑hopping loop.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Warm‑start: retrieve the previous run if present
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Try progressively larger lengths – larger n gives a lower theoretical bound.
    lengths = [8000, 4000, 2000]

    global_best_seq: np.ndarray | None = None
    global_best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Helper for maintaining the top‑K candidates (simple sorted list)
    # ------------------------------------------------------------------
    def _update_top(lst: list[tuple[float, np.ndarray]],
                    val: float,
                    seq: np.ndarray,
                    max_keep: int = 12) -> None:
        lst.append((val, seq))
        lst.sort(key=lambda x: x[0])          # smallest first
        if len(lst) > max_keep:
            lst.pop()

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        top_candidates: list[tuple[float, np.ndarray]] = []

        # --------------------------------------------------------------
        # 1. Analytic seeds (cheap)
        # --------------------------------------------------------------
        analytic = _make_analytic_seeds(n, None if previous is None else _resample(previous, n))
        for s in analytic:
            val = _objective(s, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[analytic] n={n:5d} new global best={val:.6f}")
            _update_top(top_candidates, val, s)

        # --------------------------------------------------------------
        # 2. Random piece‑wise‑linear seeds (broader exploration)
        # --------------------------------------------------------------
        rng = np.random.default_rng()
        max_random = 2500
        count_random = 0
        while time.time() < deadline - 30.0 and count_random < max_random:
            cand = _random_piecewise_linear(n, rng)
            val = _objective(cand, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = cand.copy()
                print(f"[random‑PL] n={n:5d} new global best={val:.6f}")
            _update_top(top_candidates, val, cand)
            count_random += 1

        # Keep only the most promising seeds for intensive polishing
        top_candidates = top_candidates[:6]

        # --------------------------------------------------------------
        # 3. Intensive polishing of each top seed
        # --------------------------------------------------------------
        for idx, (val, seq) in enumerate(top_candidates):
            if time.time() > deadline:
                break
            print(f"\n--- Polishing candidate #{idx+1} (n={n}) ---")

            # 3a) L‑BFGS cascade (smooth‑max)
            seq_lbfgs, val_lbfgs = _lbfgs_optimize(seq, n, deadline)
            if val_lbfgs < global_best_val:
                global_best_val = val_lbfgs
                global_best_seq = seq_lbfgs.copy()
                print(f"[polish‑L‑BFGS] n={n:5d} best={val_lbfgs:.6f}")

            # 3b) Adam refinement (optional – cheap)
            seq_adam, val_adam = _adam_optimize(seq_lbfgs, n, deadline, max_iters=8000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[polish‑Adam] n={n:5d} best={val_adam:.6f}")

            # 3c) Exact sub‑gradient descent (mirror descent)
            seq_sg, val_sg = _subgradient_refine(global_best_seq, n, deadline,
                                                max_iters=20000)
            if val_sg < global_best_val:
                global_best_val = val_sg
                global_best_seq = seq_sg.copy()
                print(f"[polish‑SubGrad] n={n:5d} best={val_sg:.6f}")

            # 3d) Targeted stochastic hill‑climb
            seq_hc, val_hc = _targeted_hill_climb(global_best_seq, n, deadline,
                                                  max_iters=20000)
            if val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[polish‑Hill] n={n:5d} best={val_hc:.6f}")

        # --------------------------------------------------------------
        # 4. Basin‑hopping loop – try to escape any remaining local minima
        # --------------------------------------------------------------
        print(f"[Basin‑hop] starting for n={n}")
        while time.time() + 5.0 < deadline:
            pert_seq, pert_val = _perturb_and_refine(global_best_seq, n, deadline)
            if pert_val < global_best_val:
                global_best_val = pert_val
                global_best_seq = pert_seq.copy()
                print(f"[Basin‑hop] n={n:5d} new best={pert_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback handling – should never be needed
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)
        print("[fallback] using uniform distribution")

    # ----------------------------------------------------------------------
    # Final verification with the provided evaluation function
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        # If the sequence is rejected (unlikely) fall back to uniform.
        print("[verification] sequence rejected – falling back to uniform")
        global_best_seq = _uniform(len(global_best_seq))
        global_best_val = _objective(global_best_seq, len(global_best_seq))

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
