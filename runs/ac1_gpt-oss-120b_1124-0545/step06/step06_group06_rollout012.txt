    (non‑smoothed) max is used together with an exponentiated‑gradient
    step.  The step‑size decays as 1/√t.

4.  **Extensive simulated‑annealing hill climbing** – at each iteration a
    random set of 2‑6 coordinates has its total mass reshuffled via a
    Dirichlet draw.  An annealing schedule slowly cools the system.
    Several independent SA runs (different random seeds) are launched.

5.  **Final polishing** – a short Adam run starting from the best
    solution found by SA.

All stages respect the supplied time budget (default 1000 s).  The
function returns a plain list of floats that can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming ∑seq == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return vectors normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Peak in the centre, linearly decreasing to the edges."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    """Flat top of relative width `width_fraction`, linear ramps on each side."""
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Right‑skewed geometric decay a_i ∝ r^i."""
    if r <= 0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometries: r**i + r**(n‑1‑i)."""
    if r <= 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^‑p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    """Standard cosine (raised‑cosine) window, zero at the ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2) / (n - 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with scalar concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse vector with non‑zeros sampled from a Dirichlet."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------


def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of normalised candidates for length `n`."""
    seeds: list[np.ndarray] = []

    # deterministic analytical families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.2))
    seeds.append(_sine_window(n))
    seeds.append(_cosine_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_sym_geom(n, r))

    # power‑law family
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # beta family
    alphas = np.linspace(0.2, 2.0, 10)
    betas = np.linspace(0.2, 2.0, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # sine^γ families
    for gamma in np.linspace(0.5, 4.0, 8):
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
        seeds.append(a / a.sum())

    # random Dirichlet draws
    for conc in (0.1, 0.3, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous global best (if length matches)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    # limit the total number to keep seed stage cheap (optional)
    if len(seeds) > 500:
        seeds = seeds[:500]

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 40000) -> tuple[np.ndarray | None, float]:
    """Adam on a smoothed max (log‑sum‑exp) with a log‑softmax parametrisation."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5            # initial temperature
    beta_mul = 1.001
    beta_max = 200.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # increase temperature (capped)
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w·conv w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # back‑prop to logits (log‑softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping (helps stability)
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.7
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray, n: int, deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent (exponentiated‑gradient)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_max = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_max]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active (max) lag
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k_max - idx
        # each term appears twice except the centre when idx==rev
        g[idx] = 2.0 * a[rev]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # for numerical stability
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 400000,
                         init_temp_factor: float = 0.03,
                         min_temp: float = 1e-10) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climbing with annealing.  Several restarts are performed."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = max(T * 0.9996, min_temp)

        k = rng.integers(2, 7)          # move 2‑6 coordinates
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        # reshuffle total mass using a Dirichlet draw
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with minimal evaluation score.
    Returns a list of floats ready for ``evaluate_sequence``.
    """
    # deterministic reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # leave a small safety margin

    # Retrieve the previous best, if it exists.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # We'll experiment with a few different lengths – longer vectors give
    # more freedom but increase per‑iteration cost.
    lengths = [3000, 4000, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – Broad seed scan (fast)
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Scanning seeds for length {n} ---")
        seeds = _collect_seeds(n, rng, previous)

        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} val={val:.6f}")

    # If nothing found, fall back to a uniform vector.
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)

    # ------------------------------------------------------------------
    # Stage 2 – Adam (smoothed max)
    # ------------------------------------------------------------------
    n = len(global_best_seq)
    print(f"\n=== Adam refinement (n={n}) – current best {global_best_val:.6f} ===")
    seq, val = _adam_optimize(global_best_seq, n, deadline, max_iters=50000)
    if seq is not None and val < global_best_val:
        global_best_seq = seq.copy()
        global_best_val = val
        print(f"[overall‑Adam] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Exact sub‑gradient mirror descent
    # ------------------------------------------------------------------
    print(f"\n=== Sub‑gradient refinement (n={n}) — best so far {global_best_val:.6f} ===")
    seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=400000)
    if seq is not None and val < global_best_val:
        global_best_seq = seq.copy()
        global_best_val = val
        print(f"[overall‑SubGrad] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing hill climbing (multiple restarts)
    # ------------------------------------------------------------------
    print(f"\n=== Simulated‑annealing (n={n}) — best so far {global_best_val:.6f} ===")
    # we launch a few short SA runs from slightly perturbed copies of the current best
    num_restarts = 3
    for rs in range(num_restarts):
        if time.time() > deadline:
            break
        # tiny random perturbation before each restart
        pert = np.clip(global_best_seq + rng.normal(scale=1e-4, size=n), 0.0, None)
        pert /= pert.sum()
        seq, val = _simulated_annealing(pert, n, deadline,
                                        max_iters=200000,
                                        init_temp_factor=0.03)
        if val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall‑SA‑run{rs}] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Final polishing (short Adam run)
    # ------------------------------------------------------------------
    print(f"\n=== Final polishing (Adam) — best so far {global_best_val:.6f} ===")
    seq, val = _adam_optimize(global_best_seq, n, deadline, max_iters=25000)
    if seq is not None and val < global_best_val:
        global_best_seq = seq.copy()
        global_best_val = val
        print(f"[final‑Adam] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Sanity check – evaluate via the provided wrapper (may clip values)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
