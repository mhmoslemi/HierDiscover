"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.  The approach is a simple, highly‑parallelisable
pipeline:

1. **Rich seed pool** – uniform, geometric, power‑law, Beta‑law,
   arcsine‑law and several Dirichlet draws (including one obtained by
   up‑sampling the previously best known sequence, if available).  All
   seeds are normalised to sum = 1 and live on the probability simplex.

2. **Mirror‑descent on the true objective** – for each seed we run a
   multiplicative‑gradient (exponentiated‑gradient) iteration that
   minimizes the exact quantity  
   `2·n·max(conv(a,a))/(sum a)^2`.  The sub‑gradient of the max‑convolution
   is cheap to compute, and the update keeps the vector on the simplex.
   We perform a large number of iterations (limited only by the global
   time budget) and keep the best iterate.

3. **Light annealing** – a short Metropolis‑Hastings walk on the best
   sequence found so far is used to escape possible shallow local minima.

The whole search respects the 1000 s budget, uses only NumPy (no external
solvers) and returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution (full length) using a real FFT.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (scale‑invariant).
    The vector is assumed non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r^i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ i^{-p} (i starts at 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑law a_i ∝ (i)^{α‑1}·(n‑i+1)^{β‑1}.
    For α,β<1 the distribution is U‑shaped (mass near the ends).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    with np.errstate(divide='ignore', invalid='ignore'):
        a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """
    Discrete analogue of the continuous arcsine law:
    a_i ∝ ((i+½)*(n‑½‑i))**exponent  with exponent<0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide='ignore', invalid='ignore'):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Gamma(α,1) draws normalised to sum = 1."""
    vec = rng.gamma(alpha, 1.0, size=n)
    s = vec.sum()
    return vec / s if s > 0 else _uniform(n)


def _interpolate(prev: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a shorter vector to length ``n``."""
    old_n = prev.shape[0]
    if old_n == n:
        return prev.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, n)
    new = np.interp(x_new, x_old, prev)
    # tiny negative artefacts from numerical interpolation are clipped
    new = np.clip(new, 0.0, None)
    s = new.sum()
    return new / s if s > 0 else _uniform(n)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of start‑vectors (already normalised to sum=1).
    """
    rng = np.random.default_rng(123456)
    seeds: list[np.ndarray] = []

    # Basic families
    seeds.append(_uniform(n))

    # Geometric families (r in (0,1))
    for r in np.linspace(0.2, 0.95, 7):
        seeds.append(_geom(n, r))

    # Power‑law families
    for p in np.linspace(0.1, 2.5, 8):
        seeds.append(_power(n, p))

    # Beta‑law families – a grid of (α,β) values.
    alphas = [0.2, 0.3, 0.5, 0.8, 1.2]
    betas = [0.2, 0.3, 0.5, 0.8, 1.2]
    for a in alphas:
        for b in betas:
            seeds.append(_beta_seed(n, a, b))

    # Arcsine‑type seeds (U‑shaped)
    for exp in [-2.0, -1.5, -1.0, -0.75, -0.5, -0.25]:
        seeds.append(_arcseed(n, exp))

    # Dirichlet draws – from sparse to dense.
    for a in [0.05, 0.1, 0.3, 0.6, 1.0, 2.0]:
        seeds.append(_dirichlet_seed(n, a, rng))

    # Mirror each seed (helps when optimum is asymmetric)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # Use previously best known construction, up‑sampled if necessary.
    if previous is not None and previous.size > 0:
        seeds.append(_interpolate(previous, n))

    # Ensure non‑empty.
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Mirror‑descent on the simplex (sub‑gradient of the max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 120000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) minimizing the exact objective.
    Returns the best sequence encountered and its objective.
    """
    a = start_seq.copy()
    # protect against zeroes – add epsilon then renormalise
    eps = 1e-30
    a = np.clip(a, eps, None)
    a /= a.sum()

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # Compute convolution and locate the maximal entry.
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # True objective (should equal 2·n·conv[max_idx] since sum≈1)
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[SubGrad] n={n} it={it:6d} best={cur_val:.6f}")

        # Sub‑gradient of the max‑convolution w.r.t a.
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient update (mirror descent on the simplex).
        lr = 0.8 / np.sqrt(it)          # learning‑rate schedule
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical safety
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Cheap Metropolis‑Hastings annealing (fine‑tuning)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that shifts a tiny fraction of mass
    between two random coordinates and occasionally adds a global jitter.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature proportional to current cost
    decay = 0.9997

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T0 * (decay ** it)

        # pick two indices and move a tiny amount of mass
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of the source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                if it % 5000 == 0:
                    print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        # every 2000 steps add a gentle Gaussian jitter
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.001, size=n)
            new_seq = cur + jitter
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            if s > 0:
                new_seq /= s
                new_val = _objective(new_seq)
                accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
                if accept:
                    cur = new_seq
                    cur_val = new_val
                    if new_val < best_val:
                        best_val = new_val
                        best = new_seq.copy()
                        print(f"[Anneal] n={n} it={it:6d} (jitter) best={new_val:.6f}")

    return best, best_val

    s = global_best_seq.sum()

        global_best_seq /= s

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "

    )
    return global_best_seq.tolist()
