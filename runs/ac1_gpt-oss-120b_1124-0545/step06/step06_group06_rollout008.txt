"""
search_step_function.py

A time‑aware optimiser for the objective

    2·n·max(conv(a,a)) / (∑a)²

The code follows four stages:

1. **Diverse seed generation** – many analytically motivated families
   (uniform, power‑law, Beta, inverse‑product‑power, sine‑window,
   random Dirichlet, sparse patterns, random permutations of the above).
   The families are explored on a fine grid (including very “U‑shaped”
   Beta parameters < 0.2) so that the search space is richer than the
   reference implementation.

2. **Adam on a smoothed max** – the vector lives on the simplex via a
   log‑softmax.  A slowly growing temperature (log‑sum‑exp) approximates
   the true max.  Adam updates (with decaying learning‑rate) are applied
   until a configurable fraction of the time budget is exhausted.

3. **Exact sub‑gradient mirror descent** – an exponentiated‑gradient step
   using the true sub‑gradient of the *unsmoothed* max further refines the
   solution.  The step‑size decays as 1/√t and the loop runs until the
   remaining budget is spent.

4. **Resolution lifting** – if time permits the best vector is linearly
   interpolated to a larger length (2×) and the whole refinement (Adam +
   sub‑gradient) is repeated.  A finer discretisation often lowers the
   constant  n·max(conv)  a little.

The routine respects the 1000 s budget, prints progress information and
returns the best sequence found (as a plain ``list`` of ``float``) ready
for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for many calls)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy).real[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming ∑seq == 1.
    (The original ``evaluate_sequence`` multiplies by (∑a)², but the
    simplex constraint makes the denominator equal to 1.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all returned **already normalised** to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^‑p   (p≥0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_product_power(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ ((i+1)*(n-i))^{-p}.
    For p>0 the shape is “U‑shaped’’ (larger near the ends).
    """
    if p <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n - i + 1.0)
    a = prod ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_power(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet with scalar concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse vector with non‑zeros drawn from a Dirichlet."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed collection (returns a list of normalised np.ndarray)
# ----------------------------------------------------------------------


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # Power‑law family (covers both decaying and uniform limits)
    for p in np.linspace(0.0, 5.0, 13):
        seeds.append(_power(n, p))

    # Beta family – we explore a *much* broader range than the reference
    alphas = np.concatenate(
        (
            np.logspace(np.log10(0.03), np.log10(0.2), 6),
            np.linspace(0.2, 3.0, 9),
        )
    )
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Inverse‑product‑power (U‑shaped) family
    for p in np.linspace(0.1, 3.0, 13):
        seeds.append(_inverse_product_power(n, p))

    # Sine‑power family
    for gamma in np.linspace(0.3, 6.0, 12):
        seeds.append(_sine_power(n, gamma))

    # Random Dirichlet draws – a few low‑concentration settings
    for conc in (0.01, 0.05, 0.1, 0.3, 0.7, 1.5):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns (different sparsities)
    for k in (3, 5, 9, 15):
        for conc in (0.1, 0.3, 0.7):
            seeds.append(_random_sparse(n, k, conc, rng))

    # Random permutations of a few promising shapes – asymmetry can help
    for base in [seeds[2], seeds[4], seeds[7]]:  # power, beta, inv‑prod
        perm = base.copy()
        rng.shuffle(perm)
        seeds.append(perm)

    # Include the previous global best (if it has the same length)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    start_vec: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on a smoothed version of ``max(conv)``.
    The vector lives on the simplex via a log‑softmax parametrisation.
    Smoothing temperature (beta) grows very slowly – this gives a more
    accurate gradient for a longer period.
    """
    n = len(start_vec)
    eps = 1e-30
    logits = np.log(start_vec + eps)      # initialise log‑softmax parameters

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta_temp = 0.4          # initial temperature
    beta_mul = 1.0005        # *very* slow increase
    beta_max = 400.0

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(logits)                # a stays on the simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing (capped)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]   # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam step
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-12
        lr0 = 0.7
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent using exponentiated‑gradient updates.
    The step‑size decays as 1/√t.  The loop runs until the deadline.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the *true* max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]                     # factor 2 as in analysis

        # exponentiated‑gradient update
        lr = 0.65 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                         # numerical stability
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

    return best_seq, best_val


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a simplex‑vector to a larger dimension.
    The resulting vector is re‑normalised to sum = 1.
    """
    old_n = len(seq)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    return new_seq / s if s > 0 else _uniform(new_n)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a list of floats.
    """
    # reproducibility
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # tiny safety cushion – we want to finish before the external timeout
    deadline = start_time + budget_s - 1.5

    # ------------------------------------------------------------------
    # Retrieve a globally stored solution (if any) – useful for incremental runs
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Stage 1 – broad seed scan (fast)
    # ------------------------------------------------------------------
    lengths = [2000, 4000]               # initial lengths – cheap to evaluate
    global_best_seq = None
    global_best_val = float("inf")
    best_len = None

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Seed scan for length {n} ---")
        seeds = _collect_seeds(n, rng, previous)

        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                best_len = n
                print(f"[seed] n={n:5d} val={val:.6f}")

    # Defensive fallback (should never happen)
    if global_best_seq is None:
        best_len = 2000
        global_best_seq = _uniform(best_len)
        global_best_val = _objective(global_best_seq)

    print(
        f"\n=== Initial best: length={best_len}  objective={global_best_val:.6f} ==="
    )

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation (smoothed max)
    # ------------------------------------------------------------------
    adam_deadline = deadline - (deadline - start_time) * 0.30   # 30 % of remaining time
    seq, val = _adam_optimize(global_best_seq, adam_deadline)
    if seq is not None and val < global_best_val:
        global_best_seq, global_best_val = seq, val
        print(f"[overall‑Adam] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Exact sub‑gradient mirror descent
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - (deadline - start_time) * 0.10   # 10 % left for this stage
    seq, val = _subgradient_refine(global_best_seq, subgrad_deadline)
    if val < global_best_val:
        global_best_seq, global_best_val = seq, val
        print(f"[overall‑SubGrad] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Resolution lifting (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:          # reserve ~30 s for final checks
        # double the resolution
        new_n = best_len * 2
        print(f"\n--- Lifting resolution to {new_n} ---")
        lifted_seq = _interpolate(global_best_seq, new_n)

        # repeat Adam + sub‑gradient on the larger grid
        lift_deadline = deadline - 5.0   # keep a few seconds spare
        seq, val = _adam_optimize(lifted_seq, lift_deadline)
        if seq is not None and val < global_best_val:
            global_best_seq, global_best_val = seq, val
            print(f"[lift‑Adam] improved global best to {val:.6f}")

        seq, val = _subgradient_refine(global_best_seq, lift_deadline)
        if val < global_best_val:
            global_best_seq, global_best_val = seq, val
            print(f"[lift‑SubGrad] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional – any exception is caught and ignored)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose the result for possible future incremental runs
    globals()["height_sequence_1"] = global_best_seq.tolist()

    return global_best_seq.tolist()
