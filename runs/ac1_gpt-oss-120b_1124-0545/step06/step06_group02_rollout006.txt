def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (length len(a)+len(b)-1)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

# Window families (each returns a *probability* vector, i.e. sum = 1)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window (0 ≤ α ≤ 1)."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shaped window, α>0, β>0."""
    x = (np.arange(1, n + 1, dtype=np.float64) - 0.5) / n
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a rich set of normalised start vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # Very basic families
    seeds.append(_uniform(n))

    # Geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # Power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # Combined geometric / power families
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # Symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Classic spectral windows
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # Tukey
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))

    # Beta windows – a surprisingly rich source of candidates
    for a in np.linspace(0.5, 5.0, 5):
        for b in np.linspace(0.5, 5.0, 5):
            seeds.append(_beta_window(n, a, b))

    # Random soft‑max vectors
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # Re‑use a previous construction if dimensions match
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # Clean‑up (clip tiny negatives caused by numerical noise)
    cleaned: list[np.ndarray] = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            cleaned.append(s / tot)
    return cleaned

# Smooth‑max objective & its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smoothed objective
        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v))).

    ``beta`` must be > 0.  The gradient w.r.t. the logits ``v`` is supplied.
    """
    a = _softmax(v)                     # probability vector
    conv = _conv_fft(a, a)               # length = 2n‑1
    shift = np.max(conv)
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta   # = (1/β)·log Σ exp(β·conv)

    obj = 2.0 * n * smooth_val

    w = exp_shifted / sum_exp            # weights for each convolution entry
    rev_a = a[::-1]

    # weighted convolution of rev_a with w (full length = 3n‑2)
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

    # chain‑rule through the soft‑max (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)          # size n
    grad_v = grad_v * (2.0 * n)          # outer factor
    return obj, grad_v


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: np.ndarray,
                     maxiter_per_stage: int = 180) -> tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS on the smooth‑max surrogate.
    Returns the best probability vector found and its true objective.
    """
    eps = 1e-30
    v = np.log(start_seq + eps)          # initialise logits

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta)

        if SCIPY_AVAILABLE:
            res = optimize.minimize(fun,
                                    v,
                                    method='L-BFGS-B',
                                    jac=True,
                                    options={'maxiter': maxiter_per_stage,
                                             'gtol': 1e-12,
                                             'ftol': 1e-12,
                                             'disp': False})
            v = res.x
        else:
            # Fallback: simple gradient descent (rarely needed)
            grad = None
            for _ in range(maxiter_per_stage):
                val, grad = _smooth_obj_and_grad(v, n, beta)
                v -= 0.1 * grad / (np.linalg.norm(grad) + 1e-12)

        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1f} → best={cur_val:.6f}")

    return best_seq, best_val

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true non‑smooth objective.
    Stops when time runs out or after ``max_iters`` iterations.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # subgradient of max‑conv (average over all active positions)
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        lr = 0.9 / np.sqrt(it)                 # diminishing learning rate
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Very cheap Metropolis walk that swaps a tiny amount of mass between
    two random positions.  Used only if a little time is left.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional Gaussian jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

def _order_alternate_ends(arr: np.ndarray) -> np.ndarray:
    """Place the largest values alternately at the left‑most and right‑most ends."""
    desc = np.sort(arr)[::-1]
    n = len(arr)
    out = np.empty(n, dtype=arr.dtype)
    left, right = 0, n - 1
    for i, v in enumerate(desc):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _random_permutation_search(base_seq: np.ndarray,
                               deadline: float,
                               max_trials: int = 120) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climb on the ordering: start from ``base_seq`` and try
    random permutations; keep the best encountered.
    """
    n = len(base_seq)
    rng = np.random.default_rng()
    best_seq = base_seq.copy()
    best_val = _objective(best_seq)

    for trial in range(max_trials):
        if time.time() > deadline:
            break
        perm = rng.permutation(n)
        seq = base_seq[perm]
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Perm] trial={trial:3d} best={val:.6f}")

    return best_seq, best_val

    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1) Build a diverse set of seeds.
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)

    # ------------------------------------------------------------------
    # 2) Quick screening – keep the top few raw seeds.
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 9
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")

    if seed_vals and seed_vals[0][1] < global_best_val:
        global_best_val = seed_vals[0][1]
        global_best_seq = seed_vals[0][0].copy()
        print(f"[seed] initial best={global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Multi‑stage L‑BFGS on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    beta_sched = np.geomspace(5.0, 2000.0, num=8)   # 5 → 2000 (geometric)
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _l_bfgs_optimize(seed_vec,
                                    n,
                                    deadline,
                                    beta_schedule=beta_sched,
                                    maxiter_per_stage=200)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall L‑BFGS #{idx}] new best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact sub‑gradient refinement (if time permits).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=80000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] refined best={val:.6f}")

    # ------------------------------------------------------------------
    # 5) Permutation‑based improvement (spreads large values apart).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        # deterministic alternating‑ends ordering
        seq_alt = _order_alternate_ends(global_best_seq)
        val_alt = _objective(seq_alt)
        if val_alt < global_best_val:
            global_best_val = val_alt
            global_best_seq = seq_alt.copy()
            print(f"[overall PermDet] improved best={val_alt:.6f}")

        # random permutation hill‑climb
        seq_perm, val_perm = _random_permutation_search(global_best_seq,
                                                        deadline,
                                                        max_trials=120)
        if val_perm < global_best_val:
            global_best_val = val_perm
            global_best_seq = seq_perm.copy()
            print(f"[overall PermRand] improved best={val_perm:.6f}")

    # ------------------------------------------------------------------
    # 6) Final lightweight annealing polish (remaining time).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=250000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] final best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.copy()
    return global_best_seq.tolist()
