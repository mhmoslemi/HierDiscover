def _smooth_max(a, beta):
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    val = shift + np.log(sum_exp) / beta
    # gradient wrt a:
    p = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(p, rev_a)
    grad_a = 2.0 * full[n-1:n-1+n]  # size n
    # gradient wrt logits:
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return val, grad_logits

"""
Search for a low‑auto‑convolution step‑function.

The algorithm proceeds in three stages for a series of increasing lengths
(e.g. 20000 → 10000 → 4000):

1. **Broad seeding** – a diverse collection of analytic (beta,
   geometric, sine‑window, uniform) and random (Dirichlet,
   piece‑wise‑linear) probability vectors of length *n* is generated.
   All vectors are ℓ₁‑normalised (sum = 1) and the *n*·max(conv(a,a))
   value is evaluated cheaply via an FFT.  The best few seeds are kept.

2. **Smooth‑max L‑BFGS refinement** – each kept seed is optimised by
   minimising a smoothed version of the objective,
   `smooth_max_β(a) = (1/β)·log Σ exp(β·(a*a))`,
   using SciPy’s L‑BFGS‑B on the log‑softmax parametrisation.
   The smoothing parameter β is increased in stages
   (2 → 5 → 15 → 50 → 150) so the optimiser first sees a gentle landscape
   and then focuses on the true maximum.

3. **Targeted hill climbing** – a cheap stochastic move that shifts a
   small amount of mass from a “donor’’ (chosen proportionally to its
   contribution to the current maximal convolution entry) to a random
   “receiver’’.  This often removes the last remaining peaks.

The best sequence found (converted to a plain Python list) is stored in
the global variable ``height_sequence_1`` for possible warm‑starts on
subsequent runs.  The routine respects the supplied time budget
(`budget_s` seconds) and prints progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level FFT convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric progression a_i ∝ r**i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Discrete half‑sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw."""
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑linear distribution via Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective helpers (assume sum(a)=1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


class SmoothMaxObjective:
    """
    Callable returning (smooth_max, grad_logits) for a given logit vector.
    The smooth max is (1/β)·log Σ exp(β·(a*a)) with a = softmax(logits).
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, logits: np.ndarray):
        a = _softmax(logits)                     # stays on the simplex
        conv = _conv_fft(a, a)                   # length 2n‑1
        shift = np.max(conv)
        # stable log‑sum‑exp
        exp_beta = np.exp(self.beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        smooth = shift + np.log(sum_exp) / self.beta

        # weights for the linear combination of conv entries
        w = exp_beta / sum_exp

        # gradient of Σ w_k·conv_k w.r.t. a (central n entries)
        rev_a = a[::-1]
        full = _conv_fft(w, rev_a)               # length 3n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_logits = a * (grad_a - dot)

        return smooth, grad_logits


# ----------------------------------------------------------------------
# Targeted stochastic hill‑climb
# ----------------------------------------------------------------------


def _targeted_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Random mass transfers with donor sampled proportionally to its
    contribution to the current maximal convolution coefficient.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        # indices that actually contribute to this entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total <= 0.0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[TargetHill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from previous run if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (larger n gives a lower theoretical bound)
    # ------------------------------------------------------------------
    lengths = [20000, 10000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        # ------------------------------------------------------------------
        # 1. Assemble a diverse set of seeds (all ℓ₁‑normalised)
        # ------------------------------------------------------------------
        seeds = []

        seeds.append(_uniform(n))
        seeds.append(_sine_window(n))

        # beta‑like shapes (U‑shaped possibilities)
        alphas = np.linspace(0.2, 2.0, 6)
        betas = np.linspace(0.2, 2.0, 6)
        for a_ in alphas:
            for b_ in betas:
                seeds.append(_beta_shape(n, a_, b_))

        # geometric families
        for r in np.linspace(0.3, 0.9, 5):
            seeds.append(_geom(n, r))

        # random Dirichlets (various concentrations)
        rng = np.random.default_rng(2024 + n)
        for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
            for _ in range(2):
                seeds.append(_random_dirichlet(n, conc, rng))

        # random piece‑wise‑linear seeds
        for _ in range(7):
            seeds.append(_random_piecewise_linear(n, np.random.default_rng()))

        # up‑sample previous solution if it exists
        if previous is not None:
            if previous.shape[0] == n:
                seeds.append(previous / previous.sum())
            else:
                # linear interpolation onto the new grid
                x_old = np.linspace(0, 1, previous.shape[0])
                x_new = np.linspace(0, 1, n)
                up = np.interp(x_new, x_old, previous)
                if up.sum() > 0:
                    seeds.append(up / up.sum())

        # ------------------------------------------------------------------
        # 2. Quick evaluation of seeds, keep the best few
        # ------------------------------------------------------------------
        top_candidates = []          # list of (value, array)
        for s in seeds:
            if time.time() > deadline:
                break
            val = _objective(s, n)    # assumes sum(s)=1
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] new global best {val:.6f} (len={n})")
            top_candidates.append((val, s))
        # keep only the best 5 seeds for polishing
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:5]

        # ------------------------------------------------------------------
        # 3. Deterministic refinement of the best seeds
        # ------------------------------------------------------------------
        for idx, (seed_val, seed_seq) in enumerate(top_candidates):
            if time.time() > deadline:
                break
            print(f"\n--- Polishing seed {idx+1}/{len(top_candidates)} (len={n}) ---")
            # start from log‑softmax of the seed
            logits = np.log(seed_seq + 1e-30)

            # schedule of smoothing parameters β
            for beta in [2.0, 5.0, 15.0, 50.0, 150.0]:
                if time.time() > deadline:
                    break
                max_iter = 500  # modest number of L‑BFGS iterations per β
                obj = SmoothMaxObjective(n, beta)

                # L‑BFGS‑B on the smooth objective
                res = minimize(obj,
                               logits,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': max_iter, 'disp': False})
                logits = res.x
                a = _softmax(logits)
                val_actual = _objective(a, n)
                if val_actual < global_best_val:
                    global_best_val = val_actual
                    global_best_seq = a.copy()
                    print(f"[L‑BFGS] β={beta:.1f} new global best {val_actual:.6f}")

                # early‑stop if improvement stagnates
                if res.success and res.fun < 1e-12:
                    break

            # 3d. Targeted stochastic hill‑climb refinement
            if time.time() < deadline - 2.0:
                cur_seq, cur_val = _targeted_hill_climb(
                    global_best_seq, n, deadline, max_iters=8000)
                if cur_val < global_best_val:
                    global_best_val = cur_val
                    global_best_seq = cur_seq.copy()
                    print(f"[Hill] new global best {cur_val:.6f}")

        # ------------------------------------------------------------------
        # Small extra L‑BFGS pass on the current best (cheap sanity step)
        # ------------------------------------------------------------------
        if global_best_seq is not None and time.time() < deadline - 5.0:
            logits = np.log(global_best_seq + 1e-30)
            obj = SmoothMaxObjective(n, beta=200.0)
            res = minimize(obj,
                           logits,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300, 'disp': False})
            a = _softmax(res.x)
            val_extra = _objective(a, n)
            if val_extra < global_best_val:
                global_best_val = val_extra
                global_best_seq = a.copy()
                print(f"[extra‑L‑BFGS] new global best {val_extra:.6f}")

    # ----------------------------------------------------------------------
    # Verification with the provided evaluate_sequence()
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as e:
        print("[verification] sequence rejected:", e)
        # fallback to uniform distribution
        n_fallback = len(global_best_seq) if global_best_seq is not None else 4000
        global_best_seq = _uniform(n_fallback)
        global_best_val = _objective(global_best_seq, n_fallback)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    # store for a possible warm‑start of a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
