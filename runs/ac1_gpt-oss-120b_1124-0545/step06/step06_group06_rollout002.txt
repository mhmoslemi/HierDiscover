"""
Advanced search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (∑a)² .

The algorithm works in four time‑aware stages:

1. **Broad seed generation** – a richer collection of analytically
   motivated families (uniform, triangular, trapezoidal, sine‑window,
   beta, arcsine, inverse‑arcsine, parabola, symmetric exponential,
   geometric, random Dirichlet, sparse).  All candidates are normalised
   to unit sum and the best is kept.

2. **Adam on a smoothed max** – the vector lives on the simplex via a
   log‑softmax.  A log‑sum‑exp with a slowly increasing temperature
   approximates the true max; Adam updates are applied until the global
   deadline is reached (more iterations than the reference code).

3. **Exact sub‑gradient mirror descent** – an exponentiated‑gradient
   step using the true sub‑gradient of the max (averaged over all
   binding indices) further refines the solution.  The learning rate
   schedule is a bit more aggressive than before.

4. **Simulated‑annealing hill climbing** – stochastic neighbourhood
   moves (Dirichlet reshuffling of a few coordinates) give the final
   push out of shallow minima.

The routine automatically chooses a problem size (length) that
balances quality and runtime, runs the stages until the 1000 s budget
is almost exhausted, and finally returns the best sequence found as a
plain ``list`` of ``float`` values ready for ``evaluate_sequence``.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for many calls)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """
    Objective assuming Σseq == 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Peak in centre, linearly decreasing to the edges."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    """Flat top of relative width `width_fraction`, linear ramps on each side."""
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(np.round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Right‑skewed geometric decay a_i ∝ r^i."""
    if r <= 0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometries: r**i + r**(n‑1‑i)."""
    if r <= 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^‑p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine law: a_i ∝ 1/√((i+1)*(n-i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 1.0) * (n - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inv_arcsine_shape(n: int) -> np.ndarray:
    """Inverse arcsine: a_i ∝ √((i+1)*(n-i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt((i + 1.0) * (n - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _parabola_shape(n: int) -> np.ndarray:
    """Parabolic tent: a_i ∝ (i+1)*(n-i)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) * (n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exponential(n: int, tau: float) -> np.ndarray:
    """Symmetric exponential decay centred in the middle."""
    if tau <= 0:
        return _uniform(n)
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-np.abs(i - centre) / tau)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with scalar concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse vector with non‑zeros sampled from a Dirichlet."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a

def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """
    Assemble a diverse set of normalised candidate vectors for length ``n``.
    """
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.2))
    seeds.append(_sine_window(n))
    seeds.append(_arcsine_shape(n))
    seeds.append(_inv_arcsine_shape(n))
    seeds.append(_parabola_shape(n))

    # Symmetric exponential with a couple of τ values
    for tau in (0.15, 0.30, 0.60, 1.20):
        seeds.append(_sym_exponential(n, tau))

    # Geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # Beta‑family grid (covers U‑shaped and bell‑shaped)
    alphas = np.linspace(0.2, 2.0, 10)
    betas = np.linspace(0.2, 2.0, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Sine^γ families
    for gamma in np.linspace(0.5, 4.0, 8):
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
        seeds.append(a / a.sum())

    # Random Dirichlet draws
    for conc in (0.1, 0.3, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns
    for k in (3, 5, 10):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # Previous global best (if compatible length)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on a smoothed version of ``max(conv)``.
    The vector lives on the simplex via a log‑softmax parametrisation.
    The smoothing coefficient (beta) starts small and grows slowly.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5            # initial smooth‑max temperature
    beta_mul = 1.001
    beta_max = 800.0      # allow a rather high temperature

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # increase smoothing (cap at beta_max)
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp over convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 15.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 1.0
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) using a
    slightly more aggressive schedule and averaging over all active
    (max‑attaining) convolution entries.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        # collect all indices that are (almost) maximal
        tol = 1e-12
        active_idx = np.where(conv >= max_val - tol)[0]

        # build sub‑gradient as average of gradients of the active entries
        g = np.zeros_like(a)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        g /= max(1.0, float(len(active_idx)))       # average

        # exponentiated‑gradient step
        lr = 1.0 / np.sqrt(it)                      # more aggressive than 0.6
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                      # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill climbing with a simple annealing schedule.
    A random subset of 2‑4 positions reshuffles its total mass via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 5)          # choose 2,3,4 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Grab previous global best if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a few problem sizes – larger sizes give finer granularity,
    # but we keep them modest so that many iterations still fit in time.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Stage 1 – broad seed scan (fast)
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Scanning seeds for length {n} ---")
        seeds = _collect_seeds(n, rng, previous)

        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                best_n = n
                print(f"[seed] n={n:5d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        best_n = n
        global_best_val = _objective(global_best_seq)

    print(
        f"\n=== Starting refinement on length {best_n} (current best {global_best_val:.6f}) ==="
    )

    # ------------------------------------------------------------------
    # Stage 2 – Adam on smoothed max (more iterations)
    # ------------------------------------------------------------------
    seq, val = _adam_optimize(
        global_best_seq, best_n, deadline, max_iters=60000
    )
    if seq is not None and val < global_best_val:
        global_best_seq = seq.copy()
        global_best_val = val
        print(f"[overall‑Adam] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient mirror descent (long run)
    # ------------------------------------------------------------------
    seq, val = _subgradient_refine(
        global_best_seq, best_n, deadline, max_iters=500000
    )
    if seq is not None and val < global_best_val:
        global_best_seq = seq.copy()
        global_best_val = val
        print(f"[overall‑SubGrad] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    seq, val = _simulated_annealing(
        global_best_seq, best_n, deadline, max_iters=500000
    )
    if seq is not None and val < global_best_val:
        global_best_seq = seq.copy()
        global_best_val = val
        print(f"[overall‑SA] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
