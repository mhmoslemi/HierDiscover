"""
Propose a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑ a)²

The algorithm is a time‑aware multi‑stage optimiser:

1. **Seed collection** – a broad collection of analytically motivated families
   (uniform, triangular, trapezoidal, geometric, power‑law, beta, sine‑window,
   random Dirichlet and sparse patterns).  All candidates are normalised to
   sum = 1 and the best seed is kept.

2. **Adam on a smooth max** – the vector lives on the simplex via a log‑softmax.
   A log‑sum‑exp with a slowly increasing temperature approximates the true max.
   Adam updates run until the global deadline, keeping the best iterate.

3. **Exact sub‑gradient mirror descent** – the true sub‑gradient of the max
   convolution entry is applied in the exponentiated‑gradient domain.
   The loop runs as long as time permits.

4. **Simulated‑annealing hill‑climbing** – random Dirichlet reshuffles of a few
   coordinates (2‑6 indices) with an exponential cooling schedule allow escape
   from shallow minima.

5. **Upscaling** – after optimisation on a medium length (≈4000) the best
   solution is duplicated to a longer length (≈8000) and the whole refinement
   (steps 2‑4) is repeated.  Larger n gives a tighter lower bound
   (≈2·n/(2n‑1) → 1) while keeping the same shape.

All stages respect the supplied time budget (default 1000 s) and the routine
returns the best sequence found as a plain ``list`` of ``float`` values.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for repeated calls)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """
    Internal objective (assuming ∑seq == 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(np.round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_power(n: int, gamma: float) -> np.ndarray:
    """sin^γ window (γ>0) – shape from flat (γ≈0) to sharply peaked."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

# Seed collection

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Assemble a diverse set of normalised candidate vectors for length ``n``.
    """
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.2))
    seeds.append(_sine_window(n))

    # Geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_sym_geom(n, r))

    # Power‑law family
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # Beta‑family grid
    alphas = np.linspace(0.2, 2.0, 10)
    betas = np.linspace(0.2, 2.0, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Sine‑power families
    for gamma in np.linspace(0.3, 5.0, 12):
        seeds.append(_sine_power(n, gamma))

    # Random Dirichlet draws
    for conc in (0.1, 0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns
    for k in (3, 5, 10, 20):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # Previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        pp = previous.astype(np.float64)
        if pp.sum() > 0:
            seeds.append(pp / pp.sum())

    return seeds

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on a smoothed max (log‑sum‑exp) respecting the global deadline.
    Returns the best sequence found and its objective value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits (on the simplex)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5            # initial smooth‑max temperature
    beta_mul = 1.001
    beta_max = 400.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # increase smoothing temperature
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp over convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.8 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) until deadline.
    Returns the best sequence found and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[SubGrad] n={n:5d} it={it:9d} best={cur_val:.6f}")

        # Build sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient step (mirror descent on the simplex)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # prevent overflow
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill climbing with annealing.  At each step a random subset
    of 2‑6 coordinates redistributes its total mass via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # Initial temperature proportional to current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)          # choose between 2 and 6 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:9d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The whole search respects the time budget
    and returns the best sequence found as a plain ``list`` of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # 2 s safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Retrieve a previously stored solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose an initial length – 4000 is a good trade‑off between speed
    # and theoretical lower bound.
    # ------------------------------------------------------------------
    n_initial = 4000
    n_long = 8000  # longer version for the final refinement (if time permits)

    # ------------------------------------------------------------------
    # Stage 0 – broad seed scan (fast, deterministic families + random)
    # ------------------------------------------------------------------
    print(f"\n--- Seed scan for length {n_initial} ---")
    seeds = _collect_seeds(n_initial, rng, previous)

    best_seq = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
            print(f"[seed] n={n_initial:5d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on the initial length
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n_initial)
        best_val = _objective(best_seq)

    seq, val = _adam_optimize(best_seq, n_initial, deadline,
                              max_iters=400_000)
    if seq is not None and val < best_val:
        best_seq, best_val = seq, val
        print(f"[overall‑Adam] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exact sub‑gradient mirror descent (initial length)
    # ------------------------------------------------------------------
    seq, val = _subgradient_refine(best_seq, n_initial, deadline,
                                   max_iters=1_500_000)
    if seq is not None and val < best_val:
        best_seq, best_val = seq, val
        print(f"[overall‑SubGrad] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing hill climbing (initial length)
    # ------------------------------------------------------------------
    seq, val = _simulated_annealing(best_seq, n_initial, deadline,
                                    max_iters=1_200_000)
    if seq is not None and val < best_val:
        best_seq, best_val = seq, val
        print(f"[overall‑SA] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Optional Stage 4 – upscale to a longer length and repeat the refinement
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 100.0:      # only if we still have a comfortable chunk of time
        # Upscale by simple duplication (preserves shape and sum)
        upscale_factor = n_long // n_initial
        if n_long % n_initial == 0:
            long_seq = np.repeat(best_seq, upscale_factor)
        else:
            # General linear interpolation when the factor is not integer
            x_old = np.linspace(0, 1, n_initial, endpoint=False)
            x_new = np.linspace(0, 1, n_long, endpoint=False)
            long_seq = np.interp(x_new, x_old, best_seq)
        long_seq = long_seq / long_seq.sum()   # renormalise to sum = 1

        print(f"\n--- Refinement on longer length {n_long} ---")
        # Adam on longer length
        seq, val = _adam_optimize(long_seq, n_long, deadline,
                                  max_iters=300_000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq, val
            print(f"[long‑Adam] improved global best to {val:.6f}")

        # Sub‑gradient on longer length
        seq, val = _subgradient_refine(best_seq, n_long, deadline,
                                       max_iters=800_000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq, val
            print(f"[long‑SubGrad] improved global best to {val:.6f}")

        # SA on longer length
        seq, val = _simulated_annealing(best_seq, n_long, deadline,
                                        max_iters=600_000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq, val
            print(f"[long‑SA] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional) – makes sure evaluate_sequence does not break.
    # ------------------------------------------------------------------
    try:
        # The external evaluator caps values to [0,1000] and rescales sum,
        # so we can safely pass our normalised vector.
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal external score = {final_score:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest internal objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # Store the result for possible reuse by a subsequent call
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
