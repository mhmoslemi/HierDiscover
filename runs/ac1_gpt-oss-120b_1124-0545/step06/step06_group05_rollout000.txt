def propose_candidate(seed=42,budget_s=1000):

    np.random.seed(seed)
    start time = time.time()
    deadline = start+budget_s-2

    n = 4000

    # generate seeds
    seeds = []
    # arcsine exponents
    for exp in np.linspace(-3.0, -0.1, 15):
        seeds.append(_arcseed(n, exp))

    # beta seeds
    for α in np.linspace(0.1, 5.0, 9):
        for β in np.linspace(0.1, 5.0, 9):
            seeds.append(_beta_seed(n, α, β))

    # uniform distribution also
    seeds.append(_uniform(n))

    # maybe add Dirichlet draws
    rng = np.random.default_rng(seed+1)
    for a in [0.1,0.3,0.5,0.8,1.5,2.0]:
        seeds.append(_dirichlet_seed(n, a, rng))

    # incorporate previous best
    prev = globals().get('height_sequence_1')
    if prev is not None:
        arr = np.asarray(prev,float)
        if arr.shape == (n,):
            seeds.append(arr/arr.sum())

    # Evaluate each seed, keep top few
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(5,len(seed_vals))
    top_seeds = [s for s,_ in seed_vals[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective(best_seq)

    print(f"Initial best objective {best_val}")

    # For each seed, perform LP improvement cycles
    for idx, f in enumerate(top_seeds):
        if time.time() > deadline: break
        f = f.copy()
        f_val = _objective(f)
        # ensure sum=1
        if abs(f.sum()-1.0) > 1e-12:
            f = f / f.sum()
        # iterate
        it = 0
        while time.time() < deadline and it < 20:
            it += 1
            conv_f = _conv_fft(f,f)
            M = conv_f.max()
            # compute indices near max (top 20 entries)
            # get sorted indices descending
            idx_sorted = np.argsort(conv_f)[::-1]
            top_idx = idx_sorted[:max(5, int(0.001*n))] # e.g., 5 or 0.1%?
            # In addition, we might include all indices where conv_f >= M*(1-0.001)
            # but we can use a threshold
            threshold = M * 0.999
            bound_idx = np.where(conv_f >= threshold)[0]
            # combine and take unique (size may be small)
            bound_idx = np.unique(np.concatenate((top_idx,bound_idx)))
            # limit number of constraints to e.g., 30
            if bound_idx.size > 30:
                bound_idx = bound_idx[:30]

            # Solve LP
            # Build A_ub and b_ub
            rows = []
            for k in bound_idx:
                row = np.zeros(n)
                # i ranges from max(0, k-(n-1)) to min(k, n-1)
                i_start = max(0, k-(n-1))
                i_end = min(k, n-1)
                # For i in that range, row[i] = f[k-i]
                # Equivalent slice
                # row[i_start:i_end+1] = f[k - np.arange(i_start,i_end+1)]
                # However direct loop perhaps fine
                idx_i = np.arange(i_start, i_end+1)
                row[idx_i] = f[k - idx_i]
                rows.append(row)

            A_ub = np.vstack(rows) if rows else None
            b_ub = np.full(len(bound_idx), M)

            # Objective: maximize sum(b) -> minimize -sum(b)
            c = np.full(n, -1.0)

            # bounds
            bounds = [(0, None)] * n

            # Solve
            res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                                         method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
            if not res.success:
                # fallback: break
                break
            b = res.x
            sum_b = b.sum()
            if sum_b <= 1e-12:
                break
            g = b / sum_b

            # mixture weight - maybe t = 0.5
            t = 0.5
            f_new = (1 - t) * f + t * g
            f_new = f_new / f_new.sum()
            f_new_val = _objective(f_new)
            # if improvement
            if f_new_val < best_val:
                best_val = f_new_val
                best_seq = f_new.copy()
                print(f"[LP] seed {idx} iter {it} new global best {best_val:.6f}")

            f = f_new
            # if improvement less than threshold break
            if abs(f_new_val - f_val) < 1e-6:
                break
            f_val = f_new_val

    # After LP improvement, final refinement via subgradient
    # apply to best_seq if time

            best_seq = seq.copy()
            print(f"[Refine SubGrad] final best {best_val:.6f}")

    # fallback ensure length and nonnegativity
    if best_seq is None:
        best_seq = _uniform(n)

    print(f"Finished with best objective {best_val}")
    return best_seq.tolist()

"""
Improved search for a low‑∞‑norm self‑convolution step function.

Algorithm
---------
1. **Rich analytic seeds** – generate many deterministic shapes
   (arcsine, Beta, uniform, Dirichlet etc.) on the probability simplex.
2. **Evaluate & keep the best few seeds**.
3. **Linear‑programming driven improvement** – given a candidate `f`,
   solve a small LP

        maximise   Σ b_i
        subject to conv(f,b) ≤ M   (only at the near‑maximal convolution
                                    indices)
        b_i ≥ 0

   The optimum `b` is a vector that can be mixed with `f`
   without raising the current maximal convolution value `M`.
   We try several mixing ratios and keep the best mixture.
   The LP is tiny (≤30 constraints, ≈4000 variables) and solved with
   ``scipy.optimize.linprog`` (HiGHS).  The process is repeated a few
   times, each iteration typically lowering the objective.
4. **Exact sub‑gradient refinement** – finally run the exponentiated‑gradient
   mirror‑descent on the true max‑convolution (as in the previous baseline)
   to polish the solution.
5. Return the best sequence seen (as a plain list of floats).

The whole routine respects the 1000 s budget, checks the wall‑clock
regularly and prints progress.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))², seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse or dense Dirichlet draw (Gamma (α,1) normalised)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """
    Discrete analogue of the continuous arcsine law.
    a_i ∝ ((i+0.5)*(n‑0.5‑i))**exponent  with exponent < 0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide='ignore', invalid='ignore'):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        return _uniform(n)
    return a / s


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) shape on the simplex.
    a_i ∝ i^{α‑1} (n‑i+1)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0))
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient of
    max(conv).  Returns the best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step on the simplex
        lr = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # protect against overflow
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming driven improvement
# ----------------------------------------------------------------------


def _lp_improve(f: np.ndarray,
                max_iters: int,
                deadline: float,
                n: int,
                global_best_val: float,
                global_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps on a probability vector `f`.
    Returns the best sequence seen in this local search and its objective.
    """
    best_seq = f.copy()
    best_val = _objective(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv_f = _conv_fft(f, f)
        M = conv_f.max()

        # Choose a small set of indices where the convolution is close to M
        thresh = M * 0.9995
        bound_idx = np.where(conv_f >= thresh)[0]
        # also add a few absolute top indices (guarantee at least 5)
        topk = np.argsort(conv_f)[::-1][:5]
        bound_idx = np.unique(np.concatenate((bound_idx, topk)))
        # limit to at most 30 constraints to keep LP tiny
        if bound_idx.size > 30:
            bound_idx = bound_idx[:30]

        # Build linear constraints: conv(f, b) <= M for selected indices
        rows = []
        for k in bound_idx:
            row = np.zeros(n, dtype=np.float64)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            # row[i] = f[k-i]
            row[idx_i] = f[k - idx_i]
            rows.append(row)

        A_ub = np.vstack(rows) if rows else None
        b_ub = np.full(len(bound_idx), M, dtype=np.float64)

        # Objective: maximise sum(b)  -> minimise -sum(b)
        c = np.full(n, -1.0, dtype=np.float64)
        bounds = [(0, None)] * n

        # Solve the LP (HiGHS) – give it a few seconds at most
        remaining = max(0.0, deadline - time.time())
        opts = {"presolve": True, "time_limit": remaining}
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                    bounds=bounds, method='highs',
                                    options=opts)

        if not res.success:
            # If LP fails, abort further LP steps for this seed
            break

        b = res.x
        if b.sum() < 1e-12:
            break

        g = b / b.sum()          # bring g onto the simplex

        # Try several mixing coefficients and keep the best mixture
        mix_candidates = [0.1, 0.25, 0.4, 0.6, 0.8]
        local_best_val = best_val
        local_best_seq = best_seq

        for t in mix_candidates:
            mix = (1 - t) * f + t * g
            mix = mix / mix.sum()
            val = _objective(mix)
            if val < local_best_val:
                local_best_val = val
                local_best_seq = mix.copy()

        if local_best_val < best_val - 1e-9:      # noticeable improvement
            best_val = local_best_val
            best_seq = local_best_seq.copy()
            f = best_seq.copy()                  # continue from improved point
            print(f"[LP] it={it:3d} new best={best_val:.6f}")
        else:
            # No improvement – stop early
            break

        # Early exit if we already beat the global best by a comfortable margin
        if best_val < global_best_val - 1e-6:
            global_best_val = best_val
            global_best_seq = best_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # tiny safety margin

    # ------------------------------------------------------------------
    # Problem dimension (4000 performed well in previous runs)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Collect a wide variety of analytic seeds
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 100)
    seeds = []

    # Uniform
    seeds.append(_uniform(n))

    # Arcsine‑type (negative exponents)
    for exp in np.linspace(-2.8, -0.2, 13):
        seeds.append(_arcseed(n, exp))

    # Beta family (α,β) – explore a grid
    for α in np.linspace(0.2, 4.0, 7):
        for β in np.linspace(0.2, 4.0, 7):
            seeds.append(_beta_seed(n, α, β))

    # Geometric & power‑law families
    for r in np.linspace(0.85, 0.999, 6):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.5, 6):
        seeds.append(_power(n, p))

    # Dirichlet draws – from sparse to dense
    for a in [0.1, 0.3, 0.6, 1.0, 2.0]:
        seeds.append(_dirichlet_seed(n, a, rng))

    # Include the previously discovered construction if available
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = arr / max(arr.sum(), 1e-12)
                seeds.append(arr)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Evaluate seeds, keep the top few
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(5, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"[Init] best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP‑based improvement on each promising seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- LP improvement for seed {idx} ---")
        # Normalise to a probability vector (sum = 1)
        f = seed_vec.copy()
        f = f / max(f.sum(), 1e-12)

        # Do a handful of LP refinement steps
        f, val = _lp_improve(f,
                             max_iters=12,
                             deadline=deadline,
                             n=n,
                             global_best_val=global_best_val,
                             global_best_seq=global_best_seq)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = f.copy()
            print(f"[Overall LP] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient refinement (mirror‑descent)
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_seq is not None:
        print("\n--- Sub‑gradient refinement ---")
        refined_seq, refined_val = _subgradient_refine(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=90000)
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[Overall SubGrad] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in case nothing was found (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[Fallback] using uniform distribution")
        global_best_val = _objective(global_best_seq)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # Return a plain Python list as required by the grader
    return global_best_seq.tolist()
