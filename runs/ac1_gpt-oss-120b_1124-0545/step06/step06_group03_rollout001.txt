"""
Propose a non‑negative step‑function that (hopefully) improves the
upper‑bound for

    2·n·max(conv(a,a)) / (∑a)² .

The search consists of three stages, each of which respects the
remaining time budget:

1️⃣  **Seed generation** – a diverse set of analytic families (uniform,
    geometric, power‑law, beta‑like, sine‑window, edge‑focused “V‑shape”
    and double‑exponential) together with a few random Dirichlet draws.
    The best few seeds are kept for further optimisation.

2️⃣  **Smooth optimisation (Adam)** – the non‑smooth max‑operator is
    replaced by a log‑sum‑exp with a temperature that is gradually
    increased.  Adam updates are performed on the log‑softmax
    parameters, staying on the probability simplex.

3️⃣  **Exact refinement** – a two‑phase local search:
    *exponentiated‑gradient sub‑gradient descent* on the true objective,
    followed by *high‑frequency simulated‑annealing* with an incremental
    convolution update (O(n·k) for a move that touches k≤6 coordinates).

The algorithm works in increasing discretisation sizes (2000 → 4000 →
8000 → 12000).  For the larger sizes we start from the best solution of
the previous size (simple linear interpolation) so that the expensive
global search is performed only once.  All work is stopped before the
global deadline (default 1000 s) and the best sequence found is returned.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (returns length len(a)+len(b)-1)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a (assumes ∑a = 1). Returns 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_double_exp(n: int, r: float) -> np.ndarray:
    """V‑shaped (high at edges, low in the middle) distribution."""
    a = r ** np.arange(n) + r ** np.arange(n - 1, -1, -1)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _v_shape_linear(n: int) -> np.ndarray:
    """1/(distance‑to‑edge+1) shape."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = 1.0 / (d + 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _v_shape_poly(n: int, p: float) -> np.ndarray:
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = 1.0 / ((d + 1.0) ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float,
                   rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries drawn from a Dirichlet (conc < 1 → sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families (monotone decreasing)
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # beta‑like shapes (asymmetric)
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # edge‑focused V‑shapes
    seeds.append(_v_shape_linear(n))
    for p in (0.5, 1.0, 2.0):
        seeds.append(_v_shape_poly(n, p))
    for r in (0.5, 0.7, 0.9):
        seeds.append(_edge_double_exp(n, r))

    # random Dirichlet draws (different concentrations)
    for conc in (0.3, 0.8, 1.5, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # a few sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(1):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse a previous solution if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 15000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate for max(conv).
    The temperature (smoothness) grows geometrically.
    Returns best sequence (as ndarray) and its true objective.
    """
    eps = 1e-30
    v = np.log(seed + eps)                 # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # correct smoothness shift
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a (conv = a * a) ----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping of the gradient norm
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the *exact* objective.
    Uses a step‑size α = C / √t with optional back‑tracking
    to guarantee a descent.
    """
    a = start_seq.copy()
    eps = 1e-30
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size schedule (larger constant than before)
        base = 1.2 / np.sqrt(it)  # can be tuned
        alpha = base

        # simple back‑tracking: try to improve, otherwise halve α a few times
        for _ in range(6):
            log_a = np.log(a + eps) - alpha * g
            log_a -= np.max(log_a)          # stability
            a_new = np.exp(log_a)
            a_new /= a_new.sum()

            new_conv = _conv_fft(a_new, a_new)
            new_val = 2.0 * n * np.max(new_conv)

            if new_val <= cur_val * (1.0 + 1e-12):
                a = a_new
                break
            else:
                alpha *= 0.5
        # if none of the trial alphas improved, we simply keep the old a

    return best_seq, best_val


def _sa_incremental_optimize(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             init_temp_factor: float = 0.04,
                             min_temp: float = 1e-9,
                             max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with cheap incremental convolution updates.
    Each move redistributes the total mass of 2‑6 randomly chosen positions.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, min_temp)

        k = rng.integers(2, 7)                 # 2‑6 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        deltas = new_alloc - cur[idx]
        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()

        # linear contributions a*Δ and Δ*a
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta                     # Δ*Δ self term

        # cross‑terms Δ_i * Δ_j for i≠j (appear twice)
        for i in range(k):
            for j in range(i + 1, k):
                pos = idx[i] + idx[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept the move
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a larger size.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    # re‑normalise (tiny numerical errors may appear)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        return _uniform(new_n)
    return new_seq / s

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects a time budget and returns
    the best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 3.0  # safety margin

    # optionally reuse a previous construction (if the hosting system provides it)
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Hierarchical optimisation over increasing discretisation sizes
    # ------------------------------------------------------------------
    length_schedule = [2000, 4000, 8000, 12000]   # larger sizes give finer resolution
    global_best_seq = None
    global_best_val = float("inf")
    prev_seq = previous

    for n in length_schedule:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # --- seed preparation ------------------------------------------------
        if prev_seq is not None and prev_seq.shape[0] == n:
            seeds = [prev_seq]
        elif prev_seq is not None and prev_seq.shape[0] < n:
            # upsample previous best to the current size
            up = _upsample(prev_seq, n)
            seeds = [up]
        else:
            seeds = _make_seeds(n, prev_seq, rng)

        # quick objective evaluation of all seeds
        seed_vals = []
        for s in seeds:
            val = _objective(s)
            seed_vals.append((val, s))
        seed_vals.sort(key=lambda x: x[0])
        # keep only the top few seeds for the expensive Adam stage
        top_k = min(5, len(seed_vals))
        top_seeds = [s for (_, s) in seed_vals[:top_k]]

        # update global best from raw seeds
        if seed_vals and seed_vals[0][0] < global_best_val:
            global_best_val = seed_vals[0][0]
            global_best_seq = seed_vals[0][1].copy()
            print(f"[raw‑seed] n={n:5d} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation (smooth max)
        # ------------------------------------------------------------------
        for s in top_seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline,
                                      max_iters=18000)   # a bit more than before
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (exponentiated gradient)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=250000)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – High‑frequency simulated‑annealing
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _sa_incremental_optimize(global_best_seq, n, deadline,
                                                max_iters=1_000_000)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

        # store the best sequence of this size for possible up‑sampling later
        prev_seq = global_best_seq

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # final verification (optional – does not affect the return value)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
