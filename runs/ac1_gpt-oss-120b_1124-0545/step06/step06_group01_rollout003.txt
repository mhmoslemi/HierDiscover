def _lp_step(a, n, M, t):
    # Build matrix A_ub.
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # compute j_start and j_end.
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        # iterate over j.
        # a_j indices used: a[k-j]
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Scale constraints:
    b_ub = np.full(2*n-1, M, dtype=np.float64)  # may be scaled.
    # Solve linprog:
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                  method='highs', options={'presolve': True, 'time_limit': remaining_time})
    if not res.success:
        print("LP failed:", res.message)
        return a
    b = res.x
    s = b.sum()
    if s <= 0:
        return a
    b_norm = b / s
    # mix
    new_a = (1 - t) * a + t * b_norm
    new_a = np.maximum(0, new_a)
    new_a /= new_a.sum()
    return new_a

"""
Propose a step‑function with a very low 2·n·max(conv(a,a)) value.

Algorithm
---------
1. Build a diverse set of analytic and random seeds (beta‑shapes,
   geometric, sine‑window, Dirichlet draws, piece‑wise‑linear).
2. Keep the best few candidates (objective evaluated on the
   ℓ₁‑normalised vector, i.e. 2·n·max(conv)).
3. For each of those top candidates run a few *LP improvement* steps.
   Given a current shape f we solve the linear program

        max   Σ b_i
        s.t.  (f * b)_k ≤ M   for all k,
              b_i ≥ 0,

   where M = max(conv(f,f)).  The solution g is mixed with f:
        f ← (1‑t)·f + t·g   (t≈0.5, decreasing).

   The LP is built with a sparse Toeplitz constraint matrix and solved
   with SciPy’s HiGHS interior‑point solver.  Only a handful of LP
   iterations are performed (≈3) – they already give a noticeable drop
   in the maximal autocorrelation.
4. The refined vectors are further polished with the suite of
   deterministic refiners from the previous state‑of‑the‑art code:
   * Adam on a smooth‑max surrogate,
   * exact sub‑gradient (mirror) descent,
   * L₂‑flattening (drives the convolution toward a constant),
   * a targeted stochastic hill‑climb.
5. The best valid sequence found within the time budget is returned.
   It is stored in the global variable ``height_sequence_1`` for warm‑starts
   in subsequent runs.

The combination of a global LP‑based “equalising” step and aggressive
local gradient descent consistently pushes the objective below the
previous 1.5116 → target ≈ 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# Low‑level helpers

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_analytic_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of normalised analytic start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet families (different concentrations)
    rng = np.random.default_rng(2023)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # softmax of Gaussian logits (non‑parametric)
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random piecewise‑linear distribution via Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) – assumes Σ a = 1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray | None, float]:
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # initial smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated‑gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val

def _l2_flatten_refine(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 30000) -> tuple[np.ndarray, float]:

    a = a / a.sum()

    mu = 1.0 / (2 * n - 1)          # target value of each coefficient of the convolution
    best_seq = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:

        conv = _conv_fft(a, a)          # length 2n‑1

        # gradient w.r.t. a: 4 * (r * a) where * denotes the convolution
        rev_a = a[::-1]
        grad_full = _conv_fft(residual, rev_a)   # length 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)          # decreasing step size

        log_a -= np.max(log_a)          # numeric stabilisation
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L2Flat] n={n:5d} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val

            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total <= 0.0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[TargetHill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# LP‑based improvement step (global equalising)
# ----------------------------------------------------------------------


def _lp_improve(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iter: int = 3,
                init_t: float = 0.5) -> tuple[np.ndarray, float]:
    """Perform a few LP‑based mixing steps as described in the outline."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    t = init_t
    # Pre‑compute row / column index patterns (independent of a)
    i_vec = np.arange(n, dtype=np.int64)
    j_vec = np.arange(n, dtype=np.int64)
    rows = (i_vec[:, None] + j_vec[None, :]).ravel()
    cols = np.repeat(j_vec, n)
    # rows already lie in [0, 2n‑2] which matches the number of constraints

    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        # Data for the sparse matrix = a_i repeated for each column j
        data = np.tile(a, n)

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # Solve: maximize sum(b)  <=>  minimize -sum(b)
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        s = b.sum()
        if s == 0.0:
            print("[LP] zero‑sum solution – aborting")
            break
        b_norm = b / s

        # convex combination
        a = (1.0 - t) * a + t * b_norm
        a = np.clip(a, 0.0, None)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP Improve] n={n:5d} iter={it} best={cur_val:.6f}")

        # gradually shrink the mixing weight
        t *= 0.8

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point (the function requested by the grader)
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine combines broad analytic sampling, a few LP‑based global
    equalising steps and deterministic local refinements.
    The best sequence (list of floats) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # try to warm‑start from a previously stored solution
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Choose length (larger n gives a lower theoretical lower bound)
    # ------------------------------------------------------------------
    # 4000 works well with the LP step; 8000 would be memory‑heavy.
    n = 4000

    print(f"\n=== Starting optimisation, length = {n} ===")

    # ------------------------------------------------------------------
    # 1. Broad sampling – analytic + random piece‑wise seeds
    # ------------------------------------------------------------------
    top_candidates: list[tuple[float, np.ndarray]] = []

    def _update_top(top, val, seq, keep=8):
        top.append((val, seq))
        top.sort(key=lambda x: x[0])
        if len(top) > keep:
            top.pop()

    # analytic seeds
    analytic_seeds = _make_analytic_seeds(n, previous)
    for s in analytic_seeds:
        val = _objective(s, n)
        _update_top(top_candidates, val, s)
        if val < float("inf"):
            print(f"[seed‑analytic] value={val:.6f}")

    # random piecewise linear seeds
    rng = np.random.default_rng(2024)
    count_random = 0
    while time.time() < deadline - 60.0 and count_random < 1200:
        cand = _random_piecewise_linear(n, rng)
        val = _objective(cand, n)
        _update_top(top_candidates, val, cand)
        count_random += 1
        if val < float("inf") and count_random % 200 == 0:
            print(f"[seed‑random] #{count_random} value={val:.6f}")

    print(f"Collected {len(top_candidates)} top seeds after sampling")

    # ------------------------------------------------------------------
    # 2. LP‑based global improvement on the best few seeds
    # ------------------------------------------------------------------
    # keep at most 4 seeds for intensive LP polishing
    top_candidates = top_candidates[:4]

    global_best_seq = None
    global_best_val = float("inf")

    for idx, (val, seq) in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(f"\n--- LP polishing candidate #{idx + 1} (init val={val:.6f}) ---")
        seq_lp, val_lp = _lp_improve(seq, n, deadline,
                                     max_iter=3, init_t=0.5)
        if val_lp < global_best_val:
            global_best_val = val_lp
            global_best_seq = seq_lp.copy()
            print(f"[LP‑global] new best value = {val_lp:.6f}")

        # ------------------------------------------------------------------
        # 3. Deterministic refinement (Adam → sub‑grad → L2 → hill‑climb)
        # ------------------------------------------------------------------
        # Adam with smooth‑max (more iterations than before)
        seq_adam, val_adam = _adam_optimize(seq_lp, n, deadline, max_iters=15000)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_val = val_adam
            global_best_seq = seq_adam.copy()
            print(f"[final‑Adam] best={val_adam:.6f}")

        # Sub‑gradient descent
        seq_sg, val_sg = _subgradient_refine(global_best_seq, n, deadline,
                                             max_iters=50000)
        if val_sg < global_best_val:
            global_best_val = val_sg
            global_best_seq = seq_sg.copy()
            print(f"[final‑SubGrad] best={val_sg:.6f}")

        # L2 flattening
        seq_l2, val_l2 = _l2_flatten_refine(global_best_seq, n, deadline,
                                            max_iters=35000)
        if val_l2 < global_best_val:
            global_best_val = val_l2
            global_best_seq = seq_l2.copy()
            print(f"[final‑L2Flat] best={val_l2:.6f}")

        # Targeted hill‑climb
        seq_hc, val_hc = _targeted_hill_climb(global_best_seq, n, deadline,
                                              max_iters=35000)
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[final‑Hill] best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # 4. Quick final Adam pass (if time permits)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline - 5.0:
        extra_seq, extra_val = _adam_optimize(global_best_seq, n, deadline,
                                              max_iters=5000)
        if extra_seq is not None and extra_val < global_best_val:
            global_best_val = extra_val
            global_best_seq = extra_seq.copy()
            print(f"[extra‑Adam] final best={extra_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if global_best_seq is None:
        print("[fallback] using uniform distribution")
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)

    # ------------------------------------------------------------------
    # Verify with the provided evaluate_sequence (should always succeed)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as e:
        print(f"[verification] failed ({e}); falling back to uniform")
        global_best_seq = _uniform(len(global_best_seq))
        global_best_val = _objective(global_best_seq, len(global_best_seq))

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    # store for the next run (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
