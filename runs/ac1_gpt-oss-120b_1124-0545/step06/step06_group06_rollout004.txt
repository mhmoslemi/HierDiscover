"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (∑a)²

The algorithm works in a time‑aware loop (default budget = 1000 s).

Main ideas
----------
1. **Broad seed scan** – a handful of analytic families (uniform,
   triangular, geometric, beta, sine‑window, random Dirichlet, sparse
   patterns) are evaluated for a few different lengths.  The best seed
   becomes the starting point.

2. **Variance‑Adam** – we first minimise the *variance* of the
   convolution (i.e. make the autocorrelation as flat as possible) by
   Adam on a smooth L2 loss:
       L = Σ (conv_k – μ)² ,   μ = 1/(2n‑1).
   The vector lives on the simplex via a log‑softmax parametrisation.
   A smooth loss is much easier to optimise than the raw max.

3. **Exact sub‑gradient descent** – after the variance stage we
   switch to the true max‑objective using the mirror‑descent sub‑gradient
   routine from the previous state‑of‑the‑art code.

4. **Simulated‑annealing hill climbing** – a cheap stochastic
   neighbourhood (reshuffle a few coordinates with a Dirichlet draw)
   helps to escape shallow minima.

5. **Iterative restarts** – while time permits we repeatedly
   perturb the current best solution, run the three refinement stages,
   and keep the best overall result.

All heavy work (convolution, gradients) is performed with FFT‑based
convolution for speed.  The routine respects the supplied time budget
and finally returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """
    Original objective (sum is assumed to be 1, i.e. the scale‑invariant
    part of the evaluate_sequence formula).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (already normalised to sum = 1)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Peak in the centre, linearly decreasing to the edges."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    """Flat top of relative width `width_fraction`, linear ramps on each side."""
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(np.round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Right‑skewed geometric decay a_i ∝ r^i."""
    if r <= 0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometries: r**i + r**(n‑1‑i)."""
    if r <= 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^‑p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a dense Dirichlet vector."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse non‑negative vector; non‑zeros drawn from a Dirichlet."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Create a modest but diverse collection of normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.2))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.98, 6):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # power‑law family
    for p in np.linspace(0.0, 3.0, 7):
        seeds.append(_power(n, p))

    # beta family (coarse grid)
    alphas = np.linspace(0.3, 2.0, 5)
    betas = np.linspace(0.3, 2.0, 5)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # sine^γ families
    for gamma in np.linspace(0.5, 4.0, 5):
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
        seeds.append(a / a.sum())

    # random Dirichlet draws
    for conc in (0.1, 0.5, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous best (if compatible length)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds

) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation of the *variance* loss
        L = Σ (conv_k – μ)² ,   μ = 1/(2n‑1)

    The vector lives on the simplex via a log‑softmax parametrisation.
    Returns the best sequence encountered (in terms of the original max‑objective)
    and its objective value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    best_seq = None
    best_val = float("inf")
    mu = 1.0 / (2 * n - 1)               # target mean of the convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                  # stays on simplex, sum == 1
        conv = _conv_fft(a, a)           # length 2n‑1
        diff = conv - mu
        loss = np.dot(diff, diff)        # L2 variance loss (unused except for monitoring)

        # track best w.r.t. original objective
        cur_obj = _objective(a)
        if cur_obj < best_val:
            best_val = cur_obj
            best_seq = a.copy()
            print(f"[AdamVar] n={n:5d} it={it:6d} obj={cur_obj:.6f} loss={loss:.6e}")

        # gradient of loss w.r.t. a
        # g = dL/d(conv) = 2 * diff
        g = 2.0 * diff
        rev_a = a[::-1]
        # full convolution of rev_a (length n) with g (length 2n‑1) → length 3n‑2
        full = _conv_fft(rev_a, g)
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # factor 2 comes from ∂conv/∂a = 2*a_{k-i}
        # convert to log‑softmax space (mirror‑descent style)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # simple clip to avoid exploding steps
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent using exponentiated‑gradient updates.
    Learning rate decays as 1/√t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max at lag max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill climbing with a simple annealing schedule.
    A random subset of 2‑4 positions reshuffles its total mass via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 5)          # choose 2, 3 or 4 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a ``list`` of ``float``.
    """
    # deterministic reproducibility
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a modest safety margin for the final conversion / printing
    deadline = start_time + budget_s - 2.0

    # possible previous global best (provided by the harness)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – quick seed scan for a few lengths
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000]  # modest set – larger lengths are tried later
    global_best_seq = None
    global_best_val = float("inf")
    best_len = None

    print("\n=== Stage 0 – seed scan ===")
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n-- n = {n} --")
        seeds = _collect_seeds(n, rng, previous)
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                best_len = n
                print(f"[seed] n={n:5d} val={val:.6f}")

    # Fallback to a uniform sequence if nothing has been found (unlikely)
    if global_best_seq is None:
        best_len = 2000
        global_best_seq = _uniform(best_len)
        global_best_val = _objective(global_best_seq)

    print(
        f"\n=== Initial best: n={best_len}  obj={global_best_val:.6f} ==="
    )

    # ------------------------------------------------------------------
    # Stage 1 – iterative refinement loop (restarts until budget is exhausted)
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 5.0:
        iteration += 1
        # Occasionally try a larger length if time permits
        if iteration % 4 == 0 and best_len < 12000:
            n = best_len + 2000
        else:
            n = best_len

        # Initialise a starting vector:
        if global_best_seq is not None and len(global_best_seq) == n:
            # perturb the current best a little (convex combination with a fresh Dirichlet)
            alpha = rng.uniform(0.0, 0.2)
            pert = rng.dirichlet(np.full(n, 1.0))
            start_vec = (1.0 - alpha) * global_best_seq + alpha * pert
            start_vec = start_vec / start_vec.sum()
        else:
            # pure random Dirichlet for a fresh length
            start_vec = _random_dirichlet(n, conc=0.5, rng=rng)

        # --------------------------------------------------------------
        # Variant 1 – variance Adam (smooth L2 loss)
        # --------------------------------------------------------------
        seq1, val1 = _adam_variance_optimize(
            start_vec, n, deadline, max_iters=12000
        )
        if seq1 is not None and val1 < global_best_val:
            global_best_seq = seq1.copy()
            global_best_val = val1
            best_len = n
            print(f"[global‑AdamVar] improved to {global_best_val:.6f}")

        # --------------------------------------------------------------
        # Variant 2 – exact sub‑gradient descent
        # --------------------------------------------------------------
        seq2, val2 = _subgradient_refine(
            global_best_seq, n, deadline, max_iters=100000
        )
        if seq2 is not None and val2 < global_best_val:
            global_best_seq = seq2.copy()
            global_best_val = val2
            best_len = n
            print(f"[global‑SubGrad] improved to {global_best_val:.6f}")

        # --------------------------------------------------------------
        # Variant 3 – simulated‑annealing hill climbing
        # --------------------------------------------------------------
        seq3, val3 = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=200000
        )
        if seq3 is not None and val3 < global_best_val:
            global_best_seq = seq3.copy()
            global_best_val = val3
            best_len = n
            print(f"[global‑SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
