def evaluate_sequence(sequence: list[float]) -> float:
    # ... security checks
    sequence = [float(x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    return float(2 * n * max_b / (sum_a**2))

"""
Low‑autocorrelation step‑function via alternating linear‑programming
------------------------------------------------------------------

The function `propose_candidate` builds a non‑negative sequence `a`
(minimum length 4000) that minimises the evaluation function

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Alternating LP** – Starting from a feasible vector (uniform or the
   previously found construction) we repeatedly
   *solve a tiny linear program* that maximises the ℓ₁‑norm of a new
   vector `g` while keeping all active convolution constraints
   `conv(a,g) ≤ M` (where `M` is the current maximal convolution).
   The LP has at most a few dozen constraints (the active set) and
   therefore solves in a fraction of a second even for `n = 4000`.
   The new vector is mixed with the current one (`a ← (1‑t)·a + t·g`);
   this strictly reduces the maximal convolution and converges within
   a handful of iterations.

2. **Exact sub‑gradient polishing** – After the LP phase we switch to
   exponentiated‑gradient (mirror‑descent) on the true non‑smooth
   objective.  This refines the solution without the smoothing
   approximation.

3. **Light Metropolis annealing** – If time remains a cheap Metropolis
   walk performs tiny mass‑swaps between random positions; this can
   escape tiny local plateaus.

All steps respect the 1000 s time budget, use only NumPy/SciPy and never
perform I/O.  The final best sequence is stored in the global variable
`height_sequence_1` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (linprog + sparse)
# ----------------------------------------------------------------------
try:
    from scipy import optimize, sparse
    SCIPY_LP_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_LP_AVAILABLE = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Sub‑gradient refinement (exponentiated gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact non‑smooth objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - i_idx
            g[i_idx] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)        # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light Metropolis annealing (final polish)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """Very cheap Metropolis walk that swaps a tiny amount of mass."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional Gaussian jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = np.full(n, 1.0 / n)
            else:
                new_seq /= new_seq.sum()
                new_seq *= np.sqrt(2.0 * n)   # keep the proper scale
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Linear‑programming step (active‑set)
# ----------------------------------------------------------------------


def _solve_lp_step(f: np.ndarray,
                   M: float,
                   active_idx: np.ndarray,
                   sum_target: float) -> np.ndarray | None:
    """
    Solve
        max 1ᵀ g
        s.t. conv(f,g)[k] ≤ M   for all k in active_idx
             g ≥ 0
    Returns the optimal g (un‑scaled).  If the LP fails, returns None.
    """
    if not SCIPY_LP_AVAILABLE:
        return None

    n = f.size
    L = active_idx.size
    rows = []
    cols = []
    data = []

    for row, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        j = np.arange(i_start, i_end + 1, dtype=int)
        f_vals = f[k - j]               # coefficient of g_j
        rows.extend([row] * len(j))
        cols.extend(j.tolist())
        data.extend(f_vals.tolist())

    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(L, n))
    b_ub = np.full(L, M + 1e-12)       # tiny slack for numerical safety
    c = -np.ones(n)                    # maximize sum(g)  → minimize –sum(g)
    bounds = [(0, None)] * n

    # Using the HiGHS solver (very fast for this size)
    res = optimize.linprog(c,
                           A_ub=A_ub,
                           b_ub=b_ub,
                           bounds=bounds,
                           method='highs',
                           options={'presolve': True, 'disp': False})

    if not res.success:
        return None
    return res.x

# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget (in
    seconds) and returns the best sequence discovered as a plain list of
    floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose length – 4000 works well.
    # ------------------------------------------------------------------
    n = 4000
    sum_target = math.sqrt(2.0 * n)   # normalization that makes the true objective = max(conv)

    # ------------------------------------------------------------------
    # Initialise from a previous construction if available, else uniform.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                f = prev_arr.copy()
                f = np.clip(f, 0.0, None)
                if f.sum() > 0.0:
                    f = f * (sum_target / f.sum())
                else:
                    f = np.full(n, sum_target / n)
            else:
                f = np.full(n, sum_target / n)
        except Exception:  # pragma: no cover
            f = np.full(n, sum_target / n)
    else:
        f = np.full(n, sum_target / n)

    best_seq = f.copy()
    best_val = float("inf")

    # store initial value
    cur_val = _objective(f)
    best_val = cur_val
    print(f"[init] objective ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1) Alternating LP phase (active‑set)
    # ------------------------------------------------------------------
    max_lp_iters = 30
    t_mix = 0.25                     # mixing coefficient (can be tuned)

    for it in range(1, max_lp_iters + 1):
        if time.time() > deadline:
            break

        # current convolution & maximal entry
        conv_f = _conv_fft(f, f)
        M = conv_f.max()

        # active set – indices whose convolution is essentially maximal
        rel_tol = 1e-9
        active_idx = np.where(conv_f >= M - rel_tol * max(1.0, M))[0]

        # solve the tiny LP
        g0 = _solve_lp_step(f, M, active_idx, sum_target)
        if g0 is None:
            print("[LP] solver failed – exiting LP loop")
            break

        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print("[LP] zero‑sum solution – exiting")
            break

        # scale g back to the required ℓ₁‑norm
        scale = sum_target / sum_g0
        g = g0 * scale

        # convex combination (mixing)
        f = (1.0 - t_mix) * f + t_mix * g

        # renormalise just in case of tiny drift
        f *= sum_target / f.sum()

        cur_val = _objective(f)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
            print(f"[LP it={it}] improved objective ≈ {best_val:.6f}")

        # optional early stop if we already beat the target
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # 2) Sub‑gradient polishing (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_refine(best_seq, n, deadline, max_iters=30000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[SubGrad] polished best ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Light annealing (final polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=200000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Anneal] final best ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, sum_target / n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    # The grader expects a plain Python list
    return best_seq.tolist()
