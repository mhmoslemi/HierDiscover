"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search combines a broad analytic/random sampling with a dedicated
linear‑programming polishing step.  The LP maximises the total mass that
can be added without raising the current maximal auto‑convolution value.
Mixing the LP direction with the current vector flattens the convolution.
After a few LP iterations a short Adam / sub‑gradient refinement is
performed.  The routine respects the supplied time budget and returns the
best sequence found (as a plain ``list[float]``).  All numbers are
non‑negative; the evaluation function will clamp them to [0,1000] if
necessary.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Fast convolution (full length) via real FFT
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_analytic_seeds(
    n: int, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Collect a diverse set of ℓ₁‑normalised analytic start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes – fairly fine grid
    alphas = np.linspace(0.1, 2.5, 9)
    betas = np.linspace(0.1, 2.5, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet families
    rng = np.random.default_rng(2023)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # softmax of Gaussian logits (non‑parametric)
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Random piece‑wise‑linear generator
# ----------------------------------------------------------------------
def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random piecewise‑linear distribution via Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Small helper to keep top‑K candidates
# ----------------------------------------------------------------------
def _update_top(
    top_list: list[tuple[float, np.ndarray]],
    cand_val: float,
    cand_seq: np.ndarray,
    max_keep: int = 8,
) -> None:
    top_list.append((cand_val, cand_seq))
    top_list.sort(key=lambda x: x[0])  # ascending by value
    if len(top_list) > max_keep:
        top_list.pop()


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray | None, float]:
    v = np.log(seed + 1e-30)  # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    beta = 5.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weighting
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k · conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # back‑prop through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) on the true max
# ----------------------------------------------------------------------
def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent step (exponentiated‑gradient)
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# LP helper – pattern for convolution matrix
# ----------------------------------------------------------------------
_LP_PATTERNS: dict[int, tuple[np.ndarray, np.ndarray]] = {}


def _lp_pattern(n: int) -> tuple[np.ndarray, np.ndarray]:
    """Return (row_idx, col_idx) for the n×n convolution matrix."""
    if n in _LP_PATTERNS:
        return _LP_PATTERNS[n]
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel().astype(np.int32)
    cols = np.repeat(np.arange(n), n).astype(np.int32)
    _LP_PATTERNS[n] = (rows, cols)
    return rows, cols


def _lp_optimize(
    f: np.ndarray,
    M: float,
    n: int,
    rows: np.ndarray,
    cols: np.ndarray,
) -> np.ndarray | None:
    """
    Solve
        max   Σ g_j
        s.t.  conv(f, g) ≤ M   (element‑wise)
              g ≥ 0
    Returns g normalised to the simplex (or None on failure).
    """
    # build A • g ≤ M  (A is the convolution matrix of f)
    data = np.repeat(f, n)  # each column i gets weight f_i in all n rows
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)  # linprog minimises; negative sign gives maximisation
    ub = M * np.ones(2 * n - 1, dtype=np.float64)

    try:
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": 5.0},
        )
    except Exception:  # pragma: no cover
        return None

    if not res.success:
        return None

    g = res.x
    total = g.sum()
    if total <= 0.0:
        return None
    return g / total  # bring back to the simplex


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine mixes a broad random/analytic
    seeding, a dedicated LP‑polishing phase (performed on a moderate
    length, usually 2000) and a short Adam / sub‑gradient finish.
    The best sequence encountered is returned as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0  # safety margin

    # warm‑start from previous run if available
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # we will work with two lengths – a smaller one for intensive LP polishing
    lengths = [2000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    # simple helper to keep a sorted top‑K list
    def _keep_top(top, val, seq):
        _update_top(top, val, seq, max_keep=8)

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        top_candidates: list[tuple[float, np.ndarray]] = []

        # --------------------------------------------------------------
        # 1. Analytic and random seeds
        # --------------------------------------------------------------
        analytic_seeds = _make_analytic_seeds(n, previous)
        for s in analytic_seeds:
            s = s / s.sum()
            conv = _conv_fft(s, s)
            cur_val = 2.0 * n * np.max(conv)
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = s.copy()
                print(f"[seed] new global best={cur_val:.6f}")
            _keep_top(top_candidates, cur_val, s)

        # piecewise‑linear random seeds (limited count)
        rng = np.random.default_rng(seed + 9999)
        pieces = 0
        while time.time() < deadline - 20.0 and pieces < 1500:
            cand = _random_piecewise_linear(n, rng)
            conv = _conv_fft(cand, cand)
            cur_val = 2.0 * n * np.max(conv)
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = cand.copy()
                print(f"[rand‑PL] new global best={cur_val:.6f}")
            _keep_top(top_candidates, cur_val, cand)
            pieces += 1

        # keep only the very best for polishing
        top_candidates = sorted(top_candidates, key=lambda x: x[0])[:5]

        # --------------------------------------------------------------
        # 2. LP‑based polishing (only on modest size to stay fast)
        # --------------------------------------------------------------
        if n <= 2000:
            rows, cols = _lp_pattern(n)
            for idx, (val, seq) in enumerate(top_candidates):
                if time.time() > deadline:
                    break
                print(f"\n--- LP polishing candidate #{idx+1} (n={n}) ---")
                f = seq.copy()
                f /= f.sum()
                # up to 12 LP rounds – stop early if no improvement
                for it in range(12):
                    if time.time() > deadline:
                        break
                    conv = _conv_fft(f, f)
                    M = np.max(conv)
                    cur_val = 2.0 * n * M
                    if cur_val < global_best_val:
                        global_best_val = cur_val
                        global_best_seq = f.copy()
                        print(f"[LP] iter={it} new global best={cur_val:.6f}")

                    g = _lp_optimize(f, M, n, rows, cols)
                    if g is None:
                        break

                    # simple line‑search on mixing coefficient t
                    best_local_val = cur_val
                    best_f = f
                    for t in (0.15, 0.30, 0.45, 0.60, 0.80):
                        new_f = (1.0 - t) * f + t * g
                        new_f /= new_f.sum()
                        new_conv = _conv_fft(new_f, new_f)
                        new_val = 2.0 * n * np.max(new_conv)
                        if new_val < best_local_val:
                            best_local_val = new_val
                            best_f = new_f
                    if best_local_val < cur_val - 1e-12:
                        f = best_f  # accept improvement
                    else:
                        break  # no further gain from LP

                # after LP phase store the polished vector
                if f is not None:
                    _keep_top(top_candidates, 2.0 * n * np.max(_conv_fft(f, f)), f)

        # --------------------------------------------------------------
        # 3. Light final refinement (Adam + Sub‑gradient) if time permits
        # --------------------------------------------------------------
        for idx, (val, seq) in enumerate(top_candidates):
            if time.time() > deadline - 5.0:
                break
            print(f"\n--- Finalising candidate #{idx+1} (n={n}) ---")
            # a quick Adam run
            seq_adam, val_adam = _adam_optimize(seq, n, deadline, max_iters=4000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[final‑Adam] best={val_adam:.6f}")

            # a quick sub‑gradient run
            seq_sg, val_sg = _subgradient_refine(seq, n, deadline, max_iters=10000)
            if val_sg < global_best_val:
                global_best_val = val_sg
                global_best_seq = seq_sg.copy()
                print(f"[final‑SubGrad] best={val_sg:.6f}")

    # ----------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = 2.0 * n * np.max(_conv_fft(global_best_seq, global_best_seq))
        print("[fallback] using uniform distribution")

    # ----------------------------------------------------------------------
    # Verify with the provided evaluate_sequence (strictly for safety)
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:  # pragma: no cover
        print("[verification] sequence rejected – falling back to uniform")
        n = len(global_best_seq)
        global_best_seq = _uniform(n)

    # store for possible warm‑starts in later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
