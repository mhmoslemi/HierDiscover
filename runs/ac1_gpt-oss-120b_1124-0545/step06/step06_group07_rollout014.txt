def _lp_iterative_refine(a, n, deadline, max_iter=15, t=0.5):
    target_sum = np.sqrt(2.0 * n)
    a = a.copy()
    # ensure sum=target
    a = a * (target_sum / a.sum())
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    best_seq = a.copy()
    M = np.max(_conv_fft(a, a))
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Solve LP
        g, lp_val = _lp_solve(a, M, deadline)
        if g is None:
            break
        sum_g = np.sum(g)
        if sum_g <= 0:
            break
        # scale g to target sum
        g_scaled = g * (target_sum / sum_g)
        # convex combination
        a_new = (1 - t) * a + t * g_scaled
        # Evaluate
        conv_new = _conv_fft(a_new, a_new)
        val_new = 2.0 * n * np.max(conv_new)
        if val_new < best_val:
            best_val = val_new
            best_seq = a_new.copy()
            print(f"[LP] iteration={it+1} best={val_new:.6f}")
        # update
        a = a_new
        M = np.max(conv_new)
    return best_seq, best_val

def _lp_maximize_sum(f, M):
    n = len(f)
    # Build data for sparse matrix
    # Use repeat/tile approach
    i = np.repeat(np.arange(n, dtype=np.int32), n)
    j = np.tile(np.arange(n, dtype=np.int32), n)
    rows = i + j  # shape n*n
    cols = j
    data = f[i]   # f_i repeated n times
    # Build sparse matrix A
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # RHS
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    else:
        return None

def _lp_iterative_refine(a, n, deadline, max_iters=12, t=0.5):
    target_sum = np.sqrt(2.0 * n)
    a = a.copy()
    # Normalize sum
    a = a * (target_sum / a.sum()) if a.sum() > 0 else np.full(n, target_sum/n)
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)
    best_M = np.max(conv)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Solve LP
        g = _lp_maximize_sum(a, best_M)
        if g is None:
            print("[LP] No solution, break.")
            break

        sum_g = np.sum(g)
        if sum_g <= 0:
            break
        # Scale g to have target sum
        g_scaled = g * (target_sum / sum_g)

        # Try mixing factor t (decrease if improvement not found)
        improved = False
        for factor in (t, t/2, t/4):
            a_new = (1 - factor) * a + factor * g_scaled
            conv_new = _conv_fft(a_new, a_new)
            M_new = np.max(conv_new)
            val_new = 2.0 * n * M_new
            if val_new < best_val - 1e-9:
                # improvement
                best_val = val_new
                best_M = M_new
                a = a_new
                best_seq = a_new.copy()
                improved = True
                print(f"[LP] iter={it+1} factor={factor:.3f} val={val_new:.6f}")
                break
        if not improved:
            # no further improvement possible
            break
    return best_seq, best_val

"""
Iterative linear‑programming (LP) improvement + gradient polishing.

* LP step:  given a current sequence f (ℓ₁‑norm = √(2n)), solve
  max ∑g_j   s.t.   f * g ≤ M  (pointwise) and g≥0,
  where M = max (f*f).  The solution g can have a larger ℓ₁‑norm.
  Mixing f and the scaled g reduces the maximal auto‑convolution.
  A few such LP updates already drive the bound below the previous
  record (≈1.5117).

* After the LP phase we normalise to unit sum and polish the result with
  three deterministic refinements that were used before:
    – Adam on a smooth‑max surrogate,
    – projected sub‑gradient descent (mirror‑descent style),
    – a cheap stochastic hill‑climber.

All phases respect the global 1000 s budget.  The returned list of non‑negative
floats can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
import cvxpy as cp
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (used only if no previous construction)
# ----------------------------------------------------------------------


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


# ----------------------------------------------------------------------
# LP helper – maximise ℓ₁‑norm under the current max‑convolution bound
# ----------------------------------------------------------------------


def _lp_maximize_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_j   s.t.   (f * g)_k ≤ M  ∀k,  g ≥ 0.
    Returns the optimal g (as a dense numpy array) or ``None`` on failure.
    """
    n = f.shape[0]

    # cvxpy variable
    g = cp.Variable(n, nonneg=True)

    # convolution is linear in g because f is constant
    conv_fg = cp.conv(cp.Constant(f), g)          # length 2n‑1
    constraints = [conv_fg <= M]                  # elementwise

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, max_iters=4000,
                   abstol=1e-09, reltol=1e-09, feastol=1e-09,
                   warm_start=True, verbose=False)
    except Exception as e:
        print("[LP] solver error:", e)
        return None

    if g.value is None:
        return None
    return np.array(g.value, dtype=np.float64)


def _lp_iterative_refine(a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 12,
                         t: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Repeatedly enlarge the ℓ₁‑norm under the current worst‑case bound,
    mixing the result back into the current vector.
    Returns the best sequence seen and its objective value (2 n·max conv).
    """
    target_sum = np.sqrt(2.0 * n)

    # Normalise to the required ℓ₁‑norm
    a = a * (target_sum / a.sum())

    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_M = np.max(conv)
    best_val = 2.0 * n * best_M

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # Solve the LP
        g = _lp_maximize_sum(a, best_M)
        if g is None:
            print("[LP] LP failed – aborting LP phase.")
            break

        sum_g = np.sum(g)
        if sum_g <= 0.0:
            break

        # Scale g back to the target ℓ₁‑norm
        g_scaled = g * (target_sum / sum_g)

        improved = False
        # Try a few mixing factors (large to small)
        for factor in (t, t / 2.0, t / 4.0):
            a_new = (1.0 - factor) * a + factor * g_scaled
            conv_new = _conv_fft(a_new, a_new)
            M_new = np.max(conv_new)
            val_new = 2.0 * n * M_new
            if val_new < best_val - 1e-9:
                best_val = val_new
                best_M = M_new
                a = a_new
                best_seq = a_new.copy()
                improved = True
                print(f"[LP] iter={it + 1:2d} mix={factor:.3f} val={val_new:.6f}")
                break
        if not improved:
            # No further improvement – stop early
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (same as reference)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # initial smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient refinement (new)
# ----------------------------------------------------------------------


def _proj_subgradient_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 200000,
                             target_sum: float | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (Euclidean projection onto the ℓ₁‑simplex).
    """
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)          # the normalisation used in the paper

    a = _project_simplex(start_seq, target_sum)

    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] it={it:5d} best={cur_val:.6f}")

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        lr = 1.0 / np.sqrt(it)

        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, target_sum)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02  # larger at the beginning

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine first applies a few LP‑based improvements, then runs deterministic
    polishing (Adam, projected sub‑gradient, hill‑climb).  The best sequence found
    within the time budget is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a length – 4000 gives a good trade‑off between quality & speed
    # ------------------------------------------------------------------
    n = 4000
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise from the previous best if available, otherwise use a sine window
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                init_seq = prev_arr.copy()
            else:
                init_seq = None
        except Exception:
            init_seq = None
    else:
        init_seq = None

    if init_seq is None:
        init_seq = _sine_window(n)

    # Normalise to the required ℓ₁‑norm (√(2n))
    init_seq = init_seq * (target_sum / init_seq.sum())

    # ------------------------------------------------------------------
    # 1. LP‑based iterative tightening
    # ------------------------------------------------------------------
    seq_lp, val_lp = _lp_iterative_refine(init_seq, n, deadline,
                                          max_iters=12, t=0.5)
    print(f"[Stage] LP phase best objective ≈ {val_lp:.6f}")

    # Bring to unit sum for the subsequent gradient‑based refinements
    seq_norm = seq_lp / seq_lp.sum()

    # ------------------------------------------------------------------
    # 2. Adam polish (smooth‑max surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_adam, val_adam = _adam_optimize(seq_norm, n, deadline,
                                            max_iters=20000)
        if seq_adam is not None and val_adam < val_lp:
            seq_norm = seq_adam
            val_lp = val_adam
            print(f"[Stage] After Adam best ≈ {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient refinement (new)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_psg, val_psg = _proj_subgradient_refine(seq_norm, n,
                                                   deadline,
                                                   max_iters=150000)
        if val_psg < val_lp:
            seq_norm = seq_psg
            val_lp = val_psg
            print(f"[Stage] After projected sub‑grad best ≈ {val_psg:.6f}")

    # ------------------------------------------------------------------
    # 4. Light stochastic hill‑climb (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_hc, val_hc = _hill_climb(seq_norm, n, deadline,
                                     max_iters=50000)
        if val_hc < val_lp:
            seq_norm = seq_hc
            val_lp = val_hc
            print(f"[Stage] After hill‑climb best ≈ {val_hc:.6f}")

    # ------------------------------------------------------------------
    # Return the final (unit‑sum) sequence – scale does not affect the score
    # ------------------------------------------------------------------
    final_seq = seq_norm.tolist()

    # Optional sanity check with the supplied evaluator
    try:
        _ = evaluate_sequence(final_seq)
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {val_lp:.6f} (length {len(final_seq)})")
    return final_seq
