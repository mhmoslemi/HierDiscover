def _local_fine_tune(seq, n, deadline):
    rng = np.random.default_rng()
    cur = seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val
    rng_step = lambda it: 0.5 / np.sqrt(it+1) # step fraction of source mass
    it=0
    while time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        delta = rng_step(it) * cur[i]
        if delta > cur[i]:
            delta = cur[i]
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_val = _objective(new_seq)
        if new_val < cur_val:
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:

"""
Propose a candidate step‑function that (hopefully) beats the current
best bound ≈ 1.5117.  The algorithm is a multi‑stage local optimiser:

1. Build a diverse pool of seeds (uniform, geometric, power‑law,
   Beta, Gaussian, Hanning, mirrored copies and an up‑scaled previous best).
2. Run a longer Adam optimisation on a smooth max‑surrogate
   (log‑sum‑exp) with a large β schedule.
3. Refine the result with a mirror‑descent (exponentiated‑gradient)
   sub‑gradient method for the exact max‑convolution, using many
   iterations and a diminishing step‑size.
4. Perform a second short Adam pass with an even larger β to tighten
   the solution.
5. Apply a cheap simulated‑annealing / random‑mass‑swap local search.
6. Add a deterministic fine‑tuning phase that moves a tiny fraction
   of mass from the most “responsible’’ coordinates to the least.
7. Symmetrise the final vector (averaging with its reverse) – this
   often reduces the peak of the self‑convolution.
8. If time permits, upscale the solution to a longer length (8000)
   and repeat steps 2‑7.

All operations stay on the simplex (non‑negative entries summing to 1),
so the objective `2·n·max(conv)/sum²` is scale‑invariant.
The code respects the 1000 s budget, saves the best sequence globally
as ``height_sequence_1`` and returns it as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for many calls)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the grader (scale‑invariant).
    seq is assumed to be non‑negative and sum(seq) > 0.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

# Seed families (all return a normalised probability vector)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sin2_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _hanning(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)
    a[a < 0] = 0.0
    return a / a.sum()


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of normalised seed vectors."""
    rng = np.random.default_rng(123456)
    seeds: list[np.ndarray] = []

    # Basic families
    seeds.append(_uniform(n))
    for r in np.linspace(0.1, 0.9, 5):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 5):
        seeds.append(_power(n, p))
    for a in [0.3, 0.5, 0.8, 1.2, 2.0]:
        seeds.append(_beta_seed(n, a, a))
        seeds.append(_beta_seed(n, a, a * 1.3))
    seeds.append(_sine_window(n))
    seeds.append(_sin2_window(n))
    seeds.append(_hanning(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_tent(n))

    # Mirrored copies (increase variety)
    mirrors = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrors)

    # Dirichlet draws
    for alpha in [0.2, 0.5, 0.9, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # Upscaled previous best when available
    if previous is not None:
        try:
            up = _upscale(previous, n)
            seeds.append(up)
        except Exception:
            pass

    # Ensure non‑empty
    if not seeds:
        seeds.append(_uniform(n))

    return seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000,
                   beta0: float = 2.0,
                   beta_target: float = 300.0,
                   beta_mul: float = 1.001,
                   lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth approximation L = (1/β)·log Σ exp(β·conv).
    Returns the best sequence seen and its *exact* objective value.
    """
    v = np.log(seed + 1e-30)               # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # point on simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # increase β gradually
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # smooth‑max weights (log‑sum‑exp trick)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical under‑flow → hard max
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient wrt a (via reversed convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Uses a 1/√t step‑size and runs until `max_iters` or time runs out.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)               # decreasing step‑size
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # improve numerical stability
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings local search that swaps tiny fragments of mass.
    The temperature decays geometrically.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature as a small fraction of the current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a modest fraction of mass from i to j
        delta = rng.random() * 0.025 * cur[i]   # ≤2.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        # occasional global jitter (helps escape flat regions)
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


def _fine_tune(seq: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """
    Deterministic fine‑tuner that repeatedly moves a tiny fraction of mass
    from a coordinate with the largest gradient (i.e. most responsible for
    the current peak) to a coordinate with the smallest gradient.
    The step size decays as 0.5/√t.
    """
    cur = seq.copy()
    best = cur.copy()
    best_val = _objective(cur)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(conv.argmax())

        # build the sub‑gradient for the active index
        grad = np.zeros_like(cur)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        grad[idx_i] = 2.0 * cur[rev_idx]

        # pick source (largest positive gradient) and destination (smallest)
        src = np.argmax(grad)
        dst = np.argmin(grad)

        if src == dst:
            # all gradients equal – nothing to improve
            break

        step = 0.5 / np.sqrt(it)          # shrinking fraction
        delta = step * cur[src]
        delta = min(delta, cur[src])

        new_seq = cur.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # evaluate
        new_val = _objective(new_seq)

        if new_val < best_val:
            best_val = new_val
            best = new_seq.copy()
            print(f"[FineTune] n={n} it={it:6d} best={new_val:.6f}")

        # accept if improvement, else keep current (no Metropolis here)
        if new_val < _objective(cur):
            cur = new_seq

        # occasional random jitter to escape possible plateaus
        if it % 5000 == 0:
            rng = np.random.default_rng()
            noisy = cur + rng.normal(scale=1e-4, size=n)
            noisy = np.clip(noisy, 0.0, None)
            if noisy.sum() == 0.0:
                noisy = _uniform(n)
            else:
                noisy /= noisy.sum()
            cur = noisy

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence (list of floats) found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # Load a previously discovered construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Length schedule – start with a moderate size, then upscale if time permits
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # Seed generation
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # quick evaluation of raw seeds; keep the top few for further work
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:5]]

        # update global best from raw seeds
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation from each top seed
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=12000,
                beta0=2.0,
                beta_target=300.0,
                beta_mul=1.001,
                lr0=0.5,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – heavy sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=200000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – second short Adam (larger β) for polishing
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _adam_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=8000,
                beta0=2.0,
                beta_target=500.0,
                beta_mul=1.001,
                lr0=0.3,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam2] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – simulated annealing
        # ------------------------------------------------------------------
        if time.time() < deadline and global_best_seq is not None:
            seq, val = _anneal_optimize(
                global_best_seq, n, deadline, max_iters=None
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 5 – deterministic fine‑tuning
        # ------------------------------------------------------------------
        if time.time() < deadline and global_best_seq is not None:
            seq, val = _fine_tune(
                global_best_seq, n, deadline, max_iters=150000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall FineTune] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Symmetrisation (often reduces the peak)
        # ------------------------------------------------------------------
        if global_best_seq is not None:
            sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[symmetrisation] n={n} improved to {sym_val:.6f}")

        # Use current best as a seed for the next (larger) length
        previous = global_best_seq

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Normalise once more for safety
    global_best_seq = np.clip(global_best_seq, 0.0, None)
    s = global_best_seq.sum()
    if s == 0.0:
        global_best_seq = _uniform(global_best_seq.shape[0])
    else:
        global_best_seq = global_best_seq / s

    # Cache for the next call
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
