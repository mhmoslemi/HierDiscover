def project_simplex(v):
    # Projects onto the simplex {x >= 0, sum(x)=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _subgrad_proj_refine(start_seq, n, deadline, max_iters=50000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # compute subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size via diminishing schedule
        alpha = 1.0 / np.sqrt(it)  # scaling factor, maybe 0.5?
        # update with projection
        a_new = a - alpha * g
        a_new = project_simplex(a_new)

        a = a_new
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGradProj] n={n} it={it} best={cur_val:.6f}")

    return best_seq, best_val

"""
Propose a high‑quality non‑negative step function.

The search proceeds in three stages for each candidate length:
1) **Rich seeding** – many analytically‑motivated families
   (uniform, geometric, power‑law, symmetric geometric, sine‑window,
   Gaussian, arcsine, Dirichlet draws, random sinusoid mixtures and
   sparse “spike” vectors).  All seeds are normalised to sum = 1 and
   kept on the simplex via a log‑softmax parametrisation.

2) **Adam on a smooth surrogate** – minimise a smooth approximation of
   the max‑convolution together with two regularisers:
      • variance regulariser λ·∑ conv²   (pushes the convolution flat)
      • entropy regulariser –μ·H(a)    (keeps the distribution spread)
   The loss is
        L(a)= (1/β)·log Σₖ exp(β·convₖ(a)) + λ·∥conv(a)∥₂² – μ·H(a)
   Gradients are back‑propagated through the soft‑max logits.
   The best Adam result over all seeds is kept.

3) **Exact refinement** – a projected sub‑gradient descent on the true
   max‑convolution (mirrored‑descent on the simplex).  The step size
   decays as 1/√t and each update is projected onto the probability
   simplex with the O(n) Euclidean‑projection algorithm of
   Duchi et al.

4) **Fine‑tuning** – a cheap Metropolis‑Hastings annealing that moves
   tiny mass between random positions and occasionally adds a global
   jitter.

The procedure explores several lengths (4000 → 8000 → 12000 → 20000)
and respects the 1000 s wall‑time budget.  The globally best sequence
found is stored in the module variable ``height_sequence_1`` and
returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for long vectors)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Duchi et al. 2008 – O(n log n) due to sorting.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen for a finite vector, but fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Gamma(α,1) draw normalised to a probability vector."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """Discrete arcsine‑type family.
    a_i ∝ ((i+½)*(n‑½‑i))**exponent   (exponent < 0 puts mass near the ends)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide="ignore", invalid="ignore"):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        return _uniform(n)
    return a / s


def _sinusoid_mix(n: int, rng: np.random.Generator, K: int = 5, scale: float = 0.2) -> np.ndarray:
    """Random linear combination of low‑frequency sine waves, shifted to be non‑negative."""
    i = np.arange(n, dtype=np.float64)
    coeffs = rng.normal(scale=scale, size=K)
    a = np.ones_like(i)  # baseline to keep positivity
    for k, c in enumerate(coeffs, start=1):
        a += c * np.sin(np.pi * k * i / n)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _spike_seed(n: int, rng: np.random.Generator, m: int = None) -> np.ndarray:
    """Very sparse seed: put random mass on m positions (default ~sqrt(n))."""
    if m is None:
        m = max(2, int(np.sqrt(n)))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.random(m)
    a[idx] = weights
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse pool of normalised seeds."""
    rng = np.random.default_rng(int(time.time()) ^ n)
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # geometric families
    for r in np.linspace(0.90, 0.99, 5):
        seeds.append(_geom(n, r))

    # symmetric geometric
    for r in np.linspace(0.90, 0.99, 4):
        seeds.append(_sym_geom(n, r))

    # power‑law
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # arcsine (negative exponents)
    for exp in [-1.2, -1.0, -0.75, -0.5, -0.25]:
        seeds.append(_arcseed(n, exp))

    # random Dirichlet draws
    for alpha in [0.02, 0.05, 0.1, 0.3, 0.7, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # sinusoid mixtures
    for K in [3, 5, 7]:
        seeds.append(_sinusoid_mix(n, rng, K=K, scale=0.15))

    # sparse spike seeds
    for m in [max(2, int(np.sqrt(n) // 2)), max(2, int(np.sqrt(n))), max(2, int(2 * np.sqrt(n)))]:
        seeds.append(_spike_seed(n, rng, m=m))

    # mirrored versions (helps asymmetric optima)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # reuse previous best if shape matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / max(previous.sum(), 1e-12))

    # ensure non‑empty
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimiser with smooth‑max + variance + entropy regularisation
# ----------------------------------------------------------------------


def _adam_optimize_entropy(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 15000,
    beta0: float = 0.5,
    beta_mul: float = 1.03,
    lr0: float = 1.0,
    lambda0: float = 2.0,
    lambda_decay: float = 0.996,
    mu0: float = 0.5,
    mu_decay: float = 0.995,
) -> tuple[np.ndarray, float]:
    """
    Adam on surrogate loss L = smooth_max(conv) + λ·‖conv‖₂² – μ·H(a).
    Returns best sequence encountered and its true objective.
    """
    v = np.log(seed + 1e-30)          # logits for softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0
    lam = lambda0
    mu = mu0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                # point on simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # ---- smooth‑max part -------------------------------------------------
        beta *= beta_mul
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of smooth‑max (conv weighted by softmax probabilities)
        rev_a = a[::-1]
        full_max = _conv_fft(rev_a, w)
        grad_max = 2.0 * full_max[n - 1 : n - 1 + n]

        # ---- variance regulariser (‖conv‖₂²) ---------------------------------
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n - 1 : n - 1 + n]

        # ---- entropy term ---------------------------------------------------
        # ∇_a ( -μ·H ) = μ·(log a + 1)
        grad_entropy = mu * (np.log(a + 1e-30) + 1.0)

        # total gradient w.r.t a
        grad_a = grad_max + lam * grad_sq + grad_entropy

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        # decay regularisers slowly
        lam *= lambda_decay
        mu *= mu_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (projected mirror‑descent)
# ----------------------------------------------------------------------


def _subgrad_proj_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # compute sub‑gradient of max(conv) at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size (diminishing)
        alpha = 0.9 / np.sqrt(it)

        # gradient step + projection onto simplex
        a = _project_simplex(a - alpha * g)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val

# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """Cheap MH walk that transfers tiny mass between random sites."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # move a tiny fraction from i to j
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            cand = cur + noise
            cand = np.clip(cand, 0.0, None)
            if cand.sum() == 0.0:
                cand = _uniform(n)
            else:
                cand /= cand.sum()
            cand_val = _objective(cand)
            accept = cand_val < cur_val or rng.random() < np.exp(-(cand_val - cur_val) / (T + 1e-12))
            if accept:
                cur = cand
                cur_val = cand_val
                if cand_val < best_val:
                    best_val = cand_val
                    best = cand.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={cand_val:.6f}")

    return best, best_val

        # ------------------------------------------------------------------

            if time.time() > deadline:
                break
            seq, val = _adam_optimize_entropy(
                seed_vec,
                n,
                deadline,
                max_iters=18000,
                beta0=0.5,
                beta_mul=1.03,
                lr0=1.0,
                lambda0=2.5,
                lambda_decay=0.995,
                mu0=0.6,
                mu_decay=0.995,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (projected).
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgrad_proj_refine(global_best_seq, n, deadline, max_iters=70000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing fine‑tuning.
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # Store the best of this length as “previous” for the next round
        previous = global_best_seq

    # ----------------------------------------------------------------------
    # Fallback – should never be needed but guards against empty result.
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    # Export for potential reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.copy()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
