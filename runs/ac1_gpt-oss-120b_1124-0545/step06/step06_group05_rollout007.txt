def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    lengths = [2000, 4000, 8000, 12000] # maybe 12000 if time.

    best_seq = None
    best_val = float('inf')

    # generate seeds and evaluate quickly
    top_candidates = []  # list of (value, seq, n)

    for n in lengths:
        # break if deadline near
        if time.time() > deadline: break
        # generate seeds
        seeds = _generate_seeds_for_n(n)
        # evaluate and keep top 5
        for s in seeds:
            val = evaluate_sequence(s.tolist())
            if val < best_val:
                best_val = val
                best_seq = s.copy()
                # print
                print(f"[seed] n={n} val={val:.6f}")

            # maintain heap for top candidates
            # but for simplicity keep a list and sort later.
            top_candidates.append((val, s, n))

        # optional: after each length, sort and keep only top_k
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]  # keep best 30 overall

    # refine with subgradient
    for val, seq, n in top_candidates:
        if time.time() > deadline: break
        # Already best candidate may be same; refine anyway
        refined_seq, refined_val = _subgradient_refine(seq, n, deadline, max_iters=30000)
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()

    # annealing fine-tune best_seq if time remains
    if best_seq is not None and time.time() < deadline:

        best_seq, best_val = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        print(f"[anneal] final val={best_val:.6f}")

    # return list
    return best_seq.tolist()

def _generate_seeds_for_n(n, previous=None):
    seeds = []
    # uniform
    seeds.append(_uniform(n))
    # geometric series
    for r in np.linspace(0.1, 0.99, 9):
        seeds.append(_geom(n, r))
    # power law p from 0 to 3
    for p in np.linspace(0, 3, 7):
        seeds.append(_power(n, p))
    # arcsine exponents
    for exp in np.linspace(-0.2, -2.0, 19):
        seeds.append(_arcseed(n, exp))
    # beta shape
    for a in [0.2, 0.5, 0.8, 1.5, 2.0, 3.0]:
        for b in [0.2, 0.5, 0.8, 1.5, 2.0, 3.0]:
            seeds.append(_beta_seed(n, a, b))
    # two-step
    for frac in [0.05, 0.1, 0.15, 0.2]:
        for w in [2, 4, 6, 8]:
            seeds.append(_twostep_seed(n, frac, w))
    # Dirichlet draws
    rng = np.random.default_rng(12345)
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))
    # random sparse
    for k in [2, 3, 5, 10]:
        seeds.append(_sparse_seed(n, k, rng))

"""
Propose a step‑function with a very low value of
evaluate_sequence(seq) ≈ 2·n·max(conv(seq,seq))/(sum(seq)²).

The algorithm works in several stages:

1. **Rich seed generation.**  A large collection of analytic families
   (uniform, geometric, power‑law, arcsine‐type, Beta‑pdf, two‑step,
   Dirichlet draws, sparse random spikes) is created for each chosen
   length.  All vectors are placed on the probability simplex
   (non‑negative, sum = 1).  A few mirrored copies are added to explore
   asymmetric shapes.

2. **Quick screening.**  Every seed is evaluated with a cheap FFT‑based
   objective; the best few (≈30) candidates across all lengths are kept.

3. **Exact sub‑gradient mirror descent.**  For each of the retained
   candidates we minimise the true max‑convolution via exponentiated‑gradient
   updates (a form of mirror descent) on the simplex.  The step‑size
   decays as 1/√t and the procedure runs until the global time budget
   expires or a fixed iteration count is reached.

4. **Simulated‑annealing fine‑tuning.**  The current global best undergoes
   a cheap Metropolis‑Hastings walk that transfers a tiny fraction of mass
   between two random positions; the temperature decays exponentially.
   Periodic global Gaussian jitter improves exploration.

5. **Local hill‑climbing.**  Finally a very small random perturbation
   (move ≤ 1 % of the source mass) is repeatedly tried while time permits;
   any improvement is kept.

All stages respect the 1000 s budget, run on a single process (but are
vectorised with NumPy and FFT for speed) and finally return the best
sequence found as a plain list of floats.  The best sequence is also stored
in the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))², seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcseed(n: int, exponent: float) -> np.ndarray:
    """Discrete arcsine‑type law a_i ∝ ((i+½)*(n‑½‑i))**exponent."""
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)
    with np.errstate(divide='ignore', invalid='ignore'):
        a = prod ** exponent
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        return _uniform(n)
    return a / s


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density discretised on the grid (i+½)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    with np.errstate(divide='ignore', invalid='ignore'):
        logpdf = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x)
    # Replace -inf (outside (0,1)) with a very small number
    logpdf = np.where(np.isfinite(logpdf), logpdf, -1e30)
    # Subtract a constant for numerical stability
    logpdf -= np.max(logpdf)
    a = np.exp(logpdf)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _twostep_seed(n: int, edge_frac: float, edge_weight: float) -> np.ndarray:
    """Two‑level shape: heavy weight on the first/last edge_frac of positions."""
    L = int(edge_frac * n)
    a = np.ones(n, dtype=np.float64)
    if L > 0:
        a[:L] = edge_weight
        a[-L:] = edge_weight
    s = a.sum()
    return a / s


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse or dense Dirichlet draw (Gamma(α,1) normalised)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    s = vec.sum()
    if s <= 0.0:
        return _uniform(n)
    return vec / s


def _sparse_seed(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Place equal mass on k random indices (k ≤ n)."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of normalised non‑negative seeds for length n."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # Basic families
    seeds.append(_uniform(n))

    # Geometric series (r close to 1 gives a slowly decaying tail)
    for r in np.linspace(0.1, 0.99, 9):
        seeds.append(_geom(n, r))

    # Power‑law decay
    for p in np.linspace(0.0, 3.0, 7):
        seeds.append(_power(n, p))

    # Arcsine‑type (negative exponents)
    for exp in np.linspace(-0.2, -2.0, 19):
        seeds.append(_arcseed(n, exp))

    # Beta distributions (α,β from 0.2 to 5)
    for a in [0.2, 0.5, 0.8, 1.5, 2.0, 3.0, 5.0]:
        for b in [0.2, 0.5, 0.8, 1.5, 2.0, 3.0, 5.0]:
            seeds.append(_beta_seed(n, a, b))

    # Two‑step (edge heavy) shapes
    for frac in [0.05, 0.1, 0.15, 0.2]:
        for w in [2.0, 4.0, 6.0, 8.0]:
            seeds.append(_twostep_seed(n, frac, w))

    # Dirichlet samples (covers sparse ↔ dense)
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # Very sparse spikes
    for k in [2, 3, 5, 10]:
        seeds.append(_sparse_seed(n, k, rng))

    # Mirror each seed (helps with asymmetric optima)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # Re‑use a previously discovered construction if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / max(previous.sum(), 1e-12))

    # Normalise and discard degenerate entries
    final = []
    for s in seeds:
        s = np.maximum(s, 0.0)
        total = s.sum()
        if total > 1e-12:
            final.append(s / total)
    return final


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 25000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best sequence encountered and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = max_idx - i_idx
        g[i_idx] = 2.0 * a[j_idx]

        # exponentiated‑gradient step (mirrored onto the simplex)
        eta = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        log_a -= np.max(log_a)          # numeric stability
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis‑Hastings walk that moves a tiny fraction of mass
    between two random positions and occasionally adds a global jitter.
    Returns the best sequence discovered.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)  # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a tiny fraction of mass from i to j
        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # periodic global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            jitter = cur + noise
            jitter = np.clip(jitter, 0.0, None)
            s = jitter.sum()
            if s > 0.0:
                jitter /= s
            else:
                jitter = _uniform(n)
            jitter_val = _objective(jitter)
            accept = jitter_val < cur_val or rng.random() < np.exp(-(jitter_val - cur_val) / (T + 1e-12))
            if accept:
                cur = jitter
                cur_val = jitter_val
                if jitter_val < best_val:
                    best_val = jitter_val
                    best = jitter.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={jitter_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found
    as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Re‑use a previously discovered construction if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0:
                previous = previous / max(previous.sum(), 1e-12)
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (short -> longer).  Longer lengths are attempted
    # only if time permits.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]
    global_best_seq = None
    global_best_val = float("inf")

    # Keep a small pool of the overall best candidates for refinement
    top_candidates: list[tuple[float, np.ndarray, int]] = []

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Exploring length {n} ===")

        # 1) Build seeds
        seeds = _make_seeds(n, previous)

        # 2) Quick evaluation of every seed; keep a global shortlist
        for s in seeds:
            val = _objective(s)
            # record global best so far
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[Seed] n={n} new global best={val:.6f}")

            top_candidates.append((val, s.copy(), n))

        # keep only the best 30 candidates overall (across lengths)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient refinement (mirror descent)
    # ------------------------------------------------------------------
    for val, seq, n in top_candidates:
        if time.time() > deadline:
            break
        refined_seq, refined_val = _subgradient_refine(seq, n, deadline, max_iters=25000)
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[Refine] n={n} improved to {refined_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        n = global_best_seq.shape[0]
        annealed_seq, annealed_val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if annealed_val < global_best_val:
            global_best_val = annealed_val
            global_best_seq = annealed_seq.copy()
            print(f"[Anneal] final best={annealed_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Tiny random hill‑climbing (last few seconds)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = global_best_seq.shape[0]
        rng = np.random.default_rng()
        new_seq = global_best_seq.copy()
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.random() * 0.01 * new_seq[i]  # ≤ 1 % of source mass
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = np.maximum(new_seq, 0.0)
        s = new_seq.sum()
        if s > 0:
            new_seq /= s
        else:
            new_seq = _uniform(n)
        val = _objective(new_seq)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = new_seq.copy()
            print(f"[Hill] n={n} improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # Extremely unlikely fallback
        global_best_seq = _uniform(2000)
        print("[Fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
