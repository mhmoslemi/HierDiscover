"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The algorithm is a 4‑phase optimizer:

1. **Deterministic sweep** – enumerate several analytic families
   (uniform, triangular, trapezoidal, sine, beta, power, geometric,
   logistic).  For each parameter setting the exact objective
   ``2·n·max(conv(a,a))`` is computed and the best few candidates are kept.
2. **Adam on a smooth max** – the max‑convolution is approximated by a
   temperature‑scaled log‑sum‑exp.  Adam optimisation works in the
   log‑softmax (simplex) space; a small regulariser that penalises the
   correlation ``a·reverse(a)`` encourages anti‑symmetry.
3. **Projected sub‑gradient descent** – after the Adam run we switch to
   the true non‑smooth objective and take sub‑gradient steps with a
   Polyak step‑size while projecting onto the probability simplex.
4. **Light simulated‑annealing polish** – a few thousand random
   Dirichlet moves finish the search.

The whole routine respects the supplied time budget (default 1000 s) and
stores the best sequence in a global variable ``height_sequence_1`` for
re‑use across runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a *probability* vector.
    Returns 2·n·max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(
                np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, steep: float = 0.01, centre: float = None) -> np.ndarray:
    """
    Monotone increasing logistic shape, shifted so that the first value is zero.
    """
    if centre is None:
        centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-steep * (i - centre)))
    a = a - a.min()                      # ensure first entry is zero
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 800_000,
                   beta_max: float = 5000.0,
                   lambda_corr: float = 0.15) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    The optimisation is performed in the logits space (log‑softmax) which
    automatically respects the simplex constraint.  An optional L2
    regularisation term ``lambda_corr * a·reverse(a)`` is added to
    encourage anti‑symmetry.
    Returns the best sequence found and its true objective.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # start temperature (smoothness)
    beta_mul = 1.0015
    beta_min = 5.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = np.exp(v - np.max(v))
        a = a / a.sum()                         # softmax → simplex

        # current true objective (cheap, exact)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam] it={it:,}  best={best_val:.6f}")

        # temperature schedule – slowly approach the true max
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # smooth max via tempered softmax over the convolution entries
        conv = _conv_fft(a, a)                  # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                              # weights sum to 1

        # gradient of the smooth surrogate w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # add correlation regulariser gradient: d/d a (λ a·R a) = 2 λ R a
        if lambda_corr > 0.0:
            grad_a += 2.0 * lambda_corr * rev_a

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # norm clipping for stability
        max_norm = 8.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses the classic Polyak step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 40_000 == 0:
                print(f"[Subgrad] it={it:,}  best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.15 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 250_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing: re‑allocate the total mass of a small
    random subset of coordinates (Dirichlet move) and accept according to
    a decaying temperature schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)                     # size of the moved block
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30_000 == 0:
                    print(f"[SA] it={it:,}  best={best_val:.6f}")

    return best_seq, best_val

def _collect_candidates(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None,
                       top_k: int,
                       deadline: float) -> list[np.ndarray]:
    """
    Systematically explore analytic families and retain the `top_k`
    candidates with smallest exact objective.
    """
    best = []  # list of (value, seq)

    def keep(val, seq):
        if len(best) < top_k:
            best.append((val, seq))
            best.sort(key=lambda x: x[0])
        elif val < best[-1][0]:
            best[-1] = (val, seq)
            best.sort(key=lambda x: x[0])

    # ----- deterministic families -----------------------------------------
    families = [
        _uniform,
        _triangular,
        lambda nn: _trapezoidal(nn, width_fraction=0.10),
        lambda nn: _trapezoidal(nn, width_fraction=0.25),
        lambda nn: _trapezoidal(nn, width_fraction=0.45),
        lambda nn: _sine_window(nn, gamma=1.0),
        lambda nn: _sine_window(nn, gamma=2.0),
        lambda nn: _sine_window(nn, gamma=3.0),
        lambda nn: _tukey_window(nn, alpha=0.2),
        lambda nn: _tukey_window(nn, alpha=0.5),
        lambda nn: _tukey_window(nn, alpha=0.8),
        lambda nn: _logistic_window(nn, steep=0.005),
        lambda nn: _logistic_window(nn, steep=0.015),
    ]

    for fn in families:
        if time.time() > deadline:
            break
        seq = fn(n)
        keep(_objective(seq), seq)

    # ----- beta shape grid -------------------------------------------------
    alphas = np.linspace(0.1, 4.0, 20)
    betas = np.linspace(0.1, 4.0, 20)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            seq = _beta_shape(n, a, b)
            keep(_objective(seq), seq)

    # ----- sine window grid -------------------------------------------------
    gammas = np.linspace(0.5, 5.0, 20)
    for g in gammas:
        if time.time() > deadline:
            break
        seq = _sine_window(n, gamma=g)
        keep(_objective(seq), seq)

    # ----- power‑law --------------------------------------------------------
    ps = np.linspace(0.0, 5.0, 20)
    for p in ps:
        if time.time() > deadline:
            break
        seq = _power(n, p)
        keep(_objective(seq), seq)

    # ----- geometric ---------------------------------------------------------
    rs = np.linspace(0.3, 0.98, 18)
    for r in rs:
        if time.time() > deadline:
            break
        seq = _geom(n, r)
        keep(_objective(seq), seq)
        seq2 = _sym_geom(n, r)
        keep(_objective(seq2), seq2)

    # ----- random Dirichlet seeds -------------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        draws = 30 if conc < 0.01 else 15
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            keep(_objective(seq), seq)

    # ----- sparse random supports -------------------------------------------
    for _ in range(120):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)                # 3 … 30 non‑zero entries
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = weights
        keep(_objective(seq), seq)

    # ----- previous solution -------------------------------------------------
    if previous is not None and previous.shape == (n,):
        seq = previous / previous.sum()
        keep(_objective(seq), seq)

    # Return only the sequences, sorted by ascending objective
    return [seq for _, seq in best]

            phase2_deadline,
            max_iters=800_000,
            beta_max=5000.0,
            lambda_corr=0.15,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  -> Updated global best after long Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: projected sub‑gradient descent ---")
        remaining = deadline - time.time()
        phase3_deadline = time.time() + min(remaining * 0.4, 300.0)
        seq_sub, val_sub = _subgrad_optimize(
            best_seq,
            n,
            phase3_deadline,
            max_iters=500_000,
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  -> Updated global best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: simulated annealing polishing ---")
        remaining = deadline - time.time()
        phase4_deadline = time.time() + min(remaining, 200.0)
        seq_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            phase4_deadline,
            max_iters=200_000,
            init_temp_factor=0.04,
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  -> Updated global best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal score reported by evaluate_sequence: {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
