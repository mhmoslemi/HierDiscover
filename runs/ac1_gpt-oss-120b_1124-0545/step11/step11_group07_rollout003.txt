"""
Propose a low‑value step‑function.

Algorithm outline
-----------------
1. **Seed pool** – generate a small diverse set of ℓ₁‑normalised candidates
   (uniform, triangular, trapezoidal, Gaussian, beta, sine, Tukey,
   random Dirichlet draws and sparse supports).  Keep the 5 best w.r.t.
   the true max‑convolution value.

2. **Intensive refinement** – each seed is refined by a short Adam run
   on a smooth log‑sum‑exp surrogate of the max‑convolution.  The
   temperature (smooth‑max β) is increased gradually up to a large
   value (β≈400).  The best result from this stage becomes the current
   global best.

3. **Exploratory restarts** – as long as time permits we repeatedly
   (a) draw a fresh random Dirichlet vector or (b) perturb the current
   best by reallocating the mass of a few randomly chosen coordinates.
   Each candidate undergoes a brief Adam optimisation (≈20 k
   iterations) followed by a short projected sub‑gradient descent.
   Whenever a better max‑convolution value is observed we update the
   global best.

4. **Final polishing** – a simple simulated‑annealing phase makes tiny
   Dirichlet reallocations on the simplex to escape shallow minima.

All stages respect the overall time budget (default 1000 s).  The final
sequence is scaled to the required ℓ₁‑norm √(2 n) before returning.

The implementation avoids lambda expressions (as required) and keeps
all helper functions at the top level.
"""

import time

# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant).  ``seq`` must be normalised to
    sum == 1.  Returns max(conv(seq, seq)).
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal_015(n: int) -> np.ndarray:
    return _trapezoidal(n, width_fraction=0.15)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window_gamma1(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window_gamma2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** 2.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_alpha02(n: int) -> np.ndarray:
    return _tukey_window(n, alpha=0.2)


def _tukey_alpha04(n: int) -> np.ndarray:
    return _tukey_window(n, alpha=0.4)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)          # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

def _collect_diverse_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    top_k: int,
    deadline: float,
) -> list[np.ndarray]:
    """
    Generate a diverse pool of ℓ₁‑normalised candidates and return the best
    `top_k` of them w.r.t. the exact max‑convolution objective.
    """
    top_vals = []  # list of (value, seq)

    def try_insert(val, seq):
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])

    # Deterministic families
    deterministic_fns = [
        _uniform,
        _triangular,
        _trapezoidal_015,
        _sine_window_gamma1,
        _sine_window_gamma2,
        _gaussian_sigma12,
        _gaussian_sigma24,
        _tukey_alpha02,
        _tukey_alpha04,
    ]

    for fn in deterministic_fns:
        seq = fn(n)
        try_insert(_objective(seq), seq)

    # Geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seq = _geom(n, r)
        try_insert(_objective(seq), seq)
        seq2 = _sym_geom(n, r)
        try_insert(_objective(seq2), seq2)

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 18):
        seq = _power_law(n, p)
        try_insert(_objective(seq), seq)

    # Random Dirichlet draws – sweep many concentration levels
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 70 if conc < 0.01 else 40
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            try_insert(_objective(seq), seq)

    # Sparse random supports
    for _ in range(250):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        try_insert(_objective(seq), seq)

    # Random beta shapes
    for _ in range(300):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 5.0)
        beta = rng.uniform(0.05, 5.0)
        seq = _beta_shape(n, alpha, beta)
        try_insert(_objective(seq), seq)

    # Random sine windows (varying exponent)
    for _ in range(200):
        if time.time() > deadline:
            break
        gamma = rng.uniform(0.2, 5.0)
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
        a = a / a.sum()
        try_insert(_objective(a), a)

    # Random Gaussian windows
    for _ in range(200):
        if time.time() > deadline:
            break
        sigma_frac = rng.uniform(0.05, 0.4)
        seq = _gaussian(n, sigma_frac=sigma_frac)
        try_insert(_objective(seq), seq)

    # Random Tukey windows
    for _ in range(150):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 0.9)
        seq = _tukey_window(n, alpha=alpha)
        try_insert(_objective(seq), seq)

    # Mixture of uniform & Gaussian
    for _ in range(150):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        try_insert(_objective(seq), seq)

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        seq = np.asarray(previous, dtype=np.float64)
        if seq.sum() > 0:
            seq = seq / seq.sum()
            try_insert(_objective(seq), seq)

    # Return only the sequences (sorted by increasing objective value)
    return [seq for _, seq in top_vals]

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    beta_max: float = 400.0,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the maximum
    convolution entry.  ``start_vec`` must already be normalised to sum=1.
    Returns the best sequence encountered (still sum=1) and its exact
    max‑convolution value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                  # point on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # raise temperature
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # convolution and smooth‑max weights
        conv = _conv_fft(a, a)            # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)       # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best sequence encountered (sum=1) and its value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]   # value that evaluate_sequence would return
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak‑style step size (using current best)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Tiny‑move simulated annealing on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)       # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence it finds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Length of the step‑function – 4096 is a power‑of‑two and a good trade‑off.
    n = 4096
    sqrt2n = np.sqrt(2.0 * n)

    print(f"\n=== Starting search (n={n}, budget={budget_s}s) ===")

    # ------------------------------------------------------------------
    # Stage 0 – diverse seed generation (≈10 % of budget)
    # ------------------------------------------------------------------
    seed_deadline = min(deadline, start_time + budget_s * 0.10)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    seeds = _collect_diverse_seeds(
        n=n,
        rng=rng,
        previous=previous,
        top_k=5,
        deadline=seed_deadline,
    )
    print(f"Collected {len(seeds)} top seeds (kept {len(seeds)}).")

    # ------------------------------------------------------------------
    # Stage 1 – short but intensive Adam runs on the seed pool
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Adam refinement on seed pool (stage 1) ---")
    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"[Seed {idx + 1}/{len(seeds)}]")
        seed_vec = seed_vec / seed_vec.sum()
        seq_adam, val_adam = _adam_optimize(
            start_vec=seed_vec,
            n=n,
            deadline=deadline,
            max_iters=30_000,      # short runs to keep budget for many restarts
            beta_max=400.0,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best (seed) = {best_val:.6f}")

    # If no seed succeeded, fall back to uniform.
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print(f"Fallback to uniform: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – many exploratory restarts (Adam + sub‑gradient)
    # ------------------------------------------------------------------
    print("\n--- Exploratory restarts (stage 2) ---")
    restart_counter = 0
    while time.time() < deadline - 20.0:          # keep ~20 s for final polishing
        restart_counter += 1
        # Choose how to initialise this restart
        if rng.random() < 0.7:
            # fresh Dirichlet draw with random concentration
            conc = np.exp(rng.uniform(-5.0, 1.5))  # roughly 0.006–4.5
            start_seq = _dirichlet_seed(n, conc, rng)
        else:
            # perturb current best on a few coordinates
            if best_seq is None:
                start_seq = _uniform(n)
            else:
                k = rng.integers(3, 12)
                idx = rng.choice(n, size=k, replace=False)
                total = best_seq[idx].sum()
                if total > 0:
                    new_alloc = rng.dirichlet(np.ones(k)) * total
                    start_seq = best_seq.copy()
                    start_seq[idx] = new_alloc
                else:
                    start_seq = _uniform(n)

        start_seq = _proj_simplex(start_seq)   # ensure sum=1

        # Quick Adam optimisation
        seq_adam, val_adam = _adam_optimize(
            start_vec=start_seq,
            n=n,
            deadline=deadline,
            max_iters=20_000,
            beta_max=400.0,
        )
        if seq_adam is None:
            continue
        cur_best_seq = seq_adam
        cur_best_val = val_adam

        # Follow‑up sub‑gradient descent (if time permits)
        if time.time() < deadline - 10.0:
            seq_sub, val_sub = _subgrad_optimize(
                start_seq=cur_best_seq,
                n=n,
                deadline=deadline,
                max_iters=10_000,
            )
            if val_sub < cur_best_val:
                cur_best_seq = seq_sub
                cur_best_val = val_sub

        if cur_best_val < best_val:
            best_seq = cur_best_seq.copy()
            best_val = cur_best_val
            elapsed = time.time() - start_time
            print(f"[Restart {restart_counter}] New global best = {best_val:.6f} (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # Stage 3 – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    print("\n--- Final simulated‑annealing polishing (stage 3) ---")
    if best_seq is not None and time.time() < deadline - 2.0:
        seq_sa, val_sa = _simulated_annealing(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=250_000,
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Polishing improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Scale to required ℓ₁‑norm and final sanity check
    # ------------------------------------------------------------------
    best_seq = best_seq * sqrt2n   # now sum ≈ sqrt(2n)

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective (official) = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse and return a plain list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
