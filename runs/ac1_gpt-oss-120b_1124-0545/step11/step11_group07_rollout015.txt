def _adam_optimize_scaled(start_vec, n, deadline, max_iters=2_000_000, beta_max=5000.0):
    eps = 1e-30
    v = np.log(start_vec + eps) # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta_temp = 5.0
    beta_mul = 1.001
    best_seq = None
    best_val = float('inf')
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)  # sum=1
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it} best={best_val:.6f}")

        # update temperature
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # compute smooth max weights
        conv_norm = conv / (cur_max + eps)  # max = 1
        w = np.exp(beta_temp * (conv_norm - 1.0))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of weighted sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length 3n-2
        grad_a = 2.0 * full[n-1:n-1+n]

        # transform to logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val

def _lp_improvement(seq, n, deadline, max_iters=5):
    a = seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    t = 0.3
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a, a)
        M = np.max(conv)
        # Build constraint matrix
        rows = []
        cols = []
        data = []
        for i in range(2*n - 1):
            start = max(0, i - (n - 1))
            end = min(i, n - 1)
            # k iterates from start to end (index in a)
            # variable index j = i - k
            for k in range(start, end + 1):
                j = i - k
                rows.append(i)
                cols.append(j)
                data.append(a[k])
        A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n), dtype=np.float64)
        b_ub = np.full(2*n-1, M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)  # maximize sum(b)
        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print(f"[LP] iteration {it}: linprog failed")
            break
        b = res.x
        sumb = b.sum()
        if sumb <= 1.0 + 1e-12:
            print(f"[LP] iteration {it}: no improvement (sum={sumb:.6f})")
            break
        g = b / sumb  # normalized
        a = (1 - t) * a + t * g
        # ensure sum 1 (should be)
        a = a / a.sum()
        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] iteration {it}: improved best={best_val:.6f}, sum(b)={sumb:.6f}")
        else:
            # no improvement, break
            print(f"[LP] iteration {it}: no improvement in objective")
            break
    return best_seq, best_val

"""
Propose a step‑function that minimises the official evaluate_sequence().
The search consists of the following stages (all respect the total time budget):

1. **Diversified seeding** – analytic families (uniform, triangular,
   geometric, β‑shapes, windows, random Dirichlet draws, sparse supports,
   mixtures) are generated and the best few are kept.

2. **Scaled‑smooth‑max Adam** – each seed is refined with Adam on a
   log‑sum‑exp surrogate of max(conv(f,f)).  The surrogate is scaled by the
   current maximum so that a very large temperature (β up to 10 000) is
   needed to resolve the tiny differences between convolution entries.

3. **Linear‑programming improvement** – with the current best shape f,
   solve the LP `max Σ b_j`  subject to  b_j≥0  and  (f * b)_i ≤ M,
   where M = max(conv(f,f)).  The LP pushes mass away from entries that
   saturate the convolution bound, yielding a strictly better shape.
   A few LP iterations (≤5) are enough.

4. **Projected sub‑gradient descent** – exact max‑convolution gradients
   are applied on the simplex to squeeze the last bits.

5. **Stochastic polishing** – a brief simulated‑annealing phase performs
   tiny Dirichlet re‑allocations of mass.

The final vector is scaled to the required ℓ₁‑norm (√(2n)) and returned.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)          # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

# Seed generation

def _collect_diverse_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    top_k: int,
    deadline: float,
) -> list[np.ndarray]:
    """
    Generate a pool of ℓ₁‑normalised candidates and keep the best ``top_k``
    according to the exact max‑convolution objective.
    """
    top_vals = []       # list of (value, seq)

    def try_insert(val, seq):
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])

    # deterministic families (always useful)
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.15),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _gaussian(n, sigma_frac=0.12),
        lambda n: _gaussian(n, sigma_frac=0.24),
        lambda n: _tukey_window(n, alpha=0.2),
        lambda n: _tukey_window(n, alpha=0.4),
    ]
    for fn in families:
        seq = fn(n)
        try_insert(np.max(_conv_fft(seq, seq)), seq)

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        for fn in (_geom, _sym_geom):
            seq = fn(n, r)
            try_insert(np.max(_conv_fft(seq, seq)), seq)

    # power‑law family
    for p in np.linspace(0.0, 5.0, 18):
        seq = _power(n, p)
        try_insert(np.max(_conv_fft(seq, seq)), seq)

    # Dirichlet draws (wide range of concentrations)
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 70 if conc < 0.01 else 40
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            try_insert(np.max(_conv_fft(seq, seq)), seq)

    # sparse random supports
    for _ in range(250):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        try_insert(np.max(_conv_fft(seq, seq)), seq)

    # random beta shapes
    for _ in range(300):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 5.0)
        beta = rng.uniform(0.05, 5.0)
        seq = _beta_shape(n, alpha, beta)
        try_insert(np.max(_conv_fft(seq, seq)), seq)

    # random sine windows
    for _ in range(200):
        if time.time() > deadline:
            break
        gamma = rng.uniform(0.2, 5.0)
        seq = _sine_window(n, gamma=gamma)
        try_insert(np.max(_conv_fft(seq, seq)), seq)

    # random Gaussian windows
    for _ in range(200):
        if time.time() > deadline:
            break
        sigma_frac = rng.uniform(0.05, 0.4)
        seq = _gaussian(n, sigma_frac=sigma_frac)
        try_insert(np.max(_conv_fft(seq, seq)), seq)

    # random Tukey windows
    for _ in range(150):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 0.9)
        seq = _tukey_window(n, alpha=alpha)
        try_insert(np.max(_conv_fft(seq, seq)), seq)

    # mixture of uniform & Gaussian
    for _ in range(150):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        try_insert(np.max(_conv_fft(seq, seq)), seq)

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        seq = np.asarray(previous, dtype=np.float64)
        if seq.sum() > 0:
            seq = seq / seq.sum()
            try_insert(np.max(_conv_fft(seq, seq)), seq)

    return [seq for _, seq in top_vals]

def _adam_optimize_scaled(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    beta_max: float = 10_000.0,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a *scaled* log‑sum‑exp (smooth‑max) surrogate.
    Returns the best simplex point found and its exact objective value
    (2 n · max conv).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        a = _softmax(v)                  # point on the simplex
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # raise temperature (slowly)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # ----- smooth‑max weights -----
        # scale conv so that the maximum becomes 1.0
        conv_scaled = conv / (cur_max + eps)
        w = np.exp(beta_temp * (conv_scaled - 1.0))   # max entry gets exponent 0
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping (helps stability)
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # simple Polyak step (fallback to diminishing step if needed)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Tiny‑move simulated‑annealing on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)       # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = 2.0 * n * np.max(_conv_fft(new, new))

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Linear‑programming improvement

def _build_constraint_matrix(a: np.ndarray) -> csr_matrix:
    """
    Construct the matrix A where A[i, j] = a[i - j] for i≥j, otherwise 0.
    Shape of A is (2n‑1, n).  Used in the LP ``(a * b) ≤ M``.
    """
    n = a.shape[0]
    # Offsets d = i - j range from 0 … n‑1
    offsets = np.arange(n, dtype=np.int64)
    # Each diagonal consists of the same value a[d] repeated n times
    diag_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_improvement(seq: np.ndarray, n: int, deadline: float, max_iters: int = 5):
    """
    Perform a few LP‑based upgrades.
    Returns (best_seq, best_obj) where best_obj = 2 n · max(conv).
    """
    a = seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    t = 0.3                     # convex‑combination weight

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        M = np.max(conv)

        # Build the constraint matrix for the current a
        A = _build_constraint_matrix(a)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise Σ b_j  ⇔ minimise -Σ b_j
        c = -np.ones(n, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            print(f"[LP] iteration {it}: linprog failed – {res.message}")
            break

        b = res.x
        sumb = b.sum()
        if sumb <= 1.0 + 1e-12:
            print(f"[LP] iteration {it}: no improvement (sum(b)={sumb:.6f})")
            break

        # Normalise the LP solution to the simplex and mix
        g = b / sumb
        a = (1.0 - t) * a + t * g
        a = a / a.sum()                     # safeguard against numerical drift

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] iteration {it}: improved best={best_val:.6f}, sum(b)={sumb:.6f}")
        else:
            print(f"[LP] iteration {it}: no objective improvement")
            break

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    # Length of the step‑function – a moderate size gives a good trade‑off.
    n = 4000
    sqrt2n = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Stage 0 – seed generation (≈12 % of total budget)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list):
        previous = np.asarray(previous, dtype=np.float64)
    else:
        previous = None

    seed_deadline = min(deadline, start_time + budget_s * 0.12)
    seeds = _collect_diverse_seeds(
        n=n,
        rng=rng,
        previous=previous,
        top_k=7,
        deadline=seed_deadline,
    )
    print(f"Collected {len(seeds)} promising seeds (kept top‑{len(seeds)}).")

    # ------------------------------------------------------------------
    # Stage 1 – scaled‑smooth‑max Adam refinements
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Adam refinement of seeds (stage 1) ---")
    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"[Seed {idx + 1}/{len(seeds)}]")
        seed_vec = seed_vec / seed_vec.sum()
        seq_adam, val_adam = _adam_optimize_scaled(
            start_vec=seed_vec,
            n=n,
            deadline=deadline,
            max_iters=1_200_000,
            beta_max=10_000.0,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after Adam: {best_val:.6f}")

    # fallback – should never happen but keeps the pipeline robust
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = 2.0 * n * np.max(_conv_fft(best_seq, best_seq))
        print(f"Fallback to uniform seed: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – linear‑programming improvement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming improvement (stage 2) ---")
        seq_lp, val_lp = _lp_improvement(best_seq, n=n, deadline=deadline, max_iters=5)
        if val_lp < best_val:
            best_seq = seq_lp.copy()
            best_val = val_lp
            print(f"  Updated global best after LP: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (stage 3) ---")
        seq_sub, val_sub = _subgrad_optimize(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=500_000,
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated global best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing (stage 4) ---")
        seq_sa, val_sa = _simulated_annealing(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=200_000,
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated global best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Scale to the required ℓ₁‑norm (√(2n)) and final sanity check.
    # ------------------------------------------------------------------
    best_seq = best_seq * sqrt2n

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective (official) ≈ {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
