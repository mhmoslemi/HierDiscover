    v = np.zeros_like(a)
    beta = beta_start

    t = 0

        # compute softmax weights

def _adam_optimize(start_vec, n, deadline,
                   lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8,
                   beta_start=1.0, beta_end=200.0):
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    phase_start = time.time()
    while time.time() < deadline:
        t += 1
        # compute conv
        conv = _conv_full_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # schedule beta
        elapsed = time.time() - phase_start
        total = deadline - phase_start
        frac = min(1.0, elapsed / total)
        beta = beta_start + (beta_end - beta_start) * frac
        # compute softmax weights
        beta_conv = beta * conv
        max_bc = np.max(beta_conv)
        exp_term = np.exp(beta_conv - max_bc)
        w = exp_term / exp_term.sum()
        # gradient
        a_rev = a[::-1]
        conv_w_rev = _conv_full_fft(w, a_rev)
        g = 2.0 * conv_w_rev[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _proj_simplex(a)
        # Optionally, print progress occasionally.
        if t % 5000 == 0:
            print(f"[Adam] it={t} beta={beta:.2f} cur={cur_val:.6f} best={best_val:.6f}")
    return best_seq, best_val

def _simulated_annealing(start_seq, n, deadline,
                         max_iters=500_000,
                         init_temp_factor=0.1,
                         min_temp=1e-12):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # exponential decay
        T = max(T * 0.9995, min_temp)
        # pick random subset size
        k = rng.integers(2, 8)   # smaller for speed
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[SA] it={it} best={best_val:.6f}")
    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that beats the previous bound
(~1.50886) for the provided ``evaluate_sequence``.  

Algorithm
---------
1. **Seed generation** – a diverse library of analytic families (uniform,
   triangular, trapezoidal, geometric, power‑law, beta‑shape, Gaussian,
   Tukey, sine‑window) together with a few random Dirichlet / exponential draws.
   For each asymmetric seed its reversed copy is also added, giving many
   asymmetric candidates that tend to reduce the autocorrelation peak.

2. **Smooth‑max gradient descent** – the non‑smooth objective
   ``max(conv)`` is replaced by a temperature‑controlled
   ``log‑sum‑exp`` (soft‑max) of the convolution values.
   Its gradient can be written as a convolution of the soft‑max weights with
   the reversed current vector, which we evaluate efficiently with FFTs.
   An Adam optimiser with a linearly increasing temperature (β) drives the
   iterates while we project onto the probability simplex after each step.

3. **Stochastic polishing** – a lightweight simulated‑annealing phase
   repeatedly reshuffles a random subset of coordinates (Dirichlet move) and
   accepts downhill moves or uphill moves according to a slowly cooling schedule.

All stages respect the 1000 s wall‑time budget.  The final best vector is stored
in the global ``height_sequence_1`` variable for reuse in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (works for arbitrary lengths)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact surrogate objective for ℓ₁‑normalised vectors (sum = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_full_fft(a, a)
    return float(2.0 * n * np.max(conv))

# Analytic families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r^i, r∈(0,1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric, mass on both ends."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12),
                             np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   beta_start: float = 1.0,
                   beta_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smoothed max convolution objective.
    The temperature β (soft‑max sharpness) is increased linearly from
    ``beta_start`` to ``beta_end`` over the allotted time.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    phase_start = time.time()

    while time.time() < deadline:
        t += 1

        # convolution a*a
        conv = _conv_full_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # temperature schedule (linear in elapsed fraction)
        elapsed = time.time() - phase_start
        total = deadline - phase_start
        frac = min(1.0, elapsed / total)
        beta = beta_start + (beta_end - beta_start) * frac

        # soft‑max weights over convolution entries
        beta_conv = beta * conv
        shift = np.max(beta_conv)            # for numerical stability
        w = np.exp(beta_conv - shift)
        w /= w.sum()

        # gradient of the smoothed objective
        a_rev = a[::-1]
        conv_w_rev = _conv_full_fft(w, a_rev)      # length = 3n‑2
        g = 2.0 * conv_w_rev[n - 1:n - 1 + n]       # slice of length n

        # Adam update with bias correction
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _proj_simplex(a)       # keep on the probability simplex

        # occasional progress report
        if t % 5000 == 0:
            print(f"[Adam] it={t:6d} beta={beta:.2f} cur={cur_val:.6f} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.08,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Light SA that repeatedly redistributes mass among a random small subset
    of coordinates (Dirichlet move).  It works on the ℓ₁‑normalised domain.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential temperature decay
        T = max(T * 0.9995, min_temp)

        # choose a tiny random block (2‑12 indices)
        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        tot = cur[idx].sum()
        if tot <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[SA] it={it:6d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # small safety margin

    # ------------------------------------------------------------------
    # Choose a reasonable length – 5000 gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) and normalise it.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    previous = prev_arr / s
                else:
                    previous = None
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a modest library of seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # evaluate all seeds with the cheap surrogate (ℓ₁‑normalised)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(3, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best from the top seed
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max Adam descent (majority of the budget).
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        # allocate ~80 % of remaining time to optimisation
        opt_deadline = time.time() + 0.80 * time_left
        # run the optimiser from each of the top seeds (time split equally)
        per_seed_deadline = opt_deadline
        for idx, seed_vec in enumerate(best_seeds):
            sub_deadline = time.time() + (opt_deadline - time.time()) / (len(best_seeds) - idx)
            print(f"\n--- Adam optimisation from seed #{idx+1} ---")
            seq_opt, val_opt = _adam_optimize(
                seed_vec, n, sub_deadline,
                lr=0.02,
                beta_start=1.0,
                beta_end=200.0
            )
            if val_opt < global_best_val:
                global_best_seq, global_best_val = seq_opt, val_opt
                print(f"[Adam] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=800_000,
            init_temp_factor=0.08
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Sanity check – evaluate with the official function.
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for possible later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
