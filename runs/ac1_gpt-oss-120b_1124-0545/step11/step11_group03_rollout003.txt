def _smooth_optimize(initial, deadline):
    a = initial.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    beta = 5.0
    iter = 0
    while time.time() < deadline:
        # compute conv a*a
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # compute weights
        # to avoid overflow, shift by max before exp
        exp_vals = np.exp(beta * (conv - max_c))
        weight_sum = exp_vals.sum()
        w = exp_vals / weight_sum
        # gradient
        w_rev = w[::-1]
        grad_full = _conv_fft(a, w_rev)
        # Need slice for gradient: indices (2*n-2) - i
        start = 2*n-2
        g_slice = grad_full[start - np.arange(n)]
        grad = 2.0 * g_slice
        # step size
        lr = 0.2 / np.sqrt(iter + 1)
        a = a - lr * grad
        # projection onto simplex
        a = _proj_simplex(a)
        # occasionally evaluate
        if iter % 200 == 0:
            val = _objective_surrogate(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                # optionally print progress
        # increase beta gradually
        if iter % 5000 == 0 and iter > 0:
            beta = min(beta * 1.5, 200.0)
        iter += 1
        # optional random perturbation to escape stagnation every some steps
        if iter % 10000 == 0:
            # random redistribute small portion
            rng = np.random.default_rng()
            # shuffle small fraction
            idx = rng.choice(n, size=int(0.01*n), replace=False)
            total = a[idx].sum()
            if total > 0:
                a[idx] = rng.dirichlet(np.ones(len(idx))) * total
        # break if iteration large relative to time left
        if time.time() > deadline - 0.1:
            break
    return best_seq, best_val

"""
Propose a non‑negative step function that (approximately) minimises the
`evaluate_sequence` objective.

Algorithm
---------
1. **Seed generation** – a rich catalogue of ℓ₁‑normalised analytic
   families (uniform, windows, Gaussian, beta, U‑shaped) together with a few
   random Dirichlet draws.  The best few seeds (by the cheap surrogate
   `2·n·max(conv)`) are retained.

2. **Smooth max‑gradient descent** – minimise the smooth approximation
   of the ‑∞ norm

        f_β(a) = (1/β)·log Σ_k exp(β·conv(a,a)_k)

   using projected gradient descent on the probability simplex.
   The smoothing parameter β is increased gradually, turning the smooth
   problem into the true max‑convolution objective.  The gradient is
   computed efficiently with two FFT‑based convolutions per iteration.

3. **Refinement** – a short sub‑gradient descent (as in the reference code)
   followed by a lightweight simulated‑annealing polishing.

All stages respect the overall time budget (default 1000 s).  The best
sequence found is stored in the global name ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) (assumes Σseq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    # fallback implementation using the Bessel‑I0 series if SciPy is absent
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    # five‑term series for I0
    a = (1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 +
         (t ** 6) / 2304.0 + (t ** 8) / 147456.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _u_shape(n: int) -> np.ndarray:
    """U‑shaped (large mass at the ends, small in the centre)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + 1.0) + 1.0 / (n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max‑convolution gradient)
# ----------------------------------------------------------------------
def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on `max(conv(a,a))`."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.15 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing (lightweight final polishing)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.06,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:      # reshape a random subset
            k = rng.integers(2, 12)   # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total > 0.0:
                new_alloc = rng.dirichlet(np.ones(k)) * total
                new = cur.copy()
                new[idx] = new_alloc
            else:
                continue
        else:                       # small transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (main optimisation engine)
# ----------------------------------------------------------------------
def _smooth_optimize(initial: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution objective using projected
    gradient descent.  The smoothing parameter β is increased gradually.
    """
    a = initial.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    beta = 5.0          # initial smoothing strength
    it = 0

    # Pre‑allocate a small array for indices used in the gradient slice
    idx_arr = np.arange(n)

    while True:
        if time.time() > deadline:
            break
        it += 1

        # 1) convolution a*a  (full length 2n‑1)
        conv = _conv_fft(a, a)

        # 2) smooth‑max weights (log‑sum‑exp trick)
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))      # safe: all ≤ 1
        w = exp_vals / exp_vals.sum()                # shape (2n‑1,)

        # 3) gradient = 2 * Σ_j a_j w_{i+j}
        w_rev = w[::-1]                               # reverse
        grad_full = _conv_fft(a, w_rev)               # length 3n‑2
        start = 2 * n - 2
        grad = 2.0 * grad_full[start - idx_arr]       # length n

        # 4) projected gradient step
        lr = 0.25 / np.sqrt(it)                       # diminishing step‑size
        a = a - lr * grad
        a = _proj_simplex(a)

        # occasional logging and best‑track
        if it % 250 == 0:
            val = _objective_surrogate(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Smooth] it={it:6d} β={beta:5.2f} surrogate={val:.6f}")

        # increase smoothing strength slowly
        if it % 2000 == 0:
            beta = min(beta * 1.3, 200.0)

        # occasional random reshuffle of a tiny fraction to escape plateaus
        if it % 5000 == 0:
            rng = np.random.default_rng()
            tiny = max(1, n // 200)                # ≈0.5 % of indices
            idx = rng.choice(n, size=tiny, replace=False)
            tot = a[idx].sum()
            if tot > 0.0:
                a[idx] = rng.dirichlet(np.ones(tiny)) * tot

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    # Choose a length – 5000 offers a good trade‑off between flexibility
    # and per‑iteration cost.
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------

    previous = globals().get("height_sequence_1")
    prev_arr = None

    # Evaluate cheap surrogate and keep the best few

    top_k = 4

    print("Top surrogate values after seeding:")

        seq_sub, val_sub = _subgradient_proj(best_seq, n, deadline,
                                             max_iters=max_iters_sub)
        if val_sub < best_val:
            best_seq, best_val = seq_sub, val_sub
            print(f"[Sub‑grad] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Simulated annealing final tweak
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(400_000, remaining * 4e4))
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              max_iters=max_iters_sa)
        if val_sa < best_val:

            print(f"[SA] final surrogate = {best_val:.6f}")

    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
