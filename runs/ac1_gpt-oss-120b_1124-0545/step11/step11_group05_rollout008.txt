"""
Propose a non‑negative step‑function that (hopefully) improves the known bound
(~1.509).  The search proceeds in four stages:

0.  Generate a very large pool of candidate shapes on a modest size (n≈2000).
    Analytic families (Beta, Tukey, Gaussian …) as well as many random
    Dirichlet draws (various concentrations) are created and ℓ¹‑normalised.

1.  Upscale the best few candidates to a larger problem size (n≈6000) and
    improve each of them with an Adam optimiser that works on a
    log‑sum‑exp (soft‑max) smooth approximation of the max‑convolution.
    The temperature β starts at 5 and grows to 5000, focusing the gradient
    onto the true maximum.

2.  Refine the current global best with a mirror‑descent / exponentiated‑
    gradient step that uses the exact sub‑gradient of the max‑convolution.
    This keeps the iterate on the probability simplex without an explicit
    projection.

3.  Final polishing with simulated‑annealing (random Dirichlet reshuffles)
    to escape possible local minima.

All heavy work respects the global 1000 s budget.  The best sequence is
returned as a plain ``list[float]`` and stored in the global name
``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate objective for an ℓ¹‑normalised sequence (∑seq == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay (0<r<1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric decay from both ends."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from a small grid to a larger one, then renorm."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    else:
        ys = _uniform(n_big)
    return ys.astype(np.float64)


def _generate_small_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a dense pool of analytic and random candidates (ℓ¹‑normalised)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for wf in np.linspace(0.02, 0.45, 12):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    for gamma in np.linspace(0.0, 6.0, 31):
        seeds.append(_sine_window(n, gamma=gamma))

    for alp in np.linspace(0.0, 0.95, 20):
        seeds.append(_tukey_window(n, alpha=alp))

    for r in np.concatenate([np.linspace(0.01, 0.98, 20), np.linspace(0.98, 0.9999, 15)]):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 7.0, 29):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 20)
    betas = np.linspace(0.2, 5.0, 20)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for sigma_frac in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma_frac))

    # random draws – many of them
    for conc in (0.001, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0):
        for _ in range(30):            # 30 draws per concentration → 240 total
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if present)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size:
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0:
                    seeds.append(arr / s)
        except Exception:
            pass

    return seeds

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   beta_max: float = 5000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max‑convolution.
    Parameters are logits of a soft‑max (probability simplex) representation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature schedule
    beta_temp = 5.0          # start a bit sharper than the original code
    beta_mul = 1.01          # increase a little faster

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # temperature increase
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # transform to log‑softmax coordinates (standard recipe)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping (helps stability)
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _expgrad_descent(start_seq: np.ndarray, n: int, deadline: float,
                    max_iters: int = 500_000,
                    eta0: float = 0.3,
                    eps_active: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient refinement using the exact
    sub‑gradient of the max‑convolution.  The iterate stays on the simplex
    automatically (no explicit projection required).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)

        # active set – indices whose value is within eps of the max
        active = conv >= (cur_max - eps_active)
        if not np.any(active):
            active = np.full_like(conv, False, dtype=bool)
            active[np.argmax(conv)] = True

        w = active.astype(float)               # weight vector for active indices
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step (mirror descent w.r.t. KL)
        eta = eta0 / np.sqrt(it)
        # keep numbers in a reasonable range: subtract max for numerical stability
        upd = np.exp(-eta * grad - np.max(-eta * grad))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = _uniform(n)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polishing that reshuffles a random subset
    of the mass via Dirichlet re‑allocation.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found
    as a list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – generate a dense pool on a modest size
    # ------------------------------------------------------------------
    n_small = 2000
    print(f"\n--- Stage 0: generating {n_small}-length seed pool ---")
    small_seeds = _generate_small_seeds(n_small, rng)
    print(f"  generated {len(small_seeds)} candidates")

    # ------------------------------------------------------------------
    # Stage 1 – upscale to a larger problem size and keep the best few
    # ------------------------------------------------------------------
    n_big = 6000
    print(f"\n--- Stage 1: upscaling to length {n_big} ---")
    large_seeds = [_upscale(vec, n_big) for vec in small_seeds]

    evaluated = [(vec, _objective(vec)) for vec in large_seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k_small = min(20, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k_small]]
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"  initial best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        seeds_to_opt = min(5, len(best_seeds))
        per_seed_budget = remaining / (seeds_to_opt + 2.0)   # reserve some for later stages
        print(f"\n--- Stage 2: Adam optimisation (≈{per_seed_budget:.1f}s per seed) ---")
        for idx, seed_vec in enumerate(best_seeds[:seeds_to_opt]):
            if time.time() > deadline:
                break
            est_iter = int(per_seed_budget / 0.003)  # rough cost per iteration ≈2 convs
            est_iter = max(30_000, min(est_iter, 250_000))
            print(f"\n[Adam] seed #{idx + 1}/{seeds_to_opt}, max_iters≈{est_iter}")
            seq_adam, val_adam = _adam_optimize(
                seed_vec, n_big, deadline,
                max_iters=est_iter,
                beta_max=5000.0,
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient refinement (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_iters = int(remaining * 1200)   # about 0.001 s per iteration
        max_iters = max(200_000, min(max_iters, 2_000_000))
        print(f"\n--- Stage 3: exponentiated‑gradient refinement (max_iters={max_iters}) ---")
        seq_exp, val_exp = _expgrad_descent(
            global_best_seq, n_big, deadline,
            max_iters=max_iters,
            eta0=0.3,
        )
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[overall‑ExpGrad] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_sa_iters = int(remaining * 900)   # ~0.001 s per iteration
        max_sa_iters = max(200_000, min(max_sa_iters, 2_000_000))
        print(f"\n--- Stage 4: simulated‑annealing polishing (max_iters={max_sa_iters}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n_big, deadline,
            max_iters=max_sa_iters,
            init_temp_factor=0.04,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluator
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
