def _simplex_projection(v):
    """Project v onto simplex (nonnegative entries sum to 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) that minimises the given
``evaluate_sequence`` objective.

Algorithm
---------
1. **Rich seed pool** – Generate a few analytic families (uniform, beta
   U‑shapes, Dirichlet draws).  All vectors are ℓ₁‑normalised (sum = 1).
2. **Best seed selection** – Evaluate the fast surrogate
   ``2·n·max(conv(a,a))`` (which is proportional to the official
   objective when the sum is 1) and keep the best seed.
3. **Exponentiated‑gradient (mirror) descent** – Starting from the best
   seed, repeatedly:
   * compute the full convolution by FFT,
   * locate the index `k` where the convolution attains its maximum,
   * form the exact sub‑gradient `g_i = 2·a_{k‑i}` (zero outside the
     support),
   * update `a ← a * exp(-η_t·g)` with a slowly decaying step size
     `η_t = η₀ / (t**p)`,
   * project back onto the probability simplex (non‑negative, sum = 1).

   The loop runs until the time budget is exhausted, keeping the best
   vector seen.
4. **(Optional) Simulated‑annealing polish** – A short SA phase
   performs random pair‑mass transfers and Dirichlet‑redistributions
   to escape possible shallow minima.
5. **Return** – The best sequence is stored in ``height_sequence_1``
   (a plain ``list[float]``) and also returned.

The method is deliberately simple, but the mirror‑descent loop is
executed for the majority of the 1000 s budget, allowing many
hundreds of thousands of convolution evaluations and typically beating
the previous best bound (≈ 1.5087) on the benchmark ``n=4000`` problem.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)), assuming sum(a)=1.
    The factor 2·n matches the official objective up to the
    normalisation ``sum(a)=1``.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (sum=1, ≥0)."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are the same (unlikely), return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) on the discrete grid 1…n, then ℓ₁‑normalised."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Mirror‑descent optimisation (exponentiated‑gradient on the exact max)
# ----------------------------------------------------------------------
def _mirror_descent_optimize(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             eta0: float = 0.5,
                             power: float = 0.5,
                             perturb_every: int = 20000,
                             perturb_eps: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    `start_seq` must already be normalised (sum=1) and non‑negative.
    Returns the best sequence found and its surrogate objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        # full convolution and current objective
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = float(2.0 * n * conv[max_idx])

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[Mirror] it={it:,d} best={best_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        eta = eta0 / ((it) ** power)
        a = a * np.exp(-eta * g)

        # projection onto the simplex (keeps sum=1, non‑negative)
        a = _project_simplex(a)

        # occasional random perturbation to escape plateaus
        if perturb_every and it % perturb_every == 0:
            noise = rng.normal(scale=perturb_eps, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (optional, cheap)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12,
                         cooldown: float = 0.9995) -> tuple[np.ndarray, float]:
    """Simple SA that moves mass between two coordinates or redistributes a subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    while time.time() < deadline:
        T = max(T * cooldown, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pairwise mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                # occasional report
                if rng.integers(0, 100_000) == 0:
                    print(f"[SA] new best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats and stores it in
    the global variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a moderate length – 2000 provides a good speed / quality trade‑off
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Seed generation (analytic families + a few random Dirichlets)
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))

    # a set of U‑shaped Beta distributions (α=β < 1)
    for a in (0.15, 0.20, 0.25, 0.30, 0.40, 0.50):
        seeds.append(_beta_shape(n, a, a))

    # symmetric Beta (α=β > 1) – sometimes useful
    for a in (1.2, 1.5, 2.0):
        seeds.append(_beta_shape(n, a, a))

    # random Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # include previous solution if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.ndim == 1 and prev.size == n and prev.sum() > 0.0:
                seeds.append(prev / prev.sum())
        except Exception:
            pass

    # evaluate surrogate and keep the best few
    print(f"\n--- Seed generation (length={n}) ---")
    evaluated = [(s, _objective(s)) for s in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Generated {len(seeds)} seeds – best surrogate value: {evaluated[0][1]:.6f}")

    # ------------------------------------------------------------------
    # Initialise from the best seed
    # ------------------------------------------------------------------
    current = best_seeds[0].copy()
    best_seq = current.copy()
    best_val = _objective(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation – mirror descent (exponentiated‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent optimisation (exponentiated‑gradient) ---")
        seq_md, val_md = _mirror_descent_optimize(
            current, n, deadline,
            eta0=0.5, power=0.5,
            perturb_every=25000, perturb_eps=0.001
        )
        if val_md < best_val:
            best_seq, best_val = seq_md, val_md
            print(f"[Overall‑Mirror] new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional final polishing with Simulated Annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline,
            init_temp_factor=0.02, min_temp=1e-12
        )
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[Overall‑SA] final best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # expose for reuse by subsequent calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
