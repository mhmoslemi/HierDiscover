"""
Propose a non‑negative step‑function that (empirically) minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Rich seed pool** – Analytic families (uniform, triangular, cosine,
   sine, Gaussian, exponential, power‑law, Beta with very small shape
   parameters, arcsine‐like ½‑½ Beta) are generated on a dense grid.
   All seeds are ℓ₁‑normalised (sum = 1) and evaluated with a fast FFT‑based
   surrogate 2·n·max(conv) (identical ordering to the official metric).

2. **Adam on a smoothed max** – The best ≈20 seeds are refined with Adam
   optimisation of the log‑sum‑exp surrogate of the max.  A temperature
   parameter is ramped from 1.0 to 500.0, the learning‑rate decays as
   1/√t, and the optimisation stops when the global time budget is
   exhausted.

3. **Mirror‑descent (exponentiated‑gradient) sub‑gradient** – The current
   best sequence undergoes exact sub‑gradient descent on the true max
   (the active convolution index) with a slower step‑size decay
   (t⁻⁰·³).

4. **Population crossover & mutation** – A tiny population (size ≤ 6) of
   top sequences is maintained.  Random convex mixtures and brief
   Dirichlet perturbations are tried; improving offspring replace the
   worst population members.

5. **Simulated‑annealing polishing** – Random pair‑wise mass transfers,
   swaps and subset redistributions are accepted with a geometric cooling
   schedule to escape shallow minima.

All stages respect the 1000 s wall‑time limit, and the best sequence found
is stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Helper utilities – convolution, soft‑max, simplex projection, etc.
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact official objective (up to the clipping performed inside
    ``evaluate_sequence``).  The sequence is assumed to be non‑negative.
    """
    n = len(seq)
    s = seq.sum()
    if s < 0.01:                     # protect against division‑by‑zero in the
        return np.inf                # official evaluator
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv) / (s * s))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – maps logits to a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta pdf on the discrete grid {1,…,n}.  Works for any α,β > 0."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑distributed probability vector."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """A smooth non‑negative curve built from a few random cosines."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a followed by simplex re‑normalisation."""
    mixed = (1.0 - alpha) * a + alpha * b
    return _project_simplex(mixed)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on the log‑softmax parametrisation with a temperature‑controlled
    log‑sum‑exp surrogate for max(conv).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    beta1, beta2 = 0.9, 0.999
    lr0 = 0.9
    beta = 1.0          # start temperature
    beta_inc = 2.0
    beta_max = 500.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # point on simplex
        conv = _conv_fft(a, a)
        cur_surrogate = 2.0 * n * np.max(conv)   # ordering identical to real objective
        if cur_surrogate < best_val:
            best_val = cur_surrogate
            best_seq = a.copy()
            # occasional progress report
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best_surrogate={best_val:.6f}")

        # ---- smooth max via log‑sum‑exp (temperature = beta) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # back‑prop through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max using exponentiated‑gradient."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12
    power = 0.35

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # compute sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update (mirror descent)
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # re‑project onto simplex
        a = _project_simplex(a)

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 1_500_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA with swaps, Dirichlet subset moves and pairwise mass transfers."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:          # plain swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:        # redistribute a random subset
            ksize = rng.integers(3, 12)
            idx = rng.choice(n, size=ksize, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(ksize, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                       # pairwise mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation – richer families and parameter grids
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # Sine window
    for gamma in np.linspace(0.3, 4.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian family
    for sigma in np.linspace(0.08, 0.45, 10):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Power‑law
    for p in np.linspace(0.0, 4.0, 9):
        seeds.append(_power(n, p))

    # Beta family – dense grid, *including* highly U‑shaped parameters*
    alphas = np.geomspace(0.04, 5.0, 12)   # from very small to >1
    betas = np.geomspace(0.04, 5.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Exponential decay
    for lam in np.linspace(0.0005, 0.015, 8):
        seeds.append(_exponential_decay(n, lam))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(_project_simplex(avg))

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a list of floats and stores it in the
    global variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a tiny safety margin for the final sanity check
    deadline = start_time + budget_s - 3.0

    # length of the step function – 4000 works well and matches the reference
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # 0) Build an extensive pool of seeds and keep the top few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 1) Adam optimisation on the smoothed max (one run per top seed)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        # allocate roughly equal time per Adam run, but cap the iteration count
        remaining = deadline - time.time()
        # each iteration is cheap – we conservatively allow up to 150k iters
        max_iters = min(150_000, int(remaining * 80))
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best surrogate = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 2) Mirror‑descent sub‑gradient refinement on the exact max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(global_best_seq,
                                                             n, deadline,
                                                             max_iters=2_500_000)
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # 3) Population crossover & mutation (exploit promising region)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Population crossover & mutation ---")
        pop_size = 6
        # initialise population with the best few candidates we have
        pop = [global_best_seq] + best_seeds[1:pop_size]
        pop = [p / p.sum() for p in pop]    # ensure exact simplex

        while time.time() < deadline:
            # pick two parents uniformly
            a, b = rng.choice(pop, size=2, replace=False)
            alpha = rng.random()
            child = _mix_shapes(a, b, alpha)

            # occasional small Dirichlet noise to escape plateaus
            if rng.random() < 0.15:
                noise = _random_dirichlet(n, 0.1, rng)
                child = _mix_shapes(child, noise, rng.random())

            child_val = _objective(child)
            if child_val < global_best_val:
                global_best_seq = child.copy()
                global_best_val = child_val
                print(f"[Crossover] improved global best = {child_val:.6f}")

            # keep the best `pop_size` individuals
            pop.append(child)
            pop.sort(key=lambda s: _objective(s))
            pop = pop[:pop_size]

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing polishing (fine‑grained local search)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(global_best_seq,
                                                       n, deadline,
                                                       max_iters=1_200_000)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
