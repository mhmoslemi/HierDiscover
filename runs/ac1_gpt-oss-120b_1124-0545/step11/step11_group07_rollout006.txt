"""
Hybrid optimisation for the step‑function bound.

The routine works in three phases:

1. **Seed preparation** – use the previously stored sequence (if any)
   and a uniform‑distribution seed.  Both are normalised to sum‑1 and
   the better one (lower exact max‑convolution) is kept.

2. **Adam refinement** – a relatively long Adam run on a smooth
   log‑sum‑exp surrogate of the convolution max reduces the objective
   quickly.  The optimiser works on the probability simplex (softmax
   parametrisation) and respects the time budget.

3. **Linear‑programming flattening** – given the current candidate `f`,
   we solve a linear program

        minimise   t
        subject to (f * g)_k ≤ t  (k = 0 … 2n‑2)
                   ∑ g_i = 1 ,  g_i ≥ 0

   which finds a distribution `g` that makes the *cross* convolution with
   `f` as flat as possible.  We then move a fraction `α` towards `g`
   (convex combination) and repeat a handful of times.  This is the
   alternating scheme described in the problem statement and is very
   effective at lowering the peak of the *self*‑convolution.

4. **Final scaling** – the best probability vector is multiplied by
   √(2 n) so that the official evaluator returns the max‑convolution
   value directly.

All steps respect the global 1000 s budget and the best sequence found
is stored in the global variable ``height_sequence_1``.
"""

import time
import math

import numpy as np
from scipy import sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
    max_k (seq * seq)[k].
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


# ----------------------------------------------------------------------
# Simple families – only a few are needed for this runner
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Adam optimisation on the smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(start_vec: np.ndarray, n: int,
                   deadline: float, max_iters: int = 1_500_000,
                   beta_max: float = 500.0) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on a smooth approximation of the maximum."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits → simplex via soft‑max
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                  # point on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6e}")

        # raise temperature slowly
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)           # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum                    # soft‑max over convolution entries

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)       # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming step (flattening the convolution)
# ----------------------------------------------------------------------


def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Sparse Toeplitz matrix T (shape (2n‑1, n)) such that T @ g = f * g
    (linear convolution).  Each diagonal contains the whole vector f.
    """
    n = len(f)
    # create n diagonals, each equal to f (length n)
    diags = [f] * n
    offsets = np.arange(n)               # 0 … n‑1
    T = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return T


def _lp_minimax(f: np.ndarray, deadline: float) -> tuple[np.ndarray | None, float]:
    """
    Solve   minimise t   subject to   (f * g)_k ≤ t,
    ∑ g_i = 1, g_i ≥ 0.
    Returns (g, t) if successful.
    """
    n = len(f)
    T = _conv_matrix(f)                 # (2n‑1, n) sparse
    # column of -1 for variable t
    minus_one = sp.csr_matrix((-np.ones((2 * n - 1, 1)), (np.arange(2 * n - 1), np.zeros(2 * n - 1))),
                               shape=(2 * n - 1, 1))
    A_ub = sp.hstack([T, minus_one], format="csr")
    b_ub = np.zeros(2 * n - 1)

    # equality: sum(g) = 1, t unrestricted (but we bound it ≥0)
    A_eq = sp.csr_matrix(np.append(np.ones(n), 0.0).reshape(1, -1))
    b_eq = np.array([1.0])

    c = np.append(np.zeros(n), 1.0)      # minimise t
    bounds = [(0, None)] * n + [(0, None)]

    # call the HiGHS LP solver (fast and accepts sparse matrices)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method="highs",
                  options={"presolve": True, "time_limit": max(0.0, deadline - time.time())})

    if not res.success:
        return None, None
    g = np.maximum(res.x[:n], 0.0)
    total = g.sum()
    if total == 0:
        return None, None
    g = g / total                       # re‑normalise to sum = 1
    t = float(res.x[-1])
    return g, t

# ----------------------------------------------------------------------
