def _project_simplex(v):
    # ensure nonnegative and sum to 1
    # Using algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    # returns projection of v onto probability simplex (sum=1, v>=0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        # all zero after projection
        return np.full_like(v, 1.0/len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _subgradient_descent(start, n, deadline, max_iters=10_000_000):
    a = start.copy()
    best_a = a.copy()
    best_val = _objective(a)
    # We'll adapt step size.
    lr0 = 0.5
    power = 0.30
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # compute gradient
        g = np.zeros_like(a)
        # For each i where i+j = max_idx and both within 0..n-1
        # Equivalent to reflect as earlier:
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # gradient descent step (add negative)
        lr = lr0 / ((it) ** power)
        # a_new = a - lr * g
        a_new = a - lr * g
        # project onto simplex
        a = _project_simplex(a_new)
        # optionally clamp negative due to numeric?
        # ensure a >=0
        a[a < 0] = 0
        # re-normalize after clamp
        s = a.sum()
        if s>0:
            a /= s
        else:
            a = _uniform(n)
    return best_a, best_val

"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

The algorithm
----------------
1. **Rich seed pool** – Analytic families (uniform, triangular,
   cosine, exponential, Gaussian, power‑law, beta, Tukey) and a few
   random Dirichlet draws are ℓ₁‑normalised (∑a = 1).  The best few
   seeds according to the cheap surrogate 2·n·max(conv) are kept.

2. **Mirror‑descent sub‑gradient** – The objective 2·n·max(conv) is
   non‑smooth but its sub‑gradient is easy: if the maximal convolution
   entry occurs at shift *k*, the gradient is the vector
   ``g[i] = 2·a[k‑i]`` (zero where the index is out of range).  We
   perform an exponentiated‑gradient (mirror) update
   ``a ← a·exp(-η·g)`` and re‑project onto the probability simplex.
   The step size follows ``η₀ / t**p`` with a relatively slow decay
   (p≈0.30) to keep the algorithm exploratory.  The routine is run
   repeatedly with different random starts until the time budget is
   exhausted.

3. **Adam polishing** – The best sequence found by the mirror step is
   refined with Adam on a log‑softmax parametrisation, using a
   high‑temperature log‑sum‑exp surrogate (β up to 1000) that
   approximates the true max.  This typically yields a small extra
   improvement.

All heavy work uses FFT‑based convolution (O(n log n)).  The code
checks the remaining time frequently, guaranteeing a return before the
1000 s budget expires.  The final sequence is stored in the global
``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised
    sequence (∑seq = 1).  The factor 2·n is exactly the quantity that
    ``evaluate_sequence`` returns up to the normalisation.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.  Implements the algorithm of
    Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only supports 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)  (ℓ₁‑normalised)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑softmax parametrisation of a
    ℓ₁‑normalised vector.  The objective is the smoothed
    log‑sum‑exp surrogate of ``max(conv)``.  The temperature ``beta``
    grows up to 1000 so that the surrogate approaches the true max.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                     # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta = 0.5
    beta_inc = 0.5
    beta_max = 1000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # a lies on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # -------- smoothed max (log‑sum‑exp) ----------
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a  (linear in a)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain‑rule through the softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the *exact*
    non‑smooth objective 2·n·max(conv).  The step size follows
    η_t = η₀ / t**p with p≈0.30 (slow decay) to keep the dynamics
    exploratory.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # hyper‑parameters (tuned empirically)
    eta0 = 0.3
    power = 0.30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative (exponentiated‑gradient) update
        eta = eta0 / ((it) ** power)
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex (KL‑projection = normalisation)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional random jitter to escape shallow plateaux
        if it % 250_000 == 0:
            jitter = np.exp(0.02 * (np.random.default_rng().normal(size=n)))
            a = a * jitter
            a = _project_simplex(a)

    return best_seq, best_val

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:

    # Gaussian with several bandwidths
    for sigma in (0.07, 0.12, 0.20):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # power‑law
    for p in (0.5, 1.5, 3.0):
        seeds.append(_power(n, p))

    # beta shapes on a grid
    for a in (0.5, 1.0, 2.0):

    for conc in (0.05, 0.2, 0.8, 2.0):

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:

    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by subsequent runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
