def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0
    n_small = 2000
    # generate seeds
    seeds = []
    # uniform
    seeds.append(_uniform(n_small))
    # trapezoidal
    for wf in np.linspace(0.01, 0.4, 12):
        seeds.append(_trapezoidal(n_small, width_fraction=wf))
    # beta symmetric fine grid
    alphas = np.concatenate([
        np.linspace(0.51, 1.0, 25, endpoint=False),
        np.linspace(1.0, 2.0, 25)
    ])
    for a in alphas:
        seeds.append(_beta_shape(n_small, a, a))
    # beta asymmetric small sample
    for a in [0.6, 0.8, 1.2]:
        for b in [0.8, 1.0, 1.2]:
            seeds.append(_beta_shape(n_small, a, b))
    # power law
    for p in np.linspace(0.0, 4.0, 25):
        seeds.append(_power(n_small, p))
    # geometric
    for r in np.linspace(0.01, 0.99, 20):
        seeds.append(_geom(n_small, r))
        seeds.append(_sym_geom(n_small, r))
    # random dirichlet
    for conc in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.5, 1.0, 2.0]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n_small, conc, rng))
    # random exponential
    for _ in range(8):
        seeds.append(_random_exponential(n_small, rng))
    # previous best if exists
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            arr = np.maximum(arr, 0.0)
            total = arr.sum()
            if total>0:
                seeds.append(arr/total)
        except Exception: pass

    # Evaluate.
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    best_small = [vec for vec, val in evaluated[:top_k]]
    print(f"Stage0: generated {len(seeds)} seeds, top {top_k} best small surrogate values.")
    for i in range(top_k):
        print(f"  #{i+1}: {evaluated[i][1]:.6f}")

    # upscale
    n_large = 5000
    large_seeds = [_upscale(v, n_large) for v in best_small]

    # evaluate large seeds
    evaluated_large = [(vec, _objective(vec)) for vec in large_seeds]
    evaluated_large.sort(key=lambda x: x[1])
    global_best_seq = evaluated_large[0][0].copy()

"""
Non‑negative step‑function search (≈1000 s).

The routine builds a large pool of analytically‑parameterised candidates,
including a new “double‑edge Gaussian” family that concentrates mass near
both ends of the interval (the most promising shape for flattening the
self‑convolution).  After a cheap surrogate evaluation on a modest size
(≈2000) we keep the best few, upscale them to the final length (≈5000)
and polish the best candidates with three optimiser phases:

1. A short Adam run on a smooth log‑sum‑exp surrogate of the max‑convolution
   (fast, helps escape shallow local minima).
2. A long projected sub‑gradient descent directly on the exact max‑convolution.
   The cheap O(n) update lets us run millions of iterations in the remaining
   budget.
3. A final simulated‑annealing “mass‑reshuffle” polish.

All phases respect the global time budget; the best sequence is returned
as a plain ``list[float]`` and stored in ``height_sequence_1`` for reuse.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate objective for a ℓ¹‑normalised sequence (∑seq == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """One‑sided geometric decay (0<r<1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_edge_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """
    Two symmetric Gaussians centred at the two ends of the interval.
    ``sigma_frac`` is the standard deviation expressed as a fraction of n.
    """
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    left = np.exp(- (i ** 2) / (2.0 * sigma ** 2))
    right = np.exp(- ((i - (n - 1)) ** 2) / (2.0 * sigma ** 2))
    a = left + right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   beta_max: float = 2500.0) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on the log‑sum‑exp surrogate of max‑convolution."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # initial logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 10.0
    beta_mul = 1.006

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 500_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a Polyak‑style diminishing step‑size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polishing that reshuffles a random subset
    of the mass (Dirichlet re‑allocation).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a small‑n sequence to a larger n."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    else:
        ys = _uniform(n_big)
    return ys.astype(np.float64)


def _generate_small_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a fairly dense pool of analytically‑parameterised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic baselines
    seeds.append(_uniform(n))
    seeds.append(_trapezoidal(n, width_fraction=0.15))

    # trapezoidal sweep
    for wf in np.linspace(0.02, 0.4, 12):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    # symmetric beta (fine grid around 0.5–1.5)
    betas = np.concatenate([
        np.linspace(0.55, 0.95, 20, endpoint=False),
        np.linspace(1.0, 2.0, 20)
    ])
    for a in betas:
        seeds.append(_beta_shape(n, a, a))

    # asymmetric beta – a few exploratory offsets
    for a in (0.7, 0.9, 1.1):
        for b in (0.7, 0.9, 1.1):
            seeds.append(_beta_shape(n, a, b))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 25):
        seeds.append(_power(n, p))

    # geometric families
    for r in np.concatenate([np.linspace(0.01, 0.90, 15),
                             np.linspace(0.90, 0.99, 10)]):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # double‑edge Gaussian – new promising family
    for sigma_frac in np.linspace(0.07, 0.30, 12):
        seeds.append(_double_edge_gaussian(n, sigma_frac))

    # random Dirichlet drawings (various concentrations)
    for conc in (0.001, 0.005, 0.02, 0.1, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # previous best (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 0:
                seeds.append(arr / s)
        except Exception:
            pass

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget and returns the best
    sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – generate and evaluate a rich collection of analytic seeds
    # ------------------------------------------------------------------
    n_small = 2000
    print(f"\n--- Stage 0: generating {n_small}‑length analytic candidates ---")
    small_seeds = _generate_small_seeds(n_small, rng)
    print(f"Generated {len(small_seeds)} candidates; evaluating surrogate objective...")

    evaluated_small = [(vec, _objective(vec)) for vec in small_seeds]
    evaluated_small.sort(key=lambda x: x[1])

    top_k_small = min(30, len(evaluated_small))
    best_small = [vec for vec, _ in evaluated_small[:top_k_small]]
    print(f"Top {top_k_small} small candidates (surrogate values):")
    for i, (_, val) in enumerate(evaluated_small[:top_k_small]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – upscale and keep the very best large‑scale seeds
    # ------------------------------------------------------------------
    n_large = 5000
    print(f"\n--- Stage 1: upscaling to length {n_large} ---")
    large_seeds = [_upscale(v, n_large) for v in best_small]

    evaluated_large = [(vec, _objective(vec)) for vec in large_seeds]
    evaluated_large.sort(key=lambda x: x[1])

    global_best_seq = evaluated_large[0][0].copy()
    global_best_val = evaluated_large[0][1]
    print(f"[Init] best surrogate after upscale = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – short Adam optimisation on a few elite seeds
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        seeds_to_opt = min(3, len(evaluated_large))
        per_seed_time = (deadline - time.time()) * 0.4 / seeds_to_opt
        max_iters = int(per_seed_time / 0.002)  # rough cost per iteration
        max_iters = max(40_000, min(max_iters, 150_000))
        print(f"\n--- Stage 2: Adam optimisation (≈{max_iters} it per seed) ---")
        for idx in range(seeds_to_opt):
            if time.time() > deadline:
                break
            seq_start = evaluated_large[idx][0]
            seq_adam, val_adam = _adam_optimize(seq_start, n_large, deadline,
                                                max_iters=max_iters,
                                                beta_max=2500.0)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long projected sub‑gradient refinement (cheap O(n) steps)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        remaining = deadline - time.time()
        # aim for a few million cheap iterations
        max_iters = int(remaining * 2500)   # empirical ~0.0004 s per iteration on n≈5000
        max_iters = max(500_000, min(max_iters, 3_000_000))
        print(f"\n--- Stage 3: sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(global_best_seq, n_large, deadline,
                                               max_iters=max_iters,
                                               init_lr=0.20)
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_sa_iters = int(remaining * 900)   # ~0.0011 s per SA step
        max_sa_iters = max(200_000, min(max_sa_iters, 2_000_000))
        print(f"\n--- Stage 4: simulated‑annealing polishing (max_iters={max_sa_iters}) ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq, n_large, deadline,
                                              max_iters=max_sa_iters,
                                              init_temp_factor=0.05)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final check with the official evaluation function
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
