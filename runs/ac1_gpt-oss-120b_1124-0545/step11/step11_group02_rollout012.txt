def _smooth_gradient_descent(start_vec, n, deadline, max_iters=..., init_lr=0.5, init_alpha=0.5, alpha_mul=1.02, max_alpha=30.0):
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    m = np.zeros_like(a)  # for Adam?
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute conv
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 5000 == 0:
                print(f"[SmoothGD] iter={t} best={best_val:.6f}")
        # compute weights
        # α schedule
        alpha = min(max_alpha, init_alpha * (alpha_mul ** t))
        # stable softmax
        c = conv.max()
        # subtract max for numerics
        exp_vals = np.exp(alpha * (conv - c))
        w = exp_vals / exp_vals.sum()
        # gradient via convolution of reversed a and w
        a_rev = a[::-1].copy()
        grad_full = _conv_fft(a_rev, w)
        # take central slice
        grad = 2.0 * grad_full[n-1:n-1 + n]
        # optional: clip gradient to avoid huge steps
        # use Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = init_lr / np.sqrt(t)  # base lr schedule
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _proj_simplex(a)
    return best_seq, best_val

def _smooth_gradient_descent(start_vec, n, deadline, init_lr=0.3, init_alpha=0.2, max_alpha=20.0):
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    lr = init_lr
    t = 0
    a_rev = None (to compute each iteration).
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        cur = 2.0*n*conv.max()
        if cur < best_val:
            best_val = cur
            best_seq = a.copy()
            if t % 5000 == 0:
                print(f"[SmoothGD] iter={t} best={best_val:.6f}")
        # compute softmax weights
        alpha = init_alpha + (max_alpha - init_alpha) * min(1.0, (t / 200000.0)) # adjust scaling maybe
        # stable softmax
        conv_max = conv.max()
        exp_vals = np.exp(alpha * (conv - conv_max))
        w = exp_vals / exp_vals.sum()
        # gradient via convolution
        a_rev = a[::-1]

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _proj_simplex(a)
    return best_seq, best_val

"""
Hybrid optimiser for the auto‑convolution bound.

The routine collects a set of analytic seed vectors, then refines them by
1. entropic mirror descent on the peak convolution entry,
2. a smooth gradient descent that minimises a soft‑max approximation of the
   ℓ∞‑convolution (softmax weighting of all entries),
3. a top‑k multi‑sub‑gradient step that flattens the whole profile, and
4. a simulated‑annealing polish.

All updates stay on the probability simplex (non‑negative, sum = 1) and the
search respects the 1000 s time budget.  The best sequence found is stored
in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective for ℓ₁‑normalised sequences (∑aᵢ = 1).

    Returns 2·n·max(conv) where conv = seq * seq.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Assemble a modest pool of ℓ₁‑normalised candidate vectors.
    Includes several deterministic families, random Dirichlet / exponential
    draws and the previous best solution (if compatible).
    """
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12),
                             np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – diverse concentration parameters
    for conc in (0.01, 0.03, 0.07, 0.15, 0.3, 0.6, 1.2, 2.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any and compatible)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _exp_mirror_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Entropic mirror descent on the active (single‑peak) convolution entry.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Mirror] it={it:6d} best={best_val:.6f}")

        idx = np.arange(n)
        rev = max_idx - idx
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)          # entropic step
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = _uniform(n)
        else:
            a = a / a.sum()

    return best_seq, best_val


def _subgradient_multi(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    init_lr: float = 0.30,
    top_k: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent averaging the gradients of the top‑k
    largest convolution entries.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        cur_max = conv[idxs].max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[MultiSub] it={it:7d} best={best_val:.6f}")

        g = np.zeros_like(a)
        for k in idxs:
            idx = np.arange(n)
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += 2.0 * a[rev[mask]]
        g /= float(len(idxs))

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _smooth_gradient_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    init_lr: float = 0.5,
    init_alpha: float = 0.2,
    max_alpha: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a soft‑max approximation of the max‑convolution.
    The soft‑max parameter α grows linearly from ``init_alpha`` to
    ``max_alpha`` over the allotted time; Adam‑style moment estimates are
    used for stability.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    phase_start = time.time()
    total_time = max(1e-12, deadline - phase_start)

    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Smooth] it={it:6d} best={best_val:.6f}")

        # schedule α
        elapsed = time.time() - phase_start
        frac = min(1.0, elapsed / total_time)
        alpha = init_alpha + (max_alpha - init_alpha) * frac

        # soft‑max weights (stable)
        conv_max = conv.max()
        exp_vals = np.exp(alpha * (conv - conv_max))
        w = exp_vals / exp_vals.sum()

        # gradient = 2 * (a_rev * w) central slice
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = init_lr / np.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing that reshuffles a random subset of
    coordinates via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)          # number of indices to move
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

"""
Hybrid optimiser for the auto‑convolution bound.

The routine collects a set of analytic seed vectors, then refines them by
1. entropic mirror descent on the peak convolution entry,
2. a smooth gradient descent that minimises a soft‑max approximation of the
   ℓ∞‑convolution (softmax weighting of all entries),
3. a top‑k multi‑sub‑gradient step that flattens the whole profile, and
4. a simulated‑annealing polish.

All updates stay on the probability simplex (non‑negative, sum = 1) and the
search respects the 1000 s time budget.  The best sequence found is stored
in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective for ℓ₁‑normalised sequences (∑aᵢ = 1).

    Returns 2·n·max(conv) where conv = seq * seq.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Assemble a modest pool of ℓ₁‑normalised candidate vectors.
    Includes deterministic families, diverse random draws and the previous
    best solution (if compatible).
    """
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate(
        [np.linspace(0.2, 0.98, 12), np.linspace(0.98, 0.9995, 8)]
    )
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – a range of concentration parameters
    for conc in (0.01, 0.03, 0.07, 0.15, 0.3, 0.6, 1.2, 2.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any and size matches)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = init_lr / np.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Light‑weight simulated‑annealing that reshuffles a random subset of
    coordinates via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)          # number of indices to move
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The routine respects the time budget
    and returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) and decide the length to use.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    n = previous.shape[0] if (previous is not None and previous.ndim == 1) else 4000
    print(f"\n--- Using vector length n = {n} ---")

    # ------------------------------------------------------------------
    # Stage 1 – seed collection
    # ------------------------------------------------------------------
    print("\n--- Collecting seeds ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    best_seed = evaluated[0][0].copy()
    global_best_seq = best_seed
    global_best_val = evaluated[0][1]
    print(f"[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deterministic refinements
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        # Mirror descent – 20 % of remaining time
        mirror_deadline = time.time() + 0.20 * time_left
        print("\n--- Exponential mirror descent ---")
        seq_m, val_m = _exp_mirror_descent(
            global_best_seq, n, mirror_deadline, max_iters=400_000, eta0=0.6
        )
        if val_m < global_best_val:
            global_best_seq, global_best_val = seq_m, val_m
            print(f"[Mirror] new global best = {val_m:.6f}")

    time_left = deadline - time.time()
    if time_left > 0:
        # Smooth gradient descent – 50 % of remaining time
        smooth_deadline = time.time() + 0.50 * time_left
        print("\n--- Smooth gradient descent (softmax) ---")
        seq_s, val_s = _smooth_gradient_descent(
            global_best_seq,
            n,
            smooth_deadline,
            init_lr=0.5,
            init_alpha=0.2,
            max_alpha=1500.0,
        )
        if val_s < global_best_val:
            global_best_seq, global_best_val = seq_s, val_s
            print(f"[Smooth] new global best = {val_s:.6f}")

    time_left = deadline - time.time()
    if time_left > 0:
        # Multi‑sub‑gradient refinement – use the remaining deterministic budget
        sub_deadline = time.time() + 0.30 * time_left
        print("\n--- Multi‑sub‑gradient refinement ---")
        seq_sub, val_sub = _subgradient_multi(
            global_best_seq,
            n,
            sub_deadline,
            max_iters=5_000_000,
            init_lr=0.30,
            top_k=10,
        )
        if val_sub < global_best_val:
            global_best_seq, global_best_val = seq_sub, val_sub
            print(f"[MultiSub] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            max_iters=1_200_000,
            init_temp_factor=0.06,
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Official evaluation raised an exception:", exc)

    # store for possible later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
