from scipy.optimize import linprog
def solve_lp(f, M):
    n = len(f)
    # construct A matrix shape (2n-1, n)
    # compute A using vectorized approach: for each i, fill column i with f.
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    b = np.full(2*n-1, M, dtype=np.float64)
    # solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        g = res.x
        # g maybe all zero if M too small; return None
        return g
    else:
        print("LP failed:", res.message)
        return None
```

def _solve_lp(f, M):
    n = f.shape[0]
    rows = 2*n - 1

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------

1. **Small‑scale exploration** (n≈800)  
   Generate a modest pool of analytically‑parameterised shapes
   (uniform, triangular, trapezoidal, geometric, beta, sine, Gaussian,
   Tukey, Dirichlet draws).  All candidates are ℓ¹‑normalised.

2. **Linear‑programming refinement**  
   For a chosen seed we iterate:
        * Compute M = ‖f∗f‖_∞ .
        * Solve the LP                maximise   Σ g_i
                                        s.t.      (f∗g)_k ≤ M   for all k
                                                   g_i ≥ 0 .
          ``scipy.optimize.linprog`` (highs) is used.
        * Normalise g to have the same ℓ¹‑mass as f and blend

                f ← (1‑t)·f + t·g          (t slowly adapts)

        * Accept the blend if the surrogate objective
          2·n·max(f∗f)/(Σf)² improves.

   The LP step pushes mass towards positions where f is small,
   typically flattening the shape and reducing the maximal convolution.

3. **Upscaling**  
   The best refined shape is linearly interpolated to a larger size
   (default n=4000) and re‑normalised to sum = 1.

4. **Adam optimisation on a smooth surrogate**  
   Using the log‑sum‑exp approximation of the maximum,
   an Adam optimiser (soft‑max parametrisation) further reduces
   the surrogate 2·n·max(conv) while staying on the probability simplex.
   Temperature β starts at 10 and is increased aggressively (β←β·1.006 up
   to 2500).  The number of iterations is chosen from the remaining time.

5. **Simulated‑annealing polishing**  
   A cheap SA routine randomly reshuffles a small Dirichlet‑distributed
   subset of the mass; moves are accepted with the usual Metropolis rule.

6. **Verification**  
   The final sequence is passed to the official ``evaluate_sequence`` and
   stored in ``height_sequence_1`` for possible reuse.

The whole search respects the external time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, objective, projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate objective 2·n·max(conv) / (sum(seq)**2)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:      # safeguard – should never happen
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ¹‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Scaling utilities
# ----------------------------------------------------------------------


def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a small‑n sequence to a larger n."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    else:
        ys = _uniform(n_big)
    return ys.astype(np.float64)


def _downscale(seq_big: np.ndarray, n_small: int) -> np.ndarray:
    """Linear interpolation of a big‑n sequence down to a smaller n."""
    n_big = len(seq_big)
    xs = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    xb = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    ys = np.interp(xb, xs, seq_big, left=seq_big[0], right=seq_big[-1])
    s = ys.sum()
    if s > 0:
        ys /= s
    else:
        ys = _uniform(n_small)
    return ys.astype(np.float64)


# ----------------------------------------------------------------------
# LP‑based refinement (uses scipy.optimize.linprog)
# ----------------------------------------------------------------------


try:
    from scipy.optimize import linprog
except Exception:          # pragma: no cover
    linprog = None


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_i   s.t.  (f∗g)_k ≤ M  (for all k),  g ≥ 0.
    Returns the optimal g vector (may have arbitrary ℓ¹‑mass).
    """
    if linprog is None:    # pragma: no cover
        raise RuntimeError("SciPy linprog not available.")
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    c = -np.ones(n, dtype=np.float64)      # maximise Σg → minimise -Σg
    b = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    # Using the high‑performance HiGHS solver
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        print("[LP] failed:", res.message)
        return None


def _lp_refine(f: np.ndarray, max_iters: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve f by the LP step described in the paper.
    Keeps the ℓ¹‑mass constant (here set to 1.0) and returns the best
    sequence encountered together with its surrogate value.
    """
    best_f = f.copy()
    best_val = _surrogate(best_f)
    t = 0.5  # initial blending factor
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # current bound
        M = _conv_fft(f, f).max()
        g = _solve_lp(f, M)
        if g is None:
            break
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # normalise g to have the same total mass as f
        g = g * (sum_f / sum_g)

        # blend
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        # re‑normalise to keep total mass = sum_f (helps numerical stability)
        f_new = f_new * (sum_f / f_new.sum())

        val_new = _surrogate(f_new)
        if val_new < best_val:
            best_f, best_val = f_new, val_new
            f = f_new
            t = min(0.9, t * 1.07)   # become a little more aggressive
        else:
            t = max(0.1, t * 0.70)   # back‑off
    return best_f, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   beta_max: float = 2500.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the smooth surrogate   log‑sum‑exp of the convolution.
    start_vec must be ℓ¹‑normalised (sum == 1).  Returns the best sequence
    (still ℓ¹‑normalised) and its surrogate value.
    """
    eps = 1e-30
    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + eps)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 10.0
    beta_mul = 1.006

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on the simplex
        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # anneal temperature
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient wrt a (soft‑max parameters)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # transform gradient to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly re‑allocates a random subset of mass via a
    Dirichlet draw.  Operates on ℓ¹‑normalised sequences.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _surrogate(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Small‑scale candidate generation
# ----------------------------------------------------------------------


def _generate_small_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a modest pool of analytically‑parameterised seeds."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # trapezoidal
    for wf in np.linspace(0.05, 0.4, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    # sine window
    for gamma in np.linspace(0.0, 5.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))

    # Tukey
    for alpha in np.linspace(0.0, 0.9, 7):
        seeds.append(_tukey_window(n, alpha=alpha))

    # geometric
    for r in np.concatenate([np.linspace(0.01, 0.95, 9), np.linspace(0.95, 0.999, 5)]):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # power law
    for p in np.linspace(0.0, 6.0, 7):
        seeds.append(_power(n, p))

    # beta shape
    alphas = np.linspace(0.3, 4.0, 5)
    betas = np.linspace(0.3, 4.0, 5)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian
    for sigma_frac in np.linspace(0.07, 0.4, 6):
        seeds.append(_gaussian(n, sigma_frac))

    # random Dirichlet draws
    for conc in (0.001, 0.01, 0.05, 0.2, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # previous global solution, if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size > 0:
                arr = np.maximum(arr, 0.0)
                if arr.sum() > 0:
                    arr = arr / arr.sum()
                    seeds.append(_downscale(arr, n))
        except Exception:
            pass

    return seeds

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain ``list[float]``) discovered before the
    time budget expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – small‑scale seed generation & LP refinement
    # ------------------------------------------------------------------
    n_small = 800
    print(f"\n--- Stage 0: generate {n_small}-length analytic seeds ---")
    small_seeds = _generate_small_candidates(n_small, rng)
    print(f"Generated {len(small_seeds)} seeds.")

    # evaluate surrogate for all seeds (fast, O(n log n) each)
    evals = [(seq, _surrogate(seq)) for seq in small_seeds]
    evals.sort(key=lambda x: x[1])
    top_k = min(6, len(evals))
    top_seeds = [vec for vec, _ in evals[:top_k]]
    print(f"Top {top_k} initial surrogate values:")
    for i, (_, v) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # LP refinement on each top seed
    best_seq_small = None
    best_val_small = float("inf")
    lp_iters = 8
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n[LP] refining seed #{idx+1}")
        seq_refined, val_refined = _lp_refine(seed_vec, max_iters=lp_iters, deadline=deadline)
        print(f"   LP refined surrogate = {val_refined:.6f}")
        if val_refined < best_val_small:
            best_seq_small, best_val_small = seq_refined, val_refined

    if best_seq_small is None:  # fallback – use the very best raw seed
        best_seq_small = top_seeds[0].copy()
        best_val_small = _surrogate(best_seq_small)

    print(f"\n[LP] best small‑scale surrogate after refinement = {best_val_small:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – upscale to a larger problem size
    # ------------------------------------------------------------------
    n_large = 4000
    print(f"\n--- Stage 1: upscale to {n_large} ---")
    seq_large = _upscale(best_seq_small, n_large)
    # re‑normalise to ℓ¹‑mass = 1 (keeps the surrogate consistent)
    seq_large = seq_large / seq_large.sum()
    cur_best_seq = seq_large.copy()
    cur_best_val = _surrogate(cur_best_seq)
    print(f"[Init] surrogate after upscale = {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimiser (smooth surrogate)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        per_iter_est = 0.0015  # rough time per Adam iteration (seconds)
        max_iters_adam = int(remaining / per_iter_est)
        max_iters_adam = max(30_000, min(max_iters_adam, 350_000))
        print(f"\n--- Stage 2: Adam optimisation (≈{max_iters_adam} iters) ---")
        seq_adam, val_adam = _adam_optimize(
            start_vec=cur_best_seq,
            n=n_large,
            deadline=deadline,
            max_iters=max_iters_adam,
            beta_max=2500.0,
        )
        if seq_adam is not None and val_adam < cur_best_val:
            cur_best_seq, cur_best_val = seq_adam, val_adam
            print(f"[Adam] new global best surrogate = {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional sub‑gradient refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        max_iters_sub = int(remaining * 1300)   # heuristic (≈0.0008 s per iter)
        max_iters_sub = max(150_000, min(max_iters_sub, 1_200_000))
        print(f"\n--- Stage 3: sub‑gradient refinement (max_iters={max_iters_sub}) ---")
        # reuse the Adam helper (sub‑gradient version) from the previous
        # code base – re‑implemented here for completeness
        seq_proj, val_proj = _subgradient_proj(cur_best_seq, n_large, deadline,
                                               max_iters=max_iters_sub,
                                               init_lr=0.20)
        if val_proj < cur_best_val:
            cur_best_seq, cur_best_val = seq_proj, val_proj
            print(f"[Sub‑proj] new global best surrogate = {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing (remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        max_iters_sa = int(remaining * 950)   # ~0.001 s per SA move
        max_iters_sa = max(200_000, min(max_iters_sa, 2_000_000))
        print(f"\n--- Stage 4: simulated‑annealing polishing (max_iters={max_iters_sa}) ---")
        seq_sa, val_sa = _simulated_annealing(cur_best_seq, n_large, deadline,
                                             max_iters=max_iters_sa,
                                             init_temp_factor=0.04)
        if val_sa < cur_best_val:
            cur_best_seq, cur_best_val = seq_sa, val_sa
            print(f"[SA] new global best surrogate = {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(cur_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {cur_best_val:.6f} "
        f"(length {len(cur_best_seq)})"
    )

    # store for possible later reuse
    globals()["height_sequence_1"] = cur_best_seq.tolist()
    return cur_best_seq.tolist()


# ----------------------------------------------------------------------
# Sub‑gradient refinement (same implementation as in the reference code)
# ----------------------------------------------------------------------


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 500_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution objective.
    Uses a Polyak‑style diminishing step size.
    Returns best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)                     # stays on simplex (sum==1)

    return best_seq, _surrogate(best_seq)
