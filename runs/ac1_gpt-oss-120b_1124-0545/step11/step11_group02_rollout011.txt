```
def evaluate_sequence(sequence):
    # ensures list, reject empty, verify each element numeric non-boolean (int or float), not NaN, not infinite.
    # convert elements to float.
    # clip each element to [0, 1000].
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))
```

```python
"""
Propose a non‑negative step‑function (probability vector) that improves
the previous bound on `evaluate_sequence`.  The algorithm works in three
phases:
   1. Generate a handful of analytic and random seeds (including the
      previous solution if available) and keep the best of them.
   2. Run a smoothed‑max Adam optimiser (gradient descent on a
      log‑sum‑exp approximation of the ∞‑norm of the auto‑convolution
      while staying on the probability simplex).  The smoothing
      parameter β is increased gradually (continuation) so that the
      optimiser first sees a smooth landscape and finally focuses on the
      true peak.
   3. A short simulated‑annealing polish that reshuffles a random
      subset of coordinates (Dirichlet moves) accepting uphill moves
      with a decaying temperature.
   The best sequence found (as a plain list of floats) is returned and
   stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for our sizes)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float = 0.2) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Objective utilities (surrogate used during the search)
# ----------------------------------------------------------------------


def _objective(vec: np.ndarray) -> float:
    """Exact value of the evaluation function when the sum of the vector equals 1."""
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


def _grad_smoothed_max(a: np.ndarray, beta: float):
    """
    Returns a triple (smooth_max, grad, max_conv) where
      smooth_max = log‑sum‑exp_β(conv)  (a smooth approximation of max(conv))
      grad        = ∇_a smooth_max  (real‑valued gradient)
      max_conv    = max(conv) (the true quantity we finally want to minimise)
    """
    n = a.shape[0]
    # 1) auto‑convolution
    conv = _conv_fft(a, a)                     # length 2n‑1
    max_c = conv.max()
    # 2) soft‑max weights, numerically stable
    exp_beta = np.exp(beta * (conv - max_c))   # values ≤ 1
    w = exp_beta / exp_beta.sum()              # probability vector, length 2n‑1
    # 3) smooth approximation of the max
    smooth = max_c + np.log(exp_beta.sum()) / beta
    # 4) gradient of the smooth max:
    #    grad_i = 2 * sum_k a_k * w_{i+k}
    a_rev = a[::-1]
    corr = _conv_fft(a_rev, w)                # length 3n‑2
    grad = 2.0 * corr[n - 1 : n - 1 + n]       # central n entries
    return smooth, grad, max_c


def _adam_optimize(init: np.ndarray,
                   n: int,
                   deadline: float,
                   total_iters: int = 300_000,
                   init_lr: float = 0.2,
                   start_beta: float = 0.5,
                   beta_mult: float = 1.03,
                   max_beta: float = 80.0):
    """
    Adam optimiser on the smoothed max.  The iterate is always projected
    back onto the probability simplex.  The smoothing parameter β is
    multiplied by ``beta_mult`` every 200 iterations (continuation).
    """
    a = init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = start_beta

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, total_iters + 1):
        if time.time() > deadline:
            break

        smooth, grad, cur_max = _grad_smoothed_max(a, beta)

        # Adam update (beta1=0.9, beta2=0.999 as in the original paper)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)

        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = init_lr * m_hat / (np.sqrt(v_hat) + 1e-10)
        a = a - step
        a = _proj_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f} beta={beta:.2f}")

        # increase smoothing (continuation)
        if it % 200 == 0:
            beta = min(max_beta, beta * beta_mult)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 400_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12):
    """
    Light‑weight simulated‑annealing polish that repeatedly redistributes
    a random small subset of coordinates via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)         # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val

"""
Propose a non‑negative step‑function (probability vector) that improves
the previous bound on ``evaluate_sequence``.  The algorithm works in three
phases:

1. **Seed generation** – a handful of analytic families (uniform,
   triangular, Gaussian) and a few random Dirichlet draws are created.
   If a previous solution exists it is embedded (zero‑padded) and added
   to the pool.  The best seed according to the exact objective is kept.

2. **Continuation Adam optimisation** – we minimise a smooth approximation
   of the ∞‑norm of the auto‑convolution:
   ``smooth_max_β(c)=max(c)+ (1/β)·log Σ exp(β·(c−max(c)))``.
   Its gradient is
   ``∇_a smooth_max = 2·corr(a, w)`` where
   ``w`` are the soft‑max weights of the convolution entries.
   Adam updates are performed on the simplex (projection after each step).
   The smoothing parameter β is multiplied by ``beta_mult`` every 200
   iterations (continuation) so that the optimiser first sees a smooth
   landscape and finally focuses on the true peak.

3. **Simulated‑annealing polish** – a lightweight SA loop that randomly
   reshuffles a small subset of coordinates via a Dirichlet move,
   accepting downhill moves always and uphill moves with a temperature
   schedule.  The best sequence found is returned and stored in the global
   ``height_sequence_1`` for reuse.

All numerical work uses FFT‑based convolutions for speed; the whole
procedure respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float = 0.2) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------


def _objective(vec: np.ndarray) -> float:
    """Exact evaluation when ``sum(vec) == 1``."""
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


def _grad_smoothed_max(a: np.ndarray, beta: float):
    """
    Returns (smooth_max, grad, true_max) where
        smooth_max = max(c) + (1/β)·log Σ exp(β·(c−max(c)))
        grad       = ∇_a smooth_max
        true_max   = max(c)
    ``c`` is the auto‑convolution of ``a``.
    """
    n = a.shape[0]

    # auto‑convolution
    conv = _conv_fft(a, a)                # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (numerically stable)
    exp_beta = np.exp(beta * (conv - max_c))   # ≤ 1
    w = exp_beta / exp_beta.sum()              # probability vector, length 2n‑1

    # smooth approximation of the maximum
    smooth = max_c + np.log(exp_beta.sum()) / beta

    # gradient: grad_i = 2· Σ_k a_k·w_{i+k}
    a_rev = a[::-1]
    corr = _conv_fft(a_rev, w)           # length 3n‑2
    grad = 2.0 * corr[n - 1 : n - 1 + n]  # central n entries
    return smooth, grad, max_c


def _adam_optimize(init: np.ndarray,
                   n: int,
                   deadline: float,
                   total_iters: int = 300_000,
                   init_lr: float = 0.2,
                   start_beta: float = 0.5,
                   beta_mult: float = 1.03,
                   max_beta: float = 80.0):
    """
    Adam optimiser on the smoothed maximum of the auto‑convolution.
    The iterate is always projected onto the probability simplex.
    The smoothing factor β is increased every 200 iterations.
    """
    a = init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = start_beta

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, total_iters + 1):
        if time.time() > deadline:
            break

        smooth, grad, _ = _grad_smoothed_max(a, beta)

        # Adam update (β₁=0.9, β₂=0.999)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)

        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = init_lr * m_hat / (np.sqrt(v_hat) + 1e-10)
        a = a - step
        a = _proj_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f} beta={beta:.2f}")

        # continuation on β
        if it % 200 == 0:
            beta = min(max_beta, beta * beta_mult)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 400_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12):
    """Lightweight Dirichlet‑move simulated annealing polish."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)                     # move 2‑12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val
