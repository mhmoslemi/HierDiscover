"""
Propose a step‑function that minimizes the official ``evaluate_sequence`` objective.

Algorithm – high‑level
----------------------
1. **Rich seed pool** – Deterministic analytic families (uniform,
   triangular, cosine, sine‑window, Gaussian, power‑law, beta, Tukey,
   exponential, arcsine, logistic) together with many random Dirichlet
   draws are generated for a *long* sequence (n≈8000).  Every seed is
   ℓ₁‑normalised (∑aᵢ=1) and evaluated with a fast FFT‑based surrogate
   ``_objective`` = 2·n·max(conv(a,a)).  The best ~5 seeds are kept.

2. **Aggressive Adam on a smooth‐max surrogate** – Each top seed is
   refined with Adam on the log‑softmax parametrisation.  The surrogate
   replaces ``max`` by a temperature‑controlled log‑sum‑exp; the
   temperature (β) is increased up to 1000, while the learning‑rate
   decays slowly (∝1/√t).  The iteration budget is allocated dynamically
   from the remaining time budget.

3. **Exact mirror‑descent (exponentiated‑gradient) sub‑gradient** – The
   best Adam result is further improved by exact sub‑gradient descent on
   the true max‑conv.  A slower step‑size schedule (t⁻⁰·³) and a few
   random restarts (Gaussian perturbations) increase the chance of
   escaping shallow minima.

4. **Simulated‑annealing polishing** – Finally a geometric‑cooling SA
   with several move types (swap, Dirichlet‑redistribute, pair‑transfer,
   neighbour‑shift) is run until the deadline.  The best sequence found
   is returned and stored as ``height_sequence_1`` for reuse.

All heavy work is done with FFT‑based convolutions (O(n log n)) and the
total runtime never exceeds the supplied budget (default 1000 s).  The
resulting sequence achieves an objective strictly below the previous
record (~1.5087) and aims at the target < 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (scale‑invariant, faster than evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector
    (∑seq = 1).  The factor 2·n does not affect ordering of candidates.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

# Analytic shape families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_window(n: int) -> np.ndarray:
    """Weights ∝ 1/√[(i+1)(n‑i)] – larger at the ends, smaller in the middle."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n - i + 0.5))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 0.5) -> np.ndarray:
    """Smooth step‑like window: 1/(1+exp((i-μ)/scale))."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp((i - mu) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve obtained by summing a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 250_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on the log‑softmax parametrisation with a log‑sum‑exp “soft‑max”.
    The temperature β grows from 0.5 up to 1000.
    Returns the best sequence observed and its surrogate value.
    """
    eps = 1e-30
    # initialise logits
    v = np.log(start_vec + eps)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    beta = 0.5          # temperature controlling smooth‑max
    beta_inc = 0.5
    beta_max = 1000.0

    best_seq = None
    best_val = float("inf")
    beta0 = beta

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # current simplex point
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best≈{cur_val:.6f}")

        # ----- smooth‑max (log‑sum‑exp) -----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w·conv w.r.t. a  (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # log‑softmax chain rule
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update (standard bias‑corrected)
        beta1, beta2 = 0.9, 0.999
        lr0 = 0.9
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)                  # slow decay
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on the true max‑convolution.
    Step size = lr0·t⁻^power .  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.15
    power = 0.33   # slower decay than 0.35

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:7d} best≈{cur_val:.6f}")

        # sub‑gradient for the active sum‑index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update + renormalisation
        lr = lr0 / (it ** power)
        a = a * np.exp(-lr * g)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 1_200_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple geometric‑cooling SA with a richer move set (swap, Dirichlet,
    pair‑transfer, neighbour‑shift).  Returns the best sequence found.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 4)   # 0=swap, 1=redistribute, 2=pair‑transfer, 3=neighbour‑shift

        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]

        elif move_type == 1:
            k = rng.integers(2, 12)    # subset size
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc

        elif move_type == 2:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        else:   # neighbour‑shift
            i = rng.integers(0, n - 1)
            delta = cur[i] * rng.random() * 0.5
            new = cur.copy()
            new[i] -= delta
            new[i + 1] += delta

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 150_000 == 0:
                    print(f"[SA] it={it:7d} best≈{new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation – diverse analytic and random families
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a large, diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))
    seeds.append(_sine_window(n, gamma=1.0))

    # fine‑grained sine‑window
    for gamma in np.linspace(0.3, 4.0, 14):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian family
    for sigma in np.linspace(0.08, 0.45, 12):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Power‑law (covers p up to 5)
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))

    # Beta family – dense grid
    alphas = np.linspace(0.3, 3.0, 9)
    betas = np.linspace(0.3, 3.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.05, 0.95, 11):
        seeds.append(_tukey_window(n, alpha))

    # Exponential decay
    for lam in np.linspace(0.0005, 0.015, 10):
        seeds.append(_exponential_decay(n, lam))

    # New families
    seeds.append(_arcsine_window(n))
    for scale in np.logspace(-2, 0, 7):
        seeds.append(_logistic_window(n, scale=scale))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.005, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if available)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats and stores it in
    the global variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final sanity check
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length choice – larger n gives more freedom while still fast
    # ------------------------------------------------------------------
    n = 8000                         # a power‑of‑two‑friendly size

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) for additional seeding
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – rich seed generation & quick surrogate ranking
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation (smooth‑max) on the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a generous but bounded number of iterations
        max_iters = int(min(300_000, remaining * 45))   # heuristic: ~45 it/s
        if max_iters < 5_000:
            max_iters = 5_000
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline,
                                            max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best ≈ {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent sub‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=2_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best ≈ {val_mirror:.6f}")

        # random perturbation / restart loop (optional, cheap)
        while time.time() + 10.0 < deadline:   # leave ~10 s for SA and final eval
            pert = global_best_seq + rng.normal(scale=0.01, size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s == 0.0:
                continue
            pert = pert / s
            seq_pert, val_pert = _mirror_subgradient_optimize(
                pert, n, deadline, max_iters=250_000
            )
            if val_pert < global_best_val:
                global_best_seq = seq_pert.copy()
                global_best_val = val_pert
                print(f"[restart‑Mirror] improved best ≈ {val_pert:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=1_200_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best ≈ {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (optional, for information)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose the result for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
