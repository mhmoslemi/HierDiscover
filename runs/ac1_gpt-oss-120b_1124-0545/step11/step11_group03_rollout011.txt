def smooth_max_minimization(a, n, deadline):
    log_a = np.log(a + 1e-30)  # avoid log(0)
    best_a = a.copy()
    best_val = objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        a = np.exp(log_a)
        a /= a.sum()
        # compute convolution
        conv_a = _conv_fft(a, a)
        max_conv = conv_a.max()
        # compute smoothed weights exp(λ * conv)
        # schedule λ
        λ = λ0 + (λ_max - λ0) * min(1.0, it / λ_ramp_iters)   # linear increase

        # compute softmax
        # to avoid overflow: subtract max conv
        exp_term = np.exp(λ * (conv_a - conv_a.max()))
        w = exp_term / exp_term.sum()

        # compute gradient  g_i = 2 * sum_k w_k * a_{k-i}
        rev_a = a[::-1]
        # convolution w * rev_a
        # use np.convolve (maybe slower)
        # Use FFT:
        L = len(w) + len(rev_a) - 1
        N = _next_pow2(L)
        fw = np.fft.rfft(w, n=N)
        fa = np.fft.rfft(rev_a, n=N)
        conv_rev = np.fft.irfft(fw * fa, n=N)[:L]
        # slice appropriate part: indices n-1 : n-1+n
        g = 2.0 * conv_rev[n-1:n-1+n]

        # update log_a via gradient descent: log_a - η * g
        # step size schedule: η = η0 / sqrt(it) perhaps.
        η = η0 / np.sqrt(it)  # produce decreasing step
        log_a = log_a - η * g

        # renormalize: ensure sum of exp(log_a) = 1. Equivalent to subtract logsumexp
        # compute logsumexp
        max_log = np.max(log_a)
        log_a = log_a - (max_log + np.log(np.sum(np.exp(log_a - max_log))))

        # evaluate surrogate objective
        a = np.exp(log_a)  # new distribution
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # optionally print improvement
            if it % 5000 == 0:
                print(f"[Smooth] iter={it} best={best_val:.6f}")

    return best_a, best_val

def _targeted_move_opt(seq, n, deadline):
    best_seq = seq.copy()
    best_val = _objective_surrogate(seq)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(seq, seq)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_range = np.arange(i_start, i_end + 1)
        j_range = max_idx - i_range
        contributions = seq[i_range] * seq[j_range]

        # identify source and dest indices based on contributions
        src_local = np.argmax(contributions)
        dst_local = np.argmin(contributions)
        src_idx = i_range[src_local]
        dst_idx = i_range[dst_local]

        # if src and dst same, break
        if src_idx == dst_idx:
            # try random move if cannot improve
            continue

        eps = min(seq[src_idx] * 0.2, 0.01)  # not exceed 0.01 mass
        if eps <= 0:
            # cannot move
            continue

        # propose new seq
        new_seq = seq.copy()
        new_seq[src_idx] -= eps
        new_seq[dst_idx] += eps

        # Ensure non-negative (already)
        new_val = _objective_surrogate(new_seq)

        if new_val < best_val - 1e-12:
            # accept
            seq = new_seq
            best_seq = new_seq.copy()
            best_val = new_val

        # else reject; maybe try random move else.
        if it % 5000 == 0:
            print(f"[Targeted] iter={it} best={best_val:.6f}")

    return best_seq, best_val

"""
Propose a step‑function that minimizes ``evaluate_sequence``.
The algorithm consists of four phases:

1. **Initialisation** – start from the previously found best sequence
   (or a uniform distribution if none is available).

2. **Smooth max‑optimisation** – treat the objective 2·n·max(conv) as a
   smooth log‑sum‑exp approximation and run a few L‑BFGS‑B steps on the
   log‑softmax parametrisation of the vector.  The smoothing parameter
   λ is increased (5 → 20 → 80) to focus progressively on the true max.

3. **Targeted mass moves** – repeatedly identify the shift where the
   self‑convolution is maximal and transfer a fraction of mass from the
   index that contributes most to that shift to the index that contributes
   the least.  The move is kept only if the surrogate objective improves.

4. **Polishing** – a momentum sub‑gradient descent (Frank‑Wolfe style)
   and a short simulated‑annealing run finish the optimisation.

All phases respect the global time budget (default 1000 s).  The best
sequence found is stored in ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS‑B)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    _SCIPY_AVAILABLE = True
except Exception:          # pragma: no cover
    _SCIPY_AVAILABLE = False
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) – assumes Σseq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‐max (log‑sum‑exp) objective and its gradient w.r.t. log‑softmax parameters
# ----------------------------------------------------------------------
def _smooth_objective_and_grad_log(log_a: np.ndarray, lam: float):
    """
    Returns (smooth_value, grad_log) where ``log_a`` are the unconstrained
    log‑softmax parameters.  The smooth value approximates
    2·n·max(conv) and the gradient is w.r.t. ``log_a``.
    """
    # softmax -> probability vector a (Σa = 1, a_i ≥ 0)
    max_log = np.max(log_a)
    exp_shift = np.exp(log_a - max_log)
    a = exp_shift / np.sum(exp_shift)

    # convolution a*a
    conv = _conv_fft(a, a)
    max_c = np.max(conv)

    # softmax weights over the convolution entries
    # (max_c is subtracted for numerical stability)
    exp_term = np.exp(lam * (conv - max_c))
    Z = np.sum(exp_term)
    smooth = max_c + np.log(Z) / lam
    w = exp_term / Z                     # w_k ≈ softmax(λ·conv_k)

    # gradient w.r.t. a : g_a = 2 * (w * rev(a))_relevant_slice
    rev_a = a[::-1]
    conv_rev = _conv_fft(w, rev_a)      # full length = (2n‑1)+(n‑1)
    n = len(a)
    g_a = 2.0 * conv_rev[n - 1:n - 1 + n]

    # chain rule: dF/dlog = diag(a)·g_a – a·(a·g_a)
    dot = np.dot(a, g_a)
    g_log = a * (g_a - dot)

    return smooth, g_log


def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    lam: float,
                    maxiter: int,
                    deadline: float) -> np.ndarray:
    """
    Runs a short L‑BFGS‑B optimisation on the smooth approximation
    (parameter λ).  Returns the improved probability vector.
    """
    if not _SCIPY_AVAILABLE:
        return a0

    log_a0 = np.log(a0 + 1e-30)

    def fun(log_a):
        if time.time() > deadline:
            raise KeyboardInterrupt
        val, grad = _smooth_objective_and_grad_log(log_a, lam)
        return val, grad

    try:
        res = _opt.minimize(fun,
                            log_a0,
                            method='L-BFGS-B',
                            jac=True,
                            options={'maxiter': maxiter, 'disp': False})
        log_opt = res.x
        # convert back to probability vector
        shifted = log_opt - np.max(log_opt)
        exp_shift = np.exp(shifted)
        a_opt = exp_shift / np.sum(exp_shift)
        return a_opt
    except KeyboardInterrupt:                     # pragma: no cover
        print("[LBFGS] time limit hit – returning current iterate")
        shifted = log_a0 - np.max(log_a0)
        exp_shift = np.exp(shifted)
        return exp_shift / np.sum(exp_shift)
    except Exception as e:                         # pragma: no cover
        print("[LBFGS] optimisation failed:", e)
        return a0


# ----------------------------------------------------------------------
# Targeted move optimisation (mass transfer on the worst shift)
# ----------------------------------------------------------------------
def _targeted_move_opt(seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer scheme.
    At each iteration the shift with maximal self‑convolution is located.
    A fraction of mass is moved from the index contributing most to that
    shift to the index contributing the least, accepting the move only
    when the surrogate objective improves.
    """
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(seq, seq)
        max_idx = int(np.argmax(conv))

        # overlapping indices for this shift
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = max_idx - i_range
        contributions = seq[i_range] * seq[j_range]

        # source = biggest contribution, dest = smallest contribution
        src_local = int(np.argmax(contributions))
        dst_local = int(np.argmin(contributions))
        src_idx = i_range[src_local]
        dst_idx = i_range[dst_local]

        if src_idx == dst_idx:
            # pick a random different destination if everything is equal
            others = np.delete(i_range, src_local)
            if others.size == 0:
                continue
            dst_idx = int(np.random.choice(others))

        # transfer a modest fraction of the source mass
        eps = seq[src_idx] * 0.20          # 20 % of the source
        if eps <= 0.0:
            continue

        new_seq = seq.copy()
        new_seq[src_idx] -= eps
        new_seq[dst_idx] += eps

        new_val = _objective_surrogate(new_seq)

        if new_val < best_val - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_val = new_val

        if it % 5000 == 0:
            print(f"[Targeted] iter={it:,d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Momentum sub‑gradient descent (Frank‑Wolfe‑type polishing)
# ----------------------------------------------------------------------
def _subgradient_proj_improved(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with momentum on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    momentum = np.zeros_like(a)
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubGrad] it={it:,d} best={best_val:.6f}")

        # sub‑gradient of the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # momentum update + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * g
        lr = 0.20 / np.sqrt(it)
        a = a - lr * momentum
        a = _proj_simplex(a)

    return best_seq, best_val

def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.06,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:          # reshape a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # transfer a small chunk between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)  # move at most 5 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:,d} best={best_val:.6f}")

    return best_seq, best_val

    print(f"\n--- Initialisation (n={n}) ---")
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        a0 = np.asarray(prev, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a0 = a0 / a0.sum()
        print("Loaded previous sequence.")
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        print("No suitable previous sequence – starting from uniform.")

    best_seq = a0.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth‑max L‑BFGS‑B optimisation.
    # --------------------------------------------------------------
    if time.time() < deadline and _SCIPY_AVAILABLE:
        print("\n--- Smooth max optimisation (L‑BFGS‑B) ---")
        for lam in (5.0, 20.0, 80.0):
            if time.time() > deadline:
                break
            maxiter = 250      # modest number of L‑BFGS iterations per λ
            best_seq = _lbfgs_optimize(best_seq, n, lam, maxiter, deadline)
            val = _objective_surrogate(best_seq)
            print(f"[LBFGS] λ={lam:.0f} surrogate={val:.6f}")
            if val < best_val:
                best_val = val

    # --------------------------------------------------------------
    # Phase 2 – targeted mass‑transfer moves.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Targeted move optimisation ---")
        remaining = deadline - time.time()
        # Rough estimate: 2000 iterations per second for n≈4000
        max_iters = int(min(300_000, remaining * 2000))
        best_seq, best_val = _targeted_move_opt(best_seq, n, deadline, max_iters=max_iters)
        print(f"[Targeted] final surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – momentum sub‑gradient descent polishing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent polishing ---")
        remaining = deadline - time.time()
        max_iters_sub = int(min(2_000_000, remaining * 1e5))
        seq_proj, val_proj = _subgradient_proj_improved(best_seq, n, deadline,
                                                        max_iters=max_iters_sub)
        if val_proj < best_val:
            best_seq, best_val = seq_proj, val_proj
            print(f"[Sub‑grad] new best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing (optional final tweak).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(500_000, remaining * 1e5))
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              max_iters=max_iters_sa)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
