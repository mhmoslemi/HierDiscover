"""
Propose a non‑negative step function (probability vector) that improves the
evaluation bound.  The algorithm works in three stages:

1. **Broad random search** – generate many analytically‑parameterised families
   (Beta‑U‑shapes with tiny exponents, Dirichlet draws with low concentration,
   random piece‑wise constant blocks, geometric / symmetric‑geometric, etc.).
   Each candidate is ℓ₁‑normalised and evaluated with a fast FFT based surrogate
   `2·n·max(conv(a,a))`.  The best candidate so far is kept.

2. **Deterministic refinement** – from the current best vector we run two
   gradient‑based optimisers that respect the simplex:
      * exponential‑weights (mirror) descent using the sub‑gradient of the
        currently maximal convolution entry;
      * projected sub‑gradient descent that averages the gradients of the
        `top_k` largest convolution entries (a flatter‑profile update).

   Both stages run for as many iterations as the remaining time budget permits.

3. **Stochastic polishing** – a lightweight simulated‑annealing loop that
   repeatedly reshuffles a random subset of coordinates (Dirichlet move) and
   accepts downhill moves or uphill moves with a gradually decreasing temperature.

The final vector (still normalised to sum = 1) is returned as a list of floats.
The best sequence is also stored in the global variable ``height_sequence_1``
for reuse across runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_max(seq: np.ndarray, fft_len: int) -> float:
    """
    Full linear convolution via real FFT and return its maximum entry.
    `seq` must be a 1‑D real array.
    """
    fft = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * len(seq) - 1]
    return float(np.max(conv))


def _objective(seq: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation for a probability vector (sum == 1)."""
    return 2.0 * n * _conv_max(seq, fft_len)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _triangular_rev(n: int) -> np.ndarray:
    """U‑shaped (minimum at centre, maximum at the ends)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_mixed(
    n: int, alpha: float, beta: float, w: float, rng: np.random.Generator
) -> np.ndarray:
    """Mixture of a Beta shape and the uniform distribution."""
    b = _beta_shape(n, alpha, beta)
    u = _uniform(n)
    a = w * b + (1.0 - w) * u
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_piecewise(
    n: int, k: int, rng: np.random.Generator
) -> np.ndarray:
    """k random constant blocks."""
    breaks = np.sort(rng.choice(n - 1, size=k - 1, replace=False)) + 1
    edges = np.concatenate(([0], breaks, [n]))
    heights = rng.random(k)
    a = np.empty(n, dtype=np.float64)
    for i in range(k):
        a[edges[i] : edges[i + 1]] = heights[i]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers (top‑level)
# ----------------------------------------------------------------------


def _mirror_descent(
    start_vec: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exponential‑weights mirror descent on the L∞‑convolution objective.
    Returns the best vector encountered and its surrogate value.
    """
    a = start_vec.copy()
    best = a.copy()
    best_val = _objective(a, n, fft_len)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current convolution and its active index
        conv = _conv_max(a, fft_len)  # we only need max value for gradient
        # obtain the index of the maximal entry (full conv needed)
        full = np.fft.irfft(np.fft.rfft(a, n=fft_len) ** 2, n=fft_len)[
            : 2 * n - 1
        ]
        max_idx = int(np.argmax(full))

        # gradient of the active entry
        idx = np.arange(n)
        rev = max_idx - idx
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        cur_val = _objective(a, n, fft_len)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            # occasional progress report
            if it % 50_000 == 0:
                print(f"[Mirror] it={it:6d} best={best_val:.6f}")

    return best, best_val


def _multi_subgradient(
    start_vec: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    init_lr: float = 0.30,
    top_k: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent averaging the gradients of the `top_k`
    biggest convolution entries.
    """
    a = start_vec.copy()
    best = a.copy()
    best_val = _objective(a, n, fft_len)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        full = np.fft.irfft(np.fft.rfft(a, n=fft_len) ** 2, n=fft_len)[
            : 2 * n - 1
        ]

        if top_k >= full.size:
            idxs = np.arange(full.size)
        else:
            idxs = np.argpartition(full, -top_k)[-top_k:]

        # average gradient over the selected indices
        g = np.zeros_like(a)
        for k_idx in idxs:
            rev = k_idx - np.arange(n)
            mask = (rev >= 0) & (rev < n)
            g[mask] += 2.0 * a[rev[mask]]
        g /= float(len(idxs))

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective(a, n, fft_len)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if it % 20_000 == 0:
                print(f"[MultiSub] it={it:7d} best={best_val:.6f}")

    return best, best_val


def _simulated_annealing(
    start_vec: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simple Dirichlet‑move simulated annealing polishing step.
    """
    rng = np.random.default_rng()
    cur = start_vec.copy()
    cur_val = _objective(cur, n, fft_len)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)  # size of the reshuffled block
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc

        new_val = _objective(new, n, fft_len)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

# Candidate generation helper (top‑level)
# ----------------------------------------------------------------------


def _add_candidate(
    vec: np.ndarray,
    n: int,
    fft_len: int,
    best_info: dict,
) -> None:
    """Evaluate ``vec`` (ℓ₁‑normalised) and possibly update the best entry."""
    s = vec.sum()
    if s == 0.0:
        return
    v = vec / s
    val = _objective(v, n, fft_len)
    if val < best_info["val"]:
        best_info["val"] = val
        best_info["seq"] = v.copy()
        print(f"[Seed] new best surrogate = {val:.6f}")

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a list of floats and stores it in the global
    variable ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose a length – 4000 gives a good compromise of quality / speed.
    # ------------------------------------------------------------------
    n = 4000
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # Load previous solution (if any) and initialise the global best.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    best_info = {"seq": None, "val": float("inf")}
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.size != n:
                # trim or pad to the chosen length
                if prev.size > n:
                    prev = prev[:n]
                else:
                    prev = np.pad(prev, (0, n - prev.size))
            _add_candidate(prev, n, fft_len, best_info)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 1 – broad random search over many analytic families.
    # ------------------------------------------------------------------
    print("\n--- Stage 1: Random seed generation ---")
    _add_candidate(_uniform(n), n, fft_len, best_info)
    _add_candidate(_triangular_rev(n), n, fft_len, best_info)

    # systematic Beta sweeps with tiny exponents (U‑shapes)
    for a in (0.02, 0.05, 0.1, 0.2):
        for b in (0.02, 0.05, 0.1, 0.2):
            _add_candidate(_beta_shape(n, a, b), n, fft_len, best_info)
            _add_candidate(_beta_mixed(n, a, b, 0.7, rng), n, fft_len, best_info)

    # geometric families
    for r in np.concatenate(

        _add_candidate(_sym_geom(n, r), n, fft_len, best_info)

    # power law shapes
    for p in np.linspace(0.0, 5.0, 15):

    # many random Dirichlet draws with a wide concentration range
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            _add_candidate(_random_dirichlet(n, conc, rng), n, fft_len, best_info)

    # random piecewise‑constant blocks
    for _ in range(300):
        k = rng.integers(2, 12)
        _add_candidate(_random_piecewise(n, k, rng), n, fft_len, best_info)

    # ------------------------------------------------------------------
    # If we still have time, continue generating cheap random candidates.
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:  # keep ~30 s for refinement
        # random Beta with log‑uniform exponents
        a_exp = 10 ** rng.uniform(-3, 0)  # between 0.001 and 1
        b_exp = 10 ** rng.uniform(-3, 0)
        w = rng.random()
        _add_candidate(
            _beta_mixed(n, a_exp, b_exp, w, rng), n, fft_len, best_info
        )
        # random low‑conc Dirichlet
        conc = 10 ** rng.uniform(-4, 0)  # between 1e‑4 and 1
        _add_candidate(_random_dirichlet(n, conc, rng), n, fft_len, best_info)

    # ------------------------------------------------------------------
    # Stage 2 – deterministic refinement (mirror descent + multi‑subgradient)
    # ------------------------------------------------------------------
    cur_seq = best_info["seq"]
    cur_val = best_info["val"]
    print(f"\n[Init] best surrogate after seeding = {cur_val:.6f}")

        )

    return cur_seq.tolist()
