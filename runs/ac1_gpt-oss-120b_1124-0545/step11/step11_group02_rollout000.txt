def _adam_opt(start_seq, n, deadline, alpha_initial=0.2, alpha_max=8.0, lr=0.2):
    a = start_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_seq = a.copy()
    best_val = _objective(a)
    it=0
    alpha = alpha_initial
    while time.time() < deadline:
        it+=1
        # compute convolution
        conv = _conv_fft(a, a)
        # compute softmax weights
        maxc = conv.max()
        # update alpha gradually
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha_initial * (1 + it * 1e-4)) # ramp up slowly
        # compute weights w = exp(alpha*(conv - maxc))
        # subtract maxc to avoid overflow
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient = 2 * (conv(w, a_rev))[n-1:n-1+n]
        a_rev = a[::-1]
        grad = 2.0 * _conv_fft(w, a_rev)[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = _uniform(n)  # fallback
        else:
            a = a / s

        # compute objective every maybe 1000 iter
        if it % 500 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val
```

"""
Propose a non‑negative step‑function (probability vector) that improves the
evaluation bound.

Algorithm
---------
1. Build a modest pool of analytically‑generated seed vectors (uniform,
   triangular, geometric, Gaussian, etc.) and, if available, the previous
   solution resampled to the current length.
2. Pick the best seed (according to the surrogate 2·n·max (a∗a)).
3. Run three optimisation phases while respecting the overall time budget.
   * **Adam with a smooth‑max surrogate** – a soft‑max approximation of the
     max‑convolution is optimised using the Adam optimiser on the simplex.
   * **Exponential‑mirror descent** – a multiplicative update that follows the
     exact sub‑gradient of the current maximal entry.
   * **Simulated‑annealing polishing** – random Dirichlet reshuffles of a
     handful of coordinates with a geometric temperature decay.
4. Post‑process: try convex mixtures of the best sequence with a few simple
   families (uniform, triangular, Gaussian, sine‑window).  The mixture can
   lower the peak by adding smooth background mass.
5. Return the best vector found (list of floats) and store it globally as
   ``height_sequence_1`` for reuse in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation for ℓ₁‑normalised sequences (sum == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _resample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    interp = np.interp(new_x, old_x, seq)
    total = interp.sum()
    if total == 0.0:
        return _uniform(new_n)
    return interp / total


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))

    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12),
                             np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlets with a range of concentration parameters
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # add a few block‑type seeds (multiple equal blocks separated by zeros)
    block_counts = [2, 3, 4]
    block_frac = 0.05
    for k in block_counts:
        block_len = max(1, int(block_frac * n / k))
        a = np.zeros(n, dtype=np.float64)
        spacing = n // k
        for i in range(k):
            start = i * spacing
            a[start:start + block_len] = 1.0
        a = a / a.sum() if a.sum() > 0 else _uniform(n)
        seeds.append(a)

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous.copy())
    elif previous is not None:
        # resample to current length
        seeds.append(_resample(previous, n))

    return seeds

def _exp_mirror_descent(start_vec: np.ndarray, n: int,
                        deadline: float, max_iters: int = 400_000,
                        eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponential‑weights (mirror) descent on the L∞‑convolution objective.
    Uses the sub‑gradient of the currently maximal convolution entry.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Mirror] it={it:6d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        idx = np.arange(n)
        rev = max_idx - idx
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        eta = eta0 / np.sqrt(it)          # decreasing learning rate
        a = a * np.exp(-eta * g)          # entropic step
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that redistributes mass among a random subset
    of coordinates (Dirichlet reshuffle).  Works as a polishing step.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _adam_opt(start_vec: np.ndarray, n: int,
              deadline: float,
              lr: float = 0.2,
              alpha_initial: float = 0.2,
              alpha_max: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth approximation of max(convolution).
    The smooth max is   (1/α)·log Σ exp(α·conv).
    We increase α gradually so that the optimisation starts from a smooth
    landscape and finishes focusing on the true maximum.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    it = 0
    alpha = alpha_initial

    while time.time() < deadline:
        it += 1
        # ---- compute convolution ----
        conv = _conv_fft(a, a)                     # length 2n-1
        maxc = conv.max()
        # ---- smooth max via soft‑max ----
        # update α slowly (geometric schedule)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha_initial * (1.0 + it / 5000.0))

        # subtract maxc for numerical stability
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / exp_vals.sum()              # soft‑max weights (sum to 1)

        # ---- gradient of smooth max ----
        a_rev = a[::-1]
        grad = 2.0 * _conv_fft(w, a_rev)[n - 1:n - 1 + n]   # length n

        # ---- Adam update ----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---- projection onto simplex ----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # ---- bookkeeping ----
        if it % 500 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

    return best_seq, best_val


def _mix_with_partner(a: np.ndarray, b: np.ndarray,
                      n: int,
                      alphas: np.ndarray = None) -> tuple[np.ndarray, float]:
    """
    Try convex mixtures (1-α)·a + α·b for a set of α values.
    Returns the mixture achieving the smallest surrogate objective.
    """
    if alphas is None:
        alphas = np.linspace(0.0, 1.0, 21)   # 0,0.05,…,1

    conv_a = _conv_fft(a, a)
    conv_b = _conv_fft(b, b)
    conv_ab = _conv_fft(a, b)

    best_seq = a
    best_val = _objective(a)

    for alpha in alphas:
        mixed_conv = ((1 - alpha) ** 2) * conv_a + \
                     (2 * alpha * (1 - alpha)) * conv_ab + \
                     (alpha ** 2) * conv_b
        cur_val = 2.0 * n * mixed_conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = ((1 - alpha) * a + alpha * b)
            # re‑normalise to stay on the simplex (tiny numerical drift possible)
            total = best_seq.sum()
            if total > 0:
                best_seq = best_seq / total
            else:
                best_seq = _uniform(n)

    return best_seq, best_val

                                                    eta0=0.6)
        if val_mirror < best_val:
            best_seq, best_val = seq_mirror, val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                             max_iters=800_000,
                                             init_temp_factor=0.04)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Mixing with simple families (optional polishing)
    # ------------------------------------------------------------------
    print("\n--- Mixing with simple families ---")
    mix_partners = [_uniform(n), _triangular(n), _gaussian(n, 0.20),
                    _sine_window(n, 1.0), _tukey_window(n, 0.4)]
    for partner in mix_partners:
        seq_mix, val_mix = _mix_with_partner(best_seq, partner, n)
        if val_mix < best_val:
            best_seq, best_val = seq_mix, val_mix
            print(f"[Mix] improved to {val_mix:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        official_val = float('inf')
        print("Warning: evaluate_sequence raised", exc)

    print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} (length {n})")

    # store for possible later reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
