    def anneal(seq, time_deadline, max_iters, T0_factor maybe 0.05):
        n = len(seq)
        cur_seq = seq.copy()
        cur_val = objective(cur_seq)
        best_seq = cur_seq.copy()
        best_val = cur_val
        T = max(T0_factor * cur_val, 1e-12)
        rng = np.random.default_rng()
        for it in range(1, max_iters+1):
            if time.time() > time_deadline:
                break
            # schedule temperature decreasing
            T *= 0.9999 # adjust factor

            # decide move type
            if rng.random() < 0.7:
                # swap two indices
                i, j = rng.choice(n, size=2, replace=False)
                # swap
                new_seq = cur_seq.copy()
                new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
                # compute new objective
                new_val = objective(new_seq)
            else:
                # mass reallocation among k indices (2-10)
                k = rng.integers(2, 11)
                idx = rng.choice(n, size=k, replace=False)
                total = cur_seq[idx].sum()
                if total == 0:
                    continue
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
                new_seq = cur_seq.copy()
                new_seq[idx] = new_alloc
                new_val = objective(new_seq)

            # acceptance
            delta = new_val - cur_val
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_seq = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = new_seq.copy()
                    if it % 5000 == 0:
                        print(f"[SA] iter {it} best {best_val:.6f}")

        return best_seq, best_val

Search for a low‑autocorrelation non‑negative step function.
The algorithm proceeds in three stages:

1. Generate many random and analytic ℓ¹‑normalised candidates of moderate length
   (n≈2000) and keep the top few according to the exact surrogate objective
   2·n·max(conv).

2. For each top seed run a long simulated‑annealing that works on the ordering
   of the values (swap moves) **and** on local mass‑reallocations.  The temperature
   schedule is aggressive and the number of iterations is scaled to the remaining
   time budget, allowing up to a few million cheap FFT evaluations.

3. Upscale the best sequence to a larger length (n≈4000) by linear interpolation
   and polish it again with a short annealing run.

Finally the best sequence is normalised and returned as a plain list of floats.
"""

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact surrogate objective 2·n·max(conv(seq,seq)) for an ℓ¹‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a small‑n sequence to a larger n."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    else:
        ys = np.full(n_big, 1.0 / n_big, dtype=np.float64)
    return ys.astype(np.float64)

def _alternating_permutation(values: np.ndarray) -> np.ndarray:

    n = len(values)
    out = np.empty(n, dtype=values.dtype)

    right = n - 1

            left += 1

def _generate_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list with a diverse collection of ℓ¹‑normalised vectors."""
    candidates = []

    # deterministic analytic families (same as earlier code)
    # Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Triangular
    i = np.arange(n, dtype=np.float64)

    tri = 1.0 - np.abs(i - centre) / centre
    tri = np.maximum(tri, 0.0)
    candidates.append(tri / tri.sum())

    # Power law

def _simulated_annealing(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int,
    init_temp_factor: float = 0.05,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the exact objective.
    Moves consist of either swapping two indices or redistributing the mass
    of a random subset (size 2‑10) via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)
    decay = 0.9997  # slower cooling for many iterations

    n = len(cur)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * decay, min_temp)

        # decide move type
        if rng.random() < 0.6:
            # swap two positions
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            # redistribute among a small random subset
            k = rng.integers(2, 11)  # 2‑10 indices
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[SA] iter {it:7d} best {best_val:.6f}")

    return best_seq, best_val

            cand.append(beta / beta.sum())

    # Random Dirichlet draws (moderate concentration)
    for conc in (0.2, 0.5, 1.0, 2.0):
        cand.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # Random exponential draws
    for _ in range(3):
        exp = rng.exponential(scale=1.0, size=n).astype(np.float64)
        cand.append(exp / exp.sum())

    # Geometric series + alternating permutation
    for r in (0.3, 0.5, 0.7):
        geom = r ** np.arange(n, dtype=np.float64)
        geom /= geom.sum()
        cand.append(geom.copy())
        # sorted descending ensures the permutation works as intended
        cand.append(_alternating_permutation(np.sort(geom)[::-1]))

    # Keep the previously discovered construction, if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    cand.append(prev_arr / s)
        except Exception:
            pass

    # Small jittered copies to broaden the pool
    jittered: list[np.ndarray] = []
    for vec in cand:
        noise = rng.normal(scale=1e-4, size=n)
        new = vec + noise
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s > 0:
            jittered.append(new / s)
    cand.extend(jittered)

    return cand


def _simulated_annealing(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int,
    init_temp_factor: float = 0.05,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the exact objective.

    Two move types are used:
      * swap two indices (pure re‑ordering);
      * redistribute the mass of a random small subset (size 2‑10)
        via a Dirichlet draw.

    The temperature is proportional to the current objective and decays
    geometrically each iteration.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)
    decay = 0.9997  # slow exponential cooling

    n = len(cur)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, min_temp)

        # decide which move to perform
        if rng.random() < 0.6:
            # --- swap two positions -------------------------------------------------
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            # --- redistribute a small random subset ---------------------------------
            k = rng.integers(2, 11)  # subset size 2‑10
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[SA] iter {it:7d} best {best_val:.6f}")

    return best_seq, best_val


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine obeys a global time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    # ----------------------------------------------------------------------
    # Stage 0 – generate a broad set of seeds (moderate size)
    # ----------------------------------------------------------------------
    n0 = 2000
    print(f"\n--- Stage 0: generating candidates of size {n0} ---")
    candidates = _generate_candidates(n0, rng)
    print(f"Generated {len(candidates)} raw candidates.")

    # evaluate each candidate exactly (surrogate objective)
    evaluated = [(vec, _objective(vec)) for vec in candidates]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    top_candidates = [vec for vec, _ in evaluated[:top_k]]
    best_seq = evaluated[0][0].copy()
    best_val = evaluated[0][1]
    print(f"Initial best surrogate = {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Stage 1 – intensive simulated‑annealing on the top seeds
    # ----------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining < 5.0:
        print("Insufficient time left for annealing.")
    else:
        secs_per_seed = remaining / (len(top_candidates) + 1.0)
        print(f"\n--- Stage 1: annealing (~{secs_per_seed:.1f}s per seed) ---")
        for idx, seed_vec in enumerate(top_candidates):
            if time.time() > deadline:
                break
            per_deadline = time.time() + secs_per_seed
            # heuristic: aim for ~1800 iterations per second
            max_iters = int(secs_per_seed * 1800)
            max_iters = max(20_000, min(max_iters, 3_000_000))
            print(f"\n[SA seed #{idx+1}/{len(top_candidates)}] max_iters≈{max_iters}")
            seq_sa, val_sa = _simulated_annealing(
                seed_vec, per_deadline, max_iters,
                init_temp_factor=0.05,
            )
            if val_sa < best_val:
                best_val = val_sa
                best_seq = seq_sa.copy()
                print(f"[overall] new global best = {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Stage 2 – upscale to a larger problem size and polish
    # ----------------------------------------------------------------------
    n_big = 4000
    print(f"\n--- Stage 2: upscaling to length {n_big} ---")
    seq_big = _upscale(best_seq, n_big)
    # re‑normalise (tiny rounding errors may appear)
    seq_big = np.maximum(seq_big, 0.0)
    s = seq_big.sum()
    if s > 0:
        seq_big /= s
    else:
        seq_big = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    val_big = _objective(seq_big)
    print(f"Surrogate after upscale = {val_big:.6f}")
    if val_big < best_val:
        best_val = val_big
        best_seq = seq_big.copy()

    # final polishing annealing on the up‑scaled sequence
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_iters = int(remaining * 1500)   # roughly 1500 cheap iterations per second
        max_iters = max(200_000, min(max_iters, 2_000_000))
        print(f"\n--- Stage 3: final annealing on length {n_big} (max_iters={max_iters}) ---")
        seq_final, val_final = _simulated_annealing(
            best_seq, deadline, max_iters,
            init_temp_factor=0.03,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final.copy()
            print(f"[final] new global best = {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Verify with the official evaluator
    # ----------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} (length {len(best_seq)})")

    # store for possible later reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
