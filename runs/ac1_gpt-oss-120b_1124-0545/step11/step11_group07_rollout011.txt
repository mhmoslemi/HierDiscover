    # f: sum=1
    M = np.max(_conv_fft(f, f))
    # Build constraint matrix A using f
    A = _build_constraint_matrix(f)   # shape (2n-1, n)
    # b_ub = np.full(2*n-1, M)
    # Solve LP
    c = -np.ones(n)  # maximize sum(b)
    # bounds
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:

    g = b / sum_b   # normalize to sum=1
    # convex combination

    f_new = _proj_simplex(f_new)
    M_new = np.max(_conv_fft(f_new, f_new))
    # return
    return f_new, M_new

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = conv[max_idx]  # max conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SubGrad] it={it} best={2*n*best_val:.6f}")
        # compute subgradient of conv entry at max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # compute step
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)
        a = a - step * g
        a = _proj_simplex(a)

"""
Propose a non‑negative step function with a very low value of
evaluate_sequence().  The algorithm works on the ℓ₁‑normalised simplex
(sum = 1) – for such a vector the official score equals 2·n·max(conv)
where conv is the self‑convolution.  The pipeline is:

1. Load the previous best construction (if any) and normalise it.
   Otherwise initialise from a random Dirichlet draw.
2. Refine the vector with a long Adam run on a smooth‑max surrogate
   of the max‑convolution.
3. Apply a few iterations of the classical alternating linear‑program
   update (the “LP polishing” described in the statement): maximise the
   total mass of a new vector b under the same max‑convolution bound,
   then convex‑combine the current vector with the normalised b.
4. Perform projected sub‑gradient descent on the exact max‑convolution
   objective for fine‑tuning.
5. Finish with a quick simulated‑annealing polishing.
6. Scale the final simplex vector to the required ℓ₁‑norm (√(2n)) and
   return it as a plain Python list.

All steps respect the supplied time budget (default 1000 s) and print
intermediate progress.  The global variable ``height_sequence_1`` stores
the best sequence for reuse in subsequent runs.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact max‑convolution for a sequence that sums to 1."""
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


def _build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that (A @ b) = conv(f, b)
    for any non‑negative vector b.
    """
    n = f.shape[0]
    # each column j contributes f_i to row i+j
    row_offsets = np.repeat(np.arange(n, dtype=np.int64), n)
    cols = np.tile(np.arange(n, dtype=np.int64), n)
    rows = row_offsets + cols
    data = np.repeat(f, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

# Simple seed families (used only if no previous solution is present)
# ----------------------------------------------------------------------


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet random vector with the given concentration."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _adam_optimize(
    start_vec: np.ndarray,

    deadline: float,
    max_iters: int = 2_000_000,
    beta_max: float = 500.0,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a smooth log‑sum‑exp surrogate of the max‑convolution.
    Returns the best sequence (sum = 1) and the corresponding objective
    (max(conv)).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        a = np.exp(v - np.max(v))
        a = a / a.sum()  # softmax → simplex point

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={2 * n * best_val:.6f}")

        # increase the temperature (smooth‑max sharpness)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # --------------- smooth max of the convolution -----------------
        conv = _conv_fft(a, a)               # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum                         # softmax weights over conv entries

        # gradient w.r.t. a (see derivation in the original code)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Works on vectors with sum = 1; returns the best vector encountered and
    its max(conv) value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SubGrad] it={it:7d} best={2 * n * best_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (using current best)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Tiny‑move simulated‑annealing on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)  # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={2 * n * best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.  Returns the best
    sequence found (scaled to the required ℓ₁‑norm √(2 n)).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Choose length n and start from a previously stored solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            n = prev_arr.shape[0]
            best_seq = prev_arr / prev_arr.sum()
            best_val = _objective(best_seq)
            print(f"Loaded previous sequence (n={n}) – current max(conv)={best_val:.6f}")
        except Exception:
            previous = None

    if previous is None:
        n = 4000
        print(f"No stored sequence – initializing a random Dirichlet vector (n={n})")
        best_seq = _dirichlet_seed(n, conc=0.2, rng=rng)
        best_seq = best_seq / best_seq.sum()
        best_val = _objective(best_seq)
        print(f"Initial max(conv)={best_val:.6f}")

    sqrt2n = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 1 – Adam refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 1: Adam refinement ---")
        seq_adam, val_adam = _adam_optimize(
            start_vec=best_seq,
            n=n,
            deadline=deadline,
            max_iters=2_000_000,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam
            best_val = val_adam
            print(f"Adam improved max(conv) to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP polishing (the alternating LP update from the statement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: LP polishing ---")
        t = 0.2                # initial convex‑combination weight
        lp_iter = 0
        while time.time() < deadline and lp_iter < 30:
            M = _objective(best_seq)          # current bound on max(conv)
            # Build the constraint matrix A (size (2n‑1) × n)
            A = _build_constraint_matrix(best_seq)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            c = -np.ones(n, dtype=np.float64)      # maximise sum(b)
            bounds = [(0, None)] * n

            remaining = max(0.0, deadline - time.time())
            res = scipy.optimize.linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": remaining},
            )

            if not res.success:
                print("LP solve failed – exiting LP phase.")
                break

            b = res.x
            sum_b = b.sum()
            if sum_b <= 0.0:
                break

            g = b / sum_b                     # normalise to sum = 1
            # convex combination and re‑projection
            new_seq = (1 - t) * best_seq + t * g
            new_seq = _proj_simplex(new_seq)

            new_val = _objective(new_seq)
            if new_val < best_val - 1e-12:
                best_seq = new_seq
                best_val = new_val
                print(f"[LP {lp_iter+1}] improved max(conv) to {best_val:.6f}")
                t = max(0.05, t * 0.9)        # slowly shrink the step size
            else:
                t = max(0.01, t * 0.8)        # if no progress, reduce t
            lp_iter += 1
            if t < 0.01:
                break

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: Sub‑gradient descent ---")
        seq_sub, val_sub = _subgrad_optimize(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=500_000,
        )
        if val_sub < best_val:
            best_seq = seq_sub
            best_val = val_sub
            print(f"Sub‑gradient improved max(conv) to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: Simulated annealing ---")
        seq_sa, val_sa = _simulated_annealing(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=300_000,
        )
        if val_sa < best_val:
            best_seq = seq_sa
            best_val = val_sa
            print(f"Simulated annealing improved max(conv) to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Scale to the required ℓ₁‑norm (√(2 n)) and final official check
    # ------------------------------------------------------------------
    final_seq = best_seq * sqrt2n

    try:
        final_score = evaluate_sequence(final_seq.tolist())
        print(f"\n=== Final official score: {final_score:.6f} ===")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_score = None

    # store for possible reuse
    globals()["height_sequence_1"] = final_seq.tolist()
    return final_seq.tolist()
