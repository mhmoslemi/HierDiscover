def _smooth_descent(a, n, deadline, max_iters=5000, init_alpha=10.0, final_alpha=1000.0):
    it=0
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    while time.time() < deadline and it < max_iters:
        # schedule alpha linearly from init to final based on progress
        alpha = init_alpha + (final_alpha - init_alpha) * (it / max_iters)
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # compute softmax weights
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum
        # gradient = 2 * conv(weights, a)
        grad = 2.0 * _conv_fft(w, a)
        # step size
        lr = 0.5 / np.sqrt(it + 1)
        a = a - lr * grad
        a = _proj_simplex(a)
        # update best
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        it+=1
    return best_seq, best_val
```

        lr = 0.5 / np.sqrt(it)

def _smooth_descent(start_seq, n, deadline, max_iters=20000, init_alpha=100.0, final_alpha=20000.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # Softmax weights for approximating max
        # linear scaling alpha
        alpha = init_alpha + (final_alpha - init_alpha) * (it / max_iters)
        # compute weights
        w = np.exp(alpha * (conv - max_val))  # subtract max to avoid overflow
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w / w_sum
        # compute gradient: 2 * convolution of w and a_rev, slice
        a_rev = a[::-1]
        tmp = _conv_fft(w, a_rev)
        grad = 2.0 * tmp[n-1:n-1+n]  # slice length n
        # step size schedule
        lr = 0.5 / np.sqrt(it + 1)   # could tune
        a = a - lr * grad
        a = _proj_simplex(a)
        # evaluate surrogate
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    return best_seq, best_val
```

Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Rich seed collection** – many analytic windows, Dirichlet draws,
   low‑frequency random curves, mixtures and the previous best solution.
2. **Score seeds** with the cheap surrogate 2·n·max(conv) (the exact
   objective up to the normalisation ‑ sum = 1).
3. **Smooth max‑descent** – minimise a smooth approximation of the
   ``max`` using a soft‑max (log‑sum‑exp) over the convolution values.
   The gradient is obtained by a second FFT; after each step the vector
   is projected onto the probability simplex.
   The smoothing parameter α is increased from a modest value to a very
   large one so that the algorithm gradually focusses on the true
   maximum.
4. **Simple sub‑gradient polishing** – the classic single‑active‑shift
   sub‑gradient descent (as used before) is applied to the best
   candidate from step 3 for a large number of iterations.
5. **Optional simulated‑annealing tweak** – aggressive stochastic
   moves with a slowly cooling schedule finish the search.

All stages respect the global deadline (default 1000 s).  The best
sequence is stored in the global name ``height_sequence_1`` and returned
as a plain list of floats.

import time

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import kaiser as _scipy_kaiser
        a = _scipy_kaiser(n, beta)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    except Exception:
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        try:
            from scipy.special import i0 as _scipy_i0
            a = _scipy_i0(beta * np.sqrt(1 - y * y))
        except Exception:
            t = beta * np.sqrt(1 - y * y)
            a = (1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 +
                 (t ** 6) / 2304.0 + (t ** 8) / 147456.0)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.15, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # beta family – systematic grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Dirichlet draws with a range of concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Heavy‑tailed exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise everything safely
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Smooth‑max descent (new)
# ----------------------------------------------------------------------
def _smooth_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    init_alpha: float = 200.0,
    final_alpha: float = 20_000.0,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv) using a soft‑max.
    The schedule for α goes from ``init_alpha`` to ``final_alpha``.
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        max_val = np.max(conv)
        # linearly increase smoothing parameter
        alpha = init_alpha + (final_alpha - init_alpha) * (it / max_iters)

        # soft‑max weights (stable)
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # gradient = 2·(w * a)  (cross‑correlation)
        a_rev = a[::-1]
        tmp = _conv_fft(w, a_rev)               # length = 3n‑2
        grad = 2.0 * tmp[n - 1:n - 1 + n]        # slice of length n

        # step size – classic diminishing schedule
        lr = 0.5 / np.sqrt(it)
        a = a - lr * grad
        a = _proj_simplex(a)

        # recompute convolution for the next iteration
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple sub‑gradient descent (exact max, single active shift)
# ----------------------------------------------------------------------
def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing learning rate
        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing (optional final polish)
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:          # reshape a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # transfer a tiny chunk
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # small safety margin

    # ------------------------------------------------------------------
    # Length of the step‑function – 4000 gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich pool of seeds and keep the best few.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    scored = [(seq, _objective_surrogate(seq)) for seq in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = min(8, len(scored))
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate after seeding = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – smooth‑max descent on a few promising seeds.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Smooth‑max descent stage ---")
        # allocate up to three seeds; the time is split evenly
        remaining_total = deadline - time.time()
        seeds_to_use = best_seeds[:3]
        for idx, seed_seq in enumerate(seeds_to_use):
            # each seed gets an equal share of the remaining time
            sub_deadline = time.time() + remaining_total / (len(seeds_to_use) - idx)
            seq, val = _smooth_descent(
                seed_seq,
                n,
                sub_deadline,
                max_iters=150_000,
                init_alpha=200.0,
                final_alpha=20_000.0,
            )
            if val < best_val:
                best_seq, best_val = seq, val
                print(f"[Smooth] new best after seed #{idx+1}: {val:.6f}")
            remaining_total = deadline - time.time()
            if remaining_total < 5.0:
                break
        print(f"[Smooth] best surrogate after this stage = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exact sub‑gradient descent polishing (single active shift).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient polishing stage ---")
        remaining = deadline - time.time()
        max_iters_sub = int(min(2_000_000, remaining * 1e5))
        seq_proj, val_proj = _subgradient_proj(
            best_seq, n, deadline, max_iters=max_iters_sub
        )
        if val_proj < best_val:
            best_seq, best_val = seq_proj, val_proj
            print(f"[Sub] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional simulated‑annealing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing refinement ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(800_000, remaining * 1e5))
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters_sa
        )
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
