def solve_lp(f, M):
    n = len(f)

"""
Propose a step‑function sequence that (approximately) minimizes
`evaluate_sequence`.  The search consists of four phases:

1. **Seed generation** – a large pool of random Dirichlet draws (various
   concentration levels) together with a few deterministic families is
   evaluated.  The best few (default 5) are kept as starting points.

2. **Linear‑program refinement** – for a current sequence `f` we solve the
   LP

        maximise    Σ g_j
        subject to  conv(f,g) ≤ M   (component‑wise)
                    g ≥ 0

   where `M = max(conv(f,f))`.  The solution `g` is renormalised to the
   simplex and mixed with `f`.  Repeating this step (few iterations) drives
   the maximal convolution down.

3. **Projected sub‑gradient descent** – a pure sub‑gradient method on the
   exact max‑convolution objective (scale‑invariant) is run for many
   iterations, always projecting back onto the probability simplex.

4. **Simulated‑annealing polishing** – tiny Dirichlet re‑allocations are
   performed to escape shallow minima.

The routine respects the supplied time budget (default 1000 s), stores the
best sequence in the global variable ``height_sequence_1`` and returns it
as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional import – linear programming with CVXPY
# ----------------------------------------------------------------------
try:
    import cvxpy as cp  # type: ignore
    _CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVXPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns ``max(conv(seq, seq))``.
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Linear‑program solve (maximise ℓ₁‑mass under current max‑convolution bound)
# ----------------------------------------------------------------------


def _lp_max_mass(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ g_j  s.t. conv(f,g) ≤ M, g ≥ 0.
    Returns the optimal vector g (might not sum to 1).
    """
    if not _CVXPY_AVAILABLE:
        # Fallback: return the current vector (no improvement)
        return f.copy()

    n = f.shape[0]

    # CVXPY objects
    f_param = cp.Parameter(n, value=f, nonneg=True)
    g = cp.Variable(n, nonneg=True)

    # Linear convolution (full length 2n‑1) – this is a linear expression in g
    conv_expr = cp.conv(f_param, g)          # type: ignore[arg-type]

    # Constraints: each entry ≤ M
    constraints = [conv_expr <= M]

    # Objective: maximise ℓ₁‑norm of g
    objective = cp.Maximize(cp.sum(g))

    prob = cp.Problem(objective, constraints)

    # Solve with ECOS (fast interior‑point for LPs)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2500, verbose=False)
    except Exception:  # pragma: no cover
        # In case the solver fails, fall back to the input vector
        return f.copy()

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return f.copy()

    g_opt = g.value
    if g_opt is None:
        return f.copy()

    # Numerical safety – enforce non‑negativity
    g_opt = np.maximum(g_opt, 0.0)
    return g_opt


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_optimize(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on ``2*n*max(conv(seq,seq))``.
    Returns the best sequence (sum=1) and the best scaled objective value
    ``2*n*max(conv)`` encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry (max_idx)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (uses the gap to the best value so far)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / math.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (tiny Dirichlet reallocations)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 400_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Small‑move simulated‑annealing on the exact objective.
    Returns the best sequence (sum=1) and its scaled objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = 2.0 * n * _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a small safety margin

    # Length of the step‑function – 4000 is a good trade‑off (fits the LP
    # comfortably and still benefits from a large number of degrees of freedom).
    n = 4000
    sqrt2n = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 0 – seed generation (≈10 % of the budget)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = [_uniform(n), _triangular(n)]
    # Random Dirichlet seeds with a spread of concentration parameters
    conc_levels = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    per_conc = 120   # total random draws ≈ 1 300 (enough for a quick scan)
    for conc in conc_levels:
        for _ in range(per_conc):
            if time.time() > deadline:
                break
            seeds.append(_dirichlet_seed(n, conc, rng))

    # Keep the top‑k seeds according to the exact objective
    top_k = 5
    top_vals = []   # list of (M, seq)
    for seq in seeds:
        M = _objective(seq)          # seq already normalised (sum=1)
        if len(top_vals) < top_k:
            top_vals.append((M, seq))
            top_vals.sort(key=lambda x: x[0])
        elif M < top_vals[-1][0]:
            top_vals[-1] = (M, seq)
            top_vals.sort(key=lambda x: x[0])

    print(f"Collected {len(top_vals)} promising seeds (kept top‑{top_k}).")
    # Extract the sequences (sorted by increasing max‑convolution)
    current_f = top_vals[0][1].copy()   # start from the best seed
    best_M = top_vals[0][0]

    # ------------------------------------------------------------------
    # Phase 1 – LP‑based refinement (≈30 % of the budget)
    # ------------------------------------------------------------------
    if _CVXPY_AVAILABLE and time.time() < deadline:
        print("\n--- LP refinement (phase 1) ---")
        lp_iters = 7
        t_mix = 0.5   # initial mixing coefficient
        for it_lp in range(lp_iters):
            if time.time() > deadline:
                break
            M = _objective(current_f)          # current max‑conv
            g_raw = _lp_max_mass(current_f, M)
            sum_g = g_raw.sum()
            if sum_g <= 1e-12:                 # degenerate solution – fallback
                g = _uniform(n)
            else:
                g = g_raw / sum_g               # renormalise to the simplex

            # Convex combination (keeps sum=1)
            current_f = (1.0 - t_mix) * current_f + t_mix * g
            # Project back onto the simplex to cure possible numerical drift
            current_f = _proj_simplex(current_f)

            new_M = _objective(current_f)
            if new_M < best_M:
                best_M = new_M
                print(f"[LP] iteration {it_lp+1:2d} improved max‑conv to {best_M:.6f}")
            # Slowly decay the mixing coefficient
            t_mix = max(0.1, t_mix * 0.85)

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (≈40 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (phase 2) ---")
        max_sg_iters = int(5e5)   # conservative; will stop early if out‑of‑time
        current_f, best_val = _subgrad_optimize(
            start_seq=current_f,
            n=n,
            deadline=deadline,
            max_iters=max_sg_iters,
        )
        best_M = best_val / (2.0 * n)  # back‑to the plain max‑conv value
        print(f"After sub‑grad best max‑conv ≈ {best_M:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing (remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing (phase 3) ---")
        max_sa_iters = int(3e5)
        current_f, best_val = _simulated_annealing(
            start_seq=current_f,
            n=n,
            deadline=deadline,
            max_iters=max_sa_iters,
        )
        best_M = best_val / (2.0 * n)
        print(f"After SA best max‑conv ≈ {best_M:.6f}")

    # ------------------------------------------------------------------
    # Scale from the simplex to the required ℓ₁‑norm (√(2n))
    # ------------------------------------------------------------------
    final_seq = current_f * sqrt2n

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq.tolist())
        print(f"\n=== Finished ===\nBest objective (official) ≈ {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for reuse and return a plain Python list.
    globals()["height_sequence_1"] = final_seq.tolist()
    return final_seq.tolist()
