#!/usr/bin/env python
"""
Propose a step‑function (ℓ1‑normalised probability vector) that improves the
evaluation bound.  The algorithm consists of three stages:

1. **Initial seed** – reuse the previous best solution (global variable
   ``height_sequence_1``) or fall back to a simple beta‑shaped distribution.
   The vector is normalised to sum = 1.

2. **Active‑set linear programming** – for the current vector `a` we solve a
   small linear program that seeks a new vector `b` (still sum = 1) with a
   *strictly* smaller maximal autocorrelation
   `max_k (a * b)_k`.  Only the most critical convolution constraints
   (the top‑K entries of the current autocorrelation) are kept, which makes
   the LP cheap (≈ a few hundred constraints).  If the LP yields a better
   bound we replace `a ← b` and repeat.

3. **Polishing** – a short simulated‑annealing sweep (Dirichlet reshuffles)
   moves a random subset of coordinates; downhill moves are always kept,
   uphill moves are accepted with a temperature schedule.

All steps respect the 1000 s budget and the final sequence is stored in the
global variable ``height_sequence_1`` for reuse.  The returned list is ready
to be fed to ``evaluate_sequence``.
"""
import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate objective for ℓ1‑normalised sequences."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used only for a fallback seed)
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta‑shaped density on {0,…,n‑1}."""
    if alpha <= 0.0 or beta <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Build the sparse matrix encoding (a * b)_k ≤ M constraints
# ----------------------------------------------------------------------
def _build_constraint_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Return a sparse matrix A of shape (2*len(a)-1, len(a)) such that
    (A @ b)[k] = (a * b)[k] = Σ_i a_i b_{k-i}   (linear in b).
    """
    n = a.shape[0]
    L = 2 * n - 1
    # each diagonal of A (offset = j) consists of the whole vector a
    # there are n diagonals, offsets 0 … n‑1
    diags = [a] * n
    offsets = np.arange(n)
    A = sp.diags(diags, offsets, shape=(L, n), format='csr')
    return A


def _build_active_constraint_matrix(a: np.ndarray, active_idx: np.ndarray) -> sp.csr_matrix:
    """
    Build a reduced constraint matrix that contains only the rows indexed by
    ``active_idx`` (a subset of 0 … 2*n‑2).  The matrix is in CSR format.
    """
    n = a.shape[0]
    rows = active_idx.shape[0]
    A_sub = sp.lil_matrix((rows, n), dtype=np.float64)

    for r, k in enumerate(active_idx):
        # for a fixed shift k, the coefficients are a_{k-j} for valid j
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # j runs inclusive
        js = np.arange(j_start, j_end + 1)
        is_ = k - js
        A_sub[r, js] = a[is_]

    return A_sub.tocsr()


# ----------------------------------------------------------------------
# Linear‑programming step (active‑set)
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray, top_k: int = 150) -> tuple[np.ndarray, float]:
    """
    Solve a small linear program that tries to find a new vector b (sum(b)=1,
    b>=0) with a strictly smaller max_{k} (a*b)_k.
    Returns (b_opt, M_opt).  If the LP does not improve the bound,
    ``b_opt`` is the original ``a`` and ``M_opt`` equals the current bound.
    """
    n = a.shape[0]
    L = 2 * n - 1

    # current convolution and bound
    conv = _conv_fft(a, a)
    M_cur = conv.max()

    # pick the most critical constraints (largest convolution entries)
    if top_k >= L:
        active = np.arange(L, dtype=np.int64)
    else:
        # indices of the top‑k entries (unsorted)
        active = np.argpartition(conv, -top_k)[-top_k:]

    # Build the reduced constraint matrix A_active * b ≤ M
    A_active = _build_active_constraint_matrix(a, active)

    # Append the variable M (scalar) with coefficient –1 to each inequality:
    #   A_active @ b - M ≤ 0   →   [A_active | -1] @ [b; M] ≤ 0
    rows = A_active.shape[0]
    Mcol = sp.csr_matrix((-np.ones((rows, 1))), shape=(rows, 1))
    A_ub = sp.hstack([A_active, Mcol], format='csr')
    b_ub = np.zeros(rows)

    # Equality: sum(b) = 1
    A_eq = sp.csr_matrix(np.hstack([np.ones((1, n)), np.zeros((1, 1))]))
    b_eq = np.array([1.0])

    # Objective: minimise M  (b variables have zero cost)
    c = np.concatenate([np.zeros(n), np.array([1.0])])
    bounds = [(0, None)] * n + [(0, None)]  # M also non‑negative

    # Solve the LP using HiGHS (fast, works with sparse matrices)
    res = opt.linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      A_eq=A_eq,
                      b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})

    if not res.success:
        # LP failed – return the original vector unchanged
        return a, M_cur

    b_opt = res.x[:n]
    M_opt = res.x[-1]

    # sanity: ensure feasibility w.r.t. the **full** set of constraints.
    # If the reduced set missed a binding row, we fall back to the original.
    full_conv = _conv_fft(a, b_opt)
    if full_conv.max() > M_opt * (1 + 1e-10):
        return a, M_cur

    return b_opt, M_opt


# ----------------------------------------------------------------------
# Light stochastic polishing (Dirichlet moves)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing polishing that reshuffles a random subset."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(0.04 * cur_val, 1e-12)   # initial temperature
    min_temp = 1e-12

    n = len(cur)
    it = 0
    while time.time() < deadline:
        it += 1
        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)                # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    A mixture of a small active‑set linear program and a final simulated‑annealing
    polish is used.  The routine respects the time budget and returns the best
    sequence found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            previous = np.maximum(previous, 0.0)
            if previous.sum() == 0.0:
                previous = None
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Length choice – a moderate 4000 gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – initialise
    # ------------------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        a = previous.copy()
        a /= a.sum()
        print("[Init] using previous solution.")
    else:
        # fallback: a symmetric Beta shape (empirically decent)
        a = _beta_shape(n, alpha=0.4, beta=0.4)
        a /= a.sum()
        print("[Init] using a Beta‑shaped seed.")

    best_seq = a.copy()
    best_val = _objective(best_seq)
    print(f"[Init] surrogate value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – active‑set linear‑programming iterations
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline and lp_iter < 12:
        lp_iter += 1
        time_left = deadline - time.time()
        if time_left < 5.0:
            break

        # solve LP that tries to reduce the maximal convolution entry
        b_opt, M_opt = _lp_improve(best_seq, top_k=150)

        # compute surrogate objective for the candidate
        cand_val = 2.0 * n * M_opt
        if cand_val < best_val - 1e-9:
            best_seq = b_opt
            best_val = cand_val
            print(f"[LP {lp_iter}] improvement: {best_val:.6f}")
        else:
            print(f"[LP {lp_iter}] no improvement (bound stays {best_val:.6f})")
            break   # stop early if LP stalls

    # ------------------------------------------------------------------
    # Stage 3 – stochastic polishing (optional, time‑limited)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, deadline)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation and return
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # store for possible later reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
