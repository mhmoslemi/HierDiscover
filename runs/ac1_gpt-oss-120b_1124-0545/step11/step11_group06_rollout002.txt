The program searches for a non‑negative step‑function (a list of heights) that minimizes  

\[
\text{score}= \frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `*` denotes the linear convolution.  
The score is invariant under scaling, therefore all candidates are normalised to
\(\sum_i a_i = 1\) – the objective reduces to `2*n*max(conv)`.

The search proceeds in five stages:

1. **Deterministic families** – a rich set of analytic windows
   (uniform, triangular, trapezoidal, sine, Tukey, Beta, Gaussian,
   exponential decay, power‑law) are sampled on a fine grid.
   Random Dirichlet vectors and sparse supports are added.
   The `top_k` best candidates are kept.

2. **Short Adam refinements** – each of the survivors is fed to an
   Adam optimiser that minimises a *smooth‑max* surrogate of the true
   objective.  Only a few · 10⁵ iterations are spent per candidate
   (fast enough to try many starts).

3. **Long Adam optimisation** – the current global best is optimised
   for up to a few · 10⁶ iterations with a slowly increasing temperature
   (the surrogate converges to the true max).

4. **Projected sub‑gradient descent** – after the smooth optimisation a
   true sub‑gradient step is taken many times (Polyak step size) to
   tighten the peak that the Adam phase may have missed.

5. **Intensive simulated annealing** – a lightweight annealer repeatedly
   reallocates the mass of a random subset of coordinates (Dirichlet
   moves).  The temperature decays exponentially; moves that improve
   the exact objective are always accepted, otherwise they are accepted
   with the usual Metropolis probability.

All phases respect the time budget (default 1000 s).  The best sequence
found is stored in the global variable `height_sequence_1` and returned as
a plain Python list, ready for `evaluate_sequence`.

```python
"""
Propose a low‑value step function for `evaluate_sequence`.

The algorithm consists of five phases:

1️⃣  systematic sweep over many analytic families (uniform, triangular,
    trapezoidal, sine, Tukey, Beta, Gaussian, exponential, power‑law)
    plus random Dirichlet / sparse seeds – keep the best few candidates.

2️⃣  short Adam optimisation (smooth‑max surrogate) of every candidate.

3️⃣  long Adam run on the current best (millions of iterations).

4️⃣  projected sub‑gradient descent on the exact max‑convolution objective.

5️⃣  intensive simulated‑annealing polishing (random Dirichlet reallocations).

All vectors are ℓ₁‑normalised (sum = 1); the objective is scale‑invariant.
The routine respects the supplied time budget and stores the best sequence
in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (sum=1)."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (used in the Adam surrogate)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat : n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, tau: float) -> np.ndarray:
    """Monotone exponential decay from the left."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / tau)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 1_500_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the smooth‐max surrogate.
    Operates in logits space (softmax) → automatically stays on the simplex.
    Returns the best sequence encountered and its true objective value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature for the smooth‑max (beta → ∞ = true max)
    beta_temp = 5.0
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    lr0 = 0.8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # current probability vector
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # increase temperature slowly
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # smooth max weights
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a (linear convolution with reversed a)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # transform to logits gradient (log‑softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a Polyak step size that references the best value seen so far.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing: repeatedly re‑allocate the total mass
    of a random subset of coordinates (Dirichlet move).  Accepts deteriorating
    moves with a temperature‑dependent Metropolis probability.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        # pick a small random subset of coordinates
        k = rng.integers(2, 14)               # 2 … 13
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

def _keep_candidate(val, seq, best, top_k):
    """Insert a candidate into the sorted best‑k list."""
    if len(best) < top_k:
        best.append((val, seq))
        best.sort(key=lambda x: x[0])
    elif val < best[-1][0]:
        best[-1] = (val, seq)
        best.sort(key=lambda x: x[0])
    return best


def _collect_candidates(n: int,
                        rng: np.random.Generator,
                        previous: np.ndarray | None,
                        top_k: int,
                        deadline: float) -> list[np.ndarray]:
    """
    Systematically explore analytic families and a few random seeds.
    Returns the top_k sequences with smallest exact objective.
    """
    best = []   # list of (value, seq)

    # ---------- deterministic families ----------
    # uniform, triangular, a few trapezoidal widths
    families = [
        _uniform,
        _triangular,
        lambda nn: _trapezoidal(nn, width_fraction=0.08),
        lambda nn: _trapezoidal(nn, width_fraction=0.18),
        lambda nn: _trapezoidal(nn, width_fraction=0.30),
    ]

    for fn in families:
        seq = fn(n)
        best = _keep_candidate(_objective(seq), seq, best, top_k)

    # sine windows
    for gamma in np.linspace(0.5, 8.0, 30):
        seq = _sine_window(n, gamma=gamma)
        best = _keep_candidate(_objective(seq), seq, best, top_k)

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 20):
        seq = _tukey_window(n, alpha=alpha)
        best = _keep_candidate(_objective(seq), seq, best, top_k)

    # Beta shape grid (coarse, then refined later if time allows)
    alphas = np.linspace(0.1, 3.0, 20)
    betas = np.linspace(0.1, 3.0, 20)
    for a in alphas:
        for b in betas:
            seq = _beta_shape(n, a, b)
            best = _keep_candidate(_objective(seq), seq, best, top_k)

    # Gaussian family
    for sigma in np.linspace(0.05 * n, 0.45 * n, 25):
        seq = _gaussian(n, sigma)
        best = _keep_candidate(_objective(seq), seq, best, top_k)

    # Exponential decay
    for tau in np.linspace(0.05 * n, 0.5 * n, 25):
        seq = _exp_decay(n, tau)
        best = _keep_candidate(_objective(seq), seq, best, top_k)

    # Power‑law
    for p in np.linspace(0.0, 5.0, 30):
        seq = _power_law(n, p)
        best = _keep_candidate(_objective(seq), seq, best, top_k)

    # ---------- random seeds ----------
    # Dirichlet draws with various concentrations
    concs = [0.001, 0.005, 0.01, 0.02, 0.05,
             0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 50 if conc < 0.01 else 30
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            best = _keep_candidate(_objective(seq), seq, best, top_k)

    # Sparse random supports (3 … 30 non‑zeros)
    for _ in range(200):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = weights
        best = _keep_candidate(_objective(seq), seq, best, top_k)

    # Mix two random candidates (convex combination)
    if previous is not None:
        # ensure we have at least one previous candidate in the pool
        prev_seq = previous / previous.sum()
        best = _keep_candidate(_objective(prev_seq), prev_seq, best, top_k)

    # forward‑backward mixtures
    pool = [seq for _, seq in best]
    for _ in range(50):
        if time.time() > deadline:
            break
        seq1, seq2 = rng.choice(pool, size=2, replace=False)
        w = rng.random()
        mixed = w * seq1 + (1.0 - w) * seq2
        mixed = _proj_simplex(mixed)
        best = _keep_candidate(_objective(mixed), mixed, best, top_k)

    # Return only the sequences, sorted by ascending objective
    return [seq for _, seq in sorted(best, key=lambda x: x[0])]

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimizes
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a tiny safety margin

    # Length of the step function – 4000 gives a good trade‑off.
    n = 4000
    print(f"\n=== Starting search (n={n}, budget={budget_s}s) ===")

    # ------------------------------------------------------------------
    # Phase 0 – deterministic + random seed sweep (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase0_deadline = min(deadline, start_time + 0.20 * budget_s)
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            prev = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev = None
    else:
        prev = None

    print("\n--- Phase 0: candidate generation ---")
    candidates = _collect_candidates(
        n=n,
        rng=rng,
        previous=prev,
        top_k=15,
        deadline=phase0_deadline,
    )
    print(f"Collected {len(candidates)} top candidates (objective values):")
    for i, seq in enumerate(candidates):
        print(f"  #{i+1:2d}: {_objective(seq):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – short Adam refinements (one pass per candidate)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Phase 1: short Adam refinements ---")
    for idx, cand in enumerate(candidates):
        if time.time() > deadline:
            break
        print(f"[Candidate {idx+1}/{len(candidates)}]")
        seq_adam, val_adam = _adam_optimize(
            cand, n, deadline, max_iters=250_000
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best (short Adam): {best_val:.6f}")

    # fallback (should never happen)
    if best_seq is None:
        best_seq = candidates[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first candidate: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam optimisation on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: long Adam optimisation ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq, n, deadline, max_iters=1_500_000
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated best after long Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: projected sub‑gradient descent ---")
        seq_sub, val_sub = _subgrad_optimize(
            best_seq, n, deadline, max_iters=1_200_000
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing polishing (heavy)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            deadline,
            max_iters=2_500_000,
            init_temp_factor=0.05,
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the official evaluator
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
