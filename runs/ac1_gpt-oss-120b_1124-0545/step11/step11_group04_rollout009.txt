"""
Propose a high‑quality step‑function (non‑negative sequence) that minimises

        2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Rich seed generation** – analytic windows (Gaussian, Kaiser, Tukey,
   DPSS, Chebyshev, etc.) are sampled densely on a grid of parameters.
   The 10 best ℓ₁‑normalised seeds are kept.

2. **Intensive local optimisation** – each seed is refined by
   *log‑space Adam* (smooth max with a temperature schedule that climbs
   up to β≈500), followed by a *projected sub‑gradient* descent on the
   exact max‑convolution, and finally a short *exponentiated‑gradient*
   (mirror‑descent) pass.  The best result over all seeds is retained.

3. **Polishing** – a simulated‑annealing hill‑climber works directly on
   the true objective while preserving the ℓ₁‑norm.  If time permits the
   best sequence is up‑sampled (linear interpolation) to twice the
   resolution and the whole refinement pipeline is repeated.

All operations respect a global deadline (default 1000 s).  The final
sequence is stored in the global name ``height_sequence_1`` and returned
as a plain list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a sequence that is already ℓ₁‑normalised.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _scipy_windows is not None and hasattr(_scipy_windows, 'kaiser'):
        a = _scipy_windows.kaiser(n, beta)
    else:
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        t = beta * np.sqrt(1 - y * y)
        # 0‑th order approximation of I0
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window; falls back to uniform."""
    if _scipy_windows is not None and hasattr(_scipy_windows, 'dpss'):
        a = _scipy_windows.dpss(n, nw, Kmax=1, return_ratios=False)[0]
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    return _uniform(n)


def _chebwin_window(n: int, att: float) -> np.ndarray:
    """Chebyshev window with given side‑lobe attenuation (dB)."""
    if _scipy_windows is not None and hasattr(_scipy_windows, 'chebwin'):
        a = _scipy_windows.chebwin(n, at=att)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    return _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    """Mixture of two symmetric Gaussians placed at shift and 1‑shift."""
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few random cosine components, clip to non‑negative and normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Optimisation primitives

def _logspace_adam_optimize(start_vec: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            beta_max: float = 500.0) -> tuple[np.ndarray | None, float]:
    """
    Adam on the log‑softmax parametrisation using a temperature‑controlled
    smooth max.  Returns the best ℓ₁‑normalised vector found and its
    (surrogate) value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                   # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5
    beta_inc = (beta_max - beta) / max_iters if max_iters > 0 else 0.0

    lr0 = 1.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                               # ℓ₁‑simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # --- smoothed max (log‑sum‑exp) ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    The step size decays like 1/√t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx_max = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx_max]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, idx_max - (n - 1))
        i_end = min(n - 1, idx_max)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx_max - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) directly on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, idx_max - (n - 1))
        i_end = min(n - 1, idx_max)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx_max - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Aggressive annealing on the true objective while preserving the ℓ₁‑norm.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:                     # reshuffle a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            tot = cur[idx].sum()
            if tot <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
            new = cur.copy()
            new[idx] = new_alloc
        else:                                      # tiny pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalisation."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Seed generation + dense parameter sweep
# ----------------------------------------------------------------------
def _find_best_seeds(n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    keep: int = 10) -> list[np.ndarray]:
    """
    Scan a broad collection of analytic families (including fine grids)
    and return the `keep` best ℓ₁‑normalised candidates.
    """
    top = []   # list of (value, vector)

    def consider(vec: np.ndarray):
        val = _objective(vec)
        top.append((val, vec))
        top.sort(key=lambda x: x[0])
        if len(top) > keep:
            top.pop()

    # Uniform baseline
    consider(_uniform(n))

    # Triangular
    consider(_triangular(n))

    # Sine windows (γ)
    for gamma in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        consider(_sine_window(n, gamma))

    # Classical windows
    consider(_hamming_window(n))
    consider(_blackman_window(n))
    consider(_nuttall_window(n))

    # Kaiser (β) – fine grid
    for beta in np.linspace(0.5, 12.0, 24):
        consider(_kaiser_window(n, beta))

    # Tukey (α) – fine grid
    for a in np.linspace(0.05, 0.95, 19):
        consider(_tukey_window(n, a))

    # Gaussian (σ) – dense grid
    for sigma in np.linspace(0.08, 0.45, 38):
        consider(_gaussian(n, sigma))

    # Double Gaussian (σ,shift)
    for sigma in (0.10, 0.12, 0.15, 0.18):
        for shift in (0.18, 0.22, 0.28, 0.32):
            consider(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))

    # Power‑law
    for p in np.linspace(0.0, 5.0, 26):
        consider(_power(n, p))

    # Beta shape – coarse grid (13×13)
    alphas = np.linspace(0.3, 4.5, 11)
    betas = np.linspace(0.3, 4.5, 11)
    for a in alphas:
        for b in betas:
            consider(_beta_shape(n, a, b))

    # DPSS (Slepian) windows – if SciPy is available
    if _scipy_windows is not None and hasattr(_scipy_windows, 'dpss'):
        for nw in np.linspace(1.8, 6.2, 45):
            consider(_dpss_window(n, nw))

    # Chebyshev windows – side‑lobe attenuation
    if _scipy_windows is not None and hasattr(_scipy_windows, 'chebwin'):
        for att in np.linspace(20, 120, 21):
            consider(_chebwin_window(n, att))

    # Random Dirichlet draws (various conc.)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            consider(_random_dirichlet(n, conc, rng))

    # Random low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            consider(_lowfreq_curve(n, nf, rng))

    # Mixture of uniform with low‑frequency curve
    for mix_alpha in (0.15, 0.35, 0.55, 0.75):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            consider(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Include previous global solution if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                consider(prev_arr / prev_arr.sum())
        except Exception:
            pass

    print(f"Seed sweep complete – best {keep} candidates retained.")
    return [vec for _, vec in top]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # small safety margin

    # ------------------------------------------------------------------
    # Stage 0 – resolution choice
    # ------------------------------------------------------------------
    n = 4000                                         # baseline size
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 1 – dense analytic seed sweep
    # ------------------------------------------------------------------
    seed_list = _find_best_seeds(n, deadline, rng, keep=10)

    # initialise global best
    global_best_seq = seed_list[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"[Init] Best surrogate from seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – focused local optimisation (run on each top seed)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seed_list):
        if time.time() > deadline:
            break
        print(f"\n--- Optimising seed #{idx + 1}/{len(seed_list)} ---")
        remaining = deadline - time.time()
        # allocate the bulk of the remaining budget to Adam for the first seed,
        # subsequent seeds get a smaller share.
        share = 0.5 if idx == 0 else 0.15
        max_iters_adam = int(min(400_000, remaining * share))
        seq_adam, val_adam = _logspace_adam_optimize(seed_vec, n,
                                                    deadline,
                                                    max_iters=max_iters_adam,
                                                    beta_max=500.0)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam, val_adam
            print(f"[Global] new best after Adam = {val_adam:.6f}")

        # Sub‑gradient projection (more exact refinement)
        remaining = deadline - time.time()
        max_iters_sub = int(min(2_000_000, remaining * 0.45))
        if max_iters_sub > 0:
            seq_sub, val_sub = _subgradient_proj(global_best_seq, n,
                                                 deadline,
                                                 max_iters=max_iters_sub)
            if val_sub < global_best_val:
                global_best_seq, global_best_val = seq_sub, val_sub
                print(f"[Global] new best after Sub‑Proj = {val_sub:.6f}")

        # Exponentiated‑gradient fine‑tuning
        remaining = deadline - time.time()
        max_iters_exp = int(min(1_200_000, remaining * 0.30))
        if max_iters_exp > 0:
            seq_exp, val_exp = _exp_gradient_optimize(global_best_seq, n,
                                                      deadline,
                                                      max_iters=max_iters_exp)
            if val_exp < global_best_val:
                global_best_seq, global_best_val = seq_exp, val_exp
                print(f"[Global] new best after Exp‑Grad = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(2_000_000, remaining * 1.0))
        seq_sa, val_sa = _simulated_annealing(global_best_seq, n,
                                              deadline,
                                              max_iters=max_iters_sa)
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[Global] new best after SA = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional up‑sampling to finer grid (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 45.0 and n <= 8000:               # enough time for up‑sample
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} for final refinement ---")
            up_seq = _upsample_sequence(global_best_seq, new_n)

            # A short Adam pass at higher resolution
            remaining = deadline - time.time()
            max_iters_up_adam = int(min(250_000, remaining * 0.4))
            seq_up, val_up = _logspace_adam_optimize(up_seq, new_n,
                                                     deadline,
                                                     max_iters=max_iters_up_adam,
                                                     beta_max=600.0)
            if seq_up is not None and val_up < global_best_val:
                global_best_seq, global_best_val = seq_up, val_up
                print(f"[Upsample] new best after Adam = {val_up:.6f}")

            # Follow‑up SA polish at high resolution
            remaining = deadline - time.time()
            if remaining > 15.0:
                max_iters_up_sa = int(min(800_000, remaining * 0.9))
                seq_up, val_up = _simulated_annealing(global_best_seq,
                                                      new_n,
                                                      deadline,
                                                      max_iters=max_iters_up_sa)
                if val_up < global_best_val:
                    global_best_seq, global_best_val = seq_up, val_up
                    print(f"[Upsample] final best after SA = {val_up:.6f}")

            n = new_n  # update the recorded length

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
