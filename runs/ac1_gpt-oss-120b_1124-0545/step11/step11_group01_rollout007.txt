"""
Propose a step‑function (non‑negative vector) that minimises  

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The routine builds a large, diverse seed pool (analytic families, Dirichlet
samples, random permutations), keeps the best few, and then performs three
phases of optimisation:

1.  Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
2.  Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
    objective with a decreasing learning‑rate.
3.  A cheap greedy “mass‑transfer’’ local search that moves a small amount
    of weight between random coordinates while the objective improves.

Periodically the ordering of the vector is changed to the
large‑small‑alternating pattern which is known to reduce the autocorrelation
peak.  The best vector seen is returned and stored in ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (without the denominator, because we keep the
    sequence ℓ₁‑normalised to 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------
def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    # normalise (should already sum to 1, but keep for safety)
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


def _riffle_order(vec: np.ndarray) -> np.ndarray:
    """Interleave two halves (like a perfect riffle shuffle)."""
    n = len(vec)
    half = n // 2
    left = vec[:half]
    right = vec[half:]
    out = np.empty_like(vec)
    out[0::2] = left
    out[1::2] = right
    # if n is odd, copy the last element
    if n % 2:
        out[-1] = vec[-1]
    s = out.sum()
    return out / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Misc random smooth curves
# ----------------------------------------------------------------------
def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed pool construction (richer than the reference version)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # Sine windows with a dense gamma grid
    for gamma in np.linspace(0.2, 12.0, 16):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussian with fine sigma grid
    for sigma in np.linspace(0.05, 0.65, 12):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # Power law (p up to 10)
    for p in np.linspace(0.0, 10.0, 22):
        seeds.append(_power(n, p))
        seeds.append(_alternate_order(_power(n, p)))

    # Beta shape – wide random sampling
    for _ in range(30):
        alpha = 10 ** rng.uniform(-2, 0.8)   # 0.01 … ~6.3
        beta = 10 ** rng.uniform(-2, 0.8)
        seeds.append(_beta_shape(n, alpha, beta))
        seeds.append(_alternate_order(_beta_shape(n, alpha, beta)))

    # Exponential rates covering several decades
    for rate in np.logspace(-5, -0.5, 14):
        seeds.append(_exponential(n, rate))
        seeds.append(_alternate_order(_exponential(n, rate)))

    # Logarithmic shapes
    for eps in np.logspace(-5, -0.5, 12):
        seeds.append(_logarithmic(n, eps))
        seeds.append(_alternate_order(_logarithmic(n, eps)))

    # Random Dirichlet draws – many concentration values
    concs = [0.005, 0.02, 0.07, 0.15, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(4):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))
            seeds.append(_riffle_order(raw))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(3):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(curve)
            seeds.append(_alternate_order(curve))
            seeds.append(_riffle_order(curve))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))
            seeds.append(_alternate_order(_mix_shapes(_uniform(n), curve, mix_alpha)))

    # Very sparse Dirichlet draws (to create spike‑like shapes)
    for _ in range(12):
        raw = rng.dirichlet(np.full(n, 0.001))
        seeds.append(raw)
        seeds.append(_alternate_order(raw))

    # Previous solution, if present
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on the logits of a simplex‑parameterisation, using a
    temperature‑controlled log‑sum‑exp surrogate for max(conv).
    Returns the best feasible vector (ℓ₁‑normalised) seen.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.9

    # temperature schedule – start smooth, finish very sharp
    beta = 0.5        # initial 1/temperature (small -> smooth)
    beta_inc = 0.5
    beta_max = 400.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex (sum = 1)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ----- smoothed max using log‑sum‑exp -----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:        # should not happen but guard against NaNs
            continue
        w /= w_sum

        # gradient of weighted sum of convolutions w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]    # central n entries

        # chain rule for the softmax (logits -> probabilities)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the true
    max‑convolution objective.
    """
    a = start_seq.copy()
    # assure proper normalisation (might be slightly off after Adam)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = _uniform(n)
    else:
        a = a / s

    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index (same as in the reference code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]          # factor 2 is fine for direction

        # multiplicative update (exponentiated gradient)
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional re‑ordering (large‑small alternation)
        if it % 200_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < best_val:
                print(f"[Mirror‑alt] it={it:7d} improved to {val_alt:.6f}")
                a = a_alt
                best_seq = a_alt.copy()
                best_val = val_alt

    return best_seq, best_val


def _local_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_moves: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap greedy hill‑climbing: repeatedly transfer a small amount of
    mass from a random coordinate to another if the objective improves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    move_scale = 0.15   # start moving up to 15 % of the donor's mass

    for mv in range(1, max_moves + 1):
        if time.time() > deadline:
            break

        if mv % 20_000 == 0:
            # slowly decay the step size – we want finer moves later
            move_scale = max(move_scale * 0.9, 1e-4)

        i, j = rng.choice(n, size=2, replace=False)
        donor = cur[i]
        if donor <= 0.0:
            continue
        delta = donor * move_scale * rng.random()
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # re‑normalise (the total mass stays the same, but numerical drift may occur)
        new_sum = new.sum()
        if new_sum <= 0.0:
            continue
        new = new / new_sum

        new_val = _objective(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_seq = new.copy()
                best_val = new_val
                # occasional log
                if mv % 100_000 == 0:
                    print(f"[Local] move={mv} new best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence (list of floats) and stores it globally as
    ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a short safety margin for the final evaluation call
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # Length – 5000 offers a slightly richer search space without a huge
    # slowdown compared with the previous 4000‑element setting.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Load previous best if it exists.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 0 – generate a very rich collection of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Quick evaluation of the exact objective on the seed pool
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on the smooth surrogate
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest number of iterations based on remaining time
        max_iters = min(250_000, int(remaining * 45))  # ~45 iters per second estimate
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # try the alternating re‑ordering of the Adam result
        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq = alt
                global_best_val = alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Mirror‑descent (exponentiated‑gradient) on the true max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_optimize(
            global_best_seq, n, deadline, max_iters=2_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

        # one final alternation attempt
        alt = _alternate_order(global_best_seq)
        alt_val = _objective(alt)
        if alt_val < global_best_val:
            global_best_seq = alt
            global_best_val = alt_val
            print(f"[overall‑Mirror‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Local greedy refinement (mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Local greedy refinement ---")
        seq_ref, val_ref = _local_refine(global_best_seq, n, deadline, max_moves=400_000)
        if val_ref < global_best_val:
            global_best_seq = seq_ref.copy()
            global_best_val = val_ref
            print(f"[overall‑Local] new global best = {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
