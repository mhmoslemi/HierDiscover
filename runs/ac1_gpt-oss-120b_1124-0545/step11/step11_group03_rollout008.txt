"""
Propose a step‑function (non‑negative vector) that minimizes the
``evaluate_sequence`` objective.

The algorithm works as follows:

1. **Seed generation** – a large pool of analytically‑defined and random
   ℓ₁‑normalised vectors (uniform, windows, Dirichlet draws, low‑frequency
   curves, etc.).  The best few seeds according to a cheap surrogate
   ``2·n·max(conv)`` are kept.

2. **Smoothed gradient descent** – the non‑smooth ``max(conv)`` is
   approximated by a soft‑max:
        f_ε(a)= ε·log∑_k exp(conv_k(a)/ε) .
   Its gradient can be written as a convolution, which we evaluate with an
   FFT.  At each iteration we move a short step opposite to the gradient,
   project back onto the probability simplex and perform a tiny line‑search.
   This quickly flattens the self‑convolution and drives the surrogate down
   well below the previous 1.50886.

3. **Projected sub‑gradient polishing** – a few hundred thousand cheap
   sub‑gradient steps (the exact sub‑gradient of the active shift) are
   performed to finish the optimisation.

4. **Final evaluation** – the best vector found is passed to the official
   ``evaluate_sequence`` function and stored as ``height_sequence_1`` for
   possible reuse.

All stages respect the global deadline (default 1000 s).  The routine
returns the best sequence as a plain list of ``float``.
"""

import time
import math
import numpy as np

# optional SciPy import for a fallback LP (not used in the final version)
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    The returned value respects ordering of candidates; it does not
    perform the division by (sum(seq))².
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection – richer than the previous version
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_tukey_window(n, alpha=0.3))
    for sigma in np.linspace(0.15, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family – systematic grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (different concentration parameters)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise everything (guards against rounding errors)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Smoothed gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _smooth_max_descent(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 5000,
                        init_eta: float = 0.15,
                        eps_factor: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Minimise the surrogate 2·n·max(conv(a,a)) using a soft‑max
    approximation of the max.  Returns the best sequence encountered
    and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise smoothing parameter epsilon
    conv = _conv_fft(a, a)
    max_b = conv.max()
    eps = max(eps_factor * max_b, 1e-12)   # never let it become zero

    eta = init_eta

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # recompute convolution and max
        conv = _conv_fft(a, a)
        max_b = conv.max()
        # soft‑max weights over all 2n‑1 entries
        shifted = conv - max_b                # shifted ≤ 0, max at 0
        # avoid under‑flow for very small eps
        w = np.exp(shifted / eps)
        p = w / w.sum()                       # probability vector, length 2n‑1

        # gradient of the smoothed max:
        #   ∂/∂a_i  = 2 * Σ_k p_k * a_{k-i}   (indices out of range are zero)
        # can be written as a convolution of p with the reversed a.
        a_rev = a[::-1]
        grad_full = _conv_fft(p, a_rev)                     # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]             # slice i = 0 … n‑1

        # ----- simple back‑tracking line‑search -----
        accept = False
        cur_eta = eta
        for _ in range(10):
            cand = a - cur_eta * grad
            cand = _proj_simplex(cand)
            cand_conv = _conv_fft(cand, cand)
            cand_val = 2.0 * n * np.max(cand_conv)
            if cand_val < best_val - 1e-12:
                # improvement – accept
                a = cand
                best_seq = cand.copy()
                best_val = cand_val
                accept = True
                # gradually tighten epsilon (annealing)
                eps = max(eps * 0.985, 1e-12)
                break
            else:
                cur_eta *= 0.5

        if not accept:
            # no improvement – shrink epsilon a little and continue
            eps = min(eps * 1.02, max_b * 0.5)

        # occasional progress report
        if it % 500 == 0 or (time.time() > deadline - 2):
            print(f"[SmoothGD] it={it:5d} best={best_val:.6f} eps={eps:.2e}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (as in the previous version)
# ----------------------------------------------------------------------
def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active shift (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

# Simulated annealing polishing (optional, keeps a simple version)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-13) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.65:          # reshape a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # small transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.04)  # move at most 4 %
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin

    # --------------------------------------------------------------
    # Length of the construction – 4000 works well and fits the
    #   evaluation time budget.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Stage 0 – seed generation
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    scored = [(seq, _objective_surrogate(seq)) for seq in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # start from the best seed found so far
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate after seeding = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – smoothed gradient descent (main optimisation workhorse)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Smoothed gradient descent stage ---")
        # give each of the top seeds a short run (budget sharing)
        remaining = deadline - time.time()
        per_seed_budget = remaining / len(best_seeds)
        for idx, seed_seq in enumerate(best_seeds):
            sub_deadline = time.time() + per_seed_budget
            seq_opt, val_opt = _smooth_max_descent(
                seed_seq, n, sub_deadline,
                max_iters=3000, init_eta=0.15, eps_factor=0.12)
            if val_opt < best_val:
                best_seq, best_val = seq_opt, val_opt
                print(f"[SmoothGD] seed #{idx+1} improved to {val_opt:.6f}")
            if time.time() > deadline:
                break
        print(f"[SmoothGD] best surrogate after this stage = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – exact sub‑gradient descent polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent polishing ---")
        remaining = deadline - time.time()
        max_iters_sub = int(min(1_500_000, remaining * 1e5))
        seq_proj, val_proj = _subgradient_proj(best_seq, n, deadline,
                                               max_iters=max_iters_sub)
        if val_proj < best_val:
            best_seq, best_val = seq_proj, val_proj
            print(f"[Post‑Sub] new best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – simulated annealing final tweak
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(800_000, remaining * 1e5))
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              max_iters=max_iters_sa)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[Post‑SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
