def _softmax_mirror_descent(start_vec, n, deadline, max_iters=500_000,
                            init_eta=0.5, init_beta=5.0, final_beta=200.0):
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute convolution
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[SoftMD] it={it} best={best_val:.6f}")
        # compute weight beta schedule linear
        beta = init_beta + (final_beta - init_beta) * min(1.0, it / max_iters)
        # stable softmax
        max_c = np.max(conv)
        exp_w = np.exp(beta * (conv - max_c))
        sum_w = exp_w.sum()
        if sum_w == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = exp_w / sum_w

        # gradient term: conv of reversed a and w
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w)
        # extract slice
        g = conv_rev[n-1:n-1+n] + a  # shape n

        # learning rate decays
        eta = init_eta / np.sqrt(it)
        a = a * np.exp(-eta * g)
        # renormalize to simplex
        total = a.sum()
        if total == 0:
            a = _uniform(n)
        else:
            a = a / total

    return best_seq, best_val

"""
Improved step‑function search.

The algorithm proceeds in three stages while respecting a 1000 s budget.

1. **Seed generation** – a modest pool of analytically‑generated families,
   a few random Dirichlet draws and the previous best solution (if any) are
   normalised to sum = 1.  The best few seeds are kept.

2. **Soft‑max mirror descent** – a smooth approximation of the L∞‑objective
   (the maximal self‑convolution entry) is minimised by an exponentiated‑gradient
   (mirror‑descent) update.  The soft‑max temperature β is increased linearly
   from a moderate start (β≈5) to a large final value (β≈200) so that the
   optimisation first follows a smooth landscape and later focuses on the true
   maximum.  The update uses the exact gradient of the smoothed objective,
   computed efficiently via two FFT‑based convolutions per iteration.

3. **Refinement** – a classic top‑k multi‑sub‑gradient descent followed by a
   lightweight simulated‑annealing polish (Dirichlet reshuffle of a random
   subset of coordinates).

All stages operate directly on the probability simplex (non‑negative,
ℓ₁‑normalised vectors).  The best sequence found is stored in the global
variable ``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation for ℓ₁‑normalised sequences (sum == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

                             np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _softmax_mirror_descent(start_vec: np.ndarray, n: int,
                            deadline: float,
                            eta0: float = 0.5,
                            beta_start: float = 5.0,
                            beta_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the L∞‑objective.
    Two FFT‑based convolutions are performed per iteration.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)                 # self‑convolution
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SoftMD] it={it:6d} best={best_val:.6f}")

        # ----- soft‑max weights ---------------------------------
        # linear schedule for β
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 200_000)
        max_c = np.max(conv)
        # stable exponentiation
        exp_w = np.exp(beta * (conv - max_c))
        w = exp_w / exp_w.sum()               # w_i ≥ 0, Σ w_i = 1

        # ----- gradient of the smoothed objective ---------------
        # g_i = Σ_k w_k a_{k-i} + a_i   (indices out of range contribute 0)
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w)        # length 3n‑2
        g = conv_rev[n - 1: n - 1 + n] + a    # slice the central n entries

        # ----- mirror‑descent (exponentiated gradient) ----------
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # re‑project onto the simplex (just renormalise – all entries stay ≥0)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _subgradient_multi(start_seq: np.ndarray, n: int,
                       deadline: float,
                       max_iters: int = 2_000_000,
                       init_lr: float = 0.30,
                       top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that averages the gradients of the
    *top_k* largest convolution entries.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        cur_max = conv[idxs].max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[MultiSub] it={it:7d} best={best_val:.6f}")

        # aggregate gradient of those entries
        g = np.zeros_like(a)
        for k in idxs:
            idx = np.arange(n)
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += 2.0 * a[rev[mask]]
        g /= float(len(idxs))

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing polish that redistributes mass
    among a random small subset of coordinates (Dirichlet move).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)                  # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the allotted time.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length that balances quality and speed
    # ------------------------------------------------------------------
    n = 5000                     # 5 k gives a richer search space yet stays fast

    # ------------------------------------------------------------------
    # Stage 1 – seed collection
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(3, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – soft‑max mirror descent (deterministic)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        md_deadline = time.time() + 0.45 * time_left
        print("\n--- Soft‑max mirror descent ---")
        # run the descent from each of the top seeds in turn (time‑sharing)
        for idx, seed_vec in enumerate(best_seeds):
            remaining = md_deadline - time.time()
            if remaining <= 0:
                break
            # give each seed a fair slice of the remaining budget
            sub_dead = time.time() + remaining / (len(best_seeds) - idx)
            seq, val = _softmax_mirror_descent(
                seed_vec, n, sub_dead,
                eta0=0.5, beta_start=5.0, beta_end=200.0
            )
            if val < global_best_val:
                global_best_seq, global_best_val = seq, val
                print(f"[SoftMD] new global best from seed {idx+1} = {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – multi‑sub‑gradient refinement
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        sub_deadline = time.time() + 0.45 * time_left
        max_iters = int((sub_deadline - time.time()) * 1800)  # 1800 it/s heuristic
        max_iters = max(200_000, min(max_iters, 4_000_000))
        print("\n--- Multi‑sub‑gradient refinement ---")
        seq, val = _subgradient_multi(
            global_best_seq, n, sub_deadline,
            max_iters=max_iters,
            init_lr=0.30,
            top_k=7
        )
        if val < global_best_val:
            global_best_seq, global_best_val = seq, val
            print(f"[MultiSub] new global best = {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq, val = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=800_000,
            init_temp_factor=0.04
        )
        if val < global_best_val:
            global_best_seq, global_best_val = seq, val
            print(f"[SA] final global best = {val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for possible later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
