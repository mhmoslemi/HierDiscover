"""
Minimal‑max‑convolution step‑function search.

The algorithm proceeds in four stages:

1. **Exploratory seed generation** – a dense collection of analytic families
   (uniform, triangular, trapezoidal, geometric, power‑law, Beta, Gaussian,
   Tukey, sine‑window) plus many random Dirichlet / exponential draws is
   created on a modest size (≈1200).  The exact surrogate objective
   `2·n·max(conv)` (for a ℓ¹‑normalised vector) is evaluated and the best
   few seeds are kept.

2. **Upscaling & heavy Adam optimisation** – the selected seeds are linearly
   interpolated to a larger size (`n≈8000`).  Each seed is then optimised with
   the Adam optimiser on the *log‑sum‑exp* smooth‐max surrogate of the
   convolution.  The temperature β is slowly increased up to a high bound
   (`β_max≈5000`), allowing the optimiser to focus on the true max as the run
   proceeds.

3. **Projected sub‑gradient refinement** – the best sequence from Adam is
   further polished by a (simpler) sub‑gradient method directly on the true
   max‑convolution objective, with Polyak‑style diminishing steps and Euclidean
   projection onto the probability simplex.

4. **Stochastic polishing** – a final simulated‑annealing pass perturbs small
   random subsets of the mass and accepts downhill moves or uphill moves with
   a slowly cooling temperature.

All stages respect the global wall‑time budget (default 1000 s).  The final
best sequence (as a list of floats) is returned and stored in the global
variable ``height_sequence_1`` for possible reuse.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate objective for a ℓ¹‑normalised sequence (∑seq == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (0<r<1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a small‑n sequence to a larger n."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    else:
        ys = _uniform(n_big)
    return ys.astype(np.float64)


def _interp_from_params(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a length‑`n` ℓ¹‑normalised sequence by linear interpolation of
    a coarse parameter vector `params` (all non‑negative).  The interpolation
    points are placed uniformly in [0, 1].
    """
    k = len(params)
    if k == 1:
        seq = np.full(n, params[0])
    else:
        xs = np.linspace(0.0, 1.0, k)
        xb = np.linspace(0.0, 1.0, n)
        seq = np.interp(xb, xs, params, left=params[0], right=params[-1])
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s > 0:
        seq /= s
    else:
        seq = _uniform(n)
    return seq

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   beta_max: float = 5000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max‑convolution.
    The optimisation works on the log‑softmax parametrisation which
    automatically enforces the simplex constraint.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature schedule
    beta_temp = 10.0
    beta_mul = 1.01   # slower increase – more stable

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex (sum=1)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # temperature increase (sharpen the smooth‑max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k conv_k  w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 8.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.7
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 2_000_000,
                      init_lr: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for active convolution index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_500_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polishing that reshuffles a random subset of the mass.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 15000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _generate_small_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a rich pool of analytic and random candidates on size `n`."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Trapezoidal sweep
    for wf in np.linspace(0.02, 0.45, 14):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    # Sine windows
    for gamma in np.linspace(0.0, 6.0, 35):
        seeds.append(_sine_window(n, gamma=gamma))

    # Tukey windows
    for alpha in np.linspace(0.0, 0.95, 22):
        seeds.append(_tukey_window(n, alpha=alpha))

    # Geometric families
    geo_r_vals = np.concatenate(
        [np.linspace(0.01, 0.98, 22), np.linspace(0.98, 0.9999, 17)]
    )
    for r in geo_r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law
    for p in np.linspace(0.0, 8.0, 33):
        seeds.append(_power(n, p))

    # Beta shapes – very wide grid (including extreme α,β)
    alphas = np.concatenate(
        (np.geomspace(0.05, 0.5, 6), np.linspace(0.6, 5.0, 12), np.geomspace(5.5, 20, 5))
    )
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in np.linspace(0.06, 0.45, 11):
        seeds.append(_gaussian(n, sigma_frac))

    # Random Dirichlet draws (diverse concentrations)
    for conc in (0.0005, 0.001, 0.005, 0.02, 0.1, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Piecewise linear random seeds (coarse control points)
    K = 15
    for _ in range(20):
        params = rng.dirichlet(np.full(K, 1.0))
        seeds.append(_interp_from_params(params, n))

    # Include previous solution if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                tot = prev_arr.sum()
                if tot > 0:
                    seeds.append(prev_arr / tot)
        except Exception:
            pass

    return seeds

    """

    Returns the best sequence as a list of floats.
    """

    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # Stage 0 – small‑scale exploratory sweep
    # ------------------------------------------------------------------
    n_small = 1200
    print(f"\n--- Stage 0: generating {n_small}‑length analytic & random seeds ---")
    small_seeds = _generate_small_seeds(n_small, rng)
    print(f"Total seeds generated: {len(small_seeds)}")

    # evaluate all small seeds (exact surrogate)
    evaluated_small = [(vec, _objective(vec)) for vec in small_seeds]
    evaluated_small.sort(key=lambda x: x[1])

    top_k_small = min(25, len(evaluated_small))
    best_small = [vec for vec, _ in evaluated_small[:top_k_small]]
    print(f"Top {top_k_small} small seeds (objective values):")
    for i, (_, val) in enumerate(evaluated_small[:top_k_small]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – upscale to a larger problem size
    # ------------------------------------------------------------------
    n_large = 8000
    print(f"\n--- Stage 1: upscaling to length {n_large} ---")
    large_seeds = [_upscale(vec, n_large) for vec in best_small]

    evaluated_large = [(vec, _objective(vec)) for vec in large_seeds]
    evaluated_large.sort(key=lambda x: x[1])
    global_best_seq = evaluated_large[0][0].copy()
    global_best_val = evaluated_large[0][1]
    print(f"[Init] best surrogate after upscale = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation (temperature‑aware)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        seeds_to_opt = min(3, len(large_seeds))
        per_seed_budget = remaining / (seeds_to_opt + 1.0)   # keep some for later stages
        print(f"\n--- Stage 2: Adam optimisation (≈{per_seed_budget:.1f}s per seed) ---")
        for idx, seed_vec in enumerate(large_seeds[:seeds_to_opt]):
            if time.time() > deadline:
                break
            # Estimate feasible iteration count (empirical 1 iter ≈ 0.0018 s for n=8000)
            est_iter = int(per_seed_budget / 0.0018)
            est_iter = max(30_000, min(est_iter, 400_000))
            print(f"\n[Adam] seed #{idx+1}/{seeds_to_opt}, max_iters≈{est_iter}")
            seq_adam, val_adam = _adam_optimize(
                seed_vec, n_large, deadline,
                max_iters=est_iter,
                beta_max=5000.0,
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        remaining = deadline - time.time()
        max_iters = int(remaining * 1500)   # about 1500 iters per second on avg
        max_iters = max(400_000, min(max_iters, 2_500_000))
        print(f"\n--- Stage 3: sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n_large, deadline,
            max_iters=max_iters,
            init_lr=0.30,
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – stochastic polishing (Simulated Annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_sa_iters = int(remaining * 900)   # approx 0.0011 s per iteration
        max_sa_iters = max(300_000, min(max_sa_iters, 2_500_000))
        print(f"\n--- Stage 4: simulated‑annealing polishing (max_iters={max_sa_iters}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n_large, deadline,
            max_iters=max_sa_iters,
            init_temp_factor=0.04,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
