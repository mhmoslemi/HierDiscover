def _smooth_max_descent(start_seq, n, deadline, max_iters=20000, eta0=0.5, gamma=50.0, uniform_mix=0.01):

    p = start_seq.copy()

    best_seq = p.copy()
    best_val = _objective_surrogate(p)

    for it in range(1, max_iters+1):
        if time.time()>deadline: break

        # compute conv p * p
        conv = _conv_fft(p, p)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = p.copy()

        # smoothing weights
        max_c = conv.max()
        # to prevent overflow, shift
        w = np.exp(gamma * (conv - max_c))  # exponentials of non-positive values
        w_sum = np.sum(w)
        if w_sum == 0:
            # fallback to argmax only
            g = np.zeros_like(p)
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end+1)
            rev_idx = max_idx - idx_i
            g[idx_i] = 2.0 * p[rev_idx]
        else:
            v = w / w_sum
            # gradient = 2 * (v * reverse(p))
            p_rev = p[::-1]
            grad = _conv_fft(v, p_rev)  # length len(v)+len(p_rev)-1
            # We need first n entries (indices 0..n-1)
            g = 2.0 * grad[:n]

        # step size
        eta = eta0 / np.sqrt(it)

        # mirror descent via exponentiation
        logp = np.log(p + 1e-30) - eta * g
        # mixing with uniform
        if uniform_mix > 0:
            # simple convex combination of logp with uniform distribution's log (which is -log n)
            log_u = -np.log(n) * np.ones_like(p)  # uniform distribution log-prob
            lam = uniform_mix / (it**0.5)  # decaying mixing
            logp = (1 - lam) * logp + lam * log_u

        p = np.exp(logp - np.max(logp))
        p /= p.sum()

    return best_seq, best_val

"""Optimization of a non‑negative step function.

The algorithm consists of three stages
--------------------------------------
1. **Initialisation** – start from the previous best sequence (if any) or
   the uniform distribution.

2. **Smooth‑max mirror descent** – minimise the quantity
   ``max(conv(p,p))`` (with the ℓ₁‑normalisation Σp=1) by replacing the
   non‑smooth maximum with a log‑sum‑exp approximation.
   An exponentiated‑gradient (mirror‑descent) update is performed and the
   iterate is always projected back onto the probability simplex.
   The update uses the exact gradient of the smooth surrogate, which
   involves a second convolution (cost ≈ 2 × FFT).  The step size, the
   smoothing parameter and a tiny mixing with the uniform distribution
   are annealed with √iteration.

3. **Simulated‑annealing polishing** – a cheap stochastic local search
   that reshuffles mass on small random subsets and accepts moves according
   to a slowly cooling temperature schedule.

The best sequence found before the deadline is stored in the global name
``height_sequence_1`` and also returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) assuming Σseq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max mirror‑descent (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------
def _smooth_max_descent(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000,
                        eta0: float = 0.9,
                        gamma: float = 40.0,
                        lam0: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise  max(conv(p,p))  on the probability simplex.
    Uses a log‑sum‑exp smooth approximation of the max and an
    exponentiated‑gradient update.
    Returns the best sequence seen and its surrogate value.
    """
    p = start_seq.copy()
    best_seq = p.copy()
    best_val = _objective_surrogate(p)
    uniform_log = -np.log(n)                     # log‑probability of uniform

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # 1) current convolution and surrogate value
        conv = _conv_fft(p, p)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = p.copy()
            if it % 5_000 == 0:
                print(f"[MD] it={it:7d} best surrogate={cur_val:.6f}")

        # 2) smooth approximation of the max
        max_c = conv.max()
        # exp(γ·(conv−max_c)) ≤ 1, so no overflow
        w = np.exp(gamma * (conv - max_c))
        w_sum = w.sum()
        if w_sum <= 0.0:
            # fall back to the pure sub‑gradient (very unlikely)
            g = np.zeros_like(p)
            i_start = max(0, max_idx - (n - 1))
            i_end = min(max_idx, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = max_idx - idx
            g[idx] = 2.0 * p[rev]
        else:
            v = w / w_sum                     # a probability distribution over the sums
            # gradient: g_i = 2·∑_k v_k·p_{k‑i}
            # compute via convolution of v with the reversed p
            grad_full = _conv_fft(v, p[::-1])
            g = 2.0 * grad_full[n - 1:n - 1 + n]

        # 3) step‑size and mixing
        eta = eta0 / math.sqrt(it)           # decreasing learning rate
        lam = lam0 / math.sqrt(it)           # decreasing uniform mixing

        # 4) exponentiated‑gradient update (mirror descent w.r.t. KL)
        logp = np.log(p + 1e-30) - eta * g
        # tiny convex combination with the uniform distribution in log‑space
        logp = (1.0 - lam) * logp + lam * uniform_log

        # 5) project back onto the simplex (just exponentiate & renormalise)
        p = np.exp(logp - np.max(logp))
        p /= p.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing (kept from the reference implementation)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_500_000,
                         init_temp_factor: float = 0.08,
                         min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:                     # reshape a random subset
            k = rng.integers(2, 12)               # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                      # tiny transfer between two slots
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)   # ≤ 5 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val
