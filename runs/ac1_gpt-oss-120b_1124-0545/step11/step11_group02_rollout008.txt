"""
Hybrid search for a step‑function that minimises the provided
``evaluate_sequence``.  The algorithm consists of three parts:

1. **Seed pool** – a collection of analytically generated vectors
   (uniform, triangular, trapezoidal, geometric, power, Gaussian,
   sine‑window, Tukey, random Dirichlet).  The best seed is kept.

2. **Deterministic refinement** – an exponential‑weights (mirror)
   descent that pushes mass away from the currently maximal convolution
   entry, followed by a multi‑sub‑gradient descent that smooths the
   whole convolution profile by averaging the gradients of the
   ``top_k`` largest entries.

3. **Stochastic polishing** – a simulated‑annealing loop that repeatedly
   reshuffles a random subset of coordinates (Dirichlet move) and
   accepts moves according to a temperature schedule.

All steps respect the shared time budget (default 1000 s) and keep the
vector on the probability simplex (non‑negative, sum = 1).  The best
sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation for ℓ₁‑normalised sequences (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _exp_mirror_descent(start_vec: np.ndarray, n: int,
                        deadline: float, max_iters: int = 400_000,
                        eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponential‑weights (mirror) descent on the L∞‑convolution objective.
    Sub‑gradient is taken from the presently maximal convolution entry.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Mirror] it={it:6d} best={best_val:.6f}")

        # sub‑gradient of active entry
        idx = np.arange(n)
        rev = max_idx - idx
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _subgradient_multi(start_seq: np.ndarray, n: int,
                       deadline: float, max_iters: int = 2_000_000,
                       init_lr: float = 0.30, top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent averaging the gradients of the
    top_k largest convolution entries.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)

        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        cur_max = conv[idxs].max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[MultiSub] it={it:7d} best={best_val:.6f}")

        # aggregate gradient of those entries
        g = np.zeros_like(a)
        for k in idxs:
            idx = np.arange(n)
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += 2.0 * a[rev[mask]]
        g /= float(len(idxs))

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that reshuffles a random subset of
    coordinates (Dirichlet move) while staying on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        # pick a random subset (2‑12 elements)
        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) and decide on the vector length
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.size > 1:
                n = prev_arr.shape[0]
            else:
                n = 4000
        except Exception:
            n = 4000
    else:
        n = 4000

    print(f"\n--- Length of search vectors: {n} ---")

    # ------------------------------------------------------------------
    # Stage 0 – build a modest pool of seeds
    # ------------------------------------------------------------------
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 5):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 5):
        seeds.append(_sine_window(n, gamma=gamma))
    for r in np.linspace(0.2, 0.99, 7):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))
    for p in np.linspace(0.0, 5.0, 6):
        seeds.append(_power(n, p))
    for sigma in (0.12, 0.20, 0.28):
        seeds.append(_gaussian(n, sigma))
    for alpha in np.linspace(0.0, 0.9, 5):
        seeds.append(_tukey_window(n, alpha=alpha))
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(3):
        seeds.append(_random_exponential(n, rng))

    if previous is not None:
        try:
            avg = np.asarray(previous, dtype=np.float64)
            s = avg.sum()
            if s > 0:
                seeds.append(avg / s)
        except Exception:
            pass

    # Evaluate seeds with cheap surrogate
    eval_seeds = [(vec, _objective(vec)) for vec in seeds]
    eval_seeds.sort(key=lambda x: x[1])
    best_seed, best_val = eval_seeds[0]
    print(f"[Seed] best surrogate value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Short mirror‑descent runs on the top few seeds (quick diversification)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 15.0:
        short_deadline = time.time() + 0.15 * time_left
        for i, (vec, _) in enumerate(eval_seeds[:5]):
            seq, val = _exp_mirror_descent(vec, n, short_deadline,
                                           max_iters=150_000, eta0=0.5)
            if val < best_val:
                best_val, best_seed = val, seq
                print(f"[ShortMirror #{i}] improved surrogate = {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – long deterministic optimisation
    # ------------------------------------------------------------------
    current_seq = best_seed.copy()
    best_seq = current_seq.copy()
    best_val = _objective(best_seq)
    print(f"[Start] deterministic optimisation, objective = {best_val:.6f}")

    # Allocate time for mirror descent, sub‑gradient, SA in that order
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # ----- Mirror descent (≈40 % of remaining time) -----
        md_deadline = time.time() + 0.40 * remaining
        seq_md, val_md = _exp_mirror_descent(best_seq, n, md_deadline,
                                             max_iters=500_000, eta0=0.6)
        if val_md < best_val:
            best_val, best_seq = val_md, seq_md
            print(f"[Mirror] new best = {best_val:.6f}")

        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # ----- Multi‑sub‑gradient (≈30 % of remaining) -----
        sg_deadline = time.time() + 0.30 * remaining
        seq_sg, val_sg = _subgradient_multi(best_seq, n, sg_deadline,
                                            max_iters=2_000_000,
                                            init_lr=0.25,
                                            top_k=8)
        if val_sg < best_val:
            best_val, best_seq = val_sg, seq_sg
            print(f"[SubGrad] new best = {best_val:.6f}")

        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # ----- Simulated annealing polish (use whatever is left) -----
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              max_iters=800_000,
                                              init_temp_factor=0.06,
                                              min_temp=1e-14)
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa
            print(f"[SA] new best = {best_val:.6f}")

        # One full cycle is enough – the loop will exit if time permits
        break

    # ------------------------------------------------------------------
    # Normalise (protect against any accidental drift) and final check
    # ------------------------------------------------------------------
    sum_a = best_seq.sum()
    if sum_a == 0.0:
        best_seq = _uniform(n)
        sum_a = best_seq.sum()
    best_seq = best_seq / sum_a

    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible later reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    print("\n=== Finished ===")
    return best_seq.tolist()
