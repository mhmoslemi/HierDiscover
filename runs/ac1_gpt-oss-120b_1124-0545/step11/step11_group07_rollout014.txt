Implementation details for LP refinement function:

def _lp_refine(f, n, M, k_constraints=300):
    """Given current distribution f (np.ndarray, sum=1), returns refined distribution."""
    # Determine active shifts (largest conv entries for f)
    conv_f = _conv_fft(f, f)
    idx_sorted = np.argsort(-conv_f)  # descending
    # select top K constraints, but ensure we have at least one each side; maybe always include idx of max.
    top_k = min(k_constraints, len(conv_f))
    active_idx = idx_sorted[:top_k]
    # Build sparse matrix A with shape (len(active_idx), n): each row k is f reversed in appropriate way: conv_f_g[k] = Σ_i f_i * g_{k-i}
    # This is linear in g: A_{row, i} = f_{k-i} if 0 <= k-i < n else 0.
    rows = []
    cols = []
    data = []
    for row_i, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i from i_start to i_end inclusive => f[k - i] index.
        # vector of length i_end - i_start + 1.
        # For each i, add entry.
        idx_i = np.arange(i_start, i_end + 1)
        idx_f = k - idx_i
        rows.extend([row_i] * len(idx_i))
        cols.extend(idx_i.tolist())
        data.extend(f[idx_f].tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))

    # Solve LP
    import cvxpy as cp
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000) # maybe set high.

    if g.value is None:
        # fallback: return original
        return f
    g_val = np.maximum(g.value, 0)
    # Normalize to sum 1
    if g_val.sum() > 0:
         g_val = g_val / g_val.sum()
    return g_val

"""
A multi‑stage optimizer for the step‑function challenge.

Stage 0 – generate a diverse pool of ℓ1‑normalised seeds (analytic families,
random Dirichlet draws, sparse supports, …).

Stage 1 – polish each seed with a medium‑length Adam run on a smooth
log‑sum‑exp surrogate of the max‑convolution.  The best of the seeds is kept.

Stage 2 – on the current global best we perform a *linear‑program* refinement.
Given the current function f we maximise Σ g_j subject to the linear constraints
(f * g)_k ≤ M (M = max (f * f)) on a small active set of “tight’’ shifts.
The solution g is mixed with f (convex combination) which provably does not
increase the objective.  A few such Frank‑Wolfe‑type steps dramatically
tighten the max‑convolution bound.

Stage 3 – a long projected sub‑gradient descent on the exact max‑convolution.

Stage 4 – cheap stochastic local‑search moves (re‑allocate a tiny Dirichlet
mass between two indices) for the last few percent of the time budget.

All stages respect a user‑supplied budget (default 1000 s) and the final
sequence is re‑scaled to ℓ1‑norm = √(2 n) so that the official evaluator
returns the peak of the convolution.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σv=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective on the probability simplex (Σseq = 1):
    returns max_k (seq * seq)_k.
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.15) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)          # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

def _collect_diverse_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    top_k: int,
    deadline: float,
) -> list[np.ndarray]:
    """
    Build a pool of candidates and keep the `top_k` smallest values of
    the exact max‑convolution objective.
    """
    top_vals = []                       # (value, seq)

    def try_insert(val, seq):
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])

    # deterministic families
    fam_fns = [
        _uniform,
        _triangular,
        lambda nn: _trapezoidal(nn, width_fraction=0.15),
        lambda nn: _sine_window(nn, gamma=1.0),
        lambda nn: _sine_window(nn, gamma=2.0),
        lambda nn: _gaussian(nn, sigma_frac=0.12),
        lambda nn: _gaussian(nn, sigma_frac=0.24),
        lambda nn: _tukey_window(nn, alpha=0.2),
        lambda nn: _tukey_window(nn, alpha=0.4),
    ]
    for fn in fam_fns:
        seq = fn(n)
        try_insert(_objective(seq), seq)

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seq = _geom(n, r)
        try_insert(_objective(seq), seq)
        seq = _geom(n, r) * 0.0  # placeholder to keep pattern (no symmetry needed)

    # power‑law family
    for p in np.linspace(0.0, 5.0, 18):
        seq = _power(n, p)
        try_insert(_objective(seq), seq)

    # random Dirichlet draws (many concentrations)
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 70 if conc < 0.01 else 40
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            try_insert(_objective(seq), seq)

    # sparse random supports
    for _ in range(250):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        try_insert(_objective(seq), seq)

    # random beta shapes
    for _ in range(300):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 5.0)
        beta = rng.uniform(0.05, 5.0)
        seq = _beta_shape(n, alpha, beta)
        try_insert(_objective(seq), seq)

    # random sine windows
    for _ in range(200):
        if time.time() > deadline:
            break
        gamma = rng.uniform(0.2, 5.0)
        seq = _sine_window(n, gamma=gamma)
        try_insert(_objective(seq), seq)

    # random Gaussian windows
    for _ in range(200):
        if time.time() > deadline:
            break
        sigma_frac = rng.uniform(0.05, 0.4)
        seq = _gaussian(n, sigma_frac=sigma_frac)
        try_insert(_objective(seq), seq)

    # random Tukey windows
    for _ in range(150):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 0.9)
        seq = _tukey_window(n, alpha=alpha)
        try_insert(_objective(seq), seq)

    # mixture of uniform & Gaussian
    for _ in range(150):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        try_insert(_objective(seq), seq)

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        seq = np.asarray(previous, dtype=np.float64)
        if seq.sum() > 0:
            seq = seq / seq.sum()
            try_insert(_objective(seq), seq)

    return [seq for _, seq in top_vals]   # sorted by increasing value

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_000_000,
    beta_max: float = 800.0,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best probability vector (sum = 1) and its exact max‑conv value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                  # stay on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # raise temperature slowly
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # smooth‑max weights over the convolution entries
        conv = _conv_fft(a, a)            # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)       # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # map gradient to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best probability vector encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (use current best)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _lp_refine(
    f: np.ndarray,
    n: int,
    M: float,
    active_k: np.ndarray,
    t: float = 0.2,
) -> np.ndarray | None:
    """
    One Frank‑Wolfe style refinement step using a tiny LP.
    Returns a new probability vector (sum = 1) or None on failure.
    """
    # Build sparse matrix A where (A @ g)_k = (f * g)_k
    rows = []
    cols = []
    data = []
    for row_id, k in enumerate(active_k):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        vals = f[k - idx_i]                     # coefficients f_{k-i}
        rows.extend([row_id] * len(idx_i))
        cols.extend(idx_i.tolist())
        data.extend(vals.tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_k), n))

    # Linear program: maximise Σ g_i   s.t. g ≥ 0, A g ≤ M
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    except Exception as e:
        # Solver failure – skip this refinement
        return None

    if g.value is None:
        return None

    g_val = np.maximum(g.value, 0.0)
    s = g_val.sum()
    if s <= 0:
        return None
    g_val = g_val / s                         # re‑normalise to ℓ1‑norm = 1

    # convex blend with the old point (guarantees non‑increase)
    new_f = (1.0 - t) * f + t * g_val
    new_f = _proj_simplex(new_f)
    return new_f


def _local_search(
    seq: np.ndarray,
    n: int,
    deadline: float,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny Dirichlet re‑allocations between two random indices.
    Accepts a move if it strictly improves the max‑convolution.
    """
    cur = seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    it = 0
    while it < max_steps and time.time() < deadline:
        it += 1
        # pick two distinct indices with non‑zero mass
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] == 0.0:
            continue
        # move a small piece from i to j (or vice‑versa)
        eps = min(cur[i], 0.002)               # up to 0.2 % of total mass
        if eps <= 0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        # re‑project onto the simplex to avoid tiny drift
        new = _proj_simplex(new)

        new_val = _objective(new)
        if new_val < best_val:
            best_val = new_val
            best_seq = new.copy()
            cur = new
            cur_val = new_val
            if it % 5000 == 0:
                print(f"[LocalSearch] it={it:7d} best={best_val:.6f}")
        else:
            # occasionally accept uphill moves (simulated‑annealing style)
            if rng.random() < 0.001:
                cur = new
                cur_val = new_val

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence found.
    """
    # ------------------------------------------------------------------
    # General set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    # Length of the step‑function – 4000 works well, but the code will
    # gracefully adapt if the time budget is very tight.
    n = 4000
    sqrt2n = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Stage 0 – seed generation (≈10 % of budget)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    seed_deadline = min(deadline, start_time + budget_s * 0.12)
    seeds = _collect_diverse_seeds(
        n=n,
        rng=rng,
        previous=previous,
        top_k=7,
        deadline=seed_deadline,
    )
    print(f"Collected {len(seeds)} promising seeds (kept top‑{len(seeds)}).")

    # ------------------------------------------------------------------
    # Stage 1 – medium‑length Adam refinements on the best seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Adam refinement of seeds (stage 1) ---")
    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"[Seed {idx + 1}/{len(seeds)}]")
        s = seed_vec / seed_vec.sum()               # probability simplex
        seq_adam, val_adam = _adam_optimize(
            start_vec=s,
            n=n,
            deadline=deadline,
            max_iters=800_000,
            beta_max=800.0,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after Adam (seed) = {best_val:.6f}")

    # Fallback to uniform if nothing succeeded
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print(f"Fallback to uniform seed: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long Adam run on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long Adam optimisation (stage 2) ---")
        seq_adam, val_adam = _adam_optimize(
            start_vec=best_seq,
            n=n,
            deadline=deadline,
            max_iters=3_000_000,
            beta_max=1200.0,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after long Adam = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – linear‑program refinement (Frank‑Wolfe style)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP refinement (stage 3) ---")
        # we will run a few refinement steps; each step solves a tiny LP
        for rp in range(5):
            if time.time() > deadline:
                break
            cur_M = _objective(best_seq)
            conv_full = _conv_fft(best_seq, best_seq)
            # active constraints: top 200 values (or 5 % of the shifts)
            top_k = min(200, len(conv_full))
            active_idx = np.argpartition(-conv_full, top_k - 1)[:top_k]
            new_seq = _lp_refine(best_seq, n, cur_M, active_idx, t=0.15)
            if new_seq is None:
                print("  LP solve failed – aborting LP stage.")
                break
            new_val = _objective(new_seq)
            if new_val < best_val:
                best_seq = new_seq
                best_val = new_val
                print(f"  LP step {rp+1}: improved to {best_val:.6f}")
            else:
                print(f"  LP step {rp+1}: no improvement")
                # early exit if stagnating
                if rp >= 2:
                    break

    # ------------------------------------------------------------------
    # Stage 4 – projected sub‑gradient descent (more iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (stage 4) ---")
        seq_sub, val_sub = _subgrad_optimize(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=1_200_000,
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated global best after sub‑grad = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – stochastic local search (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Local search polishing (stage 5) ---")
        seq_loc, val_loc = _local_search(
            best_seq,
            n=n,
            deadline=deadline,
            max_steps=250_000,
        )
        if val_loc < best_val:
            best_seq = seq_loc
            best_val = val_loc
            print(f"  Updated global best after local search = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Scale to the required ℓ₁ norm (√(2n)) and final sanity check
    # ------------------------------------------------------------------
    best_seq = best_seq * sqrt2n

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective (official) ≈ {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    # Store for reuse and return plain list
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
