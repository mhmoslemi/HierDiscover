"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm follows a three‑stage strategy:

1. **Broad candidate sweep** – deterministic families (Beta, cosine‑power,
   Tukey, geometric, quadratic, etc.) together with many random Dirichlet
   draws and sparse supports.  The 20 best candidates (lowest exact objective)
   are retained.
2. **Focused Adam optimisation** – a long Adam run on the single best
   candidate.  The Adam optimiser works in the log‑softmax (logits) space,
   uses a smoothly increasing temperature to approximate the max‑operator,
   and stops automatically when the time budget expires.
3. **Fine polishing** – a short projected sub‑gradient descent followed by a
   lightweight simulated‑annealing phase, both of which respect the remaining
   time.

All vectors are kept on the probability simplex (sum = 1).  The routine
honours the supplied ``budget_s`` (default 1000 s) and stores the final
sequence in the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a probability vector (sum=1).
    Returns 2·n·max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a point on the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all returned already normalised to sum = 1)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_power(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π (i+0.5)/n) ^ power."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Standard Tukey window (α in (0,1])."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p  (p ≥ 0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _parabolic(n: int) -> np.ndarray:
    """a_i ∝ i·(n-1-i) (zero at both ends, peak in centre)."""
    i = np.arange(n, dtype=np.float64)
    a = i * (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 2_000_000,
                   beta0: float = 5.0,
                   beta_mul: float = 1.005,
                   beta_max: float = 2000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Operates on logits (log‑softmax) so the simplex constraint is automatic.
    Returns the best sequence found and its exact objective value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = beta0
    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # temperature update (approach true max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # convolution a*a
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # soft‑max over conv entries

        # gradient of Σ w_k·conv_k w.r.t a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional norm clipping for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 1.0
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 700_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution objective.
    Uses a simple Polyak step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing that randomly re‑allocates the mass of a
    small subset of coordinates (Dirichlet move).  Accepts worsening moves
    according to a decreasing temperature schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)                     # size of the perturbed subset
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 40_000 == 0:
                    print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

def _keep_best(best: list[tuple[float, np.ndarray]],
               val: float,
               seq: np.ndarray,
               top_k: int) -> None:
    """Maintain a sorted list of the best *top_k* (value, seq) pairs."""
    if len(best) < top_k:
        best.append((val, seq))
        best.sort(key=lambda x: x[0])
    elif val < best[-1][0]:
        best[-1] = (val, seq)
        best.sort(key=lambda x: x[0])


def _collect_candidates(n: int,
                       rng: np.random.Generator,
                       previous: np.ndarray | None,
                       top_k: int,
                       deadline: float) -> list[np.ndarray]:
    """
    Deterministically sweep analytic families, add random seeds and a few
    convex combinations of the current elites, returning the best *top_k*
    normalised sequences.
    """
    best: list[tuple[float, np.ndarray]] = []

    # ----- deterministic families -------------------------------------------------
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.08),
        lambda n: _trapezoidal(n, width_fraction=0.18),
        lambda n: _trapezoidal(n, width_fraction=0.30),
        lambda n: _beta_shape(n, 2.0, 2.0),          # symmetric parabola‑like
        lambda n: _beta_shape(n, 0.5, 0.5),          # U‑shaped
        lambda n: _cosine_power(n, 1.0),
        lambda n: _cosine_power(n, 2.0),
        lambda n: _cosine_power(n, 4.0),
        lambda n: _tukey_window(n, 0.3),
        lambda n: _tukey_window(n, 0.6),
        lambda n: _geom(n, 0.95),
        lambda n: _sym_geom(n, 0.95),
        _parabolic,
    ]
    for fn in families:
        if time.time() > deadline:
            break
        seq = fn(n)
        _keep_best(best, _objective(seq), seq, top_k)

    # ----- beta shape grid (broader ranges) ------------------------------------
    alphas = np.concatenate([np.linspace(0.05, 0.5, 8),
                             np.linspace(0.6, 4.0, 12)])
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            seq = _beta_shape(n, a, b)
            _keep_best(best, _objective(seq), seq, top_k)

    # ----- cosine‑power grid ----------------------------------------------------
    powers = np.array([0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 7.0])
    for p in powers:
        if time.time() > deadline:
            break
        seq = _cosine_power(n, p)
        _keep_best(best, _objective(seq), seq, top_k)

    # ----- geometric grid -------------------------------------------------------
    rs = np.linspace(0.70, 0.99, 16)
    for r in rs:
        if time.time() > deadline:
            break
        seq = _geom(n, r)
        _keep_best(best, _objective(seq), seq, top_k)
        seq2 = _sym_geom(n, r)
        _keep_best(best, _objective(seq2), seq2, top_k)

    # ----- power‑law grid -------------------------------------------------------
    ps = np.linspace(0.0, 4.5, 15)
    for p in ps:
        if time.time() > deadline:
            break
        seq = _power_law(n, p)
        _keep_best(best, _objective(seq), seq, top_k)

    # ----- random Dirichlet draws ------------------------------------------------
    concs = [0.002, 0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 30 if conc < 0.01 else 15
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            _keep_best(best, _objective(seq), seq, top_k)

    # ----- sparse random supports -----------------------------------------------
    for _ in range(200):
        if time.time() > deadline:
            break
        k = rng.integers(3, 41)           # 3 … 40 non‑zero entries
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = weights
        _keep_best(best, _objective(seq), seq, top_k)

    # ----- convex combinations of current elites ---------------------------------
    # Ensure we have at least two elites before mixing.
    elites = [seq for _, seq in best]
    if len(elites) >= 2:
        for _ in range(100):
            if time.time() > deadline:
                break
            i, j = rng.choice(len(elites), size=2, replace=False)
            lam = rng.random()
            mixed = lam * elites[i] + (1.0 - lam) * elites[j]
            mixed = mixed / mixed.sum()
            _keep_best(best, _objective(mixed), mixed, top_k)

    # ----- previous solution (if any) -------------------------------------------
    if previous is not None and previous.shape == (n,):
        seq = previous / previous.sum()
        _keep_best(best, _objective(seq), seq, top_k)

    # Return just the sequences, sorted by objective.
    best.sort(key=lambda x: x[0])
    return [seq for _, seq in best[:top_k]]

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    # --------------------------------------------------------------
    # Choose a length that balances resolution and speed.
    # --------------------------------------------------------------
    n = 5000
    print(f"\n=== Starting search (n={n}, budget={budget_s}s) ===")

    # --------------------------------------------------------------
    # Phase 0 – systematic generation of diverse candidates.
    # --------------------------------------------------------------
    phase0_deadline = min(deadline, start_time + 0.15 * budget_s)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    print("\n--- Phase 0: deterministic families + random seeds ---")
    candidates = _collect_candidates(
        n, rng, previous, top_k=20, deadline=phase0_deadline
    )
    print(f"Collected {len(candidates)} top candidates (objective values):")
    for i, seq in enumerate(candidates):
        print(f"  #{i+1:02d}: {_objective(seq):.6f}")

    # --------------------------------------------------------------
    # Phase 1 – short Adam refinements on a few elite candidates.
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    short_adam_iters = 250_000
    print("\n--- Phase 1: short Adam refinements (max_iters={}) ---".format(
        short_adam_iters))
    for idx, cand in enumerate(candidates[:8]):          # only the very best
        if time.time() > deadline:
            break
        print(f"[Candidate {idx+1}/{len(candidates[:8])}]")
        seq_adam, val_adam = _adam_optimize(
            cand, n, deadline, max_iters=short_adam_iters
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after short Adam: {best_val:.6f}")

    if best_seq is None:
        # fallback – use the absolute best raw candidate
        best_seq = candidates[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to raw best candidate: {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long Adam optimisation on the current best.
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Phase 2: long Adam optimisation (until deadline) ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            deadline,
            max_iters=2_500_000,          # generous upper bound; stops by time
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after long Adam: {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (exact max).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: projected sub‑gradient descent ---")
        seq_sub, val_sub = _subgrad_optimize(
            best_seq, n, deadline, max_iters=800_000
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated global best after sub‑grad: {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing polishing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline, max_iters=400_000
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated global best after SA: {best_val:.6f}")

    # --------------------------------------------------------------
    # Final sanity check using the official evaluator.
    # --------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # store for possible reuse and return a plain Python list
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
