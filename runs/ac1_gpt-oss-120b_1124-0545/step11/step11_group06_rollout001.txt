"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Deterministic seeding** – generate a rich set of analytic families
   (uniform, triangular, trapezoidal, sine, Tukey, power‑law, geometric,
   Beta, symmetric‑geometric) plus a large number of random Dirichlet
   vectors.  Keep the best *pop_size* candidates (probability vectors).

2. **Genetic optimisation** – maintain a population of size ``pop_size``.
   Repeatedly create offspring by crossover (random mask) and mutation
   (partial Dirichlet remix).  After each offspring is evaluated the
   worst individual is discarded if the child is better.  An adaptive
   mutation strength ``alpha`` is exponentially decayed.  The loop runs
   until the time budget leaves a small reserve for final polishing.

3. **Local polishing** – run a few thousand steps of the Adam optimiser
   (smooth‑max surrogate) on the current best vector, followed by a
   short projected sub‑gradient descent and a lightweight simulated
   annealing phase.

All sequences are normalised to unit ℓ₁‑norm; the objective is
scale‑invariant, so the true score returned by ``evaluate_sequence`` is
simply ``2·n·max(conv)``.  The best sequence is stored in the global
variable ``height_sequence_1`` for reuse by subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities (convolution, power‑of‑2 helper)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the target objective for a *probability* vector.
    Returns 2·n·max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (used in sub‑gradient descent)
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers – Adam on smooth‑max surrogate, projected sub‑gradient,
# and a lightweight simulated‑annealing polishing.
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 800_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Operates in the logits space of a soft‑max, so the simplex constraint
    is always satisfied.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # initial smooth‑max temperature
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = np.exp(v - np.max(v))
        a = a / a.sum()                 # soft‑max → simplex

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # increase temperature (approach true max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # smooth‑max weights for the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a simple Polyak step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 200_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Very lightweight simulated‑annealing: repeatedly re‑allocate the total
    mass of a small random subset of coordinates (Dirichlet move).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 14)                     # size of the moved subset
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Candidate generation (deterministic families + random seeds)
# ----------------------------------------------------------------------


def _collect_initial_candidates(n: int, rng: np.random.Generator,
                               previous: np.ndarray | None,
                               top_k: int,
                               deadline: float) -> list[np.ndarray]:
    """
    Generate a diverse pool of analytic and random candidates.
    Returns the ``top_k`` vectors with smallest exact objective.
    """
    best = []   # list of (value, seq)

    def keep(val, seq):
        if len(best) < top_k:
            best.append((val, seq))
            best.sort(key=lambda x: x[0])
        elif val < best[-1][0]:
            best[-1] = (val, seq)
            best.sort(key=lambda x: x[0])

    # ----- deterministic families ---------------------------------------
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, 0.12),
        lambda n: _trapezoidal(n, 0.25),
        lambda n: _trapezoidal(n, 0.45),
        lambda n: _sine_window(n, 0.8),
        lambda n: _sine_window(n, 1.2),
        lambda n: _sine_window(n, 2.0),
        lambda n: _sine_window(n, 3.0),
        lambda n: _tukey_window(n, 0.2),
        lambda n: _tukey_window(n, 0.5),
        lambda n: _tukey_window(n, 0.8),
        lambda n: _power_law(n, 0.3),
        lambda n: _power_law(n, 0.7),
        lambda n: _power_law(n, 1.5),
        lambda n: _geom(n, 0.7),
        lambda n: _sym_geom(n, 0.7),
    ]
    for fn in families:
        if time.time() > deadline:
            break
        seq = fn(n)
        keep(_objective(seq), seq)

    # ----- Beta shape grid (coarse) ------------------------------------
    alphas = np.linspace(0.2, 5.0, 15)
    betas = np.linspace(0.2, 5.0, 15)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            seq = _beta_shape(n, a, b)
            keep(_objective(seq), seq)

    # ----- Random Dirichlet seeds (various concentrations) -------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 60 if conc < 0.01 else 30
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            keep(_objective(seq), seq)

    # ----- Sparse random supports ---------------------------------------
    for _ in range(200):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)           # 3 … 30 non‑zero entries
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = weights
        keep(_objective(seq), seq)

    # ----- Previous solution (if any) -----------------------------------
    if previous is not None and previous.shape == (n,):
        seq = previous / previous.sum()
        keep(_objective(seq), seq)

    # Return only the sequences (sorted increasingly by objective)
    return [seq for _, seq in best]


# ----------------------------------------------------------------------
# Genetic operators
# ----------------------------------------------------------------------


def _crossover(parent1: np.ndarray, parent2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random mask crossover – returns a child on the simplex."""
    mask = rng.random(parent1.shape[0]) < 0.5
    child = np.where(mask, parent1, parent2)
    s = child.sum()
    if s == 0.0:
        # degenerate case – fall back to uniform
        child = _uniform(len(parent1))
    else:
        child = child / s
    return child


def _mutate(seq: np.ndarray, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """
    Partial Dirichlet remix: keep a fraction (1‑α) of the current shape,
    replace the remaining α with a fresh Dirichlet sample.
    """
    if alpha <= 0.0:
        return seq
    rand = rng.dirichlet(np.ones_like(seq))
    new = (1.0 - alpha) * seq + alpha * rand
    s = new.sum()
    if s == 0.0:
        return _uniform(len(seq))
    return new / s

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimizes
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Choose a resolution that balances quality and speed.
    n = 4000
    print(f"\n=== Starting search (n={n}, budget={budget_s}s) ===")

    # ------------------------------------------------------------------
    # Phase 0 – deterministic & random seed sweep (produces a pool)
    # ------------------------------------------------------------------
    phase0_deadline = min(deadline, start_time + 0.15 * budget_s)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    print("\n--- Phase 0: candidate generation ---")
    candidate_pool = _collect_initial_candidates(
        n, rng, previous, top_k=50, deadline=phase0_deadline
    )
    print(f"Collected {len(candidate_pool)} candidates after Phase 0.")

    # ------------------------------------------------------------------
    # Initialise population (elitist GA)
    # ------------------------------------------------------------------
    pop_size = 30
    elite_size = 5
    # evaluate the pool and keep the best ``pop_size`` vectors
    pop = []
    for seq in candidate_pool:
        val = _objective(seq)
        pop.append((val, seq))
    pop.sort(key=lambda x: x[0])
    pop = pop[:pop_size]

    best_val, best_seq = pop[0]
    print(f"Initial best objective: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Genetic optimisation (major part of the budget)
    # ------------------------------------------------------------------
    print("\n--- Phase 1: genetic optimisation ---")
    # GA hyper‑parameters
    offsprings_per_round = 25
    mutation_alpha = 0.25            # initial remix weight
    min_alpha = 0.01
    decay = 0.995                    # geometric decay per round

    # Reserve a small chunk of time for the final polishing phases
    reserve_seconds = 120.0

    round_counter = 0
    while time.time() < deadline - reserve_seconds:
        round_counter += 1
        for _ in range(offsprings_per_round):
            # parent selection – pick two parents from the current population
            # (uniformly from the elite half improves exploitation)
            idx1 = rng.integers(pop_size // 2)
            idx2 = rng.integers(pop_size // 2)
            p1 = pop[idx1][1]
            p2 = pop[idx2][1]

            child = _crossover(p1, p2, rng)
            child = _mutate(child, rng, mutation_alpha)

            child_val = _objective(child)

            # If child beats the worst individual, replace it
            if child_val < pop[-1][0]:
                pop[-1] = (child_val, child)
                pop.sort(key=lambda x: x[0])
                # Update global best
                if child_val < best_val:
                    best_val, best_seq = child_val, child.copy()
        # decay mutation strength
        mutation_alpha = max(min_alpha, mutation_alpha * decay)

        if round_counter % 10 == 0:
            print(f"[GA] round {round_counter:3d} – best={best_val:.6f} "
                  f"α={mutation_alpha:.4f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement on the global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: Adam refinement ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq, n, deadline, max_iters=800_000
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq, best_val = seq_adam.copy(), val_adam
            print(f"  Adam improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: sub‑gradient descent ---")
        seq_sub, val_sub = _subgrad_optimize(
            best_seq, n, deadline, max_iters=400_000
        )
        if val_sub < best_val:
            best_seq, best_val = seq_sub.copy(), val_sub
            print(f"  Sub‑grad improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline, max_iters=200_000, init_temp_factor=0.04
        )
        if val_sa < best_val:
            best_seq, best_val = seq_sa.copy(), val_sa
            print(f"  SA improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective (official) ≈ {final_score:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_score = best_val   # fall back to internal estimate

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
