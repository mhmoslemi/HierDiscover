        # optionally run Adam

            print(...)  

        # try random permutations of adam result (including seed result)

"""
Propose a step‑function (non‑negative vector) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm is a multi‑stage search:

1.   Build a rich pool of analytic “seed” shapes (uniform,
     triangular, sine, Gaussian, Beta, Laplace, mixtures, Dirichlet draws,
     previous best, …) and ℓ₁‑normalise them.
2.   Keep the best few seeds (by the surrogate objective 2·n·max(conv)).
3.   Starting from the overall best seed, run a long
     exponentiated‑gradient (mirror‑subgradient) descent on the exact
     max‑convolution objective.
4.   Apply a permutation‑local‑search (swap‑based hill‑climbing) that
     tries to reduce the peak by re‑ordering the entries.
5.   Run a second mirror descent to polish the reordered vector.
6.   Finish with an extensive simulated‑annealing (SA) polish that uses
     swap, mass‑transfer and Dirichlet‑redistribution moves.
7.   Return the best vector (list of floats) and store it as the global
     name ``height_sequence_1`` for possible reuse.

All steps respect the global time budget (default 1000 s).  The
implementation uses only NumPy and the supplied ``evaluate_sequence``.

# Basic convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate (the numerator of evaluate_sequence when sum(seq)=1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Re‑order as large, small, 2nd‑large, 2nd‑small, … and renormalise."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    # renormalise to keep ℓ₁‑norm = 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    """Two‑sided exponential centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.07, offset_frac: float = 0.35) -> np.ndarray:
    """
    Two narrow Gaussians placed symmetrically around the centre.
    offset_frac ∈ (0, 0.5) determines the distance from the centre.
    """
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

            seeds.append(normalised)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,

    Adam on the log‑softmax parameters using a temperature‑controlled

    (already ℓ₁‑normalised) and its surrogate value.
    """

    lr0 = 0.8

    beta_inc = 0.5
    beta_max = 300.0

    best_val = float("inf")

        it += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the true
    max‑convolution.  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.30          # a slightly larger start step size than the reference

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # Sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional aggressive re‑ordering
        if it % 60_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {val_alt:.6f}")

    return best_seq, best_val


def _perm_hill_climb(start_seq: np.ndarray, n: int,
                     deadline: float,
                     max_swaps: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Simple swap‑based hill‑climbing on the permutation of the entries.
    Tries to lower the max‑convolution peak while keeping the multiset
    of values unchanged.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    it = 0
    # a very mild temperature for occasional uphill moves
    T0 = cur_val * 0.005
    T = T0

    while it < max_swaps and time.time() < deadline:
        it += 1
        # geometric cooling
        T = max(T * 0.9999, 1e-12)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 20_000 == 0:
                    print(f"[Perm] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 3_000_000,
                                 init_temp_factor: float = 0.02,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polish with three move types.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a small random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence (list of floats) and stores it in the global
    variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Length – 4000 offers a good trade‑off between granularity and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best, if any, and normalise it.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Phase 1 – generate a diverse seed pool.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise the global best with the very best seed.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – quick Adam refinements on the top seeds.
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest number of Adam iterations based on remaining time
        max_iters = int(min(120_000, remaining * 40))   # roughly 40 iters/sec
        max_iters = max(5_000, max_iters)
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam.copy(), val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # Try the alternating permutation of the Adam result
        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq, global_best_val = alt, alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – long mirror‑descent (sub‑gradient) optimisation.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=5_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror.copy(), val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – permutation hill‑climbing (re‑ordering only).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation hill‑climbing ---")
        seq_perm, val_perm = _perm_hill_climb(
            global_best_seq, n, deadline, max_swaps=80_000
        )
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[overall‑Perm] new global best = {val_perm:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – second mirror descent (fine polishing after re‑ordering).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Second mirror‑descent polishing ---")
        seq_m2, val_m2 = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=2_000_000
        )
        if val_m2 < global_best_val:
            global_best_seq, global_best_val = seq_m2.copy(), val_m2
            print(f"[overall‑Mirror2] new global best = {val_m2:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – simulated‑annealing polishing (final refinement).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=3_000_000,
            init_temp_factor=0.02,
            min_temp=1e-12
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final sanity‑check with the official evaluator.
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
