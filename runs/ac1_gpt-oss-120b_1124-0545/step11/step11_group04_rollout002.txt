def smooth_obj_and_grad(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    shift = max_conv  # subtract for stability
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.exp(beta * (conv - shift)).sum()
    # objective value:
    val = 2.0 * n * (shift + np.log(sum_exp) / beta)
    # gradients:
    w = exp_beta / sum_exp
    # derivative w.r.t a:
    # gradient_a = 2 * conv_fft(a_rev, w)
    rev_a = a[::-1]
    grad_a = 2.0 * _conv_fft(rev_a, w)
    # convert to gradient wrt logits:
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return val, grad_v

def _lbfgs_smooth_optimize(start_vec, n, deadline, beta_schedule, max_iter_per_beta):
    v = np.log(start_vec + 1e-30)
    for beta in beta_schedule:
        # define closure f(v):
        def f(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_conv = np.max(conv)
            shift = max_conv
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            val = 2.0 * n * (shift + np.log(sum_exp) / beta)
            w = exp_beta / sum_exp
            # gradient w.r.t a:
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]
            # gradient w.r.t v:
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return val, grad_v
        # Use SciPy minimize
        maxiter = max_iter_per_beta.get(beta, 500)
        # define callback
        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

            res = scipy.optimize.minimize(fun=f, x0=v, method='L-BFGS-B', jac=True,

        except KeyboardInterrupt:
            print(f"[L-BFGS] Time out at beta={beta}")
            break
    # final vector:
    best_a = _softmax(v)
    return best_a

"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the full linear convolution.
The algorithm:

1. Build a large, diverse pool of ℓ₁‑normalised “seed’’ vectors
   (analytic windows, random Dirichlet/exponential draws, low‑frequency
   cosine curves, mixtures, plus the previously‑found solution).

2. Keep the top few seeds according to a cheap surrogate
   `2·n·max(conv)` (the true objective for ℓ₁‑normalised vectors).

3. For each top seed run a **continuation L‑BFGS** optimiser:
   * the objective is a smooth approximation of the max,
     `log‑sum‑exp_beta(conv)/beta` (β starts small and is increased).
   * Gradients are obtained analytically via convolution
     (the same formulas as in the original code).
   * L‑BFGS is called via SciPy with a time‑out callback.

4. Take the best result, optionally polish it with a few
   sub‑gradient / exponentiated‑gradient passes.

5. If time permits, up‑sample the vector (linear interpolation) to a
   finer grid (2× points) and repeat the continuation L‑BFGS
   optimisation.  Finer discretisation often yields a lower true bound.

All stages respect an overall deadline (default 1000 s).  The final
vector is stored in the global name ``height_sequence_1`` and returned
as a plain list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    # fall‑back implementation if SciPy is unavailable
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # using numpy’s modified Bessel function (approx)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a large, diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian families
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    # Power‑law
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise everything robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimises the smoothed max‐convolution objective using L‑BFGS‑B.
    Returns the ℓ₁‑normalised vector (softmax of the logits).
    """
    if _opt is None:          # pragma: no cover
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        # --- objective & gradient for a given beta --------------------
        def fun(v_local):
            a = _softmax(v_local)                     # probability simplex
            conv = _conv_fft(a, a)                     # full autocorrelation
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smoothed objective (log‑sum‑exp)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # weights w_i = softmax(beta * conv_i)
            w = exp_beta / sum_exp

            # grad wrt a : 2 * conv(rev_a, w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # grad wrt logits via Jacobian of softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # --- callback to enforce the global deadline -----------------
        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    # final vector on the simplex
    best_a = _softmax(v)
    return best_a

# Sub‑gradient / exponentiated‑gradient refinements (same as earlier)
# ----------------------------------------------------------------------
def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Upsampling utilities
# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    # ------------------------------------------------------------------
    # Stage 0 – initial resolution
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and cheap surrogate ranking
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap surrogate evaluation (max‑convolution)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"Best surrogate after seeding (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS optimisation on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Current best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all seeds.")
            break
        remaining = deadline - time.time()
        # allocate a fraction of the remaining budget to this seed
        max_iters_per_beta = max(200, int(min(3000, remaining * 0.09 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={max_iters_per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=max_iters_per_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

        # short sub‑gradient polishing on the current global best
        remaining = deadline - time.time()
        if remaining > 2.0:
            max_sub = int(min(300_000, remaining * 0.07 * 1e5))
            seq_sub, val_sub = _subgradient_proj(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubProj] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient refinement (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(300_000, remaining * 0.07 * 1e5))
        if max_exp > 0:
            print("\n--- Exponentiated‑gradient refinement ---")
            seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – up‑sampling to a finer grid (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample_sequence(global_best_seq, new_n)

            # L‑BFGS on the higher resolution
            max_iters_per_beta_up = int(min(2500, time_left * 0.08 * 1e5))
            up_opt = _lbfgs_smooth_optimize(up_seq,
                                            new_n,
                                            deadline,
                                            max_iter_per_beta=max_iters_per_beta_up)
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] new best after up‑sampling = {global_best_val:.6f}")

            # final tiny polishing with sub‑gradient (if any time remains)
            if time.time() < deadline:
                rem = deadline - time.time()
                max_sub2 = int(min(150_000, rem * 0.05 * 1e5))
                if max_sub2 > 0:
                    seq_sub2, val_sub2 = _subgradient_proj(global_best_seq,
                                                          n,
                                                          deadline,
                                                          max_iters=max_sub2)
                    if val_sub2 < global_best_val:
                        global_best_seq = seq_sub2.copy()
                        global_best_val = val_sub2
                        print(f"[Final‑SubProj] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
