def _local_improvement(seq, n, deadline, max_iters=200000, step_scale=0.01):

    cur = seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    # step_scale is relative to total mass (which sum to 1). Typically e.g., 0.01.
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # choose two indices i,j distinct
        i, j = rng.choice(n, size=2, replace=False)
        # compute transfer amount
        delta = cur[j] * step_scale * rng.random()
        if delta == 0:
            continue
        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        # ensure non-negativity
        if new[j] < 0:
            continue
        new = _proj_simplex(new)  # ensure sum=1 (might not needed as we kept sum, but maintain numeric)
        new_val = _objective(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val
                # optionally print progress.

    return best, best_val

"""
Optimiser for the “minimise max‑self‑convolution” objective.

The programme works in five lightweight phases:

1. **Broad analytical seeding** – a very large pool of
   ℓ₁‑normalised families (geometric, power‑law, β‑shapes,
   mixtures, random Dirichlet draws, sparse supports) is sampled.
   All candidates are evaluated with a fast FFT‑convolution and the
   best `top_k` are retained.

2. **Targeted Adam refinement** – each retained seed is refined with a
   smooth‑max surrogate (log‑sum‑exp) of the convolution.  The temperature
   is ramped up to a large value (`β≈2000`) so that the surrogate
   converges to the true maximum.  The optimisation works in the
   log‑softmax (simplex) coordinates and respects the time budget.

3. **Long Adam polishing** – a final long‑run Adam optimisation on the
   single best sequence from phase 2.

4. **Projected sub‑gradient descent** – exact max‑convolution gradients
   are used to squeeze the remaining tiny bits of improvement.

5. **Simple local search** – tiny stochastic mass‑transfer moves on the
   simplex are accepted only when they reduce the exact objective.
   This little “hill‑climbing” phase often pushes the value a few
   10⁻⁶ further down.

All phases check the global deadline (`budget_s` seconds).  The best
sequence found is scaled to the required ℓ₁‑norm `sqrt(2n)` and returned
as a plain Python list.  The sequence is also stored in the global
variable ``height_sequence_1`` so that future runs can reuse it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
    max_k (seq * seq)[k].
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑normalised, non‑negative)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex Σ v_i = 1, v_i ≥ 0.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i ∝ r^i (r < 1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped discrete density."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    return a / a.sum()


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few non‑zero entries (3 – 30)."""
    k = rng.integers(3, 31)
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of a uniform and a narrow Gaussian."""
    lam = rng.random()
    sigma_frac = rng.uniform(0.05, 0.3)
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    g = np.exp(-((i - mu) ** 2) / (2.0 * (sigma_frac * n) ** 2))
    g = g / g.sum()
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

def _collect_diverse_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    top_k: int,
    deadline: float,
) -> list[np.ndarray]:
    """
    Generate a large pool of ℓ₁‑normalised candidates and retain the
    `top_k` best (according to the exact max‑convolution objective).
    """
    top_vals: list[tuple[float, np.ndarray]] = []   # (value, seq)

    def try_insert(val: float, seq: np.ndarray) -> None:
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])

    # deterministic families (always useful)
    for fn in [_uniform, _triangular]:
        seq = fn(n)
        try_insert(_objective(seq), seq)

    # geometric sweep – we go all the way up to r≈0.9999
    for r in np.linspace(0.80, 0.9999, 120):
        seq = _geom(n, r)
        try_insert(_objective(seq), seq)

    # power‑law sweep – fine grid
    for p in np.linspace(0.0, 5.0, 120):
        seq = _power(n, p)
        try_insert(_objective(seq), seq)

    # beta‑shape random draws
    for _ in range(200):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.1, 4.5)
        beta = rng.uniform(0.1, 4.5)
        seq = _beta_shape(n, alpha, beta)
        try_insert(_objective(seq), seq)

    # random Dirichlet draws – many concentration levels
    conc_vals = [0.0005, 0.001, 0.005, 0.02, 0.1, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        draws = 80 if conc < 0.01 else 40
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            try_insert(_objective(seq), seq)

    # sparse random supports
    for _ in range(200):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        try_insert(_objective(seq), seq)

    # mixture of uniform & Gaussian (a useful heuristic)
    for _ in range(200):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        try_insert(_objective(seq), seq)

    # previous solution (if any)
    if previous is not None:
        prev = np.asarray(previous, dtype=np.float64)
        if prev.shape == (n,) and prev.sum() > 0.0:
            prev = prev / prev.sum()
            try_insert(_objective(prev), prev)

    # keep the best `top_k`
    return [seq for _, seq in top_vals]

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    beta_max: float = 2000.0,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on the log‑sum‑exp (smooth‑max) surrogate.
    Returns the best sequence seen (sum = 1) and its exact objective.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        a = _softmax(v)                  # point on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.8f}")

        # increase smooth‑max temperature
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # convolution and soft‑max weights
        conv = _conv_fft(a, a)           # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient w.r.t. a (smooth‑max of the convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)       # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping (stability)
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters (standard)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best sequence encountered and its exact value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = conv[max_idx]               # exact max‑conv (sum = 1)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.8f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Tiny‑move simulated‑annealing on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)                 # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.8f}")

    return best_seq, best_val


def _local_hill_climb(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    transfer_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Very small stochastic mass‑transfer moves.
    Only moves mass from one index to another (keeps the sum exactly 1).
    Accepts a move if it strictly decreases the exact objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        # transfer a small fraction of the mass at j towards i
        delta = cur[j] * transfer_frac * rng.random()
        if delta <= 0.0:
            continue
        if delta > cur[j]:
            delta = cur[j]

        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        # non‑negativity is guaranteed by construction
        new_val = _objective(new)

        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[HillClimb] it={it:7d} best={best_val:.8f}")

    return best_seq, best_val

    # ------------------------------------------------------------------
    # Phase 5 – tiny random hill‑climbing moves
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 5: local hill‑climbing ---")
        seq_hc, val_hc = _local_hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=200_000,
            transfer_frac=0.01,
        )
        if val_hc < best_val:
            best_seq = seq_hc.copy()
            best_val = val_hc
            print(f"  Global best updated (hill‑climb): {best_val:.8f}")

    # ------------------------------------------------------------------
    # Rescale the simplex vector to the required ℓ₁‑norm (√(2n))
    # ------------------------------------------------------------------
    best_seq = best_seq * sqrt2n

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective (official) ≈ {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    # Store for reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
