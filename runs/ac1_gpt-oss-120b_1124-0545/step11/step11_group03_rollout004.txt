def _adam_projected_optimize(f0, n, deadline, max_iters=500000):
    f = f0.copy()
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1=0.9; beta2=0.999
    epsilon=1e-8
    lr=0.5  # adjust later
    best_f = f.copy()
    best_val = _objective_surrogate(best_f)
    it = 0
    while time.time()<deadline and (max_iters is None or it<max_iters):
        conv = _conv_fft(f, f)
        max_val = conv.max()
        # optionally compute early break if conv[max] is decreasing?
        # compute gradient (subgradient)
        # find all indices where conv close to max (within tolerance 1e-12)
        max_idx = np.flatnonzero(conv == max_val)
        # compute subgradient by summing contributions for each idx
        g = np.zeros_like(f)
        for shift in max_idx:
            i_start = max(0, shift - (n-1))
            i_end = min(n-1, shift)
            idx_i = np.arange(i_start, i_end+1)
            rev_idx = shift - idx_i
            g[idx_i] += 2.0 * f[rev_idx]
        # Normalize gradient? Not needed.
        it += 1
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        f = f - step
        # project onto simplex
        f = _proj_simplex(f)
        # compute surrogate
        cur_val = _objective_surrogate(f)
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()
        # optional: reduce lr over time
        if it % 100000 == 0:
            lr *= 0.5

def _adam_optimize(initial, n, deadline, max_iters=None):
   a = initial.copy()
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   lr0 = 0.5  # initial learning rate
   best = a.copy()
   best_val = _objective_surrogate(a)
   it = 0
   while time.time() < deadline and (max_iters is None or it < max_iters):
       # compute conv and gradient
       conv = _conv_fft(a, a)
       max_val = conv.max()
       # find all indices achieving max within tolerance micro
       max_shifts = np.where(np.abs(conv - max_val) < 1e-12)[0]
       g = np.zeros_like(a)
       for shift in max_shifts:
           i_start = max(0, shift - (n - 1))
           i_end = min(n - 1, shift)
           idx = np.arange(i_start, i_end + 1)
           rev = shift - idx
           g[idx] += 2.0 * a[rev]
       # update
       it += 1
       m = beta1 * m + (1 - beta1) * g
       v = beta2 * v + (1 - beta2) * (g ** 2)
       m_hat = m / (1 - beta1 ** it)
       v_hat = v / (1 - beta2 ** it)
       step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
       a = a - step
       a = _proj_simplex(a)
       cur_val = _objective_surrogate(a)
       if cur_val < best_val:
           best_val = cur_val
           best = a.copy()
       # optionally adapt lr schedule
       if it % 200000 == 0:
           lr0 *= 0.5
           print(f"[Adam] iter={it} best={best_val:.6f}")
   return best, best_val
```

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Rich seed pool** – deterministic analytic windows, systematic
   sweeps of Beta, Gaussian, power‑law, sine‑window, Kaiser and Tukey
   families, random Dirichlet draws, exponential draws, low‑frequency
   cosine curves and sparse random spikes.

2. **Rapid surrogate ranking** – each seed is ℓ₁‑normalised (sum = 1)
   and its surrogate 2·n·max(conv) is computed with an FFT‑based full
   convolution.  The best *k* seeds are retained.

3. **Projected Adam optimisation** – starting from the best seed,
   minimise the true non‑smooth objective using the sub‑gradient of the
   active autocorrelation shift, a Adam‑style adaptive step size and a
   projection onto the probability simplex after every step.

4. **Optional up‑sampling** – if time permits the best vector is
   interpolated to a larger length (≈2×) and refined again; the finer
   discretisation can lower the objective further.

The routine respects the given time budget (default 1000 s) and stores
the final sequence in the global name ``height_sequence_1`` for possible

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import kaiser as _scipy_kaiser
        a = _scipy_kaiser(n, beta)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a higher resolution, then ℓ₁‑normalise."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq /= s
    return new_seq

def _adam_optimize(initial: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the non‑smooth objective.
    Returns the best vector seen and its surrogate value.
    """
    a = initial.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5            # base learning rate
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        shift = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, shift - (n - 1))
        i_end = min(n - 1, shift)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = shift - idx
        g[idx] = 2.0 * a[rev]

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _proj_simplex(a)

        if it % 200_000 == 0:

            print(f"[Adam] it={it:,} best surrogate={best_val:.6f}")

    return best_seq, best_val

        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
