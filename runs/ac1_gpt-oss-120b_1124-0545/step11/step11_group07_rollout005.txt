import scipy.sparse as sp
import scipy.optimize as opt

def lp_step(f, M):
    n = len(f)
    rows = 2*n - 1
    # Build matrix once per iteration
    A = sp.diags([f]*n, offsets=range(n), shape=(rows, n), format='csc')
    b_ub = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64) # maximize sum(b)
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None, None
    b = res.x
    sumb = b.sum()
    return b, sumb

"""
Propose a low‑value step‑function.

The algorithm consists of three phases:

1. **Frank‑Wolfe style linear‑programming reduction**
   – Starting from a simplex‑normalised vector `f` we compute
     `M = max(conv(f,f))`.  We then solve the LP
       max  Σ b_j
       s.t.  b ≥ 0 ,  conv(f,b) ≤ M.
     The optimal `b` increases the total mass while keeping all
     convolution entries ≤ the current maximum.
     A convex combination `f ← (1‑t)·f + t·(b/Σb)` (with a small step
     size `t`) strictly reduces `M`.  We repeat this step a few
     times (≈ 15) – each iteration cheap ( ≈ 0.1 s) and rapidly
     drives the objective down.

2. **Heavy Adam optimisation**
   – On the best vector from phase 1 we run Adam on a smooth
     log‑sum‑exp surrogate of the max‑convolution.  The temperature
     (smooth‑max parameter) is slowly increased to a large value,
     guaranteeing convergence to the true max.  The run is long
     (up to the remaining budget) and keeps the best exact point
     found.

3. **Projected sub‑gradient polishing**
   – The exact max‑convolution objective is finally refined by a
     projected sub‑gradient method with a Polyak step size.  A few
     hundred thousand iterations squeeze the last bits of improvement.

All stages obey the global time budget (default 1000 s).  The final
sequence is scaled to the required ℓ₁‑norm √(2 n) and returned as a
plain Python list.  The best sequence is cached in the global variable
`height_sequence_1` for reuse by later runs.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a simplex‑normalised sequence (sum = 1):
    max(conv(seq, seq)).
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _uniform(n: int) -> np.ndarray:
    """Uniform simplex‑vector of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Linear‑programming step (Frank–Wolfe)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1)×n matrix A such that A @ b = conv(f, b).
    Row k, column i contains f_{k‑i} when the index is in range,
    otherwise 0.  Implemented with a stack of diagonals.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # the same vector `f` appears on each diagonal with offset = column index
    A = sp.diags([f] * n, offsets=range(n), shape=(rows, n), format="csc")
    return A


def _lp_step(f: np.ndarray, M: float) -> tuple[np.ndarray, float] | tuple[None, None]:
    """
    Solve   max Σ b_j   s.t.   b ≥ 0 ,   conv(f, b) ≤ M .
    Returns the optimal b and its sum.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                 # (2n‑1) × n   sparse
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)         # maximise Σ b_j  → minimise –Σ b_j
    bounds = [(0, None)] * n

    # use the Highs interior‑point LP solver (sparse aware)
    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None, None
    b = res.x
    return b, b.sum()


# ----------------------------------------------------------------------
# Adam optimiser on a smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    beta_max: float = 500.0,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation of the exact max‑convolution objective
    via a log‑sum‑exp smooth approximation.
    Returns (best_sequence, best_exact_value).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        a = _softmax(v)               # current point on simplex (∑=1)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.9f}")

        # increase temperature (smooth‑max becomes sharper)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # convolution and smooth‑max weighting
        conv = _conv_fft(a, a)          # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum                    # softmax over convolution entries

        # gradient of Σ w_k·conv_k w.r.t. a  (see derivation in original code)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)       # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping (stability)
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution
    objective.  Returns the best sequence encountered and its exact
    value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]          # same scaling used later
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.9f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (using best value seen so far)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # 0.5 s safety margin

    # Length of the step‑function – a moderate size gives fast LP solves.
    n = 3000
    sqrt2n = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise from a previous run if available, otherwise uniform.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                f = prev_arr / prev_arr.sum()
            else:
                raise ValueError
        except Exception:
            f = _uniform(n)
    else:
        f = _uniform(n)

    best_f = f.copy()
    best_val = _objective(f)
    print(f"\n[Init] n={n}, initial objective = {best_val:.9f}")

    # ------------------------------------------------------------------
    # Phase 1 – Frank‑Wolfe LP reduction (≈10 % of budget)
    # ------------------------------------------------------------------
    max_lp_iters = 15
    lp_deadline = min(deadline, start_time + budget_s * 0.12)
    t_step = 0.15                     # step size for convex combination

    print("\n--- Phase 1: LP‑based reduction ---")
    for lp_iter in range(max_lp_iters):
        if time.time() > lp_deadline:
            break
        M = _objective(f)             # current max‑convolution
        b, sumb = _lp_step(f, M)
        if b is None or sumb is None:
            print("[LP] Solver failed – exiting LP phase")
            break
        if sumb <= 1.0 + 1e-12:
            # cannot increase mass any more → stop
            print("[LP] No mass increase possible – terminating LP phase")
            break
        g = b / sumb                  # normalise to sum = 1
        # convex update; keep t modest to ensure monotone decrease
        f = (1.0 - t_step) * f + t_step * g
        f = f / f.sum()               # re‑project onto simplex (numerical safety)

        cur_val = _objective(f)
        if cur_val < best_val:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline:

        # allocate the bulk of the remaining budget to Adam

        seq_adam, val_adam = _adam_optimize(
            start_vec=best_f,
            n=n,
            deadline=deadline,
            max_iters=max_iters_adam,
        )
        if seq_adam is not None and val_adam < best_val:
            best_f = seq_adam.copy()
            best_val = val_adam
            print(f"[Adam] improved best to {best_val:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent for final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: Sub‑gradient polishing ---")
        seq_sub, val_sub = _subgrad_optimize(
            start_seq=best_f,
            n=n,
            deadline=deadline,

        )
        if val_sub < best_val:
            best_f = seq_sub.copy()
            best_val = val_sub

    # ------------------------------------------------------------------
    # Scale to the required ℓ₁‑norm √(2 n) and final sanity check
    # ------------------------------------------------------------------
    final_seq = best_f * sqrt2n

    try:
        final_val = evaluate_sequence(final_seq.tolist())
        print(f"\n=== Finished ===\nOfficial objective ≈ {final_val:.9f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    # Cache for possible reuse.
    globals()["height_sequence_1"] = final_seq.tolist()
    return final_seq.tolist()
