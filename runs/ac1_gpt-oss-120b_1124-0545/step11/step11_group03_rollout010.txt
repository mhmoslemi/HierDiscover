"""
search for a non‑negative step function that minimises `evaluate_sequence`.

Approach
--------
* Choose a moderately large length (default 8000).  The theoretical lower
  bound of the objective is 1, so a longer sequence gives more freedom to
  flatten the self‑convolution.
* Build a rich pool of ℓ₁‑normalised seed vectors (analytic windows,
  beta‑shapes, Dirichlet draws, low‑frequency curves, and the previous
  solution if available).
* Keep the few best seeds (by the fast FFT‑based surrogate
  ``2·n·max(conv)``) and start a long simulated‑annealing run from the
  best one.  Moves consist of
    – reshuffling the mass of a random subset (Dirichlet redistribution);
    – transferring a small chunk between two positions;
    – swapping two entries.
* The temperature is slowly reduced (geometric cooling).  Every accepted
  move is projected back onto the probability simplex to keep the sum equal
  to 1.
* The annealing runs until the global deadline (default 1000 s) is reached.
* Finally the best candidate is evaluated with the official ``evaluate_sequence``
  function and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Collect a diverse set of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_tukey_window(n, 0.2))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_tukey_window(n, 0.8))

    for sigma in np.linspace(0.12, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any) – pad or truncate to the current length
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size < n:
            padded = np.zeros(n, dtype=np.float64)
            padded[:arr.size] = arr
            arr = padded
        elif arr.size > n:
            arr = arr[:n]
        s = arr.sum()
        if s > 0:
            seeds.append(arr / s)

    # ensure proper normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s <= 0:
            cleaned.append(_uniform(n))
        else:
            cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Simulated annealing (aggressive)
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_seq: np.ndarray,
    deadline: float,
    init_temp_factor: float = 0.07,
    cooling: float = 0.99997,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Aggressive annealing that reshuffles mass on random subsets,
    transfers small chunks and occasionally swaps two entries.
    Returns the best sequence seen and its surrogate value.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature proportional to the current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while time.time() < deadline:
        it += 1
        T = max(T * cooling, min_temp)

        move_type = rng.random()
        if move_type < 0.55:                     # reshuffle a random subset
            k = rng.integers(3, 15)             # subset size 3 … 14
            idx = rng.choice(len(cur), size=k, replace=False)
            total = cur[idx].sum()
            if total > 0.0:
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
                new = cur.copy()
                new[idx] = new_alloc
            else:
                continue
        elif move_type < 0.85:                  # transfer a chunk between two positions
            i, j = rng.choice(len(cur), size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.08)  # at most 8 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps
        else:                                   # swap two entries
            i, j = rng.choice(len(cur), size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]

        # re‑normalise (the moves preserve the sum except possible rounding)
        s = new.sum()
        if s > 0:
            new = new / s
        else:
            continue

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    # --------------------------------------------------------------
    # Length selection – 8000 gives more degrees of freedom while staying fast.
    # --------------------------------------------------------------
    n = 8000
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)

            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)

    # --------------------------------------------------------------

    scored = [(seq, _objective_surrogate(seq)) for seq in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # start from the overall best seed
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate after seeding = {best_val:.6f}")

    # --------------------------------------------------------------
    # Simulated annealing – the heavy‑lifting phase.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing (main optimisation) ---")
        best_seq, best_val = _simulated_annealing(best_seq, deadline)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Optional short polishing with a few gradient steps.
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(200_000, remaining * 2e4))

        a = best_seq.copy()
        cur_best = best_seq.copy()
        cur_best_val = best_val

        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break
            # compute convolution and locate the active shift
            conv = _conv_fft(a, a)
            k = int(np.argmax(conv))

            # sub‑gradient for the active shift (as in the reference code)
            g = np.zeros_like(a)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] = 2.0 * a[rev_idx]

            # step‑size schedule – a slightly larger start than the reference
            lr = 0.30 / np.sqrt(it)
            a = a - lr * g
            a = _proj_simplex(a)

            val = _objective_surrogate(a)
            if val < cur_best_val:
                cur_best_val = val
                cur_best = a.copy()
                if it % 50000 == 0:
                    print(f"[Grad] it={it:7d} best={val:.6f}")

        if cur_best_val < best_val:
            best_seq, best_val = cur_best, cur_best_val
            print(f"[Grad] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
