Propose a non‑negative step‑function (probability vector) that improves

1.  A modest pool of analytic seeds (uniform, triangular, geometric, …)
    plus the previous best solution (if any).  The best seed (by the fast
    surrogate 2·n·max(conv)) is selected.

2.  Adam‑style gradient descent on a *smooth* approximation of the
    max‑convolution (log‑sum‑exp with a temperature α).  The distribution
    is represented by logits and transformed with a soft‑max to stay on the
    probability simplex.  Gradients are obtained by a single FFT‑based
    cross‑correlation per iteration (≈ O(n log n)).

3.  A short simulated‑annealing polish that reshuffles mass among a random
    subset of coordinates (Dirichlet move) and accepts uphill moves with a
    cooling schedule.

The routine respects the 1000 s budget, prints intermediate progress and
returns the best sequence found.  The result is stored in the global
variable ``height_sequence_1`` for reuse in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation for ℓ₁‑normalised sequences (sum == 1).
    Returns 2·n·max(convolution) – the quantity we minimise before
    calling the official ``evaluate_sequence``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max producing a probability vector."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    return exp / np.sum(exp)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _interpolate(old: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_len = len(old)
    if old_len == new_len:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new, 0.0)
    s = new.sum()
    return new / s if s > 0 else _uniform(new_len)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12),
                             np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any) interpolated to the new length
    if previous is not None and previous.shape[0] > 0:
        seeds.append(_interpolate(previous, n))

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha: float = 20.0,
    lr: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth approximation of max⁡(a*a).
    The distribution is represented by logits (w) and transformed with
    a soft‑max to stay on the probability simplex.
    Returns the best sequence found (probability vector) and its
    surrogate objective 2·n·max(conv).
    """
    # initialise logits; add a tiny epsilon to avoid log(0)
    w = np.log(start_a + 1e-20)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = _softmax(w)
    best_val = _objective(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(w)                # probability vector, sum=1
        conv = _conv_fft(a, a)         # full convolution, length 2n‑1
        max_conv = np.max(conv)

        # smooth approximation of the max (log‑sum‑exp)
        shifted = conv - max_conv      # for numerical stability
        weight = np.exp(alpha * shifted)
        p = weight / np.sum(weight)    # length 2n‑1, non‑negative, sums to 1

        # gradient w.r.t. a: 2 * cross‑correlation of p and a
        a_rev = a[::-1]
        conv_pa = _conv_fft(p, a_rev)                # length 3n‑2
        start = n - 1
        grad_a = 2.0 * conv_pa[start:start + n]      # size n

        # chain rule through the soft‑max
        avg = np.sum(grad_a * a)
        grad_w = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep the best surrogate seen so far (occasionally report)
        if it % 2000 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] it={it:6d} best surrogate={best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing polish that reshuffles a random
    subset of coordinates via a Dirichlet move.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)                 # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length that balances quality and speed
    # ------------------------------------------------------------------
    n = 6000                     # empirically a good trade‑off

    # ------------------------------------------------------------------
    # Stage 1 – collect a modest set of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(3, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best with the best surrogate seed
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    try:
        global_best_official = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        global_best_official = float("inf")
    print(f"\n[Init] best surrogate = {global_best_val:.6f}, "
          f"official = {global_best_official:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation from the top seeds
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        runs = len(best_seeds)
        # keep a fraction of the remaining budget for polishing
        time_for_optim = 0.85 * time_left
        time_per_run = time_for_optim / runs

        for idx, seed_vec in enumerate(best_seeds):
            run_deadline = time.time() + time_per_run
            # heuristic: about 2500 iterations per second of wall‑clock time
            max_iters = int((run_deadline - time.time()) * 2500)
            max_iters = max(30_000, min(max_iters, 300_000))
            print(f"\n--- Adam optimisation run {idx + 1}/{runs} ---")
            seq_adam, surrogate_val = _adam_optimize(
                seed_vec, n, run_deadline,
                max_iters=max_iters,
                alpha=25.0,
                lr=0.5,
            )
            # evaluate the official objective on the result
            try:
                official_val = evaluate_sequence(seq_adam.tolist())
            except Exception:
                official_val = float("inf")
            print(f"[Adam] run {idx + 1}: surrogate={surrogate_val:.6f}, "
                  f"official={official_val:.6f}")

            if official_val < global_best_official:
                global_best_seq = seq_adam.copy()
                global_best_val = surrogate_val
                global_best_official = official_val
                print(f"   *** New global best (official) = {global_best_official:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, surrogate_val = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=800_000,
            init_temp_factor=0.04,
        )
        try:
            official_val = evaluate_sequence(seq_sa.tolist())
        except Exception:
            official_val = float("inf")
        print(f"[SA] final surrogate={surrogate_val:.6f}, official={official_val:.6f}")
        if official_val < global_best_official:
            global_best_seq = seq_sa.copy()
            global_best_val = surrogate_val
            global_best_official = official_val
            print(f"   *** New global best after SA = {global_best_official:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}\n"
        f"Best official value ≈ {global_best_official:.6f}\n"
        f"Length = {len(global_best_seq)}"
    )

    # store for possible later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
