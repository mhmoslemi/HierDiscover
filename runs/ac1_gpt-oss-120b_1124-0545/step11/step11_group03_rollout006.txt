Propose a step‐function (non‑negative vector) that minimises
evaluate_sequence(sequence).

Algorithm
---------
1. **Rich seed pool** – a large collection of analytic families
   (uniform, triangular, various windows, exponential, power‑law,
   beta, Dirichlet draws, low‑frequency curves).  All seeds are ℓ₁‑
   normalised.

2. **Sparse LP refinement** – for a current shape f compute
   M = max(conv(f,f)).  Solve the linear program

        maximise   1ᵀb
        s.t.       conv(f,b) ≤ M      (for every shift)
                   b ≥ 0

   The convolution constraints are represented by a *single*
   sparse Toeplitz matrix (size (2n‑1)×n).  The LP is solved with

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective (2·n·max(conv(seq,seq))) assuming Σseq = 1.
    The value equals the true objective when Σseq = 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i = r**i, normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform for a little extra diversity
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous best (if present)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final clean‑up – enforce ℓ₁‑norm
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned


# --------------------------------------------------------------
# Sparse LP utilities
# --------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n sparse matrix A where A[k, j] = f[k‑j] when
    0 ≤ k‑j < n, otherwise 0.  This encodes the convolution
    constraints conv(f, b) ≤ M.
    """
    n = len(f)
    m = 2 * n - 1

    # generate all (i, j) pairs, i = row contribution from f, j = column index of b
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)

    # each i repeats n times, each j tiled n times
    I = np.repeat(i, n)          # shape (n*n,)
    J = np.tile(j, n)            # shape (n*n,)

    rows = I + J                 # shift index k = i + j
    cols = J

    # All rows are < 2n‑1 by construction
    data = f[I]

    # Build CSR matrix
    A = csr_matrix((data, (rows, cols)), shape=(m, n), dtype=np.float64)
    return A


def _lp_optimize(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max 1ᵀb   s.t.   conv(f,b) ≤ M,   b ≥ 0 .
    Returns the optimal b (not normalised) or None on failure.
    """
    if linprog is None or csr_matrix is None:                     # pragma: no cover
        return None

    n = len(f)
    A = _build_conv_matrix(f)                     # (2n‑1)×n sparse

    # linprog solves   min cᵀx   s.t.   A_ub x ≤ b_ub,  x ≥ 0.
    c = -np.ones(n, dtype=np.float64)             # maximise sum(b)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception as exc:                       # pragma: no cover
        print("[LP] exception:", exc)
    return None


def _lp_refine(seq: np.ndarray, n: int, deadline: float,
               max_outer: int = 12,
               init_mix: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP and mix the result with the current vector.
    Returns the refined vector and its surrogate value.
    """
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    mix_weight = init_mix

    for outer in range(1, max_outer + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(best_seq, best_seq)
        M = np.max(conv)

        b = _lp_optimize(best_seq, M)
        if b is None:
            print("[LP] LP failed – terminating LP phase.")
            break

        s = b.sum()
        if s <= 0.0:
            break
        b_norm = b / s  # ℓ₁‑normalised

        # try a few mixing ratios, decreasing if necessary
        improved = False
        for t in (mix_weight, mix_weight * 0.5, mix_weight * 2.0):
            cand = (1.0 - t) * best_seq + t * b_norm
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            val = _objective_surrogate(cand)
            if val < best_val - 1e-9:
                best_seq, best_val = cand, val
                improved = True
                print(f"[LP] iter={outer:2d} mix={t:.3f} surrogate={val:.6f}")
                break

        if not improved:
            mix_weight *= 0.5
            if mix_weight < 1e-4:
                break
    return best_seq, best_val


# --------------------------------------------------------------
# Sub‑gradient descent (exact max‑convolution gradient)
# --------------------------------------------------------------
def _subgradient_proj(start_seq: np.ndarray, n: int,
                      deadline: float,
                      max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Sub] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# --------------------------------------------------------------
# Simulated annealing polishing
# --------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 150_000,
                         init_temp_factor: float = 0.06,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Aggressive annealing that reshuffles mass on random subsets.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:          # reshape a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    evaluate_sequence objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin

    # ---------------------------
    # Length of the candidate
    # ---------------------------
    n = 3500               # balances expressiveness and LP memory

    # ---------------------------
    # Seed generation
    # ---------------------------
    print(f"\n--- Generating seeds (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Total seeds generated: {len(seeds)}")

    scored = [(seq, _objective_surrogate(seq)) for seq in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"\nTop {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # start from the best seeded shape
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # ---------------------------
    # LP refinement stage
    # ---------------------------
    if time.time() < deadline:
        print("\n--- LP refinement stage ---")
        best_seq, best_val = _lp_refine(best_seq, n, deadline,
                                        max_outer=10, init_mix=0.30)
        print(f"[LP] surrogate after refinement = {best_val:.6f}")

    # ---------------------------
    # Sub‑gradient descent polishing
    # ---------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent polishing ---")
        remaining = deadline - time.time()
        max_iters_sub = int(min(250_000, remaining * 2e5))
        seq_proj, val_proj = _subgradient_proj(best_seq, n, deadline,
                                               max_iters=max_iters_sub)
        if val_proj < best_val:
            best_seq, best_val = seq_proj, val_proj
            print(f"[Post‑Sub] new best surrogate = {best_val:.6f}")

    # ---------------------------
    # Simulated annealing polishing
    # ---------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(150_000, remaining * 2e5))
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              max_iters=max_iters_sa)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[Post‑SA] final surrogate = {best_val:.6f}")

    # --------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                                          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
