   ``Sβ(a)=β⁻¹·log Σ exp(β·conv(a,a))`` using Adam on the log‑softmax

4. **Simulated‑annealing polishing** – Simple SA with three move types

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _obj_normed(seq: np.ndarray) -> float:
    """
    Surrogate objective for a *ℓ₁‑normalised* sequence.
    Returns 2·n·max(conv(seq,seq)).
    """
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    a = seq / s
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve obtained by summing a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # sine windows
    for gamma in np.linspace(0.3, 4.0, 14):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian
    for sigma in np.linspace(0.06, 0.4, 12):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Power‑law
    for p in np.linspace(0.0, 5.0, 12):
        seeds.append(_power(n, p))

    # Beta
    alphas = np.linspace(0.2, 3.5, 9)
    betas = np.linspace(0.2, 3.5, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey
    for alpha in np.linspace(0.05, 0.95, 12):
        seeds.append(_tukey_window(n, alpha))

    # Exponential decay
    for lam in np.linspace(0.0003, 0.02, 12):
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet draws
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # Low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 120_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on the smoothed max (log‑sum‑exp) of the convolution.
    The temperature β is increased up to 800, giving a very tight surrogate.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits (log‑softmax parametrisation)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.85                                 # a bit larger than before
    beta = 0.5
    beta_inc = 0.7
    beta_max = 800.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                       # current point on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)      # true (unnormalised) surrogate

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smooth max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n elements

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_descent(start_seq: np.ndarray, n: int, deadline: float,
                    max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true objective.
    Larger initial step‑size and a slower decay (power = 0.30) give a
    finer final refinement.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _obj_normed(a)

    lr0 = 0.45          # increased step‑size
    power = 0.30        # slower decay

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 2_000_000,
                         init_temp_factor: float = 0.06,
                         min_temp: float = 1e-13) -> tuple[np.ndarray, float]:
    """
    Simple SA with a very slowly cooling temperature.
    Three move types (swap, Dirichlet redistribution, pair‑transfer) are
    employed.  Any improvement resets the temperature to its current
    value, encouraging a thorough exploration of the neighbourhood.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _obj_normed(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-13)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling (extremely slow)
        T = max(T * 0.9999, min_temp)

        move_type = rng.integers(0, 3)   # 0=swap, 1=redistribute, 2=pair‑transfer
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 15)    # size of the subset
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _obj_normed(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _optimise_length(n: int, deadline: float, previous: np.ndarray | None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` within the time window ending at ``deadline``.
    Returns the best sequence found together with its surrogate value.
    """
    rng = np.random.default_rng()
    print(f"\n=== Optimising length {n} ===")
    # ---- seed generation -------------------------------------------------
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # cheap evaluation of all seeds, keep the best few
    evaluated = [(vec, _obj_normed(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # start from the best seed
    global_best_seq = best_seeds[0].copy()
    global_best_val = _obj_normed(global_best_seq)
    print(f"[Init] Best surrogate = {global_best_val:.6f}")

    # ---- Adam optimisation ------------------------------------------------
    # allocate roughly 30 % of the remaining budget to Adam
    remaining = deadline - time.time()
    if remaining > 20.0:  # need at least a few seconds
        adam_deadline = time.time() + 0.30 * remaining
        for idx, seed_vec in enumerate(best_seeds, 1):
            if time.time() > adam_deadline:
                break
            # a modest iteration budget that scales with the time left
            iters = int(min(150_000, (adam_deadline - time.time()) * 80))
            print(f"\n[Adam] run {idx}/{len(best_seeds)} – max_iters={iters}")
            seq_adam, val_adam = _adam_optimize(seed_vec, n, adam_deadline, max_iters=iters)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ---- Mirror‑descent refinement ---------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        mirror_deadline = time.time() + 0.40 * remaining
        print("\n[Mirror] fine‑tuning")
        seq_mirror, val_mirror = _mirror_descent(global_best_seq, n, mirror_deadline,
                                                max_iters=5_000_000)
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ---- Simulated annealing polishing -----------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sa_deadline = time.time() + remaining
        print("\n[SA] polishing")
        seq_sa, val_sa = _simulated_annealing(global_best_seq, n, sa_deadline,
                                              max_iters=3_000_000)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    return global_best_seq, global_best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allotted time budget.
    The function explores several sequence lengths (2000‑5000) and returns
    the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin for the final sanity check
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a modest list of lengths – we start from the smallest,
    # continuing while we still have at least 30 seconds left.
    # ------------------------------------------------------------------
    candidate_lengths = [2000, 3000, 4000, 5000]

    # retrieve previous global solution, if any
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    overall_best_seq = None
    overall_best_val = float("inf")

    for n in candidate_lengths:
        now = time.time()
        if now >= deadline - 30.0:          # leave a safe buffer for later lengths
            print(f"Time low – skipping remaining lengths (next would be {n}).")
            break
        sub_deadline = min(deadline, now + (deadline - now) * 0.9)  # a bit of slack
        seq, val = _optimise_length(n, sub_deadline, previous)
        print(f"[Length {n}] best surrogate = {val:.6f}")

        if val < overall_best_val:
            overall_best_val = val
            overall_best_seq = seq.copy()
            print(f"*** New overall best (length {n}) = {val:.6f}")

        # expose the current leader for the next length (may help warm‑start)
        previous = overall_best_seq.copy()

    # ------------------------------------------------------------------
    # Final official evaluation (optional, just for information)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(overall_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {overall_best_val:.6f} "
        f"(length {len(overall_best_seq)})"
    )

    globals()["height_sequence_1"] = overall_best_seq.tolist()
    return overall_best_seq.tolist()
