def lp_iterate(f, M, n, t):
    # Solve LP to maximize sum(g) under conv(f,g) <= M.
    import cvxpy as cp
    g = cp.Variable(n)
    # Constraints: g >= 0
    constraints = [g >= 0]
    # Build convolution constraints
    # Considering conv length = 2n-1.
    # For each k in 0..2n-2:
    # conv_f_g[k] = sum_i f_i * g_{k-i} (if indices permitted)
    # We'll use loops.
    for k in range(2*n - 1):
        # compute indices i where j = k - i within [0,n-1]
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # Build linear expression sum f[i] * g[k-i]
        expr = 0
        for i in range(i_min, i_max + 1):
            expr += f[i] * g[k - i]
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000) # or other params
    g_opt = g.value
    return g_opt

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]
    # conv constraints
    # Build matrix A (2n-1 x n)
    # Using sparse matrix to speed: create list of entries.
    import scipy.sparse as sp
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        for i in range(i_min, i_max+1):
            rows.append(k)
            cols.append(k - i)  # index of g
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # define constraints: A @ g <= M
    constraints.append(A @ g <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    g_opt = g.value
    return g_opt

"""
Propose a low‑value step function for ``evaluate_sequence``.

The approach combines three ideas:

1. **Linear‑programming refinement on a small scale.**
   Starting from a simple distribution (uniform or a random Dirichlet)
   we solve a linear program that maximises the total mass of a new
   sequence ``g`` while keeping the convolution ``f*g`` below the
   current worst convolution value ``M`` of ``f``.
   After a few alternating updates ``f ← (1‑t)·f + t·g`` the
   maximum of ``f*f`` drops dramatically (the objective becomes the
   maximum convolution when the ℓ¹‑norm is fixed).

2. **Upscaling to the full size (4000) and fine‑tuning** with a
   smooth‑max Adam optimiser (log‑sum‑exp surrogate).  The optimiser
   works in log‑softmax space so the simplex constraint is automatic.

3. **Final polishing** with a cheap projected sub‑gradient descent
   and simulated annealing.  All phases respect the global time budget
   (default 1000 s) and store the best sequence in the global variable
   ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the official objective for a *probability* vector.
    (seq must be non‑negative and sum to 1.)
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a point on the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c * i)  (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * i)
    s = a.sum()
    return a / s


def _gen_normal(n: int, p: float, sigma: float) -> np.ndarray:
    """Generalised normal: a_i ∝ exp(-|i-centre|^p / sigma)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) ** p / sigma)
    s = a.sum()
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Candidate generation (small scale)
# ----------------------------------------------------------------------


def _collect_candidates_small(n: int, rng: np.random.Generator,
                             top_k: int, deadline: float) -> list[np.ndarray]:
    """
    Exhaustively search a handful of analytically tractable families
    (including a few new ones) and keep the ``top_k`` candidates with
    smallest exact objective.
    """
    best = []                     # list of (value, seq)

    def keep(val, seq):
        if len(best) < top_k:
            best.append((val, seq))
            best.sort(key=lambda x: x[0])
        elif val < best[-1][0]:
            best[-1] = (val, seq)
            best.sort(key=lambda x: x[0])

    # ---- classic families -------------------------------------------------
    families = [
        _uniform,
        lambda n: _exp_decay(n, c=0.5),
        lambda n: _exp_decay(n, c=1.0),
        lambda n: _exp_decay(n, c=2.0),
        lambda n: _gen_normal(n, p=2.0, sigma=200.0),
        lambda n: _gen_normal(n, p=4.0, sigma=100.0),
        lambda n: _beta_shape,
        lambda n: _sine_window,
        lambda n: _tukey_window,
    ]

    # systematic scan over a few parameters
    for fn in families:
        if fn is _beta_shape:
            alphas = np.linspace(0.2, 5.0, 12)
            betas = np.linspace(0.2, 5.0, 12)
            for a in alphas:
                for b in betas:
                    seq = _beta_shape(n, a, b)
                    keep(_objective(seq), seq)
        elif fn is _tukey_window:
            alps = np.linspace(0.1, 0.9, 12)
            for a in alps:
                seq = _tukey_window(n, a)
                keep(_objective(seq), seq)
        elif fn is _sine_window:
            gammas = np.linspace(0.5, 5.0, 12)
            for g in gammas:
                seq = _sine_window(n, gamma=g)
                keep(_objective(seq), seq)
        else:
            # plain call (no additional parameters)
            seq = fn(n)
            keep(_objective(seq), seq)

    # ---- random Dirichlet seeds -----------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 30 if conc < 0.01 else 20
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            keep(_objective(seq), seq)

    # ---- sparse supports -------------------------------------------------
    for _ in range(80):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)                # 3 … 30 non‑zero entries
        idx = rng.choice(n, size=k, replace=False)
        w = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = w
        keep(_objective(seq), seq)

    return [seq for _, seq in best]

# Linear‑programming based refinement (small n)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return the (2n‑1)×n matrix A with (A g)[k] = Σ_i f[i]·g[k‑i]."""
    n = f.shape[0]
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(k - i)          # index of g
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_one_step(f: np.ndarray, M: float, solver='ECOS') -> np.ndarray:
    """
    One LP step: maximise Σ g_j subject to g ≥ 0 and (f * g) ≤ M.
    ``f`` is a probability vector (sum = 1).  Returns the raw optimal
    g (not renormalised).
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=solver, warm_start=False, max_iters=2000, verbose=False)
    if g.value is None:
        # fallback – return the current f (should never happen)
        return f.copy()
    return np.maximum(g.value, 0.0)


def _lp_refine(f_start: np.ndarray, iters: int,
               step: float, deadline: float) -> np.ndarray:
    """
    Alternating LP refinement (the method described in the statement).
    The sequence stays on the simplex at all times (sum = 1).
    """
    f = f_start.copy()
    n = f.shape[0]
    for i in range(iters):
        if time.time() > deadline:
            break
        M = np.max(_conv_fft(f, f))
        g_raw = _lp_one_step(f, M)
        s = g_raw.sum()
        if s <= 0.0:
            break
        g = g_raw / s                     # renormalise to sum = 1
        f = (1.0 - step) * f + step * g   # convex combination
    return f

def _upscale(seq_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from ``len(seq_small)`` to ``n_large``."""
    x_small = np.linspace(0.0, 1.0, seq_small.shape[0])
    x_large = np.linspace(0.0, 1.0, n_large)
    seq_large = np.interp(x_large, x_small, seq_small)
    seq_large = np.maximum(seq_large, 0.0)
    total = seq_large.sum()
    if total == 0.0:
        return _uniform(n_large)
    return seq_large / total

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 1_200_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Operates in log‑softmax space, so the simplex constraint is automatic.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # start temperature (smooth‑max)
    beta_mul = 1.001
    beta_max = 500.0         # final temperature – higher is sharper

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # raise temperature slowly
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clipping for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a Polyak step size referencing the current best value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Light‑weight simulated‑annealing with Dirichlet moves on a random
    subset of coordinates.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30000 == 0:
                    print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimizes
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Small‑scale LP phase – find a very good shape on a modest grid.
    # ------------------------------------------------------------------
    n_small = 800                     # manageable size for LP
    print(f"\n=== Phase 0: LP‑based shape discovery (n={n_small}) ===")

    phase0_deadline = min(deadline, start_time + 0.20 * budget_s)
    candidates_small = _collect_candidates_small(
        n_small, rng, top_k=12, deadline=phase0_deadline
    )
    print(f"  generated {len(candidates_small)} deterministic/random seeds")

    # Run a few LP refinement rounds starting from each candidate
    best_small = None
    best_small_val = float("inf")
    lp_iters = 12
    lp_step = 0.25                    # convex‑combination weight

    print("\n--- LP refinement of the small candidates ---")
    for idx, cand in enumerate(candidates_small):
        if time.time() > deadline:
            break
        print(f"[LP] start candidate {idx + 1}/{len(candidates_small)}")
        refined = _lp_refine(cand, iters=lp_iters, step=lp_step,
                             deadline=deadline)
        val = _objective(refined)
        print(f"    candidate {idx + 1} post‑LP objective: {val:.6f}")
        if val < best_small_val:
            best_small_val = val
            best_small = refined.copy()
            print(f"    ** new global best (small) = {best_small_val:.6f}")

    # ------------------------------------------------------------------
    # Upscale to the full size expected by the evaluator.
    # ------------------------------------------------------------------
    n_large = 4000
    print(f"\n=== Upscaling to n={n_large} ===")
    if best_small is None:
        # fallback – use a uniform distribution
        best_small = _uniform(n_small)
        best_small_val = _objective(best_small)

    upscaled = _upscale(best_small, n_large)
    upscaled = np.maximum(upscaled, 0.0)
    upscaled = upscaled / upscaled.sum()   # strict ℓ¹‑normalisation
    print(f"Upscaled objective (pre‑refine): {_objective(upscaled):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam refinement (long run).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Phase 1: long Adam optimisation ===")
        seq_adam, val_adam = _adam_optimize(
            upscaled, n_large, deadline, max_iters=1_300_000
        )
        if seq_adam is not None and val_adam < best_small_val:
            upscaled = seq_adam.copy()
            best_small_val = val_adam
            print(f"  Updated best after Adam: {best_small_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Phase 2: projected sub‑gradient descent ===")
        seq_sub, val_sub = _subgrad_optimize(
            upscaled, n_large, deadline, max_iters=900_000
        )
        if val_sub < best_small_val:
            upscaled = seq_sub.copy()
            best_small_val = val_sub
            print(f"  Updated best after sub‑grad: {best_small_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Phase 3: simulated annealing polishing ===")
        seq_sa, val_sa = _simulated_annealing(
            upscaled, n_large, deadline, max_iters=350_000, init_temp_factor=0.05
        )
        if val_sa < best_small_val:
            upscaled = seq_sa.copy()
            best_small_val = val_sa
            print(f"  Updated best after SA: {best_small_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(upscaled.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_small_val:.6f} "
        f"(length {len(upscaled)})"
    )

    globals()["height_sequence_1"] = upscaled.tolist()
    return upscaled.tolist()
