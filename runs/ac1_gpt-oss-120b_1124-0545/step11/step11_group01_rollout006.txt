def _lp_optimize(f, M):
    n = len(f)
    offsets = np.arange(n)
    data = [np.full(n - d, f[d]) for d in range(n)]

    if res.success:

        return g
    else:
        return None

            break

"""
Optimisation via alternating LP (max‑sum sub‑problem) → mirror‑descent →
simulated‑annealing.

The LP stage (theoretical “alternating” update from the statement) 
maximises the ℓ₁‑norm of a new step‑function g under the constraint 
‖f∗g‖∞ ≤ M (where M = ‖f∗f‖∞).  The optimal g is obtained by a linear
program.  After a few LP steps we switch to exponentiated‑gradient
(mirror) sub‑gradient descent on the exact max‑convolution objective and
finally polish with a simple SA routine.  The whole pipeline is limited
to the supplied time budget (≈1000 s) and returns the best sequence found.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (exact, not the surrogate)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the target functional:
        2·n·max(conv(seq,seq)) / (sum(seq)²)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a <= 0.0:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Simple ordering helpers
# ----------------------------------------------------------------------
def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, … (then renormalise)."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(vec, 1.0 / len(vec))


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on 2·n·max(conv).
    Returns best sequence and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20  # initial learning rate

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry (active index)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise (positivity already holds)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional large/small alternation
        if it % 50_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {val_alt:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (swap / mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 800_000,
                                 init_temp_factor: float = 0.03,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple SA with three move types. Returns best sequence and value.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling schedule
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# LP sub‑problem (max‑sum g under convolution bound)
# ----------------------------------------------------------------------
def _lp_optimize(f: np.ndarray, M: float, deadline: float) -> np.ndarray | None:
    """
    Solve
        maximise   Σ g_i
        subject to   (f * g)_k ≤ M  for all k,
                     g_i ≥ 0.
    Returns the optimiser g (not normalised), or None on failure.
    """
    n = len(f)

    # construct sparse convolution matrix A (size (2n‑1) × n)
    # each diagonal offset d (>=0) contains the constant f[d]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    c = -np.ones(n, dtype=np.float64)                 # minimise -Σg  → maximise Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    # a tiny safety margin for the LP solver so it does not run forever
    time_left = max(0.0, deadline - time.time())
    # SciPy’s HiGHS accepts a time_limit option (seconds); guard against
    # older SciPy versions by catching a possible KeyError.
    opts = {"presolve": True}
    try:
        opts["time_limit"] = time_left * 0.9
    except Exception:
        pass

    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options=opts)

    if res.success:
        return res.x
    else:
        print(f"[LP] linprog failed – status {res.status}: {res.message}")
        return None

        prev_arr = None

            x_new = np.linspace(0, old_len - 1, n, dtype=np.float64)
            f = np.interp(x_new, x_old, prev_arr)
    else:
        # fall back to a smooth analytic seed (beta distribution)
        # beta parameters chosen heuristically

            a = left * right
            s = a.sum()
            return a / s if s > 0 else np.full(m, 1.0 / m)
