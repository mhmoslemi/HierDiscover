"""
Propose a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  The search proceeds in several phases:

1) **Broad analytic & random seed sweep (n≈1500).**  Hundreds of parameterised
   families (Beta, sine, Tukey, geometric, power‑law, Gaussian,
   super‑Gaussian etc.) together with many Dirichlet draws are generated.
   Each candidate is ℓ¹‑normalised and its exact surrogate objective
   `2·n·max(conv)` is computed via an FFT‑based convolution.  The best
   ~30 seeds are retained.

2) **Upscaling to a larger size (n≈4000).**  The retained seeds are linearly
   interpolated to a longer length and the best few (≤5) are selected.

3) **Heavy Adam optimisation on a smooth “log‑sum‑exp” surrogate.**  The
   distribution is represented by soft‑max logits, guaranteeing non‑negativity
   and the simplex constraint.  The temperature β is annealed up to 5000,
   and the Adam schedule runs for as many iterations as the remaining time
   budget permits.

4) **Brief sub‑gradient refinement** on the true max‑convolution objective
   (projected sub‑gradient descent) to polish the Adam result.

5) **Long simulated‑annealing polishing** that repeatedly reshuffles a
   random subset of the mass (Dirichlet re‑allocation).  The schedule is
   aggressive but cheap because each move only needs a single FFT‑based
   convolution.

6) **Final verification** with the official `evaluate_sequence` call.

All stages respect the global 1000 s budget; intermediate progress and
objective values are printed.  The best sequence found is stored in the
global variable `height_sequence_1` for reuse in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate objective for a ℓ¹‑normalised sequence (∑seq == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ¹‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (0<r<1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped density on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _supergaussian(n: int, sigma_frac: float = 0.2, exponent: float = 4.0) -> np.ndarray:
    """exp(-|x|^exponent) – a more peaked Gaussian for exponent>2."""
    if sigma_frac <= 0.0 or exponent <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((np.abs(i - mu) / sigma) ** exponent))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   beta_max: float = 5000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max‑convolution.
    Parameters are logits of the soft‑max representation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 10.0
    beta_mul = 1.004

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)             # exact surrogate (not smoothed)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] iter={it:6d} best={best_val:.6f}")

        # temperature increase
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        # soft‑max weights (log‑sum‑exp)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 300_000,
                      init_lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a simple diminishing step‑size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] iter={it:7d} best={cur_val:.6f}")

        # sub‑gradient for active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.05,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polishing that reshuffles a random subset of the mass.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        # pick a random subset size (3‑15)
        k = rng.integers(3, 16)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        # re‑allocate using Dirichlet
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] iter={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a small‑n sequence to a larger n."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    else:
        ys = _uniform(n_big)
    return ys.astype(np.float64)


def _add_candidate(pool: list[tuple[float, np.ndarray]],
                   seq: np.ndarray, max_size: int) -> None:
    """Maintain a sorted pool of best candidates (by surrogate value)."""
    val = _objective(seq)
    if len(pool) < max_size:
        pool.append((val, seq))
        pool.sort(key=lambda x: x[0])
    else:
        if val < pool[-1][0]:
            pool[-1] = (val, seq)
            pool.sort(key=lambda x: x[0])


def _generate_small_seeds(n: int, rng: np.random.Generator,
                          top_k: int = 30) -> list[np.ndarray]:
    """
    Create a dense pool of parameterised candidates on a modest n.
    The function returns the best `top_k` seeds (already ℓ¹‑normalised).
    """
    pool: list[tuple[float, np.ndarray]] = []

    # deterministic analytic families
    _add_candidate(pool, _uniform(n), top_k)
    _add_candidate(pool, _triangular(n), top_k)

    # Trapezoidal sweep
    for wf in np.linspace(0.02, 0.45, 12):
        _add_candidate(pool, _trapezoidal(n, width_fraction=wf), top_k)

    # Sine windows – extended gamma range
    for gamma in np.linspace(0.0, 12.0, 31):
        _add_candidate(pool, _sine_window(n, gamma=gamma), top_k)

    # Tukey windows
    for alpha in np.linspace(0.0, 0.95, 20):
        _add_candidate(pool, _tukey_window(n, alpha=alpha), top_k)

    # Geometric families – dense near 1
    r_vals = np.concatenate([
        np.linspace(0.01, 0.98, 20),
        np.linspace(0.98, 0.9999, 20),
        np.logspace(np.log10(0.9999), np.log10(0.999999), 10)
    ])
    for r in r_vals:
        _add_candidate(pool, _geom(n, r), top_k)
        _add_candidate(pool, _sym_geom(n, r), top_k)

    # Power‑law family
    for p in np.linspace(0.0, 10.0, 25):
        _add_candidate(pool, _power(n, p), top_k)

    # Beta shape – finer grid, include very small exponents
    alphas = np.concatenate([
        np.linspace(0.05, 0.5, 12),
        np.linspace(0.5, 5.0, 15),
        np.linspace(5.0, 10.0, 8)
    ])
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            _add_candidate(pool, _beta_shape(n, a, b), top_k)

    # Gaussian windows
    for sigma_frac in np.linspace(0.08, 0.45, 9):
        _add_candidate(pool, _gaussian(n, sigma_frac), top_k)

    # Super‑Gaussian (more flexible tails)
    for sigma_frac in np.linspace(0.08, 0.45, 7):
        for expnt in [2.0, 3.0, 4.0, 6.0, 8.0]:
            _add_candidate(pool, _supergaussian(n, sigma_frac, expnt), top_k)

    # Random Dirichlet draws – various concentrations
    for conc in (0.001, 0.005, 0.02, 0.1, 0.5, 1.0, 2.0, 5.0):
        for _ in range(6):
            _add_candidate(pool, _random_dirichlet(n, conc, rng), top_k)

    # Random exponential draws
    for _ in range(12):
        _add_candidate(pool, _random_exponential(n, rng), top_k)

    # Include previously stored solution if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    _add_candidate(pool, prev_arr / s, top_k)
        except Exception:
            pass

    # Return only the sequences (sorted by objective ascending)
    return [vec for _, vec in pool]

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The routine respects the time budget and returns the best sequence found
    as a list of floats.
    """
    # ------------------------------------------------------------------
    # Preparations
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – exploratory sweep on a modest size (n≈1500)
    # ------------------------------------------------------------------
    n_small = 1500
    print(f"\n--- Stage 0: exhaustive sweep (n={n_small}) ---")
    small_best = _generate_small_seeds(n_small, rng, top_k=30)
    print(f"  -> retained top {len(small_best)} candidates after sweep")

    # ------------------------------------------------------------------
    # Stage 1 – upscale to a larger problem size (n≈4000)
    # ------------------------------------------------------------------
    n_large = 4000
    print(f"\n--- Stage 1: upscaling to n={n_large} ---")
    large_candidates = [_upscale(vec, n_large) for vec in small_best]

    # Evaluate the up‑scaled candidates (exact surrogate)
    evaluated = [(vec, _objective(vec)) for vec in large_candidates]
    evaluated.sort(key=lambda x: x[1])

    # Keep the best few for heavy optimisation
    top_k_large = 5
    top_large = [vec for vec, _ in evaluated[:top_k_large]]
    global_best_seq = top_large[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"[init] best surrogate after upscale = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation (temperature‑aware)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # allocate about 60 % of remaining time to Adam, the rest for polishing
        adam_budget = remaining * 0.60
        seeds_to_opt = min(len(top_large), 3)          # keep it modest
        per_seed_budget = adam_budget / max(seeds_to_opt, 1)
        print(f"\n--- Stage 2: Adam optimisation ({seeds_to_opt} seeds, ≈{per_seed_budget:.1f}s each) ---")

        for idx, seed_vec in enumerate(top_large[:seeds_to_opt], 1):
            if time.time() > deadline:
                break
            # rough iteration budget (empirical 0.0018 s per iter for n≈4000)
            est_iter = int(per_seed_budget / 0.0018)
            est_iter = max(20_000, min(est_iter, 250_000))
            print(f"[Adam] seed #{idx}/{seeds_to_opt}, max_iters≈{est_iter}")
            seq_adam, val_adam = _adam_optimize(
                seed_vec, n_large, deadline,
                max_iters=est_iter,
                beta_max=5000.0,
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – short sub‑gradient refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_iters = int(remaining * 1200)   # about 1 200 iterations per sec
        max_iters = max(100_000, min(max_iters, 500_000))
        print(f"\n--- Stage 3: sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n_large, deadline,
            max_iters=max_iters,
            init_lr=0.5,
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        max_sa_iters = int(remaining * 950)   # ~0.001 s per iteration
        max_sa_iters = max(200_000, min(max_sa_iters, 2_000_000))
        print(f"\n--- Stage 4: simulated‑annealing polishing (max_iters={max_sa_iters}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n_large, deadline,
            max_iters=max_sa_iters,
            init_temp_factor=0.05,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
