"""
Propose a low‑value step‑function.

Algorithm
---------
1. **Initialisation** – start from the best known sequence (if any) or
   the uniform distribution, normalised so that the ℓ₁‑norm equals 1.
2. **Linear‑programming refinement** – repeatedly solve the LP
   ``max Σ g_j``  subject to  ``g ≥ 0``  and  ``conv(f,g) ≤ M`` where
   ``M = max(conv(f,f))``.  The LP concentrates mass on “quiet’’ indices,
   making the convolution more flat.  After each LP we mix the previous
   vector with the normalised LP solution and keep the improvement.
   Only a few dozen LP iterations are needed; the constraint matrix
   is built as a sparse Toeplitz matrix (≈ n²⁄2 non‑zeros).
3. **Adam polishing** – run Adam on the smooth log‑sum‑exp surrogate of
   the max‑convolution starting from the LP‑refined point.
4. **Projected sub‑gradient descent** – exact max‑convolution sub‑gradient
   steps squeeze the last bits.
5. **Simulated‑annealing polishing** – tiny Dirichlet re‑allocations on
   the simplex escape shallow local minima.
6. **Scaling** – finally scale the simplex vector to the required ℓ₁‑norm
   ``√(2 n)`` and return it as a plain Python list (the official evaluator
   then returns exactly ``max(conv)``).

All stages respect a 1000 s wall‑clock budget and keep the best
sequence found.  The code uses only NumPy, SciPy (sparse LP) and the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant). ``seq`` is assumed to sum to 1;
    the function returns max(conv(seq, seq)).
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimisers (Adam, sub‑gradient, simulated annealing)
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 500_000,
                   beta_max: float = 2000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Returns (best_sequence, best_exact_value).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        a = _softmax(v)                  # point on simplex (sum=1)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.9f}")

        # raise temperature
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # surrogate (smooth max) gradient
        conv = _conv_fft(a, a)           # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum                    # softmax over convolution entries

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)       # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # back‑prop to logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best sequence encountered and its exact value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]          # exact evaluation (scale‑invariant)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.9f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (using current best)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Tiny‑move simulated‑annealing on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)       # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.9f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement utilities
# ----------------------------------------------------------------------


def _build_conv_ub_matrix(f_vec: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (shape (2n-1, n)) such that
    A @ g = conv(f_vec, g)  (full linear convolution).
    Only non‑negative offsets appear, so the matrix is upper‑banded.
    """
    n = f_vec.shape[0]
    rows = 2 * n - 1
    # each diagonal d (0 ≤ d < n) is constant f[d]
    offsets = np.arange(n)                     # 0 … n‑1
    data = [np.full(n - d, f_vec[d], dtype=np.float64) for d in range(n)]
    A = sp.diags(data, offsets, shape=(rows, n), format='csr')
    return A


def _lp_refine(f: np.ndarray, n: int, deadline: float,
               max_outer: int = 30, mix_factor: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based refinement steps.
    Returns the improved sequence (still ℓ₁‑normalised to 1) and its max‑conv.
    """
    best_f = f.copy()
    best_val = _objective(best_f)

    outer = 0
    while time.time() < deadline and outer < max_outer:
        M = _objective(f)                     # current max conv
        A = _build_conv_ub_matrix(f)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)     # maximise sum(g)  ⇔ minimise -sum(g)

        # time limit for the LP is the remaining budget (rounded down)
        time_left = max(0.0, deadline - time.time())
        lp_options = {"presolve": True}
        if time_left > 0:
            lp_options["time_limit"] = time_left

        res = opt.linprog(c, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method="highs",
                          options=lp_options)

        if not res.success:
            print(f"[LP] iteration {outer}: failed ({res.message})")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g                     # project back onto simplex

        # simple line‑search on the mixing factor
        t = mix_factor
        while t > 1e-4:
            new_f = (1 - t) * f + t * g_norm
            new_f = _proj_simplex(new_f)
            new_val = _objective(new_f)
            if new_val < M:
                break
            t *= 0.5

        if new_val < best_val:
            best_val = new_val
            best_f = new_f.copy()
            print(f"[LP] iter={outer:2d} improved max‑conv to {best_val:.9f}")

        # accept the new point if it improves the objective
        if new_val < M:
            f = new_f
        else:
            # no progress – stop early
            break

        outer += 1

    return best_f, best_val
