"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)^2)

which is exactly the value returned by `evaluate_sequence` because scaling
cancels out.  The algorithm improves upon the previous version by:

* building the LP constraints with a **sparse Toeplitz matrix** (via
  `scipy.sparse.diags`) – this makes the linear program feasible even for
  n=4000;
* performing a **grid line‑search** on the mixing weight after every LP
  direction, instead of a fixed tiny step;
* using a modest number of Frank‑Wolfe outer iterations (≤12) and then
  polishing with exact sub‑gradient descent and a simulated‑annealing
  phase (as before).

All helpers are defined at module level; the only public entry point is
`propose_candidate`.  The routine respects the 1000 s time budget,
stores the best sequence in the global name `height_sequence_1` and
returns it as a plain list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (sum=1, v≥0).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(seq,seq)) for a sequence that is ℓ₁‑normalised
    (i.e. sum(seq)==1).  This is exactly the value returned by
    `evaluate_sequence` for such a vector.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                       (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                       (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Exact sub‑gradient descent (same as before)
# ----------------------------------------------------------------------
def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """Frank‑Wolfe‑style sub‑gradient descent on the true objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:          # reshape a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # transfer a small chunk between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)  # move at most 5 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement – **sparse** version
# ----------------------------------------------------------------------
def _lp_optimize_sparse(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max 1ᵀb   s.t. b ≥ 0 , conv(f,b) ≤ M .
    Returns the optimal (un‑normalised) b or None on failure.
    """
    if linprog is None or diags is None:
        print("[LP] scipy.linprog or scipy.sparse.diags not available.")
        return None

    n = len(f)
    m = 2 * n - 1

    # Build the Toeplitz constraint matrix using constant diagonals.
    # Entry A[i, j] = f[i - j] for 0 ≤ i - j < n, else 0.
    offsets = np.arange(-(n - 1), n, dtype=int)          # length = 2n-1
    data = []
    for d in offsets:
        length = n - abs(d)
        # constant value f[abs(d)] placed on this diagonal
        diag = np.full(length, f[abs(d)], dtype=np.float64)
        data.append(diag)

    A = diags(data, offsets, shape=(m, n), format='csr')

    # Linear program: minimise -sum(b)  (i.e. maximise sum(b))
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, M, dtype=np.float64)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
        else:
            print("[LP] linprog did not succeed:", res.message)
            return None
    except Exception as exc:  # pragma: no cover
        print("[LP] linprog raised exception:", exc)
        return None


def _lp_refine(seq: np.ndarray,
               n: int,
               deadline: float,
               max_outer: int = 12,
               t_grid: tuple[float, ...] = (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5)
               ) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style refinement using a sparse LP at each outer iteration.
    Returns the refined vector and its surrogate value.
    """
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)

    for outer in range(1, max_outer + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(best_seq, best_seq)
        M = np.max(conv)

        b = _lp_optimize_sparse(best_seq, M)
        if b is None:
            print("[LP] optimisation failed – aborting LP phase.")
            break

        s = b.sum()
        if s <= 0.0:
            print("[LP] LP returned zero mass – aborting.")
            break
        b_norm = b / s

        # try a grid of mixing weights
        improved = False
        for t in t_grid:
            cand = (1.0 - t) * best_seq + t * b_norm
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand = cand / cand_sum
            val = _objective_surrogate(cand)
            if val < best_val - 1e-9:
                best_seq, best_val = cand, val
                improved = True
                print(f"[LP] iter={outer} mix={t:.3f} surrogate={val:.6f}")
                break        # accept the first improvement (grid is ordered from small to large)

        if not improved:
            # no improvement with any tried t → stop early
            print(f"[LP] iter={outer} – no improving mix found; terminating.")
            break

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin for final evaluation

    # --------------------------------------------------------------
    # Length selection – 4000 gives a good balance.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Stage 0 – seed generation
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    scored = [(seq, _objective_surrogate(seq)) for seq in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate after seeding = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – LP refinement (sparse version)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP refinement stage (sparse) ---")
        best_seq, best_val = _lp_refine(best_seq, n, deadline,
                                        max_outer=12,
                                        t_grid=(0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5))
        print(f"[LP] after refinement surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – exact sub‑gradient descent polishing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent polishing ---")
        remaining = deadline - time.time()
        max_iters_sub = int(min(2_500_000, remaining * 1e5))
        seq_proj, val_proj = _subgradient_proj(best_seq, n, deadline,
                                               max_iters=max_iters_sub)
        if val_proj < best_val - 1e-12:
            best_seq, best_val = seq_proj, val_proj
            print(f"[Post‑Sub] new best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – simulated annealing polishing (optional).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(1_500_000, remaining * 1e5))
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              max_iters=max_iters_sa)
        if val_sa < best_val - 1e-12:
            best_seq, best_val = seq_sa, val_sa
            print(f"[Post‑SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
