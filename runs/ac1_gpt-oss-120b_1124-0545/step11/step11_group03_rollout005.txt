"""
Propose a near‑optimal non‑negative step function.

The algorithm starts from the best known sequence (if any) or from a
random Dirichlet draw, then performs a long projected‑gradient descent
with a correctly‑computed sub‑gradient (the previous script used an
approximation).  Momentum (heavy‑ball) is added to accelerate convergence
and a simple exponential‑gradient variant is run afterwards for extra
polishing.  After optimisation the vector is trimmed of leading / trailing
zeros (which only increase the length factor in the objective) and the
result is returned.

The routine respects the global 1000 s deadline, uses only NumPy and
standard library, and stores the final sequence in the global variable
``height_sequence_1`` for possible reuse in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    It is exactly the numerator of ``evaluate_sequence`` (the denominator is 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seeds (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet and normalise to sum 1."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------
def _maxconv_and_index(a: np.ndarray) -> tuple[float, int]:
    """Return (max(conv), index) for the full convolution of a with itself."""
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _subgradient_momentum_opt(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 5_000_000,
    beta: float = 0.9,
    lr0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Heavy‑ball sub‑gradient descent on the true max‑convolution objective.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    vel = np.zeros_like(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # convolution and active shift
        conv_val, k = _maxconv_and_index(a)

        cur_val = 2.0 * n * conv_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[HB‑SG] iter={it:,} best surrogate={best_val:.6f}")

        # --- sub‑gradient of conv[k] -------------------------------------------------
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx

        g = np.zeros_like(a)
        # derivative contribution a_j added to position i, and a_i to position j
        g[idx] += a[jdx]
        g[jdx] += a[idx]          # double count when i==j (produces 2*a_i)

        # momentum update
        vel = beta * vel + (1.0 - beta) * g

        # step‑size schedule
        lr = lr0 / np.sqrt(it)

        a = a - lr * vel
        a = _proj_simplex(a)       # stay on the simplex

    return best_seq, best_val


def _exp_gradient_opt(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2_000_000,
    lr0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (multiplicative‑weights) descent.
    Guarantees non‑negativity automatically.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv_val, k = _maxconv_and_index(a)
        cur_val = 2.0 * n * conv_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[Exp‑Grad] iter={it:,} best surrogate={best_val:.6f}")

        # gradient of the active shift (same as in heavy‑ball)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] += a[jdx]
        g[jdx] += a[idx]

        # learning‑rate schedule
        lr = lr0 / np.sqrt(it)

        # multiplicative update, stay non‑negative
        a = a * np.exp(-lr * g)

        # re‑normalise (exact ℓ₁ = 1)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = _uniform(n)

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Initial sequence (reuse previous if available)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and prev:
        a0 = np.asarray(prev, dtype=np.float64)
        if a0.size > 0 and a0.sum() > 0:
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
            print("[Init] using previously stored sequence")
        else:
            a0 = None
    else:
        a0 = None

    # if no good previous, fall back to a random Dirichlet seed (moderate size)
    if a0 is None:
        n = 4000
        a0 = _random_dirichlet(n, conc=0.5, rng=rng)
        print("[Init] generated fresh Dirichlet seed")
    else:
        n = a0.shape[0]

    # Keep a reference to the best sequence found so far
    best_seq = a0.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] initial surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Heavy‑ball sub‑gradient refinement (major work)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Heavy‑ball sub‑gradient refinement ---")
        seq_hb, val_hb = _subgradient_momentum_opt(best_seq, deadline)
        if val_hb < best_val:
            best_seq, best_val = seq_hb, val_hb
            print(f"[HB] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exponential‑gradient polishing (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        seq_exp, val_exp = _exp_gradient_opt(best_seq, deadline)
        if val_exp < best_val:
            best_seq, best_val = seq_exp, val_exp
            print(f"[Exp] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final trimming of leading / trailing near‑zero entries
    # ------------------------------------------------------------------
    eps = 1e-12
    nz = np.where(best_seq > eps)[0]
    if nz.size > 0:
        trimmed = best_seq[nz[0] : nz[-1] + 1]
        if trimmed.shape[0] != best_seq.shape[0]:
            print(f"[Trim] removed {best_seq.shape[0] - trimmed.shape[0]} zero tails")
        best_seq = trimmed

    # ------------------------------------------------------------------
    # 5️⃣  Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
