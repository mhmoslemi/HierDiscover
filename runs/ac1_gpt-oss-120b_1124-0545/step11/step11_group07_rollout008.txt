"""
Hybrid search for a low‑value step function.

The routine proceeds as follows

1. Load the previously‑found best sequence (if any).  If none is
   available, initialise a simple β‑distribution and normalise it to
   sum = 1.

2. Compute the full linear convolution a*a once (FFT).

3. Perform a fast greedy local search that repeatedly moves a small
   amount of mass from a point that heavily participates in the
   current maximal convolution entry to a random other point.
   The convolution is updated *incrementally* – only three entries
   (2·src, 2·dst and src+dst) change – so each iteration costs only
   O(1) Python operations plus a cheap ``np.max`` over ~8000 values.
   Moves are accepted only when they do not increase the maximal
   convolution value, guaranteeing monotone improvement.  The step
   size is annealed.

4. After the time budget (≈1000 s) the best sequence found is scaled
   to the required ℓ₁‑norm (√(2n)) and returned.  The official
   evaluator is called once for a sanity check.

The method dramatically reduces the number of expensive FFTs (only
one) and spends the bulk of the budget on cheap, exact  updates,
which is well‑suited for squeezing out the last few percent of
improvement over the previous best bound (~1.509 → target ≤ 1.503).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """β‑distribution on {0,…,n‑1}, normalised to sum = 1."""
    if alpha <= 0.0 or beta <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Incremental convolution update for a mass move src → dst
# ----------------------------------------------------------------------


def _apply_move(seq: np.ndarray,
                conv: np.ndarray,
                src: int,
                dst: int,
                eps: float) -> None:
    """Move ``eps`` mass from ``src`` to ``dst`` and update ``conv`` in‑place."""
    a_src = seq[src]
    a_dst = seq[dst]

    # indices that change
    idx_src2 = src * 2
    idx_dst2 = dst * 2
    idx_sum = src + dst

    # store old values (caller may need them for a rollback)
    old_src2 = conv[idx_src2]
    old_dst2 = conv[idx_dst2]
    old_sum = conv[idx_sum]

    # delta updates (see derivation in the docstring)
    conv[idx_src2] = old_src2 - 2.0 * a_src * eps + eps * eps
    conv[idx_dst2] = old_dst2 + 2.0 * a_dst * eps + eps * eps
    conv[idx_sum] = old_sum + 2.0 * eps * (a_src - a_dst) - 2.0 * eps * eps

    # update the sequence
    seq[src] = a_src - eps
    seq[dst] = a_dst + eps

    # clip tiny negatives that can appear due to rounding
    if conv[idx_src2] < 0.0:
        conv[idx_src2] = 0.0
    if conv[idx_dst2] < 0.0:
        conv[idx_dst2] = 0.0
    if conv[idx_sum] < 0.0:
        conv[idx_sum] = 0.0


def _revert_move(seq: np.ndarray,
                 conv: np.ndarray,
                 src: int,
                 dst: int,
                 eps: float,
                 old_src2: float,
                 old_dst2: float,
                 old_sum: float) -> None:
    """Undo a move; restores ``seq`` and the three affected conv entries."""
    seq[src] += eps
    seq[dst] -= eps
    conv[src * 2] = old_src2
    conv[dst * 2] = old_dst2
    conv[src + dst] = old_sum

    n = 4000

        except Exception:
            seq = None
    else:

        # fall back to a simple β‑shape (α≈0.7, β≈0.7) – fairly good start
        seq = _beta_shape(n, alpha=0.7, beta=0.7)
    seq = seq / seq.sum()          # ensure exact ℓ₁‑normalisation

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    conv = _conv_fft(seq, seq)     # length 2n‑1
    cur_max = float(np.max(conv))
    best_seq = seq.copy()
    best_conv = conv.copy()
    best_val = cur_max
    it = 0
    print(f"Initial max(conv) = {cur_max:.8f} → objective ≈ {2 * n * cur_max:.6f}")

    # --------------------------------------------------------------
    # Fast greedy local search (mass relocation)
    # --------------------------------------------------------------
    move_frac = 0.15               # start with up‑to 15 % of source mass

    decay = 0.9999                 # geometric decay per 1 000 accepted moves
    accepted = 0

        dst = rng.integers(0, n)

        # 4) compute a small transfer amount
        a_src = seq[src]
        if a_src <= 0.0:

        eps = rng.random() * a_src * move_frac
        if eps <= 0.0:
            continue

        # 5) save old convolution entries (for possible rollback)
        idx_src2 = src * 2
        idx_dst2 = dst * 2
        idx_sum = src + dst
        old_src2 = conv[idx_src2]
        old_dst2 = conv[idx_dst2]
        old_sum = conv[idx_sum]

        new_max = float(np.max(conv))

            accepted += 1

                best_seq = seq.copy()
                best_conv = conv.copy()
                stagnation_counter = 0

            else:
                stagnation_counter += 1
        else:
            # reject – restore previous state
            _revert_move(seq, conv, src, dst, eps,
                         old_src2, old_dst2, old_sum)
            stagnation_counter += 1

        it += 1

        # decay the step‑size every thousand accepted moves
        if accepted and accepted % 1000 == 0:
            move_frac = max(min_frac, move_frac * decay)

        # occasional escape from stagnation: perform a random perturbation
        if stagnation_counter > 20000:

            old_sum = conv[idx_sum]

    # --------------------------------------------------------------
