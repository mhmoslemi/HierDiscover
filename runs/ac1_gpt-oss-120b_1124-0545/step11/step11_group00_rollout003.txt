"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Initial seed** – If a previous solution exists (global
   ``height_sequence_1``) it is used, otherwise a uniform vector of
   length ``n`` is taken.  The vector is ℓ₁‑normalised (sum = 1).

2. **Linear‑programming tightening** – For the current sequence ``f``
   compute ``M = max(conv(f,f))``.  The constraint
   ``conv(f,g) ≤ M`` is linear in ``g``; we therefore solve a linear
   program that *maximises* ``∑ g_i`` under these constraints and
   ``g_i ≥ 0``.  The optimal ``g`` is then renormalised to have the same
   ℓ₁‑norm as ``f`` and mixed with ``f`` using a small convex step.
   Repeating this step (≈ 15 – 20 times) monotonically decreases the
   objective.

3. **Fine‑grained refinement** – After the LP phase a few thousand
   mirror‑descent (exponentiated‑gradient) iterations are applied to
   polish the solution w.r.t. the true max‑convolution objective.

4. **Optional simulated‑annealing polishing** – A short SA run may
   further improve the result.

All heavy work respects the global deadline (default 1000 s).  The
final sequence is stored in ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv) for a ℓ₁‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) sub‑gradient optimiser
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12            # initial step‑size factor
    power = 0.35          # step‑size decay exponent

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (lightweight version)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 300_000,
                                 init_temp_factor: float = 0.05,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles mass among a few indices."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        move = rng.integers(0, 3)
        if move == 0:                         # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move == 1:                       # redistribute a random subset
            k = rng.integers(2, 10)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new = cur.copy()
            new[idx] = rng.dirichlet(np.ones(k)) * total
        else:                                 # pairwise mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming tightening (core of the new search)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1, n)) such that
    A @ g = conv(f, g) for any g ≥ 0.
    Row k, column j contains f[k‑j] when 0 ≤ k‑j < n else 0.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j carries the vector f shifted down by j rows
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_tighten(f: np.ndarray,
                n: int,
                deadline: float,
                max_lp_iters: int = 20) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP:
        maximise  Σ g_i
        s.t.     conv(f, g) ≤ M   (M = max(conv(f,f)))   (2n‑1 constraints)
                 g_i ≥ 0
    and mix the result back into f.
    Returns the best sequence found and its surrogate value.
    """
    best_seq = f.copy()
    best_val = _objective(best_seq)
    t = 0.2                     # convex‑mixing factor

    for it in range(1, max_lp_iters + 1):
        if time.time() > deadline:
            break

        conv_fg = _conv_fft(f, f)
        M = np.max(conv_fg)

        # construct the linear constraints
        A = _build_conv_matrix(f)          # (2n‑1) × n
        b = np.full(2 * n - 1, M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)  # maximise Σ g_i  ⇔ minimise –Σ g_i
        bounds = [(0.0, None)] * n

        # ---- solve LP -------------------------------------------------
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # LP failure – fall back to tiny random perturbation and continue
            print(f"[LP] iteration {it} failed (msg: {res.message})."
                  " Applying a random perturbation.")
            perturb = np.random.default_rng().dirichlet(np.ones(n) * 0.01) * 1e-4
            f = (1 - t) * f + t * perturb
            f = f / f.sum()
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # normalise g to the same ℓ₁‑norm as f (which is 1)
        g = g / sum_g

        # convex combination
        f = (1 - t) * f + t * g
        f = f / f.sum()

        cur_val = _objective(f)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
            print(f"[LP] it={it:2d} improved surrogate = {cur_val:.6f}")

    return best_seq, best_val
