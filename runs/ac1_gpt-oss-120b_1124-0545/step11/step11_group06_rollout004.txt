    lin_res = scipy.optimize.linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                                     bounds=[(0, None)]*n,
                                     method='highs')
    if not lin_res.success:
        print(f"LP iteration {it+1} failed: {lin_res.message}")
        break
    g = lin_res.x
    g_sum = g.sum()
    if g_sum > 0:
        g = g / g_sum
    else:
        g = np.full(n, 1.0/n)
    # mixing factor
    t = t_initial * (0.8 ** it)  # decreasing
    f = (1 - t) * f + t * g
    f = f / f.sum()
    cur_val = _objective(f)
    print(f"LP iter {it+1}: objective={cur_val:.6f} (t={t:.4f})")
    if cur_val < best_val:
        best_val = cur_val
        best_seq = f.copy()
        print(f"  New global best: {best_val:.6f}")

"""
Search for a low‑value step function.

Algorithm
---------
1. Initialise a probability vector `f` (non‑negative, Σf = 1).  
   The start is taken from the previous best solution if available,
   otherwise a simple decreasing geometric shape is used.
2. Repeatedly apply the “alternating‑LP” update from the problem
   statement:
      * Let M = max (f * f)   (linear convolution).
      * Solve the linear programme
            maximise   Σ g_i
            subject to (f * g)_k ≤ M  for all k = 0 … 2n‑2
                       g_i ≥ 0
        where the convolution matrix is a lower‑triangular Toeplitz
        matrix built from `f`.
      * Rescale g to Σg = 1 and mix it with the current iterate:
            f ← (1‑t)·f + t·g   (t shrinks each iteration).
   The LP reduces the peak of the self‑convolution, consequently the
   objective 2·n·max(f*f) / (Σf)².
3. Polish the best vector found so far with a short Adam optimiser on a
   smooth‑max surrogate of the true max‑convolution.
4. Return the best sequence as a plain Python list and store it in the
   global variable ``height_sequence_1`` for possible reuse.
The routine respects the supplied time budget (default 1000 s) and prints
progress information.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz
from scipy import sparse

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, objective, projection)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a *probability* vector (Σseq = 1).
    Returns 2·n·max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (used in Adam optimiser)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on a log‑sum‑exp (smooth‑max) surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 400_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv(a,a)).
    Operates in logits space of a soft‑max so the simplex constraint is
    automatically satisfied.
    Returns the best sequence found and its true objective value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # initial temperature for log‑sum‑exp
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # raise temperature slowly (approach true max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)                  # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # softmax over conv entries

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # a small safety margin so that the call to evaluate_sequence never
    # triggers the timeout guard inside the evaluation function.
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a length – 2000 gives a good trade‑off between quality and
    # LP solve time (the bound grows roughly like O(n²) in the LP).
    # ------------------------------------------------------------------
    n = 2000
    print(f"\n=== Starting search (n={n}, budget={budget_s}s) ===")

    # ------------------------------------------------------------------
    # Initialise f.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None and isinstance(previous, list) and len(previous) == n:
        f = np.asarray(previous, dtype=np.float64)
        print("  Using previous solution (exact length match).")
    elif previous is not None and isinstance(previous, list):
        # Resample previous vector to the new length by linear interpolation
        prev_arr = np.asarray(previous, dtype=np.float64)
        x_old = np.linspace(0, 1, len(prev_arr))
        x_new = np.linspace(0, 1, n)
        f = np.interp(x_new, x_old, prev_arr)
        print("  Resampled previous solution to new length.")
    else:
        # Simple decreasing geometric shape – a reasonably good start.
        r = 0.997
        f = r ** np.arange(n, dtype=np.float64)
        print("  No previous solution – started from a geometric decay.")

    f = np.maximum(f, 0.0)
    s = f.sum()
    if s == 0.0:
        f = np.full(n, 1.0 / n)
    else:
        f = f / s

    best_seq = f.copy()
    best_val = _objective(best_seq)
    print(f"Initial objective: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Alternating linear‑programming updates.
    # ------------------------------------------------------------------
    max_lp_iters = 12
    t0 = 0.4  # initial mixing weight
    print("\n--- Phase 1: LP alternating updates ---")
    for lp_it in range(1, max_lp_iters + 1):
        if time.time() > deadline:
            break

        # current peak of the self‑convolution
        M = _conv_fft(f, f).max()

        # Build the lower‑triangular Toeplitz convolution matrix.
        # A_ub @ g  =  f * g   (linear convolution)
        c_vec = np.concatenate([f, np.zeros(n - 1)])      # first column
        r_vec = np.zeros(n)                               # first row (zeros → lower triangular)
        A_dense = toeplitz(c_vec, r_vec)                  # shape (2n‑1, n)
        A_ub = sparse.csr_matrix(A_dense)                # convert to sparse for the LP solver
        del A_dense                                         # free memory

        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        # maximise Σ g_i  → minimise -Σ g_i
        lin_res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs')
        if not lin_res.success:
            print(f"[LP {lp_it}] solver failed: {lin_res.message}")
            break

        g = lin_res.x
        g_sum = g.sum()
        if g_sum > 0.0:
            g = g / g_sum
        else:
            g = np.full(n, 1.0 / n)

        # Decrease mixing weight slowly
        t = t0 * (0.85 ** (lp_it - 1))
        f = (1.0 - t) * f + t * g
        f = f / f.sum()          # re‑normalise to stay on the simplex

        cur_val = _objective(f)
        print(f"[LP {lp_it:2d}] objective={cur_val:.6f}   (mix t={t:.4f})")
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
            print(f"   ** New global best after LP **")

    # ------------------------------------------------------------------
    # Phase 2 – short Adam polishing on the best candidate.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: Adam fine‑tuning ---")
        seq_adam, val_adam = _adam_optimize(best_seq, n, deadline,
                                            max_iters=400_000)
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"   Adam improvement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        # The evaluator will also clip/normalise, but we keep the vector
        # as it is (the algorithm works with a probability vector).
        ev = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal evaluate_sequence value: {ev:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
