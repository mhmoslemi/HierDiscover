   (c * c)[k] = 1 for every k ≤ n‑1.  After ℓ₁‑normalisation they give a
   very flat convolution and already improve the uniform bound (2 → ~1.57).

2. **Permutation search** – The order of the coefficients influences the
   maximum entry of the convolution.  Starting from the “alternating”
   permutation (largest, smallest, second‑largest, …) we run a short,
   temperature‑controlled simulated‑annealing that swaps two positions.
   The surrogate objective ``_objective`` (2·n·max(conv)) is cheap (FFT).

3. **Mirror‑subgradient refinement** – From the best permuted sequence we
   perform many multiplicative‑gradient updates (exponentiated‑gradient) on
   the true max‑convolution.  The update is cheap (O(n)) because only the
   active shift contributes to the gradient.

All steps respect the global deadline (default 1000 s).  The final
sequence is stored in the global name ``height_sequence_1`` and returned
as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic “optimal” kernel (central binomial coefficients)
# ----------------------------------------------------------------------
def _central_binomial(n: int) -> np.ndarray:
    """
    Generate the first n central‑binomial coefficients
    c_i = C(2i,i) / 4**i  (c_0 = 1, c_{i+1}=c_i * (2i+1)/(2i+2)).
    The raw vector is returned (not normalised).
    """
    c = np.empty(n, dtype=np.float64)
    c[0] = 1.0
    for i in range(1, n):
        c[i] = c[i - 1] * (2 * i - 1) / (2 * i)
    return c


def _alternating_order(arr: np.ndarray) -> np.ndarray:
    """
    Return an “alternating” permutation: largest, smallest, 2‑nd largest,
    2‑nd smallest, …  (helps to keep large numbers far apart).
    """
    a = np.sort(arr)          # ascending
    lo, hi = 0, len(a) - 1
    out = np.empty_like(a)
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    return out


# ----------------------------------------------------------------------
# Mirror‑subgradient optimizer (exact max, multiplicative updates)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 deadline: float,
                                 lr0: float = 0.30,
                                 power: float = 0.38) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while True:
        it += 1
        if time.time() > deadline:
            break

        # linear convolution to locate the active shift
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = float(2 * len(a) * conv[max_idx])

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient for the active shift (only the overlapping indices)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (len(a) - 1))
        i_end = min(len(a) - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative (exponentiated) update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (swap‑based moves)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 deadline: float,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly swaps two positions.  The surrogate objective
    (the same as used in the mirror step) is evaluated with a fast FFT.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature proportional to the current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    while time.time() < deadline:
        # geometric cooling
        T = max(T * 0.9995, min_temp)

        # propose a swap
        i, j = rng.choice(len(cur), size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 987654321, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain ``list[float]`` and registers it
    as the global variable ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Length – 4000 offers a good compromise between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build an initial candidate from the central‑binomial family
    # ------------------------------------------------------------------
    raw_c = _central_binomial(n)                   # raw (unnormalised) coefficients
    raw_c /= raw_c.sum()                           # ℓ₁‑normalise   →  sum = 1
    init_seq = _alternating_order(raw_c)           # spread large values apart
    print("[Init] central‑binomial alternated surrogate = {:.6f}".format(_objective(init_seq)))

    # ------------------------------------------------------------------
    # Phase 1 – Simulated‑annealing on permutations (cheap swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- SA on permutations (phase 1) ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            init_seq, deadline, init_temp_factor=0.04, min_temp=1e-12
        )
        print("[SA‑phase1] best surrogate = {:.6f}".format(val_sa))
    else:
        seq_sa, val_sa = init_seq, _objective(init_seq)

    # ------------------------------------------------------------------
    # Phase 2 – Mirror‑subgradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑subgradient refinement (phase 2) ---")
        seq_mir, val_mir = _mirror_subgradient_optimize(
            seq_sa, deadline, lr0=0.30, power=0.38
        )
        print("[Mirror] best surrogate = {:.6f}".format(val_mir))
    else:
        seq_mir, val_mir = seq_sa, val_sa

    # ------------------------------------------------------------------
    # Phase 3 – Final short SA to escape possible local minima
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final SA polishing (phase 3) ---")
        seq_final, val_final = _simulated_annealing_optimize(
            seq_mir, deadline, init_temp_factor=0.02, min_temp=1e-12
        )
        print("[SA‑final] best surrogate = {:.6f}".format(val_final))
    else:
        seq_final, val_final = seq_mir, val_mir

    # ------------------------------------------------------------------
    # Sanity check with the official evaluator (not timed)
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(seq_final.tolist())
        print("\n[Eval] Official evaluate_sequence value = {:.6f}".format(official_val))
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse
    globals()["height_sequence_1"] = seq_final.tolist()
    return seq_final.tolist()
