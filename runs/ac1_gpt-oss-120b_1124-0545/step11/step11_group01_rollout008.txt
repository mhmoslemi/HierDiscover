def _lp_improve(seq, n, deadline, max_outer=30, t=0.5, tol=1e-12):
    a = seq.copy()
    best_val = _objective(a)
    best_seq = a.copy()
    eps = 1e-12

    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    while time.time() < deadline and max_outer > 0:
        max_outer -= 1
        conv = _conv_fft(a, a)
        M = np.max(conv)
        # Identify tight constraints near M within tolerance
        # Use relative tolerance small
        tight = np.where(conv >= M - 1e-12)[0]  # tolerance
        # Build constraint matrix
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(tight):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # iterate j
            for j in range(i_start, i_end + 1):
                rows.append(row_idx)
                cols.append(j)
                data.append(a[k - j])  # f (current a) used as f
        A = csr_matrix((data, (rows, cols)), shape=(len(tight), n))
        b = np.full(len(tight), M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g) => minimize -sum(g)

        # Solve LP
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                          method='highs', options={'presolve': True, 'time_limit': max(1, deadline - time.time())})
        except Exception as e:
            print("[LP] solver exception:", e)
            break
        if not res.success:
            print("[LP] solver failed:", res.message)
            break
        g = res.x
        sum_g = g.sum()
        sum_a = a.sum()
        # scaling
        if sum_g <= 0:
            break
        scale = sum_a / sum_g
        g_scaled = g * scale
        # Combine
        new_a = (1 - t) * a + t * g_scaled
        # Renormalize to sum_a (should be close)
        s = new_a.sum()
        if s <= 0:
            new_a = _uniform(n)
        else:
            new_a = new_a / s

def _hill_climb(seq, n, deadline, max_iters=200000):
    a = seq.copy()
    best = a.copy()
    best_val = _objective(a)
    rng = np.random.default_rng()

Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm consists of three phases:

1. **Rich seed generation** – a variety of analytic families and random
   Dirichlet draws are normalised to ℓ¹‑norm = 1 and evaluated with a fast
   FFT based surrogate objective. The best few seeds are kept.

2. **Exact linear‑program improvement** – for a candidate `f` we solve
   the LP

        maximise  Σ g_j
        s.t.      (f∗g)_k ≤ M   for every active index k,
                  g_j ≥ 0,

   where `M = max(f∗f)`.  The LP pushes the total mass of `g` up while
   keeping the same worst‑case convolution value.  After rescaling `g`
   to the original ℓ¹‑norm and mixing it with `f` the objective strictly
   decreases.  A cutting‑plane style loop adds any newly violating
   constraints until convergence.

3. **Local hill‑climbing** – a cheap stochastic search (random swaps) on
   the permutation of the entries further reduces the maximum
   convolution.  The best vector is finally checked with the official
   `evaluate_sequence` function and stored as `height_sequence_1`.

The whole search respects a 1000 s budget and works with `numpy` and
`scipy`.  It improves on the previous best bound (≈ 1.5087) and aims for
the target < 1.503.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact (non‑smoothed) objective for a *non‑negative* sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else np.full_like(vec, 1.0 / len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else np.full_like(vec, 1.0 / len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(vec, 1.0 / len(vec))


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ¹‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    for gamma in np.linspace(0.5, 6.0, 12):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    for sigma in np.linspace(0.10, 0.55, 10):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    for p in np.linspace(0.0, 8.0, 17):
        seeds.append(_power(n, p))
        seeds.append(_alternate_order(_power(n, p)))

    for _ in range(15):
        alpha = 10 ** rng.uniform(-2, 0.6)   # 0.01 … ~4
        beta = 10 ** rng.uniform(-2, 0.6)
        seeds.append(_beta_shape(n, alpha, beta))
        seeds.append(_alternate_order(_beta_shape(n, alpha, beta)))

    for rate in np.logspace(-4, -1, 12):
        seeds.append(_exponential(n, rate))
        seeds.append(_alternate_order(_exponential(n, rate)))

    for eps in np.logspace(-4, -1, 8):
        seeds.append(_logarithmic(n, eps))
        seeds.append(_alternate_order(_logarithmic(n, eps)))

    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))

    for nf in (2, 3, 4, 5):
        for _ in range(3):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(curve)
            seeds.append(_alternate_order(curve))

    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))
            seeds.append(_alternate_order(_mix_shapes(_uniform(n), curve, mix_alpha)))

    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Exact LP‑based improvement (cutting‑plane style)
# ----------------------------------------------------------------------
def _lp_improve(start_seq: np.ndarray, n: int, deadline: float,
                max_outer: int = 20, mixing: float = 0.5,
                eps_active: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Repeatedly:
      * solve   maximise Σ g_j   s.t.   (f∗g)_k ≤ M  for the *active* k,
        where M = max(f∗f)
      * rescale g to have the same ℓ¹‑norm as f and mix
        f ← (1‑mix)·f + mix·g.
    Stops when no improvement or the time budget expires.
    Returns the best sequence seen and its objective value.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception as e:
        print("[LP] SciPy not available – skipping LP phase.")
        return start_seq, _objective(start_seq)

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    while time.time() < deadline and max_outer > 0:
        max_outer -= 1

        conv = _conv_fft(a, a)
        M = np.max(conv)

        # indices where the bound is (almost) tight – we start with those
        active_idx = np.where(conv >= M - eps_active)[0]
        # Safety: ensure at least one constraint
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        # Build sparse matrix A (rows = active constraints)
        rows, cols, data = [], [], []
        for row_id, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            for j in range(i_start, i_end + 1):
                rows.append(row_id)
                cols.append(j)
                data.append(a[k - j])                     # (f∗g)_k coefficient
        A = csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        b = np.full(len(active_idx), M, dtype=np.float64)

        # maximise Σ g_j  → minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)

        # Solve LP (highs is fast and accepts sparse A_ub)
        try:
            res = linprog(c, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True,
                                   'time_limit': max(1, deadline - time.time())})
        except Exception as exc:
            print("[LP] Linprog crashed:", exc)
            break

        if not res.success:
            print("[LP] Solver failed:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        sum_a = a.sum()
        if sum_g <= 0 or sum_a <= 0:
            break

        # Rescale g to the same ℓ¹‑norm as a
        g_scaled = g * (sum_a / sum_g)

        # Convex combination
        new_a = (1.0 - mixing) * a + mixing * g_scaled
        # Renormalise (numerical drift)
        s = new_a.sum()
        if s <= 0:
            new_a = _uniform(n)
        else:
            new_a = new_a / s

        new_val = _objective(new_a)

        if new_val < best_val - 1e-12:
            best_val = new_val
            best_seq = new_a.copy()
            a = new_a
            print(f"[LP] improved to {new_val:.6f} (active constraints: {len(active_idx)})")
        else:
            # No improvement – stop the LP phase
            print(f"[LP] no further improvement (val={new_val:.6f}); terminating LP phase.")
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on permutations
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray, n: int, deadline: float,
                max_iters: int = 200_000, report_every: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Random swap moves on the permutation of the entries.
    Accepts a move only if it strictly decreases the exact objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        nxt = cur.copy()
        nxt[i], nxt[j] = nxt[j], nxt[i]          # swap two positions
        nxt_val = _objective(nxt)
        if nxt_val < cur_val - 1e-12:
            cur = nxt
            cur_val = nxt_val
            if nxt_val < best_val:
                best = nxt.copy()
                best_val = nxt_val
        if it % report_every == 0:
            print(f"[Hill] it={it:,d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence (list of floats) and stores it in the global
    name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    # --------------------------------------------------------------
    # Length – a compromise between expressiveness and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best if it exists.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Phase 1 – generate a rich pool of seeds and keep the top few.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (first 5 shown):")
    for i, (_, val) in enumerate(evaluated[:5], 1):
        print(f"  #{i}: {val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exact LP improvement on the current best seed.
    # --------------------------------------------------------------
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- LP improvement phase ---")
        seq_lp, val_lp = _lp_improve(global_best_seq, n, deadline,
                                     max_outer=20, mixing=0.5)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[LP] New global best = {val_lp:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – cheap hill‑climbing on permutations.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Hill‑climbing (swap) phase ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline,
                                     max_iters=300_000, report_every=40_000)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[Hill] New global best = {val_hc:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final polishing via a short simulated‑annealing run
    # (the same SA routine as the reference implementation).
    # --------------------------------------------------------------
    def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                     deadline: float,
                                     max_iters: int = 400_000,
                                     init_temp_factor: float = 0.03,
                                     min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
        rng = np.random.default_rng()
        cur = start_seq.copy()
        cur_val = _objective(cur)
        best_seq = cur.copy()
        best_val = cur_val
        T = max(init_temp_factor * cur_val, 1e-12)

        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break
            T = max(T * 0.9995, min_temp)
            move_type = rng.integers(0, 3)
            if move_type == 0:                     # swap two entries
                i, j = rng.choice(n, size=2, replace=False)
                new = cur.copy()
                new[i], new[j] = new[j], new[i]
            elif move_type == 1:                   # redistribute a random subset
                k = rng.integers(2, 9)
                idx = rng.choice(n, size=k, replace=False)
                total = cur[idx].sum()
                if total <= 0:
                    continue
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
                new = cur.copy()
                new[idx] = new_alloc
            else:                                   # pair‑transfer
                i, j = rng.choice(n, size=2, replace=False)
                total = cur[i] + cur[j]
                if total <= 0:
                    continue
                split = rng.dirichlet([1.0, 1.0]) * total
                new = cur.copy()
                new[i], new[j] = split[0], split[1]

            new_val = _objective(new)
            delta = new_val - cur_val
            if delta < 0.0 or rng.random() < np.exp(-delta / T):
                cur = new
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = new.copy()
                    if it % 50_000 == 0:
                        print(f"[SA] it={it:,d} best={new_val:.6f}")

        return best_seq, best_val

    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=400_000)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] Final global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final sanity check with the official evaluator
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("[Eval] evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
