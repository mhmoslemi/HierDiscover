"""
Propose a low‑value step function for `evaluate_sequence`.

The search proceeds in several stages:

1. **Systematic sweep** over a rich set of analytic families
   (uniform, triangular, trapezoidal, sine‑window, Tukey,
   beta‑distribution, power‑law, symmetric geometric, “polynomial‑abs”).
   The best few (default 12) candidates according to the exact objective
   are kept.

2. **Adam‑based smooth‑max optimisation**.
   The max‑convolution is approximated by a soft‑max with a temperature
   that starts very low (almost averaging) and is slowly increased
   to a large value (tight approximation of the true max).  All iterates
   stay on the probability simplex by working in log‑softmax space.
   Short runs are performed for every top candidate; a longer run is
   performed on the current global best.

3. **Upsampling & fine‑tuning**.
   The best sequence is linearly interpolated to a larger size
   (typically twice as many points) – this provides a finer discretisation
   that can further flatten the self‑convolution.  The upsampled vector
   is again optimised with a longer Adam run.

4. **Projected sub‑gradient descent** (optional polishing) and a final
   sanity check with the official `evaluate_sequence`.

The routine respects a time budget (default 1000 s) and stores the
result in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a probability vector.
    Returns 2·n·max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _poly_abs(n: int, p: float) -> np.ndarray:
    """a_i ∝ max(0, 1 - |i - centre|/centre)^p."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - (np.abs(i - centre) / centre) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 800_000,
                   beta_start: float = 0.2,
                   beta_end: float = 500.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Operates in the log‑softmax (logits) space, hence the simplex constraint
    is automatically satisfied.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = beta_start
    beta_mul = 1.0015  # slow increase
    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        beta_temp = min(beta_temp * beta_mul, beta_end)

        conv = _conv_fft(a, a)                  # linear conv (len 2n‑1)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # softmax over conv entries

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # transform to logits coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a simple Polyak‑type step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------


def _collect_candidates(n: int,
                       rng: np.random.Generator,
                       previous: np.ndarray | None,
                       top_k: int,
                       deadline: float) -> list[np.ndarray]:
    """
    Systematically explore analytic families (and random seeds) and retain
    the `top_k` candidates with smallest exact objective.
    """
    best = []  # list of (value, seq)

    def keep(val, seq):
        if len(best) < top_k:
            best.append((val, seq))
            best.sort(key=lambda x: x[0])
        elif val < best[-1][0]:
            best[-1] = (val, seq)
            best.sort(key=lambda x: x[0])

    # Deterministic families ----------------------------------------------------
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.1),
        lambda n: _trapezoidal(n, width_fraction=0.25),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _sine_window(n, gamma=3.0),
        lambda n: _tukey_window(n, alpha=0.3),
        lambda n: _tukey_window(n, alpha=0.6),
    ]
    for fn in families:
        seq = fn(n)
        keep(_objective(seq), seq)

    # Beta shape grid (extended ranges) ----------------------------------------
    alphas = np.linspace(0.2, 6.0, 22)
    betas = np.linspace(0.2, 6.0, 22)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            seq = _beta_shape(n, a, b)
            keep(_objective(seq), seq)

    # Sine window grid -----------------------------------------------------------
    gammas = np.linspace(0.5, 6.0, 26)
    for g in gammas:
        if time.time() > deadline:
            break
        seq = _sine_window(n, gamma=g)
        keep(_objective(seq), seq)

    # Power‑law grid -------------------------------------------------------------
    ps = np.linspace(0.0, 5.0, 26)
    for p in ps:
        if time.time() > deadline:
            break
        seq = _power_law(n, p)
        keep(_objective(seq), seq)

    # Geometric / symmetric geometric --------------------------------------------
    rs = np.linspace(0.2, 0.99, 25)
    for r in rs:
        if time.time() > deadline:
            break
        seq = _geom(n, r)
        keep(_objective(seq), seq)
        seq2 = _sym_geom(n, r)
        keep(_objective(seq2), seq2)

    # Polynomial‑abs families ----------------------------------------------------
    for p in np.linspace(1.0, 5.0, 17):
        if time.time() > deadline:
            break
        seq = _poly_abs(n, p)
        keep(_objective(seq), seq)

    # Random Dirichlet seeds -----------------------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 30 if conc < 0.01 else 15
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            keep(_objective(seq), seq)

    # Sparse random supports ------------------------------------------------------
    for _ in range(120):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)  # 3 … 30 non‑zero entries
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = weights
        keep(_objective(seq), seq)

    # Previous solution -----------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        seq = previous / previous.sum()
        keep(_objective(seq), seq)

    # Return sequences sorted by ascending objective
    return [seq for _, seq in best]

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a base length.  4000 is a good trade‑off; we will later
    # optionally up‑sample to a larger length.
    # ------------------------------------------------------------------
    n0 = 4000
    print(f"\n=== Starting search (base n={n0}, budget={budget_s}s) ===")

    # ------------------------------------------------------------------
    # Phase 0 – deterministic and random seed sweep (≈20 % of budget)
    # ------------------------------------------------------------------
    phase0_deadline = min(deadline, start_time + 0.20 * budget_s)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    print("\n--- Phase 0: candidate sweep ---")
    candidates = _collect_candidates(
        n0, rng, previous, top_k=12, deadline=phase0_deadline
    )
    print(f"Collected {len(candidates)} top candidates (objective values):")
    for i, seq in enumerate(candidates):
        print(f"  #{i+1}: {_objective(seq):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – short Adam refinements for each candidate
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Phase 1: short Adam refinements ---")
    for idx, seed_vec in enumerate(candidates):
        if time.time() > deadline:
            break
        print(f"[Candidate {idx+1}/{len(candidates)}]")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n0, deadline, max_iters=200_000,
            beta_start=0.1, beta_end=300.0
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after short Adam: {best_val:.6f}")

    # Fallback to the best raw candidate if nothing improved
    if best_seq is None:
        best_seq = candidates[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first candidate: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – a longer Adam optimisation on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: long Adam optimisation (base length) ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq, n0, deadline,
            max_iters=1_200_000,
            beta_start=0.05, beta_end=500.0
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after long Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – optional up‑sampling and fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        scale = 2                # double the resolution
        n1 = n0 * scale
        print(f"\n--- Phase 3: up‑sampling to n={n1} and fine‑tuning ---")
        x_old = np.linspace(0.0, 1.0, n0)
        x_new = np.linspace(0.0, 1.0, n1)
        up_seq = np.interp(x_new, x_old, best_seq)
        up_seq = up_seq / up_seq.sum()

        seq_adam, val_adam = _adam_optimize(
            up_seq, n1, deadline,
            max_iters=800_000,
            beta_start=0.05, beta_end=500.0
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            n0 = n1
            print(f"  Updated global best after up‑sampled Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected sub‑gradient descent for polishing (if time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: projected sub‑gradient polishing ---")
        seq_sub, val_sub = _subgrad_optimize(
            best_seq, n0, deadline, max_iters=500_000
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated global best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective reported by evaluator: {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
