def refine_sequence(seq, n, deadline):
    cur = seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    # parameters
    max_iters = 2000000  # as many as time permits
    # step sizes
    # We'll use small epsilon drawn from Beta distribution or uniform.
    # Accept if improvement, else reject.

    while time.time() < deadline:
        # Decide move type randomly
        move_type = rng.integers(0, 4)
        new = cur.copy()
        if move_type == 0:  # swap two
            i, j = rng.choice(n, size=2, replace=False)
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:  # transfer epsilon
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.uniform(0, min(cur[i], 0.01))
            # ensure small relative change
            new[i] -= eps
            new[j] += eps
        elif move_type == 2:  # redistribute subset
            k = rng.integers(2, 6)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new[idx] = new_alloc
        else: # random perturb of a few entries
            # pick random subset and apply Gaussian noise then reproject to simplex
            k = rng.integers(2, 10)
            idx = rng.choice(n, size=k, replace=False)
            noise = rng.normal(scale=0.01, size=k)
            new[idx] = np.maximum(new[idx] + noise, 0.0)
            # reproject to simplex
            new = new / new.sum()
        # evaluate new
        new_val = _objective(new)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (0.0001 + 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val
        # optionally occasional reordering
        if rng.random() < 0.001:
            alt = _alternate_order(cur)
            alt_val = _objective(alt)
            if alt_val < cur_val:
                cur = alt
                cur_val = alt_val
                if alt_val < best_val:
                    best = alt.copy()
                    best_val = alt_val

    return best, best_val

def _sidon_sparse_seed(n):
    # generate indices using greedy algorithm
    S = []
    sums = set()
    for i in range(n):
        # test if adding i introduces duplicate sums
        duplicate = False
        for s in S:
            if (i + s) in sums:
                duplicate = True
                break
        if not duplicate and (i + i) not in sums:
            # add i
            for s in S:
                sums.add(i + s)
            sums.add(i + i)
            S.append(i)
        # stop if size close to sqrt(2*n)
        if len(S) >= int(np.sqrt(2 * n)):
            break
    if not S:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    weight = 1.0 / len(S)
    for idx in S:
        a[idx] = weight
    return a / a.sum()

"""
Improved search for a step‑function that minimises

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm builds a diverse pool of analytic seeds (including a
greedy Sidon‑set seed), runs a short Adam optimisation on the
log‑sum‑exp surrogate, then performs a long multiplicative mirror‑
descent on the exact max‑convolution objective.  After that a
projected additive sub‑gradient descent is applied for fine‑tuning,
followed by a quick random‑permutation search.  The best sequence
found is stored in the global name ``height_sequence_1`` and also
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------
def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, … (then normalised)."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Misc random smooth curves
# ----------------------------------------------------------------------
def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Sparse Sidon‑set seed (greedy construction)
# ----------------------------------------------------------------------
def _greedy_sidon_seed(n: int) -> np.ndarray:
    """
    Greedy construction of a Sidon set (all pairwise sums distinct)
    and equal weight on its elements.
    """
    S = []
    sums = set()
    for i in range(n):
        conflict = False
        for s in S:
            if (i + s) in sums:
                conflict = True
                break
        if conflict or (i + i) in sums:
            continue
        # add i to the set
        for s in S:
            sums.add(i + s)
        sums.add(i + i)
        S.append(i)
        # stop when we have roughly √n elements (theoretical bound)
        if len(S) >= int(np.sqrt(n)):
            break
    a = np.zeros(n, dtype=np.float64)
    if S:
        w = 1.0 / len(S)
        a[S] = w
    return a / a.sum() if a.sum() > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed pool construction (richer than the reference version)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # sine window across a fine grid of gamma
    for gamma in np.linspace(0.5, 6.0, 12):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussian with a finer sigma grid
    for sigma in np.linspace(0.10, 0.55, 10):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # power law up to p=8 (coarse grid) and alternating order
    for p in np.linspace(0.0, 8.0, 17):
        seeds.append(_power(n, p))
        seeds.append(_alternate_order(_power(n, p)))

    # beta family covering a wide range, including very U‑shaped cases
    for _ in range(15):
        alpha = 10 ** rng.uniform(-2, 0.6)   # between 0.01 and ~4
        beta = 10 ** rng.uniform(-2, 0.6)
        seeds.append(_beta_shape(n, alpha, beta))
        seeds.append(_alternate_order(_beta_shape(n, alpha, beta)))

    # exponential shapes (rates covering several decades)
    for rate in np.logspace(-4, -1, 12):
        seeds.append(_exponential(n, rate))
        seeds.append(_alternate_order(_exponential(n, rate)))

    # logarithmic shapes (different epsilons)
    for eps in np.logspace(-4, -1, 8):
        seeds.append(_logarithmic(n, eps))
        seeds.append(_alternate_order(_logarithmic(n, eps)))

    # Random Dirichlet draws – different concentration parameters and orderings
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_desc(raw)[::-1])
            seeds.append(_alternate_order(raw))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))
            seeds.append(_alternate_order(_lowfreq_curve(n, nf, rng)))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))
            seeds.append(_alternate_order(_mix_shapes(_uniform(n), curve, mix_alpha)))

    # Greedy Sidon‑set seed (sparse, equal weights)
    seeds.append(_greedy_sidon_seed(n))
    seeds.append(_alternate_order(_greedy_sidon_seed(n)))

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 150_000) -> tuple[np.ndarray | None, float]:
    """Adam on logits using a temperature‑controlled log‑sum‑exp surrogate."""
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    # smoothing temperature schedule
    beta = 0.5
    beta_inc = 0.5
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the true max‑conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20  # larger initial step than the reference

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient (active index)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional re‑ordering – large‑small alternation often drops the peak
        if it % 50_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {val_alt:.6f}")

    return best_seq, best_val


def _subgradient_descent_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Additive projected sub‑gradient descent on the exact max‑convolution.
    Uses a decreasing step size and Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.30  # initial learning rate

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active indices (tolerance for numerical noise)
        tol = max_val * 1e-12 + 1e-15
        act = np.where(conv >= max_val - tol)[0]

        # gradient: average of sub‑gradients for all active indices
        g = np.zeros_like(a)
        for k in act:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += a[j_range]
        g = 2.0 * g / max(1, len(act))

        lr = lr0 / np.sqrt(it)
        a_new = a - lr * g
        a_new = _proj_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < best_val:
            best_val = new_val
            best_seq = a_new.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad] it={it:7d} best={new_val:.6f}")

        a = a_new

    return best_seq, best_val


def _random_permutation_improve(start_seq: np.ndarray, n: int,
                               deadline: float,
                               attempts: int = 5000) -> tuple[np.ndarray, float]:
    """
    Try many random reorderings of the *same* multiset of values.
    Keeps the best ordering found.
    """
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)
    rng = np.random.default_rng()
    for _ in range(attempts):
        if time.time() > deadline:
            break
        perm = rng.permutation(n)
        cand = best_seq[perm]
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    return best_seq, best_val

        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
