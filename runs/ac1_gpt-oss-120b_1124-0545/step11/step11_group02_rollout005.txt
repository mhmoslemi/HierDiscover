"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises ``evaluate_sequence``.

The algorithm consists of three phases:

1. **Seed generation** – a modest pool of analytic families (uniform,
   triangular, geometric, beta, Gaussian, Tukey, etc.) together with the
   previous best solution is assembled and normalised to sum = 1.
   The best surrogate (2·n·max conv) seed is selected.

2. **Smooth‑max Adam optimisation** – starting from the best seed we
   minimise a smooth approximation of the ℓ∞‑convolution objective.
   For a given temperature ``α`` the weights
   ``w_k ∝ exp(α·conv_k)`` define a softmax over the convolution entries.
   The gradient of the weighted sum is computed analytically via a single
   FFT‑based convolution (no explicit O(n²) loops).  An Adam optimiser
   with Euclidean projection onto the probability simplex updates the
   vector while ``α`` is slowly increased, eventually approximating the
   true max‑convolution sub‑gradient.  The best vector seen during the
   run is retained.

3. **Stochastic polishing** – a light simulated‑annealing (SA) loop
   randomly reshuffles a small subset of coordinates (Dirichlet move) and
   accepts downhill moves or uphill moves with a geometric temperature
   schedule.  This helps escape tiny plateaus left by the deterministic
   optimiser.

All steps respect the overall time budget (default 1000 s).  The final
vector is returned as a plain ``list[float]`` and stored in the global
variable ``height_sequence_2`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12),
                             np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # Random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Additional random mixtures to increase diversity
    for _ in range(8):
        lam = rng.random()
        a = rng.random()
        mix = lam * seeds[0] + (1 - lam) * _random_dirichlet(n, a + 0.01, rng)
        mix = mix / mix.sum()
        seeds.append(mix)

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised → 2·n·max conv)

def _objective(seq: np.ndarray) -> float:
    """Exact surrogate evaluation for sequences with sum = 1."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth‑max gradient (softmax weighting of convolution entries)
# ----------------------------------------------------------------------


def _smooth_max_grad(seq: np.ndarray, alpha: float):
    """
    Returns the full convolution (seq*seq) and the gradient of the
    soft‑maxed objective  Σ w_k·conv_k  where w_k ∝ exp(alpha·conv_k).
    """
    n = seq.shape[0]

    # 1. convolution of seq with itself
    conv = _conv_fft(seq, seq)               # length L = 2n‑1
    max_c = conv.max()

    # 2. softmax weights (numerically stable)
    exp_arg = alpha * (conv - max_c)
    w_raw = np.exp(exp_arg)
    w = w_raw / w_raw.sum()                  # shape (L,)

    # 3. gradient: g_j = 2 * Σ_i seq_i * w_{i+j}
    #    This is a correlation of seq with w, which we obtain via one FFT‑conv
    w_rev = w[::-1]                          # reverse the weight vector
    conv_a_wr = _conv_fft(seq, w_rev)        # length n + L - 1

    L = conv.shape[0]                        # = 2n‑1
    idx = L - 1 - np.arange(n)               # select the appropriate slice
    grad = 2.0 * conv_a_wr[idx]               # shape (n,)

    return conv, grad


# ----------------------------------------------------------------------
# Adam optimiser for the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   init_lr: float = 0.02,
                   init_alpha: float = 0.10,
                   alpha_mul: float = 1.001,
                   max_alpha: float = 80.0,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    The temperature ``alpha`` is steadily increased so that the
    distribution concentrates on the current maximal convolution entry.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    alpha = init_alpha

    while time.time() < deadline and t < max_iters:
        t += 1
        conv, grad = _smooth_max_grad(a, alpha)

        cur_val = float(2.0 * n * conv.max())
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 5000 == 0:
                print(f"[Adam] it={t:7d} val={cur_val:.6f} best={best_val:.6f} α={alpha:.3f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _proj_simplex(a)

        # increase temperature (smooth‑max becomes sharper)
        alpha = min(alpha * alpha_mul, max_alpha)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (lightweight)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple SA that reshuffles a random subset of coordinates (Dirichlet move).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)               # affect 2–12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    The best sequence is returned as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Retrieve possible previous solution (named height_sequence_2)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_2")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Choose length (4000 is a good trade‑off, power of two is not required)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Phase 1 – Seed collection & selection
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        adam_deadline = time.time() + 0.70 * time_left
        print("\n--- Smooth‑max Adam optimisation ---")
        seq_adam, val_adam = _adam_optimize(
            global_best_seq,
            n,
            adam_deadline,
            init_lr=0.02,
            init_alpha=0.10,
            alpha_mul=1.001,
            max_alpha=80.0,
            max_iters=5_000_000
        )
        if val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam, val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            max_iters=800_000,
            init_temp_factor=0.04
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible later reuse
    globals()["height_sequence_2"] = global_best_seq.tolist()
    return global_best_seq.tolist()
