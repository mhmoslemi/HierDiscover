def _adam_optimize(start_vec, n, deadline, max_iters=1_200_000):
    eps = 1e-30
    v = np.log(start_vec + eps)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001
    beta_max = 1000.0 # maybe 1000 to be more sharp

    best_seq = None
    best_val = float('inf')

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(v)   # shape n
        # compute conv once
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # optional print every 100k

        # update temperature
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # compute soft weights for smooth max
        shift = max_conv   # shift for numerical stability
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to uniform weights
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of weighted sum conv w.r.t a:
        rev_a = a[::-1]
        # conv of rev_a with w yields length len(a)+len(w)-1 = 2n-1 + 2n-1 -1? Wait w length = 2n-1.
        # But we only need a part of length n: convolution rev_a * w yields length n+(2n-1)-1 = 3n-2.
        # full = _conv_fft(rev_a, w) -> get vector length 3n-2.
        # gradient = 2 * full[n-1:n-1+n] as earlier.
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        # convert to gradient w.r.t logits:
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip norm
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update:
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.8 / np.sqrt(it)  # as before

        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)

        m_hat = m / (1 - beta1 ** it)
        v2_hat = v2 / (1 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)
```

"""
Propose a step‐function (non‑negative sequence) that minimizes
evaluate_sequence().  The algorithm is a multi‑phase heuristic:

1. **Deterministic / random candidate sweep** – a variety of analytic
   families (uniform, triangular, trapezoidal, beta, sine, Tukey,
   power‑law, geometric, symmetric geometric, Gaussian, bimodal gap)
   together with many random Dirichlet seeds.  The 𝑘 best candidates
   (smallest exact objective) are retained.

2. **Short Adam runs** – each retained candidate is refined by an
   Adam optimiser that works on the *log‑softmax* parametrisation
   (so the ℓ₁‑simplex is always satisfied).  The smooth‑max surrogate
   uses an exponentially increasing temperature and the gradient is
   obtained via a single FFT‑based convolution per iteration
   (the exact max value is read from the same convolution, avoiding a
   second FFT).

3. **Long Adam on the current best** – the best sequence from phase 2
   is further optimised with many more Adam iterations.

4. **Projected sub‑gradient descent** – optimisation of the true
   max‑convolution objective using the Polyak step‑size rule.

5. **Simulated annealing polishing** – random Dirichlet “mass‑re‑
   allocation’’ moves are accepted with a decreasing temperature schedule.

All phases respect the overall time budget (default 1000 s) and the
final sequence is stored in the global variable ``height_sequence_1``.
The implementation runs on the CPU only and uses only NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation of the objective (sum = 1)."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_window(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _bimodal_gap_window(n: int, gap_frac: float) -> np.ndarray:
    """Two uniform blocks at the ends leaving a central gap."""
    if not (0.0 < gap_frac < 1.0):
        return _uniform(n)
    gap_len = int(round(gap_frac * n))
    side_len = (n - gap_len) // 2
    if side_len == 0:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    a[:side_len] = 1.0
    a[-side_len:] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 1_200_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    The sequence stays on the probability simplex via the soft‑max of logits.
    Returns (best_sequence, best_objective) where best_objective is the
    exact value from evaluate_sequence (i.e. 2·n·max(conv)).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001
    beta_max = 800.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex

        # ---- one FFT convolution gives both max and surrogate weights ----
        conv = _conv_fft(a, a)               # length 2n‑1
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress print
            if it % 200_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # ---- smooth‑max weight vector ------------------------------------
        beta_temp = min(beta_temp * beta_mul, beta_max)
        shift = max_conv
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a -------------------------
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ---- transform to log‑softmax coordinates -----------------------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- gradient clipping ------------------------------------------
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update ------------------------------------------------
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.8 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a Polyak step size that references the current best value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple annealing: repeatedly redistributes the total mass of a random
    subset of coordinates (Dirichlet move).  Accepts uphill moves with
    probability exp(-Δ / T).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)                # size of the moved set
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

def _collect_candidates(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None,
                       top_k: int,
                       deadline: float) -> list[np.ndarray]:
    """
    Systematically explore analytic families and retain the `top_k`
    candidates with smallest exact objective.
    """
    best = []      # list of (value, seq)

    def keep(val, seq):
        if len(best) < top_k:
            best.append((val, seq))
            best.sort(key=lambda x: x[0])
        elif val < best[-1][0]:
            best[-1] = (val, seq)
            best.sort(key=lambda x: x[0])

    # ---- deterministic families -------------------------------------------------
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.12),
        lambda n: _trapezoidal(n, width_fraction=0.25),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _sine_window(n, gamma=3.0),
        lambda n: _tukey_window(n, alpha=0.3),
        lambda n: _tukey_window(n, alpha=0.6),
    ]
    for fn in families:
        seq = fn(n)
        keep(_objective(seq), seq)

    # ---- beta shape grid -------------------------------------------------------
    alphas = np.linspace(0.1, 5.0, 20)
    betas = np.linspace(0.1, 5.0, 20)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            seq = _beta_shape(n, a, b)
            keep(_objective(seq), seq)

    # ---- sine window grid -------------------------------------------------------
    gammas = np.linspace(0.5, 5.0, 20)
    for g in gammas:
        if time.time() > deadline:
            break
        seq = _sine_window(n, gamma=g)
        keep(_objective(seq), seq)

    # ---- tukey window grid ------------------------------------------------------
    alps = np.linspace(0.1, 0.9, 15)
    for a in alps:
        if time.time() > deadline:
            break
        seq = _tukey_window(n, alpha=a)
        keep(_objective(seq), seq)

    # ---- power‑law ---------------------------------------------------------------
    ps = np.linspace(0.0, 5.0, 15)
    for p in ps:
        if time.time() > deadline:
            break
        seq = _power(n, p)
        keep(_objective(seq), seq)

    # ---- geometric ---------------------------------------------------------------
    rs = np.linspace(0.2, 0.98, 15)
    for r in rs:
        if time.time() > deadline:
            break
        seq = _geom(n, r)
        keep(_objective(seq), seq)
        seq2 = _sym_geom(n, r)
        keep(_objective(seq2), seq2)

    # ---- Gaussian windows --------------------------------------------------------
    sigmas = np.linspace(0.1 * n, 0.4 * n, 12)
    for sigma in sigmas:
        if time.time() > deadline:
            break
        seq = _gaussian_window(n, sigma)
        keep(_objective(seq), seq)

    # ---- Bimodal gap windows ----------------------------------------------------
    gaps = np.linspace(0.1, 0.6, 12)
    for gap in gaps:
        if time.time() > deadline:
            break
        seq = _bimodal_gap_window(n, gap)
        keep(_objective(seq), seq)

    # ---- random Dirichlet seeds --------------------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 40 if conc < 0.01 else 20
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            keep(_objective(seq), seq)

    # ---- sparse random supports -------------------------------------------------
    for _ in range(120):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)               # 3 … 30 non‑zero entries
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = weights
        keep(_objective(seq), seq)

    # ---- previous solution -------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        seq = previous / previous.sum()
        keep(_objective(seq), seq)

    # Return only the sequences, sorted by ascending objective
    return [seq for _, seq in best]

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    evaluate_sequence().  The routine respects the time budget and stores the
    resulting best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    final_deadline = start_time + budget_s - 1.0    # small safety margin

    # -------------------- configuration ---------------------------------------
    n = 4000                          # length of the sequence
    top_k = 20                        # keep this many candidates from phase 0

    # -------------------- Phase 0 – candidate sweep ---------------------------
    phase0_deadline = min(final_deadline, start_time + 0.07 * budget_s)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    print(f"\n=== Starting search (n={n}, budget={budget_s}s) ===")
    print("\n--- Phase 0: deterministic families + random seeds ---")
    candidates = _collect_candidates(
        n, rng, previous, top_k=top_k, deadline=phase0_deadline
    )
    print(f"Collected {len(candidates)} top candidates (objective values):")
    for i, seq in enumerate(candidates):
        print(f"  #{i+1}: {_objective(seq):.6f}")

    # -------------------- Phase 1 – short Adam runs -------------------------
    phase1_deadline = min(final_deadline,
                          start_time + 0.37 * budget_s)   # 0.07 + 0.30
    best_seq = None
    best_val = float("inf")
    print("\n--- Phase 1: short Adam refinements ---")
    for idx, seed_vec in enumerate(candidates):
        if time.time() > phase1_deadline:
            break
        print(f"[Candidate {idx+1}/{len(candidates)}]")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, phase1_deadline, max_iters=180_000
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after short Adam: {best_val:.6f}")

    # Fallback if nothing improved
    if best_seq is None:
        best_seq = candidates[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first candidate: {best_val:.6f}")

    # -------------------- Phase 2 – long Adam on best ------------------------
    phase2_deadline = min(final_deadline,
                          start_time + 0.67 * budget_s)   # +0.30
    print("\n--- Phase 2: long Adam optimisation ---")
    seq_long, val_long = _adam_optimize(
        best_seq, n, phase2_deadline, max_iters=2_200_000
    )
    if seq_long is not None and val_long < best_val:
        best_seq = seq_long.copy()
        best_val = val_long
        print(f"  Updated global best after long Adam: {best_val:.6f}")

    # -------------------- Phase 3 – projected sub‑gradient -------------------
    phase3_deadline = min(final_deadline,
                          start_time + 0.92 * budget_s)   # +0.25
    print("\n--- Phase 3: projected sub‑gradient descent ---")
    seq_sub, val_sub = _subgrad_optimize(
        best_seq, n, phase3_deadline, max_iters=800_000
    )
    if val_sub < best_val:
        best_seq = seq_sub.copy()
        best_val = val_sub
        print(f"  Updated global best after sub‑grad: {best_val:.6f}")

    # -------------------- Phase 4 – simulated annealing ----------------------
    print("\n--- Phase 4: simulated annealing polishing ---")
    seq_sa, val_sa = _simulated_annealing(
        best_seq, n, final_deadline, max_iters=300_000, init_temp_factor=0.04
    )
    if val_sa < best_val:
        best_seq = seq_sa.copy()
        best_val = val_sa
        print(f"  Updated global best after SA: {best_val:.6f}")

    # -------------------- Final sanity check ---------------------------------
    try:
        final_eval = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal evaluate_sequence return value: {final_eval:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
