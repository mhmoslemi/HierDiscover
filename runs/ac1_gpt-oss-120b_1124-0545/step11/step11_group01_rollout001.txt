def projected_subgradient_optimize(start_seq, n, deadline, max_iters=500_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    lr = 0.5
    for it in range(1,max_iters+1):
        if time.time()>deadline: break
        # compute conv and max idx
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0*n*conv[max_idx]  # equal to max conv when sum=1? Actually sum=1 => denominator 1; but now we normalized sum=1 not sqrt(2n).             best_val = cur_val
            best_seq = a.copy()
        # compute subgradient g
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)

"""
Propose a step‑function that minimises

    2·n·max(conv(seq,seq)) / (sum(seq)^2)

The implementation follows a multi‑stage pipeline:

* **Seed generation** – deterministic analytic families, random Dirichlet
  draws, multi‑modal Gaussian mixtures and low‑frequency random curves,
  all ℓ₁‑normalised.  The number of seeds is kept modest to stay within the
  time budget.

* **Surrogate optimisation** – Adam on the log‑softmax parametrisation
  with a temperature‑controlled log‑sum‑exp approximation of the
  max‑convolution.

* **Exact sub‑gradient descent** – two complementary variants:
    – multiplicative (mirror‑descent) updates (as used previously);
    – additive updates with Euclidean projection onto the probability
      simplex.

  Both are run on the top seeds; periodic “alternating” re‑ordering
  (large, small, 2nd‑large, …) is applied to break peaks.

* **Permutation‑only simulated annealing** – starting from the best
  sequence, a long SA run that only swaps positions (the multiset of
  values stays fixed).  This explores the ordering space more thoroughly
  than the previous SA, which also changed magnitudes.

The whole search respects a global deadline (default 1000 s).  The best
sequence found (as a plain list of floats) is stored in the global name
``height_sequence_1`` and returned.

The code uses only NumPy and the given ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used during the search.
    The sequence is assumed to be ℓ₁‑normalised (sum = 1),
    therefore the value equals ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _multi_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of a few Gaussian bumps."""
    k = rng.integers(2, 6)          # number of components
    seq = np.zeros(n, dtype=np.float64)
    for _ in range(k):
        centre = rng.integers(0, n)
        sigma_frac = rng.uniform(0.02, 0.10)
        sigma = sigma_frac * n
        i = np.arange(n, dtype=np.float64)
        comp = np.exp(-((i - centre) ** 2) / (2.0 * sigma ** 2))
        seq += comp
    s = seq.sum()
    return seq / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Ordering helpers
# ----------------------------------------------------------------------
def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


def _spacing_order(vec: np.ndarray, fraction: float = 0.20) -> np.ndarray:
    """
    Places the top ``fraction`` of values at evenly spaced positions,
    fills the remaining slots with the rest in descending order.
    """
    n = len(vec)
    desc = np.sort(vec)[::-1]
    L = max(1, int(fraction * n))
    big = desc[:L]
    small = desc[L:]
    out = np.empty_like(vec)
    # evenly spaced indices for the big values
    positions = np.linspace(0, n - 1, L, dtype=int)
    out[positions] = big
    mask = np.ones(n, dtype=bool)
    mask[positions] = False
    out[mask] = small
    s = out.sum()
    return out / s if s > 0 else _uniform(n)

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # sine‑window family
    for gamma in np.linspace(0.5, 6.0, 9):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussian family
    for sigma in np.linspace(0.10, 0.55, 8):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # power‑law family
    for p in np.linspace(0.0, 8.0, 13):
        seeds.append(_power(n, p))
        seeds.append(_alternate_order(_power(n, p)))

    # beta family
    for _ in range(12):
        alpha = 10 ** rng.uniform(-2, 0.6)
        beta = 10 ** rng.uniform(-2, 0.6)
        shape = _beta_shape(n, alpha, beta)
        seeds.append(shape)
        seeds.append(_alternate_order(shape))

    # exponential family
    for rate in np.logspace(-4, -1, 10):
        seeds.append(_exponential(n, rate))
        seeds.append(_alternate_order(_exponential(n, rate)))

    # logarithmic family
    for eps in np.logspace(-4, -1, 6):
        shape = _logarithmic(n, eps)
        seeds.append(shape)
        seeds.append(_alternate_order(shape))

    # Dirichlet draws with varied concentrations
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))
            seeds.append(_spacing_order(raw, fraction=0.15))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            c = _lowfreq_curve(n, nf, rng)
            seeds.append(c)
            seeds.append(_alternate_order(c))
            seeds.append(_spacing_order(c, fraction=0.25))

    # Random multi‑Gaussian mixtures
    for _ in range(10):
        seeds.append(_multi_gaussian(n, rng))
        seeds.append(_alternate_order(_multi_gaussian(n, rng)))
        seeds.append(_spacing_order(_multi_gaussian(n, rng), fraction=0.2))

    # mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            mix = _mix_shapes(_uniform(n), curve, mix_alpha)
            seeds.append(mix)
            seeds.append(_alternate_order(mix))
            seeds.append(_spacing_order(mix, fraction=0.18))

    # previous best (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on log‑softmax parameters, using a temperature‑controlled
    log‑sum‑exp surrogate for max(conv).  Returns the best feasible
    vector seen (ℓ₁‑normalised) and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 0.5          # initial temperature
    beta_inc = 0.5
    beta_max = 400.0    # higher final temperature for sharper max

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)        # exact objective (since sum=1)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smoothed max via log‑sum‑exp ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the true
    max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20  # initial learning rate

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional re‑ordering
        if it % 50_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {val_alt:.6f}")

    return best_seq, best_val


def _additive_subgradient_optimize(start_seq: np.ndarray, n: int,
                                  deadline: float,
                                  max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Projected additive sub‑gradient descent on the exact max‑convolution.
    Uses Euclidean projection onto the simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Additive] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        step = 0.8 / np.sqrt(it)

        # simple backtracking line search (few attempts)
        for _ in range(5):
            cand = a - step * g
            cand = _proj_simplex(cand)
            cand_val = _objective(cand)
            if cand_val <= cur_val:
                a = cand
                break
            step *= 0.5
        else:
            # if no improvement, keep current a
            pass

        # occasional alternating re‑ordering as a heuristic jump
        if it % 30_000 == 0:
            alt = _alternate_order(a)
            alt_val = _objective(alt)
            if alt_val < cur_val:
                a = alt
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()
                    print(f"[Additive‑alt] it={it:7d} improved alt to {alt_val:.6f}")

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 1_200_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that only permutes the multiset of values.
    Useful to explore the ordering space deeper than the previous SA.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9997, min_temp)

        # swap two positions
        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA‑perm] it={it:7d} best={new_val:.6f}")

        # every now and then try a “large‑small” block swap
        if it % 200_000 == 0:
            block = int(0.05 * n)
            if block >= 2:
                idx = rng.choice(n, size=block, replace=False)
                perm = np.random.permutation(block)
                alt = cur.copy()
                alt[idx] = alt[idx][perm]
                alt_val = _objective(alt)
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = alt.copy()
                    cur = alt
                    cur_val = alt_val
                    print(f"[SA‑block] it={it:7d} improved block to {alt_val:.6f}")

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the official
    ``evaluate_sequence`` objective.  The function respects the time
    budget, returns the best sequence (list of floats) and stores it in
    the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0                 # safety margin

    # ------------------------------------------------------------------
    # Choose a length that balances quality and speed.
    # If we have > 300 s from the start we try a longer sequence.
    # ------------------------------------------------------------------
    if budget_s >= 300:
        n = 8000
    else:
        n = 4000
    print(f"\n=== Starting search (length = {n}) ===")

    # ------------------------------------------------------------------
    # Load previous best if it exists.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – seed generation & quick ranking
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # dynamic iteration budget (≈ 45 iters per second)
        max_iters = min(250_000, int(remaining * 45))
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # Try an alternating re‑ordering of the Adam result
        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq = alt
                global_best_val = alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Mirror‑descent sub‑gradient optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=1_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Additive projected sub‑gradient optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Additive projected sub‑gradient optimisation ---")
        seq_add, val_add = _additive_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=400_000
        )
        if val_add < global_best_val:
            global_best_seq = seq_add.copy()
            global_best_val = val_add
            print(f"[overall‑Additive] new global best = {val_add:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated‑annealing polishing (ordering only)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing (ordering) polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=1_200_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
