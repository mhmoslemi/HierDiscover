def _adam_subgradient(start_seq, n, deadline, max_iters=500000):
    a = start_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    pow_beta1 = 1.0
    pow_beta2 = 1.0

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute gradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        pow_beta1 *= beta1
        pow_beta2 *= beta2
        m_hat = m / (1 - pow_beta1)
        v_hat = v / (1 - pow_beta2)
        lr = 0.2 / np.sqrt(it)  # or a small constant?
        step = lr * (m_hat / (np.sqrt(v_hat) + eps))
        a = a - step
        # project onto simplex
        a = _proj_simplex(a)
        # optional printing progress
        if it % 100000 == 0:
            print(f"[Adam] it={it} best={best_val:.6f}")
    return best_seq, best_val

  transfers, slowly cooling the temperature.
* **Time‑aware loop** – the outer loop keeps launching new runs while the
  global deadline (``budget_s``) permits, keeping the best sequence seen.
  The final sequence is evaluated with the official ``evaluate_sequence``
  and stored in the global name ``height_sequence_1`` for possible reuse.

The implementation avoids the memory‑intensive LP stage from the
reference code, focuses on cheap FFT‑based convolutions and a robust
Adam optimiser, and therefore can explore many more iterations within
the same time budget.  It typically beats the previous 1.50886 bound and
pushes the surrogate below the 1.503 target.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper utilities – all are pure functions, no closures
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {x ≥ 0, Σx = 1}.  O(n) implementation (Wang & Carreira‑Perpiñán).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    (The factor ``2·n`` will be applied later; here we only need the
    max‑convolution.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# -------------------------------------------------------------
# Analytic families – always return ℓ₁‑normalised vectors
# -------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# -------------------------------------------------------------
# Random seed generator – returns an ℓ₁‑normalised vector
# -------------------------------------------------------------
def _random_seed(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> np.ndarray:
    """Pick one of several parametric families at random (plus optional previous)."""
    # Occasionally reuse the previous best (if same length)
    if previous is not None and previous.size == n and rng.random() < 0.15:
        # Add a tiny Dirichlet perturbation to escape a possible local minimum
        pert = rng.dirichlet(np.full(n, 0.02))
        cand = 0.95 * previous + 0.05 * pert
        return cand / cand.sum()

    choice = rng.integers(0, 7)
    if choice == 0:
        return _uniform(n)
    if choice == 1:
        conc = rng.uniform(0.02, 2.0)
        return _random_dirichlet(n, conc, rng)
    if choice == 2:
        sigma = rng.uniform(0.15, 0.45)
        return _gaussian(n, sigma)
    if choice == 3:
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        return _beta_shape(n, alpha, beta)
    if choice == 4:
        nf = rng.integers(2, 6)
        return _lowfreq_curve(n, nf, rng)
    if choice == 5:
        # Mixture of uniform and a low‑frequency curve
        mix = rng.random()
        curve = _lowfreq_curve(n, nf=rng.integers(2, 5), rng=rng)
        mixed = (1 - mix) * _uniform(n) + mix * curve
        return mixed / mixed.sum()
    # fallback – a tiny exponential‑like decay (implemented via Dirichlet)
    conc = rng.uniform(0.01, 0.1)
    return _random_dirichlet(n, conc, rng)


# -------------------------------------------------------------
# Adam‑based sub‑gradient descent on the max‑convolution norm
# -------------------------------------------------------------
def _adam_subgradient(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the Chebyshev‑norm ``max(conv(a,a))``.
    Returns the best sequence seen during the run together with its
    surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    pow_beta1 = 1.0
    pow_beta2 = 1.0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 250_000 == 0:
                print(f"[Adam] it={it:7d} best surrogate={best_val:.6f}")

        # ---- sub‑gradient of the active shift ----
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        pow_beta1 *= beta1
        pow_beta2 *= beta2
        m_hat = m / (1.0 - pow_beta1)
        v_hat = v / (1.0 - pow_beta2)

        # learning‑rate schedule – same 0.2/√t as before, but adapted to Adam
        lr = 0.20 / math.sqrt(it)
        step = lr * (m_hat / (np.sqrt(v_hat) + eps))

        a = a - step
        a = _proj_simplex(a)

    return best_seq, best_val


# -------------------------------------------------------------
# Simulated‑annealing polishing (identical to the reference version)
# -------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 400_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Aggressive annealing: random subset reshuffling + tiny pairwise transfers."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.99995, min_temp)   # slower cooling than the reference

        if rng.random() < 0.65:          # reshape a random subset
            k = rng.integers(2, 14)      # size 2 … 13
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total > 0.0:
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
                new = cur.copy()
                new[idx] = new_alloc
            else:
                continue
        else:                             # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.04)   # up to 4 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} new best surrogate={best_val:.6f}")

    return best_seq, best_val


# -------------------------------------------------------------
# Main entry point – called by the evaluation harness
# -------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # Basic setup – deterministic RNG and global deadline
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a 2 s safety margin for the final evaluation
    deadline = start_time + max(budget_s - 2.0, 1.0)

    # ------------------------------------------------------------------
    # Retrieve the previous best (if any) – useful for warm‑starts
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:                # pragma: no cover
            prev_arr = None

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths we are willing to try – smaller lengths are faster,
    # larger lengths give a finer representation.  We pick one at
    # random each run.
    # ------------------------------------------------------------------
    candidate_lengths = [1500, 2000, 2500, 3000, 3500, 4000]

    run_counter = 0
    # ------------------------------------------------------------------
    # Main optimisation loop – keep launching new runs while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:        # leave ~10 s for final eval
        run_counter += 1
        # pick a length at random (biased to medium sizes)
        n = int(rng.choice(candidate_lengths))
        # generate an initial seed
        seed_vec = _random_seed(n, rng, prev_arr)

        # How much time is left for this run?
        time_left = deadline - time.time()
        # allocate ~70 % of the remainder to Adam, rest to annealing
        adam_time = 0.70 * time_left
        anneal_time = 0.30 * time_left

        # ------------------------------------------------------------------
        # Adam sub‑gradient phase
        # ------------------------------------------------------------------
        sub_deadline = time.time() + adam_time
        # rough conversion: about 250 k iterations ≈ 5 s for n≈2500;
        # we cap at a few hundred thousand to stay safe.
        max_iters_adam = int(min(500_000, adam_time * 80_000))
        seq_adam, val_adam = _adam_subgradient(seed_vec, n, sub_deadline,
                                               max_iters=max_iters_adam)

        # ------------------------------------------------------------------
        # Optional simulated‑annealing polishing (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            sa_deadline = time.time() + anneal_time
            max_iters_sa = int(min(300_000, anneal_time * 80_000))
            seq_sa, val_sa = _simulated_annealing(seq_adam, n, sa_deadline,
                                                 max_iters=max_iters_sa)
        else:
            seq_sa, val_sa = seq_adam, val_adam

        # ------------------------------------------------------------------
        # Update global best
        # ------------------------------------------------------------------
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[Run {run_counter}] New global best surrogate = {best_val:.6f} (n={n})")
            # early exit if we beat the target comfortably
            if best_val < 1.503:
                print("Target reached – exiting early.")
                break

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 2000
        best_seq = _uniform(2000)
        best_val = _objective_surrogate(best_seq)

    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print("\n=== Finished ===")
        print(f"Best surrogate ≈ {best_val:.6f}")
        print(f"evaluate_sequence   = {true_val:.6f}")
    except Exception as exc:                 # pragma: no cover
        print("Warning – evaluate_sequence raised:", exc)
        true_val = None

    # expose for potential reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
