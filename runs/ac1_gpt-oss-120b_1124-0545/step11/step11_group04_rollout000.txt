    from scipy.signal import windows as _scipy_windows
except Exception:                     # pragma: no cover
    _scipy_windows = None


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _asymmetric_triangular(n: int, peak_frac: float = 0.3) -> np.ndarray:
    """Linear increase to a peak, then linear decrease to zero at the end."""
    peak = int(peak_frac * (n - 1))
    if peak <= 0:
        return _uniform(n)
    left = np.linspace(0, 1, peak + 1, endpoint=False)
    right = np.linspace(1, 0, n - peak, endpoint=False)
    a = np.concatenate([left, right])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _scipy_windows is not None and hasattr(_scipy_windows, 'kaiser'):
        a = _scipy_windows.kaiser(n, beta)
    else:
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        t = beta * np.sqrt(1 - y * y)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    if _scipy_windows is not None and hasattr(_scipy_windows, 'dpss'):
        a = _scipy_windows.dpss(n, nw, Kmax=1, return_ratios=False)[0]
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    return _uniform(n)


def _chebwin_window(n: int, att: float) -> np.ndarray:
    if _scipy_windows is not None and hasattr(_scipy_windows, 'chebwin'):
        a = _scipy_windows.chebwin(n, at=att)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    return _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15, shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few random low‑frequency cosine components, clip and normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed aggregation
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Build a large, diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_asymmetric_triangular(n, peak_frac=0.25))
    seeds.append(_exponential_decay(n, lam=0.003))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_power(n, p=0.0))
    seeds.append(_power(n, p=1.0))
    seeds.append(_power(n, p=2.0))

    # Beta family – a coarse grid
    for a in np.linspace(0.5, 5.0, 7):
        for b in np.linspace(0.5, 5.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # Window families
    for beta in (2, 4, 6, 8, 12):
        seeds.append(_kaiser_window(n, beta))
    for nw in (2.0, 3.0, 4.0):
        seeds.append(_dpss_window(n, nw))
    for att in (30, 40, 60):
        seeds.append(_chebwin_window(n, att))
    seeds.append(_tukey_window(n, alpha=0.5))

    # Random draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform (adds a flat background)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly – each entry non‑negative, sum = 1
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _logspace_adam_optimize(start_vec: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    """Adam on the log‑softmax parametrisation (smooth max)."""
    eps = 1e-30
    v = np.log(start_vec + eps)             # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # temperature for log‑sum‑exp smoothing
    beta_inc = 0.2
    beta_max = 800.0

    lr0 = 1.0                # base learning rate (log‑space)

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probability simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smoothed max using log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule – raise to sharpen the approximation
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a decreasing step size and Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[Sub] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient corresponding to the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak‑style diminishing step size
        lr = 0.30 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _local_search(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 400_000,
                  rng: np.random.Generator | None = None) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic local search.
    Randomly reshuffles mass on small subsets; always accepts strict improvements.
    """
    if rng is None:
        rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    move_frac = 0.07  # move up to 7 % of source mass per pairwise transfer

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        if rng.random() < 0.65:  # reshuffle a random subset
            k = rng.integers(2, 12)            # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.ones(k)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:  # simple pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], move_frac * cur[i])
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 25000 == 0:
                    print(f"[LS] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, keeping ℓ₁‑norm = 1."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq /= new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Reserve a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Stage 0 – problem size
    # ------------------------------------------------------------------
    n = 4000  # good trade‑off between expressiveness and speed

    # ------------------------------------------------------------------
    # Stage 1 – seed generation + quick surrogate ranking
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # Evaluate surrogate (very cheap) and keep the top performers
    evals = [(vec, _objective(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – log‑space Adam on a few promising seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        # allocate a modest slice of remaining time per Adam run
        remaining = deadline - time.time()
        max_iters = int(min(150_000, remaining * 0.12 * 1e5))
        if max_iters <= 0:
            break
        print(f"\n--- Adam run {idx+1}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _logspace_adam_optimize(seed_vec, n, deadline,
                                                     max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Global‑Adam] new best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long projected sub‑gradient descent from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_sub = int(min(1_200_000, remaining * 0.45 * 1e5))
        if max_iters_sub > 0:
            print(f"\n--- Projected sub‑gradient (max_iters={max_iters_sub}) ---")
            seq_sub, val_sub = _subgradient_proj(global_best_seq, n, deadline,
                                                 max_iters=max_iters_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[Global‑Sub] new best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – greedy local stochastic search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_ls = int(min(600_000, remaining * 0.30 * 1e5))
        if max_iters_ls > 0:
            print("\n--- Local stochastic search ---")
            seq_ls, val_ls = _local_search(global_best_seq, n, deadline,
                                           max_iters=max_iters_ls, rng=rng)
            if val_ls < global_best_val:
                global_best_seq = seq_ls.copy()
                global_best_val = val_ls
                print(f"[Global‑LS] new best = {val_ls:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – optional up‑sampling to a finer grid (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        # Need a reasonable chunk to make up‑sampling worthwhile
        if time_left > 30.0:
            n2 = n * 2                         # 8000 points
            print(f"\n--- Upsampling to n={n2} for final polishing ---")
            up_seq = _upsample_sequence(global_best_seq, n2)

            # A short Adam run at higher resolution
            remaining = deadline - time.time()
            max_iters_up_adam = int(min(120_000, remaining * 0.20 * 1e5))
            seq_up_adam, val_up_adam = _logspace_adam_optimize(
                up_seq, n2, deadline, max_iters=max_iters_up_adam)
            if seq_up_adam is not None and val_up_adam < global_best_val:
                global_best_seq = seq_up_adam.copy()
                global_best_val = val_up_adam
                print(f"[Upsample‑Adam] new best = {val_up_adam:.6f}")

            # Followed by a brief sub‑gradient phase
            remaining = deadline - time.time()
            max_iters_up_sub = int(min(400_000, remaining * 0.20 * 1e5))
            if max_iters_up_sub > 0:
                seq_up_sub, val_up_sub = _subgradient_proj(
                    global_best_seq, n2, deadline, max_iters=max_iters_up_sub)
                if val_up_sub < global_best_val:
                    global_best_seq = seq_up_sub.copy()
                    global_best_val = val_up_sub
                    print(f"[Upsample‑Sub] new best = {val_up_sub:.6f}")

            # Final quick local search on the fine grid
            remaining = deadline - time.time()
            max_iters_up_ls = int(min(200_000, remaining * 0.15 * 1e5))
            if max_iters_up_ls > 0:
                seq_up_ls, val_up_ls = _local_search(
                    global_best_seq, n2, deadline,
                    max_iters=max_iters_up_ls, rng=rng)
                if val_up_ls < global_best_val:
                    global_best_seq = seq_up_ls.copy()
                    global_best_val = val_up_ls
                    print(f"[Upsample‑LS] new best = {val_up_ls:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (if available)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for potential reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
