"""
Propose a non‑negative step‑function that (hopefully) beats the best known bound.
The algorithm works in four phases:

1. **Broad seed generation** – analytic windows, power‑laws, beta‑shapes and a
   dense set of random Dirichlet draws (different concentration parameters).
   All seeds are ℓ¹‑normalised.

2. **Intensive Adam optimisation** – each of the best few seeds is optimised
   on the smooth log‑sum‑exp surrogate of the max‑convolution.
   The temperature β is increased aggressively up to a large β_max so that the
   surrogate quickly focuses on the true max.

3. **Heavy sub‑gradient descent** – after Adam the exact non‑smooth objective
   is tackled with a projected sub‑gradient method (Euclidean projection onto
   the simplex).

4. **Targeted local search** – a cheap hill‑climbing loop that repeatedly
   moves a small fraction of mass away from the indices that contribute the
   most to the current maximal convolution entry.  This often reduces the
   bound further when the previous stages have stalled.

The routine respects a global time budget (default 1000 s) and returns the
best sequence found as a plain ``list[float]``.  The final sequence is stored
in the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate for an ℓ¹‑normalised sequence (∑seq == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (0<r<1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   beta_max: float = 5000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max‑convolution.
    ``start_vec`` must already sum to 1.
    Returns the best sequence found (still ℓ¹‑normalised) and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 10.0
    beta_mul = 1.008  # faster temperature rise

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # increase temperature
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 500_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a Polyak‑style diminishing step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polishing that reshuffles a random subset
    of the mass (Dirichlet‑reallocation) with a simple exponential cooling schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _targeted_local_search(start_seq: np.ndarray, n: int, deadline: float,
                           max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that repeatedly moves a small fraction of the mass
    away from indices that contribute the most to the current maximal convolution.
    """
    rng = np.random.default_rng()
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_val = _objective(seq)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(seq, seq)
        max_idx = int(np.argmax(conv))

        # indices that appear in the maximal term
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        contrib = seq[idx_range] * seq[max_idx - idx_range]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(i_start, i_end + 1)

        # pick a random destination (different index)
        j = rng.integers(0, n)
        if j == i:
            continue

        # move a modest fraction of the mass from i to j
        alpha = rng.uniform(0.05, 0.15)          # 5‑15 % of a_i
        delta = seq[i] * alpha
        if delta <= 0.0:
            continue

        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # renormalise (tiny drift may appear)
        new_seq /= new_seq.sum()

        val = _objective(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq.copy()
            seq = new_seq
            if it % 5000 == 0:
                print(f"[Local] it={it:6d} new best={best_val:.6f}")

    return best_seq, best_val

def _generate_initial_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diversified pool of analytic‑parameterised and random candidates."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for wf in np.linspace(0.02, 0.45, 12):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    for gamma in np.linspace(0.0, 6.0, 31):
        seeds.append(_sine_window(n, gamma=gamma))

    for alpha in np.linspace(0.0, 0.95, 20):
        seeds.append(_tukey_window(n, alpha=alpha))

    for r in np.concatenate([np.linspace(0.01, 0.98, 20), np.linspace(0.98, 0.9999, 15)]):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 7.0, 29):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 20)
    betas = np.linspace(0.2, 5.0, 20)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for sigma_frac in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma_frac))

    # Random Dirichlet draws – use a wide range of concentration parameters
    concs = np.logspace(-4, 1, 12)                     # from very spiky to very smooth
    for conc in concs:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Include the previous solution if it exists
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape[0] > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                tot = prev_arr.sum()
                if tot > 0.0:
                    seeds.append(prev_arr / tot)
        except Exception:
            pass

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats).  The routine respects the time
    budget ``budget_s`` (default 1000 s).  Progress information is printed to stdout.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – generate and evaluate a large pool of seeds
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n--- Stage 0: generating {n}‑length seed pool ---")
    seeds = _generate_initial_seeds(n, rng)
    print(f"Generated {len(seeds)} seeds, evaluating surrogate objective...")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = min(10, len(evaluated))
    best_candidates = [vec for vec, _ in evaluated[:top_k]]
    global_best_seq = best_candidates[0].copy()
    global_best_val = evaluated[0][1]
    print(f"Initial best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – intensive Adam optimisation on the top candidates
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        per_seed_budget = remaining * 0.7 / len(best_candidates)
        print(f"\n--- Stage 1: Adam optimisation (≈{per_seed_budget:.1f}s per seed) ---")
        for idx, seed_vec in enumerate(best_candidates):
            if time.time() > deadline:
                break
            # aim for roughly per_seed_budget seconds of work
            est_iter = int(per_seed_budget / 0.0015)
            est_iter = max(60_000, min(est_iter, 700_000))
            print(f"\n[Adam] seed #{idx + 1}/{len(best_candidates)}, max_iters≈{est_iter}")
            seq_adam, val_adam = _adam_optimize(
                seed_vec, n, deadline,
                max_iters=est_iter,
                beta_max=8000.0,
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – sub‑gradient refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        remaining = deadline - time.time()
        max_iters = int(remaining * 1800)            # aim for roughly 0.00055 s per iter
        max_iters = max(200_000, min(max_iters, 2_000_000))
        print(f"\n--- Stage 2: sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            init_lr=0.25,
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – targeted local hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_iters = int(remaining * 1500)   # a few hundred thousand cheap iterations
        max_iters = max(100_000, min(max_iters, 800_000))
        print(f"\n--- Stage 3: targeted local search (max_iters={max_iters}) ---")
        seq_loc, val_loc = _targeted_local_search(
            global_best_seq, n, deadline,
            max_iters=max_iters,
        )
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[overall‑Local] new global best = {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        remaining = deadline - time.time()
        max_iters = int(remaining * 950)   # about 0.001 s per iteration
        max_iters = max(200_000, min(max_iters, 2_500_000))
        print(f"\n--- Stage 4: simulated‑annealing polishing (max_iters={max_iters}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            init_temp_factor=0.06,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
