def _lbfgs_optimize(start_a, n, deadline, beta_start=10.0, beta_end=200.0, steps=3):
    # steps: number of beta increments
    a = start_a.copy()
    v = np.log(a + eps)
    for s in range(steps):
        beta = beta_start * (beta_end / beta_start) ** (s / (steps-1))
        # define wrapper function
        def fun(v_vec):
            a_vec = _softmax(v_vec)
            conv = _conv_fft(a_vec, a_vec)
            shift = np.max(conv)
            exp_terms = np.exp(beta * (conv - shift))
            w_sum = exp_terms.sum()
            smooth_val = shift + np.log(w_sum) / beta
            # objective = 2 * n * smooth_val
            obj = 2.0 * n * smooth_val
            # gradient
            w = exp_terms / w_sum
            rev_a = a_vec[::-1]
            full = _conv_fft(rev_a, w)  # length > n maybe 2n - 1
            grad_a = 2.0 * full[n-1:n-1+n]
            grad_v = a_vec * (grad_a - np.dot(grad_a, a_vec))
            grad = 2.0 * n * grad_v
            return obj, grad
        # call minimize
        res = minimize(fun, v, method='L-BFGS-B', jac=True,
                       options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
        v = res.x
        # optional early break if time near deadline
        if time.time() > deadline:
            break
    # return final a, objective surrogate
    best_a = _softmax(v)
    best_val = _objective_surrogate(best_a) # compute exact surrogate (2n*max)
    return best_a, best_val
```

"""
Propose a low‑autocorrelation step function.

The algorithm follows a three‑stage strategy:

1. **Seed collection** – a rich pool of analytic windows, random
   Dirichlet draws and the best sequence from a previous run are
   assembled and ℓ₁‑normalised.  The top few seeds (by the cheap surrogate
   2·n·max(conv)) are kept.

2. **Smooth‑max L‑BFGS optimisation** – each seed is refined by a
   continuation method that minimises a smoothed version of the true
   objective.
   The smoothing uses a log‑sum‑exp over the convolution values with a
   temperature β that is increased (10 → 50 → 200).  The optimisation
   variable is a log‑softmax vector, so the simplex constraints
   (non‑negativity and unit‑sum) are enforced automatically.  A SciPy
   L‑BFGS‑B call supplies fast quasi‑Newton convergence.

3. **Exact‑max refinement** – after the smooth optimisation the
   sequence is further polished with a projected sub‑gradient method on
   the true max‑convolution objective.  If time permits the best
   sequence is up‑sampled to twice the resolution and the whole
   optimisation pipeline is repeated on the finer grid.

The whole routine respects the global deadline (default 1000 s) and
stores the final sequence in the global name ``height_sequence_1``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy).real[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised
    vector (∑seq = 1).  The returned value is exactly the value that
    evaluate_sequence would compute, up to the sum‑squared scaling.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _triangular(n: int) -> np.ndarray:

    a = 1.0 - np.abs(i - c) / c

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows as _scipy_windows  # type: ignore
    except Exception:  # pragma: no cover
        _scipy_windows = None

    if _scipy_windows is not None and hasattr(_scipy_windows, "kaiser"):
        a = _scipy_windows.kaiser(n, beta)
    else:
        # Pure‑numpy fallback (coarse approximation)
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        t = beta * np.sqrt(1 - y * y)

        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _double_gaussian(n: int, sigma_frac: float = 0.15, shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (richer than before)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # DPSS windows
    for nw in (2.0, 3.0, 4.0, 5.0):
        seeds.append(_dpss_window(n, nw))

    # Chebyshev windows
    for att in (30, 40, 60, 80):
        seeds.append(_chebwin_window(n, att))

    # Gaussian families
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family – grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution, if provided
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # normalise robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Smooth‑max objective for L‑BFGS (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------
class _SmoothMaxObj:
    """Callable pair (fun, jac) for a fixed β."""

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def fun(self, v: np.ndarray) -> float:
        a = _softmax(v)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        exp_term = np.exp(self.beta * (conv - shift))
        w_sum = exp_term.sum()
        smooth_val = shift + np.log(w_sum) / self.beta
        return float(2.0 * self.n * smooth_val)

    def jac(self, v: np.ndarray) -> np.ndarray:
        a = _softmax(v)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        exp_term = np.exp(self.beta * (conv - shift))
        w_sum = exp_term.sum()
        w = exp_term / w_sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3·n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]
        grad_v = a * (grad_a - np.dot(grad_a, a))
        return 2.0 * self.n * grad_v


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smooth objective
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_a: np.ndarray,
                    n: int,
                    deadline: float,
                    betas: tuple[float, ...] = (10.0, 50.0, 200.0),
                    iters_per_beta: int = 800) -> tuple[np.ndarray, float]:
    """
    Run a continuation L‑BFGS optimisation.
    Returns the best ℓ₁‑normalised vector found and its exact surrogate value.
    """
    if minimize is None:
        # SciPy unavailable – fall back to the start vector
        return start_a, _objective_surrogate(start_a)

    v = np.log(start_a + 1e-30)      # logits (softmax guarantees simplex)
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in betas:
        if time.time() > deadline:
            break
        obj = _SmoothMaxObj(n, beta)
        try:
            res = minimize(obj.fun,
                           v,
                           jac=obj.jac,
                           method='L-BFGS-B',
                           options={'maxiter': iters_per_beta,
                                    'ftol': 1e-12,
                                    'disp': False})
        except Exception as exc:  # pragma: no cover
            print("[L‑BFGS] optimisation failed:", exc)
            continue
        v = res.x
        a = _softmax(v)
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS β={beta}] new best surrogate = {val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact‑max subgradient refinement (projected onto simplex)
# ----------------------------------------------------------------------
def _subgradient_refine(start_a: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    """
    a = start_a.copy()
    best = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]
        lr = 0.20 / np.sqrt(it)          # decreasing step size
        a = a - lr * g
        a = _proj_simplex(a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if it % 25000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")
    return best, best_val


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) keeping ℓ₁‑norm
# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = up / up.sum()
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats and stores it in
    the global variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Stage 0 – choose resolution
    # ------------------------------------------------------------------
    n = 4000                         # baseline grid size (good trade‑off)

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and cheap surrogate ranking
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # evaluate cheap surrogate (2·n·max(conv)) and keep the top performers
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS smooth‑max optimisation on the best seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS run {idx+1}/{len(best_seeds)} ---")
        seq_lbfgs, val_lbfgs = _lbfgs_optimize(seed_vec, n, deadline,
                                               betas=(10.0, 50.0, 200.0),
                                               iters_per_beta=800)
        if seq_lbfgs is not None and val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs.copy()
            global_best_val = val_lbfgs
            print(f"[Overall] new global best surrogate = {val_lbfgs:.6f}")

        # ------------------------------------------------------------------
        # Sub‑gradient refinement on the current global best
        # ------------------------------------------------------------------
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a fraction of the remaining budget
        max_iters_sub = int(min(500_000, remaining * 0.30 * 1e5))
        if max_iters_sub > 0:
            seq_sub, val_sub = _subgradient_refine(global_best_seq, n,
                                                   deadline,
                                                   max_iters=max_iters_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[Overall‑SubGrad] new global best surrogate = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional up‑sampling to a finer grid
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 40.0:                      # need some time for the extra work
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} for further refinement ---")
            up_seq = _upsample_sequence(global_best_seq, new_n)

            # L‑BFGS on the finer grid (fewer iterations, higher β)
            seq_up, val_up = _lbfgs_optimize(up_seq, new_n, deadline,
                                             betas=(30.0, 100.0, 300.0),
                                             iters_per_beta=600)
            if val_up < global_best_val:
                global_best_seq = seq_up.copy()
                global_best_val = val_up
                print(f"[Upsample‑L‑BFGS] new global best surrogate = {val_up:.6f}")

            # final subgradient polish on the finer grid
            if time.time() < deadline:
                remaining = deadline - time.time()
                max_iters_up_sub = int(min(300_000, remaining * 0.20 * 1e5))
                if max_iters_up_sub > 0:
                    seq_up_sub, val_up_sub = _subgradient_refine(global_best_seq,
                                                                 new_n,
                                                                 deadline,
                                                                 max_iters=max_iters_up_sub)
                    if val_up_sub < global_best_val:
                        global_best_seq = seq_up_sub.copy()
                        global_best_val = val_up_sub
                        print(f"[Upsample‑SubGrad] new global best surrogate = {val_up_sub:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (if available)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
