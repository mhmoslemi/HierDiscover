"""
Propose a step‑function (probability vector) that minimizes the bound
``evaluate_sequence``.  The search runs three stages within a 1000 s budget:

1. **Seed generation** – a modest collection of analytic families
   (uniform, triangular, beta, sine‑window, Gaussian, etc.) together
   with the previous best solution (resized if necessary).  All seeds are
   ℓ₁‑normalised.

2. **Smoothed max‑convolution descent** – we minimise a smooth
   approximation of the objective
   ``J(p)=2·n·softmax(conv(p,p))`` where ``softmax`` is a
   temperature‑controlled log‑sum‑exp.  The gradient of the soft‑max
   is computed via a correlation (FFT based) and an Adam‑style update
   with Euclidean projection onto the probability simplex.  The
   temperature (β) is slowly increased so that the surrogate
   converges to the true maximum.  This stage uses the full budget
   except a small portion reserved for polishing.

3. **Stochastic polishing** – a lightweight simulated‑annealing loop
   that reshuffles a random subset of coordinates (Dirichlet move) and
   accepts downhill moves or uphill moves with a geometric temperature
   schedule.

The best sequence found (according to the cheap surrogate
``_objective`` and finally the official ``evaluate_sequence``) is stored
in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(p: np.ndarray) -> float:
    """Exact evaluation for ℓ₁‑normalised vectors."""
    n = p.shape[0]
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        out = np.asarray(seq, dtype=np.float64)
    else:
        x_old = np.linspace(0.0, 1.0, old_n)
        x_new = np.linspace(0.0, 1.0, new_n)
        out = np.interp(x_new, x_old, seq).astype(np.float64)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s > 0.0:
        out /= s
    else:
        out = _uniform(new_n)
    return out


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a small pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine windows
    for gamma in np.linspace(0.5, 6.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))

    # beta families – include very small exponents
    alphas = np.concatenate([np.linspace(0.02, 0.2, 5), np.linspace(0.3, 1.5, 7)])
    betas = alphas
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussians with several widths
    for sigma in (0.08, 0.15, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – very low concentration
    for conc in (0.001, 0.01, 0.05, 0.2, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any) – resized if length differs
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            seeds.append(_resize_sequence(prev, n))
        except Exception:
            pass

    return seeds

# Optimisers
# ----------------------------------------------------------------------


def _smooth_descent(start_vec: np.ndarray, n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style descent on a smoothed version of the max‑convolution
    objective.  The temperature β of the soft‑max is slowly increased so
    that the surrogate converges to the true max.
    """
    p = start_vec.copy()
    best_p = p.copy()
    best_val = _objective(p)

    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.4          # base learning rate
    it = 0

    beta = 1.0         # start temperature
    beta_target = 30.0

    while time.time() < deadline:
        it += 1

        # convolution p * p
        conv = _conv_fft(p, p)
        max_c = conv.max()

        # soft‑max weights
        shifted = conv - max_c      # stability
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fall back to exact max
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of the smoothed objective:
        #   g = 4 * n * correlation(w, p)
        w_rev = w[::-1]
        grad_full = _conv_fft(w_rev, p)          # length 3n‑2
        g = grad_full[n - 1:n - 1 + n]          # slice of length n
        g = 4.0 * n * g

        # Adam update (additive) and simplex projection
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps) * m_hat
        p = p - step
        p = _proj_simplex(p)

        # keep the best surrogate value
        cur_val = _objective(p)
        if cur_val < best_val:
            best_val = cur_val
            best_p = p.copy()
            if it % 5000 == 0:
                print(f"[SmoothDescent] it={it:6d} best={best_val:.6f}")

        # slowly increase the temperature
        if beta < beta_target:
            beta = min(beta_target, beta + 0.01)

    return best_p, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_000_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing polishing.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 14)                 # move between 2 and 13 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence as a list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # Previous solution (if present) – may have a different length
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the vector – a power‑of‑two is convenient for FFTs
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Stage 1 – collect seeds and pick a promising starter
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # cheap surrogate evaluation for all seeds
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    best_seed = evaluated[0][0].copy()
    best_surrogate = evaluated[0][1]
    print(f"[Init] best surrogate = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smoothed descent on the max‑convolution objective
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 10.0:          # need at least a few seconds
        descent_deadline = time.time() + 0.80 * time_left
        print("\n--- Smoothed max‑convolution descent ---")
        seq_descent, val_descent = _smooth_descent(
            best_seed, n, descent_deadline
        )
        if val_descent < best_surrogate:
            best_seed, best_surrogate = seq_descent, val_descent
            print(f"[Descent] new best surrogate = {val_descent:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seed, n, deadline,
            max_iters=1_200_000,
            init_temp_factor=0.04
        )
        if val_sa < best_surrogate:
            best_seed, best_surrogate = seq_sa, val_sa
            print(f"[SA] final surrogate = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seed.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_surrogate:.6f} "
        f"(length {len(best_seed)})"
    )

    # store for possible later reuse
    globals()["height_sequence_1"] = best_seed.tolist()
    return best_seed.tolist()
