"""
Improved search for a low‑peak step function.

The algorithm works in three stages:

1. **Diversified seed generation** – a large pool of ℓ₁‑normalised
   candidates (analytic windows, Dirichlet draws, low‑frequency curves,
   random mixtures).  The 8 best seeds are kept.

2. **Exponentiated‑gradient descent** – a mirror‑descent step on the
   simplex that follows the sub‑gradient of the active convolution
   index.  This converges much faster than the plain additive
   sub‑gradient used previously.  Each seed receives a limited number
   of iterations (controlled by the remaining time budget).  The best
   result over all runs is kept.

3. **Aggressive simulated‑annealing polish** – a final stochastic
   refinement (subset reshuffle / small transfers) that removes any
   remaining spikes.

All operations respect the global deadline (default 1000 s).  The
resulting sequence is stored in the global name ``height_sequence_1``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy).real[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate for the official objective (valid for ℓ₁‑normalised vectors)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)          # assumes Σ seq = 1


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection – richer than the previous version
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.2))
    seeds.append(_gaussian(n, sigma_frac=0.35))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family – systematic grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # Random Dirichlet draws (different concentration parameters)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential (heavy‑tailed)
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with the uniform baseline
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise everything safely
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the ℓ∞‑convolution objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.6,
    eps_prune: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the sub‑gradient of the *active*
    convolution index.  The update is multiplicative:
        a_i ← a_i·exp(−η·g_i)
    followed by ℓ₁‑renormalisation.

    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current convolution and the worst shift
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 50_000 == 0:
                print(f"[ExpGD] it={it:7d} best={cur_val:.6f}")

        # ---- sub‑gradient of the active shift (same as in the original code) ----
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]          # ∂/∂a_i of conv[max_idx]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * g)

        # prune numerical noise (helps sparsity and stability)
        a[a < eps_prune] = 0.0

        # project back onto the simplex
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            # all entries vanished – fall back to uniform
            a = _uniform(n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing (unchanged core, but a bit more aggressive)
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-13,
) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.65:          # reshape a random subset
            k = rng.integers(3, 15)          # subset size 3 … 14
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.04)  # move ≤ 4 % of source
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted time.

    Returns the best sequence (plain ``list[float]``) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Length selection – a larger n gives a lower theoretical floor.
    # 6000 works comfortably within the time budget on a single core.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Stage 0 – gather a rich pool of seeds, keep the best few.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    scored = [(seq, _objective_surrogate(seq)) for seq in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – exponentiated‑gradient descent from each top seed.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient descent stage ---")
        for idx, seed_seq in enumerate(best_seeds):
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            # allocate a fraction of the remaining budget to each seed
            max_iters = int(min(300_000, remaining * 150_000))
            seq, val = _exp_gradient_descent(
                seed_seq, n, deadline, max_iters=max_iters, eta0=0.6
            )
            if val < best_val:
                best_val = val
                best_seq = seq
                print(f"[ExpGD] seed #{idx+1} improved surrogate to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – finer additive sub‑gradient polishing (optional).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Additive sub‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 120_000))

        def _additive_subgrad(a, max_iters_local):
            """Plain additive sub‑gradient descent (exact copy of the original)."""
            best_local = a.copy()
            best_local_val = _objective_surrogate(a)
            for it in range(1, max_iters_local + 1):
                if time.time() > deadline:
                    break
                conv = _conv_fft(a, a)
                max_idx = int(np.argmax(conv))
                cur_val = 2.0 * n * conv[max_idx]

                if cur_val < best_local_val:
                    best_local_val = cur_val
                    best_local = a.copy()

                # sub‑gradient for the active shift
                g = np.zeros_like(a)
                i_start = max(0, max_idx - (n - 1))
                i_end = min(n - 1, max_idx)
                idx_i = np.arange(i_start, i_end + 1, dtype=int)
                rev_idx = max_idx - idx_i
                g[idx_i] = 2.0 * a[rev_idx]

                lr = 0.20 / np.sqrt(it)
                a = a - lr * g
                a = _proj_simplex(a)
            return best_local, best_local_val

        seq, val = _additive_subgrad(best_seq, max_iters)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[AddSub] improved surrogate to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated annealing polishing (final tweak).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(1_200_000, remaining * 100_000))
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              max_iters=max_iters_sa)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa
            print(f"[SA] final surrogate = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
