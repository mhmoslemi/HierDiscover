"""
Multi‑stage optimisation for the non‑negative step‑function.

The algorithm proceeds as follows:
    1️⃣  Construct a large, diverse pool of ℓ₁‑normalised seeds.
        • Analytic families (uniform, triangular, trapezoidal, Gaussian,
          sine‑window, Tukey, β‑shapes) are sampled densely in their
          hyper‑parameters.
        • Random Dirichlet draws with many concentration levels.
        • Sparse supports and a uniform–Gaussian mixture.
        • The previously‑found best solution (if any) is also added.

    2️⃣  Keep the best *TOP_K* seeds (according to the exact
        max‑convolution value) and refine each of them with Adam on a
        smooth‑max surrogate of the convolution.  The temperature (β)
        is slowly increased up to a large value (default 1500) so that
        the surrogate approaches the true max.

    3️⃣  Run a long Adam optimisation starting from the globally best
        seed obtained in step 2.

    4️⃣  Apply projected sub‑gradient descent on the exact max‑convolution
        objective for a large number of iterations (Polyak step size
        with simplex projection).

    5️⃣  Polish the result with a lengthy simulated‑annealing phase that
        moves mass among a small random subset of indices at each step.

All stages respect the supplied time budget (default 1000 s).  The
best sequence found is scaled to the required ℓ₁‑norm √(2 n) and stored
in the global variable ``height_sequence_1`` for reuse in subsequent
runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant). ``seq`` is assumed to sum to 1;
    returns max(conv(seq, seq)).
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)          # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3_000_000,
    beta_max: float = 1500.0,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Returns (best_sequence, best_exact_value).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.0012

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        a = _softmax(v)                  # current point on simplex (sum=1)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.9f}")

        # increase temperature
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # convolution and smooth‑max weights
        conv = _conv_fft(a, a)           # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum                    # softmax over convolution entries

        # gradient of weighted sum of convolutions wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)       # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best sequence encountered and its exact value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a)          # exact max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.9f}")

        # sub‑gradient of the active entry (indices that contribute to max_idx)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size using the current best value
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Tiny‑move simulated‑annealing on the simplex, with a long schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)               # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.9f}")

    return best_seq, best_val

def _collect_seeds(n: int, rng: np.random.Generator, prior: np.ndarray | None,
                   deadline: float) -> list[np.ndarray]:
    """

    Returns a list of the best few candidates (sorted by increasing max‑conv).
    """
    TOP_K = 10
    top_vals: list[tuple[float, np.ndarray]] = []   # (value, seq)

    def try_insert(val: float, seq: np.ndarray):
        if len(top_vals) < TOP_K:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])

    # ----- deterministic analytic families -----
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.15),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _gaussian(n, sigma_frac=0.12),
        lambda n: _gaussian(n, sigma_frac=0.24),
        lambda n: _tukey_window(n, alpha=0.2),
        lambda n: _tukey_window(n, alpha=0.4),
    ]
    for fn in families:
        seq = fn(n)
        try_insert(_objective(seq), seq)

    # ----- geometric families -----
    for r in np.linspace(0.2, 0.98, 12):
        for fn in (_geom, _sym_geom):
            seq = fn(n, r)
            try_insert(_objective(seq), seq)

    # ----- power‑law family -----
    for p in np.linspace(0.0, 5.0, 18):
        seq = _power(n, p)
        try_insert(_objective(seq), seq)

    # ----- dense β‑shape sweep (α,β) -----
    alist = np.linspace(0.1, 3.0, 13)   # 0.1 … 3.0
    for alpha in alist:
        for beta in alist:
            if time.time() > deadline:
                break
            seq = _beta_shape(n, alpha, beta)
            try_insert(_objective(seq), seq)

    # ----- sine‑window γ sweep -----
    for gamma in np.linspace(0.2, 5.0, 25):
        if time.time() > deadline:
            break
        seq = _sine_window(n, gamma=gamma)
        try_insert(_objective(seq), seq)

    # ----- Gaussian σ sweep -----
    for sigma_frac in np.linspace(0.05, 0.4, 18):
        if time.time() > deadline:
            break
        seq = _gaussian(n, sigma_frac=sigma_frac)
        try_insert(_objective(seq), seq)

    # ----- random Dirichlet draws -----
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 120 if conc < 0.01 else 60
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            try_insert(_objective(seq), seq)

    # ----- sparse random supports -----
    for _ in range(250):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        try_insert(_objective(seq), seq)

    # ----- mixture of Uniform & Gaussian -----
    for _ in range(200):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        try_insert(_objective(seq), seq)

    # ----- previous solution (if any) -----
    if prior is not None and prior.shape == (n,):
        seq = prior.copy()
        if seq.sum() > 0:
            seq = seq / seq.sum()
            try_insert(_objective(seq), seq)

    # Return only the sequences (sorted by increasing value)
    return [seq for _, seq in top_vals]

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.  Returns the best
    sequence found.
    """
    # ------------------------------------------------------------------
    # Global set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    n = 4000                                   # length of the step‑function
    sqrt2n = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Retrieve previously found solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prior = None
    if previous is not None:
        try:
            prior = np.asarray(previous, dtype=np.float64)
        except Exception:
            prior = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds (≈10 % of budget)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seed_deadline = min(deadline, start_time + budget_s * 0.12)
    seeds = _collect_seeds(
        n=n,
        rng=rng,
        prior=prior,
        deadline=seed_deadline,
    )
    print(f"Collected {len(seeds)} promising seeds (kept top‑{len(seeds)}).")

    # ------------------------------------------------------------------
    # Stage 1 – Adam refinements of the top seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Adam refinement of seeds (stage 1) ---")
    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"[Seed {idx + 1}/{len(seeds)}]")
        seed_vec = seed_vec / seed_vec.sum()
        seq_adam, val_adam = _adam_optimize(
            start_vec=seed_vec,
            n=n,
            deadline=deadline,
            max_iters=1_500_000,
            beta_max=1200.0,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"   New global best after Adam: {best_val:.9f}")

    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print(f"Fallback to uniform seed: {best_val:.9f}")

    # ------------------------------------------------------------------
    # Stage 2 – long Adam optimisation on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long Adam optimisation (stage 2) ---")
        seq_adam, val_adam = _adam_optimize(
            start_vec=best_seq,
            n=n,
            deadline=deadline,
            max_iters=5_000_000,
            beta_max=1800.0,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"   Updated global best after long Adam: {best_val:.9f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient descent (more iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (stage 3) ---")
        seq_sub, val_sub = _subgrad_optimize(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=2_000_000,
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"   Updated global best after sub‑grad: {best_val:.9f}")

    # ------------------------------------------------------------------
    # Stage 4 – aggressive simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing (stage 4) ---")
        seq_sa, val_sa = _simulated_annealing(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=1_200_000,
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"   Updated global best after SA: {best_val:.9f}")

    # ------------------------------------------------------------------
    # Rescale to the required ℓ₁‑norm √(2n) and final check
    # ------------------------------------------------------------------
    best_seq = best_seq * sqrt2n

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective (official) ≈ {final_val:.9f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse and return plain Python list
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
