def lp_max_sum_under_conv(f, M):
    n = len(f)
    # Build A matrix
    diag_data = [np.full(n, f_i, dtype=np.float64) for f_i in f]  # list of n arrays
    offsets = np.arange(n)
    A = sp.diags(diag_data, offsets, shape=(2*n-1, n), format='csc')
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum
    bounds = [(0, None)] * n
    # Solve linear program
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    return res.x

"""
Propose a low‑value step‑function.

Algorithm
---------
1. **Seed generation** – a lightweight collection of analytically generated
   families (uniform, triangular, β‑shapes, windows, Dirichlet draws, etc.)
   is built and the best few (default 3) are kept.

2. **Conditional‑gradient refinement** – starting from the best seed we run a
   Frank‑Wolfe–style iteration:

   * Let `M = max(conv(f,f))`.
   * Solve a linear program:
         maximise   Σ g_i
         subject to   (f * g)[k] ≤ M   for all k,
                     g_i ≥ 0.
     The constraints are linear because the convolution is linear in `g`.
   * Normalise `g` to a probability vector and perform a line‑search on the
     convex combination `h(α) = (1‑α)·f + α·g` (α∈[0,1]).
     The maximal convolution coefficient of `h(α)` is a convex function of α,
     so a ternary search quickly finds the optimal α.
   * If the new maximum is strictly smaller than `M` we accept the update and
     repeat.  The loop stops when the time budget is exhausted or no further
     improvement is possible.

3. **Fine‑grained polishing** – a short sub‑gradient descent (exact max‑conv)
   and a final simulated‑annealing phase are applied to squeeze the last bits.

4. **Rescaling & verification** – the final vector is scaled to the required
   ℓ₁‑norm `√(2n)` and evaluated with the official `evaluate_sequence`.

All heavy steps respect the global time budget (`budget_s`).  The best
sequence is stored in the global variable ``height_sequence_1`` and returned
as a plain Python list.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant). ``seq`` is assumed to sum to 1;
    the function returns max(conv(seq, seq)).
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)          # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

# ----------------------------------------------------------------------
# Conditional‑gradient (Frank‑Wolfe) refinement based on a linear program
# ----------------------------------------------------------------------


def _lp_max_sum_under_conv(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve:
        maximise   Σ g_i
        subject to (f * g)[k] ≤ M   for all k,
                   g_i ≥ 0.

    Returns the optimal `g` (not normalised).  The constraints are expressed
    via a sparse Toeplitz matrix constructed with `scipy.sparse.diags`.
    """
    n = f.shape[0]
    # Build a (2n‑1) × n matrix where each column i is f_i repeated on the
    # sub‑diagonal with offset i.
    # Using `diags` we create `n` constants diagonals.
    diag_data = [np.full(n, coeff, dtype=np.float64) for coeff in f]
    offsets = np.arange(n)                     # 0 … n‑1
    A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format="csc")
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  ⇔ minimise –Σ g_i
    bounds = [(0, None)] * n

    # HiGHS is very fast for these moderate‑size LPs.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def _line_search_optimal_alpha(f: np.ndarray, g: np.ndarray,
                               conv_f: np.ndarray,
                               conv_g: np.ndarray,
                               conv_fg: np.ndarray) -> float:
    """
    For the convex function

        φ(α) = max_t [ (1‑α)²·conv_f[t] + 2α(1‑α)·conv_fg[t] + α²·conv_g[t] ]

    find the minimising α∈[0,1] by ternary (golden‑section) search.
    """
    lo, hi = 0.0, 1.0
    for _ in range(30):   # enough for double‑precision accuracy
        a1 = lo + (hi - lo) / 3.0
        a2 = hi - (hi - lo) / 3.0

        mix1 = (1 - a1) ** 2 * conv_f + 2 * a1 * (1 - a1) * conv_fg + a1 ** 2 * conv_g
        mix2 = (1 - a2) ** 2 * conv_f + 2 * a2 * (1 - a2) * conv_fg + a2 ** 2 * conv_g

        if mix1.max() < mix2.max():
            hi = a2
        else:
            lo = a1
    return (lo + hi) / 2.0


def _conditional_gradient_refine(start: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 30) -> np.ndarray:
    """
    Frank‑Wolfe style refinement where the linear sub‑problem is solved by a
    small LP.  Returns a (probability) vector with hopefully lower max‑conv.
    """
    f = start.copy()
    best_val = _objective(f)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        M = best_val
        try:
            g_raw = _lp_max_sum_under_conv(f, M)
        except Exception as exc:
            print(f"[CG] LP failed at iteration {it}: {exc}")
            break

        sum_g = g_raw.sum()
        if sum_g <= 0:
            print("[CG] LP returned zero vector – terminating.")
            break
        g = g_raw / sum_g                     # normalise to sum‑one

        # Pre‑compute convolutions needed for line search
        conv_f = _conv_fft(f, f)
        conv_g = _conv_fft(g, g)
        conv_fg = _conv_fft(f, g)

        # Find optimal mixing weight α
        alpha = _line_search_optimal_alpha(f, g, conv_f, conv_g, conv_fg)

        if alpha <= 0.0:
            # No beneficial direction found
            break

        # Update f
        f = (1 - alpha) * f + alpha * g
        f = f / f.sum()                       # re‑project to simplex (numerical safety)

        new_val = _objective(f)
        if new_val < best_val - 1e-12:
            best_val = new_val
            print(f"[CG] iter {it:3d} improved max‑conv = {best_val:.6f}")
        else:
            # No improvement – stop early
            break

    return f


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑conv)
# ----------------------------------------------------------------------


def _subgrad_optimize(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry (as described in the problem statement)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq


# ----------------------------------------------------------------------
# Stochastic polishing (tiny Dirichlet reallocations)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000,
                         init_temp_factor: float = 0.04) -> np.ndarray:
    """
    Tiny‑move simulated‑annealing on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, 1e-12)

        k = rng.integers(2, 13)       # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # small safety margin

    # Length of the step‑function – 2000 offers a good trade‑off between
    # optimisation speed and achievable bound.
    n = 2000
    sqrt2n = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Stage 0 – seed generation (≈8 % of total budget)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    seed_deadline = min(deadline, start_time + budget_s * 0.08)
    seeds = _collect_diverse_seeds(
        n=n,
        rng=rng,
        previous=previous,
        top_k=5,
        deadline=seed_deadline,
    )
    print(f"Collected {len(seeds)} promising seeds (kept top‑{len(seeds)}).")

    # ------------------------------------------------------------------
    # Stage 1 – pick the best seed and run a short LP‑based refinement
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Selecting initial seed ---")
    for idx, cand in enumerate(seeds):
        val = _objective(cand)
        print(f"[Seed {idx+1}] max‑conv = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Stage 2 – conditional‑gradient (LP) refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Conditional‑gradient LP refinement (stage 2) ---")
        refined = _conditional_gradient_refine(
            start=best_seq,
            n=n,
            deadline=deadline,
            max_iters=30,
        )
        refined_val = _objective(refined)
        print(f"[CG] post‑refine max‑conv = {refined_val:.6f}")
        if refined_val < best_val:
            best_seq = refined
            best_val = refined_val

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient descent (fine tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (stage 3) ---")
        seq_sub = _subgrad_optimize(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=400_000,
        )
        sub_val = _objective(seq_sub)
        print(f"[Sub‑grad] max‑conv = {sub_val:.6f}")
        if sub_val < best_val:
            best_seq = seq_sub
            best_val = sub_val

    # ------------------------------------------------------------------
    # Stage 4 – stochastic polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing (stage 4) ---")
        seq_sa = _simulated_annealing(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=200_000,
        )
        sa_val = _objective(seq_sa)
        print(f"[SA] max‑conv = {sa_val:.6f}")
        if sa_val < best_val:
            best_seq = seq_sa
            best_val = sa_val

    # ------------------------------------------------------------------
    # Scale to the required ℓ₁‑norm (√(2n)) and final sanity check
    # ------------------------------------------------------------------
    best_seq = best_seq * sqrt2n

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective (official) ≈ {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
