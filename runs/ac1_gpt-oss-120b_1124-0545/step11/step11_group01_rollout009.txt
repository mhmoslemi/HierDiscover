def _perm_anneal_optimize(start_seq, n, deadline, max_iters=500_000, init_temp_factor=0.02):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    # Temperature schedule: start T proportional to init_temp_factor * cur_val
    T = max(init_temp_factor * cur_val, 1e-12)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # cooling schedule
        T = max(T * 0.9998, 1e-12)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]  # swap

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_seq = new.copy()
                best_val = new_val
                if it % 50000 == 0:
                    print(f"[Perm‑SA] it={it} best={new_val:.6f}")
    return best_seq, best_val

"""
search algorithm for minimise 2·n·max(conv(seq,seq))/(sum(seq)²)

The routine proceeds in several stages while staying inside the allotted
time‑budget (≈1000 s).  A pool of ℓ₁‑normalised “seed” shapes (uniform,
triangular, windows, random Dirichlet, and sparse spike patterns) is
generated.  The best few seeds are refined with an Adam optimiser that
optimises a temperature‑controlled log‑sum‑exp surrogate for the max‑
convolution.  The resulting vector is then polished by a long mirror‑
descent (exponentiated‑gradient) sub‑gradient descent on the *exact*
max‑convolution objective.  Finally a permutation‑only simulated‑annealing
phase shuffles the entries (preserving the multiset of weights) – this
often lowers the peak autocorrelation further because the objective depends
only on the ordering of the values.  The best feasible vector found is
stored globally as ``height_sequence_1`` and returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate (exact) objective for a ℓ₁‑normalised sequence:
    2·n·max(conv(seq,seq)).
    The normalization (sum(seq)=1) is assumed – the true objective of
    ``evaluate_sequence`` is homogeneous and therefore unchanged by scaling.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _spike_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """m non‑zero equal spikes (m ≤ n)."""
    if m <= 0:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=m, replace=False)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------
def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Seed pool construction (richer than the reference version)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # sine window
    for gamma in np.linspace(0.5, 6.0, 12):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussian
    for sigma in np.linspace(0.10, 0.55, 10):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # power law
    for p in np.linspace(0.0, 8.0, 17):
        seeds.append(_power(n, p))
        seeds.append(_alternate_order(_power(n, p)))

    # beta family
    for _ in range(15):
        alpha = 10 ** rng.uniform(-2, 0.6)   # 0.01 … ~4
        beta = 10 ** rng.uniform(-2, 0.6)
        seeds.append(_beta_shape(n, alpha, beta))
        seeds.append(_alternate_order(_beta_shape(n, alpha, beta)))

    # exponential
    for rate in np.logspace(-4, -1, 12):
        seeds.append(_exponential(n, rate))
        seeds.append(_alternate_order(_exponential(n, rate)))

    # logarithmic
    for eps in np.logspace(-4, -1, 8):
        seeds.append(_logarithmic(n, eps))
        seeds.append(_alternate_order(_logarithmic(n, eps)))

    # Dirichlet draws with various concentrations
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))

    # low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))
            seeds.append(_alternate_order(_lowfreq_curve(n, nf, rng)))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))
            seeds.append(_alternate_order(_mix_shapes(_uniform(n), curve, mix_alpha)))

    # Sparse spike patterns – often useful for flattening autocorrelation
    for m in (5, 10, 20, 40, 80, 160):
        seeds.append(_spike_seed(n, m, rng))
        seeds.append(_alternate_order(_spike_seed(n, m, rng)))

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 150_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on the logits (softmax) of the vector, using a temperature‑controlled
    log‑sum‑exp surrogate for max(conv).  Returns the best feasible vector seen
    (already ℓ₁‑normalised) and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    # smoothing temperature schedule
    beta = 0.5
    beta_inc = 0.5
    beta_max = 500.0   # sharper than reference

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the true
    max‑convolution objective.  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.25  # a bit smaller than reference – smoother convergence

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient (active index)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional re‑ordering – large‑small alternation can lower the peak
        if it % 50_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {val_alt:.6f}")

    return best_seq, best_val


def _perm_anneal_optimize(start_seq: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that only permutes the entries of the vector
    (the multiset of values stays unchanged).  Swaps are accepted
    according to a decreasing temperature schedule.
    Returns the best sequence seen.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9998, min_temp)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]   # swap

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_seq = new.copy()
                best_val = new_val
                if it % 50_000 == 0:
                    print(f"[Perm‑SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given budget (seconds).
    Returns the best sequence (list of floats) and stores it globally as
    ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # Length – a larger n gives a better theoretical bound while still
    # being fast enough for FFT‑based convolutions.
    # --------------------------------------------------------------
    n = 6000

    # --------------------------------------------------------------
    # Load previous best if it exists
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Phase 1 – seed generation & quick screening
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation (smooth surrogate) on top seeds
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # Allocate a modest number of iterations based on remaining time
        max_iters = min(200_000, int(remaining * 40))   # empirical scaling
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # try an alternating re‑ordering of the Adam result
        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq = alt
                global_best_val = alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Mirror‑descent sub‑gradient optimisation (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline,
            max_iters=2_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Permutation‑only simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation‑only simulated annealing ---")
        remaining = deadline - time.time()
        max_iters = int(min(1_000_000, remaining * 800))   # roughly 800 moves/second
        seq_perm, val_perm = _perm_anneal_optimize(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            init_temp_factor=0.02
        )
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[overall‑Perm‑SA] new global best = {val_perm:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – Very light smoothing (optional)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = global_best_seq.copy()
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        a = np.convolve(a, kernel, mode='same')
        a = np.clip(a, a_min=0.0, a_max=None)
        a = a / a.sum()
        val_smooth = _objective(a)
        if val_smooth < global_best_val:
            global_best_seq = a.copy()
            global_best_val = val_smooth
            print(f"[Smooth] new global best = {val_smooth:.6f}")

    # --------------------------------------------------------------
    # Final sanity‑check with the official evaluator
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
