def _greedy_local_search(start_seq, n, deadline, max_iters=100000):
    a = start_seq.copy()
    cur_val = _objective(a)
    rng = np.random.default_rng()
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # choose source index weighted by a (prefer larger values)
        i = rng.choice(n, p=a)
        # choose target index uniformly ignoring i
        j = rng.choice(n-1)
        if j >= i:
            j += 1
        # epsilon = fraction of a[i]
        eps = a[i] * rng.random() * 0.5  # up to 50% of source
        if eps <= 1e-12:
            continue
        new = a.copy()
        new[i] -= eps
        new[j] += eps
        # Evaluate
        new_val = _objective(new)
        if new_val < cur_val:
            a = new
            cur_val = new_val
            # optional print
            if it % 10000 == 0:
                print(f"[Greedy] it {it} cur {cur_val:.6f}")
    return a
```

   max‑convolution, starting from each of the top candidates (but the total
   runtime of this phase is limited to ~55 % of the overall budget).
3. Projected sub‑gradient descent on the true max‑convolution objective
   (Polyak step size) – ~20 % of the budget.
4. Light simulated‑annealing (Dirichlet moves of a random subset) – ~15 %.
5. Greedy local search: repeatedly shift a small amount of mass from one
   index to another, accepting only improving moves – the remaining time.
6. The best sequence found is returned (as a plain Python list) and stored
   in the global variable ``height_sequence_1`` for possible reuse.
All steps respect the 1000 s time budget and avoid filesystem/network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective of ``evaluate_sequence`` (scale‑invariant)."""
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    w = e / e.sum()
    return w


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_frac: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_frac * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray | None, float]:
    """Adam on the log‑sum‑exp surrogate of max‑convolution."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits (stay on simplex after softmax)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # initial smoothing parameter
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)          # probability vector on simplex
        cur_val = _objective(a)   # exact objective (used for bookkeeping)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # increase temperature → tighter smooth‑max
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)            # full convolution, length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                         # softmax over convolution entries

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)        # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # transform to logits coordinates (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.8 / np.sqrt(it)   # decaying learning‑rate

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        # Polyak step size towards current best
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)
        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Light SA: re‑allocate the total mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)                # size of the moved subset
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30000 == 0:
                    print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


def _greedy_local_search(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 120_000) -> np.ndarray:
    """
    Simple hill‑climbing: move a fraction of mass from a high‑mass index
    to a random target.  Only improving moves are kept.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    cur_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # pick a source proportional to its current mass (larger masses move more)
        if a.sum() == 0:
            break
        src = rng.choice(n, p=a)
        # pick a different target uniformly
        tgt = rng.integers(n - 1)
        if tgt >= src:
            tgt += 1
        eps = a[src] * rng.random() * 0.5   # move up to 50 % of source mass
        if eps < 1e-14:
            continue
        new = a.copy()
        new[src] -= eps
        new[tgt] += eps
        new_val = _objective(new)
        if new_val < cur_val:
            a = new
            cur_val = new_val
            if it % 20000 == 0:
                print(f"[Greedy] it={it:,} cur={cur_val:.6f}")

    return a


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------


def _collect_candidates(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None,
                       top_k: int,
                       deadline: float) -> list[np.ndarray]:
    """Deterministic sweep + random seeds, keeping the `top_k` best."""
    best = []  # list of (value, seq)

    def keep(val, seq):
        if len(best) < top_k:
            best.append((val, seq))
            best.sort(key=lambda x: x[0])
        elif val < best[-1][0]:
            best[-1] = (val, seq)
            best.sort(key=lambda x: x[0])

    # ----- analytic families -------------------------------------------------
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_frac=0.12),
        lambda n: _trapezoidal(n, width_frac=0.25),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _sine_window(n, gamma=3.0),
        lambda n: _tukey_window(n, alpha=0.3),
        lambda n: _tukey_window(n, alpha=0.6),
        lambda n: _gaussian(n, sigma=0.3 * n),
        lambda n: _gaussian(n, sigma=0.15 * n),
        lambda n: _geom(n, r=0.95),
        lambda n: _sym_geom(n, r=0.95),
    ]
    for fn in families:
        seq = fn(n)
        keep(_objective(seq), seq)

    # ----- Beta grid ---------------------------------------------------------
    alphas = np.linspace(0.05, 4.0, 35)
    betas = np.linspace(0.05, 4.0, 35)
    for a in alphas:
        for b in betas:
            seq = _beta_shape(n, a, b)
            keep(_objective(seq), seq)

    # ----- Power‑law ---------------------------------------------------------
    ps = np.linspace(0.0, 6.0, 35)
    for p in ps:
        seq = _power(n, p)
        keep(_objective(seq), seq)

    # ----- Geometric ---------------------------------------------------------
    rs = np.linspace(0.2, 0.99, 30)
    for r in rs:
        seq = _geom(n, r)
        keep(_objective(seq), seq)
        seq2 = _sym_geom(n, r)
        keep(_objective(seq2), seq2)

    # ----- Random Dirichlet seeds -------------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 80 if conc < 0.01 else 40
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            keep(_objective(seq), seq)

    # ----- Sparse random supports -------------------------------------------
    for _ in range(250):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)           # support size 3 … 30
        idx = rng.choice(n, size=k, replace=False)
        w = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = w
        keep(_objective(seq), seq)

    # ----- previous solution -------------------------------------------------
    if previous is not None and previous.shape == (n,):
        seq = previous / previous.sum()
        keep(_objective(seq), seq)

    # Return only the sequences, sorted ascending by objective
    return [seq for _, seq in best]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    `evaluate_sequence`.  The routine respects the given time budget and
    returns the best sequence found (also stored as ``height_sequence_1``).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline_total = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose the length of the step function.
    #   4000 gives a good balance; if a previous solution exists we keep its length.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) > 0:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.size > 0:
                n = int(prev_arr.size)
                prev_arr = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
                n = 4000
        except Exception:
            prev_arr = None
            n = 4000
    else:
        prev_arr = None
        n = 4000

    print(f"\n=== Starting search (n={n}, budget={budget_s}s) ===")

    # ------------------------------------------------------------------
    # Phase 0 – deterministic & random seed sweep (≈15 % of budget)
    # ------------------------------------------------------------------
    phase0_deadline = min(deadline_total, start_time + 0.15 * budget_s)
    print("\n--- Phase 0: candidate sweep ---")
    candidates = _collect_candidates(
        n, rng, prev_arr, top_k=20, deadline=phase0_deadline
    )
    print(f"Collected {len(candidates)} top candidates (objective values):")
    for i, seq in enumerate(candidates[:5]):
        print(f"  #{i+1}: {_objective(seq):.6f}")

    # pick the best from the sweep as a starting point
    best_seq = candidates[0].copy()
    best_val = _objective(best_seq)
    for seq in candidates[1:]:
        val = _objective(seq)
        if val < best_val:
            best_seq, best_val = seq.copy(), val
    print(f"[Init] best objective from sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation (≈55 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline_total, start_time + 0.55 * budget_s)
    print("\n--- Phase 1: long Adam optimisation ---")
    seq_adam, val_adam = _adam_optimize(best_seq, n, phase1_deadline)
    if seq_adam is not None and val_adam < best_val:
        best_seq, best_val = seq_adam.copy(), val_adam
        print(f"[Adam] improvement to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (≈20 % of budget)
    # ------------------------------------------------------------------
    phase2_deadline = min(deadline_total, start_time + 0.75 * budget_s)
    if time.time() < phase2_deadline:
        print("\n--- Phase 2: projected sub‑gradient descent ---")
        seq_sub, val_sub = _subgrad_optimize(best_seq, n, phase2_deadline)
        if val_sub < best_val:
            best_seq, best_val = seq_sub.copy(), val_sub
            print(f"[Subgrad] improvement to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing polishing (≈15 % of budget)
    # ------------------------------------------------------------------
    phase3_deadline = min(deadline_total, start_time + 0.90 * budget_s)
    if time.time() < phase3_deadline:
        print("\n--- Phase 3: simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, phase3_deadline)
        if val_sa < best_val:
            best_seq, best_val = seq_sa.copy(), val_sa
            print(f"[SA] improvement to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – greedy local search (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline_total:
        print("\n--- Phase 4: greedy local search ---")
        seq_greedy = _greedy_local_search(best_seq, n, deadline_total)
        val_greedy = _objective(seq_greedy)
        if val_greedy < best_val:
            best_seq, best_val = seq_greedy.copy(), val_greedy
            print(f"[Greedy] final objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[Verify] evaluate_sequence returns {ev:.6f}")
    except Exception as exc:
        print("[Verify] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # store for potential reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
