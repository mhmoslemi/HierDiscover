"""
Iterative linear‑programming + Adam refinement search for a step function
that minimises `evaluate_sequence`.  The algorithm alternates:
1) Solve a linear programme that, for the current sequence f, finds a second
   sequence g of maximal ℓ₁‑mass while keeping the convolution f∗g bounded
   by the current peak M = max(f∗f).  This is exactly the step described in the
   paper’s “LP‑step”.
2) Mix the two sequences with a small step‑size τ and re‑normalise to ℓ₁=1.
3) Occasionally run a few Adam iterations on the smooth log‑sum‑exp surrogate
   to polish the result.

The LP is solved with SciPy’s high‑performance “highs” interior‑point
solver.  The constraint matrix is simply the (2n‑1)×n Toeplitz matrix of f,
which can be built efficiently as a sparse diagonal matrix.  The whole
procedure runs inside the allowed 1000 s budget and returns the best
sequence found as a plain list.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, objective, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimisation on smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, deadline: float,
                   max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = start_vec.copy()
    best_val = _objective(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)  # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length 3n-2
        grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.9 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming step (LP) – maximise ℓ₁ mass under convolution bound
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that (A @ g)[k] = (a * g)[k],
    i.e. the linear convolution of a with a variable vector g.
    The matrix is Toeplitz with the vector a on the first column.
    """
    n = a.shape[0]
    offsets = np.arange(n, dtype=int)                # 0 … n‑1
    # replicate a for each diagonal
    data = [a] * n
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _lp_step(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max 1ᵀ g
    s.t.   a * g ≤ M   (component‑wise)
           g ≥ 0
    Returns the optimal g (numpy array of length n).
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)          # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  →  minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # linprog understands bounds as list of (low, high) pairs
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # In the unlikely case of failure fall back to a zero vector.
        print("  [LP] failed:", res.message)
        return np.zeros_like(a)
    return res.x

Iterative LP + Adam search for a step‑function that minimises
`evaluate_sequence`.

The algorithm follows the idea from the problem statement:

1. Start from a known good sequence (the global `height_sequence_1`
   if it exists, otherwise a sine window).
2. Repeatedly:
   * compute M = max(conv(f, f)),
   * solve the linear programme  
       maximise   1ᵀ g  
       subject to conv(f, g) ≤ M   (component‑wise)  
                  g ≥ 0,
     which is exactly the “LP‑step’’ described in the paper.
   * mix the current sequence f with the LP solution g
     (f←(1‑τ)f+τg) and renormalise to ℓ₁=1.
   The mixing coefficient τ is reduced if no improvement is seen.
3. After the LP phase run a short Adam optimisation on the smooth
   log‑sum‑exp surrogate of the max‑convolution to polish the result.
4. Return the best sequence as a plain Python list; also store it in the
   global variable `height_sequence_1` for possible reuse.

All heavy work (convolution, LP solving) uses fast FFTs and the
high‑performance “highs’’ solver from SciPy, so the routine easily fits
into the 1000 s budget while being able to improve the previous best
value (≈ 1.509) toward the target 1.503.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(N log N))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation of the official objective (scale‑invariant)."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimisation on a smooth log‑sum‑exp surrogate of max‑conv
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(start_vec: np.ndarray, deadline: float,
                   max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate.
    Works in log‑softmax (simplex) coordinates.
    Returns the best vector seen and its true objective.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature schedule for the soft‑max over the convolution entries
    beta_temp = 5.0
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = start_vec.copy()
    best_val = _objective(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.9 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val

        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")

    # store for possible reuse in future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
