"""
Propose a low‑autocorrelation non‑negative step function.
--------------------------------------------------------------------
The search consists of several stages:

1️⃣  Small‑scale LP improvement.
    Starting from a uniform vector of length *n_small* (≈200) we
    iteratively solve the linear program

        max   Σ b_j
        s.t.  b_j ≥ 0,
              (f * b)_k ≤ M   for all k,

    where M = max_k (f*f)_k.
    The LP adds mass in directions that do not increase the current
    worst‑case convolution.  After a few LP iterations we obtain an
    improved small‑scale shape.

2️⃣  Upscaling.
    The small shape is linearly interpolated to the target size
    *n_big* (≈4000) and renormalised to ℓ¹‑norm 1.

3️⃣  Heavy optimisation (Adam on a smooth “log‑sum‑exp’’ surrogate).
    Starting from the up‑scaled seed we run the Adam optimiser from the
    original code for a time‑budgeted number of iterations.

4️⃣  Local refinement (projected sub‑gradient descent).

5️⃣  Stochastic polishing (simulated annealing).

All stages respect the global 1000 s budget.  The best sequence found
is stored in the global variable ``height_sequence_1`` and returned as a
plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution, linprog

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate for a ℓ¹‑normalised sequence (∑seq == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Interpolation utilities
# ----------------------------------------------------------------------


def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from a small sequence to a larger one."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    else:
        ys = np.full(n_big, 1.0 / n_big)
    return ys.astype(np.float64)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   beta_max: float = 2500.0) -> tuple[np.ndarray | None, float]:
    eps = 1e-30
    v = np.log(start_vec + eps)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 10.0
    beta_mul = 1.006

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 500_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _lp_improve(f: np.ndarray, max_iters: int, deadline: float) -> np.ndarray:
    """
    Perform a few iterations of the LP step described in the statement.
    The LP solves:
        max   Σ b_j
        s.t.  b_j ≥ 0,
              (f * b)_k ≤ M   for all k,
    where M = max_k (f*f)_k.
    After each LP we convex‑combine the current f with the normalised
    LP solution.
    """
    n = len(f)
    eps = 1e-8                       # avoid zero weights in constraints
    t = 0.2                          # mixing parameter
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current worst‑case convolution value
        conv_ff = _conv_fft(f, f)
        M = float(conv_ff.max())

        # build the constraint matrix A (shape n×n)
        # (f * b)[k] = Σ_i f[i] * b[k-i]   (zero‑padding outside)
        A = np.zeros((n, n), dtype=np.float64)
        for k in range(n):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                A[k, j] = f[i] + eps   # tiny epsilon guarantees boundedness

        # linear programme: maximise Σ b_j  ⇔ minimise −Σ b_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(n, M), bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        if not res.success:
            # LP failed – abort further LP refinement
            break

        b = res.x
        # normalise the LP solution
        b_sum = b.sum()
        if b_sum <= 0:
            break
        b = b / b_sum

        # convex combination and re‑normalise
        f_new = (1.0 - t) * f + t * b
        f_new = f_new / f_new.sum()

        # stop if no progress
        if _objective(f_new) >= _objective(f) * (1 - 1e-6):
            break
        f = f_new
    return f

    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Stage 0 – small LP refinement (n≈200)
    # --------------------------------------------------------------
    n_small = 200
    print(f"\n--- Stage 0: LP refinement on size {n_small} ---")
    # start from a uniform vector
    f_small = np.full(n_small, 1.0 / n_small, dtype=np.float64)

    f_small = _lp_improve(f_small, max_iters=6, deadline=deadline)
    val_small = _objective(f_small)
    print(f"[LP] best small surrogate = {val_small:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – upscale to the target size
    # --------------------------------------------------------------
    n_large = 4000
    print(f"\n--- Stage 1: upscaling to {n_large} ---")
    f_large = _upscale(f_small, n_large)
    best_seq = f_large.copy()
    best_val = _objective(best_seq)
    print(f"[Init] surrogate after upscale = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – Adam optimisation (temperature‑aware)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        per_seed_budget = remaining * 0.5
        est_iters = max(20_000, int(per_seed_budget / 0.0015))
        est_iters = min(est_iters, 300_000)
        print(f"\n--- Stage 2: Adam optimisation (≈{est_iters} iters) ---")
        seq_adam, val_adam = _adam_optimize(best_seq, n_large, deadline,
                                            max_iters=est_iters,
                                            beta_max=2500.0)
        if seq_adam is not None and val_adam < best_val:
            best_seq, best_val = seq_adam, val_adam
            print(f"[Adam] new global best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – sub‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        remaining = deadline - time.time()
        max_iters = int(remaining * 1500)
        max_iters = max(200_000, min(max_iters, 1_500_000))
        print(f"\n--- Stage 3: sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(best_seq, n_large, deadline,
                                               max_iters=max_iters,
                                               init_lr=0.20)
        if val_proj < best_val:
            best_seq, best_val = seq_proj, val_proj
            print(f"[Proj] new global best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_sa_iters = int(remaining * 950)
        max_sa_iters = max(200_000, min(max_sa_iters, 2_000_000))
        print(f"\n--- Stage 4: simulated‑annealing polishing (max_iters={max_sa_iters}) ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n_large, deadline,
                                              max_iters=max_sa_iters,
                                              init_temp_factor=0.04)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluator
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
