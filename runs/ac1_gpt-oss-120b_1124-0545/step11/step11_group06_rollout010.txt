"""
Propose a low‑value step function for `evaluate_sequence`.

The algorithm proceeds in two resolution levels:

1. **Coarse search (n_small≈2000).**  A deterministic sweep of several
   analytic families (uniform, triangular, trapezoidal, sine‑window,
   logistic) together with many random Dirichlet samples yields a set of
   promising candidates (kept on the simplex, sum=1).

2. **Refinement at full resolution (n_big≈4000).**  Each coarse candidate
   is up‑sampled by linear interpolation, then refined with a long Adam
   optimisation on a smooth‑max surrogate of the max‑convolution.
   The current best candidate is further polished with a projected
   sub‑gradient descent and a brief simulated‑annealing phase.

All phases respect the time budget (default 1000 s) and constantly
check the deadline.  The best sequence found is stored in the global
variable ``height_sequence_1`` for possible reuse.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a *probability* vector.
    Returns 2·n·max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal_width_12(n: int) -> np.ndarray:
    return _trapezoidal(n, width_fraction=0.12)


def _trapezoidal_width_25(n: int) -> np.ndarray:
    return _trapezoidal(n, width_fraction=0.25)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_gamma_1(n: int) -> np.ndarray:
    return _sine_window(n, gamma=1.0)


def _sine_gamma_2(n: int) -> np.ndarray:
    return _sine_window(n, gamma=2.0)


def _sine_gamma_3(n: int) -> np.ndarray:
    return _sine_window(n, gamma=3.0)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_alpha_2(n: int) -> np.ndarray:
    return _tukey_window(n, alpha=0.2)


def _tukey_alpha_5(n: int) -> np.ndarray:
    return _tukey_window(n, alpha=0.5)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, beta: float = 1.0, shift: float = 0.0) -> np.ndarray:
    """Decreasing sigmoid shape a_i ∝ 1/(1+exp(beta*(i-shift)))"""
    i = np.arange(n, dtype=np.float64)
    z = beta * (i - shift)
    # Clip to avoid overflow in exp
    z = np.clip(z, -700.0, 700.0)
    a = 1.0 / (1.0 + np.exp(z))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 1_200_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits (stay on simplex after softmax)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # start temperature
    beta_mul = 1.001
    beta_max = 800.0         # higher than before for tighter approximation

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # raise temperature slowly (approach true max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)                  # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # softmax over conv entries

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Very lightweight simulated‑annealing: randomly re‑allocate the total
    mass of a small random subset of coordinates (Dirichlet move).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30_000 == 0:
                    print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger size, then re‑normalise to sum=1."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    xs_old = np.linspace(0.0, 1.0, old_n)
    xs_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(xs_new, xs_old, seq)
    new_seq = new_seq / max(new_seq.sum(), 1e-30)
    return new_seq


def _keep_best(best: list, val: float, seq: np.ndarray, top_k: int):
    """Insert (val,seq) into the sorted list of best candidates."""
    if len(best) < top_k:
        best.append((val, seq))
        best.sort(key=lambda x: x[0])
    elif val < best[-1][0]:
        best[-1] = (val, seq)
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Candidate generation (deterministic sweep + random seeds)
# ----------------------------------------------------------------------


def _collect_candidates(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None,
                       top_k: int, deadline: float) -> list[np.ndarray]:
    """
    Systematically explore analytic families and retain the `top_k`
    candidates with smallest exact objective.
    """
    best: list[tuple[float, np.ndarray]] = []   # (value, seq)

    # ---- deterministic families -------------------------------------------------
    families = [
        _uniform,
        _triangular,
        _trapezoidal_width_12,
        _trapezoidal_width_25,
        _sine_gamma_1,
        _sine_gamma_2,
        _sine_gamma_3,
        _tukey_alpha_2,
        _tukey_alpha_5,
    ]

    for fn in families:
        seq = fn(n)
        _keep_best(best, _objective(seq), seq, top_k)

    # ---- beta shape grid -------------------------------------------------------
    alphas = np.linspace(0.1, 3.0, 30)
    betas = np.linspace(0.1, 3.0, 30)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            seq = _beta_shape(n, a, b)
            _keep_best(best, _objective(seq), seq, top_k)

    # ---- power‑law ---------------------------------------------------------------
    ps = np.linspace(0.0, 5.0, 30)
    for p in ps:
        if time.time() > deadline:
            break
        seq = _power(n, p)
        _keep_best(best, _objective(seq), seq, top_k)

    # ---- geometric ---------------------------------------------------------------
    rs = np.linspace(0.2, 0.98, 25)
    for r in rs:
        if time.time() > deadline:
            break
        seq = _geom(n, r)
        _keep_best(best, _objective(seq), seq, top_k)
        seq2 = _sym_geom(n, r)
        _keep_best(best, _objective(seq2), seq2, top_k)

    # ---- logistic windows (vary steepness and shift) ---------------------------
    betas_log = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    shifts = [0.0, n * 0.25, n * 0.5]
    for b_val in betas_log:
        for sh in shifts:
            if time.time() > deadline:
                break
            seq = _logistic_window(n, beta=b_val, shift=sh)
            _keep_best(best, _objective(seq), seq, top_k)

    # ---- random Dirichlet seeds --------------------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 80 if conc < 0.01 else 40
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            _keep_best(best, _objective(seq), seq, top_k)

    # ---- sparse random supports -------------------------------------------------
    for _ in range(200):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)           # 3 … 30 non‑zero entries
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = weights
        _keep_best(best, _objective(seq), seq, top_k)

    # ---- previous solution -------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        seq = previous / max(previous.sum(), 1e-30)
        _keep_best(best, _objective(seq), seq, top_k)

    # Return only the sequences, sorted by ascending objective
    return [seq for _, seq in best]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimizes
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – coarse search on a smaller grid.
    # ------------------------------------------------------------------
    n_small = 2000
    phase0_deadline = min(deadline, start_time + 0.15 * budget_s)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    print(f"\n=== Phase 0: coarse sweep (n={n_small}) ===")
    candidates_small = _collect_candidates(
        n_small, rng, previous, top_k=30, deadline=phase0_deadline
    )
    print(f"Coarse sweep yielded {len(candidates_small)} candidates.")

    # ------------------------------------------------------------------
    # Upsample to full resolution.
    # ------------------------------------------------------------------
    n_big = 4000
    print(f"\n=== Upsampling {len(candidates_small)} candidates to n={n_big} ===")
    candidates_big = []
    for i, seq_small in enumerate(candidates_small):
        seq_big = _upsample(seq_small, n_big)
        candidates_big.append(seq_big)

    # ------------------------------------------------------------------
    # Phase 1 – short Adam refinements of each upsampled candidate.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Phase 1: short Adam refinements ---")
    for idx, seed_vec in enumerate(candidates_big):
        if time.time() > deadline:
            break
        print(f"[Candidate {idx + 1}/{len(candidates_big)}]")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n_big, deadline, max_iters=200_000
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after short Adam: {best_val:.6f}")

    # Fallback in the unlikely case nothing succeeded
    if best_seq is None:
        best_seq = candidates_big[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first candidate: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam optimisation on the current best.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: long Adam optimisation ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq, n_big, deadline, max_iters=2_000_000
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after long Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: projected sub‑gradient descent ---")
        seq_sub, val_sub = _subgrad_optimize(
            best_seq, n_big, deadline, max_iters=1_200_000
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated global best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n_big, deadline, max_iters=400_000, init_temp_factor=0.05
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated global best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
