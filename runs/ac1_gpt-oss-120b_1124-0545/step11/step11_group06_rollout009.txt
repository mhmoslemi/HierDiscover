"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().  The algorithm mixes several ideas:

1.  A deterministic sweep of simple analytic families + many random
    Dirichlet seeds – the best *top_k* candidates are kept.
2.  Each top candidate is refined by a short Adam run on a smooth
    log‑sum‑exp surrogate of the max‑convolution (the code from the
    previous solution is retained but the iteration budget is reduced
    to allow more time for later phases).
3.  A linear‑programming “direction” step (the classic
    `max(conv(f,g)) ≤ M` LP) is applied a few times.  It finds a
    non‑negative vector g that maximises its ℓ₁‑mass while keeping
    `conv(f,g)` below the current peak M.  Mixing f and the scaled
    solution g strictly reduces the peak in practice.
4.  A long projected sub‑gradient descent (one convolution per
    iteration) is performed – this is cheap and can run for many
    hundred‑thousands of steps.
5.  Finally a lightweight simulated‑annealing polishing phase.

All phases respect a global 1000 s budget and the best sequence found
is stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a probability vector (sum=1).
    Returns 2·n·max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate (same code as before,
    but with a tighter iteration limit that we can call many times).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # initial smooth‑max temperature
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = np.exp(v - np.max(v))
        a = a / a.sum()                     # softmax → simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # raise temperature slowly (approach true max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)                  # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # soft‑max over conv entries

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a Polyak step size (reference to the current best value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        # Polyak step size (if we have a better value we aim for it)
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            # fallback to diminishing steps
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)          # keep on simplex

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple SA: randomly re‑allocate the total mass of a small random subset.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)                # size of the moved block
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming direction step (the classic “M‑LP”)
# ----------------------------------------------------------------------


def _lp_direction_step(f: np.ndarray, M: float, n: int) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀb   s.t.  conv(f, b) ≤ M  , b ≥ 0.
    Returns the optimal b (not normalised).  If LP fails, returns None.
    """
    L = 2 * n - 1

    # Build sparse Toeplitz matrix A where A @ b = conv(f, b)
    # Column k of A contains f shifted by k positions.
    A = sp.lil_matrix((L, n), dtype=np.float64)
    for k in range(n):
        start = k
        end = k + n
        A[start:end, k] = f  # broadcasting puts the whole f in the slice
    A = A.tocsr()

    c = -np.ones(n, dtype=np.float64)                # minimise -sum(b)  → maximise sum(b)
    ub = M * np.ones(L, dtype=np.float64)

    # SciPy’s HiGHS backend handles sparse matrices efficiently.
    res = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None
    return res.x


def _lp_improve(f: np.ndarray, n: int, deadline: float,
                mix_factors=(0.1, 0.2, 0.3, 0.5),
                max_iters: int = 5) -> np.ndarray:
    """
    Apply a few LP‑based improvement steps to f.
    Each step solves the M‑LP, rescales the solution to sum‑1 and mixes
    it back into f with a small factor.  Returns the (potentially) improved f.
    """
    a = f.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        M = np.max(conv)

        b = _lp_direction_step(a, M, n)
        if b is None:
            break
        # Scale to have the same ℓ₁‑mass (which is 1)
        s = b.sum()
        if s <= 0:
            break
        b = b / s

        # Try a few mixing coefficients, keep the best improvement.
        best_mix = None
        best_val = _objective(a)
        for t in mix_factors:
            cand = (1.0 - t) * a + t * b
            val = _objective(cand)
            if val < best_val:
                best_val = val
                best_mix = cand
        if best_mix is not None:
            a = best_mix
        else:
            # No improvement found – stop early.
            break
    return a

def _collect_candidates(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None,
                       top_k: int, deadline: float) -> list[np.ndarray]:
    """
    Systematically explore analytic families and a large set of random
    Dirichlet seeds.  Keep the `top_k` candidates with smallest exact
    objective.
    """
    best = []  # list of (value, seq)

    def keep(val, seq):
        if len(best) < top_k:
            best.append((val, seq))
            best.sort(key=lambda x: x[0])
        elif val < best[-1][0]:
            best[-1] = (val, seq)
            best.sort(key=lambda x: x[0])

    # Deterministic families -------------------------------------------------
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.12),
        lambda n: _trapezoidal(n, width_fraction=0.25),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _sine_window(n, gamma=3.0),
        lambda n: _tukey_window(n, alpha=0.3),
        lambda n: _tukey_window(n, alpha=0.6),
    ]
    for fn in families:
        seq = fn(n)
        keep(_objective(seq), seq)

    # Beta shape grid --------------------------------------------------------
    alphas = np.linspace(0.1, 3.0, 25)
    betas = np.linspace(0.1, 3.0, 25)
    for a in alphas:
        for b in betas:
            seq = _beta_shape(n, a, b)
            keep(_objective(seq), seq)

    # Sine‑window grid -------------------------------------------------------
    gammas = np.linspace(0.5, 5.0, 25)
    for g in gammas:
        seq = _sine_window(n, gamma=g)
        keep(_objective(seq), seq)

    # Tukey‑window grid ------------------------------------------------------
    alps = np.linspace(0.1, 0.9, 20)
    for a in alps:
        seq = _tukey_window(n, alpha=a)
        keep(_objective(seq), seq)

    # Power‑law ---------------------------------------------------------------
    ps = np.linspace(0.0, 5.0, 20)
    for p in ps:
        seq = _power(n, p)
        keep(_objective(seq), seq)

    # Geometric families ------------------------------------------------------
    rs = np.linspace(0.2, 0.98, 20)
    for r in rs:
        seq = _geom(n, r)
        keep(_objective(seq), seq)
        seq2 = _sym_geom(n, r)
        keep(_objective(seq2), seq2)

    # Random Dirichlet seeds --------------------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 60 if conc < 0.01 else 30
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            keep(_objective(seq), seq)

    # Sparse random supports -------------------------------------------------
    for _ in range(200):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)           # 3 … 30 non‑zero entries
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = weights
        keep(_objective(seq), seq)

    # Previous solution -------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        seq = previous / previous.sum()
        keep(_objective(seq), seq)

    return [seq for _, seq in best]

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

            previous = None

    for i, seq in enumerate(candidates):
        print(f"  #{i+1}: {_objective(seq):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – short Adam refinements for each candidate (≈15 % budget)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Phase 1: short Adam refinements ---")
    phase1_deadline = min(deadline, start_time + 0.20 * budget_s)

        if time.time() > phase1_deadline:

        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, phase1_deadline, max_iters=80_000

        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after short Adam: {best_val:.6f}")

    # If nothing improved, fall back to the best from Phase 0.
    if best_seq is None:
        best_seq = candidates[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to best Phase‑0 candidate: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based directional improvement (≈5 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: LP‑directional improvement ---")

        print(f"  After LP step best objective: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long projected sub‑gradient descent (≈30 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: long projected sub‑gradient descent ---")
        subgrad_deadline = min(deadline, start_time + 0.55 * budget_s)
        seq_sub, val_sub = _subgrad_optimize(
            best_seq, n, subgrad_deadline, max_iters=1_000_000
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final simulated‑annealing polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: simulated annealing polishing ---")
        sa_deadline = deadline
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, sa_deadline, max_iters=400_000, init_temp_factor=0.04
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
