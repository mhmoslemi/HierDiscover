"""
Propose a non‑negative step‑function that minimizes the official evaluation
function.  The algorithm uses

1.  A fast seed‑generation phase (deterministic families + random Dirichlet
    draws) to obtain a promising ℓ₁‑normalised candidate.
2.  A multi‑stage L‑BFGS optimisation of a smooth‑max surrogate of the
    convolution maximum (increasing the temperature β to approach the true
    max).
3.  Projected sub‑gradient descent on the exact max‑convolution objective
    for final polishing.
4.  A short simulated‑annealing phase to escape possible shallow minima.

All stages respect the global time budget (default 1000 s).  The best
sequence found is scaled to sum = √(2 n) (as required by the official
evaluator) and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a sequence that already sums to 1.
    Returns max(conv(seq, seq)).
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int) -> np.ndarray:
    """Trapezoid with flat width ≈30 % of the support."""
    width_fraction = 0.15
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window_1(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window_2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** 2.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_12(n: int) -> np.ndarray:
    sigma_frac = 0.12
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_24(n: int) -> np.ndarray:
    sigma_frac = 0.24
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_02(n: int) -> np.ndarray:

def _tukey_04(n: int) -> np.ndarray:
    return _tukey_window(n, alpha=0.4)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)          # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian_24(n)  # use the 0.24‑σ version as a base
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate for L‑BFGS
# ----------------------------------------------------------------------


class _SmoothMaxGrad:
    """
    Callable returning (value, gradient) of the smooth‑max surrogate
    f(v) = Σ w_i·conv_i ,   w_i = softmax(β·conv_i) .
    The variables are v (logits) and a = softmax(v) lives on the simplex.
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v: np.ndarray):
        a = _softmax(v)
        conv = _conv_fft(a, a)                     # length 2n‑1
        shift = np.max(conv)
        w = np.exp(self.beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # degenerate – return a large value and zero gradient
            return shift, np.zeros_like(v)
        w = w / w_sum                              # normalized weights

        # gradient w.r.t. a (treated as if w were constant)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # chain rule to logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # smooth‑max value (expectation of conv under w)
        val = np.dot(conv, w)
        return float(val), grad_v.astype(np.float64)


class _TimeCallback:
    """Callback used in SciPy minimise to abort when the global deadline is hit."""
    def __init__(self, deadline: float):
        self.deadline = deadline

    def __call__(self, xk):
        if time.time() > self.deadline:
            raise RuntimeError("Time limit reached")


def _lbfgs_optimize(v_start: np.ndarray,
                    n: int,
                    beta: float,
                    max_iter: int,
                    deadline: float) -> np.ndarray:
    """Run L‑BFGS‑B on the smooth‑max surrogate (up to the deadline)."""
    try:
        import scipy.optimize as opt
    except Exception as exc:
        # SciPy not available – fall back to the original Adam routine
        print("SciPy not available, falling back to Adam.")
        from math import inf
        return v_start

    obj = _SmoothMaxGrad(n, beta)
    cb = _TimeCallback(deadline)

    try:
        res = opt.minimize(fun=obj,
                           x0=v_start,
                           method='L-BFGS-B',
                           jac=True,
                           callback=cb,
                           options={'maxiter': max_iter,
                                    'disp': False})
        return res.x
    except RuntimeError:
        # timeout – return current iterate
        return v_start


# ----------------------------------------------------------------------
# Sub‑gradient optimisation on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgrad_optimize(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best sequence encountered and its exact value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)                     # max(conv)  (sum=1)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = conv[max_idx]                  # max(conv)  (sum=1)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.8f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (using current best)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (tiny Dirichlet re‑allocations)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 400_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Very small‑move simulated annealing on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)               # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 20000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.8f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence it finds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # Length of the step‑function – 4000 is a good trade‑off.
    n = 4000
    sqrt2n = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 0. Seed generation (≈12 % of the total budget)
    # --------------------------------------------------------------
    print("\n--- Seed generation ---")
    seed_deadline = start_time + budget_s * 0.12

    best_val = float("inf")
    best_seq = None

    # deterministic analytic families (no lambdas)
    families = [_uniform,
                _triangular,
                _trapezoidal,
                _sine_window_1,
                _sine_window_2,
                _gaussian_12,
                _gaussian_24,
                _tukey_02,
                _tukey_04]

    for fn in families:
        seq = fn(n)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        for fn in (_geom, _sym_geom):
            seq = fn(n, r)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()

    # power‑law family
    for p in np.linspace(0.0, 5.0, 18):
        seq = _power(n, p)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # random Dirichlet draws – sweep many concentration levels
    concs = [0.001, 0.005, 0.01, 0.02, 0.05,
             0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 70 if conc < 0.01 else 40
        for _ in range(draws):
            if time.time() > seed_deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()

    # sparse random supports
    for _ in range(250):
        if time.time() > seed_deadline:
            break
        seq = _sparse_seed(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # random beta shapes
    for _ in range(300):
        if time.time() > seed_deadline:
            break
        alpha = rng.uniform(0.05, 5.0)
        beta = rng.uniform(0.05, 5.0)
        seq = _beta_shape(n, alpha, beta)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # random mixtures of uniform & Gaussian
    for _ in range(150):
        if time.time() > seed_deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # previous solution if any
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.size == n and prev_arr.sum() > 0:
                prev_arr = prev_arr / prev_arr.sum()
                val = _objective(prev_arr)
                if val < best_val:
                    best_val = val
                    best_seq = prev_arr.copy()
        except Exception:
            pass

    # Ensure we have a valid seed
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)

    print(f"Best seed after generation: max conv = {best_val:.8f}")

    # --------------------------------------------------------------
    # 1. L‑BFGS optimisation of the smooth‑max surrogate
    # --------------------------------------------------------------
    print("\n--- L‑BFGS optimisation (smooth‑max) ---")
    # work on log‑softmax variables
    v = np.log(best_seq + 1e-30)
    # a small schedule of β values – increase to tighten the surrogate
    beta_schedule = [20.0, 200.0, 2000.0]
    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        max_iter = 2000
        print(f"  β = {beta:.0f}, max_iter = {max_iter}")
        v = _lbfgs_optimize(v_start=v,
                            n=n,
                            beta=beta,
                            max_iter=max_iter,
                            deadline=deadline)
        a_candidate = _softmax(v)
        cand_val = _objective(a_candidate)
        print(f"    after β={beta:.0f} → max conv = {cand_val:.8f}")
        if cand_val < best_val:
            best_val = cand_val
            best_seq = a_candidate.copy()
            print("    ** New global best from L‑BFGS **")

    # --------------------------------------------------------------
    # 2. Projected sub‑gradient descent on the exact objective
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (exact max) ---")
        seq_sub, val_sub = _subgrad_optimize(start_seq=best_seq,
                                             n=n,
                                             deadline=deadline,
                                             max_iters=800_000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub.copy()
            print(f"  Updated global best after sub‑grad: {best_val:.8f}")

    # --------------------------------------------------------------
    # 3. Simulated‑annealing polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(start_seq=best_seq,
                                              n=n,
                                              deadline=deadline,
                                              max_iters=400_000)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"  Updated global best after SA: {best_val:.8f}")

    # --------------------------------------------------------------
    # 4. Scale to the required ℓ₁‑norm (√(2 n)) and final check
    # --------------------------------------------------------------
    best_seq = best_seq * sqrt2n          # now sum = √(2 n)

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nOfficial objective ≈ {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
