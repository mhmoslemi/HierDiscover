def _expgrad_optimize(seed_seq, n, deadline, max_iters):
    a = seed_seq.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    N1 = _next_pow2(2*n-1)
    N2 = _next_pow2(3*n-2)
    lam_final = 15000.0
    lr0 = 0.5
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # schedule
        lam = lam_final * (it / max_iters)
        # compute conv_a
        a_fft = np.fft.rfft(a, n=N1)
        conv_a_full = np.fft.irfft(a_fft * a_fft, n=N1)[:2*n-1]
        max_conv = conv_a_full.max()
        # compute weights
        # using lam>0
        w_raw = np.exp(lam * (conv_a_full - max_conv))
        w = w_raw / w_raw.sum()
        # Compute gradient: conv(w, a_rev)
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=N2)
        rev_fft = np.fft.rfft(a_rev, n=N2)
        conv_wr_full = np.fft.irfft(w_fft * rev_fft, n=N2)[:3*n-2]
        grad = 2.0 * conv_wr_full[n-1:n-1+n]
        # exponential gradient update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum
        # evaluate objective
        cur_val = 2 * n * max_conv
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[ExpGrad] it={it} best={best_val:.6f}")
    return best_seq, best_val

Algorithm:
- Generate a diverse set of ℓ₁‑normalised candidate step heights (analytic windows,
  Dirichlet draws, low‑frequency sinusoidal curves, mixtures).
- Pick the few candidates with the lowest surrogate value 2·n·max(conv).
- For each selected seed run a smoothed‑max gradient descent on the simplex.
  The smoothing is a soft‑max over the auto‑convolution values; the gradient
  is computed efficiently with FFT‑based convolutions.  Updates are performed
  in log‑space (exponentiated gradient) and the step‑size as well as the
  soft‑max temperature are annealed.
- Keep the best sequence found and finally evaluate it with
  ``evaluate_sequence``.

import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective_surrogate(seq):
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families ----------------------------------------------------------
def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n, gamma=1.0):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n, sigma_frac):
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    s = a.sum()

# Exponentiated‑gradient optimiser -------------------------------------------
def _expgrad_optimize(seed_seq, n, deadline, max_iters=200_000,
                      lr0=0.5, lam_max=15000.0):
    """Smoothed‑max descent on the simplex using exponentiated updates."""
    a = np.asarray(seed_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = _uniform(n)
    else:
        a = a / a_sum
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    # pre‑compute FFT lengths
    N1 = _next_pow2(2 * n - 1)          # for a*a
    N2 = _next_pow2(3 * n - 2)          # for w*a_rev
    eps = 1e-30
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # linear schedule for the smoothing parameter
        lam = lam_max * (it / max_iters)   # gradually increase
        # a*a convolution (full)
        a_fft = np.fft.rfft(a, n=N1)
        conv_a = np.fft.irfft(a_fft * a_fft, n=N1)[:2 * n - 1]
        max_conv = np.max(conv_a)
        # soft‑max weights on the convolution entries
        w_raw = np.exp(lam * (conv_a - max_conv))
        w = w_raw / w_raw.sum()
        # gradient = 2 * (w * a_rev) convolution
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=N2)
        rev_fft = np.fft.rfft(a_rev, n=N2)
        conv_wr = np.fft.irfft(w_fft * rev_fft, n=N2)[:3 * n - 2]
        grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
        # exponentiated‑gradient step in log‑space
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * grad
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum <= 0:
            a = _uniform(n)
        else:
            a /= a_sum
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")
    return best_seq, best_val

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # FFT sizes required for the two convolutions we use
    N1 = _next_pow2(2 * n - 1)      # for a * a
    N2 = _next_pow2(3 * n - 2)      # for w * a_rev
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linearly increase the smoothing (soft‑max) parameter
        lam = lam_max * (it / max_iters)

        # ----- a * a (full) -------------------------------------------------
        a_fft = np.fft.rfft(a, n=N1)
        conv_a = np.fft.irfft(a_fft * a_fft, n=N1)[:2 * n - 1]
        max_conv = np.max(conv_a)

        # ----- soft‑max weights over convolution entries --------------------
        w_raw = np.exp(lam * (conv_a - max_conv))
        w = w_raw / w_raw.sum()

        # ----- gradient = 2 * (w ⋆ a_rev) ---------------------------------
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=N2)
        rev_fft = np.fft.rfft(a_rev, n=N2)
        conv_wr = np.fft.irfft(w_fft * rev_fft, n=N2)[:3 * n - 2]

        # the relevant slice corresponds to indices n‑1 … n‑1+n‑1
        grad = 2.0 * conv_wr[n - 1 : n - 1 + n]

        # ----- exponentiated‑gradient step (log‑space) ----------------------
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * grad
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum <= 0:
            a = _uniform(n)
        else:
            a /= a_sum

        cur_val = 2.0 * n * max_conv
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective.
    """
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # a tiny safety margin

    # length of the sequence – a good trade‑off between quality and speed
    n = 4000

    # ------------------------------------------------------------------
    # 1) Seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"[Info] generated {len(seeds)} seed vectors (n={n}).")

    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])

    top_k = 5
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"[Info] top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"   #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Optimisation from each promising seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, start=1):
        now = time.time()
        if now >= deadline:
            break
        remaining = deadline - now
        # allocate a roughly equal slice of the remaining budget
        # estimate 300–400 simple iterations per second for n≈4000
        est_iters = int(min(120_000, remaining * 350))
        if est_iters < 5_000:
            continue
        print(f"\n[Run {idx}] optimisation start – allowed iters = {est_iters}")
        seq, val = _expgrad_optimize(seed_vec, n, deadline,
                                     max_iters=est_iters,
                                     lr0=0.5, lam_max=15000.0)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Run {idx}] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
