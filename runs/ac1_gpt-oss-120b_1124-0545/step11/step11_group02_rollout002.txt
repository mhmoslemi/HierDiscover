"""
Propose a non‑negative step‑function (probability vector) that improves the
current best bound on ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Seed pool** – deterministic analytic families (uniform, triangular,
   geometric, beta, Gaussian, …) plus a few random Dirichlet draws and the
   previous best solution (if any). All vectors are ℓ₁‑normalised (sum = 1).

2. **Differential Evolution (DE)** – a population‑based global optimiser.
   Each individual is projected onto the probability simplex after mutation
   and crossover. The fitness is the inexpensive surrogate
   ``_objective`` = 2·n·max(a*a). DE explores the search space far more
   aggressively than pure gradient methods.

3. **Mirror‑descent refinement** – the best DE candidate is fine‑tuned with
   an exponential‑weights (entropic) mirror‑descent update that follows the
   gradient of the currently maximal convolution entry.

4. **Simulated‑annealing polishing** – a lightweight annealing loop that
   redistributes mass among a random subset of coordinates (Dirichlet move)
   and accepts uphill moves according to a geometric temperature schedule.

All stages respect the 1000 s time budget and the final sequence is stored
in the global ``height_sequence_1`` for reuse across runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate for ``evaluate_sequence`` when sum(seq) == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is a zero vector – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12),
                             np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # random exponential draws
    for _ in range(5):
        a = rng.exponential(scale=1.0, size=n)
        s = a.sum()
        seeds.append(a / s if s > 0 else _uniform(n))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _exp_mirror_descent(start_vec: np.ndarray, n: int,
                        deadline: float, max_iters: int = 400_000,
                        eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Entropic mirror descent on the L∞‑convolution objective.
    Sub‑gradient is taken at the currently maximal convolution entry.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # current surrogate value
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Mirror] it={it:6d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        idx = np.arange(n)
        rev = max_idx - idx
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        eta = eta0 / math.sqrt(it)          # decreasing learning rate
        a = a * np.exp(-eta * g)            # entropic step
        a = np.maximum(a, 0.0)

        if a.sum() == 0.0:
            a = _uniform(n)
        else:
            a = a / a.sum()                  # projection onto simplex

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing polishing.
    Randomly reshuffles a subset of coordinates using a Dirichlet move.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, min(20, n // 10 + 2))   # size of the reshuffle set
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _de_optimize(pop: np.ndarray, pop_vals: np.ndarray, n: int,
                 deadline: float,
                 F_range: tuple[float, float] = (0.5, 0.9),
                 CR_range: tuple[float, float] = (0.4, 0.9)) -> tuple[np.ndarray, float]:
    """
    Differential Evolution (DE/rand/1/bin) on the surrogate objective.
    Population vectors are maintained on the probability simplex.
    """
    rng = np.random.default_rng()
    popsize = pop.shape[0]
    best_idx = int(np.argmin(pop_vals))
    best_seq = pop[best_idx].copy()
    best_val = float(pop_vals[best_idx])

    while time.time() < deadline:
        order = rng.permutation(popsize)
        for i in order:
            if time.time() >= deadline:
                break

            # three distinct indices different from i
            idxs = rng.choice([j for j in range(popsize) if j != i],
                              size=3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]

            F = rng.uniform(*F_range)
            mutant = a + F * (b - c)

            CR = rng.uniform(*CR_range)
            cross_mask = rng.random(n) < CR
            if not cross_mask.any():
                cross_mask[rng.integers(0, n)] = True

            trial = np.where(cross_mask, mutant, pop[i])
            trial = _proj_simplex(trial)

            val = _objective(trial)
            if val < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_seq = trial.copy()
        # occasional status report
        if int(time.time()) % 30 == 0:
            print(f"[DE] current best = {best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev = None
    else:
        prev = None

    # ------------------------------------------------------------------
    # Length of the candidate
    # ------------------------------------------------------------------
    n = 4000   # a compromise between expressiveness and speed

    # ------------------------------------------------------------------
    # Stage 1 – collect a diverse pool of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, prev)

    # add a few extra random Dirichlet draws with varied concentrations
    for _ in range(30):
        conc = 10 ** rng.uniform(-3, 1)   # between 0.001 and 10
        seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # evaluate all seeds with the fast surrogate
    seed_vals = [_objective(s) for s in seeds]
    seed_tuples = list(zip(seeds, seed_vals))
    seed_tuples.sort(key=lambda x: x[1])
    best_seq = seed_tuples[0][0].copy()
    best_val = seed_tuples[0][1]
    print(f"Initial best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Differential Evolution (global search)
    # ------------------------------------------------------------------
    popsize = 60
    # initialise population with the best few seeds + random draws
    population = [vec for vec, _ in seed_tuples[:min(10, len(seed_tuples))]]
    while len(population) < popsize:
        conc = rng.uniform(0.01, 5.0)
        population.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    pop = np.stack(population)                     # shape (popsize, n)
    pop_vals = np.array([_objective(v) for v in pop])

    # update best from the initial population
    idx = int(np.argmin(pop_vals))
    if pop_vals[idx] < best_val:
        best_val = float(pop_vals[idx])
        best_seq = pop[idx].copy()

    time_left = deadline - time.time()
    if time_left > 10.0:
        de_deadline = time.time() + time_left * 0.55
        print(f"\n--- Differential Evolution (≈{de_deadline - time.time():.1f}s) ---")
        best_seq, best_val = _de_optimize(pop, pop_vals, n, de_deadline)
        print(f"[DE] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Mirror‑descent refinement (local exploitation)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        mirror_deadline = time.time() + time_left * 0.35
        print(f"\n--- Mirror Descent refinement (≈{mirror_deadline - time.time():.1f}s) ---")
        seq_m, val_m = _exp_mirror_descent(best_seq, n, mirror_deadline,
                                           max_iters=600_000, eta0=0.6)
        if val_m < best_val:
            best_seq, best_val = seq_m, val_m
            print(f"[Mirror] new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing (final fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated Annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              max_iters=800_000,
                                              init_temp_factor=0.04)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] final global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # store for possible reuse in later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
