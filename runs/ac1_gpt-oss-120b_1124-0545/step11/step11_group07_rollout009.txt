"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm (≈1000 s budget)

1. **Diverse seeding** – generate a small pool of analytical families
   (uniform, triangular, windows, Dirichlet draws, sparse supports, …)
   and keep the best few (default 5) after exact evaluation.

2. **Adam on a smooth max** – each seed is refined with Adam on a
   log‑sum‑exp surrogate of the convolution maximum.  The temperature is
   raised gradually up to a large value (β≈500) so that the surrogate
   converges to the true max.  The best among all seeds becomes the current
   global optimum.

3. **Long Adam run** – a longer Adam optimisation starting from the
   global optimum (≈4 M iterations or until the budget expires).

4. **Quadratic‑programme refinement** – solve a linear programme

        max  Σ b_i
        s.t.   (f * b)_k ≤ M   ∀k ,  b_i ≥ 0

   where ``f`` is the current vector (Σf=1) and ``M`` = max(f*f).
   The optimiser returns a direction ``g`` that does not increase the
   max‑convolution.  A convex combination ``f←(1‑t)f+t·g`` (t≈0.2) is
   performed a few times.

5. **Projected sub‑gradient descent** – a few hundred thousand steps of
   exact sub‑gradient descent on the true ``max(conv(f,f))``.

6. **Simulated‑annealing polishing** – tiny Dirichlet re‑allocations on
   random subsets to escape shallow minima.

7. **Optional symmetrisation** – replace ``f`` by ``(f+rev(f))/2`` and keep
   the improvement if any.

Finally the simplex vector is scaled to the required ℓ₁‑norm
``√(2·n)`` (the official evaluator multiplies by this factor) and the
best sequence is stored in the global ``height_sequence_1`` variable.

All phases respect the total time budget; intermediate progress and
objective values are printed.
"""

import time
import numpy as np
from scipy import optimize, sparse

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant). ``seq`` is assumed to sum to 1;
    the function returns max(conv(seq, seq)).
    """
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all ℓ₁‑normalised to sum = 1)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)          # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

def _collect_diverse_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    top_k: int,
    deadline: float,
) -> list[np.ndarray]:
    """
    Generate a diverse pool of ℓ₁‑normalised candidates and return the best
    `top_k` of them w.r.t. the exact objective.
    """
    top_vals = []  # (value, seq)

    def try_insert(val, seq):
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])

    # deterministic families (always useful)
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.15),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _gaussian(n, sigma_frac=0.12),
        lambda n: _gaussian(n, sigma_frac=0.24),
        lambda n: _tukey_window(n, alpha=0.2),
        lambda n: _tukey_window(n, alpha=0.4),
    ]
    for fn in families:
        seq = fn(n)
        try_insert(_objective(seq), seq)

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        for fn in (_geom, _sym_geom):
            seq = fn(n, r)
            try_insert(_objective(seq), seq)

    # power‑law family
    for p in np.linspace(0.0, 5.0, 18):
        seq = _power(n, p)
        try_insert(_objective(seq), seq)

    # random Dirichlet draws – sweep many concentration levels
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 70 if conc < 0.01 else 40
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            try_insert(_objective(seq), seq)

    # sparse random supports
    for _ in range(250):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        try_insert(_objective(seq), seq)

    # random beta shapes
    for _ in range(300):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 5.0)
        beta = rng.uniform(0.05, 5.0)
        seq = _beta_shape(n, alpha, beta)
        try_insert(_objective(seq), seq)

    # random sine windows (varying exponent)
    for _ in range(200):
        if time.time() > deadline:
            break
        gamma = rng.uniform(0.2, 5.0)
        seq = _sine_window(n, gamma=gamma)
        try_insert(_objective(seq), seq)

    # random Gaussian windows
    for _ in range(200):
        if time.time() > deadline:
            break
        sigma_frac = rng.uniform(0.05, 0.4)
        seq = _gaussian(n, sigma_frac=sigma_frac)
        try_insert(_objective(seq), seq)

    # random Tukey windows
    for _ in range(150):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 0.9)
        seq = _tukey_window(n, alpha=alpha)
        try_insert(_objective(seq), seq)

    # mixture of uniform & Gaussian
    for _ in range(150):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        try_insert(_objective(seq), seq)

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        seq = np.asarray(previous, dtype=np.float64)
        if seq.sum() > 0:
            seq = seq / seq.sum()
            try_insert(_objective(seq), seq)

    # Return only the sequences (sorted by increasing value)
    return [seq for _, seq in top_vals]

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    beta_max: float = 500.0,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Returns (best_sequence, best_exact_value).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        a = _softmax(v)                  # current point on simplex (sum=1)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6e}")

        # raise temperature
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # convolution and smooth‑max weights
        conv = _conv_fft(a, a)           # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum                    # softmax over convolution entries

        # gradient wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)       # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best sequence encountered and its exact value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]   # the official scaling (unused here)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6e}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (using current best)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Tiny‑move simulated‑annealing on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)       # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6e}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programme refinement (LP) – small number of heavy steps
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, n: int, deadline: float, t: float = 0.2) -> tuple[np.ndarray, float]:
    """
    One LP‑based refinement step.
    Solves   max Σ b_i  s.t. (f * b)_k ≤ M, b ≥ 0,
    where M = max(f * f).  Returns the new vector (convex combination)
    and its objective value.
    """
    if time.time() >= deadline:
        return f, _objective(f)

    M = _objective(f)                     # current max‑convolution
    # Build the (2n‑1) × n convolution matrix A such that A @ b = f * b
    offsets = np.arange(n, dtype=int)
    # each diagonal stores a constant f[d] on all n entries
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    c = -np.ones(n, dtype=np.float64)      # maximise Σ b → minimise –Σ b
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # solve linear programme
    res = optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        print("[LP] linprog failed:", res.message)
        return f, M

    b0 = res.x
    sum_b0 = b0.sum()
    if sum_b0 <= 1e-12:
        print("[LP] zero‑mass solution")
        return f, M

    # Normalise to sum = 1 and take a convex blend
    g = b0 / sum_b0
    new = (1.0 - t) * f + t * g
    new = _proj_simplex(new)               # keep exact sum = 1
    new_val = _objective(new)
    return new, new_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # length of the step‑function – a good trade‑off between flexibility
    # and speed.
    n = 4000
    sqrt2n = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 0 – seed generation (≈10 % of total budget)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    seed_deadline = min(deadline, start_time + budget_s * 0.12)
    seeds = _collect_diverse_seeds(
        n=n,
        rng=rng,
        previous=previous,
        top_k=5,
        deadline=seed_deadline,
    )
    print(f"Collected {len(seeds)} promising seeds (kept top‑{len(seeds)}).")

    # ------------------------------------------------------------------
    # Phase 1 – Adam refinement of the seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Adam refinement of seeds (phase 1) ---")
    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"[Seed {idx + 1}/{len(seeds)}]")
        seed_vec = seed_vec / seed_vec.sum()
        seq_adam, val_adam = _adam_optimize(
            start_vec=seed_vec,
            n=n,
            deadline=deadline,
            max_iters=1_500_000,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after seed Adam: {best_val:.6e}")

    # If nothing succeeded, fall back to uniform.
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print(f"Fallback to uniform seed: {best_val:.6e}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam run on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long Adam optimisation (phase 2) ---")
        seq_adam, val_adam = _adam_optimize(
            start_vec=best_seq,
            n=n,
            deadline=deadline,
            max_iters=4_000_000,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after long Adam: {best_val:.6e}")

    # ------------------------------------------------------------------
    # Phase 3 – a few LP refinements
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP refinement (phase 3) ---")
        lp_steps = 0
        while time.time() < deadline and lp_steps < 3:
            new_seq, new_val = _lp_refine(best_seq, n, deadline, t=0.2)
            if new_val < best_val:
                best_seq = new_seq
                best_val = new_val
                print(f"[LP] step {lp_steps + 1} improved value {best_val:.6e}")
            else:
                print(f"[LP] step {lp_steps + 1} no improvement")
            lp_steps += 1

    # ------------------------------------------------------------------
    # Phase 4 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (phase 4) ---")
        seq_sub, val_sub = _subgrad_optimize(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=600_000,
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated global best after sub‑grad: {best_val:.6e}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing (phase 5) ---")
        seq_sa, val_sa = _simulated_annealing(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=500_000,
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated global best after SA: {best_val:.6e}")

    # ------------------------------------------------------------------
    # Phase 6 – optional local perturbation + short sub‑grad runs
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # Small random perturbation of the current best
        sigma = 0.02
        pert = best_seq + sigma * rng.normal(size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _proj_simplex(pert)

        seq_local, val_local = _subgrad_optimize(
            start_seq=pert,
            n=n,
            deadline=deadline,
            max_iters=200_000,
        )
        if val_local < best_val:
            best_seq = seq_local.copy()
            best_val = val_local
            print(f"[Local] improvement to {best_val:.6e}")
        else:
            # no improvement – break after a few attempts
            break

    # ------------------------------------------------------------------
    # Phase 7 – optional symmetrisation (often beneficial)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sym = (best_seq + best_seq[::-1]) / 2.0
        sym = _proj_simplex(sym)
        sym_val = _objective(sym)
        if sym_val < best_val:
            best_seq = sym
            best_val = sym_val
            print(f"[Symmetrise] improved to {best_val:.6e}")

    # ------------------------------------------------------------------
    # Scale the simplex vector to the required ℓ₁‑norm (√(2·n))
    # ------------------------------------------------------------------
    best_seq = best_seq * sqrt2n

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest official objective ≈ {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
