    best_seq = a.copy()
    best_val = _objective(a)
    beta = 0.5
    beta_max = 500.0
    inc = 0.2  # as step
    lr0 = 0.2

    it = 1
    while time.time() < deadline:
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute weight vector
        shift = max_val
        w = np.exp(beta * (conv - shift))
        ws = w.sum()
        if ws == 0.0:
            # fallback to uniform weight for stability
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= ws
        # gradient
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)
        grad = 2.0 * grad_full[n-1:n-1+n]   # central n entries
        # update
        it += 1
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * grad)
        sum_a = a.sum()
        if sum_a <= 0.0:
            a = _uniform(n)

                a = alt
                if alt_val < best_val:

"""
Propose a near‑optimal non‑negative step function.

Algorithm
---------
1. **Rich seed pool** – deterministic families (uniform, triangular,
   sine, Gaussian, power‑law, beta, Tukey, exponential, logarithmic,
   low‑frequency random curves) and Dirichlet draws are generated and
   ℓ₁‑normalised.

2. **Quick screening** – the surrogate objective
   ``2·n·max(conv(seq,seq))`` (identical to the official score for
   ℓ₁‑normalised vectors) is evaluated for every seed, the best few
   are kept.

3. **Smooth mirror descent** – starting from the best seed we perform
   an exponentiated‑gradient (mirror) descent on the *exact* objective,
   but the sub‑gradient is taken with respect to a **log‑sum‑exp**
   smoothing of the max.  The temperature `β` is increased gradually
   (0.5 → 500) so that early iterations see a smooth landscape and the
   last iterations focus on the true maximum.  After each update the
   vector is re‑projected onto the probability simplex (simple normalisation).

   Every 100 000 iterations we also try the “large‑small alternating”
   ordering; if it improves the surrogate we switch to it.

4. **Simulated‑annealing polish** – a cheap SA that repeatedly swaps
   entries or redistributes mass among a few coordinates, accepting
   downhill moves or uphill moves with a slowly cooling temperature.

5. **Final check** – the resulting sequence is evaluated with the
   official ``evaluate_sequence`` function, stored in the global name
   ``height_sequence_1`` and returned.

The whole procedure respects the 1000 s budget (a 2 s safety margin)
and works for any previously stored solution (`height_sequence_1`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (surrogate for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a length‑n sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, … (keeps ℓ₁ norm)."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    # renormalise (should already sum to 1, but guard against round‑off)
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a rich list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # sine window over a grid of gamma
    for gamma in np.linspace(0.5, 6.0, 12):
        w = _sine_window(n, gamma)
        seeds.append(w)
        seeds.append(_alternate_order(w))

    # Gaussian family
    for sigma in np.linspace(0.10, 0.55, 10):
        g = _gaussian(n, sigma)
        seeds.append(g)
        seeds.append(_alternate_order(g))

    # Power‑law family
    for p in np.linspace(0.0, 8.0, 17):
        pvec = _power(n, p)
        seeds.append(pvec)
        seeds.append(_alternate_order(pvec))

    # Beta family (wide range)
    for _ in range(15):
        alpha = 10 ** rng.uniform(-2, 0.6)   # ~0.01 … 4
        beta = 10 ** rng.uniform(-2, 0.6)
        b = _beta_shape(n, alpha, beta)
        seeds.append(b)
        seeds.append(_alternate_order(b))

    # Exponential family
    for rate in np.logspace(-4, -1, 12):
        e = _exponential(n, rate)
        seeds.append(e)
        seeds.append(_alternate_order(e))

    # Logarithmic family
    for eps in np.logspace(-4, -1, 8):
        l = _logarithmic(n, eps)
        seeds.append(l)
        seeds.append(_alternate_order(l))

    # Dirichlet draws with several concentrations
    for conc in (0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_alternate_order(raw))
            seeds.append(np.sort(raw)[::-1])      # descending

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            c = _lowfreq_curve(n, nf, rng)
            seeds.append(c)
            seeds.append(_alternate_order(c))

    # Mixtures of uniform & low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curv = _lowfreq_curve(n, nf, rng)
            m = _mix_shapes(_uniform(n), curv, mix_alpha)
            seeds.append(m)
            seeds.append(_alternate_order(m))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Smooth mirror‑descent optimiser (exponential weights on a smoothed max)
# ----------------------------------------------------------------------
def _smooth_mirror_optimize(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            beta_max: float = 500.0,
                            beta_inc: float = 0.5,
                            lr0: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *log‑sum‑exp* smoothed max.
    Returns the best vector seen (ℓ₁‑normalised) and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    beta = 0.5           # initial temperature
    it = 0

    while time.time() < deadline:
        it += 1
        # ----- convolution & surrogate objective -----
        conv = _conv_fft(a, a)                 # length 2·n‑1
        max_c = np.max(conv)
        cur_val = 2.0 * n * max_c

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ----- smoothed weights -----
        shift = max_c
        w = np.exp(beta * (conv - shift))     # max exponent = 0
        w_sum = w.sum()
        if w_sum == 0.0:                       # numerical safety
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient of Σ w_k·conv_k -----
        rev_a = a[::-1]
        # convolve rev_a (size n) with w (size 2·n‑1) → length 3·n‑2
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # optional clipping to avoid overflow in the exponent
        grad = np.clip(grad, -1e4, 1e4)

        # ----- multiplicative (mirror) update -----
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # ----- temperature schedule -----
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

        # ----- occasional re‑ordering (large‑small alternation) -----
        if it % 100_000 == 0:
            alt = _alternate_order(a)
            alt_val = _objective(alt)
            if alt_val < cur_val:
                a = alt
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()

        # guard against tiny time overrun
        if time.time() > deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (same as in the reference implementation)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 800_000,
                                 init_temp_factor: float = 0.03,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA with swap / mass‑transfer moves."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                                    # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                                  # redistribute among k entries
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                                  # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

        # occasional re‑ordering (large‑small) inside SA
        if it % 100_000 == 0:
            alt = _alternate_order(cur)
            alt_val = _objective(alt)
            if alt_val < best_val:
                best_seq = alt.copy()
                best_val = alt_val
                cur = alt.copy()
                cur_val = alt_val

    return best_seq, best_val

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The function respects the 1000 s time budget
    and stores the best sequence in the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # small safety margin

    # ------------------------------------------------------------------
    # Length of the candidate (4000 works well for the current budget)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous best if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – generate a rich pool of seed vectors
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Collected {len(seeds)} raw seeds.")

    # Evaluate quickly with the surrogate objective (ℓ₁‑normalised)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate among seeds: {evaluated[0][1]:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth mirror descent starting from the best seed
    # ------------------------------------------------------------------
    current = top_seeds[0].copy()
    cur_val = _objective(current)
    best_seq = current.copy()
    best_val = cur_val
    print(f"\n[Init] Starting smooth mirror descent, initial value = {cur_val:.6f}")

    if time.time() < deadline:
        seq_m, val_m = _smooth_mirror_optimize(current, n, deadline)
        if val_m < best_val:
            best_seq, best_val = seq_m, val_m
            print(f"[Mirror] New best after smooth mirror = {val_m:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing polish (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(best_seq, n, deadline)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] New best after polishing = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official function
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} (length {n})")
    return best_seq.tolist()
