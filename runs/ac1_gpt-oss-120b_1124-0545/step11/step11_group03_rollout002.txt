def adam_optimize(f_init, n, deadline, max_iters=1_000_000):
    f = f_init.copy()
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.02  # base step
    best_f = f.copy()
    best_val = surrogate(f)
    for t in range(1, max_iters+1):
        if time.time() > deadline: break
        conv = conv_fft(f, f)
        # find active shift index
        k = int(np.argmax(conv))
        # compute gradient
        grad = np.zeros_like(f)
        # For all i where 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev = k - idx
        grad[idx] = 2.0 * f[rev]
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad**2)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        f = f - step
        # projection onto simplex
        f = proj_simplex(f)  # ensure non-negative sum=1
        # evaluate surrogate
        val = surrogate(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            # optionally print
            if t % 50000 == 0:
                print(f"[Adam] it={t} best={best_val:.6f}")
    return best_f, best_val

"""
Enhanced search for a non‑negative step function that minimises
the evaluate_sequence objective.

Algorithm
---------
1. **Seed generation** – a rich library of analytic windows,
   random Dirichlet draws and a simple exponential‑decay shape.
   All seeds are ℓ₁‑normalised (sum = 1) and the ten best w.r.t.
   the surrogate objective 2·n·max(conv) are kept.

2. **Exponentiated‑gradient refinement** – for each good seed we run a
   short mirror‑descent (Exponential‑Gradient) loop.  The update is
   `x_i ← x_i·exp(-η·g_i)` where `g` is the sub‑gradient of the current
   maximal convolution entry, followed by a simplex projection.
   This quickly drives the convolution peak down.

3. **Long‑run EG polishing** – the current best vector is refined for as
   many iterations as time permits (up to ~8·10⁵ steps).  A decreasing
   stepsize η = η₀/√t is used; the algorithm works entirely with
   FFT‑based convolutions, thus scaling to thousands of variables.

4. **Simulated‑annealing polishing** – an aggressive SA phase reshuffles
   mass on random subsets and transfers tiny chunks between positions.
   Temperature is slowly cooled; any improvement in the surrogate is
   accepted (or occasionally accepted probabilistically).

5. **Final evaluation** – the best sequence is passed to the official
   `evaluate_sequence` function, stored globally as ``height_sequence_1``
   and returned as a plain Python list.

All stages respect the global deadline (default 1000 s).  The code uses
only NumPy (no heavy LP matrices), runs on a single process and fits
the 2‑CPU limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate: 2·n·max(conv(seq, seq))  (assumes Σseq = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    """Monotonically decreasing exponential, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    # Beta family (coarse grid)
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    # Tukey windows
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise all (safety)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with KL divergence (exponential‑gradient).
    Returns the best vector seen and its surrogate value.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size
        eta = eta0 / math.sqrt(it)

        # multiplicative update; clip exponent to avoid overflow/underflow
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # projection onto the simplex
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val

# Simulated‑annealing polish
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # geometric cooling
        T = max(T * 0.99995, min_temp)

        # two types of move
        if rng.random() < 0.6:
            # reshuffle a random subset (size 3‑12)
            k = rng.integers(3, 13)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # small transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)  # at most 7 % of source mass
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

    # ------------------------------------------------------------------
    try:

        true_val = None

    return best_seq.tolist()
