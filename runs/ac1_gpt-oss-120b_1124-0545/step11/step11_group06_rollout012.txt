"""
Improved search for a low‑value step function.

The algorithm expands the analytic families explored in the previous
implementation (adds monotone decreasing families, exponential families
with r≈1, polynomial decays, etc.)  It keeps the best few candidates,
refines them with Adam on a smooth‐max surrogate,
followed by projected sub‑gradient descent and a short simulated‑annealing
phase.  Finally it scans all cyclic rotations of the best vector – a
cheap operation that often reduces the max‑convolution because the
sequence is not symmetric.  All work respects the 1000 s budget and
stores the result in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a *probability* vector.
    Returns 2·n·max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all normalized to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decay(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)  (strictly decreasing, no zeros)."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _poly_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p  (p ≥ 0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-α·i/n).  α controls the decay speed."""
    if alpha <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i = r**i  (0<r<1)."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p  (p ≥ 0)."""
    if p < 0.0:
        return _uniform(n)
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 800_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Works in logits space of a soft‑max (simplex constraint is automatic).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # start with moderate smoothing, temperature grows slowly
    beta_temp = 5.0
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # increase temperature (approach true max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)                  # linear convolution (length 2n‑1)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # softmax over conv entries

        # grad of Σ w_k·conv_k w.r.t a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # transform to logits space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Clip norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a simple diminishing step size based on the current best value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        # Polyak step size (uses current best)
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 200_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Very lightweight simulated‑annealing: randomly re‑allocate the total
    mass of a small random subset of coordinates (Dirichlet move).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)            # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30_000 == 0:
                    print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Candidate generation (deterministic sweep + random seeds)
# ----------------------------------------------------------------------


def _collect_candidates(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None,
                       top_k: int,
                       deadline: float) -> list[np.ndarray]:
    """
    Systematically explore analytic families (including new monotone
    decreasing ones) and retain the `top_k` candidates with smallest
    exact objective.
    """
    best = []  # (value, seq)

    def keep(val, seq):
        if len(best) < top_k:
            best.append((val, seq))
            best.sort(key=lambda x: x[0])
        elif val < best[-1][0]:
            best[-1] = (val, seq)
            best.sort(key=lambda x: x[0])

    # ---- deterministic families -------------------------------------------------
    families = [
        _uniform,
        _linear_decay,
    ]
    for fn in families:
        seq = fn(n)
        keep(_objective(seq), seq)

    # ---- polynomial decay ( (n‑i)^p ) -----------------------------------------
    for p in np.linspace(0.5, 5.0, 25):
        if time.time() > deadline:
            break
        seq = _poly_decay(n, p)
        keep(_objective(seq), seq)

    # ---- exponential decay (exp(-α i/n)) ---------------------------------------
    for alpha in np.linspace(0.2, 15.0, 60):
        if time.time() > deadline:
            break
        seq = _exponential_decay(n, alpha)
        keep(_objective(seq), seq)

    # ---- geometric decay (r close to 1) ---------------------------------------
    # use r = exp(-α/n) so that the effective decay parameter is α
    for alpha in np.linspace(0.2, 15.0, 60):
        if time.time() > deadline:
            break
        r = np.exp(-alpha / n)
        seq = _geom(n, r)
        keep(_objective(seq), seq)

    # ---- power‑law (1/(i+1)^p) ------------------------------------------------
    for p in np.linspace(0.0, 5.0, 30):
        if time.time() > deadline:
            break
        seq = _power(n, p)
        keep(_objective(seq), seq)

    # ---- random Dirichlet seeds ------------------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 30 if conc < 0.01 else 15
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            keep(_objective(seq), seq)

    # ---- sparse random supports -------------------------------------------------
    for _ in range(120):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)           # 3 … 30 non‑zero entries
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = weights
        keep(_objective(seq), seq)

    # ---- previous solution -------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        seq = previous / previous.sum()
        keep(_objective(seq), seq)

    # Return only the sequences (sorted by ascending objective)
    return [seq for _, seq in best]

    for i, seq in enumerate(candidates):
        print(f"  #{i+1}: {_objective(seq):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – short Adam refinements of the top candidates.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Phase 1: short Adam refinements ---")
    for idx, seed_vec in enumerate(candidates):
        if time.time() > deadline:
            break
        print(f"[Candidate {idx+1}/{len(candidates)}]")

        if seq_adam is not None and val_adam < best_val:

            best_val = val_adam
            print(f"  New global best after short Adam: {best_val:.6f}")

    # Fallback – should never happen

        best_seq = candidates[0].copy()
        best_val = _objective(best_seq)

    # Phase 2 – longer Adam optimisation on the current best.

        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after long Adam: {best_val:.6f}")

            best_val = val_sub

    return best_seq.tolist()
