def _lp_refine(a, n, deadline, max_iters=20, top_k=20, t=0.1):
    a = a.copy()
    sum_target = a.sum()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(max_iters):
        if time.time()>deadline:
            break
        conv = _conv_fft(a, a)
        M = conv.max()
        # get indices of top values (including ties)
        # We'll pick top_k unique largest
        top_inds = np.argpartition(conv, -top_k)[-top_k:]  # unsorted
        # build matrix A (top_inds rows)
        rows = []
        for k in top_inds:
            # compute row: for i from max(0, k-(n-1)) to min(n-1,k):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # create a zero row
            row = np.zeros(n, dtype=np.float64)
            # row[i] = a[k-i]
            # Equivalent: row[i_start:i_end+1] = a[i_start:i_end+1 not? Wait.
            # For each i in [i_start,i_end], conv[k] includes a[i] * a[k-i].
            # row[i] is coefficient for g_i in conv(f,g)[k] = sum_i f_i * g_{k-i}
            # So row[i] = a_{k-i}.
            # So indices for g_i correspond to i, and coefficient is a_{k-i}.
            # That's a reversed order within the range.
            # We'll compute vector slice:
            slice_g_i = np.arange(i_start, i_end+1)
            slice_f = a[k - slice_g_i]
            row[slice_g_i] = slice_f
            rows.append(row)
        A = np.stack(rows) # shape (top_k, n)
        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        # use high precision
        res = linprog(c, A_ub=A, b_ub=np.full(top_k, M), bounds=bounds, method='highs')
        if not res.success:
            # skip LP and break
            break
        g = res.x
        # scale g to have same sum as a (target)
        s_g = g.sum()
        if s_g > 0:
            g = g * (sum_target / s_g)
        else:
            # fallback: keep current a
            break
        # convex combination update
        a = (1 - t) * a + t * g
        # re-normalize to sum target
        a = a * (sum_target / a.sum())
        # evaluate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # optionally reduce t
            # t = max(0.01, t * 0.9)
    return best_seq, best_val

"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm builds a very large and diverse pool of candidate shapes,
ranks them with a cheap surrogate, and then refines the most promising
ones with several optimisation stages:

1.   Analytic windows (Hamming, Kaiser, DPSS, …) and a massive set of
     randomised families (Dirichlet draws, Gaussian mixtures, low‑frequency
     cosines, etc.).
2.   Log‑space Adam on the log‑softmax parametrisation (smooth max via
     log‑sum‑exp) – fast first‑order descent.
3.   Exact sub‑gradient projected descent on the true max‑convolution.
4.   Mirror‑descent (exponentiated‑gradient) on the simplex.
5.   A custom linear‑programming refinement: maximise the ℓ₁‑mass of a
     vector g under the active convolution constraints of the current
     iterate (solved with `scipy.optimize.linprog`).  The solution is
     mixed with the current iterate.
6.   Stochastic‑annealing polishing.
7.   Final evaluation with the official `evaluate_sequence`.

The whole search respects a global deadline (default 1000 s).  The best
sequence found is stored in the global name ``height_sequence_1`` and
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for LP and some windows)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None

try:
    from scipy.signal import windows as _scipy_windows
except Exception:  # pragma: no cover
    _scipy_windows = None

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """smallest power of two ≥ x"""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """full linear convolution using a real FFT (O(n log n))"""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """numerically stable softmax – returns a vector on the probability simplex"""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for a vector that is
    ℓ₁‑normalised (∑seq = 1).  The returned value is exactly the quantity
    returned by ``evaluate_sequence`` when the sum equals 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _scipy_windows is not None and hasattr(_scipy_windows, 'kaiser'):
        a = _scipy_windows.kaiser(n, beta)
    else:
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        t = beta * np.sqrt(1 - y * y)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window; falls back to uniform."""
    if _scipy_windows is not None and hasattr(_scipy_windows, 'dpss'):
        a = _scipy_windows.dpss(n, nw, Kmax=1, return_ratios=False)[0]
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    return _uniform(n)


def _chebwin_window(n: int, att: float) -> np.ndarray:
    """Chebyshev window with side‑lobe attenuation (dB)."""
    if _scipy_windows is not None and hasattr(_scipy_windows, 'chebwin'):
        a = _scipy_windows.chebwin(n, at=att)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    return _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15, shift_frac: float = 0.25) -> np.ndarray:
    """Mixture of two symmetric Gaussians placed at shift and 1‑shift."""
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `conc` (scalar)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few random low‑frequency cosine components, clip & renormalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_gaussian_mix(n: int, rng: np.random.Generator,
                         components: int = 3) -> np.ndarray:
    """Random mixture of `components` Gaussians with random centres and widths."""
    weights = rng.dirichlet(np.ones(components))
    mus = rng.uniform(0, n - 1, size=components)
    sigmas = rng.uniform(0.04 * n, 0.30 * n, size=components)
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for w, mu, sigma in zip(weights, mus, sigmas):
        a += w * np.exp(-((t - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_all_seeds(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a very large pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # ---------- deterministic analytic families ----------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for nw in (2.0, 2.5, 3.0, 3.5, 4.0, 4.5, 5.0):
        seeds.append(_dpss_window(n, nw))
    for att in (30, 40, 60, 80):
        seeds.append(_chebwin_window(n, att))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # ---------- random families ----------
    # Dirichlet draws with many concentration levels
    conc_vals = np.logspace(-2, 1, 15)  # 0.01 … 10
    for conc in conc_vals:
        for _ in range(40):  # 600 draws total
            seeds.append(_random_dirichlet(n, conc, rng))

    # Heavy‑tailed exponentials
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(20):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random mixtures of Gaussian components
    for _ in range(300):
        seeds.append(_random_gaussian_mix(n, rng, components=3))

    # Mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Include the previous solution if available
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise everything robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

def _logspace_adam_optimize(start_vec: np.ndarray, n: int,
                            deadline: float,
                            max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑softmax parametrisation.
    Returns (best_vector, best_surrogate_value).  The vector is ℓ₁‑normalised.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # initial temperature for soft‑max smoothing
    beta_inc = 0.12
    beta_max = 200.0

    lr0 = 1.0                # learning‑rate in log‑space

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[LogAdam] it={it:6d} best={cur_val:.6f}")

        # --- smooth max via log‑sum‑exp -------------------------------------------------
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w·conv wrt a (convolution with reversed w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient through the softmax (log‑softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam step (learning‑rate decays as 1/√t)
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int,
                      deadline: float,
                      max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Euclidean projection onto the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / np.sqrt(it)          # diminishing step‑size
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(start_seq: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) directly on the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (same as in _subgradient_proj)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)          # decreasing step
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)              # back to simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 1_000_000,
                         init_temp_factor: float = 0.06,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:          # reshuffle a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # small pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)  # move ≤5 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _lp_refine(start_seq: np.ndarray, n: int,
               deadline: float,
               max_iters: int = 20,
               top_k: int = 25,
               step_size: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    At each iteration we maximise the ℓ₁‑mass of a vector g under the *active*
    convolution constraints of the current iterate (i.e. those indices where
    conv(f,f) is close to its maximum).  The solution is convex‑combined with
    the current iterate.
    Returns the best sequence seen and its surrogate value.
    """
    if linprog is None:
        # SciPy unavailable – skip this stage.
        return start_seq.copy(), _objective_surrogate(start_seq)

    a = start_seq.copy()
    target_sum = a.sum()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()

        # pick the top‑k indices (most binding constraints)
        idxs = np.argpartition(conv, -top_k)[-top_k:]
        # build the constraint matrix A (shape top_k × n)
        A = np.zeros((len(idxs), n), dtype=np.float64)
        for row, k in enumerate(idxs):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            # coefficient for g[i] is f_{k-i}
            A[row, i_range] = a[k - i_range]

        # Linear programme: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=bounds, method='highs')
        if not res.success:
            # If the LP fails, abort the refinement.
            break
        g = res.x
        s = g.sum()
        if s > 0:
            g = g * (target_sum / s)          # re‑scale to the original ℓ₁‑mass
        else:
            break

        # convex update
        a = (1.0 - step_size) * a + step_size * g
        # renormalise exactly
        a = a * (target_sum / a.sum())

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LPRefine] it={it:2d} best={cur_val:.6f}")

        # optional step‑size decay (optional, here we keep it constant)

    return best_seq, best_val


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalisation."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin to ensure we can call evaluate_sequence at the end
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Stage 0 – choose resolution
    # ------------------------------------------------------------------
    n = 4000                        # baseline length (good trade‑off)

    # ------------------------------------------------------------------
    # Stage 1 – massive seed generation and cheap surrogate ranking
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    raw_seeds = _collect_all_seeds(n, rng, prev_arr)
    print(f"Generated {len(raw_seeds)} candidate seeds (including random draws).")

    # Evaluate surrogate cheaply and keep the top performers
    top_k = 30
    evaluations = []
    for vec in raw_seeds:
        val = _objective_surrogate(vec)
        evaluations.append((vec, val))
    evaluations.sort(key=lambda x: x[1])
    best_seeds = [vec for vec, _ in evaluations[:top_k]]

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Log‑space Adam optimisation on top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 0.08 * 1e5))
        print(f"\n--- Log‑space Adam run {idx+1}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _logspace_adam_optimize(seed_vec, n, deadline,
                                                     max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Exact sub‑gradient projection (more accurate max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_sub = int(min(1_800_000, remaining * 0.45 * 1e5))
        print("\n--- Sub‑gradient projection refinement ---")
        seq_sub, val_sub = _subgradient_proj(global_best_seq, n, deadline,
                                             max_iters=max_iters_sub)
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_exp = int(min(1_200_000, remaining * 0.30 * 1e5))
        print("\n--- Exponentiated‑gradient refinement ---")
        seq_exp, val_exp = _exp_gradient_optimize(global_best_seq, n,
                                                  deadline, max_iters=max_iters_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[overall‑ExpGrad] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Linear‑programming refinement (new idea)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_lp = min(20, int(remaining // 2))  # each LP is cheap but we keep a ceiling
        print("\n--- Linear‑programming refinement ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, n, deadline,
                                    max_iters=max_iters_lp, top_k=30, step_size=0.13)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_sa = int(min(800_000, remaining * 1e5))
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq, n,
                                              deadline, max_iters=max_iters_sa)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 7 – optional up‑sampling for final fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} for final refinement ---")
            up_seq = _upsample_sequence(global_best_seq, new_n)

            # brief Adam on the higher resolution
            max_iters_up = int(min(150_000, time_left * 0.30 * 1e5))
            seq_up, val_up = _logspace_adam_optimize(up_seq, new_n,
                                                     deadline, max_iters=max_iters_up)
            if seq_up is not None and val_up < global_best_val:
                global_best_seq = seq_up.copy()
                global_best_val = val_up
                print(f"[Upsample‑Adam] new global best = {val_up:.6f}")

            # quick exponentiated‑gradient pass
            remaining = deadline - time.time()
            if remaining > 5.0:
                max_iters_up_exp = int(min(300_000, remaining * 0.25 * 1e5))
                seq_up_exp, val_up_exp = _exp_gradient_optimize(
                    global_best_seq, new_n, deadline, max_iters=max_iters_up_exp)
                if val_up_exp < global_best_val:
                    global_best_seq = seq_up_exp.copy()
                    global_best_val = val_up_exp
                    print(f"[Upsample‑ExpGrad] new global best = {val_up_exp:.6f}")

                # final tiny SA polish
                remaining = deadline - time.time()
                if remaining > 5.0:
                    seq_up_sa, val_up_sa = _simulated_annealing(
                        global_best_seq, new_n, deadline,
                        max_iters=int(min(200_000, remaining * 1e5)))
                    if val_up_sa < global_best_val:
                        global_best_seq = seq_up_sa.copy()
                        global_best_val = val_up_sa
                        print(f"[Upsample‑SA] final global best = {val_up_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (if available)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
