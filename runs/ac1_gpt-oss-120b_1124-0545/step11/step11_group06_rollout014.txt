"""
Propose a step‑function (non‑negative list) that minimizes
``evaluate_sequence``.  The strategy improves on the previous
implementation by

1. using a richer family of analytic shapes (beta, tukey, power,
   exponential, piece‑wise constant, etc.) and a much finer random
   sweep,
2. allocating the 1000 s budget into several stages (candidate sweep,
   many short Adam runs, a long Adam refinement, projected sub‑gradient
   descent and a final simulated‑annealing polish),
3. keeping the currently best sequence and constantly checking the
   remaining time so that the function always returns a valid answer
   before the timeout.

The optimisation works on the simplex (ℓ₁‑normalised vectors) – the
objective is scale‑invariant, therefore we normalise to sum = 1 at each
step.  The optimiser uses a smooth log‑sum‑exp surrogate for the max‑
convolution (Adam) and a true sub‑gradient descent for the exact
max‑convolution, followed by a lightweight simulated‑annealing phase.
All steps are pure NumPy, no external solvers are required.
"""

import time

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT, exact objective)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(N log N))."""
    L = x.shape[0] + y.shape[0] - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the official objective for a *probability*
    vector (∑seq = 1).  Returns 2·n·max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    """Trapezoid with a flat top of given width fraction."""
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑like shape, asymmetrical for α≠β."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """Sine window raised to a power γ."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Standard Tukey window."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Pure geometric decay (strictly decreasing)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Exponential decay e^{-λ·i}."""
    if lam <= 0.0:
        return _uniform(n)
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p (p may be negative)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet_seq(n: int, rng: np.random.Generator,
                         conc: float) -> np.ndarray:
    """Draw a Dirichlet vector with the given concentration."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_constant_seq(n: int, rng: np.random.Generator,
                            min_blocks: int = 2, max_blocks: int = 10) -> np.ndarray:
    """Random contiguous blocks with different heights."""
    nb = rng.integers(min_blocks, max_blocks + 1)
    # pick nb‑1 cut points between 1 and n‑1
    cuts = np.sort(rng.choice(n - 1, size=nb - 1, replace=False)) + 1
    lengths = np.diff(np.concatenate(([0], cuts, [n])))
    heights = rng.dirichlet(np.ones(nb)).astype(np.float64)
    seq = np.repeat(heights, lengths)
    seq = seq.astype(np.float64)
    seq /= seq.sum()
    return seq

# Optimisers (Adam on log‑sum‑exp surrogate, projected sub‑gradient,
# simulated annealing)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max‑convolution.
    Operates in the log‑softmax (logits) space to stay on the simplex.
    Returns the best sequence encountered and its true objective.
    """
    eps = 1e-30
    a0 = np.clip(start_vec, eps, None)                # avoid log(0)
    v = np.log(a0)                                     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # smooth‑max temperature (starts low → smooth)
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = np.exp(v - np.max(v))                       # softmax, stays on simplex
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # raise temperature (approach true max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)                           # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                                       # soft‑max over conv entries

        # gradient of Σ w_k·conv_k wrt a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                      # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]            # central n entries

        # transform gradient to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.8 / np.sqrt(it)                          # decaying stepsize

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a simple Polyak step‑size based on the current best value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient at the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Light‑weight simulated annealing based on random Dirichlet moves
    on a small random subset of coordinates.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)                         # size of the moved subset
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30_000 == 0:
                    print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

def _maybe_keep(best: list, val: float, seq: np.ndarray, top_k: int):
    """Insert (val,seq) into the sorted list of best candidates."""
    if len(best) < top_k:
        best.append((val, seq))
        best.sort(key=lambda x: x[0])
    elif val < best[-1][0]:
        best[-1] = (val, seq)
        best.sort(key=lambda x: x[0])


def _collect_candidates(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None,
                       top_k: int, deadline: float) -> list[np.ndarray]:
    """
    Systematically explore many analytic families and a large random
    sample, retaining the `top_k` candidates with smallest exact
    objective values.
    """
    best = []      # list of (value, seq)

    # ---- deterministic families -------------------------------------------------
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.12),
        lambda n: _trapezoidal(n, width_fraction=0.25),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _sine_window(n, gamma=3.0),
        lambda n: _tukey_window(n, alpha=0.3),
        lambda n: _tukey_window(n, alpha=0.5),
        lambda n: _tukey_window(n, alpha=0.7),
    ]
    for fn in families:
        seq = fn(n)
        _maybe_keep(best, _objective(seq), seq, top_k)

    # ---- random beta shapes ----------------------------------------------------
    for _ in range(400):
        if time.time() > deadline:
            break
        a = 10.0 ** rng.uniform(-2, 1)          # α in [0.01,10]
        b = 10.0 ** rng.uniform(-2, 1)          # β in [0.01,10]
        seq = _beta_shape(n, a, b)
        _maybe_keep(best, _objective(seq), seq, top_k)

    # ---- random power‑law ------------------------------------------------------
    for _ in range(300):

            break
        p = rng.uniform(-5.0, 5.0)
        seq = _power(n, p)
        _maybe_keep(best, _objective(seq), seq, top_k)

    # ---- random geometric / exponential ----------------------------------------
    for _ in range(300):
        if time.time() > deadline:

        r = rng.uniform(0.7, 0.9999)            # geometric ratio
        seq = _geom(n, r)
        _maybe_keep(best, _objective(seq), seq, top_k)
        lam = rng.uniform(0.001, 0.1)           # exponential decay rate
        seq2 = _exp_decay(n, lam)
        _maybe_keep(best, _objective(seq2), seq2, top_k)

    # ---- random tukey windows --------------------------------------------------
    for _ in range(200):
        if time.time() > deadline:

        a = rng.uniform(0.1, 0.9)
        seq = _tukey_window(n, a)
        _maybe_keep(best, _objective(seq), seq, top_k)

    # ---- random Dirichlet seeds ------------------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        draws = 60 if conc < 0.01 else 40
        for _ in range(draws):

            seq = _random_dirichlet_seq(n, rng, conc)

        idx = rng.choice(n, size=k, replace=False)

        seq = np.zeros(n, dtype=np.float64)

        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, phase4_deadline,
            max_iters=400_000, init_temp_factor=0.05
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated global best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
