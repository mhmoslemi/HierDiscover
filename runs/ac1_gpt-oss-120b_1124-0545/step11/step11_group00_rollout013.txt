"""
Propose a step‑function that (hopefully) improves on the previous best
objective (≈ 1.5087).  The method follows three stages:

1. **Rich Seed Generation** – In addition to the families used before we
   add “reverse‑power” shapes (a_i ∝ (n–i)ᵖ with p possibly negative)
   and a finer grid for the beta‑distribution.  All seeds are ℓ₁‑normalised
   (Σ a = 1) and are ranked by the cheap surrogate
   `2·n·max(conv(a,a))`.  The top few seeds are kept.

2. **Exact Mirror‑Descent Sub‑gradient** – We run a long exponentiated‑gradient
   (mirror) optimisation on the *true* objective  
   `F(a) = 2·n·max_k Σ_i a_i a_{i+k}`.
   The sub‑gradient at the active lag `k*` is
   `g_i = 2·a_{k*‑i}` (zero outside the overlap).  An
   adaptive step‑size `η_t = η₀·t⁻ᵖ` (`η₀=0.25, p=0.33`) is used.
   The iterate is multiplied by `exp(‑η·g)` and projected back onto the
   simplex (normalise).  The routine is very cheap (one FFT per iteration)
   and we devote the bulk of the time budget to it.

3. **Polishing by Simulated‑Annealing** – A lightweight SA that repeatedly
   performs three simple mass‑transfer moves (swap, Dirichlet‑redistribute,
   pair‑transfer) and accepts them with a slowly cooling schedule.
   This can escape shallow local minima left by the gradient phase.

All stages respect the global deadline (default 1000 s).  The best sequence
found is stored in the global name `height_sequence_1` and returned as a

# FFT utilities (O(n log n))

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv


def _surrogate_obj(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(a,a)) for an ℓ₁‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (p≥0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _reverse_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^{p}   (p can be negative or positive)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (n - idx + 1) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (
                1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1))
            )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑distributed probability vector."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Mirror‑descent sub‑gradient optimiser (exact max)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.25          # initial step size factor
    power = 0.33        # slower decay than 0.5
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:7d}  best surrogate={cur_val:.6f}")

        # sub‑gradient at the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative (exponentiated) update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # project back onto simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 init_temp_factor: float = 0.05,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Lightweight SA with three move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-14)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            # simple swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            # Dirichlet redistribution on a random subset
            k = rng.integers(2, 15)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # pairwise mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:7d}  best surrogate={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed collection – richer families
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))

    # fine sine‑window grid
    for gamma in np.linspace(0.3, 5.0, 15):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian grid
    for sigma in np.linspace(0.06, 0.45, 12):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # power‑law (decreasing) and reverse‑power (increasing)
    for p in np.linspace(-3.0, 3.0, 13):
        if p >= 0:
            seeds.append(_power(n, p))
        else:
            seeds.append(_reverse_power(n, -p))

    # beta grid (more dense)
    alphas = np.linspace(0.2, 4.0, 9)
    betas = np.linspace(0.2, 4.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.05, 0.95, 11):
        seeds.append(_tukey_window(n, alpha))

    # exponential decay
    for lam in np.linspace(0.0003, 0.02, 10):
        seeds.append(_exponential_decay(n, lam))

    # random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            dr = _random_dirichlet(n, conc, rng)
            seeds.append(dr)                       # raw
            seeds.append(np.sort(dr)[::-1])        # descending
            seeds.append(np.sort(dr))              # ascending

    # low‑frequency random curves
    for nf in (2, 3, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.1, 0.3, 0.6):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    # ensure all seeds are proper probability vectors
    cleaned = []
    for v in seeds:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 31337, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats and stores it in the
    global variable ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final check

    # ------------------------------------------------------------------
    # Length – a bit larger than the reference (4000) to give more
    # degrees of freedom while staying comfortably fast.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Retrieve the previous solution (if any) to enrich the seed pool
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a large pool of analytic & random seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n = {n}) ---")
    seed_list = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seed_list)} raw seeds.")

    # Quick surrogate ranking
    evals = [(vec, _surrogate_obj(vec)) for vec in seed_list]
    evals.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _ in evals[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evals[:top_k], 1):
        print(f"  #{i:2d}: {val:.6f}")

    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Mirror‑descent sub‑gradient optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent optimisation (exact max) ---")
        # Run mirror descent on the best seed; if time permits we also
        # give a second shot to the second‑best seed.
        seq_m1, val_m1 = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=5_000_000
        )
        if val_m1 < global_best_val:
            global_best_seq = seq_m1
            global_best_val = val_m1
            print(f"[overall‑Mirror] new best surrogate = {val_m1:.6f}")

        # optional second seed (only if we still have a considerable budget)
        remaining = deadline - time.time()
        if remaining > 30.0:
            seq_m2, val_m2 = _mirror_subgradient_optimize(
                top_seeds[1], n, deadline, max_iters=3_000_000
            )
            if val_m2 < global_best_val:
                global_best_seq = seq_m2
                global_best_val = val_m2
                print(f"[overall‑Mirror‑2] new best surrogate = {val_m2:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=2_000_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa
            global_best_val = val_sa
            print(f"[overall‑SA] final best surrogate = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (optional, just for information)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose result for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()

**Explanation of why this may beat the previous result**

* The seed pool is markedly richer (reverse‑power shapes, a finer β‑grid,
  many more Dirichlet draws, and additional low‑frequency curves), giving the
  optimiser more promising starting points.

* The mirror‑descent phase works directly on the *exact* max‑convolution
  rather than on a smoothed surrogate, and it runs for several million
  iterations with a very gentle step‑size decay (`η₀=0.25, p=0.33`).  This
  typically pushes the objective below the earlier 1.5087 plateau.

* The final simulated‑annealing pass uses a slower cooling schedule and
  three complementary move types, allowing the algorithm to escape the
  shallow local minima that often trap the earlier optimizer.

* The length is increased to 5000 (still cheap for FFT‑based convolutions) –
  the extra degrees of freedom give the optimiser more room to reduce the
  peak of the auto‑correlation, which is exactly what the objective
  measures.

All stages respect the 1000 s budget, and the best sequence found is
returned as a simple list of floats ready for `evaluate_sequence`.
