"""
search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

The optimizer works in several stages:

1. **Candidate sweep** – analytic families (uniform, triangular, beta,
   sine, tukey, power, geometric, exponential) and many random Dirichlet
   seeds are generated and the best few are kept.

2. **Long Adam on a log‑sum‑exp surrogate** – the surrogate approximates the
   max‑convolution; the temperature is slowly increased up to β≈600.
   Adam works in the soft‑max (log‑probability) space, guaranteeing the
   ℓ₁‑simplex constraint automatically.

3. **Projected sub‑gradient descent** – after the smooth phase we
   optimise the true max‑convolution directly with a Polyak‑type step size.

4. **Heavier simulated‑annealing** – random Dirichlet moves are applied
   with a slowly decreasing temperature.

5. **Greedy mass‑reallocation** – a fast hill‑climbing that moves a tiny
   amount of probability mass from the most‑responsible coordinates to the
   least‑responsible ones.

All phases respect the global time budget (default 1000 s).  The routine
stores the best sequence found in ``height_sequence_1`` and returns a plain
Python list that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np
import cvxpy as cp  # optional, not used in the final version.

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a *probability* vector.
    Returns 2·n·max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    """Monotonically decreasing exponential, normalised."""
    if rate <= 0.0:
        return _uniform(n)
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 1_200_000,
                   beta_max: float = 600.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Operates in the logits space of a soft‑max, i.e. the simplex.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # initial smooth‑max temperature
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # raise temperature slowly (approach true max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)                  # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # softmax over conv entries

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip norm for stability
        max_norm = 8.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a simple Polyak step size based on the best value seen so far.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing: re‑allocate the total mass of a random
    small subset of coordinates (Dirichlet move).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30_000 == 0:
                    print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


def _greedy_reallocation(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 150_000,
                         step_frac: float = 0.002) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: repeatedly move a tiny amount of mass
    from the coordinate that contributes most to the current max‑conv
    to the coordinate that contributes least.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_max = int(np.argmax(conv))

        # indices that appear in the active shift
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        active = np.arange(i_start, i_end + 1)

        # pick source (largest a among active) and target (smallest a overall)
        src = active[np.argmax(a[active])]
        tgt = np.argmin(a)  # global minimum (could be zero)

        if src == tgt:
            # random fallback
            src, tgt = rng.choice(active, size=2, replace=False)

        delta = step_frac * a[src]  # move a small fraction of src mass
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        # re‑project onto simplex to keep exact sum‑1 (numerical drift)
        a_new = _proj_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < best_val:
            best_val = new_val
            best_seq = a_new.copy()
            a = a_new
            if it % 20_000 == 0:
                print(f"[Greedy] it={it:,} best={best_val:.6f}")
        else:
            # occasionally accept a worsening move (tiny probability) to escape plateaus
            if rng.random() < 0.001:
                a = a_new

    return best_seq, best_val


# ----------------------------------------------------------------------
# Candidate generation (deterministic sweep + random seeds)
# ----------------------------------------------------------------------


def _collect_candidates(n: int,
                       rng: np.random.Generator,
                       previous: np.ndarray | None,
                       top_k: int,
                       deadline: float) -> list[np.ndarray]:
    """
    Systematically explore analytic families and retain the `top_k`
    candidates with smallest exact objective.
    """
    best = []  # list of (value, seq)

    def keep(val, seq):
        if len(best) < top_k:
            best.append((val, seq))
            best.sort(key=lambda x: x[0])
        elif val < best[-1][0]:
            best[-1] = (val, seq)
            best.sort(key=lambda x: x[0])

    # ---- deterministic families -------------------------------------------------
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.12),
        lambda n: _trapezoidal(n, width_fraction=0.25),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _sine_window(n, gamma=3.0),
        lambda n: _tukey_window(n, alpha=0.5),
    ]
    for fn in families:
        seq = fn(n)
        keep(_objective(seq), seq)

    # ---- beta shape grid -------------------------------------------------------
    alphas = np.linspace(0.1, 3.0, 30)
    betas = np.linspace(0.1, 3.0, 30)
    for a in alphas:
        for b in betas:
            seq = _beta_shape(n, a, b)
            keep(_objective(seq), seq)

    # ---- sine window grid -------------------------------------------------------
    gammas = np.linspace(0.5, 5.0, 30)
    for g in gammas:
        seq = _sine_window(n, gamma=g)
        keep(_objective(seq), seq)

    # ---- tukey window grid ------------------------------------------------------
    alps = np.linspace(0.1, 0.9, 25)
    for a in alps:
        seq = _tukey_window(n, alpha=a)
        keep(_objective(seq), seq)

    # ---- power‑law ---------------------------------------------------------------
    ps = np.linspace(0.0, 5.0, 30)
    for p in ps:
        seq = _power(n, p)
        keep(_objective(seq), seq)

    # ---- geometric ---------------------------------------------------------------
    rs = np.linspace(0.2, 0.98, 25)
    for r in rs:
        seq = _geom(n, r)
        keep(_objective(seq), seq)
        seq2 = _sym_geom(n, r)
        keep(_objective(seq2), seq2)

    # ---- exponential -------------------------------------------------------------
    rates = np.linspace(0.0005, 0.02, 20)
    for rate in rates:
        seq = _exponential(n, rate)
        keep(_objective(seq), seq)

    # ---- random Dirichlet seeds --------------------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 60 if conc < 0.01 else 30
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            keep(_objective(seq), seq)

    # ---- sparse random supports -------------------------------------------------
    for _ in range(200):
        if time.time() > deadline:
            break
        k = rng.integers(3, 31)           # 3 … 30 non‑zero entries
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(np.ones(k)).astype(np.float64)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = weights
        keep(_objective(seq), seq)

    # ---- previous solution -------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        seq = previous / previous.sum()
        keep(_objective(seq), seq)

    # Return only the sequences, sorted by ascending objective
    return [seq for _, seq in best]

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimizes
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length of the step function.
    # 4000 gives a good compromise between expressiveness and speed.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}, budget={budget_s}s) ===")

    # ------------------------------------------------------------------
    # Phase 0 – systematic candidate generation.
    # ------------------------------------------------------------------
    phase0_deadline = min(deadline, start_time + 0.18 * budget_s)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    print("\n--- Phase 0: deterministic and random seed sweep ---")
    candidates = _collect_candidates(
        n, rng, previous, top_k=15, deadline=phase0_deadline
    )
    print(f"Collected {len(candidates)} top candidates (objective values):")
    for i, seq in enumerate(candidates):
        print(f"  #{i+1}: {_objective(seq):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – short Adam refinements (more iterations than before).
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Phase 1: short Adam refinements ---")
    for idx, seed_vec in enumerate(candidates):
        if time.time() > deadline:
            break
        print(f"[Candidate {idx+1}/{len(candidates)}]")
        seq_adam, val_adam = _adam_optimize(
            seed_vec,
            n,
            deadline,
            max_iters=300_000,     # double the previous short run
            beta_max=600.0,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after short Adam: {best_val:.6f}")

    # Fallback if nothing improved (unlikely)
    if best_seq is None:
        best_seq = candidates[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first candidate: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – longer Adam optimisation.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: long Adam optimisation ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            deadline,
            max_iters=3_000_000,
            beta_max=600.0,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after long Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: projected sub‑gradient descent ---")
        seq_sub, val_sub = _subgrad_optimize(
            best_seq,
            n,
            deadline,
            max_iters=1_600_000,
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated global best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            deadline,
            max_iters=500_000,
            init_temp_factor=0.05,
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated global best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – cheap greedy mass reallocation.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 5: greedy mass reallocation ---")
        seq_gr, val_gr = _greedy_reallocation(
            best_seq,
            n,
            deadline,
            max_iters=300_000,
            step_frac=0.0015,
        )
        if val_gr < best_val:
            best_seq = seq_gr.copy()
            best_val = val_gr
            print(f"  Updated global best after greedy phase: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This implementation expands the search space, runs substantially longer Adam and sub‑gradient phases, raises the smooth‑max temperature to β ≈ 600 for a tighter surrogate, adds an explicit exponential family, ups the number of random seeds, and finishes with a lightweight greedy reallocation step. All phases observe the overall 1000 s time budget, and the best sequence found is stored in the global variable ``height_sequence_1`` for possible reuse.
