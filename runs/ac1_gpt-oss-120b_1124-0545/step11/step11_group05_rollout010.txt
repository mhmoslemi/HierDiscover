"""
Propose a step‑function (non‑negative list) that lowers the bound
2·n·max(conv) and therefore the value returned by evaluate_sequence().
The strategy expands on the previous approach:

1.  Sample a dense set of analytic families (Beta, sine, Tukey, power,
    geometric, etc.) on a modest size (n≈2000) and keep the best few.
2.  Upscale those candidates to a larger length n≈6000 (more degrees of
    freedom → flatter convolution is possible).
3.  Refine the best up‑scaled candidate with a continuation of
    L‑BFGS optimisation on a smooth log‑sum‑exp approximation of the
    max‑convolution.  The temperature β is increased up to 6000 so that
    the surrogate becomes very close to the true max.
4.  Perform a long projected sub‑gradient descent on the exact max‑convolution
    (the true objective) to polish the solution.
5.  Finish with a fast simulated‑annealing sweep that redistributes
    small random mass blocks.
All stages respect the overall 1000 s budget.  The final sequence is
stored in the global ``height_sequence_1`` variable for reuse.
"""

import time
import numpy as np
import scipy.optimize

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate objective for a ℓ¹‑normalised sequence (∑seq == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (0<r<1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,

    """

    v2 = np.zeros_like(v)

    beta_temp = 10.0
    beta_mul = 1.006

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # temperature increase
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 500_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a Polyak‑style diminishing step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polishing that reshuffles a random subset
    of the mass (Dirichlet re‑allocation).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _lbfgs_optimize(start_vec: np.ndarray, n: int, deadline: float,
                    beta: float = 3000.0, maxfun: int = 200_000) -> tuple[np.ndarray, float]:
    """
    L‑BFGS optimisation on a log‑sum‑exp surrogate of max‑convolution.
    The variables are logits; the returned vector is a probability distribution.
    """
    eps = 1e-30
    v0 = np.log(start_vec + eps)  # initialise logits

    # estimate per function evaluation cost (≈2 FFTs)
    est_per_call = 0.0015

    # safety: bound the number of calls to respect the deadline
    remaining = deadline - time.time()
    max_allowed = max(1, int(remaining / est_per_call))
    maxfun = min(maxfun, max_allowed)

    def func_and_grad(v):
        a = _softmax(v)                     # probabilities (sum = 1)
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        # log‑sum‑exp approximation of the max
        approx_M = shift + (1.0 / beta) * np.log(w_sum) if w_sum > 0 else shift
        w = w / w_sum if w_sum > 0 else np.full_like(conv, 1.0 / conv.size)

        # gradient w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # objective (exact surrogate = 2·n·approx_M)
        f_val = 2.0 * n * approx_M
        # gradient of the surrogate w.r.t. v
        grad_v = grad_v * (2.0 * n)

        return f_val, grad_v

    # run L‑BFGS‑B; it accepts a function returning (value, grad)
    try:
        opt_v, f_opt, info = scipy.optimize.fmin_l_bfgs_b(
            func_and_grad, v0, maxfun=maxfun, disp=False)
    except Exception as exc:
        print("[L‑BFGS] optimisation aborted:", exc)
        opt_v = v0

    best_seq = _softmax(opt_v)
    # compute the true surrogate (exact max) for reporting
    best_val = float(_objective(best_seq))
    return best_seq, best_val

def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a small‑n sequence to a larger n."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    else:
        ys = _uniform(n_big)
    return ys.astype(np.float64)


def _generate_small_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a fairly dense pool of analytically‑parameterised candidates on a modest n."""
    seeds: list[np.ndarray] = []

    # deterministic bases
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Trapezoidal sweep
    for wf in np.linspace(0.02, 0.45, 12):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    # Sine windows – finer gamma sweep
    for gamma in np.linspace(0.0, 8.0, 41):
        seeds.append(_sine_window(n, gamma=gamma))

    # Tukey windows – finer α sweep
    for alpha in np.linspace(0.0, 0.95, 25):
        seeds.append(_tukey_window(n, alpha=alpha))

    # Geometric families – expanded search
    for r in np.concatenate([np.linspace(0.01, 0.98, 25),
                             np.linspace(0.98, 0.9999, 20)]):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family – broader p range
    for p in np.linspace(0.0, 10.0, 31):
        seeds.append(_power(n, p))

    # Beta shape – finer grid
    alphas = np.linspace(0.2, 5.0, 25)
    betas = np.linspace(0.2, 5.0, 25)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in np.linspace(0.05, 0.45, 9):
        seeds.append(_gaussian(n, sigma_frac))

    # Random Dirichlet draws – diverse concentration
    for conc in (0.001, 0.005, 0.02, 0.1, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponentials
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Previous solution (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape[0] > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                tot = prev_arr.sum()
                if tot > 0:
                    seeds.append(prev_arr / tot)
        except Exception:
            pass

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The routine respects the time budget and returns
    the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – modest‑scale analytic sweep
    # ------------------------------------------------------------------
    n_small = 2000
    print(f"\n--- Stage 0: analytic sweep (n={n_small}) ---")
    small_seeds = _generate_small_seeds(n_small, rng)
    print(f"Generated {len(small_seeds)} analytic seeds.")

    evaluated_small = [(vec, _objective(vec)) for vec in small_seeds]
    evaluated_small.sort(key=lambda x: x[1])

    top_k_small = min(30, len(evaluated_small))
    best_small = [vec for vec, _ in evaluated_small[:top_k_small]]
    print(f"Top {top_k_small} analytic seeds (surrogate values):")
    for i, (_, val) in enumerate(evaluated_small[:top_k_small]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – upscale to a larger problem size
    # ------------------------------------------------------------------
    n_large = 6000
    print(f"\n--- Stage 1: upscale to n={n_large} ---")
    large_seeds = [_upscale(vec, n_large) for vec in best_small]

    evaluated_large = [(vec, _objective(vec)) for vec in large_seeds]
    evaluated_large.sort(key=lambda x: x[1])
    global_best_seq = evaluated_large[0][0].copy()
    global_best_val = evaluated_large[0][1]
    print(f"[Init] best surrogate after upscale = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS refinement (smooth surrogate)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        lbfgs_budget = remaining * 0.45
        maxfun = int(lbfgs_budget / 0.0015)
        maxfun = max(10_000, min(maxfun, 300_000))
        print(f"\n--- Stage 2: L‑BFGS optimisation (β≈6000, maxfun={maxfun}) ---")
        try:
            seq_lbfgs, val_lbfgs = _lbfgs_optimize(
                global_best_seq, n_large, deadline, beta=6000.0, maxfun=maxfun)
            if val_lbfgs < global_best_val:
                global_best_seq = seq_lbfgs.copy()
                global_best_val = val_lbfgs
                print(f"[L‑BFGS] new global best = {val_lbfgs:.6f}")
        except Exception as exc:
            print("[L‑BFGS] failed:", exc)

    # ------------------------------------------------------------------
    # Stage 3 – sub‑gradient refinement (exact max objective)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        remaining = deadline - time.time()
        max_iters = int(remaining * 2000)   # generous scaling
        max_iters = max(200_000, min(max_iters, 2_000_000))
        print(f"\n--- Stage 3: sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n_large, deadline,
            max_iters=max_iters,
            init_lr=0.20,
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[Sub‑Proj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_sa_iters = int(remaining * 800)   # conservative estimate
        max_sa_iters = max(200_000, min(max_sa_iters, 1_500_000))
        print(f"\n--- Stage 4: simulated‑annealing polishing (max_iters={max_sa_iters}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n_large, deadline,
            max_iters=max_sa_iters,
            init_temp_factor=0.04,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
