    # Build matrix A (conv matrix)
    c = np.concatenate([f, np.zeros(n-1)])
    r = np.concatenate([[f[0]], np.zeros(n-1)])

    # objective minimize -sum(g)
    c_obj = -np.ones(n)
    # bounds for variables: (0, None)
    bounds = [(0, None) for _ in range(n)]

    if not res.success:
        # fallback: return uniform
        return np.full(n, 1.0 / n)
    return res.x

def _interpolate(seq, n_new):
    n_old = len(seq)
    xs = (np.arange(n_old, dtype=np.float64) + 0.5) / n_old
    xb = (np.arange(n_new, dtype=np.float64) + 0.5) / n_new
    ys = np.interp(xb, xs, seq, left=seq[0], right=seq[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    return ys
```

def _lp_optimize(f, M, sum_bound=5.0):
    n = len(f)
    # Ensure positivity lower bound
    f_eps = np.maximum(f, 1e-12)  # ensure all entries >0
    # Build Toeplitz matrix
    c = np.concatenate([f_eps, np.zeros(n-1)])
    r = np.concatenate([[f_eps[0]], np.zeros(n-1)])
    A = scipy.linalg.toeplitz(c, r)
    # Add sum bound
    A_ub = np.vstack([A, np.ones(n, dtype=np.float64)])
    b_ub = np.concatenate([np.full(2*n-1, M, dtype=np.float64), np.array([sum_bound])])
    # Solve
    res = scipy.optimize.linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,
                                 method='highs')
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)
    return res.x
```

"""
Propose a step‑function sequence that minimizes the official
evaluate_sequence() score.

Algorithm
---------
1. Initialise a non‑negative probability vector `f` (sum = 1) by
   interpolating the best known sequence (`height_sequence_1`) or,
   falling back to a triangular shape.
2. **Linear‑programming step** – given `f` and the current bound
   M = max(conv(f,f)), solve a linear program

        maximise   sum(g)
        subject to conv(f,g) ≤ M   (2·n−1 constraints)
                    g ≥ 0
                    sum(g) ≤ B   (B = 5, prevents unboundedness)

   The solution `g` is normalised to sum(g)=1 and mixed with `f`
   (convex combination with a decreasing mixing weight `t`).
   The mix reduces the maximum convolution, and the process is
   repeated while time permits.
3. **Fine‑tuning** – the best vector from step 2 is refined with an
   Adam optimiser that smooths the max‑convolution via a
   log‑sum‑exp surrogate.  Temperature β is increased up to 10 000,
   and the optimisation runs as long as the time budget allows.
4. **Polishing** – a short simulated‑annealing run reshuffles a
   random subset of the mass to squeeze out any remaining slack.
5. The final vector is returned (as a plain list of floats) and also
   stored in the global variable `height_sequence_1` for reuse.

All stages respect the overall time budget (default 1000 s).  The
search works on a moderate size (n≈2000–2500) so that the LP matrix
fits comfortably in memory while still yielding a tight bound.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate objective for a ℓ¹‑normalised sequence (∑seq == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used only for a fallback initialisation)
# ----------------------------------------------------------------------


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _interpolate(seq: np.ndarray, n_new: int) -> np.ndarray:
    """
    Linear interpolation of an arbitrary‑length sequence to length n_new.
    The result is re‑normalised to sum = 1.
    """
    n_old = len(seq)
    xs = (np.arange(n_old, dtype=np.float64) + 0.5) / n_old
    xb = (np.arange(n_new, dtype=np.float64) + 0.5) / n_new
    ys = np.interp(xb, xs, seq, left=seq[0], right=seq[-1])
    tot = ys.sum()
    if tot > 0:
        ys /= tot
    else:
        ys = np.full(n_new, 1.0 / n_new)
    return ys.astype(np.float64)


# ----------------------------------------------------------------------
# Linear‑programming optimiser (step 1)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray, M: float, sum_bound: float = 5.0) -> np.ndarray:
    """
    Solve the LP:
        maximise   sum(g)
        subject to conv(f,g) ≤ M   (2·n−1 constraints)
                    g ≥ 0
                    sum(g) ≤ sum_bound
    Returns the optimal g (may be all‑zeros if the problem is infeasible).
    """
    n = len(f)
    # Ensure every entry participates in the constraints (tiny epsilon)
    f_eps = np.maximum(f, 1e-12)

    # Build the full convolution matrix A (size (2n‑1)×n)
    c = np.concatenate([f_eps, np.zeros(n - 1, dtype=np.float64)])
    r = np.concatenate([[f_eps[0]], np.zeros(n - 1, dtype=np.float64)])
    A = scipy.linalg.toeplitz(c, r)          # dense, but n≈2500 → ~100 MiB
    # Add the sum‑bound row
    A_ub = np.vstack([A, np.ones(n, dtype=np.float64)])
    b_ub = np.concatenate([np.full(2 * n - 1, M, dtype=np.float64),
                           np.array([sum_bound], dtype=np.float64)])

    # Objective: maximise sum(g) → minimise -sum(g)
    c_obj = -np.ones(n, dtype=np.float64)

    # Bounds for each variable: g_i ≥ 0
    bounds = [(0.0, None) for _ in range(n)]

    # Solve
    res = scipy.optimize.linprog(
        c=c_obj,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )
    if not res.success:
        # Fallback to a uniform distribution – never crashes later
        return np.full(n, 1.0 / n, dtype=np.float64)
    return res.x


# ----------------------------------------------------------------------
# Adam optimiser (step 3)
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   beta_max: float = 10_000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max‑convolution.
    Parameters are logits of the soft‑max representation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 10.0
    beta_mul = 1.006

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # temperature increase
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (step 4)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Stochastic polishing that reshuffles a random subset of the mass.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val
