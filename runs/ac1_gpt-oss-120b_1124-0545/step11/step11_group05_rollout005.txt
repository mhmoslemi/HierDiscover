"""
Propose a non‑negative step function that (hopefully) improves the bound on
evaluate_sequence().  The method combines three ideas:

1. **Warm‑start** – reuse the best known sequence (global variable
   ``height_sequence_1``).  If it exists we linearly upscale it to a larger
   length (8000) to give the optimiser more degrees of freedom.

2. **Exact‑max Adam optimisation** – we treat the objective
   ``2·n·max(conv(a,a))`` (with ``sum(a)=1``) and run Adam on the
   *log‑softmax* parametrisation of the vector.  The gradient is the exact
   sub‑gradient of the max‑convolution (no log‑sum‑exp smoothing).  Several
   random Dirichlet seeds are also tried and the best result is kept.

3. **Momentum projected sub‑gradient descent** – after the Adam phase we
   perform a few hundred thousand iterations of a momentum sub‑gradient method
   on the true max‑convolution objective, with Euclidean projection onto the
   probability simplex after each step.

All stages respect the global time budget (default 1000 s).  The final
sequence is stored in ``height_sequence_1`` for possible reuse and also
returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate objective for an ℓ¹‑normalised sequence (∑seq == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple shape generators (used for random restarts)
# ----------------------------------------------------------------------


def _random_dirichlet_seq(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex with Dirichlet( conc,…,conc )."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a small‑n sequence to a larger n (preserves ℓ¹‑norm)."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    ys = np.maximum(ys, 0.0)
    tot = ys.sum()
    if tot > 0.0:
        ys /= tot
    else:
        ys = np.full(n_big, 1.0 / n_big)
    return ys.astype(np.float64)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_exact_optimize(start_vec: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the *exact* max‑convolution objective.
    Parameters are logits of the soft‑max representation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # initialise logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.6

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                       # stays on simplex
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        cur_val = 2.0 * n * max_conv          # exact surrogate value

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[AdamExact] it={it:6d} best={best_val:.6f}")

        # --- sub‑gradient of max(conv) w.r.t. a -----------------------
        max_idx = int(np.argmax(conv))

        # partner[i] = a[max_idx - i] when the index is valid, else 0
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i = np.arange(i_start, i_end + 1, dtype=int)
        partner[i] = a[max_idx - i]
        grad_a = 2.0 * partner                 # sub‑gradient ∂/∂a max(conv)

        # Transform gradient to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # --- Adam update ---------------------------------------------
        beta_temp = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - beta_temp * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _momentum_subgrad_opt(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with heavy‑ball momentum on the exact
    max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    velocity = np.zeros_like(a)
    beta = 0.9                 # momentum coefficient
    init_lr = 0.5

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[MomSubGrad] it={it:7d} best={best_val:.6f}")

        # gradient of max conv (same as in AdamExact)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i = np.arange(i_start, i_end + 1, dtype=int)
        partner[i] = a[max_idx - i]
        grad = 2.0 * partner

        # heavy‑ball momentum
        velocity = beta * velocity + (1.0 - beta) * grad
        lr = init_lr / np.sqrt(it)

        a = a - lr * velocity
        a = _proj_simplex(a)   # stay on the probability simplex

    return best_seq, best_val

        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate (2·n·max(conv)) ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # store for later reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
