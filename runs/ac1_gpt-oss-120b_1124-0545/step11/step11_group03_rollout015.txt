"""
Propose a step‑function (non‑negative list) that minimises the supplied
``evaluate_sequence`` objective.

The core idea
---------------
* The objective equals 2·n·max(conv(a,a)) when the ℓ₁‑norm of ``a`` is 1.
* ``max(conv)`` is the ℓ∞‑norm of the *anti‑diagonal* sums of the outer
  product a·aᵀ.  Their total mass is 1, hence the theoretical lower bound of
  the objective is 2·n/(2·n‑1)≈1.  The current best (≈1.508) is far from this
  limit.
* We minimise a *smooth* approximation of the max‑convolution and
  periodically perform a greedy “mass‑transfer’’ move that directly reduces
  the dominant anti‑diagonal.  The smooth objective is

        Φ_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])

  whose gradient can be written using only convolution operations,
  allowing an O(n log n) update.

* After each gradient step we renormalise onto the probability simplex
  (project onto ℓ₁‑norm = 1, a_i≥0).  The optimisation runs until the
  global deadline (≈1000 s) is reached, keeping the best vector seen.

The implementation
------------------
* Generates a rich set of analytic seeds (uniform, power‑law, beta,
  windows, exponential, Dirichlet …) and also re‑uses the previous
  solution if present.
* Starts from the best seed (lowest surrogate 2·n·max(conv)).
* Alternates between
    – Adam‑style gradient descent on the smooth max (α grows during the run),
    – a cheap targeted transfer that moves a small amount of mass from the
      index that contributes most to the current worst anti‑diagonal to an
      index that contributes the least.
* Occasionally applies a tiny random shake (pairwise transfer) to escape
  shallow plateaus.
* Finally evaluates the best vector with the official ``evaluate_sequence``
  function and stores it globally as ``height_sequence_1`` for the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric tools (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w | w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Fast non‑negative normalisation (approximate projection)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – the value returned by evaluate_sequence
    when Σseq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _simplex_normalise(a)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return _simplex_normalise(mixed)

# Seed generation – diverse pool of ℓ₁‑normalised vectors

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # power‑law grid
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # beta family – a coarse grid covering U‑shapes (α,β<1) and bell shapes
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (different concentration parameters)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential – heavy‑tailed
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # mixture of uniform and a random Dirichlet (adds diversity)
    for mix in (0.2, 0.5, 0.8):
        base = _uniform(n)
        noisy = _random_dirichlet(n, 0.5, rng)
        seeds.append(_mix_shapes(base, noisy, mix))

    # previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.shape == (n,) and arr.sum() > 0.0:
            seeds.append(_simplex_normalise(arr))

    # final clean‑up (ensure true normalisation)
    cleaned = []
    for v in seeds:
        v = _simplex_normalise(np.asarray(v, dtype=np.float64))
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _smooth_max_gradient_step(a: np.ndarray,
                              alpha: float,
                              lr: float,
                              m: np.ndarray,
                              v: np.ndarray,
                              beta1: float,
                              beta2: float,
                              eps: float) -> tuple[np.ndarray, np.ndarray, np.ndarray, float]:
    """
    One Adam‑style step on the smooth max objective.
    Returns (new_a, new_m, new_v, max_conv_before_projection).
    """
    n = a.shape[0]

    # convolution a*a  (full)
    conv = _conv_fft(a, a)                         # length 2n‑1

    # soft‑max weights (stable)
    shift = conv.max()
    w = np.exp(alpha * (conv - shift))
    w /= w.sum()

    # gradient = 2 * (a_rev * w)  on the slice that corresponds to a‑indices
    a_rev = a[::-1]
    grad_full = _conv_fft(a_rev, w)               # length 3n‑2
    grad = 2.0 * grad_full[n - 1:n - 1 + n]

    # Adam update
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1)
    v_hat = v / (1.0 - beta2)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

    # project back onto the simplex (fast renormalisation)
    a_new = _simplex_normalise(a_new)

    return a_new, m, v, conv.max()


def _targeted_transfer(a: np.ndarray, conv: np.ndarray, k_max: int, rng: np.random.Generator) -> np.ndarray:
    """
    Greedy “mass‑transfer’’ move:
    move a small chunk from the index that contributes most to the worst
    anti‑diagonal (k_max) to an index that contributes least.
    """
    n = a.shape[0]
    idx = np.arange(n)
    partner = k_max - idx
    valid = (partner >= 0) & (partner < n)

    # contribution of each index to the problematic shift
    contrib = np.zeros(n, dtype=np.float64)
    contrib[valid] = a[idx[valid]] * a[partner[valid]]

    i_max = int(np.argmax(contrib))                 # worst contributor
    # choose a target with minimal contribution (zero is ideal)
    min_contrib = contrib.min()
    candidates = np.where(contrib == min_contrib)[0]
    # avoid self‑transfer
    if i_max in candidates and candidates.size > 1:
        candidates = candidates[candidates != i_max]
    j_min = int(rng.choice(candidates))

    # transfer a modest amount – never more than 5 % of the source mass
    eps = min(a[i_max] * 0.05, 0.01)
    if eps > 0.0:
        a[i_max] -= eps
        a[j_min] += eps

    # re‑normalise (sum is unchanged, only non‑negativity can be violated)
    a = np.maximum(a, 0.0)
    return a

        cur_val = _objective_surrogate(cur)
        best_seq_final = cur.copy()
        best_val_final = cur_val
        T = max(1e-3 * cur_val, 1e-12)

        while time.time() < deadline:
            T = max(T * 0.9995, 1e-14)

            # two‑point mass transfer (the classic SA move)
            i, j = rng.choice(n, size=2, replace=False)
            eps_move = rng.random() * min(cur[i], 0.04)   # up to 4 % of source
            if eps_move == 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps_move
            cand[j] += eps_move
            cand = _simplex_normalise(cand)

            cand_val = _objective_surrogate(cand)
            delta = cand_val - cur_val
            if delta < 0.0 or rng.random() < math.exp(-delta / T):
                cur, cur_val = cand, cand_val
                if cand_val < best_val_final:
                    best_val_final, best_seq_final = cand_val, cand.copy()
        best_seq = best_seq_final
        best_val = best_val_final
        print(f"[final SA] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – Evaluate with the official evaluator and store the result.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== finished ===\nsurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning – evaluate_sequence raised:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
