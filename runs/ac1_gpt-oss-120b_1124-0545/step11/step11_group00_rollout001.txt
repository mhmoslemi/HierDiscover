"""
Propose a step‑function that minimizes the official ``evaluate_sequence`` score.

Algorithm
---------
1. **Rich seed pool** – deterministic analytic families (uniform,
   triangular, sine‑window, Gaussian, beta, Tukey, exponential, cosine)
   together with a large number of random Dirichlet / low‑frequency curves.
   All seeds are ℓ₁‑normalised (sum = 1).  The cheap surrogate
   ``_objective`` (2·n·max(conv)) is evaluated for every seed and the
   best *k* are kept.

2. **Adam on a temperature‑controlled log‑sum‑exp surrogate** – each of the
   retained seeds is optimised with Adam on the log‑softmax
   parametrisation.  The surrogate replaces the non‑smooth max by a
   log‑sum‑exp whose temperature is slowly increased up to 500, so the
   optimiser finally sees the true max.

3. **Mirror‑descent (exponentiated‑gradient) on the exact max** – from the
   best Adam result we run a very long mirror‑descent.  The sub‑gradient
   is computed from *all* (not only a single) active constraints,
   i.e. every convolution index whose value is within a tiny tolerance of
   the current maximum.  This yields a stronger descent direction.

4. **Simulated‑annealing polishing** – random swaps, Dirichlet‑redistribute,
   pair‑transfer and block‑swap moves are accepted with a geometric cooling
   schedule, allowing the algorithm to escape shallow minima.

5. **Final evaluation** – the best sequence found is converted to a plain
   list, stored globally as ``height_sequence_1`` and returned.

The whole routine respects a hard time budget (default 1000 s) and never
exceeds it.  It works for any length; we use ``n = 4000`` which offers a
good trade‑off between quality and speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective (the factor 2·n is irrelevant for ranking):
    2·n·max(conv(seq,seq))  for an ℓ₁‑normalised sequence (Σseq = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random / low‑frequency helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve obtained by summing a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,

    Adam optimisation on log‑softmax parameters using a log‑sum‑exp
    surrogate for the max.  A temperature β is increased slowly up to
    β_max = 500.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta = 0.5          # initial temperature
    beta_inc = 0.5
    beta_max = 500.0

    best_seq = None
    best_val = float("inf")
    it_print = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # current probability vector (sum = 1)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            it_print += 1
            if it_print % 5_000 == 0:
                print(f"[Adam] it={it:7d}  best={cur_val:.6f}")

        # --- smooth max via log‑sum‑exp ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerical issue – skip gradient update
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on the *exact* max.
    The sub‑gradient is built from *all* active constraints
    (convolution entries within a tiny tolerance of the current maximum).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12          # initial step‑size factor
    power = 0.35        # slower decay than ½
    no_imp_counter = 0
    IMP_STOP = 300_000  # stop early if no improvement for a while

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_imp_counter = 0
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:7d}  best={cur_val:.6f}")
        else:
            no_imp_counter += 1
            if no_imp_counter >= IMP_STOP:
                # stagnation – break early
                break

        # ---------- sub‑gradient from all nearly‑active indices ----------
        # absolute tolerance; also a relative one to cope with tiny values
        tol_abs = 1e-12
        tol_rel = 1e-8
        active = np.where(conv >= max_val - max(tol_abs, tol_rel * max_val))[0]

        # weight vector w for the active set (simple 1‑weight each)
        w = np.zeros_like(conv)
        w[active] = 1.0

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]          # central n entries

        # multiplicative (exponentiated‑gradient) update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * grad_a)

        # project back to the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 1_500_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple SA that performs four move types:
        0 – swap two coordinates,
        1 – redistribute mass over a random subset,
        2 – split‑merge between two coordinates,
        3 – block‑swap of two contiguous segments.
    The temperature decays geometrically.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 4)   # include the block‑swap

        if move_type == 0:   # swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]

        elif move_type == 1:  # redistribute a random subset
            k = rng.integers(2, 12)    # subset size
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc

        elif move_type == 2:  # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        else:  # block‑swap (move_type == 3)
            block_len = rng.integers(1, 6)          # small block
            i = rng.integers(0, n - block_len + 1)
            j = rng.integers(0, n - block_len + 1)
            if i == j:
                continue
            new = cur.copy()
            blk_i = new[i:i + block_len].copy()
            blk_j = new[j:j + block_len].copy()
            new[i:i + block_len] = blk_j
            new[j:j + block_len] = blk_i

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d}  best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed collection – richer parameter grids
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # Sine windows (dense gamma grid)
    for gamma in np.linspace(0.3, 4.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian windows
    for sigma in np.linspace(0.08, 0.45, 10):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Power‑law family
    for p in np.linspace(0.0, 4.0, 9):
        seeds.append(_power(n, p))

    # Beta family
    alphas = np.linspace(0.3, 3.0, 7)
    betas = np.linspace(0.3, 3.0, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.05, 0.95, 9):
        seeds.append(_tukey_window(n, alpha))

    # Exponential decay
    for lam in np.linspace(0.0005, 0.015, 8):
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending
            # alternating high‑low ordering
            a = np.sort(raw)
            lo, hi = 0, len(a) - 1
            alt = np.empty_like(a)
            pos = 0
            while lo <= hi:
                if pos % 2 == 0:
                    alt[pos] = a[hi]
                    hi -= 1
                else:
                    alt[pos] = a[lo]
                    lo += 1
                pos += 1
            seeds.append(alt)

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Incorporate previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best sequence as a plain list of floats and stores it in
    the global variable ``height_sequence_1``.
    """
    # ------------------------------------------------------------------
    # Setup and time management
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length of the candidate – 4000 works well on the platform
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve a previously found solution (if any) to seed the search
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich pool of analytic / random seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Fast surrogate evaluation and keep the best few
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20                       # a slightly larger pool than before
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # initialise the global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on the smooth log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest proportion of the remaining budget to each run
        max_iters = min(180_000, int(remaining * 55))   # roughly 0.02 s per iter
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent sub‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=5_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=1_500_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (informational)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:                      # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by subsequent calls
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
