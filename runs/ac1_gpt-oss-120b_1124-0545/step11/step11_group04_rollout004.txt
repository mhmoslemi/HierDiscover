def _lp_g_step(f, M):
    n = len(f)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    b = np.full(rows, M)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None

"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines three ideas:

1. **Large, diverse analytic seeds** (uniform, windows, random Dirichlet, etc.).
2. **Log‑space Adam optimisation** on the simplex (soft‑max parametrisation) to
   drive the smoothed ℓ∞‑norm of the convolution down.
3. **Linear‑programming “mix‑step”**:
   for the current candidate f (∑f=1) we compute the maximal
   self‑convolution value M = max(conv(f,f)).
   We then solve the LP

        maximise   ∑ g_j
        subject to conv(f,g) ≤ M   (all shifts)
                   g_j ≥ 0

   The solution g can carry more ℓ₁‑mass while respecting the
   same convolution bound.  A convex combination (1‑τ)·f + τ·(g/∑g) is
   searched over τ∈[0,1] and the best mixture is kept.
   Repeating this step strictly reduces the objective.

After a few LP refinements the best vector is up‑sampled to a finer
grid (8000 points) and polished again with Adam and a short simulated‑
annealing run.  The whole procedure respects the 1000 s deadline and
stores the final sequence in the global name ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports
# ----------------------------------------------------------------------
try:
    from scipy.signal import windows as _scipy_windows
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    _scipy_windows = None
    linprog = None

# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """smallest power of two ≥ x"""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector
    (∑seq = 1).  The returned value is proportional to the true
    evaluate_sequence value and respects the ordering of candidates.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _scipy_windows is not None and hasattr(_scipy_windows, 'kaiser'):
        a = _scipy_windows.kaiser(n, beta)
    else:
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        t = beta * np.sqrt(1 - y * y)
        # 8‑term series approximation of I0(t)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    if _scipy_windows is not None and hasattr(_scipy_windows, 'dpss'):
        a = _scipy_windows.dpss(n, nw, Kmax=1, return_ratios=False)[0]
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    return _uniform(n)


def _chebwin_window(n: int, att: float) -> np.ndarray:
    if _scipy_windows is not None and hasattr(_scipy_windows, 'chebwin'):
        a = _scipy_windows.chebwin(n, at=att)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    return _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15, shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few low‑frequency cosine components, clip to non‑negative and normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

    seeds: list[np.ndarray] = []

    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):

    for nw in (2.0, 3.0, 4.0, 5.0):
        seeds.append(_dpss_window(n, nw))

    for att in (30, 40, 60, 80):

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(5):

    for nf in (2, 3, 4, 5):

            seeds.append(_lowfreq_curve(n, nf, rng))

    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    cleaned = []
    for v in seeds:

        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑softmax parametrisation.
    Returns the best ℓ₁‑normalised vector found and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5          # initial temperature for smoothed max
    beta_inc = 0.1
    beta_max = 200.0    # final temperature (higher → tighter approximation)

    lr0 = 1.0            # initial learning rate in log‑space

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probability simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[LogAdam] it={it:6d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient on logits (log‑softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update (learning‑rate decays like 1/√t)
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:          # reshuffle a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # small pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid (keeps the shape) and ℓ₁‑renormalises."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Linear‑programming “mix” step
# ----------------------------------------------------------------------
def _solve_lp_for_g(f_vec: np.ndarray, M: float, deadline: float) -> np.ndarray | None:
    """
    Solve   maximise ∑ g_j   subject to   conv(f,g) ≤ M,   g ≥ 0.
    Returns the solution g (not normalised) or None on failure/timeout.
    """
    if linprog is None:
        return None
    if time.time() > deadline:
        return None

    n = f_vec.shape[0]
    rows = 2 * n - 1
    # Build the (rows × n) matrix A where A @ g = conv(f,g)
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f_vec
    b = np.full(rows, M, dtype=np.float64)

    # maximise ∑ g   → minimise –∑ g
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n

    # Use HiGHS (default) – fast for dense medium‑scale problems
    res = linprog(c=c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final call

    # ------------------------------------------------------------------
    # Stage 0 – base resolution (small enough for LP)
    # ------------------------------------------------------------------
    n0 = 2000
    print(f"\n=== Starting search (base n = {n0}) ===")

    # ------------------------------------------------------------------
    # Stage 1 – seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n0:
                # down‑sample or up‑sample to the required size
                prev_arr = _upsample_sequence(prev_arr, n0)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n0, rng, prev_arr)
    print(f"Collected {len(seeds)} seeds")

    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    best_seq, best_val = evals[0]
    print(f"[Init] Best surrogate (n={n0}) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the current best seed
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 0.30 * 1e5))
        print(f"Running Adam (max_iters={max_iters})")
        seq_adam, val_adam = _logspace_adam_optimize(best_seq, n0, deadline,
                                                    max_iters=max_iters)
        if seq_adam is not None and val_adam < best_val:
            best_seq, best_val = seq_adam, val_adam
            print(f"[Adam] New best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – LP‑based mixing refinements (few iterations)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Allow a short grace period for one more LP step
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # current bound
        conv_cur = _conv_fft(best_seq, best_seq)
        M = np.max(conv_cur)

        # solve LP
        g = _solve_lp_for_g(best_seq, M, deadline)
        if g is None:
            print("[LP] Solver failed or timed out – terminating LP loop")
            break

        s = g.sum()
        if s <= 0.0:
            print("[LP] Obtained zero‑mass solution – terminating")
            break
        g_norm = g / s

        # search for best mixing coefficient τ
        best_tau = None
        best_tau_val = best_val
        for tau in np.linspace(0.1, 0.9, 9):
            cand = (1.0 - tau) * best_seq + tau * g_norm
            cand_val = _objective_surrogate(cand)
            if cand_val < best_tau_val:
                best_tau_val = cand_val
                best_tau = tau
        if best_tau is not None and best_tau_val < best_val - 1e-9:
            best_seq = (1.0 - best_tau) * best_seq + best_tau * g_norm
            best_val = best_tau_val
            print(f"[LP] Mixed with τ={best_tau:.3f} → surrogate {best_val:.6f}")
        else:
            print("[LP] No improving τ found – exiting LP refinement")
            break

    # ------------------------------------------------------------------
    # Stage 4 – Upsample to a finer grid and final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_up = 8000
        print(f"\nUpsampling to n={n_up}")
        best_seq = _upsample_sequence(best_seq, n_up)

        # Adam on the finer grid
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_iters = int(min(400_000, remaining * 0.30 * 1e5))
            print(f"Fine‑grid Adam (max_iters={max_iters})")
            seq_up_adam, val_up_adam = _logspace_adam_optimize(best_seq, n_up,
                                                              deadline,
                                                              max_iters=max_iters)
            if seq_up_adam is not None and val_up_adam < best_val:
                best_seq, best_val = seq_up_adam, val_up_adam
                print(f"[Fine‑Adam] New best surrogate = {best_val:.6f}")

        # Final short simulated annealing polish
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_iters = int(min(200_000, remaining * 1e5))
            print(f"Final SA polish (max_iters={max_iters})")
            seq_sa, val_sa = _simulated_annealing(best_seq, n_up, deadline,
                                                  max_iters=max_iters)
            if val_sa < best_val:
                best_seq, best_val = seq_sa, val_sa
                print(f"[Final SA] New best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
