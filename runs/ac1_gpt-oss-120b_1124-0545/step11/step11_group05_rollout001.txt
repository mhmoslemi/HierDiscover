"""
Propose a non‑negative step‑function that (hopefully) improves the bound.

Algorithm overview
------------------
1. **Analytic seed sweep** – generate a large pool of parameterised
   families (uniform, triangular, trapezoidal, geometric, power, sine,
   beta, Gaussian, Tukey, random Dirichlet/exp).  All seeds are
   ℓ¹‑normalised (sum = 1) and the exact surrogate objective
   ``2·n·max(conv(seq,seq))`` is evaluated.  The best few seeds are kept.

2. **Iterated linear‑programming refinement** – for the current seed *f*
   compute ``M = max(conv(f,f))`` and solve the LP

        maximise   Σ g_j
        subject to  conv(f,g) ≤ M   (element‑wise)
                    g_j ≥ 0

   The LP is small (≈2n constraints, n variables) and is solved with
   CVXOPT/ECOS.  The solution ``g`` is rescaled to a probability vector
   ``f ← g / Σg``.  This step strictly reduces the objective (if possible).
   A few iterations (typically ≤ 5) are performed.

3. **Upscaling** – the refined small‑scale distribution is linearly
   interpolated to a larger length (≈ 8000).  The interpolation keeps the
   ℓ¹‑norm equal to 1.

4. **Heavy simulated‑annealing polishing** – starting from the upscaled
   vector we run a long, temperature‑controlled SA that repeatedly
   reshuffles the mass of a random subset (Dirichlet re‑allocation).
   Each move is accepted with the Metropolis criterion.  The schedule
   is tuned to the remaining time budget.

5. **Final verification** – the best sequence found is passed to the
   provided ``evaluate_sequence`` function and stored globally as
   ``height_sequence_1`` for possible reuse.

The whole routine respects the given time budget (default 1000 s) and
returns a plain ``list[float]``.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate objective for a ℓ¹‑normalised sequence (∑seq == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return ℓ¹‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (0<r<1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _generate_small_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a fairly dense pool of analytically‑parameterised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic bases
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # trapezoidal sweep
    for wf in np.linspace(0.02, 0.45, 12):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    # sine windows
    for gamma in np.linspace(0.0, 6.0, 31):
        seeds.append(_sine_window(n, gamma=gamma))

    # tukey windows
    for alpha in np.linspace(0.0, 0.95, 20):
        seeds.append(_tukey_window(n, alpha=alpha))

    # geometric families
    for r in np.concatenate([np.linspace(0.01, 0.98, 20), np.linspace(0.98, 0.9999, 15)]):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # power‑law family
    for p in np.linspace(0.0, 7.0, 29):
        seeds.append(_power(n, p))

    # beta shapes
    alphas = np.linspace(0.2, 5.0, 20)
    betas = np.linspace(0.2, 5.0, 20)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma_frac))

    # random Dirichlet draws
    for conc in (0.001, 0.005, 0.02, 0.1, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                tot = prev_arr.sum()
                if tot > 0:
                    seeds.append(prev_arr / tot)
        except Exception:
            pass

    return seeds


# ----------------------------------------------------------------------
# Upscaling / interpolation
# ----------------------------------------------------------------------


def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a small‑n sequence to a larger n."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    else:
        ys = _uniform(n_big)
    return ys.astype(np.float64)


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    One LP iteration:
        maximise   Σ g_j
        subject to  conv(f,g) ≤ M   (for all shifts)
                    g_j ≥ 0
    Returns the optimal g (ℓ¹‑normalised later) or None on failure.
    """
    import cvxpy as cp

    n = len(f)
    m = 2 * n - 1

    # Build the convolution matrix A (size m×n) such that A @ g = conv(f,g)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]  # column i contributes to rows i … i+n‑1

    # CVXPY model
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(m)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception as e:
        print("[LP] Solver exception:", e)
        return None

    if g.value is None or prob.status not in ("optimal", "optimal_inaccurate"):
        print("[LP] Solver status:", prob.status)
        return None

    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    if g_val.sum() < 1e-12:
        return None
    return g_val


# ----------------------------------------------------------------------
# Simulated annealing polishing
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_000_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polishing that repeatedly redistributes the mass of a
    random subset (Dirichlet re‑allocation).  Returns best sequence and its
    surrogate objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        # pick a random subset of indices (2 – 15 elements)
        k = rng.integers(2, 15)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        # re‑allocate via Dirichlet
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The routine respects the time budget and returns the best sequence found
    as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # 1 s safety margin

    # ------------------------------------------------------------------
    # Stage 0 – analytic sweep on modest size
    # ------------------------------------------------------------------
    n_small = 1500
    print(f"\n--- Stage 0: exploring {n_small}‑length analytic families ---")
    small_seeds = _generate_small_seeds(n_small, rng)
    print(f"Generated {len(small_seeds)} candidates.")

    evaluated = [(vec, _objective(vec)) for vec in small_seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    best_small = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – LP refinement (few iterations)
    # ------------------------------------------------------------------
    # Take the very best small seed as starting point
    f = best_small[0].copy()
    best_val = _objective(f)
    print(f"\n[LP] initial surrogate = {best_val:.6f}")

    max_lp_iters = 5
    for lp_it in range(1, max_lp_iters + 1):
        if time.time() > deadline - 5.0:
            print("[LP] Time low – aborting further LP steps.")
            break

        M = _conv_fft(f, f).max()
        g = _lp_step(f, M)
        if g is None:
            print("[LP] LP step failed – stopping.")
            break

        S = g.sum()
        if S <= 0.0:
            print("[LP] Sum of g non‑positive – stopping.")
            break

        f_new = g / S
        val_new = _objective(f_new)
        print(f"[LP] iteration {lp_it}: surrogate = {val_new:.6f}  (sum g = {S:.6f})")

        if val_new < best_val - 1e-12:  # improvement
            f = f_new
            best_val = val_new
        else:
            print("[LP] No improvement – terminating LP loop.")
            break

    # ------------------------------------------------------------------
    # Stage 2 – upscale to a larger problem size
    # ------------------------------------------------------------------
    n_large = 8000
    print(f"\n--- Stage 2: upscaling to length {n_large} ---")
    f_large = _upscale(f, n_large)
    val_large = _objective(f_large)
    print(f"[Upscale] surrogate after upscaling = {val_large:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # Estimate iterations (roughly 0.0012 s per iteration for n≈8000)
        est_iters = int(remaining / 0.0012)
        est_iters = max(200_000, min(est_iters, 2_000_000))
        print(f"\n--- Stage 3: simulated‑annealing polishing (≈{est_iters} iters) ---")
        seq_sa, val_sa = _simulated_annealing(
            f_large, n_large, deadline,
            max_iters=est_iters,
            init_temp_factor=0.04,
        )
        if val_sa < val_large:
            f_large = seq_sa
            best_val = val_sa
            print(f"[SA] new best surrogate = {best_val:.6f}")
        else:
            best_val = val_large
            print(f"[SA] best surrogate unchanged = {best_val:.6f}")
    else:
        print("[SA] Not enough time left for annealing.")

    # ------------------------------------------------------------------
    # Final verification with the official evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(f_large.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate (2·n·max(conv)) ≈ {best_val:.6f} "
        f"(length {len(f_large)})"
    )

    # Store for possible later reuse
    globals()["height_sequence_1"] = f_large.tolist()
    return f_large.tolist()
