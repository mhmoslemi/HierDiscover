"""
Propose a non‑negative step function that (hopefully) beats the previous bound.

Algorithm (≈1000 s total)
---------------------------------
0.  Analytic “seed” generation on a modest size (n≈1200).  A rich
    library of families (uniform, triangular, trapezoidal, sine,
    Tukey, geometric, beta, Gaussian) plus a few random Dirichlet /
    exponential draws is built.  The exact surrogate objective
    2·n·max(conv) (with ∑a = 1) is evaluated and the best `k_small`
    seeds are kept.

1.  Random convex combinations of the best seeds are sampled
    (Dirichlet weights) to enrich the pool.

2.  All shortlisted seeds are linearly interpolated to a large length
    `n_big` (≈8000).  The best `k_big` (≈3) of these are selected as
    starting points for the heavy optimisation.

3.  **Adam on a smooth‑max surrogate**.
    The sequence is represented by logits of a soft‑max (staying on
    the probability simplex).  A temperature β starts at 5 and grows
    slowly (β←β·1.0015, capped at 5000).  The gradient of the
    log‑sum‑exp surrogate is obtained via a weighted convolution;
    Adam (β1=0.9, β2=0.999) updates the logits.  For each seed we run
    as many iterations as the remaining time budget permits, keeping
    the best iterate.

4.  **Projected sub‑gradient descent on the true max‑convolution**.
    From the best Adam result we run a Polyak‑style sub‑gradient method,
    projecting after every step onto the probability simplex.

5.  **Simulated‑annealing polishing**.
    Random Dirichlet re‑allocation of a small random subset of indices
    (2‑12 positions) is performed; Metropolis acceptance with a slowly
    cooling temperature refines the solution.

6.  The final sequence is verified with the official `evaluate_sequence`
    and stored in the global variable `height_sequence_1`.

All stages respect the global time budget; intermediate bounds are
printed for monitoring.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate objective for a L1‑normalised sequence (∑seq == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return L1‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _upscale(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a short sequence to length n_big, then renormalise."""
    n_small = len(seq_small)
    xs = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    xb = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    ys = np.interp(xb, xs, seq_small, left=seq_small[0], right=seq_small[-1])
    total = ys.sum()
    if total > 0:
        ys /= total
    else:
        ys = _uniform(n_big)
    return ys.astype(np.float64)


def _generate_small_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a dense pool of analytically parameterised candidates (L1‑normalised)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.02, 0.45, 12):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    for gamma in np.linspace(0.0, 6.0, 31):
        seeds.append(_sine_window(n, gamma=gamma))

    for alpha in np.linspace(0.0, 0.95, 20):
        seeds.append(_tukey_window(n, alpha=alpha))

    for r in np.concatenate([np.linspace(0.01, 0.98, 20), np.linspace(0.98, 0.9999, 15)]):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 7.0, 29):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 20)
    betas = np.linspace(0.2, 5.0, 20)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for sigma_frac in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma_frac))

    # random Dirichlet draws
    for conc in (0.001, 0.005, 0.02, 0.1, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # previous solution, if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    seeds.append(prev_arr / s)
        except Exception:
            pass

    return seeds


def _random_convex_combination(seeds: list[np.ndarray],
                               m: int,
                               rng: np.random.Generator) -> list[np.ndarray]:
    """Sample `m` random convex combinations of the given seeds."""
    combos: list[np.ndarray] = []
    if len(seeds) < 2:
        return combos
    for _ in range(m):
        weights = rng.dirichlet(np.ones(len(seeds)))
        combo = np.zeros_like(seeds[0])
        for w, s in zip(weights, seeds):
            combo += w * s
        # renormalise to sum 1 (numerical drift)
        combo_sum = combo.sum()
        if combo_sum > 0:
            combo /= combo_sum
        else:
            combo = _uniform(len(combo))
        combos.append(combo)
    return combos

                   max_iters: int = 300_000,
                   beta_start: float = 5.0,
                   beta_mul: float = 1.0015,
                   beta_max: float = 5000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Parameters are logits of the soft‑max representation.
    Returns best sequence found and its surrogate value.
    """
    eps = 1e-30
    logits = np.log(start_vec + eps)          # initialise logits

    m = np.zeros_like(logits)
    v2 = np.zeros_like(logits)

    beta_temp = beta_start

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(logits)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = float(2.0 * n * np.max(conv))

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # increase temperature (smoothness -> sharp)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # smooth‑max weighting
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # convert to gradient w.r.t. logits (log‑softmax coords)
        dot = np.dot(grad_a, a)
        grad_logits = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_logits)
        if norm > max_norm:
            grad_logits = grad_logits * (max_norm / norm)

        # Adam update (classic schedule)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        logits = logits - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a Polyak‑style diminishing step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = float(2.0 * n * np.max(conv))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = float(2.0 * n * conv[max_idx])

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polishing that reshuffles a random subset of the mass.
    Each move is a Dirichlet re‑allocation of `k` randomly chosen indices.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = float(2.0 * n * np.max(_conv_fft(cur, cur)))
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc

        new_val = float(2.0 * n * np.max(_conv_fft(new, new)))
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
    `evaluate_sequence`.  Returns a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – cheap analytic exploration (n≈1200)
    # ------------------------------------------------------------------
    n_small = 1200
    print(f"\n--- Stage 0: Analytic exploration (n={n_small}) ---")
    small_seeds = _generate_small_seeds(n_small, rng)
    print(f"Generated {len(small_seeds)} analytic seeds.")

    # exact surrogate evaluation on the small size
    small_evals = [(vec, _objective(vec)) for vec in small_seeds]
    small_evals.sort(key=lambda x: x[1])

    top_k_small = 12
    best_small = [vec for vec, _ in small_evals[:top_k_small]]
    print(f"Top {top_k_small} small seeds (surrogate values):")
    for i, (_, val) in enumerate(small_evals[:top_k_small]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0b – random convex combinations of the elite seeds
    # ------------------------------------------------------------------
    print("\n--- Stage 0b: Random convex combinations ---")
    combo_seeds = _random_convex_combination(best_small, m=30, rng=rng)
    print(f"Generated {len(combo_seeds)} convex‑combination seeds.")
    # evaluate combos
    combo_evals = [(vec, _objective(vec)) for vec in combo_seeds]
    combo_evals.sort(key=lambda x: x[1])

    # keep best of combos together with original top seeds
    merged = best_small + [vec for vec, _ in combo_evals[:6]]
    print(f"After combos, total candidates for up‑scaling: {len(merged)}")

    # ------------------------------------------------------------------
    # Stage 1 – upscale to a large problem size
    # ------------------------------------------------------------------
    n_large = 8000
    print(f"\n--- Stage 1: Upscaling to n={n_large} ---")
    large_candidates = [_upscale(vec, n_large) for vec in merged]

    large_evals = [(vec, _objective(vec)) for vec in large_candidates]
    large_evals.sort(key=lambda x: x[1])
    global_best_seq = large_evals[0][0].copy()
    global_best_val = large_evals[0][1]
    print(f"[Init] best surrogate after upscale = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy Adam optimisation (use top few seeds)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining < 5.0:
        print("Not enough time for Adam optimisation.")
    else:
        # pick the top 3 upscaled seeds for Adam
        seeds_to_opt = [vec for vec, _ in large_evals[:3]]
        per_seed_budget = remaining / (len(seeds_to_opt) + 1.0)
        print(f"\n--- Stage 2: Adam optimisation (≈{per_seed_budget:.1f}s per seed) ---")
        for idx, seed_vec in enumerate(seeds_to_opt):
            if time.time() > deadline:
                break
            est_iters = int(per_seed_budget / 0.0015)
            est_iters = max(50_000, min(est_iters, 400_000))
            print(f"\n[Adam] seed #{idx+1}/{len(seeds_to_opt)} – max_iters≈{est_iters}")
            seq_adam, val_adam = _adam_optimize(
                seed_vec,
                n_large,
                deadline,
                max_iters=est_iters,
                beta_start=5.0,
                beta_mul=1.0015,
                beta_max=5000.0,
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        remaining = deadline - time.time()
        max_iters = int(remaining * 1200)   # heuristic: ~0.0008 s per iter
        max_iters = max(200_000, min(max_iters, 1_500_000))
        print(f"\n--- Stage 3: Sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_sub, val_sub = _subgradient_optimize(
            global_best_seq,
            n_large,
            deadline,
            max_iters=max_iters,
            init_lr=0.20,
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_sa_iters = int(remaining * 900)   # ~0.0011 s per SA step
        max_sa_iters = max(200_000, min(max_sa_iters, 2_000_000))
        print(f"\n--- Stage 4: Simulated‑annealing polishing (max_iters={max_sa_iters}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            n_large,
            deadline,
            max_iters=max_sa_iters,
            init_temp_factor=0.04,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
