start_time = time.time()

    i = random index from active interval (based on conv max)

    iters+=1

        elapsed = time.time() - start_time
        fraction = elapsed / (deadline - start_time)
        step_frac = max(min_step_frac, init_step_frac * (1 - fraction))

return best_a,best_val

"""
Improved search for a step‑function that minimises ``evaluate_sequence``.

The design follows the same overall structure as the reference code but
adds three important changes that make the optimisation more aggressive
within the same 1000 s wall‑time:

1. **Larger beta‑schedule & more iterations** – we continue the
   continuation L‑BFGS to very large β (up to 1e14) and give each β a
   generous iteration budget.  This yields a much tighter smooth
   approximation of the exact max‑convolution.

2. **Extended stochastic local search** – after the smooth optimisation
   we run a *long* mass‑transfer hill‑climb (up to a few million steps)
   with a decaying step‑size and occasional targeted moves that explicitly
   move mass away from the active (maximal) convolution lag.

3. **Richer seed pool** – we generate many analytic families (including a
   wider sweep of power‑law exponents and shifts) and keep a larger elite
   set for the expensive L‑BFGS phase.

All operations respect the 1000 s budget, updating the global variable
``height_sequence_1`` with the best sequence found.
"""
import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (non‑smooth)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²; a is assumed non‑negative."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (active lag only)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    # indices i that contribute to lag idx
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g = np.zeros_like(a)
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of 2·n·max(conv) with a log‑sum‑exp.
    Returns (obj, grad) where gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised (sum = 1)
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # smooth max
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # soft‑max over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through the simplex softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    ``start_vec`` must already be ℓ₁‑normalised.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0, 50_000_000.0, 100_000_000.0,
            500_000_000.0, 1_000_000_000.0, 5_000_000_000.0,
            10_000_000_000.0, 50_000_000_000.0, 100_000_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits of the probability vector

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(
                fun=_smooth_objective,
                x0=v,
                args=(n, float(beta)),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-12, "disp": False},
            )
            v = res.x
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{‑p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Seed‑pool construction (augmented)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse set of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # power‑law sweep – larger exponent & shift ranges
    p_vals = np.linspace(0.1, 2.5, 20)
    shifts = np.linspace(0, 500, 12)
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # random Dirichlet families (very sparse to very dense)
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # pairwise mixtures to explore the convex hull
    base = np.arange(len(seeds))
    for _ in range(300):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(20):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final cleanup – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Stochastic mass‑transfer hill‑climb (long version)
# ----------------------------------------------------------------------
def _mass_transfer_search(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         start_frac: float = 0.02,
                         end_frac: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Long random mass‑transfer hill‑climb.
    ``start_frac`` is the initial proportion of mass moved from a source index.
    It decays linearly to ``end_frac``.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    rng = np.random.default_rng(123456789)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # linearly decaying step size
        frac = start_frac - (start_frac - end_frac) * (it / max_iters)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = frac * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


def _targeted_search(start_a: np.ndarray,
                     n: int,

    """Targeted hill‑climb that preferentially moves mass away from the active lag."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

        it += 1
        # decay step size

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)

        j = rng.integers(0, n)
        if i == j:
            continue
        delta = frac * a[i]
        if delta <= 0.0:

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:

            a = cand

            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long exponentiated‑gradient descent from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(12_000_000, remaining * 0.35 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            a = global_best_seq.copy()
            best_a = a.copy()
            best_val = _objective_exact(a)

            for it in range(1, max_iters + 1):
                # sub‑gradient of the exact max‑convolution
                g = _subgradient_maxconv(a)
                eta = 2.5 / np.sqrt(it)           # step‑size schedule
                log_a = np.log(a + 1e-30) - eta * g
                a = _softmax(log_a)

                if it % 250_000 == 0:
                    cur = _objective_exact(a)
                    if cur < best_val:
                        best_val = cur
                        best_a = a.copy()
                    print(f"[EG {it}] current best = {best_val:.6f}")

                if time.time() > deadline:
                    break

            val_eg = _objective_exact(best_a)
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = best_a.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – extensive stochastic local search (mass‑transfer + targeted)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # try to spend roughly half of the remaining time on local search
        local_time = remaining * 0.48
        lm_deadline = time.time() + local_time

        # random mass‑transfer phase
        print("\n--- Random mass‑transfer hill‑climb (long) ---")
        seq_rand, val_rand = _mass_transfer_search(
            global_best_seq,
            n,
            lm_deadline,
            max_iters=2_500_000,
            start_frac=0.025,
            end_frac=0.001,
        )
        print(f"[RandomLocal] best = {val_rand:.6f}")
        if val_rand < global_best_val:
            global_best_seq = seq_rand.copy()
            global_best_val = val_rand
            print(f"[Update] New best after random local search = {global_best_val:.6f}")

        # targeted phase (uses the active lag)
        if time.time() < deadline:
            remaining = deadline - time.time()
            lm_deadline2 = time.time() + remaining * 0.48
            print("\n--- Targeted local search (active‑lag) ---")
            seq_tgt, val_tgt = _targeted_search(
                global_best_seq,
                n,
                lm_deadline2,
                max_iters=1_600_000,
                start_frac=0.018,
                end_frac=0.001,
            )
            print(f"[TargetedLocal] best = {val_tgt:.6f}")
            if val_tgt < global_best_val:
                global_best_seq = seq_tgt.copy()
                global_best_val = val_tgt
                print(f"[Update] New best after targeted local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final high‑β polishing L‑BFGS (very tight smoothing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1200, int(min(2500, remaining * 0.04 * 1e5)))
        print("\n--- Final polishing L‑BFGS (extra‑high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000_000.0, 10_000_000_000.0,
                           50_000_000_000.0, 100_000_000_000.0,
                           500_000_000_000.0, 1_000_000_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation via the provided security‑checked function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
