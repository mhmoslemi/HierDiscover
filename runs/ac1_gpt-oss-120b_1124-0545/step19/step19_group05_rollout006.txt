def _permute_zigzag(v):
    n = len(v)
    indices = np.empty(n, dtype=int)
    left = 0
    right = n-1
    i = 0
    while left <= right:
        indices[i] = left
        i += 1
        left += 1
        if i < n:
            indices[i] = right
            i += 1
            right -= 1
    return v[indices]
```

def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1

    n = 4000

    # previous sequence
    previous = globals().get("height_sequence_1")
    prev_arr = np.asarray(previous, dtype=np.float64) if previous is not None else None

    # seed generation
    seeds = _collect_seeds_v2(n, rng, prev_arr) # new function includes new families

    # evaluate seeds
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    # keep elite
    seeds.sort(key=_objective_exact)
    elite = seeds[:min(30,len(seeds))]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # Stage 1: smooth L-BFGS continuation
    beta_sched = [1.,5.,20.,80.,200.,500.,1e3,2e3,5e3,1e4,2e4,5e4,
                  1e5,2e5,5e5,1e6,2e6,5e6,1e7,5e7,1e8,5e8,1e9,5e9,1e10,5e10,1e11]
    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline: break
        per_beta = max(800, int(min(3000, (deadline - time.time()) * 0.05 * 1e5)))
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline, beta_schedule=beta_sched,
                                          max_iter_per_beta=per_beta)
        val_opt = _objective_exact(seq_opt)
        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt

    # Stage 2: EG restarts
    remaining = deadline - time.time()
    if remaining > 0:
        restarts = 12
        iters_per_restart = int(min(800000, remaining * 0.12 / restarts))
        for r in range(restarts):
            if time.time() > deadline: break
            conc = rng.uniform(0.01, 1.5)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize(start_vec, n, deadline,
                                                    max_iters=iters_per_restart, eta0=2.5)
            if val_eg < global_best_val:
                global_best_seq, global_best_val = seq_eg.copy(), val_eg

    # Stage 3: long EG from best
    remaining = deadline - time.time()
    if remaining>0:
        max_iters = int(min(10_000_000, remaining*0.35*1e5))
        seq_eg, val_eg = _exp_gradient_optimize(global_best_seq, n, deadline,
                                                max_iters=max_iters, eta0=3.0)
        if val_eg < global_best_val:
            global_best_seq, global_best_val = seq_eg.copy(), val_eg

    # Stage 4: stochastic local search
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(300_000, remaining*50))
        seq_loc, val_loc = _local_search(global_best_seq, n, deadline,
                                         iters=iters, step_frac=0.010)
        if val_loc < global_best_val:
            global_best_seq, global_best_val = seq_loc.copy(), val_loc

    if time.time() < deadline:
        remaining = deadline - time.time()
        iters_tgt = int(min(200_000, remaining*35))
        seq_tgt, val_tgt = _targeted_local_search(global_best_seq, n, deadline,
                                            iters=iters_tgt, step_frac=0.009)
        if val_tgt < global_best_val:
            global_best_seq, global_best_val = seq_tgt.copy(), val_tgt

    # Stage 5: permutation heuristics
    if time.time() < deadline:
        # generate permutations
        perms = [global_best_seq, global_best_seq[::-1]]
        # shift permutations
        for shift in [n//4, n//2, n//3]:
            perms.append(_rotate(global_best_seq, shift))
        perms.append(_permute_zigzag(global_best_seq))
        # sorted
        perms.append(np.sort(global_best_seq)[::-1])
        # random shuffles limited
        for _ in range(5):
            perms.append(rng.permuted(global_best_seq))
        # evaluate
        for p in perms:
            val = _objective_exact(p)
            if val < global_best_val:
                global_best_seq, global_best_val = p.copy(), val

        # random swap hill climbing
        seq_perm_opt, val_perm_opt = _perm_swap_hill_climb(global_best_seq, n, deadline,
                                                          max_swaps=5000)
        if val_perm_opt < global_best_val:
            global_best_seq, global_best_val = seq_perm_opt.copy(), val_perm_opt

    # Stage 6: final polishing L-BFGS
    if time.time() < deadline:
        per_beta = max(800, int(min(2500, (deadline - time.time()) * 0.04 * 1e5)))
        seq_final = _lbfgs_smooth_optimize(global_best_seq, n, deadline,
                                           beta_schedule=[1e7,1e8,5e8,1e9,5e9,1e10],
                                           max_iter_per_beta=per_beta)
        val_final = _objective_exact(seq_final)
        if val_final < global_best_val:
            global_best_seq, global_best_val = seq_final.copy(), val_final

    # final evaluate using evaluate_sequence
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print("\nFinished: surrogate = {:.6f}, evaluate_sequence = {:.6f}".format(
            global_best_val, true_val))
    except Exception as exc:
        print("evaluate_sequence error:", exc)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

"""
Optimiser for ``evaluate_sequence`` – a non‑convex min‑max
problem on the (full) self‑convolution of a non‑negative step‑function.

Idea
----
1. **Rich seed pool** – many analytic families (uniform, triangular,
   Gaussian, a broad sweep of Beta(a,b) with a,b∈(0.1,2), power‑law,
   sin‑shapes, semi‑circular, polynomial tents, random Dirichlet,
   sparse random, low‑frequency cosine mixtures) and mixtures/rotations of
   them.  All vectors are ℓ₁‑normalised (∑a_i = 1).

2. **Continuation L‑BFGS** – minimise a log‑sum‑exp smooth surrogate of the
   max‑convolution.  The smoothing parameter β is increased along a long
   schedule (up to 1e11) while the optimiser is limited by the remaining
   wall‑time.

3. **Exponentiated‑gradient (mirror descent)** – many short EG restarts from
   Dirichlet draws, followed by a long EG run from the current best.
   EG works directly on the exact objective using the sub‑gradient of the
   active convolution lag.

4. **Stochastic mass‑transfer hill‑climbs** – a plain random transfer
   (increase a_j, decrease a_i) and a *targeted* version that moves mass
   away from the indices that dominate the maximal convolution lag.

5. **Permutation heuristics** – the objective depends heavily on the
   ordering of the same multiset of numbers.  We evaluate several
   deterministic permutations (reverse, rotation, “zig‑zag’’ interleaving,
   sorted) and a cheap random‑swap hill‑climb to discover a better ordering.

6. **Final high‑β polishing** – a last L‑BFGS run with very large β values.

All stages respect a 1000 s wall‑time budget.  The best sequence found
(ℓ₁‑normalised) is stored in the global ``height_sequence_1`` variable
and returned as a plain list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (n = len(a))."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a))) w.r.t. a (active lag only)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g = np.zeros_like(a)
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth surrogate of the exact objective.
    Returns (obj, grad) where the gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised (∑a = 1)
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through the simplex softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 800) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    ``start_vec`` must already be ℓ₁‑normalised.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3, 1e4, 2e4, 5e4,
            1e5, 2e5, 5e5, 1e6, 2e6, 5e6, 1e7,
            5e7, 1e8, 5e8, 1e9, 5e9, 1e10, 5e10,
            1e11,
        ]

    v = np.log(start_vec + 1e-30)  # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        try:
            res = _opt.minimize(
                fun=_smooth_objective,
                x0=v,
                args=(n, beta_val),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-12, "disp": False},
            )
            v = res.x
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic mass‑transfer hill‑climbs
# ----------------------------------------------------------------------
def _local_search(start_a: np.ndarray,
                  n: int,
                  deadline: float,
                  iters: int = 100_000,
                  step_frac: float = 0.012) -> tuple[np.ndarray, float]:
    """Random mass transfers, accept only improving moves."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


def _targeted_local_search(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           iters: int = 80_000,
                           step_frac: float = 0.010) -> tuple[np.ndarray, float]:
    """Hill‑climb biased toward the active convolution lag."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for _ in range(iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)

        i = rng.integers(i_start, i_end + 1)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_shape(n: int) -> np.ndarray:
    """a_i ∝ sin(π * i / (n‑1))."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_shape_pow(n: int, power: float = 2.0) -> np.ndarray:
    """sin‑shape raised to a power (>0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _semicircle_shape(n: int) -> np.ndarray:
    """Upper half‑circle on a discrete grid."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (i - c) / c
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _poly_tent(n: int, power: float = 1.5) -> np.ndarray:
    """Tent‑like shape: a_i ∝ ((i+1)*(n-i))^{power}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) * (n - i)) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


def _permute_zigzag(v: np.ndarray) -> np.ndarray:
    """Interleave extremes: 0, n‑1, 1, n‑2, 2, …"""
    n = len(v)
    order = np.empty(n, dtype=int)
    left, right = 0, n - 1
    pos = 0
    while left <= right:
        order[pos] = left
        pos += 1
        left += 1
        if pos < n:
            order[pos] = right
            pos += 1
            right -= 1
    return v[order]


def _perm_swap_hill_climb(seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_swaps: int,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Random pairwise swaps, keep improving swaps."""
    a = seq.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()

    for _ in range(max_swaps):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        cand = a.copy()
        cand[i], cand[j] = cand[j], cand[i]   # swap
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

# Seed‑pool construction (augmented)

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # deterministic base families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_gaussian(n, 0.10))
    seeds.append(_beta_shape(n, 2.0, 2.0))          # symmetric beta
    seeds.append(_beta_shape(n, 0.5, 0.5))          # arcsine‑like
    seeds.append(_beta_shape(n, 0.8, 1.2))
    seeds.append(_beta_shape(n, 1.2, 0.8))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.005))

    # new analytic families
    seeds.append(_sin_shape(n))
    seeds.append(_sin_shape_pow(n, power=2.0))
    seeds.append(_sin_shape_pow(n, power=3.0))
    seeds.append(_semicircle_shape(n))
    seeds.append(_poly_tent(n, power=1.2))
    seeds.append(_poly_tent(n, power=2.0))

    # power‑law sweep – wide exponent and shift range
    p_vals = np.linspace(0.05, 0.9, 12)
    shifts = [0.0, 1.0, 5.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # random Dirichlet families (including very sparse)
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families (different densities)
    for frac in (0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine mixtures
    for nf in (2, 3, 4, 5, 6):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures – explore convex hull
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – non‑negativity + ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

def propose_candidate(seed: int = 123,

                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed collection (including warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Evaluate seeds – keep the elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # keep a few elite seeds for heavy optimisation
    seeds.sort(key=_objective_exact)
    elite = seeds[:min(20, len(seeds))]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on smooth surrogate (aggressive)
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1e3, 2e3, 5e3, 1e4, 2e4, 5e4,
        1e5, 2e5, 5e5, 1e6, 2e6, 5e6, 1e7,
        5e7, 1e8, 5e8, 1e9, 5e9, 1e10, 5e10,
        1e11,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break
        # allocate a modest per‑beta budget that respects the remaining time
        per_beta = max(800, int(min(2500,
                                    (deadline - time.time()) * 0.07 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β≈{per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient restarts (short runs)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 12
        iters_per_restart = int(min(600_000, remaining * 0.10 / restarts))
        print("\n--- EG restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.01, 1.4)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.4,
            )
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long EG run from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(9_000_000, remaining * 0.30 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=2.8,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – stochastic local searches (mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(250_000, remaining * 45))
        if iters > 0:
            print("\n--- Random mass‑transfer local search ---")
            seq_loc, val_loc = _local_search(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.011,
            )
            print(f"[RandomLocal] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after random local search = {global_best_val:.6f}")

        remaining = deadline - time.time()
        iters_tgt = int(min(180_000, remaining * 30))
        if iters_tgt > 0:
            print("\n--- Targeted local search (active‑lag focus) ---")
            seq_tgt, val_tgt = _targeted_local_search(
                global_best_seq,
                n,
                deadline,
                iters=iters_tgt,
                step_frac=0.009,
            )
            print(f"[TargetedLocal] best = {val_tgt:.6f}")
            if val_tgt < global_best_val:
                global_best_seq = seq_tgt.copy()
                global_best_val = val_tgt
                print(f"[Update] New best after targeted local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – permutation heuristics
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation heuristics ---")
        perm_cands = []

        perm_cands.append(global_best_seq)                     # identity
        perm_cands.append(global_best_seq[::-1])               # reverse
        perm_cands.append(_rotate(global_best_seq, n // 3))   # rotate
        perm_cands.append(_rotate(global_best_seq, n // 2))   # rotate ½
        perm_cands.append(_permute_zigzag(global_best_seq))   # zig‑zag
        perm_cands.append(np.sort(global_best_seq)[::-1])     # descending sort
        perm_cands.append(np.sort(global_best_seq))           # ascending sort

        for _ in range(5):
            perm_cands.append(rng.permuted(global_best_seq))

        for p in perm_cands:
            val = _objective_exact(p)
            if val < global_best_val:
                global_best_seq = p.copy()
                global_best_val = val
                print(f"[Perm] improved to {global_best_val:.6f}")

        # cheap random‑swap hill climbing
        swaps_allowed = 5000
        seq_perm, val_perm = _perm_swap_hill_climb(
            global_best_seq,
            n,
            deadline,
            max_swaps=swaps_allowed,
            rng=rng,
        )
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[PermSwap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(2500,
                                    remaining * 0.04 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5e6, 1e7, 5e7, 1e8, 5e8, 1e9, 5e9, 1e10],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation via the provided security‑checked function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
