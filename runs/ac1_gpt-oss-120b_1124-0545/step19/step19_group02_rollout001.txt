# ----------------------------------------------------------------------
# FFT based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(seq)=1)
#   obj = 2 * n * max(conv(seq,seq))
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities – we stay on the simplex throughout optimisation
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    # protect against overflow for very extreme parameters
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add isotropic Gaussian noise, clip to ≥0 and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Mixing utilities
# ----------------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mix α·a + (1‑α)·b (already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    """Ternary search for best convex combination of a and b."""
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if best_val is None or val1 < best_val:
                best_val = val1
                best_seq = seq1
        else:
            lo = m1
            if best_val is None or val2 < best_val:
                best_val = val2
                best_seq = seq2
    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective_surrogate(best_seq)
    else:
        mid = (lo + hi) / 2.0
        cand = _mix_shapes(a, b, mid)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq = cand
            best_val = cand_val
    return best_seq, best_val

# Exponentiated‑gradient (mirror‑descent) optimiser on exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    # 5. Multi‑start exponentiated‑gradient polishing (exact max)
    # --------------------------------------------------------------
    # we will run a long EG on the current best, then several short
    # restarts starting from perturbed versions.
    if time.time() < deadline:
        print("\n--- Long EG polishing of current best ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq,
                                          n,
                                          deadline,
                                          max_iters=1_500_000,
                                          eta0=6.0)
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg
            print(f"  Updated after long EG: {best_val:.6f}")

    # additional random restarts (perturb‑then‑EG)
    num_restarts = 4
    for r in range(num_restarts):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.25  # share remaining time
        sigma = 0.003 + 0.002 * r
        pert = _perturb(best_seq, rng, sigma=sigma)
        print(f"\n--- Restart {r+1}: perturb σ={sigma:.4f} ---")
        seq_r, val_r = _exp_gradient_opt(pert,
                                         n,
                                         sub_deadline,
                                         max_iters=300_000,
                                         eta0=5.5)
        if val_r < best_val:
            best_seq, best_val = seq_r, val_r
            print(f"  Restart {r+1} improved best: {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. (Optional) final short L‑BFGS smoothing to polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        try:
            import scipy.optimize as _opt_sc
        except Exception:  # pragma: no cover
            _opt_sc = None

        if _opt_sc is not None:
            # reuse the smoother from the previous script (log‑sum‑exp)
            def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                                       n: int,
                                       deadline: float,
                                       beta_schedule=None,
                                       max_iter_per_beta: int = 6000):
                if beta_schedule is None:
                    beta_schedule = [1.0, 5.0, 20.0, 80.0,
                                     200.0, 500.0, 1000.0, 2000.0,
                                     5000.0, 20000.0, 50000.0,
                                     100000.0, 200000.0, 500000.0,
                                     1000000.0]

                v = np.log(start_vec + 1e-30)

                for beta in beta_schedule:
                    if time.time() > deadline:
                        break
                    beta_val = float(beta)

                    def fun(v_local):
                        a = _softmax(v_local)
                        conv = _conv_fft(a, a)
                        max_c = np.max(conv)
                        shift = max_c
                        exp_beta = np.exp(beta_val * (conv - shift))
                        sum_exp = exp_beta.sum()
                        obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

                        w = exp_beta / sum_exp
                        rev_a = a[::-1]
                        full = _conv_fft(rev_a, w)
                        grad_a = 2.0 * full[n - 1:n - 1 + n]

                        dot = np.dot(grad_a, a)
                        grad_v = a * (grad_a - dot)
                        return obj, grad_v

                    def callback(_):
                        if time.time() > deadline:
                            raise KeyboardInterrupt()

                    try:
                        res = _opt_sc.minimize(fun=fun,
                                               x0=v,
                                               method='L-BFGS-B',
                                               jac=True,
                                               callback=callback,
                                               options={'maxiter': max_iter_per_beta,
                                                        'disp': False,
                                                        'ftol': 1e-12})
                        v = res.x
                    except KeyboardInterrupt:
                        break
                    except Exception:  # pragma: no cover
                        break
                return _softmax(v)

            print("\n--- Final L‑BFGS polishing ---")
            seq_lbfgs_final = _lbfgs_smooth_optimize(best_seq,
                                                     n,
                                                     deadline,
                                                     max_iter_per_beta=5000)
            val_lbfgs_final = _objective_surrogate(seq_lbfgs_final)
            if val_lbfgs_final < best_val:
                best_seq, best_val = seq_lbfgs_final, val_lbfgs_final
                print(f"  L‑BFGS improved best to {best_val:.6f}")
        else:
            print("\nSciPy not available – skipping L‑BFGS polishing.")

    # --------------------------------------------------------------
    # 7. Compute the true evaluation (the costly part)
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
