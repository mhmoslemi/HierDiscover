```
def _project_simplex(v):
    # v can be any real vector
    n = v.shape[0]
    u = np.sort(v)[::-1] # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Rich seed generation** – Build a large collection of ℓ₁‑normalised
   vectors using many analytic families (uniform, triangular, power‑law,
   inverse‑power (U‑shaped), exponential, beta‑distribution, Dirichlet
   draws, sparse masks, low‑frequency cosine sums, and mixed/rotated
   variants).  The pool contains a few thousand candidates.

2. **Elite selection** – Evaluate the exact objective (2·n·max(conv)/∑²)
   for all seeds, keep the top‐k (default 15) as elite starting points.

3. **Projected sub‑gradient descent** – For every elite vector run a
   simple mirror‑descent on the simplex:
   * sub‑gradient of the active convolution lag,
   * step size ηₜ = η₀/√t,
   * Euclidean projection onto the probability simplex.
   The method is cheap, monotonic (accepts only improving steps) and
   quickly reduces the exact max‑convolution value.

4. **Aggressive stochastic local search** – Starting from the best
   vector found, perform many random “mass‑transfer’’ moves:
   pick indices i≠j, move a small fraction of a[i] to a[j] and keep the
   change if the true objective improves.  The number of iterations is
   scaled to the remaining wall‑time.

5. **Final evaluation** – Return the best sequence found; store it in the
   global ``height_sequence_1`` for warm‑starts in future runs.

All steps respect the 1000 s budget, use only NumPy and the provided
``evaluate_sequence`` function, and keep every entry non‑negative.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_exact(a: np.ndarray) -> float:
    """Exact evaluation of the target objective for a (assumed ℓ₁‑normalised)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Returned vector has the same length as a.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))               # active lag
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (always returned normalised to sum=1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay_rev(n: int, lam: float) -> np.ndarray:
    """Mirror of _exponential_decay."""
    return _exponential_decay(n, lam)[::-1]


def _power_law(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+shift+1)^{-p}  (decreasing shape)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inv_power_law(n: int, p: float) -> np.ndarray:
    """
    U‑shaped distribution:
    a_i ∝ (i+1)^{-p} + (n-i)^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse list of ℓ₁‑normalised vectors of length n."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_power_law(n, p=0.5, shift=0.0))
    seeds.append(_inv_power_law(n, p=0.8))
    seeds.append(_exponential_decay(n, lam=0.02))
    seeds.append(_exponential_decay_rev(n, lam=0.02))

    # sweep of power‑law (decreasing)
    for p in np.linspace(0.2, 1.0, 7):
        for sh in (0.0, 5.0, 20.0):
            seeds.append(_power_law(n, p, shift=sh))

    # sweep of inverse‑power (U‑shape)
    for p in np.linspace(0.3, 1.4, 9):
        seeds.append(_inv_power_law(n, p))

    # exponential decays, finer λ sweep
    for lam in np.logspace(-4, -0.3, 9):
        seeds.append(_exponential_decay(n, lam))
        seeds.append(_exponential_decay_rev(n, lam))

    # beta‑distribution family (wide log‑uniform grid)
    for _ in range(120):
        alpha = 10 ** rng.uniform(-2, 1)      # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, alpha, beta))

    # random Dirichlet draws (different concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(20):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for frac in (0.01, 0.02, 0.05, 0.1):
        for _ in range(8):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine constructions
    for nf in (2, 3, 4, 5):
        for _ in range(10):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # rotations & mixes of existing seeds – increase diversity
    base = np.arange(len(seeds))
    for _ in range(250):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    for _ in range(20):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (warm‑start)
    if previous is not None and previous.shape == (n,) and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce non‑negativity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        a = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
            s = a.sum()
        cleaned.append(a / s)
    return cleaned

# Optimisation primitives

def _pgd_maxconv(start_a: np.ndarray,
                 deadline: float,
                 max_iters: int = 20000,
                 eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses simplex projection after each step and accepts only improving moves.
    """
    a = start_a.copy()
    cur_val = _objective_exact(a)
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        eta = eta0 / np.sqrt(it)
        g = _subgradient_maxconv(a)
        a_candidate = _project_simplex(a - eta * g)
        cand_val = _objective_exact(a_candidate)

        if cand_val < cur_val:
            a = a_candidate
            cur_val = cand_val

        if cand_val < best_val:
            best_val = cand_val
            best_a = a_candidate.copy()

    return best_a, best_val


def _local_search(start_a: np.ndarray,
                  n: int,
                  deadline: float,
                  iters: int = 300000,
                  step_frac: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: transfer a small fraction of mass from i to j.
    Accepts only improvements.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val
