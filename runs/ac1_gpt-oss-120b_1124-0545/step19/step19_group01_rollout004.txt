def _smooth_loss_and_grad(a, beta):
    n = len(a)
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    # compute softmax weights w
    # to avoid overflow, shift conv by max.
    exp_terms = np.exp(beta * (conv - shift))
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp  # length 2n-1
    # gradient w.r.t a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)  # length len(rev_a)+len(w)-1 = n+2n-2 = 3n-2
    grad_a = 2.0 * full[n-1:n-1+n]  # slice
    # compute gradient w.r.t v
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    loss = 2.0 * n * (shift + np.log(sum_exp) / beta)
    return loss, grad_v

"""
Propose a low‑autocorrelation step function.

The algorithm proceeds in several stages:

1. **Rich seed pool** – deterministic families (uniform, triangular,
   sine‑window, Gaussian, power‑law, exponential‑decay, beta‑shape,
   Dirichlet draws, low‑frequency cosine curves) and many random
   permutations of these shapes.

2. **Quick surrogate screening** – each seed is normalised (ℓ₁‑norm = 1)
   and its surrogate objective `2·n·max(conv(seq,seq))` is evaluated.
   The best few (default 15) are kept.

3. **Refinement by Adam on a smooth max‑convolution** – for each promising
   seed we run a mirror‑descent (Adam) optimiser in log‑space on the
   smooth approximation `log‑sum‑exp` of the convolution maximum.
   The smoothing parameter β is increased through a schedule
   `[1,5,20,…,1e6]`.  The best sequence seen is retained.

4. **Final polishing** – a long Adam run on the current global best
   (more iterations per β) to push the objective a little further.

5. **Official evaluation** – the resulting sequence is fed to the supplied
   `evaluate_sequence` function, stored in the global variable
   ``height_sequence_1`` and returned.

The whole search respects a 1000 s budget, constantly checking the time
and keeping the best sequence discovered so far.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector, sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine components (clipped to non‑negative)."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smooth max‑convolution loss (log‑sum‑exp) and its gradient in log‑space
# ----------------------------------------------------------------------
def _smooth_loss_and_grad(a: np.ndarray, beta: float):
    """
    Returns (loss, grad_v) where:
      a      = softmax(v) (ℓ₁‑normalised)
      loss   = 2·n·(max(conv) + log‑sum‑exp(beta·(conv-max))/beta)
      grad_v = gradient w.r.t. v (log‑space).
    """
    n = len(a)
    conv = _conv_fft(a, a)                        # length 2n‑1
    shift = np.max(conv)                          # for numerical stability
    # softmax weights for the smooth max
    exp_terms = np.exp(beta * (conv - shift))
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp                       # length 2n‑1, sums to 1

    # gradient w.r.t. a (see derivation in comments)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                    # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]          # slice of length n
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)                   # chain rule through softmax

    loss = 2.0 * n * (shift + np.log(sum_exp) / beta)
    return loss, grad_v


# ----------------------------------------------------------------------
# Adam optimiser for the smooth surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   schedule=None,
                   steps_per_beta: int = 1000,
                   lr0: float = 0.3):
    """
    Mirror‑descent in log‑space using Adam on the smooth max‑convolution.
    Returns (best_seq, best_surrogate_value) found before the deadline.
    """
    if schedule is None:
        schedule = [1, 5, 20, 80, 200, 500, 1000,
                    2000, 5000, 20000, 50000, 100000,
                    200000, 500000, 1000000]

    n = len(a0)
    v = np.log(a0 + 1e-30)               # log‑probability vector
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = a0.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in schedule:
        if time.time() > deadline:
            break
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(v)
            loss, grad_v = _smooth_loss_and_grad(a, beta)

            t += 1
            m = b1 * m + (1.0 - b1) * grad_v
            v2 = b2 * v2 + (1.0 - b2) * (grad_v ** 2)

            m_hat = m / (1.0 - b1 ** t)
            v_hat = v2 / (1.0 - b2 ** t)

            # modest learning‑rate decay with β (larger β → more peaked loss)
            lr = lr0 / (beta ** 0.25 + 1e-12)

            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep the best according to the exact (non‑smooth) surrogate
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The best sequence found (under the time budget) is returned and saved in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem dimension – 4000 offers a good compromise.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous solution (if any) for warm‑starting.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        arr = np.maximum(arr, 0.0)
        if arr.sum() > 0.0:
            prev_arr = arr / arr.sum()

    # --------------------------------------------------------------
    # 1. Massive seed generation
    # --------------------------------------------------------------
    seeds = []

    # Deterministic families (quite dense sweeps)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.2, 4.0, 30):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma_frac in np.linspace(0.05, 0.35, 30):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.0, 5.0, 25):
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.2, 50):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Random beta‑shape draws
    for _ in range(120):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws (wide range of concentrations)
    for conc in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(10):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(40):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(10):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform & exponential (different alphas)
    for alpha in (0.2, 0.5, 0.8):
        for rate in (0.001, 0.005, 0.02):
            curve = _exp_decay(n, rate)
            seeds.append(_mix_shapes(_uniform(n), curve, alpha))

    # Random permutations of a few promising bases
    base_shapes = [_exp_decay(n, 0.0035), _sym_exp_decay(n, 0.0045),
                   _beta_shape(n, 2.5, 2.5), _gaussian(n, 0.18)]
    for base in base_shapes:
        for _ in range(6):
            seeds.append(rng.permutation(base))

    # Include the previous best and perturbed copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(8):
            seeds.append(_perturb(prev_arr, rng, sigma=0.005))

    # --------------------------------------------------------------
    # Normalise and cheap surrogate evaluation of all seeds
    # --------------------------------------------------------------
    normalized_seeds = []
    for s in seeds:
        s = np.maximum(s, 0.0)
        total = s.sum()
        if total <= 0.0:
            continue
        normalized_seeds.append(s / total)

    evaluated = []
    for vec in normalized_seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(15, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds after quick surrogate screening ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 2. Initialise global best
    # --------------------------------------------------------------
    if top_seeds:
        global_best_seq = top_seeds[0].copy()
        global_best_val = _objective_surrogate(global_best_seq)
    else:
        global_best_seq = _uniform(n)
        global_best_val = _objective_surrogate(global_best_seq)

    print(f"\n[Init] Global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Multi‑start refinement (Adam on smooth max‑convolution)
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate ~12 % of the remaining budget to each remaining seed
        remaining = deadline - time.time()
        remaining_seeds = len(top_seeds) - idx
        share = remaining * 0.12 / max(1, remaining_seeds)
        sub_deadline = time.time() + share

        print(f"\n[Seed {idx+1}] Refinement budget ≈ {share:.1f}s")

        seq_opt, val_opt = _adam_optimize(seed_vec,
                                          sub_deadline,
                                          schedule=[1, 5, 20, 80, 200, 500, 1000,
                                                    2000, 5000, 20000, 50000,
                                                    100000, 200000, 500000,
                                                    1000000],
                                          steps_per_beta=800,
                                          lr0=0.3)

        # The optimiser already returns the best surrogate value it saw
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"   *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final long polishing (more iterations per β)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (extended Adam run) ---")
        final_seq, final_val = _adam_optimize(global_best_seq,
                                              deadline,
                                              schedule=[1, 5, 20, 80, 200, 500, 1000,
                                                        2000, 5000, 20000, 50000,
                                                        100000, 200000, 500000,
                                                        1000000],
                                              steps_per_beta=2500,
                                              lr0=0.3)
        if final_val < global_best_val:
            global_best_val = final_val
            global_best_seq = final_seq.copy()
            print(f"[Final] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
