"""
Propose a step‑function that minimises ``evaluate_sequence``.
----------------------------------------------------------------
The algorithm proceeds in two stages:

1. **Diverse seeding** – a collection of analytically‑defined families
   (uniform, linear / quadratic decay, exponentials, beta‑shapes and random
   Dirichlet draws) is generated, normalised to ℓ₁‑sum = 1 and evaluated with
   the exact objective.  The best seed provides a good starting point.

2. **Incremental simulated annealing** – the current best vector is refined
   by repeatedly moving a small amount of mass from a (weighted) source index
   to a random destination index.  The convolution is updated *in‑place* using
   the exact algebraic effect of the move (O(n) per step) and the new
   objective (2·n·max(conv)) is computed.  A linear cooling schedule
   (temperature → 0) allows occasional uphill moves early on; the step size
   also decays, enabling fine‑grained final polishing.  Periodically the
   full convolution is recomputed to eliminate accumulated round‑off error.

The whole search respects the 1000 s wall‑time limit; the best sequence found
is stored in the global ``height_sequence_1`` variable and returned as a plain
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    n = len(a)
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a)
    max_conv = np.max(conv)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decrease(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _quadratic_decrease(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** 2
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a modest but diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decrease(n))
    seeds.append(_quadratic_decrease(n))

    # exponential decay (both directions) – a few λ values
    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        seeds.append(_exponential_decay(n, lam))
        seeds.append(_exponential_decay(n, lam)[::-1])

    # beta‑shape sweep
    for _ in range(30):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta  = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, alpha, beta))

    # random Dirichlet draws with varying concentration
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Incremental simulated annealing optimiser
# ----------------------------------------------------------------------
def _anneal_optimize(start_a: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Refine ``start_a`` by mass‑transfer moves while updating the convolution
    incrementally.  Returns the best vector seen.
    """
    n = len(start_a)
    a = start_a.copy()                     # ℓ₁‑normalised
    conv = _conv_fft(a)                    # length = 2n‑1
    factor = 2.0 * n / (a.sum() ** 2)      # constant ≈ 2·n when sum≈1
    cur_val = factor * np.max(conv)

    best_a = a.copy()
    best_val = cur_val

    # annealing schedule parameters
    total_time = deadline - time.time()
    step_start, step_end = 0.05, 0.001          # move fraction
    temp_start, temp_end = 0.02, 1e-6           # temperature (objective units)

    it = 0
    recompute_interval = 50000                 # refresh full convolution

    while time.time() < deadline:
        it += 1
        # linear cooling for temperature and step size
        remaining = deadline - time.time()
        frac = max(0.0, remaining / total_time)
        step = step_end + (step_start - step_end) * frac
        temp = temp_end + (temp_start - temp_end) * frac

        # choose source index weighted by current mass, destination uniformly
        i = rng.choice(n, p=a)                 # heavier entries move more often
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if i == j:
            continue

        # propose a fraction of the source mass
        delta = step * a[i] * rng.random()
        if delta <= 0.0:
            continue
        d = float(delta)

        # store old values for possible revert
        ai_old = a[i]
        aj_old = a[j]

        # --- update convolution using the analytical formula ---
        # slice updates (each creates a temporary O(n) array)
        conv[i:i + n] -= 2.0 * d * a
        conv[j:j + n] += 2.0 * d * a
        conv[2 * i]     += d * d
        conv[2 * j]     += d * d
        conv[i + j]     -= 2.0 * d * d

        # update the vector itself
        a[i] = ai_old - d
        a[j] = aj_old + d

        # compute new objective
        new_val = factor * np.max(conv)

        # acceptance rule (greedy + occasional uphill)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp((cur_val - new_val) / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert both vector and convolution
            a[i] = ai_old
            a[j] = aj_old
            conv[i:i + n] += 2.0 * d * a
            conv[j:j + n] -= 2.0 * d * a
            conv[2 * i]     -= d * d
            conv[2 * j]     -= d * d
            conv[i + j]     += 2.0 * d * d

        # occasional full recomputation to curb drift
        if it % recompute_interval == 0:
            a = a / a.sum()
            conv = _conv_fft(a)
            cur_val = factor * np.max(conv)

        # lightweight progress reports
        if it % 200000 == 0:
            elapsed = time.time() - (deadline - total_time)
            print(f"[Anneal] it={it:,} best={best_val:.6f} cur={cur_val:.6f} "
                  f"time={elapsed:.1f}s")

    return best_a
