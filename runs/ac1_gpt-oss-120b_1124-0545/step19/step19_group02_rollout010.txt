def random_transfer(seq, rng, max_step):
    i = rng.integers(len(seq))
    j = rng.integers(len(seq))
    if i == j:
        return seq, 0, i, j
    eps = max_step * rng.random()
    # ensure eps <= seq[i], else limit.
    eps = min(eps, seq[i])
    if eps == 0:
        return seq, 0, i, j
    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    # maintain normalization automatically.
    return new_seq, eps, i, j
```

def _adam_optimize(start_seq, n, deadline, max_iters=500000):
    a = start_seq.copy()
    logits = np.log(a + 1e-30)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    best_seq = a.copy()
    best_val = _objective_surrogate(a) # using surrogate
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # compute subgradient at any max index
        idx = np.argmax(conv)
        # compute gradient vector g_a
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = idx - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]
        # gradient wrt logits
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        # adam update
        m = beta1 * m + (1 - beta1) * grad_logits
        v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        logits -= step
        # recompute a
        # to avoid overflow
        max_logit = np.max(logits)
        a = np.exp(logits - max_logit)
        a_sum = a.sum()
        a = a / a_sum

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # optional: occasionally apply random noise to escape local minima
        if it % 50000 == 0:
            # maybe reset momentum
            m[:] = 0
            v[:] = 0
            # maybe add jitter
            a = _perturb(a, rng, sigma=0.001)
            logits = np.log(a + 1e-30)
    return best_seq, best_val
```

    a = np.maximum(a, eps)
    a = a / a.sum()
    logits = np.log(a + 1e-30)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.5
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while True:
        if time.time() > deadline or it >= max_iters:
            break
        it += 1
        # compute conv
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # subgradient at argmax
        k = int(np.argmax(conv))
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        m = beta1 * m + (1 - beta1) * grad_logits
        v = beta2 * v + (1 - beta2) * (grad_logits * grad_logits)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        logits -= step

        # recompute a via stable softmax
        max_log = np.max(logits)
        exp_vals = np.exp(logits - max_log)
        a = exp_vals / exp_vals.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 50000 == 0:
            # optional jitter to escape stagnation
            # basically reset momentum
            m.fill(0)
            v.fill(0)
            # small random perturbation
            a = _perturb(a, rng, sigma=0.001)
            logits = np.log(a + 1e-30)
    return best_seq, best_val

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in several stages:

1. **Rich seed generation** – a broad collection of analytic windows,
   exponential / logistic decays, power‑law tails, Dirichlet samples,
   low‑frequency random curves and a few simple ramps.  All seeds are
   ℓ₁‑normalised (sum = 1).

2. **Fast surrogate evaluation** – for ℓ₁‑normalised vectors the denominator
   is constant, so the objective reduces to ``2·n·max(conv)``.  Convolution
   is performed by a real FFT for speed.

3. **Coarse selection** – the best few seeds (by surrogate value) are kept.

4. **Continuation L‑BFGS** – a smooth approximation of the max (log‑sum‑exp)
   is minimised in log‑softmax space using L‑BFGS‑B with a long schedule of
   increasing smoothness parameters.

5. **Adam‑based mirror‑descent** – a sub‑gradient of the exact (non‑smoothed)
   max‑convolution is taken at the active offset and an Adam optimiser
   updates the log‑probabilities.  This often pushes the solution a tiny
   amount further than plain exponentiated‑gradient descent.

6. **Final mixing & polishing** – the best result is mixed with its reversal
   (a cheap convex‑combination search) and a short exponentiated‑gradient
   polish.

The whole procedure respects a global time budget (default 1000 s).  The
best sequence found is stored in ``height_sequence_1`` for reuse in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _ramp(n: int) -> np.ndarray:
    """Linear decreasing ramp: a[i] ∝ (n-i)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _reverse_ramp(n: int) -> np.ndarray:
    """Linear increasing ramp (mirror of _ramp)."""
    i = np.arange(n, dtype=np.float64)
    a = i + 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

# Mixing utilities
# ----------------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    """Ternary search for the best convex combination of a and b."""
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if best_val is None or val1 < best_val:
                best_val = val1
                best_seq = seq1
        else:
            lo = m1
            if best_val is None or val2 < best_val:
                best_val = val2
                best_seq = seq2
    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective_surrogate(best_seq)
    else:
        mid = (lo + hi) / 2.0
        cand = _mix_shapes(a, b, mid)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq = cand
            best_val = cand_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space so we stay on the probability simplex.
    """
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑based mirror‑descent optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_refine(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 rng: np.random.Generator,
                 max_iters: int = 2_000_000,
                 eta0: float = 0.5,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps_adam: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam mirror‑descent on the true max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    # initialise
    a = np.maximum(start_seq, 1e-30)
    a = a / a.sum()
    logits = np.log(a + 1e-30)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while True:
        if time.time() > deadline or it >= max_iters:
            break
        it += 1

        conv = _conv_fft(a, a)
        max_c = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t logits (mirror‑descent)
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits * grad_logits)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # stable softmax
        max_log = np.max(logits)
        exp_vals = np.exp(logits - max_log)
        a = exp_vals / exp_vals.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional jitter to avoid stagnation
        if it % 100_000 == 0:
            m.fill(0.0)
            v.fill(0.0)
            a = _perturb(a, rng, sigma=0.001)
            logits = np.log(a + 1e-30)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a non‑negative step‑function that minimises

    and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so we can still call evaluate_sequence at the end
    deadline = start_time + budget_s - 8.0

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # 1. Seed generation (rich family)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic families (coarse grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_ramp(n))
    seeds.append(_reverse_ramp(n))

    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 14):
        seeds.append(_gaussian(n, sigma))
    for sigma in (0.10, 0.12):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))
    for p in np.linspace(0.0, 2.0, 13):
        seeds.append(_power(n, p))

    # Beta‑shape grid
    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    for alpha in np.linspace(0.1, 0.9, 7):
        seeds.append(_tukey_window(n, alpha))

    # Exponential / logistic families (new)
    for gamma in np.linspace(0.1, 8.0, 16):
        seeds.append(_exp_decay(n, gamma))
    for steep in np.linspace(0.5, 30.0, 10):
        for shift in np.linspace(0.1, 0.9, 9):
            seeds.append(_logistic_decay(n, steep, shift))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Alternating Dirichlet
    seeds.append(_alternating_dirichlet(n, rng))

    # Perturbations of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(4):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"\nTop {top_k} surrogate values among seeds:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Pairwise mixing of top seeds (convex combinations)
    # ------------------------------------------------------------------
    print("\n--- Pairwise mixing of top seeds ---")
    mix_deadline = deadline - 2.0
    mixture_candidates = []

    for i in range(len(top_seeds)):
        for j in range(i + 1, len(top_seeds)):
            if time.time() > mix_deadline:
                break
            seq_mix, val_mix = _best_mixed_pair(top_seeds[i],
                                                top_seeds[j],
                                                mix_deadline,
                                                steps=25)
            mixture_candidates.append((seq_mix, val_mix))

    # Combine raw top seeds and mixtures
    all_candidates = evaluated[:top_k] + mixture_candidates
    all_candidates.sort(key=lambda x: x[1])
    best_seq = all_candidates[0][0].copy()
    best_val = all_candidates[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (L‑BFGS smoothing)
    # ------------------------------------------------------------------
    refined_candidates = []
    for idx, (seed_vec, _) in enumerate(all_candidates[:4]):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.45 / (4 - idx)

        print(f"\n[Refine L‑BFGS #{idx+1}] starting from seed (surrogate {_objective_surrogate(seed_vec):.6f})")
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec,
                                           n,
                                           sub_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1000.0,
                                                          2000.0, 5000.0,
                                                          20000.0, 50000.0,
                                                          100000.0, 200000.0,
                                                          500000.0, 1000000.0],
                                           max_iter_per_beta=8000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        # Store candidate
        refined_candidates.append((seq_lbfgs, val_lbfgs))

        if val_lbfgs < best_val:
            best_seq = seq_lbfgs
            best_val = val_lbfgs
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final Adam refinement on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Adam refinement (exact max) ---")
        seq_adam, val_adam = _adam_refine(best_seq,
                                          n,
                                          deadline,
                                          rng,
                                          max_iters=5_000_000,
                                          eta0=0.7)   # slightly larger step size
        if val_adam < best_val:
            best_seq = seq_adam
            best_val = val_adam
            print(f"[Final Adam] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Mixing with reversal (sometimes reduces the peak further)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rev_seq = best_seq[::-1]
        mix_deadline = deadline - 1.0
        seq_mix, val_mix = _best_mixed_pair(best_seq, rev_seq,
                                            mix_deadline,
                                            steps=30)
        if val_mix < best_val:
            best_seq = seq_mix
            best_val = val_mix
            print(f"[Mix with reversal] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
