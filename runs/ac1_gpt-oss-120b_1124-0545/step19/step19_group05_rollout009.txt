"""
Propose a step‑function that minimises ``evaluate_sequence``.

The optimiser works in several stages:

1. **Exploratory grid search (small n)** –  A quick sweep over several
   parametric families (Beta, exponential, power‑law, sinᵖ, Gaussian,
   polynomial tent) on a modest length (n≈800).  The best few parameter
   settings (by the exact objective) are kept and later re‑instantiated
   at the final problem size.

2. **Seed pool construction** – Deterministic families, random
   Dirichlet draws, low‑frequency curves, mixtures, rotations and the
   top candidates from the grid search (re‑scaled to the final length).

3. **Continuation L‑BFGS on a smooth log‑sum‑exp surrogate** –  Aggressive
   continuation with a long β‑schedule and many iterations per β.

4. **Exponentiated‑gradient (mirror descent) restarts** –  Exact sub‑gradient
   of the max‑convolution is used; many short restarts are performed
   (large step‑size constant) to explore the landscape.

5. **Stochastic local search** –  Two complementary hill‑climbs:
    * random mass‑transfer,
    * targeted moves that withdraw mass from the active convolution lag.

6. **Simulated‑annealing polishing** –  Small random perturbations are
   accepted with a temperature schedule, allowing the optimiser to
   escape shallow local minima.

7. **Final high‑β L‑BFGS polishing** –  Very large β values tighten the
   smooth approximation to the true max‑convolution.

All stages respect the overall 1000 s wall‑time budget.  The best
sequence found (ℓ₁‑normalised, i.e. ``sum == 1``) is stored in the global
variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (∑a assumed >0)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (active lag only)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g = np.zeros_like(a)
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of the exact objective.

    Returns (obj, grad) where the gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                 # ℓ₁‑normalised (sum = 1)
    conv = _conv_fft(a, a)          # length 2n‑1
    shift = np.max(conv)            # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp          # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)      # length 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through the simplex softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    ``start_vec`` must already be ℓ₁‑normalised.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0, 50_000_000.0, 100_000_000.0,
            500_000_000.0, 1_000_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)  # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(
                fun=_smooth_objective,
                x0=v,
                args=(n, float(beta)),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_beta,
                         "ftol": 1e-12,
                         "disp": False},
            )
            v = res.x
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val

# Stochastic local searches
# ----------------------------------------------------------------------
def _random_mass_transfer(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         iters: int = 150_000,
                         step_frac: float = 0.012) -> tuple[np.ndarray, float]:
    """Plain random mass transfer hill‑climb (accept only improving)."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(42)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


def _targeted_mass_transfer(start_a: np.ndarray,
                            n: int,
                            deadline: float,
                            iters: int = 120_000,
                            step_frac: float = 0.010) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass away from the
    currently active (maximal) convolution lag.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(314159)

    for _ in range(iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i = rng.integers(i_start, i_end + 1)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         iters: int = 200_000,
                         init_temp: float = 1e-3,
                         final_temp: float = 1e-6,
                         step_frac: float = 0.008) -> tuple[np.ndarray, float]:
    """
    Small random perturbations accepted with a temperature schedule.
    """
    a = start_a.copy()
    cur_val = _objective_exact(a)
    best_val = cur_val
    best_a = a.copy()
    rng = np.random.default_rng(20230621)

    # exponential temperature decay
    log_T0 = np.log(init_temp)
    log_T1 = np.log(final_temp)
    for t in range(1, iters + 1):
        if time.time() > deadline:
            break
        T = np.exp(log_T0 + (log_T1 - log_T0) * (t - 1) / (iters - 1))

        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = step_frac * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()  # re‑normalise to stay on the simplex

        cand_val = _objective_exact(cand)
        dE = cand_val - cur_val
        if dE < 0 or rng.random() < np.exp(-dE / T):
            a = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cand.copy()
    return best_a, best_val

# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    return a / a.sum()


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    return a / a.sum()


def _sin_power_shape(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** power
    return a / a.sum()


def _semicircle_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (i - c) / c
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, 1.0))
    return a / a.sum()


def _poly_tent(n: int, power: float = 1.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) * (n - i)) ** power
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)

# Small‑n grid search (discover promising parameter settings)
# ----------------------------------------------------------------------
def _grid_search_small_n(n_small: int,
                         top_k: int = 12,
                         deadline: float | None = None):
    """
    Perform a lightweight sweep over several parametric families on a
    small problem size.  The best ``top_k`` candidates (by the exact objective)
    are returned as tuples ``(generator, params)`` that can later be instantiated
    at any larger ``n``.
    """
    candidates = []

    # ----- Beta family -------------------------------------------------
    alphas = np.linspace(0.3, 5.0, 12)
    betas = np.linspace(0.3, 5.0, 12)
    for a in alphas:
        for b in betas:
            seq = _beta_shape(n_small, float(a), float(b))
            val = _objective_exact(seq)
            candidates.append( ( _beta_shape, (float(a), float(b)), val) )

    # ----- Exponential decay -------------------------------------------
    lams = np.linspace(0.0005, 0.03, 10)
    for lam in lams:
        seq = _exponential_decay(n_small, float(lam))
        val = _objective_exact(seq)
        candidates.append( ( _exponential_decay, (float(lam),), val) )

    # ----- Power‑law ----------------------------------------------------
    p_vals = np.linspace(0.2, 1.3, 10)
    shifts = [0.0, 2.0, 5.0, 15.0, 30.0, 60.0]
    for p in p_vals:
        for sh in shifts:
            seq = _power_law(n_small, float(p), float(sh))
            val = _objective_exact(seq)
            candidates.append( ( _power_law, (float(p), float(sh)), val) )

    # ----- Sin‑power ----------------------------------------------------
    powers = np.linspace(0.6, 4.0, 10)
    for pw in powers:
        seq = _sin_power_shape(n_small, float(pw))
        val = _objective_exact(seq)
        candidates.append( ( _sin_power_shape, (float(pw),), val) )

    # ----- Gaussian ------------------------------------------------------
    sigmas = [0.07, 0.12, 0.18, 0.25]    # fraction of n
    for sf in sigmas:
        seq = _gaussian(n_small, float(sf))
        val = _objective_exact(seq)
        candidates.append( ( _gaussian, (float(sf),), val) )

    # ----- Polynomial tent -----------------------------------------------
    powers = [0.8, 1.2, 1.6, 2.2]
    for pw in powers:
        seq = _poly_tent(n_small, float(pw))
        val = _objective_exact(seq)
        candidates.append( ( _poly_tent, (float(pw),), val) )

    # sort and keep top_k
    candidates.sort(key=lambda x: x[2])
    best = candidates[:top_k]

    # Return only generator + param tuples (no values)
    return [(gen, params) for (gen, params, _) in best]

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_exponential_decay(n, 0.001))

    # newly added analytic families
    seeds.append(_sin_shape(n))
    seeds.append(_semicircle_shape(n))
    seeds.append(_poly_tent(n, power=1.2))
    seeds.append(_poly_tent(n, power=2.0))

    # power‑law sweep – extended range
    p_vals = np.linspace(0.2, 0.9, 12)
    shifts = [0.0, 5.0, 20.0, 50.0, 100.0, 200.0]
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # Random Dirichlet draws (sparse to dense)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse vectors
    for frac in (0.02, 0.04, 0.08, 0.15):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixtures
    base = np.arange(len(seeds))
    for _ in range(150):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], float(alpha)))

    # Rotations of promising seeds
    for _ in range(10):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], int(shift)))

    # ---------- Grid‑search warm‑start (small‑n -> large‑n) ----------
    small_n = max(200, n // 5)  # about 800 for n=4000
    deadline = time.time() + 5.0  # quick cut‑off for the sweep
    grid_candidates = _grid_search_small_n(small_n, top_k=10, deadline=deadline)
    for gen, params in grid_candidates:
        try:
            # instantiate at the final size
            seq = gen(n, *params)
            seeds.append(seq)
        except Exception:   # pragma: no cover
            continue

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up: non‑negativity + ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main entry point

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between fidelity and speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed collection (including warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors (including grid‑search).")

    # ------------------------------------------------------------------
    # Evaluate seeds, retain elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # keep the top‑k seeds for heavy optimisation
    seeds.sort(key=_objective_exact)
    elite = seeds[:min(35, len(seeds))]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on smooth surrogate (aggressive)
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 50_000_000.0, 100_000_000.0,
        500_000_000.0, 1_000_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break
        # allocate a slice of the remaining budget (≈7 % per run)
        per_beta = max(1000, int(min(3000,
                        (deadline - time.time()) * 0.07 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β≈{per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient restarts (many short runs)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 20
        iters_per_restart = int(min(500_000, remaining * 0.10 / restarts))
        print("\n--- EG restarts (short runs) ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.01, 1.4)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.3,
            )
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long EG run from current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(12_000_000, remaining * 0.30 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=2.7,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – stochastic local searches
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(300_000, remaining * 45))
        if iters > 0:
            print("\n--- Random mass‑transfer local search ---")
            seq_loc, val_loc = _random_mass_transfer(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.011,
            )
            print(f"[RandomLocal] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after random local search = {global_best_val:.6f}")

        remaining = deadline - time.time()
        iters_tgt = int(min(180_000, remaining * 30))
        if iters_tgt > 0:
            print("\n--- Targeted local search (active‑lag focus) ---")
            seq_tgt, val_tgt = _targeted_mass_transfer(
                global_best_seq,
                n,
                deadline,
                iters=iters_tgt,
                step_frac=0.010,
            )
            print(f"[TargetedLocal] best = {val_tgt:.6f}")
            if val_tgt < global_best_val:
                global_best_seq = seq_tgt.copy()
                global_best_val = val_tgt
                print(f"[Update] New best after targeted local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters_anneal = int(min(250_000, remaining * 25))
        if iters_anneal > 0:
            print("\n--- Simulated‑annealing polishing ---")
            seq_anneal, val_anneal = _simulated_annealing(
                global_best_seq,
                n,
                deadline,
                iters=iters_anneal,
                init_temp=2e-3,
                final_temp=5e-7,
                step_frac=0.009,
            )
            print(f"[Anneal] best = {val_anneal:.6f}")
            if val_anneal < global_best_val:
                global_best_seq = seq_anneal.copy()
                global_best_val = val_anneal
                print(f"[Update] New best after annealing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1200, int(min(3000, remaining * 0.04 * 1e5)))
        high_beta_sched = [5_000_000.0, 1e7, 5e7,
                           1e8, 5e8, 1e9, 5e9,
                           1e10]
        print("\n--- Final high‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=high_beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation via the provided security‑checked function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
