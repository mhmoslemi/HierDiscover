"""
Propose a candidate step‑function (non‑negative heights) that minimizes
``evaluate_sequence``.  The approach builds many diverse seed shapes
(analytic windows, random Dirichlet / exponential draws, parametrised
Beta / power families, Chebyshev & Kaiser windows, etc.), keeps the
best few according to the exact surrogate objective

    obj(a) = 2·n·max(conv(a,a))            (a is ℓ₁‑normalised)

and improves the best one by
  * linear‑programming “boost” (LP) that adds ℓ₁‑mass where the convolution
    slack permits,
  * a long exponentiated‑gradient (mirror‑descent) sub‑gradient descent on
    the exact max‑convolution,
  * a continuation L‑BFGS optimisation that smooths the max with a
    log‑sum‑exp surrogate.

The problem size is set to *n = 2000* (small enough for the LP matrix
while still large enough to give a good bound).  The whole pipeline is
bounded by the supplied time budget (default 1000 s).  The final
sequence is stored in the global ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for LP, L‑BFGS and extra windows)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    import scipy.signal as _sig
except Exception:  # pragma: no cover

    _sp = None

# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to ℓ₁‑norm = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _chebwin_window(n: int, attenuation: float) -> np.ndarray:
    # SciPy's get_window accepts ('chebwin', attenuation)
    if _sig is None:
        return _uniform(n)
    try:
        a = _sig.get_window(('chebwin', attenuation), n, sym=False)
        a = np.asarray(a, dtype=np.float64)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    except Exception:  # pragma: no cover
        return _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    n = len(a)
    check_interval = 2000  # check time every this many iterations

    for it in range(1, max_iters + 1):
        # time‑check (periodic to save overhead)
        if it % check_interval == 0 and time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        # SciPy not available – fall back to the input vector unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0, 2000000.0]

    n = len(start_vec)
    v = np.log(start_vec + 1e-30)  # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (log‑softmax space)
            w = exp_beta / sum_exp          # weights for each entry of conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)     # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Linear‑programming boost step (adds ℓ₁‑mass where slack permits)
# ----------------------------------------------------------------------
def _build_conv_ub_matrix(f: np.ndarray) -> _sp.csr_matrix:
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that A @ g = conv(f, g).
    The matrix is used for the LP constraint conv(f, g) ≤ M.
    """
    n = f.shape[0]
    # Rows correspond to i + j for all i,j ∈ [0, n‑1]
    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel().astype(np.int32)
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    data = np.repeat(f, n)
    A = _sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_max_sum(given_f: np.ndarray, M: float, deadline: float):
    """
    Solve   max Σ g_i   s.t. conv(given_f, g) ≤ M, g ≥ 0.
    Returns the optimal vector g (dense).
    """
    if _opt is None or _sp is None:
        raise RuntimeError("SciPy required for LP step")

    n = given_f.shape[0]
    A = _build_conv_ub_matrix(given_f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimise -Σg_i  ⇔ maximise Σg_i
    bounds = [(0, None)] * n

    # hi‑gHS solves quickly; enforce remaining time budget
    time_left = max(0.5, deadline - time.time())
    res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                       method='highs',
                       options={'presolve': True,
                                'time_limit': time_left})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def _ternary_mix(a: np.ndarray, b: np.ndarray, deadline: float,
                 steps: int = 30) -> tuple[np.ndarray, float]:
    """Best convex combination of a and b by ternary search on α∈[0,1]."""
    lo, hi = 0.0, 1.0
    best_seq = None
    best_val = float('inf')
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if val1 < best_val:
                best_val, best_seq = val1, seq1
        else:
            lo = m1
            if val2 < best_val:
                best_val, best_seq = val2, seq2
    mid = (lo + hi) / 2.0
    seq_mid = _mix_shapes(a, b, mid)
    val_mid = _objective_surrogate(seq_mid)
    if val_mid < best_val:
        best_seq, best_val = seq_mid, val_mid
    return best_seq, best_val


def _lp_refine(initial_f: np.ndarray,
               deadline: float,
               max_lp_iters: int = 6,
               mix_steps: int = 25,
               min_improve: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based refinement iterations starting from `initial_f`.
    Returns the best ℓ₁‑normalised sequence and its surrogate value.
    """
    f = initial_f.copy()
    n = f.shape[0]
    best_val = _objective_surrogate(f)
    best_f = f.copy()
    print(f"[LP‑refine] start surrogate = {best_val:.6f}")

    for it in range(max_lp_iters):
        if time.time() > deadline:
            break
        conv_ff = _conv_fft(f, f)
        M = float(np.max(conv_ff))
        try:
            g = _lp_max_sum(f, M, deadline)
        except Exception as exc:
            print(f"  LP iteration {it} failed: {exc}")
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            print(f"  LP iteration {it}: no ℓ₁‑norm increase (sum_g={sum_g:.6f})")
            break
        # Scale g back to ℓ₁‑norm = 1 and mix
        g_scaled = g / sum_g
        f_new, val_new = _ternary_mix(f, g_scaled, deadline, steps=mix_steps)
        f_new = f_new / f_new.sum()
        val_new = _objective_surrogate(f_new)
        print(f"  LP iteration {it}: new surrogate = {val_new:.6f} (sum_g={sum_g:.6f})")
        if val_new < best_val - min_improve:
            best_val, best_f = val_new, f_new.copy()
            f = f_new
        else:
            print(f"  LP iteration {it}: improvement too small, stopping.")
            break
    return best_f, best_val


# ----------------------------------------------------------------------
# Candidate generation utilities
# ----------------------------------------------------------------------
def _window_signal(name: str, n: int, param=None) -> np.ndarray:
    """
    Return a SciPy signal window of given name and optional param,
    normalised to ℓ₁‑norm = 1.
    """
    if _sig is None:
        return _uniform(n)
    try:
        if param is None:
            w = _sig.get_window(name, n, sym=False)
        else:
            w = _sig.get_window((name, param), n, sym=False)
        w = np.asarray(w, dtype=np.float64)
        s = w.sum()
        if s <= 0.0:
            return _uniform(n)
        return w / s
    except Exception:  # pragma: no cover
        return _uniform(n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The pipeline consists of:

    1. Broad seed generation (analytic families, random Dirichlet/exp,
       low‑frequency curves, many window functions, plus random β‑shape).
    2. Cheap surrogate evaluation → keep the top‑few raw seeds.
    3. Pairwise convex mixing of the top seeds (ternary search).
    4. Linear‑programming refinement (few iterations, modest n).
    5. Long exponentiated‑gradient sub‑gradient descent.
    6. Continuation L‑BFGS smoothing.
    7. Final evaluation with ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size (a compromise between quality and LP memory)
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Retrieve previous solution if present (helps with warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Broad seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # --- classic analytic families (coarse grids) --------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.2, 3.0, 8):
        seeds.append(_sine_window(n, gamma=gamma))

    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    for sigma in np.linspace(0.08, 0.24, 9):
        seeds.append(_gaussian(n, sigma))

    for alpha in np.linspace(0.2, 5.0, 6):
        for beta in np.linspace(0.2, 5.0, 6):
            seeds.append(_beta_shape(n, alpha, beta))

    for tuk_alpha in np.linspace(0.1, 0.9, 5):
        seeds.append(_tukey_window(n, tuk_alpha))

    for kaiser_beta in np.linspace(2.0, 12.0, 6):
        seeds.append(_kaiser_window(n, kaiser_beta))

    # Chebyshev windows (different sidelobe attenuation)
    for att in (30, 60, 90, 120):
        seeds.append(_chebwin_window(n, att))

    # SciPy generic windows
    generic_win_names = ['hann', 'hamming', 'blackman', 'blackmanharris',
                         'flattop', 'parzen', 'bartlett']
    for name in generic_win_names:
        seeds.append(_window_signal(name, n))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random mixtures of two windows (adds diversity)
    for mix_alpha in (0.2, 0.5, 0.8):
        w1 = _uniform(n)
        w2 = _tukey_window(n, 0.4)
        seeds.append(_mix_shapes(w1, w2, mix_alpha))

    # Alternating Dirichlet (already defined)
    seeds.append(_alternating_dirichlet(n, rng))

    # Random β‑shape parameters (log‑uniform sampling)
    for _ in range(150):
        a = 10 ** rng.uniform(-2, 1)   # between 0.01 and 10
        b = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, a, b))

    # Perturb previous best if available
    if prev_arr is not None and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(5):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Surrogate evaluation of all seeds (cheap)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Pairwise convex mixing of the top seeds (cheap ternary search)
    # ------------------------------------------------------------------
    best_seq = top_seeds[0]
    best_val = _objective_surrogate(best_seq)

    mix_deadline = deadline - 30.0   # leave room for LP + polishing
    for i in range(top_k):
        for j in range(i + 1, top_k):
            if time.time() > mix_deadline:
                break
            seq_i = top_seeds[i]
            seq_j = top_seeds[j]
            mixed, val_mix = _ternary_mix(seq_i, seq_j, mix_deadline, steps=20)
            if val_mix < best_val:
                best_val, best_seq = val_mix, mixed

    # ------------------------------------------------------------------
    # 4. Linear‑programming refinement (if memory permits)
    # ------------------------------------------------------------------
    lp_deadline = deadline - 15.0
    refined_seq, refined_val = _lp_refine(best_seq, lp_deadline,
                                          max_lp_iters=8,
                                          mix_steps=30)

    # ------------------------------------------------------------------
    # 5. Long exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- EG polishing (long run) ---")
        eg_deadline = deadline - 7.0
        seq_eg, val_eg = _exp_gradient_opt(refined_seq,
                                           eg_deadline,
                                           max_iters=3_000_000,
                                           eta0=8.0)
        if val_eg < refined_val:
            refined_seq, refined_val = seq_eg, val_eg
            print(f"  EG improved surrogate to {refined_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Continuation L‑BFGS smoothing (final polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS smoothing ---")
        lbfgs_deadline = deadline
        seq_lbfgs = _lbfgs_smooth_optimize(refined_seq,
                                           lbfgs_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1000.0,
                                                          2000.0, 5000.0,
                                                          20000.0, 50000.0,
                                                          100000.0, 200000.0,
                                                          500000.0, 1000000.0,
                                                          2000000.0],
                                           max_iter_per_beta=5000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < refined_val:
            refined_seq, refined_val = seq_lbfgs, val_lbfgs
            print(f"  L‑BFGS improved surrogate to {refined_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation with the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(refined_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {refined_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the best sequence for the next run (same length as used)
    globals()["height_sequence_1"] = refined_seq.tolist()
    return refined_seq.tolist()
