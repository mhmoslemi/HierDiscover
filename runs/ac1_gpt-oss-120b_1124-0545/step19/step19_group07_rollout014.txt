def propose_candidate(budget_s=1000, seed=12345, **kwargs):

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1

    n = 4000

    # Build seeds list
    seeds = []

    # Uniform baseline
    seeds.append(np.full(n, 1.0/n))

    # Fractions for sparse
    fractions = [0.3,0.4,0.5,0.6,0.7,0.8]
    for frac in fractions:
        m = max(2, int(frac * n))
        for _ in range(15):
            idx = rng.choice(n, size=m, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[idx] = 1.0 / m
            seeds.append(a)

    # Random Dirichlet low conc (0.01-0.2)
    for conc in [0.02,0.05,0.1,0.2]:
        for _ in range(10):
            a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            seeds.append(a)

    # Random reorderings: zigzag
    for _ in range(10):
        a = rng.dirichlet(np.full(n, 0.5)).astype(np.float64)
        seed_a = _zigzag(a)
        seeds.append(seed_a)

    # Evaluation: compute objective
    best_val = np.inf
    best_seq = None
    for a in seeds:
        # ensure sum >0
        s = a.sum()
        if s <= 0: continue
        a = a / s
        val = _objective_exact(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    print(f"[Init] best objective = {best_val:.6f}")

    # Stochastic local search from best_seq
    if best_seq is None: return [0.0]*n

    # parameters
    iters = int(min(200_000, (deadline - time.time()) * 100))
    # implement hill climbing
    a = best_seq.copy()
    best_val_local = best_val
    best_a_local = a.copy()
    rng2 = np.random.default_rng(seed+1)

    for it in range(iters):
        if time.time() > deadline:
            break
        # propose move: pick two indices (could be same)
        i = rng2.integers(n)
        j = rng2.integers(n)
        if i == j or a[i] <= 0:
            continue
        # move portion
        delta = a[i] * rng2.uniform(0.001, 0.01)  # move small fraction
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalize? sum remains 1 (since delta moved)
        # maybe set small negative to zero
        if a_new[i] < 0: a_new[i]=0.0
        # recompute objective
        val = _objective_exact(a_new)
        if val < best_val_local:
            best_val_local = val
            best_a_local = a_new.copy()
            a = a_new
            # optionally print progress occasionally
            if it % 10000 == 0:
                print(f"[Local] iter {it} best = {best_val_local:.6f}")

    print(f"[Local] final best = {best_val_local:.6f}")

    global_best_seq = best_a_local
    global_best_val = best_val_local

    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"evaluate_sequence = {true_val:.6f}")
    except Exception as e:
        print("evaluate_sequence raised:", e)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

"""
A fast combinatorial construction for a low‑autocorrelation step function.

The idea:
* Choose a subset S⊂{0,…,n‑1} of positions and put equal mass on them.
  The convolution of the resulting vector consists of the ordered‑pair
  counts Ck=|{(i,j)∈S×S : i+j=k}| multiplied by w² where w=1/|S|.
  The evaluation function is   2·n·max(Ck)/|S|², i.e. ρ·(with ρ=maxC/averageC).

* For a random subset the ratio ρ≈2 (the uniform case).  By swapping
  positions we can lower maxC towards the average C≈|S|²/(2n) and
  therefore drive ρ down to ≈1.0 → objective ≈1.0.

* The algorithm builds a random set S of size m=n/2, computes the
  ordered‑pair count array C with a convolution, then performs a
  simple hill‑climbing: repeatedly try to exchange one element of S
  with an element outside S; accept the swap only if the new max(C)
  does not increase.  This cheap local search quickly reduces ρ.

* After the combinatorial phase we keep the resulting vector and run a
  short exponentiated‑gradient optimisation on the exact objective to
  polish the result.

The whole routine respects the 1000 s wall‑time limit, stores the best
sequence in the global ``height_sequence_1`` and returns it as a list
of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective & helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) – only the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = np.exp(log_a)
        a = a / a.sum()                     # renormalise (softmax)

        # occasional checkpoint
        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val

        last = S_list[-1]
        S_list[idx_out] = last
        S_list.pop()
        in_S[i_out] = False

        # ---- add i_in ----------------------------------------------------
        for j in S_list:                 # S_list now lacks i_out
            C[i_in + j] += 2
        C[2 * i_in] += 1
        S_list.append(i_in)
        in_S[i_in] = True

        new_max = int(C.max())
        if new_max <= current_max:
            # accept the swap
            current_max = new_max
        else:
            # ---- revert (undo i_in) ---------------------------------------
            for j in S_list:             # S_list currently contains i_in
                if j == i_in:
                    C[2 * i_in] -= 1
                else:
                    C[i_in + j] -= 2
            S_list.pop()
            in_S[i_in] = False

            # ---- revert (restore i_out) ------------------------------------
            for j in S_list:             # S_list now back to former members
                C[i_out + j] += 2
            C[2 * i_out] += 1
            S_list.append(i_out)
            in_S[i_out] = True

        swaps_done += 1
        # occasional progress report
        if swaps_done % 5_000 == 0:
            print(f"[Hill] swaps {swaps_done}, maxC={current_max}")

    # ------------------------------------------------------------------
    # Phase 3 – build the actual sequence (equal weights on S)
    # ------------------------------------------------------------------
    a = np.zeros(n, dtype=np.float64)
    w = 1.0 / m
    a[S_list] = w

    init_val = _objective_exact(a)
    print(f"[Result] after combinatorial phase objective = {init_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – short exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a modest budget for the EG stage (up to 300 k updates)
        max_iters_eg = int(min(300_000, remaining * 80_000))
        if max_iters_eg > 0:
            a, val_eg = _exp_gradient_optimize(
                a,
                n,
                deadline,
                max_iters=max_iters_eg,
                eta0=2.0,
            )
            print(f"[EG] polished objective = {val_eg:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(a.tolist())
        print(f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = a.tolist()
    return a.tolist()
