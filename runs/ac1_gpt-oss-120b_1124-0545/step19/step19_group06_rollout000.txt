def _project_to_simplex(v):
    # ensure non-negative, sum =1
    # technique: shift v by subtracting max to maintain numeric.

Pseudo:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

Return w.

def _sgd_proj_optimize(start_a, n, deadline, max_iters, eta0=0.5):
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    it=0
    while it<max_iters and time.time() < deadline:
        it+=1
        g = _subgradient_maxconv(a)
        step = eta0 / np.sqrt(it)
        a_new = a - step * g
        a = _project_to_simplex(a_new)
        # occasional random perturbation: with prob small e.g., if it%50000==0
        if it % 50000 == 0:
            noise = np.random.default_rng().normal(scale=0.001, size=n)
            a = _project_to_simplex(a + noise)
        if it % 10000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val

"""
Propose a candidate step‑function that improves the max‑convolution bound.

The algorithm builds a rich pool of seed shapes (uniform, triangular,
power‑law, exponential, beta, Gaussian, cosine‑mixture, random Dirichlet
etc.), keeps the best few, and refines them with three complementary
optimisers:

1. Continuation L‑BFGS on a smooth log‑sum‑exp surrogate (as before).
2. Projected sub‑gradient descent on the exact non‑smooth objective,
   using a simplex projection after each step.
3. A short simulated‑annealing hill‑climb for final polishing.

The stages are timed so that the total wall‑clock budget (≈1000 s) is
respected.  The best sequence found is stored in the global variable
``height_sequence_1`` for warm‑starts in later calls.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# FFT based convolution (real, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (onto ℓ₁‑simplex)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to avoid tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Analytic families (all returned vectors are ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, mu: float, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int, width: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.maximum(0.0, np.cos(np.pi * (x - center) / width))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_halfwave(n: int) -> np.ndarray:
    """Positive half‑sine bump on [0, n‑1]."""
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Exact (non‑smooth) objective and sub‑gradient for the active lag
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (∑a)²  (sum may differ from 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (active lag only)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    # for each i in the overlapping interval set g[i] = 2*a[idx-i]
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) & continuation L‑BFGS
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate of the max‑convolution objective."""
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)
    shift = np.max(conv)                 # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑propagation through the outer softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0,
        ]
    # start from log‑odds to keep positivity unconstrained
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun, v, method="L-BFGS-B", jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "ftol": 1e-12,
                                         "disp": False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break
    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _sgd_projected_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent (simplex projection after each step)."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    it = 0
    rng = np.random.default_rng()

    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        step = eta0 / np.sqrt(it)
        a_tmp = a - step * g
        a = _project_to_simplex(a_tmp)

        # occasional tiny random jitter to escape flats
        if it % 50000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            a = _project_to_simplex(a + noise)

        if it % 20000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local search (mass‑transfer hill‑climb)
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 300_000,
                         init_temp: float = 2.0) -> tuple[np.ndarray, float]:
    a = start_a.copy()
    cur_val = _objective_exact(a)
    best_a = a.copy()
    best_val = cur_val
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.random() * 0.02
        delta = frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = _objective_exact(cand)
        if cand_val < cur_val:
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_a = cand.copy()
                best_val = cand_val
        else:
            progress = it / max_iters
            T = init_temp * max(0.05, 1.0 - progress)
            if rng.random() < np.exp(-(cand_val - cur_val) / T):
                a = cand
                cur_val = cand_val
    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set) – optional but useful
# ----------------------------------------------------------------------
def _lp_refine(start_a: np.ndarray,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_steps: int = 5,
               top_k: int = 200,
               mix_ratio: float = 0.15) -> tuple[np.ndarray, float]:
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for step in range(max_steps):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        M = np.max(conv)

        # choose the most critical lags
        if top_k >= len(conv):
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(-conv, top_k)[:top_k]

        rows, cols, data = [], [], []
        for r, k in enumerate(idxs):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                rows.append(r)
                cols.append(j)
                data.append(a[i])
        A = _sp.csr_matrix((data, (rows, cols)), shape=(len(idxs), n))
        b_ub = np.full(len(idxs), M, dtype=np.float64)
        bounds = [(0, None)] * n
        c = np.full(n, -1.0, dtype=np.float64)

        time_left = deadline - time.time()
        lp_options = {"presolve": True,
                      "time_limit": max(1.0, time_left * 0.2)}

        try:
            res = _opt.linprog(c, A_ub=A, b_ub=b_ub,
                               bounds=bounds,
                               method="highs",
                               options=lp_options)
        except Exception:
            break
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break
        g_norm = g / s
        new = (1.0 - mix_ratio) * a + mix_ratio * g_norm
        new_val = _objective_exact(new)
        if new_val < best_val:
            best_a = new.copy()
            best_val = new_val
        a = new
    return best_a, best_val

# Seed‑pool construction (extended with a sine shape)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.0004))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_gaussian(n, mu=n / 2, sigma=n / 5))
    seeds.append(_raised_cosine(n, width=n * 0.8))
    seeds.append(_sine_halfwave(n))

    # dense grid of power‑law exponents
    for p in np.linspace(0.30, 0.80, 31):
        seeds.append(_power_law(n, float(p)))

    # random Dirichlet families
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures of existing seeds
    base = np.arange(len(seeds))
    for _ in range(120):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(20):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best, if any
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up (non‑negativity + ℓ₁‑normalisation)
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

        if max_iters > 0:
            print("\n--- Simulated‑annealing local search ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq, n,
                                                  deadline, rng,
                                                  max_iters=max_iters,
                                                  init_temp=2.0)
            print(f"[SA] best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[Update] New best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional LP refinement (active‑set)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        steps = 5
        print("\n--- LP refinement (active‑set) ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, n, deadline,
                                    rng, max_steps=steps,
                                    top_k=250, mix_ratio=0.12)
        print(f"[LP] best = {val_lp:.6f}")
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[Update] New best after LP = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final high‑β L‑BFGS polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(2500, remaining * 0.07 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq, n, deadline,
                                           beta_schedule=[500_000.0, 1_000_000.0,
                                                          2_000_000.0, 5_000_000.0],
                                           max_iter_per_beta=per_beta)
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in future calls
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
