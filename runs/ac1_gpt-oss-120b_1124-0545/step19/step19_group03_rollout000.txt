class SmoothObjective:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta

    def __call__(self, v):
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shift = max_c
        exp_beta = np.exp(self.beta * (conv - shift))
        sum_exp = exp_beta.sum()
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / self.beta)

        # gradient w.r.t a
        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length = 3n-2
        grad_a = 2.0 * full[self.n-1:self.n-1 + self.n]

        # gradient w.r.t v
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v

"""
Propose a non‑negative step‑function (height sequence) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The algorithm loads the previously best sequence (if any) and performs a
deep refinement using a smooth log‑sum‑exp surrogate of the max‑convolution,
optimised with L‑BFGS‑B on the log‑softmax (probability simplex) parametrisation.
After a schedule of increasing smoothing parameters (beta) we run an
exponentiated‑gradient (mirror‑descent) polishing pass with many iterations,
followed by a short simulated‑annealing fine‑tuning.  All work respects the
global time budget (default 1000 s) and the final sequence is stored in the
global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalize."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Sub‑gradient is taken from *all* active (near‑maximal) indices.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # Work in log‑space; this preserves positivity and the ℓ₁‑norm automatically.
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # Active set: all indices within a small relative tolerance of the max.
        eps = max_val * 1e-6
        active = np.where(conv >= max_val - eps)[0]

        # Accumulate sub‑gradient contributions from every active index.
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth surrogate for L‑BFGS (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------
class _SmoothMax:
    """
    Callable object for the smooth max‑convolution surrogate.
    The objective is
        2*n* (shift + log(sum(exp(beta*(conv - shift))))/beta)
    where shift = max(conv) for numerical stability.
    The gradient is given in the log‑softmax (v) space.
    """
    __slots__ = ("n", "beta")

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v: np.ndarray) -> tuple[float, np.ndarray]:
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shift = max_c
        exp_beta = np.exp(self.beta * (conv - shift))
        sum_exp = exp_beta.sum()
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / self.beta)

        # Gradient w.r.t a (smooth max derivative)
        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # Chain rule to v‑space (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.01,
                         T_end: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Lightweight SA: Gaussian perturbations with a linearly decreasing temperature.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)  # perturbation shrinks with temperature

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous solution if present
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr = prev_arr / s
                else:
                    prev_arr = None
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # --------------------------------------------------------------
    # Initial seed – either the previous best or a generic analytic shape
    # --------------------------------------------------------------
    if prev_arr is not None:
        init_seq = prev_arr.copy()
        print("[Info] Loaded previous best sequence.")
    else:
        # a simple sine window turns out to be a decent starting point
        init_seq = _sine_window(n, gamma=1.2)
        print("[Info] No previous sequence – starting from sine window.")

    # --------------------------------------------------------------
    # Deep L‑BFGS refinement on the smooth max surrogate
    # --------------------------------------------------------------
    if _opt is not None:
        print("\n--- L‑BFGS refinement (smooth max) ---")
        # work in log‑softmax space
        v = np.log(init_seq + 1e-30)

        # a fairly fine‑grained schedule of smoothing parameters
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0
        ]

        # keep track of the best sequence seen during the whole schedule
        best_seq = init_seq.copy()
        best_val = _objective_surrogate(best_seq)

        for beta in beta_schedule:
            if time.time() > deadline:
                break
            print(f"  * beta = {beta:.0e}")

            smooth_obj = _SmoothMax(n, beta)

            # callback to enforce the time limit
            def _cb(_):
                if time.time() > deadline:
                    raise KeyboardInterrupt()

            try:
                res = _opt.minimize(
                    fun=smooth_obj,
                    x0=v,
                    method="L-BFGS-B",
                    jac=True,
                    callback=_cb,
                    options={"maxiter": 5000, "disp": False, "ftol": 1e-15}
                )
                v = res.x
            except KeyboardInterrupt:
                print("[L‑BFGS] Time limit reached – exiting early.")
                break
            except Exception as e:   # pragma: no cover
                print(f"[L‑BFGS] Exception at beta={beta}: {e}")
                break

            # evaluate the current point on the *exact* surrogate
            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"    > Improved surrogate to {best_val:.6f}")

        # after the schedule, set the sequence to the best found
        current_seq = best_seq.copy()
        current_val = best_val
        print(f"\n[After L‑BFGS] best surrogate = {current_val:.6f}")
    else:
        # SciPy not available – fall back to the initial sequence
        print("\n[Warning] SciPy not available – skipping L‑BFGS.")
        current_seq = init_seq.copy()
        current_val = _objective_surrogate(current_seq)

    # --------------------------------------------------------------
    # Final exponentiated‑gradient polishing (many iterations)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing (high‑precision) ---")
        seq_eg, val_eg = _exp_gradient_opt_multi(
            current_seq,
            n,
            deadline,
            max_iters=2_500_000,
            eta0=4.5
        )
        if val_eg < current_val:
            current_seq = seq_eg
            current_val = val_eg
            print(f"[EG] Improved surrogate to {current_val:.6f}")

    # --------------------------------------------------------------
    # Final simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Simulated Annealing fine‑tuning ---")
        seq_sa, val_sa = _simulated_annealing(
            current_seq,
            n,
            deadline,
            rng,
            max_attempts=120_000,
            start_sigma=0.001,
            T_start=5e-3,
            T_end=1e-9
        )
        if val_sa < current_val:
            current_seq = seq_sa
            current_val = val_sa
            print(f"[SA] Improved surrogate to {current_val:.6f}")

    # --------------------------------------------------------------
    # Official evaluation via the supplied function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(current_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {current_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = current_seq.tolist()
    return current_seq.tolist()
