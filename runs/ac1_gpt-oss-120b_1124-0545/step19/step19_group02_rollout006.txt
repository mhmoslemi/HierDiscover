def _optimize_mixture(seeds, n, deadline, beta_schedule=None, max_iter=10000):
    K = len(seeds)
    # Ensure seeds each sum to 1 (ell1 normalized)
    seeds = [s / s.sum() for s in seeds]
    v0 = np.full(K, 1.0 / K) # starting weights
    x0 = np.log(v0 + 1e-30) # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(logits):
            w = _softmax(logits)  # mixture weights
            a = np.tensordot(w, seeds, axes=1)  # shape (n)
            # compute smoothing objective and gradient
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            w_conv = exp_beta / sum_exp
            rev_a = a[::-1]
            # convolution rev_a w -> full length (2n-1)
            full = _conv_fft(rev_a, w_conv)
            grad_a = 2.0 * full[n-1:n-1+n]  # length n

            # gradient wrt mixture weights
            # compute dot(grad_a, seed_i) for each i
            # seeds[i] shape (n)
            grad_w = np.array([np.dot(grad_a, s) for s in seeds])  # size K

            # chain rule through softmax
            dot = np.dot(w, grad_w)
            grad_logits = w * (grad_w - dot)

            return obj, grad_logits

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun, x0=x0,
                                method='L-BFGS-B', jac=True,
                                callback=callback,
                                options={'maxiter': 5000, 'disp': False, 'ftol': 1e-12})
            x0 = res.x
        except KeyboardInterrupt:
            break

    # Compute final mixture vector
    final_w = _softmax(x0)
    final_a = np.tensordot(final_w, seeds, axes=1)
    final_obj = _objective_surrogate(final_a) # can verify

    return final_a, final_obj

def _optimize_mixture(seeds_mat, n, deadline, start_logits=None, beta_schedule=None):
    if _opt is None: return seeds_mat[0] (first) - but we can combine simple mixing.

    K = seeds_mat.shape[0]
    if start_logits is None:
        start_logits = np.log(np.full(K, 1.0/K) + 1e-30)   # uniform

    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 20000.0, 50000.0, 100000.0, 200000.0, 500000.0]

    logits = start_logits.copy()
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        def fun(logits_local):
            w = _softmax(logits_local)
            a = np.dot(w, seeds_mat)          # shape (n,)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = np.sum(exp_beta)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w_conv = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w_conv)
            grad_a = 2.0 * full[n-1:n-1+n]

            # gradient wrt w (mixture weights)
            grad_w = np.dot(seeds_mat, grad_a)  # shape (K,)
            dot = np.dot(w, grad_w)
            grad_logits = w * (grad_w - dot)
            return obj, grad_logits

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun, x0=logits, method='L-BFGS-B',
                                jac=True, callback=callback,
                                options={'maxiter': 6000, 'disp': False, 'ftol': 1e-12})
            logits = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[Mixture L‑BFGS] exception at β={beta_val}: {e}")
            break

    final_w = _softmax(logits)
    final_seq = np.dot(final_w, seeds_mat)
    final_val = _objective_surrogate(final_seq)
    return final_seq, final_val

"""
Enhanced search for a low‑value step‑function.

The algorithm proceeds in three stages:

1. **Rich seed pool** – A large collection of analytic windows
   (uniform, triangular, Hamming, Blackman, Kaiser, Gaussian, …),
   random Dirichlet draws and perturbations of the previously best
   solution.  All seeds are ℓ₁‑normalised (sum = 1).

2. **Convex‑mixture optimisation** – The *K* best seeds (K≈10)
   are combined with weights 𝑤∈Δ_K (simplex).  Using a log‑softmax
   parametrisation the problem is solved with L‑BFGS‑B on a smooth
   approximation of the “max‑convolution’’ objective.
   The smoothing parameter β follows an increasing schedule, which
   drives the optimiser towards the true (non‑smooth) objective.

3. **Exact polishing** – The resulting vector is refined by an
   exponentiated‑gradient (mirror‑descent) step on the exact max‑convolution.
   A final call to ``evaluate_sequence`` returns the true score.

The routine respects a total wall‑clock budget of *budget_s* seconds
(≈1000 s).  If the time expires early the best vector found so far is
returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS‑B)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Convex‑mixture optimisation (low‑dimensional L‑BFGS on weights)
# ----------------------------------------------------------------------
def _optimize_mixture(seeds_mat: np.ndarray,
                     n: int,
                     deadline: float,
                     start_logits: np.ndarray = None,
                     beta_schedule=None,
                     max_iter_per_beta: int = 6000):
    """
    Optimise a convex combination of the rows of *seeds_mat*.
    The weights live on the probability simplex (softmax of logits).
    A smooth approximation of ``max(conv)`` via log‑sum‑exp is
    minimised with L‑BFGS‑B while the smoothing parameter β is
    increased according to *beta_schedule*.
    Returns the best mixed sequence and its exact surrogate value.
    """
    if _opt is None:                     # SciPy unavailable – fall back
        best_seq = seeds_mat[0].copy()
        return best_seq, _objective_surrogate(best_seq)

    K = seeds_mat.shape[0]

    if start_logits is None:
        start_logits = np.log(np.full(K, 1.0 / K) + 1e-30)

    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0]

    logits = start_logits.copy()

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(logits_local):
            w = _softmax(logits_local)                 # mixture weights
            a = np.dot(w, seeds_mat)                  # combined vector, shape (n,)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = np.sum(exp_beta)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. ``a`` (smooth max)
            w_conv = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w_conv)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # slice of length n

            # gradient w.r.t. mixture weights (before softmax)
            grad_w = np.dot(seeds_mat, grad_a)        # shape (K,)

            # chain rule through softmax
            dot = np.dot(w, grad_w)
            grad_logits = w * (grad_w - dot)

            return obj, grad_logits

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun, x0=logits,
                                method='L-BFGS-B', jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            logits = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[Mixture L‑BFGS] exception at β={beta_val}: {e}")
            break

    final_w = _softmax(logits)
    final_seq = np.dot(final_w, seeds_mat)
    final_val = _objective_surrogate(final_seq)
    return final_seq, final_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:

    n = 4000

    previous = globals().get("height_sequence_1")

            prev_arr = None

    # --------------------------------------------------------------

    alphas = np.linspace(0.2, 5.0, 7)

        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible subsequent runs
    globals()["height_sequence_1"] = best_mix_seq.tolist()
    return best_mix_seq.tolist()
