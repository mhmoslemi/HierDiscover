"""
Improved step‑function search.

The target objective is

    2 * n * max(conv(a, a)) / (sum(a)**2)

with non‑negative a.  The algorithm works in three phases:

1. **Broad seeding** – a large pool of analytically‑defined windows
   (uniform, sine, Gaussian, beta‑shapes, Tukey, arcsine, …) together
   with many random Dirichlet / exponential draws.  All seeds are
   ℓ₁‑normalised.

2. **Fast surrogate ranking** – the exact surrogate 2·n·max(conv) is
   evaluated for every seed; the best few (default 8) are kept.

3. **Refinement** – each elite seed is fed to a continuation L‑BFGS
   optimiser that minimises a smooth ``log‑sum‑exp`` approximation of
   the max‑convolution.  The schedule uses gradually increasing β so
   that the smooth problem approaches the original non‑smooth one.
   After L‑BFGS a short exponentiated‑gradient (mirror‑descent) run
   (exact max, sub‑gradient from all active shifts) is applied for
   final polishing.

All work respects the global time budget (default 1000 s).  The best
sequence found is stored in ``height_sequence_1`` for future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:                     # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    return np.hamming(n).astype(np.float64) / np.hamming(n).sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """a_i ∝ 1 / sqrt(i·(n‑i‑1))  (ends are singular, capped by 0)."""
    i = np.arange(n, dtype=np.float64) + 0.5  # avoid zero
    a = 1.0 / np.sqrt(i * (n - i + 0.5))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - c))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1500):
    """Minimise a smooth surrogate of the max‑convolution using L‑BFGS‑B.
    The optimisation works in log‑softmax space, guaranteeing positivity
    and unit‑sum constraints.
    """
    if _opt is None:
        # SciPy not available → return the start vector unchanged
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)       # work in log‑space

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # ℓ₁‑normalised
            conv = _conv_fft(a, a)                     # full linear conv
            max_c = np.max(conv)
            shift = max_c                               # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth max = shift + log(sum_exp)/beta
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max derivative)
            w = exp_beta / sum_exp                     # soft‑weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # slice back to length n

            # chain‑rule for the log‑softmax parameterisation
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # stop early if we run out of time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact max‑convolution mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 200_000,
                            eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution.
    Sub‑gradient is averaged over *all* active (maximal) indices.
    Returns a pair (best_seq, best_val).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑space representation automatically enforces positivity & ℓ₁‑norm
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active set – indices whose value is within tiny tolerance of the max
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    ``evaluate_sequence`` routine.  The routine respects the time budget
    (default 1000 s) and stores the best sequence found in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical bound.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 1.  Seed generation – diverse analytic families + random draws
    # ------------------------------------------------------------------
    seeds = []

    # Analytic windows (coarse parameter grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_tukey_window(n, 0.8))
    seeds.append(_arcsine_shape(n))

    for gamma in np.linspace(0.2, 6.0, 12):
        seeds.append(_sine_window(n, gamma))

    for sigma_frac in np.linspace(0.05, 0.35, 10):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.0, 6.0, 10):
        seeds.append(_power(n, p))

    for a in np.linspace(0.5, 5.0, 5):
        for b in np.linspace(0.5, 5.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (wide concentration range)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves (few frequencies → smooth shapes)
    for nf in (2, 3, 4, 5, 6):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixed shapes – blend uniform with a low‑frequency curve
    base = _uniform(n)
    for _ in range(10):
        curve = _lowfreq_curve(n, 4, rng)
        alpha = rng.random()
        seeds.append(_mix_shapes(base, curve, alpha))

    # ------------------------------------------------------------------
    # Normalise every seed to ℓ₁ = 1 and discard pathological ones
    # ------------------------------------------------------------------
    clean_seeds = []
    for vec in seeds:
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s < 1e-12:
            continue
        clean_seeds.append(vec / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in clean_seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:                     # pragma: no cover
        raise RuntimeError("No valid seeds generated before deadline.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))      # keep only the most promising
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:02d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (L‑BFGS → short EG polish)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # Allocate a slice of remaining budget for this seed
        remaining = deadline - time.time()
        # reserve roughly 12 % for each upcoming seed + final polish
        sub_deadline = time.time() + remaining * 0.12 / max(1, (top_k - si))

        # ----- (a) continuation L‑BFGS on the smooth surrogate -----
        seq_opt = _lbfgs_smooth_optimize(seed_vec.copy(),
                                         n,
                                         sub_deadline,
                                         beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                        200.0, 500.0, 1_000.0,
                                                        2_000.0, 5_000.0,
                                                        20_000.0, 50_000.0,
                                                        100_000.0, 200_000.0,
                                                        500_000.0, 1_000_000.0],
                                         max_iter_per_beta=1200)
        val_opt = _objective_surrogate(seq_opt)

        # ----- (b) short exponentiated‑gradient polishing (exact max) -----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt_multi(seq_opt.copy(),
                                                     n,
                                                     sub_deadline,
                                                     max_iters=80_000,
                                                     eta0=5.0)
            if val_eg < val_opt:
                seq_opt, val_opt = seq_eg, val_eg

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"  *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing on the best sequence (long EG run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (exponentiated‑gradient) ---")
        final_seq, final_val = _exp_gradient_opt_multi(global_best_seq.copy(),
                                                       n,
                                                       deadline,
                                                       max_iters=500_000,
                                                       eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
