"""
Search for a non‑negative step function (length 4000) minimizing
evaluate_sequence.  The algorithm builds a diverse pool of analytic
and random seeds, mixes them randomly, and then applies a continuation
L‑BFGS with a smooth max surrogate followed by an exponentiated‑gradient
polish.  The remaining time is spent on a simple stochastic hill‑climber
that moves mass between coordinates.  The best sequence found is stored
in the global variable ``height_sequence_1``.
"""

import time, math, numpy as np

# Helper: next power of two, FFT convolution
def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(x, y):
    L = len(x)+len(y)-1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx*fy, n=N)[:L]
    return conv

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0/len(v))
    return e / s

def _objective_surrogate(seq):
    # seq is assumed to sum to 1
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families
def _uniform(n): return np.full(n, 1.0/n)
def _triangular(n):
    i = np.arange(n, dtype=np.float64)
    c = (n-1)/2.0
    a = 1.0 - np.abs(i-c)/c
    a = np.maximum(a,0)
    return a / a.sum()
def _hamming_window(n):
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46*np.cos(2*np.pi*i/(n-1))
    return a / a.sum()
def _blackman_window(n):
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5*np.cos(2*np.pi*i/(n-1)) + 0.08*np.cos(4*np.pi*i/(n-1))
    return a / a.sum()
def _kaiser_window(n, beta):
    i = np.arange(n, dtype=np.float64)
    alpha = (n-1)/2.0
    y = (i-alpha)/alpha
    t = beta*np.sqrt(1 - y*y)
    a = np.i0(t)
    return a / a.sum()
def _gaussian(n, sigma_frac):
    sigma = sigma_frac*n
    mu = (n-1)/2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i-mu)**2)/(2*sigma**2))
    return a / a.sum()
def _exp_decay(n, gamma):
    i = np.arange(n, dtype=np.float64)/(n-1)
    a = np.exp(-gamma*i)
    return a / a.sum()
def _logistic_decay(n, steep, shift):
    x = np.arange(n, dtype=np.float64)/(n-1)
    a = 1.0/(1.0+np.exp(steep*(x-shift)))
    return a / a.sum()
def _random_dirichlet(n, conc, rng):
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)
def _random_exponential(n, rng):
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()
def _perturb(seq, rng, sigma=0.002):
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy,0)
    s = noisy.sum()
    if s==0:
        return seq.copy()
    return noisy / s

# LBFGS smooth optimiser
try:
    import scipy.optimize as _opt
except Exception:
    _opt = None

def _lbfgs_smooth_optimize(start_vec, n, deadline,
                           beta_schedule=None,
                           max_iter_per_beta=4000):
    if _opt is None:
        return start_vec
    if beta_schedule is None:
        beta_schedule = [1.0,5.0,20.0,80.0,200.0,500.0,1000.0,
                         2000.0,5000.0,20000.0,50000.0,100000.0,
                         200000.0,500000.0,1000000.0]
    v = np.log(start_vec + 1e-30)
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
            # gradient w.r.t a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)   # length 3n-2
            grad_a = 2.0 * full[n-1:n-1+n]
            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()
        try:
            res = _opt.minimize(fun=fun, x0=v, method='L-BFGS-B',
                                jac=True, callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            # fall back on current v
            break
    return _softmax(v)

# exponentiated gradient (mirror descent) exact max
def _exp_gradient_opt(start_seq, n, deadline,
                      max_iters=200_000, eta0=6.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # subgradient for entry k
        g = np.zeros_like(a)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        i_idx = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = k - i_idx
        g[i_idx] = 2.0 * a[rev_idx]
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length 4000 that minimizes
    ``evaluate_sequence``.  The algorithm proceeds as:
    1. Build a library of analytic and random seeds.
    2. Keep the best few seeds (according to the exact surrogate).
    3. Generate a large set of random convex mixtures of those seeds.
    4. Run a continuation L‑BFGS on the top mixtures, followed by a
       short exponentiated‑gradient polish.
    5. Spend the remaining budget on a cheap stochastic hill‑climber
       that moves a tiny amount of mass between random coordinates.
    6. Return the best sequence (list of floats) and store it in
       ``height_sequence_1`` for the next round.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0   # 3 s safety margin
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------------
    seeds = []
    # Analytic
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 12):
        seeds.append(_gaussian(n, sigma))
    for gamma in np.linspace(0.1, 8.0, 14):
        seeds.append(_exp_decay(n, gamma))
    for steep in np.linspace(0.5, 25.0, 12):
        for shift in np.linspace(0.1, 0.9, 8):
            seeds.append(_logistic_decay(n, steep, shift))

    # Random Dirichlet with many concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    # Random exponentials
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Perturb previous
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(6):
            seeds.append(_perturb(prev_vec, rng, sigma=0.004))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation and keep the best
    # ------------------------------------------------------------------
    evaluated = []
    for s in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(s)
        evaluated.append((s, val))
    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Random convex mixtures of the top seeds
    # ------------------------------------------------------------------
    mix_candidates = []
    num_mixes = 4000
    for _ in range(num_mixes):
        if time.time() > deadline:
            break
        k_mix = rng.integers(2, min(5, len(top_seeds)) + 1)   # 2..5 seeds
        idx = rng.choice(len(top_seeds), size=k_mix, replace=False)
        weights = rng.dirichlet(np.ones(k_mix))
        cand = np.zeros(n, dtype=np.float64)
        for w, i in zip(weights, idx):
            cand += w * top_seeds[i]
        # already normalised because each term sums to 1 and weights sum to 1
        val = _objective_surrogate(cand)
        mix_candidates.append((cand, val))

    # Keep best mixture candidates
    mix_candidates.sort(key=lambda x: x[1])
    best_mix = mix_candidates[:10]   # keep ten best mixtures
    # Also keep the original top seeds as candidates
    all_candidates = evaluated[:top_k] + best_mix
    all_candidates.sort(key=lambda x: x[1])
    global_best_seq = all_candidates[0][0].copy()
    global_best_val = all_candidates[0][1]
    print(f"Best surrogate after mixing stage: {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement with L‑BFGS + short EG on a few promising candidates
    # ------------------------------------------------------------------
    refine_count = min(5, len(all_candidates))
    for idx in range(refine_count):
        if time.time() >= deadline:
            break
        seed_vec = all_candidates[idx][0]
        # give each refinement a slice of remaining time
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.3   # 30 % of the rest
        print(f"\n[Refine {idx+1}] start surrogate { _objective_surrogate(seed_vec):.6f}")

        # L‑BFGS smoothing
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec, n, sub_deadline,
                                           max_iter_per_beta=3000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        # short EG polish (at most 80 k iters)
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs, n,
                                               sub_deadline,
                                               max_iters=80_000,
                                               eta0=5.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # pick best
        cand_seq, cand_val = min([(seed_vec, _objective_surrogate(seed_vec)),
                                   (seq_lbfgs, val_lbfgs),
                                   (seq_eg, val_eg)],
                                  key=lambda x: x[1])
        print(f"[Refine {idx+1}] candidate surrogate = {cand_val:.6f}")
        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = cand_seq.copy()
            print(f"  >>> Global best improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Stochastic hill‑climber for the remaining budget
    # ------------------------------------------------------------------
    print("\n--- Stochastic hill‑climber ---")
    a = global_best_seq.copy()
    log_a = np.log(a + 1e-30)
    best_val = global_best_val
    it = 0
    while time.time() < deadline:
        it += 1
        # pick two indices, move a small fraction of mass
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to transfer, relative to a_i
        delta = a[i] * rng.random() * 0.02   # up to 2 % of a_i
        if delta <= 0:
            continue
        # propose move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non‑negativity (a_new[i] already >=0)
        # Renormalisation not needed (total sum unchanged)
        # compute surrogate
        val_new = _objective_surrogate(a_new)
        if val_new < best_val:
            a = a_new
            best_val = val_new
            log_a = np.log(a + 1e-30)
            # occasional print
            if it % 5000 == 0:
                print(f"Iter {it}: improved surrogate = {best_val:.6f}")

    global_best_seq = a
    global_best_val = best_val
    print(f"After hill‑climber: surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the supplied function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()

"""
Search for a non‑negative step‑function (length 4000) that minimizes the
provided ``evaluate_sequence`` (i.e. 2·n·max(conv(a,a))/sum(a)^2).

Algorithm
---------
1. Build a rich pool of analytic windows (uniform, triangular, Hamming,
   Blackman, Kaiser, Gaussian, exponential, logistic, DPSS …) and a set
   of random Dirichlet / exponential draws.  The previous best solution
   (global variable ``height_sequence_1``) is also inserted and perturbed.

2. Evaluate the exact surrogate 2·n·max(conv) for every seed (the surrogate
   equals the true objective for sequences normalised to sum = 1).

3. Keep the top few seeds and generate a large number of random convex
   mixtures of them (2‑5 components, Dirichlet weights).  The best mixtures
   are added to the candidate set.

4. Refine the best candidates with a continuation L‑BFGS optimiser
   (smooth‐max surrogate) followed by a short exponentiated‑gradient
   (mirror‑descent) polish on the exact max.

5. Use the remaining time for a stochastic hill‑climber that moves a tiny
   amount of mass between random coordinates; whenever an improvement is
   found the best surrogate is updated.

6. If a little time is left, run a final EG polish on the best sequence,
   evaluate it with ``evaluate_sequence`` and store it in
   ``height_sequence_1`` for the next round.

The whole routine respects the 1000 s time budget and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    return a / a.sum()


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# DPSS (Slepian) windows – optional, fall back to uniform if unavailable
def _dpss_window(n: int, NW: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        w = windows.dpss(n, NW, sym=False).astype(np.float64)
        return w / w.sum()
    except Exception:
        return _uniform(n)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:   # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space so we stay on the probability simplex.
    """
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t a (see derivation in the original script)
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # Abort if we run out of time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000,
                     eta0: float = 6.0):
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of the active entry (convolution matrix J_k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

        seeds.append(_dpss_window(n, NW))

    # Random Dirichlet draws with a wide range of concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Perturb the previous best (if we have one)
    if prev_vec is not None:
        seeds.append(prev_vec)  # exact copy
        for _ in range(6):
            seeds.append(_perturb(prev_vec, rng, sigma=0.004))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of every seed
    # ------------------------------------------------------------------
    evaluated = []
    for s in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(s)
        evaluated.append((s, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"\nTop {top_k} seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Random convex mixtures of the top seeds
    # ------------------------------------------------------------------
    mix_candidates = []
    num_mixes = 4000
    for _ in range(num_mixes):
        if time.time() > deadline:
            break
        k_mix = rng.integers(2, min(5, len(top_seeds)) + 1)   # 2‑5 components
        idx = rng.choice(len(top_seeds), size=k_mix, replace=False)
        weights = rng.dirichlet(np.ones(k_mix))
        cand = np.zeros(n, dtype=np.float64)
        for w, i in zip(weights, idx):
            cand += w * top_seeds[i]          # each seed already sums to 1
        val = _objective_surrogate(cand)
        mix_candidates.append((cand, val))

    mix_candidates.sort(key=lambda x: x[1])
    best_mixes = mix_candidates[:12]      # keep a dozen promising mixtures

    # Merge the raw top seeds and the best mixtures
    all_candidates = evaluated[:top_k] + best_mixes
    all_candidates.sort(key=lambda x: x[1])

    global_best_seq = all_candidates[0][0].copy()
    global_best_val = all_candidates[0][1]
    print(f"\nInitial best surrogate after mixing: {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement with L‑BFGS + short EG on a few promising candidates
    # ------------------------------------------------------------------
    refine_cnt = min(5, len(all_candidates))
    for idx in range(refine_cnt):
        if time.time() >= deadline:
            break

        seed_vec = all_candidates[idx][0]
        # Allocate a fair share of the remaining budget (≈60 % of the rest)
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.6 / (refine_cnt - idx)

        print(f"\n[Refine {idx+1}] start surrogate { _objective_surrogate(seed_vec):.6f}")

        # 1) L‑BFGS on smooth max
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec, n, sub_deadline,
                                           max_iter_per_beta=2000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        # 2) Short EG polish (max 80 k iters)
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs, n,
                                               sub_deadline,
                                               max_iters=80_000,
                                               eta0=5.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # Choose the best among raw seed, LBFGS result, and EG polish
        cand_seq, cand_val = min([(seed_vec, _objective_surrogate(seed_vec)),
                                  (seq_lbfgs, val_lbfgs),
                                  (seq_eg, val_eg)],
                                 key=lambda x: x[1])

        print(f"[Refine {idx+1}] candidate surrogate = {cand_val:.6f}")

        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = cand_seq.copy()
            print(f"  >>> Global best improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Stochastic hill‑climber on the remaining time
    # ------------------------------------------------------------------
    print("\n--- Stochastic hill‑climber ---")
    a = global_best_seq.copy()
    best_val = global_best_val
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.02   # move up to 2 % of a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # total mass unchanged, non‑negativity guaranteed by construction
        val_new = _objective_surrogate(a_new)
        if val_new < best_val:
            a = a_new
            best_val = val_new
            if it % 5000 == 0:
                print(f"Iter {it}: improved surrogate = {best_val:.6f}")

    global_best_seq = a
    global_best_val = best_val
    print(f"\nAfter hill‑climber: surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final short EG polish if a tiny slice of time is left
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        seq_final, val_final = _exp_gradient_opt(global_best_seq, n,
                                                 deadline,
                                                 max_iters=100_000,
                                                 eta0=5.0)
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final
            print(f"Final EG polish improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation with the supplied function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
