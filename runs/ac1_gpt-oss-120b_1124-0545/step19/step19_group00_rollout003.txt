"""
Propose a step‑function (list of non‑negative heights) that tries to
minimise the evaluation function `evaluate_sequence`.  The algorithm
consists of four stages:

1. **Extensive parametric sweep** – we generate a few hundred candidate
   vectors from a rich collection of analytic windows (sine, Gaussian,
   Kaiser, Tukey, β‑shape, power‑law, etc.) and a handful of random
   Dirichlet / low‑frequency curves.  Every candidate is ℓ₁‑normalised
   and scored by the cheap surrogate
        2·n·max(conv(a,a))          (a has unit sum)
   which equals the true objective for normalised vectors.

2. **Mirror‑descent (exponentiated‑gradient) refinement** – the best
   few candidates from the sweep are refined by many EG steps
   (sub‑gradient descent on the exact max‑convolution).  The iterates
   live in log‑softmax space so the simplex constraint is satisfied
   automatically.

3. **Continuation L‑BFGS on a smooth max** – starting from the best
   EG result we run a sequence of L‑BFGS optimisations on the
   log‑sum‑exp smooth approximation of the max, increasing the
   smoothing parameter β gradually (the “continuation” trick).

4. **Local stochastic polishing** – a cheap simulated‑annealing‑like
   hill‑climber moves a small amount of mass between two random
   coordinates, accepts improving moves and slowly reduces the step‑size.

The whole pipeline respects the 1000 s wall‑time limit; the best vector
found is stored in the global ``height_sequence_1`` variable and also
returned as a plain Python list.
"""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with ℓ₁‑norm = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)               # Modified Bessel function I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Cosine‑tapered window.  α=0 ⇒ uniform, α=1 ⇒ Hann."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution (with support on {1,…,n})."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add small Gaussian noise, clip, re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        # SciPy not available – fall back to the input vector unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0]

    # work in log‑softmax space so we stay on the simplex automatically
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (softmax space)
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The method consists of a parametric
    sweep, EG refinement, continuation L‑BFGS smoothing and a short
    stochastic polishing stage.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – reuse the previous length if a solution already exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple)) and len(prev) > 0:
        n = len(prev)
    else:
        n = 4000

    # ------------------------------------------------------------------
    # Helper to safely normalise a vector to unit ℓ₁‑norm
    # ------------------------------------------------------------------
    def _norm(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return _uniform(n)
        return v / s

    # ------------------------------------------------------------------
    # 1. Broad parametric sweep (cheap surrogate evaluation)
    # ------------------------------------------------------------------
    print(f"\n--- Parametric sweep (n={n}) ---")
    top_k = 12                     # keep this many for the next stage
    best_heap: list[tuple[float, np.ndarray]] = []  # (value, vec)

    def _maybe_insert(vec: np.ndarray):
        """Insert vec into best‑heap if it belongs to the top‑k."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            best_heap.append((val, vec))
            best_heap.sort(key=lambda x: x[0])
        else:
            if val < best_heap[-1][0]:
                best_heap[-1] = (val, vec)
                best_heap.sort(key=lambda x: x[0])

    # uniform baseline
    _maybe_insert(_uniform(n))

    # ---- Analytic families with moderately dense grids -----------------
    # Sine window (γ)
    for gamma in np.linspace(0.01, 1.0, 25):
        _maybe_insert(_sine_window(n, gamma))

    # Gaussian (σ fraction of n)
    for sigma_frac in np.linspace(0.07, 0.30, 16):
        _maybe_insert(_gaussian(n, sigma_frac))

    # Kaiser window (β parameter)
    for beta in np.linspace(0.5, 20.0, 20):
        _maybe_insert(_kaiser_window(n, beta))

    # Tukey window (α)
    for alpha in np.linspace(0.0, 1.0, 12):
        _maybe_insert(_tukey_window(n, alpha))

    # Beta shape (α,β) – fairly coarse but covers many regimes
    alist = [0.1, 0.3, 0.7, 1.0, 2.0, 5.0, 10.0]
    for a_ in alist:
        for b_ in alist:
            _maybe_insert(_beta_shape(n, a_, b_))

    # Power law (p)
    for p in np.linspace(0.0, 2.5, 15):
        _maybe_insert(_power(n, p))

    # Low‑frequency random curves (2–5 sinusoids)
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            _maybe_insert(_lowfreq_curve(n, nf, rng))

    # Random Dirichlet draws
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            _maybe_insert(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        _maybe_insert(_random_exponential(n, rng))

    # Perturbations of the previous best (if any)
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.size == n and prev_arr.sum() > 0:
            base = _norm(prev_arr)
            _maybe_insert(base)
            for sigma in (0.002, 0.004, 0.006):
                for _ in range(3):
                    _maybe_insert(_perturb(base, rng, sigma=sigma))

    # Sort the heap (ascending by surrogate value)
    best_heap.sort(key=lambda x: x[0])
    top_seqs = [vec for _, vec in best_heap[:top_k]]
    best_val, best_seq = best_heap[0]
    print("\nTop candidates after sweep (surrogate):")
    for i, (val, _) in enumerate(best_heap[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Exponentiated‑gradient refinement on the top candidates
    # ------------------------------------------------------------------
    eg_time = max(0.0, deadline - time.time() - 30.0)  # keep ~30 s for later steps
    eg_deadline = time.time() + eg_time
    print(f"\n--- EG refinement (budget ≈ {eg_time:.1f}s) ---")
    for idx, seq in enumerate(top_seqs):
        if time.time() > eg_deadline:
            break
        # run EG for a moderate number of iterations (depends on time left)
        remaining = eg_deadline - time.time()
        # aim for roughly 200 k iterations per candidate, but never exceed time
        max_iters = min(400_000, int(2e5 * (remaining / eg_time + 0.1)))
        seq_opt, val_opt = _exp_gradient_opt(seq.copy(),
                                             n,
                                             eg_deadline,
                                             max_iters=max_iters,
                                             eta0=5.0)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"  Candidate {idx+1}: new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS smoothing (smooth max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0 and _opt is not None:
        print("\n--- L‑BFGS continuation smoothing ---")
        # Ensure the start vector has strictly positive entries for log‑softmax
        start_vec = np.maximum(best_seq, 1e-12)
        start_vec = start_vec / start_vec.sum()
        seq_lbfgs = _lbfgs_smooth_optimize(start_vec,
                                           n,
                                           deadline,
                                           beta_schedule=[1, 5, 20, 80,
                                                          200, 500, 1000,
                                                          2000, 5000,
                                                          20000, 50000,
                                                          100000, 200000,
                                                          500000, 1000000],
                                           max_iter_per_beta=5000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print(f"  L‑BFGS improvement: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Local stochastic polishing (tiny mass moves)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        print("\n--- Local stochastic polishing ---")
        seq = best_seq.copy()
        cur_val = best_val
        step_frac = 0.05               # initial fraction of source mass to move
        it = 0
        while time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # propose moving a fraction of a[i] to a[j]
            delta = step_frac * seq[i] * rng.uniform(0.5, 1.5)
            if delta > seq[i]:
                delta = seq[i]
            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            val_new = _objective_surrogate(new_seq)
            if val_new < cur_val:
                seq = new_seq
                cur_val = val_new
                if cur_val < best_val:
                    best_val, best_seq = cur_val, seq.copy()
                    print(f"  iter {it:6d}: new best surrogate = {best_val:.6f}")
                # slightly increase aggressiveness after a success
                step_frac = min(0.2, step_frac * 1.02)
            else:
                # slowly decay step size on failures
                step_frac = max(1e-4, step_frac * 0.9995)
            it += 1
            if it % 5000 == 0:
                # give a quick status update
                print(f"  iter {it:6d}: cur surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final true evaluation and bookkeeping
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    # Store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
