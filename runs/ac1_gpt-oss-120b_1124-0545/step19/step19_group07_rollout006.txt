"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.  The optimiser works in several phases:

1. **Diversified seed pool** – many analytic families (uniform,
   linear / quadratic decay, exponentials, Gaussians, Beta, power‑law,
   logistic, low‑frequency cosines, random Dirichlet draws, sparse
   patterns, zig‑zag permutations and mixtures).

2. **Heavy L‑BFGS on a smooth log‑sum‑exp surrogate** – continuation
   from low to ultra‑high β, 4000 L‑BFGS iterations per β.  The
   variables are logits (unconstrained) and the soft‑max yields an
   ℓ₁‑normalised candidate.

3. **Exponentiated‑gradient (EG) on the exact max‑convolution** – many
   long runs from random Dirichlet starts and from the current best.
   The step‑size ηₜ = η₀/√t, η₀≈2.5.  Sub‑gradient of the max
   convolution is computed analytically.

4. **Simulated‑annealing fine‑tuning** – mass‑transfer moves
   (δ taken from a random source index and added to a random target)
   are accepted with Metropolis probability at a temperature that
   cools linearly from a modest start to a tiny final value.

5. **Final ultra‑high‑β L‑BFGS polishing** – a last short L‑BFGS run
   with β∈{5e9,1e12} to push the exact max a little lower.

The best vector is stored globally as ``height_sequence_1`` and also
returned as a plain list of Python floats.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → ℓ₁‑normalised (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective and sub‑gradient
# ----------------------------------------------------------------------


def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    Returns g where (∂/∂a) max(conv) = g (the active lag only).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp approximation of max(conv).  Returns (obj, grad_v) where
    the gradient is taken w.r.t. the logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # length 2n‑1
    shift = np.max(conv)                 # for numeric stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # soft‑max over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through the soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float],
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """Continuation L‑BFGS on the log‑sum‑exp surrogate."""
    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun, v, method="L-BFGS-B",
                                jac=True, callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "disp": False,
                                         "ftol": 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the exact max‑convolution
# ----------------------------------------------------------------------


def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 2.5) -> tuple[np.ndarray, float]:
    """Mirror‑descent (EG) on the exact objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    cur_val = best_val
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_exact(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional cheap checkpoint (avoid excessive function calls)
        if it % 200_000 == 0:
            pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000,
                         start_temp: float = 0.05,
                         end_temp: float = 0.0005,
                         start_frac: float = 0.12,
                         end_frac: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Simple SA: move a fraction of mass from i → j.
    Temperature (and step‑size) cool linearly from start_* to end_*.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    cur_val = best_val

    rng = np.random.default_rng(1122334455)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear cooling
        t_frac = it / max_iters
        temp = start_temp * (1 - t_frac) + end_temp * t_frac
        frac = start_frac * (1 - t_frac) + end_frac * t_frac

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] == 0.0:
            continue

        delta = frac * a[i] * rng.random()
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = _objective_exact(cand)
        delta_val = cand_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / max(temp, 1e-12)):
            # accept
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay_rev(n: int, lam: float) -> np.ndarray:
    """Mirror of _exponential_decay."""
    return _exponential_decay(n, lam)[::-1]


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _quadratic_decrease(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic(n: int, k: float, mid: float) -> np.ndarray:
    """Logistic curve, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-k * (i - mid)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _zigzag(v: np.ndarray) -> np.ndarray:
    """Interleave largest / smallest values (produces a non‑trivial ordering)."""
    s = np.sort(v)[::-1]               # descending
    out = np.empty_like(s)
    left, right = 0, len(s) - 1
    i = 0
    while left <= right:
        out[i] = s[left]
        i += 1
        left += 1
        if left <= right:
            out[i] = s[right]
            i += 1
            right -= 1
    out = out / out.sum()
    return out


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_linear_decrease(n))
    seeds.append(_quadratic_decrease(n))

    # power‑law sweep
    for p in np.linspace(0.35, 0.75, 9):
        for sh in [0.0, 2.0, 10.0, 30.0, 80.0, 150.0]:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # exponential decays (both directions) – finer λ sweep
    for lam in np.logspace(-4, -0.5, 9):
        seeds.append(_exponential_decay(n, lam))
        seeds.append(_exponential_decay_rev(n, lam))

    # logistic curves
    for k in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for mid in [n * 0.25, n * 0.5, n * 0.75]:
            seeds.append(_logistic(n, k, float(mid)))
            seeds.append(_logistic(n, k, float(mid))[::-1])

    # beta sweep
    for _ in range(30):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, alpha, beta))

    # random Dirichlet draws (different concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(6):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # zig‑zag permutations of a few Dirichlet draws
    for _ in range(12):
        v = _random_dirichlet(n, 0.5, rng)
        seeds.append(_zigzag(v))

    # pairwise mixtures of existing seeds
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(15):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

            else:
                start_vec = global_best_seq.copy()
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.5,
            )
            print(f"[EG {r + 1}/{max_restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate about 20 % of remaining time to SA
        sa_iters = int(min(600_000, remaining * 30_000))  # heuristic
        print("\n--- Simulated‑Annealing fine‑tuning ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            max_iters=sa_iters,
            start_temp=0.04,
            end_temp=0.0003,
            start_frac=0.10,
            end_frac=0.001,
        )
        print(f"[SA] best = {val_sa:.6f}")
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[Update] New best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final ultra‑high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1200, int(min(4000, remaining * 0.05 * 1e5)))
        print("\n--- Final polishing L‑BFGS (ultra‑high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000_000.0, 1_000_000_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
