   m = np.zeros_like(log_a)
   v = np.zeros_like(log_a)
   t=0
   best_seq = start_seq.copy()
   best_val = _objective_surrogate(start_seq)

   while t < max_iters and time.time() < deadline:
       t +=1
       a = _softmax(log_a)
       conv = _conv_fft(a, a)

       # we can fix beta a large number to approximate max gradient.

"""
Advanced search for a low‑autocorrelation non‑negative step function.

The algorithm builds a large and diverse pool of seed vectors, keeps the
best few according to the exact surrogate 2·n·max(conv(a,a)) (with Σa = 1),
and refines each seed through three complementary local‑optimisation stages:

1. Permutation‑based simulated‑annealing hill‑climbing.
2. Adam‑optimiser on a smooth max‑convolution (log‑sum‑exp) objective.
3. Targeted “max‑product” mass‑transfer moves that directly shrink the
   dominant convolution entry.

The final best sequence is polished with a long Adam run and (if time allows)
one more max‑product pass.  The whole routine respects the 1000 s budget,
stores the best sequence in the global variable ``height_sequence_1``,
and returns it as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (Σa = 1) ------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine components, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Extra seed families
# ----------------------------------------------------------------------
def _alternating_binary(n: int) -> np.ndarray:
    """0/1 alternating pattern, normalised."""
    a = np.zeros(n, dtype=np.float64)
    a[::2] = 1.0
    s = a.sum()
    return a / s


def _random_block_pattern(n: int, rng: np.random.Generator,
                          num_blocks: int = 10) -> np.ndarray:
    """Random piecewise‑constant shape with ``num_blocks`` blocks."""
    if num_blocks >= n:
        return _uniform(n)
    cuts = sorted(rng.choice(np.arange(1, n), size=num_blocks - 1, replace=False))
    indices = [0] + cuts + [n]
    a = np.empty(n, dtype=np.float64)
    for i in range(num_blocks):
        start, end = indices[i], indices[i + 1]
        val = rng.random() + 0.1  # keep >0
        a[start:end] = val
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Permutation‑based hill climbing (simulated annealing)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Swap‑heavy annealing on the surrogate."""
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # Gaussian jitter
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              beta: float = 1000.0,
              lr: float = 0.05,
              max_iters: int = 800_000,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smooth max of the convolution."""
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # stabiliser
        exp_beta = np.exp(beta * (conv - shift))
        w = exp_beta / exp_beta.sum()            # smooth arg‑max weights

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)              # gradient on log‑probabilities

        # Adam update on the log‑space variables
        m = beta1 * m + (1 - beta1) * grad_v
        v = beta2 * v + (1 - beta2) * (grad_v ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted max‑product mass‑transfer hill climbing
# ----------------------------------------------------------------------
def _max_product_hill(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly identifies the dominant convolution contribution
    (the product a_i*a_j that belongs to the maximal conv entry)
    and moves a tiny fraction of mass from the larger factor to a random
    low‑mass position.  This directly shrinks the current peak.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(cur, cur)
        k = int(np.argmax(conv))

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idxs = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idxs
        products = cur[idxs] * cur[rev]

        # index of the worst (largest) product
        max_pos = int(np.argmax(products))
        i_max = idxs[max_pos]
        j_max = rev[max_pos]

        # pick a target index with relatively small weight
        # (simple uniform draw that is unlikely to be i_max or j_max)
        t = rng.integers(0, n)
        while t == i_max or t == j_max:
            t = rng.integers(0, n)

        # move a tiny fraction of mass from the larger of the two contributors
        donor = i_max if cur[i_max] >= cur[j_max] else j_max
        delta = 0.001 * cur[donor]      # 0.1 % of the donor weight

        if delta <= 0.0:
            continue

        new = cur.copy()
        new[donor] -= delta
        new[t] += delta

        val = _objective_surrogate(new)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best = new.copy()
                best_val = val

    return best, best_val
