def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 8192  # choose based on time

    # Create seeds
    seeds = []
    # uniform
    seeds.append(np.full(n, 1.0/n))
    # triangular
    # symmetrical triangular shape
    i = np.arange(n)
    c = (n-1)/2
    tri = 1 - np.abs(i-c)/c
    tri[tri<0]=0
    seeds.append(tri/tri.sum())

    # Sine windows gamma
    for gamma in np.linspace(0.5, 4.0, 15):
        a = np.sin(np.pi * (i+1) / (n+1))**gamma
        seeds.append(a/a.sum())
    # Gaussian
    for sigma_frac in np.linspace(0.05, 0.35, 15):
        mu = (n-1)/2
        sigma = sigma_frac*n
        a = np.exp(-((i-mu)**2)/(2*sigma**2))
        seeds.append(a/a.sum())
    # Exp decay
    for r in np.logspace(-6, -0.5, 30):
        a = np.exp(-r*i)
        seeds.append(a/a.sum())
        a = np.exp(-r*np.abs(i-(n-1)/2))
        seeds.append(a/a.sum())
    # Beta shape random
    for _ in range(100):
        alpha = rng.uniform(0.2,5.0)
        beta = rng.uniform(0.2,5.0)
        a = ( (i+1)**(alpha-1) * (n-i)**(beta-1) )
        seeds.append(a/a.sum())
    # Random Dirichlet with varied concentration
    for conc in [0.01,0.05,0.1,0.2,0.5,1.0,2.0]:
        seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))
    # Random exponential draws
    for _ in range(20):
        seeds.append(rng.exponential(scale=1.0,size=n))
    # Lowfreq curves (cosine)
    for nf in range(2,7):
        for _ in range(5):
            a = np.zeros(n)
            t = np.arange(n)
            for k in range(1,nf+1):
                phase = rng.uniform(0,2*np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp*np.cos(2*np.pi*k*t/n + phase)
            a = np.maximum(a,0)
            seeds.append(a/a.sum() if a.sum()>0 else np.full(n,1.0/n))
    # Kaiser windows if SciPy available
    try:
        from scipy.signal import windows
        for beta in np.linspace(0.5,10,10):
            a = windows.kaiser(n,beta,sym=False)
            seeds.append(a/a.sum())
    except Exception:
        pass
    # DPSS windows
    try:
        from scipy.signal import windows
        for NW in [2,3,4,5,6,7]:
            a = windows.dpss(n,NW, Kmax=1, sym=False)[0]
            seeds.append(a/a.sum())
    except Exception:
        pass

    # Random permutations of some shapes
    base_shapes = [seeds[2], seeds[3], seeds[5]] # just pick some
    for base in base_shapes:
        for _ in range(3):
            seeds.append(rng.permutation(base))

    # previous best
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (arr.size,): # unknown length maybe different; ignore if diff size
            # if not same length, we may interpolate to n.
            if arr.shape[0] == n:
                seeds.append(arr/arr.sum())
            else:
                # resize via interpolation
                x_old = np.linspace(0,1,arr.shape[0])
                x_new = np.linspace(0,1,n)
                arr_interp = np.interp(x_new, x_old, arr)
                seeds.append(arr_interp/arr_interp.sum())
    # Clip and normalize each seed
    normalized_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        if s.ndim != 1:
            continue
        s = np.maximum(s,0.0)
        s_sum = s.sum()
        if s_sum <= 0:
            continue
        normalized_seeds.append(s/s_sum)

    # Evaluate surrogate for each seed, keep top_k
    evaluated = []
    for vec in normalized_seeds:
        if time.time()>deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec,val))

"""
Propose a low‑autocorrelation step‑function.

Algorithm
---------
1. Generate a large and diverse seed pool (analytic windows, Dirichlet
   draws, random cosine sums, Kaiser/DPSS windows, permutations etc.).
2. Normalise each seed to sum = 1 and evaluate the cheap surrogate
   ``2·n·max(conv(a,a))`` (exact because the sum is 1).  Keep the best few.
3. For each top seed run a three‑stage refinement
   * a permutation‑based simulated‑annealing hill‑climber,
   * exponentiated‑gradient (mirror‑descent) on the true max‑convolution,
   * an optional L‑BFGS optimisation of a smooth log‑sum‑exp surrogate.
4. Keep the best sequence found globally and finish with a long
   exponentiated‑gradient polishing run.
5. Return the sequence (list of floats).  The best sequence is also stored
   in the global variable ``height_sequence_1`` for possible continuation.

The routine respects a total wall‑clock budget (default 1000 s) and always
maintains the current global best, so a timeout will still yield a valid
candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for L‑BFGS smoothing and extra windows)
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

try:                     # pragma: no cover
    from scipy.signal import windows as _win
except Exception:       # pragma: no cover
    _win = None

# ----------------------------------------------------------------------
# Basic numeric utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – valid because sum(seq) = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine components, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _win is None:
        return _uniform(n)
    a = _win.kaiser(n, beta, sym=False)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, NW: float) -> np.ndarray:
    if _win is None:
        return _uniform(n)
    a = _win.dpss(n, NW, Kmax=1, sym=False)[0]
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the objective 2·n·max(conv(a,a)).
    Operates in log‑softmax space so the simplex constraint is enforced.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    eps = 1e-30
    log_a = np.log(a + eps)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs counted twice)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local stochastic search that swaps two positions or perturbs the vector.
    Accepts worsening moves according to an annealing schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:          # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                           # small Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # accept with probability exp(−Δ/T)
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq, cur_val = ns, val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)        # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

            break

        remaining = deadline - time.time()
        # allocate a roughly equal fraction for each remaining seed,
        # keeping a small buffer for the final polish.
        share = remaining * 0.15 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}/{top_k}] time share ≈ {share:.1f}s")

        # a) permutation‑based hill‑climb
        seq_perm, val_perm = _perm_hill_climb(seed_vec.copy(),
                                              n,
                                              sub_deadline,
                                              rng,
                                              max_iters=250_000)

        # b) exponentiated‑gradient refinement
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_perm,
                                               n,
                                               sub_deadline,
                                               max_iters=600_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_perm, val_perm

        # c) optional L‑BFGS smooth optimisation
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1_000.0,
                                                              2_000.0, 5_000.0,
                                                              20_000.0, 50_000.0,
                                                              100_000.0],
                                               max_iter_per_beta=1500)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # choose best among the three refinements
        candidates = [(seq_perm, val_perm),
                      (seq_eg, val_eg),
                      (seq_lbfgs, val_lbfgs)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq, global_best_val = local_best_seq, local_best_val
            print(f"  *** New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long EG polishing (use the remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_200_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store the best sequence for possible future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
