   windows (uniform, triangular, Hamming, Tukey, sine, Gaussian,
   power‑law, exponential, low‑frequency cosine sums, beta‑shapes)
   plus a handful of random Dirichlet/exponential draws.
   All seeds are ℓ₁‑normalised.

2. **Surrogate pre‑ranking** – fast evaluation of the exact
   objective (the “surrogate”) for every seed; the best few are kept.

3. **Multi‑start local refinement** – each top seed is improved by
   three stages:
      * a quick *exponentiated‑gradient* (mirror‑descent) pass,
      * an *Adam* optimiser on a smooth log‑sum‑exp approximation of the
        max‑convolution (continuation on a schedule of β),
      * a short *simulated‑annealing* pass.

4. **Final polishing** – a long Adam run from the current global best,
   followed by a final low‑temperature annealing.

All heavy work respects the global deadline; after the deadline the
routine evaluates the best candidate with the supplied ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS – kept for safety)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    a = np.hamming(n).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalize."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 5.0):
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smooth log‑sum‑exp surrogate of max‑conv
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_total_iters: int = 500_000,
              beta_schedule=None,
              lr: float = 0.5,
              beta1: float = 0.9,
              beta2: float = 0.999,
              epsilon: float = 1e-8):
    """
    Adam optimisation in log‑softmax space on the smooth surrogate

        2·n· (shift + log Σ exp(β·(conv−shift)) / β)

    that approximates the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    log_a = np.log(start_seq + 1e-30)
    a = _softmax(log_a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    for beta in beta_schedule:
        beta_val = float(beta)
        # allocate ~equal number of iterations to each β
        iters_this_beta = max_total_iters // len(beta_schedule)

        for _ in range(iters_this_beta):
            if time.time() > deadline:
                return best_seq, best_val
            it += 1

            # ---- smooth max ----
            conv = _conv_fft(a, a)
            shift = np.max(conv)                 # for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                # soft‑max weights over shifts

            # ---- gradient w.r.t. a ----
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # ---- chain rule to log‑softmax space ----
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
            a = _softmax(log_a)

            # ---- keep best ----
            if (it % 2000) == 0:          # cheap periodic check
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    cur_val = _objective_surrogate(a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.01,
                         T_end: float = 1e-6):
    """Lightweight SA: Gaussian perturbations with linear temperature decay."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    stores the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – 4000 matches the reference implementation
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation – a diverse but bounded pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic families – coarse grids (kept relatively small for speed)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_tukey_window(n, 0.75))

    for gamma in np.linspace(0.5, 5.0, 12):
        seeds.append(_sine_window(n, gamma))

    for sigma_frac in np.linspace(0.05, 0.35, 12):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.0, 4.0, 12):
        seeds.append(_power(n, p))

    for rate in np.logspace(-4, -0.4, 12):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Beta‑shape draws – wide ranges
    for _ in range(30):
        a = rng.uniform(0.05, 8.0)
        b = rng.uniform(0.05, 8.0)
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.001, 0.01, 0.1, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(20):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixed shapes & permutations
    bases = [_gaussian(n, 0.22), _sine_window(n, 1.2), _beta_shape(n, 2.0, 2.0)]
    for base in bases:
        for _ in range(3):
            seeds.append(np.random.permutation(base))
        for mix_alpha in (0.2, 0.5, 0.8):
            curve = _lowfreq_curve(n, 4, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Perturbations of the previously best sequence (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(6):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds (limit to keep time)
    # ------------------------------------------------------------------
    max_seed_eval = 300               # cap to keep within budget
    evaluated = []
    for idx, vec in enumerate(seeds):
        if time.time() > deadline or len(evaluated) >= max_seed_eval:
            break
        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total == 0.0:
            continue
        vec = vec / total
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        # Extreme fallback – uniform distribution
        global_best_seq = _uniform(n)
        global_best_val = _objective_surrogate(global_best_seq)
    else:
        evaluated.sort(key=lambda x: x[1])
        top_k = min(8, len(evaluated))
        top_seeds = [vec for vec, _ in evaluated[:top_k]]

        print("\nTop seeds (surrogate values):")
        for i, (_, v) in enumerate(evaluated[:top_k]):
            print(f"  #{i+1}: {v:.6f}")

        # ------------------------------------------------------------------
        # 3. Initialise global best
        # ------------------------------------------------------------------
        global_best_seq = top_seeds[0].copy()
        global_best_val = _objective_surrogate(global_best_seq)
        print(f"\n[Init] current global best surrogate = {global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 4. Multi‑start refinement on the top seeds
        # ------------------------------------------------------------------
        for si, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline:
                break

            # Allocate a slice of the remaining budget for this seed
            remaining = deadline - time.time()
            # keep about 12 % of the remaining budget per upcoming seed + final polish
            sub_deadline = time.time() + remaining * 0.12 / max(1, (top_k - si))

            # ---- (a) quick exponentiated‑gradient pass ----
            seq_eg, val_eg = _exp_gradient_opt_multi(seed_vec.copy(),
                                                     n,
                                                     sub_deadline,
                                                     max_iters=150_000,
                                                     eta0=5.0)

            # ---- (b) Adam refinement on smooth surrogate ----
            if time.time() < sub_deadline:
                seq_adam, val_adam = _adam_opt(seq_eg,
                                               n,
                                               sub_deadline,
                                               max_total_iters=300_000,
                                               lr=0.5)
            else:
                seq_adam, val_adam = seq_eg, val_eg

            # ---- (c) short simulated‑annealing pass ----
            hill_deadline = sub_deadline - 0.3   # reserve a margin
            if time.time() < hill_deadline:
                seq_sa, val_sa = _simulated_annealing(seq_adam,
                                                      n,
                                                      hill_deadline,
                                                      rng,
                                                      max_attempts=100_000,
                                                      start_sigma=0.004)
            else:
                seq_sa, val_sa = seq_adam, val_adam

            # Choose the best among all refinements
            candidates = [
                (seed_vec, _objective_surrogate(seed_vec)),
                (seq_eg, val_eg),
                (seq_adam, val_adam),
                (seq_sa, val_sa)
            ]
            local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

            print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

            if local_best_val < global_best_val:
                global_best_seq = local_best_seq.copy()
                global_best_val = local_best_val
                print(f"    *** New global best = {global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 5. Final polishing – long Adam run + a final SA pass
        # ------------------------------------------------------------------
        if time.time() < deadline:
            print("\n--- Final long Adam polishing ---")
            final_seq, final_val = _adam_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_total_iters=2_000_000,
                                             lr=0.6)
            if final_val < global_best_val:
                global_best_seq = final_seq.copy()
                global_best_val = final_val
                print(f"[Final Adam] improved to {global_best_val:.6f}")

            if time.time() < deadline:
                print("\n--- Final simulated annealing ---")
                final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                                  n,
                                                                  deadline,
                                                                  rng,
                                                                  max_attempts=150_000,
                                                                  start_sigma=0.001,
                                                                  T_start=5e-3,
                                                                  T_end=1e-8)
                if final_sa_val < global_best_val:
                    global_best_seq = final_sa_seq.copy()
                    global_best_val = final_sa_val
                    print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
