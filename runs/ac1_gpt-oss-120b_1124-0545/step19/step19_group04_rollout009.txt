def _exp_fourier_shape(n, num_freq, rng):
    # generate random amplitude for each frequency (including zero freq)
    # amplitude as normal distribution; phase random.
    # sum over k (including 0) a_k cos(2π k i / n + φ_k)
    t = np.arange(n)
    potential = np.zeros(n)
    for k in range(1, num_freq+1):
        amp = rng.normal(scale=0.5)  # random amplitude
        phase = rng.uniform(0, 2*np.pi)
        potential += amp * np.cos(2*np.pi * k * t / n + phase)
    # optional offset to ensure positivity after exponent
    # exponentiate
    a = np.exp(potential)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s

"""
search for a step‑function (list of non‑negative heights) that minimizes
evaluate_sequence().  The method follows the successful multi‑stage pipeline
used previously (seed generation → smooth L‑BFGS → exponentiated‑gradient
mirror‑descent → stochastic mass‑transfer local search → high‑β polishing →
swap‑optimisation) and adds a few new ingredients:

* a richer seed pool (sin^p, exponential‑Fourier, finer power‑law grid,
  extra low‑frequency cosine mixes)
* longer L‑BFGS runs (more iterations per β)
* more EG restarts and a longer final EG pass
* a post‑swap EG polishing step
* slightly more aggressive swap search (more swaps when time permits)

All work is bounded by the supplied time budget (≈1000 s).  The best
sequence found is stored globally under the name ``height_sequence_1`` for
warm‑starts in future calls.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, p>0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_pow_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^p(π (i+0.5)/n), 0<p."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_fourier_shape(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """
    a_i = exp( Σ_k A_k cos(2π k i / n + φ_k) ), clipped to ≥0 and ℓ₁‑normalised.
    Random amplitudes A_k are drawn from N(0,0.5²) and phases uniform.
    """
    t = np.arange(n, dtype=np.float64)
    potential = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        amp = rng.normal(scale=0.5)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        potential += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    # shift to avoid overflow before exponentiation
    potential -= np.max(potential)
    a = np.exp(potential)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# exact objective & sub‑gradient helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the active lag.
    Returns a vector g with g_j = 2·a_{k‑j} for indices participating in the
    maximal convolution entry (k is the argmax index).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g

# smooth surrogate (log‑sum‑exp) for continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate of the max‑convolution objective."""
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive stochastic local search (mass transfers)
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 150_000,
    step_frac: float = 0.006,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Swap‑based combinatorial optimisation
# ----------------------------------------------------------------------
def _swap_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing on the ordering of a fixed multiset of heights.
    Only indices i and j are swapped; the ℓ₂‑norm (hence the value of M)
    changes only at the two convolution positions 2*i and 2*j, which we
    update in O(1).  The method runs until the time budget expires or the
    iteration limit is reached.
    """
    a = start_a.copy()
    conv = np.convolve(a, a)                     # full convolution (size 2n‑1)
    sum_a = a.sum()
    cur_max = conv.max()
    cur_obj = 2.0 * n * cur_max / (sum_a * sum_a)

    best_a = a.copy()
    best_val = cur_obj

    rng = np.random.default_rng(987654321)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        vi = a[i]
        vj = a[j]
        if vi == vj:
            continue
        d = vj - vi                       # change that will be applied at position i
        pos2i = i + i
        pos2j = j + j
        old2i = conv[pos2i]
        old2j = conv[pos2j]

        # update convolution entries impacted by the swap
        conv[pos2i] = old2i + 2.0 * d * vi + d * d
        conv[pos2j] = old2j - 2.0 * d * vj + d * d

        # perform the swap
        a[i], a[j] = vj, vi

        max_conv = conv.max()
        obj = 2.0 * n * max_conv / (sum_a * sum_a)

        if obj < cur_obj:                 # accept only improving moves
            cur_obj = obj
            if obj < best_val:
                best_val = obj
                best_a = a.copy()
        else:
            # revert everything
            a[i], a[j] = vi, vj
            conv[pos2i] = old2i
            conv[pos2j] = old2j

        if it % 500_000 == 0:
            print(f"[Swap] iter {it:,} – best obj = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed‑pool construction – enriched with new families
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families (always present)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.0004))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # ---- NEW ---- dense grid of power‑law exponents (finer)
    for p in np.linspace(0.10, 2.00, 38):
        seeds.append(_power_law(n, float(p)))

    # sin^p shapes (smooth bell‑like curves)
    for p in np.linspace(0.5, 5.0, 30):
        seeds.append(_sin_pow_shape(n, float(p)))

    # exponential‑Fourier families (random low‑dim Fourier expansions)
    for nf in (4, 6, 8, 10):
        for _ in range(6):
            seeds.append(_exp_fourier_shape(n, nf, rng))

    # random Dirichlet families – diverse concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families – few non‑zero entries
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(6):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine curves (more frequencies than before)
    for nf in (2, 3, 4, 5, 6):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures of existing seeds (helps explore the convex hull)
    base = np.arange(len(seeds))
    for _ in range(150):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(20):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – ensure non‑negativity + ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimizes ``evaluate_sequence``.
    A 1000 s budget is respected; the best sequence is stored globally as
    ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # problem size – keep the same as the reference implementation
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # collect seeds (including warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # quick exact evaluation of seeds, keep the elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # keep the top‑k for heavy L‑BFGS work
    seeds.sort(key=_objective_exact)
    top_k = min(12, len(seeds))
    elite = seeds[:top_k]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=1500,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – multiple exponentiated‑gradient restarts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 20
        iters_per_restart = int(min(400_000, remaining * 0.08 / restarts))
        print("\n--- EG restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.01, 1.2)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.0,
            )
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long EG run from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(8_000_000, remaining * 0.35 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=2.5,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – aggressive stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(300_000, remaining * 90))
        if iters > 0:
            print("\n--- Stochastic local search (mass transfers) ---")
            seq_loc, val_loc = _local_search(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.006,
            )
            print(f"[LocalSearch] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1200, int(min(4000, remaining * 0.07 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – swap‑based ordering optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(3_000_000, remaining * 3_000_000))
        if max_swaps > 0:
            print("\n--- Swap‑optimisation (ordering search) ---")
            seq_swap, val_swap = _swap_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_swaps,
            )
            print(f"[Swap‑optimisation] best = {val_swap:.6f}")
            if val_swap < global_best_val:
                global_best_seq = seq_swap.copy()
                global_best_val = val_swap
                print(f"[Update] New best after swap optimisation = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 7 – post‑swap EG polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(4_000_000, remaining * 0.15 * 1e5))
        if max_iters > 0:
            print("\n--- Post‑swap EG polishing ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=2.5,
            )
            print(f"[Post‑swap EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after post‑swap EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in future invocations
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
