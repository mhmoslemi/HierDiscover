"""
Propose a new step–function sequence.

Strategy
--------
1. **Random‐Dirichlet seeding** – generate many Dirichlet vectors with
   concentrations covering several orders of magnitude (0.001 … 10)
   and keep the best (smallest surrogate value).

2. **Mirror‑descent (exponentiated‑gradient) refinement** – the
   surrogate objective is
        f(a) = 2·n·max(conv(a,a))          (a lives on the simplex)
   The exact sub‑gradient of the active convolution entry is
        g_i = 2·a_{k-i}   for i such that 0 ≤ i , k-i < n,
   where k = argmax(conv(a,a)).  Working in log‑space with a softmax
   automatically respects the non‑negativity and ℓ₁‑normalisation
   constraints.  A decreasing stepsize η_t = η0 / sqrt(t) is used.
   The loop runs until the wall‑time budget expires.

3. **Convex mixing with the uniform vector** – after the EG phase we
   optionally improve the result by a cheap ternary search over the
   mixture α·a + (1‑α)·u (u = uniform, ℓ₁‑normalised).

4. **Final evaluation** – the best sequence found is evaluated with the
   provided ``evaluate_sequence`` routine and stored globally for the
   next run.

The whole routine respects a 1000 s wall‑time budget, prints progress
and returns a list of non‑negative heights (sum ≈ 1).  No large
matrices are built, so it works well inside the limited memory
environment.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers (FFT convolution, softmax, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (∑=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact when ∑a_i = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a sum‑1 vector."""
    n = seq.size
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution objective.

    Returns the best sequence seen and its surrogate value.
    """
    n = start_seq.size
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)                     # stay in log‑softmax space
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current convolution and index of the active entry
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # stepsize schedule η_t = η0 / sqrt(t)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report (lightweight)
        if it % 200_000 == 0:
            print(f"[EG] iter {it:,} – best surrogate = {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Convex mixing (ternary search) with the uniform vector
# ----------------------------------------------------------------------
def _ternary_mix(a: np.ndarray, b: np.ndarray,
                 deadline: float, steps: int = 30):
    """Best convex combination α·a+(1‑α)·b by ternary search on α∈[0,1]."""
    lo, hi = 0.0, 1.0
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = a * m1 + b * (1.0 - m1)
        seq2 = a * m2 + b * (1.0 - m2)
        seq1 /= seq1.sum()
        seq2 /= seq2.sum()
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)

        if val1 < val2:
            hi = m2
            if val1 < best_val:
                best_val, best_seq = val1, seq1
        else:
            lo = m1
            if val2 < best_val:
                best_val, best_seq = val2, seq2

    # final midpoint evaluation
    mid = (lo + hi) / 2.0
    seq_mid = a * mid + b * (1.0 - mid)
    seq_mid /= seq_mid.sum()
    val_mid = _objective_surrogate(seq_mid)
    if val_mid < best_val:
        best_seq, best_val = seq_mid, val_mid

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine follows the strategy
    described above and respects a wall‑time budget of *budget_s*
    seconds.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin for final eval

    # ------------------------------------------------------------------
    # Fixed problem size (same as in the previous run)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution if it exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        cur_best = np.asarray(prev, dtype=np.float64)
        if cur_best.sum() > 0:
            cur_best = cur_best / cur_best.sum()
        else:
            cur_best = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur_best = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    best_val = _objective_surrogate(cur_best)
    print(f"Initial surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Random Dirichlet seeding (broad exploration)
    # ------------------------------------------------------------------
    print("\n--- Random Dirichlet seeding ---")
    # allocate roughly the first 30 % of the budget to this stage
    seed_deadline = start_time + 0.3 * (deadline - start_time)

    # concentrations spanning many orders of magnitude
    while time.time() < seed_deadline:
        # draw a concentration uniformly in log‑space between 1e‑3 and 10
        conc = 10 ** rng.uniform(-3, 1)
        cand = rng.dirichlet(np.full(n, conc))
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val, cur_best = val, cand
            print(f"  new seed found – surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Mirror‑descent (exponentiated‑gradient) refinement
    # ------------------------------------------------------------------
    print("\n--- Exponentiated‑gradient refinement ---")
    cur_best, best_val = _exp_gradient_opt(cur_best, deadline, max_iters=5_000_000,
                                           eta0=5.0)
    print(f"EG finished – best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Optional mixing with uniform (tiny extra gain sometimes)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    if time.time() < deadline:
        cur_best, best_val = _ternary_mix(cur_best, uniform, deadline, steps=30)
        print(f"After mixing with uniform – surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(cur_best.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    # Store for the next run
    globals()["height_sequence_1"] = cur_best.tolist()
    return cur_best.tolist()
