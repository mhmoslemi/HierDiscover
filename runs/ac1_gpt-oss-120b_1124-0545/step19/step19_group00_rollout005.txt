def _adam_opt(start_seq, n, deadline, max_iters=200_000, lr=0.2, beta_start=1.0, beta_factor=1.5, beta_max=20000.0, beta_step=5000):
    a = start_seq.copy()
    v = np.log(a + 1e-30)
    m = np.zeros_like(v)
    v_exp = np.zeros_like(v)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    beta = beta_start
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        cur_val = 2 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        shift = max_conv
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]  # shape n
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam updates
        m = 0.9 * m + 0.1 * grad_v
        v_exp = 0.999 * v_exp + 0.001 * (grad_v * grad_v)  # Actually using beta2 = 0.999
        # compute bias-corrected first and second mhat, vhat
        m_hat = m / (1 - 0.9**it)
        v_hat = v_exp / (1 - 0.999**it)
        v = v - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # increase beta occasionally
        if it % beta_step == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return best_seq, best_val

"""
Propose a candidate step‑function by (1) generating many simple analytic
distributions, (2) keeping a few with the best surrogate value
    S(a) = 2·n·max(conv(a,a))      (a is ℓ₁‑normalised),
and (3) polishing each with a fast Adam optimiser that minimises a
log‑sum‑exp smooth approximation of the max‑convolution.
The optimiser works in the log‑softmax (mirror‑descent) space so the
ℓ₁‑constraint is satisfied automatically.  A temperature‑like
parameter β controls the smoothness; it is increased gradually so the
procedure first makes large‑scale changes and later focuses on the true
maximum.  The best sequence seen during the whole run is stored in the
global variable ``height_sequence_1`` and returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """exact objective for a vector normalised to sum=1: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return a *probability* vector (ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet with the same concentration for all coordinates."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr: float = 0.2,
                   beta_start: float = 1.0,
                   beta_factor: float = 1.5,
                   beta_max: float = 20_000.0,
                   beta_update_every: int = 5_000) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent/Adam on the log‑sum‑exp surrogate of max conv.
    Returns the best (ℓ₁‑normalised) vector found and its exact surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # work in log‑softmax space
    m1 = np.zeros_like(v)                 # first moment
    m2 = np.zeros_like(v)                 # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    beta = beta_start
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # Current probabilities
        a = _softmax(v)

        # Convolution and exact max (needed for shift)
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)

        # Keep the best exact surrogate seen
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ----- gradient of smoothed max -----
        shift = max_conv                       # numeric stabiliser
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp                # soft‑max weights over the 2n‑1 entries

        rev_a = a[::-1]
        # grad_a = 2·(rev_a * w)  (full convolution, then central n entries)
        full = _conv_fft(rev_a, w)            # length = 2n‑1
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update in log‑softmax space -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v * grad_v)

        m1_hat = m1 / (1.0 - beta1 ** it)
        m2_hat = m2 / (1.0 - beta2 ** it)

        v -= lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # ----- increase smoothness (β) gradually -----
        if it % beta_update_every == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    # final evaluation of the best sequence
    best_val = _objective_surrogate(best_seq)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The algorithm:

        1. create a large pool of simple analytic or random “seed” vectors,
           all ℓ₁‑normalised;
        2. keep the few best seeds according to the cheap surrogate
           2·n·max(conv);
        3. starting from each seed, run the Adam optimiser described above
           (log‑sum‑exp surrogate) for a time‑budgeted number of iterations;
        4. keep the best sequence ever seen and store it globally.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation / printing
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – keep the same length as in the previous run
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve the previous best sequence (if any) and normalise it
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                s = arr.sum()
                if s > 0:
                    prev_vec = arr / s
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Seed generation – a fairly diverse collection
    # ------------------------------------------------------------------
    seeds = []

    # a handful of analytic families (including the heavily U‑shaped Beta)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for a in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0):
        for b in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # low‑frequency random curves – they tend to produce smoother, less
    # “triangular” convolution patterns
    for nf in (2, 3, 4, 5, 6):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random Dirichlet draws with a spread of concentration parameters
    for conc in (0.05, 0.1, 0.3, 0.7, 1.5, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # include the previous best (if any) together with a few perturbed copies
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(5):
            noise = rng.normal(scale=0.003, size=n)
            noisy = np.maximum(prev_vec + noise, 0.0)
            s = noisy.sum()
            if s > 0:
                seeds.append(noisy / s)

    # ------------------------------------------------------------------
    # 2. Surrogate evaluation of all seeds (cheap) and keep the top‑few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(4, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed evaluation (surrogate) ---")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  Seed #{i + 1}: surrogate = {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam optimisation from each top seed (time‑budgeted)
    # ------------------------------------------------------------------
    best_seq = None
    best_surrogate = float('inf')

    # allocate roughly equal time for each seed optimisation run
    remaining_seeds = len(top_seeds)
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        # give each run a slice of the remaining budget (at least a few seconds)
        slice_deadline = min(deadline,
                             time.time() + (deadline - time.time()) / max(remaining_seeds, 1))
        print(f"\n--- Optimising seed #{idx + 1} (remaining seeds: {remaining_seeds}) ---")
        opt_seq, opt_val = _adam_optimize(seed_vec, n, slice_deadline,
                                          max_iters=300_000,
                                          lr=0.3,
                                          beta_start=1.0,
                                          beta_factor=1.5,
                                          beta_max=100_000.0,
                                          beta_update_every=4_000)
        print(f"    Optimisation result surrogate = {opt_val:.6f}")
        if opt_val < best_surrogate:
            best_surrogate = opt_val
            best_seq = opt_seq.copy()
        remaining_seeds -= 1

    # ------------------------------------------------------------------
    # 4. If time permits, perform a few random‑restart short runs
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # short run budget (≈ 5 s)
        run_deadline = time.time() + 5.0
        rand_vec = _random_dirichlet(n, conc=rng.uniform(0.02, 2.0), rng=rng)
        opt_seq, opt_val = _adam_optimize(rand_vec, n, run_deadline,
                                          max_iters=80_000,
                                          lr=0.3,
                                          beta_start=1.0,
                                          beta_factor=1.5,
                                          beta_max=50_000.0,
                                          beta_update_every=2_000)
        if opt_val < best_surrogate:
            best_surrogate = opt_val
            best_seq = opt_seq.copy()
            print(f"\n[Random restart] new best surrogate = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation (exact objective)
    # ------------------------------------------------------------------
    true_val = None
    if best_seq is not None:
        try:
            true_val = evaluate_sequence(best_seq.tolist())
            print(f"\n=== Finished ===\nBest surrogate ≈ {best_surrogate:.6f}, "
                  f"true evaluate_sequence = {true_val:.6f}")
        except Exception as exc:      # pragma: no cover
            print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the best sequence for the next run
    globals()["height_sequence_1"] = best_seq.tolist() if best_seq is not None else []
    return best_seq.tolist() if best_seq is not None else []
