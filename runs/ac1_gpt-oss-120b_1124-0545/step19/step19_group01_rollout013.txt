def _anneal_refine(start_seq, n, deadline, rng, max_iters=200_000):
   cur = start_seq.copy()
   cur_val = _objective_surrogate(cur)
   best_seq = cur.copy()
   best_val = cur_val
   # schedule
   T0 = cur_val * 0.01   # temperature initial maybe 1% of objective
   Tmin = cur_val * 1e-6
   for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        T = max(Tmin, T0 * (1 - it / max_iters))  # linear cooling
        # decide move
        if rng.random() < 0.5: # swap
            i,j = rng.integers(n, size=2)
            if i==j: continue
            ns = cur.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:
            # perturb a few entries
            idxs = rng.choice(n, size=2, replace=False)
            delta = rng.normal(scale=0.01)  # maybe adjust
            ns = cur.copy()
            ns[idxs[0]] += delta
            ns[idxs[1]] -= delta
            # ensure nonneg
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s==0: continue
            ns = ns / s
        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0 and rng.random() < math.exp(-(val-cur_val)/T)):
            cur = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
   return best_seq, best_val
```

"""
Propose a step‑function sequence (non‑negative heights) that minimises
``evaluate_sequence``.  The algorithm is a multi‑stage stochastic optimiser:

1. **Rich seed pool** – analytic families (uniform, exponential, Gaussian,
   power‑law, beta‑shape, low‑frequency cosine) plus random Dirichlet draws,
   random piecewise‑linear curves and a few very sparse Dirichlet samples.
   For each family many random permutations are generated; the surrogate
   objective (2·n·max(conv(a,a))) is evaluated cheaply using FFT.

2. **Selection of promising seeds** – the best `top_k` seeds (by surrogate)
   are kept for intensive refinement.

3. **Permutation‑based simulated annealing** – for each selected seed a
   hill‑climbing / simulated‑annealing loop performs swaps of two positions
   and small Gaussian perturbations of the values.  The temperature is
   linearly cooled; the surrogate is evaluated after every move and the
   best sequence seen is remembered.

4. **Exponentiated‑gradient polishing** – starting from the best permutation
   result, mirror‑descent (a sub‑gradient method on the exact max‑convolution)
   is applied for many iterations.  This directly reduces the peak of the
   self‑convolution.

5. **Optional L‑BFGS smoothing** – if SciPy is present, the smoothed
   max‑convolution (log‑sum‑exp) is minimised with L‑BFGS‑B for a short
   final touch.

6. **Final long EG polishing** – the current global best sequence is again
   refined with a long exponentiated‑gradient run.

All steps respect the overall 1000 s budget; the globally best sequence is
stored in ``height_sequence_1`` for possible continuation runs.

The code contains no closures or lambdas, only top‑level functions, and
uses only `numpy` (and optionally `scipy`).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Very sparse Dirichlet draw – keep only the top 20 % of entries."""
    vec = rng.dirichlet(np.full(n, conc))
    thresh = np.percentile(vec, 80)   # keep the largest 20 %
    vec = np.where(vec >= thresh, vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return _uniform(n)
    return vec / total


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine components, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑linear curve with ``num_breaks`` interior break‑points."""
    # pick break points (including the two ends)
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    # random heights at the break points
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution (sub‑gradient on the active index)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two positions or perturbs the vector.
    Accepts worsening moves with a simple annealing schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing schedule parameters
    T_start = 1e-3 * cur_val          # temperature proportional to objective magnitude
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # linear temperature cooling
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # small Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # accept worse move with probability exp(−Δ/T)
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val

# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp) – kept from baseline
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,

    """

    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The workflow:
      * generate a huge pool of analytic / random seeds,
      * keep the best few (by surrogate objective),
      * for each: permutation annealing → exponentiated‑gradient polishing,
      * finally a long EG polishing on the global best.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – 4000 is a good compromise between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (helps continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Massive seed generation (many families + random permutations)
    # ------------------------------------------------------------------
    seeds = []

    # Analytic families – a fairly dense grid
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.1, 3.5, 20):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma_frac in np.linspace(0.05, 0.35, 20):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.0, 4.0, 20):
        seeds.append(_power(n, p))

    # Exponential decay – finer sweep
    for rate in np.logspace(-5, -0.3, 40):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Beta‑shape random sampling (more of them)
    for _ in range(120):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws – broad concentration spectrum
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(15):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Very sparse Dirichlet draws (aimed at tiny support)
    for conc in [0.001, 0.005, 0.01]:
        for _ in range(8):
            seeds.append(_random_sparse_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(40):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(10):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Piecewise‑linear random curves
    for breaks in (3, 5, 8):
        for _ in range(12):
            seeds.append(_piecewise_linear(n, breaks, rng))

    # Mixtures of uniform & various shapes
    mix_alphas = (0.2, 0.5, 0.8)
    for alpha in mix_alphas:
        for rate in (0.001, 0.005, 0.02):
            curve = _exp_decay(n, rate)
            seeds.append(_mix_shapes(_uniform(n), curve, alpha))

    # Random permutations of a few promising bases (break symmetry aggressively)
    base_shapes = [_exp_decay(n, 0.003), _sym_exp_decay(n, 0.004),
                   _beta_shape(n, 2.0, 2.0), _gaussian(n, 0.18)]
    for base in base_shapes:
        for _ in range(12):
            seeds.append(np.random.permutation(base))

    # Include the previous best (if any) and copies with small perturbations
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(8):
            seeds.append(_perturb(prev_arr, rng, sigma=0.004))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds – keep the top‑k
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (permutation hill‑climb + EG + optional L‑BFGS)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # give each remaining seed roughly 12 % of the left time
        share = remaining * 0.12 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] start refinement, time budget ≈ {share:.1f}s")

        # ---- (a) permutation‑based hill‑climbing ----
        seq_perm, val_perm = _perm_hill_climb(seed_vec.copy(),
                                              n,
                                              sub_deadline,
                                              rng,
                                              max_iters=220_000)

        # ---- (b) exponentiated‑gradient polishing ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_perm,
                                               n,
                                               sub_deadline,
                                               max_iters=350_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_perm, val_perm

        # ---- (c) optional L‑BFGS smooth optimisation ----
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1_000.0,
                                                              2_000.0, 5_000.0,
                                                              20_000.0, 50_000.0,
                                                              100_000.0, 200_000.0,
                                                              500_000.0, 1_000_000.0],
                                               max_iter_per_beta=2000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # ---- (d) a short second permutation refinement (helps escape ordering traps) ----
        if time.time() < sub_deadline:
            seq_perm2, val_perm2 = _perm_hill_climb(seq_lbfgs.copy(),
                                                   n,
                                                   sub_deadline,
                                                   rng,
                                                   max_iters=80_000)
        else:
            seq_perm2, val_perm2 = seq_lbfgs, val_lbfgs

        # Pick the best of all refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_perm, val_perm),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_perm2, val_perm2)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long EG polishing (as much time as left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_400_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. One final short permutation tweak (tiny budget, may shave a bit)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 30.0   # 30 s max
        seq_last, val_last = _perm_hill_climb(global_best_seq.copy(),
                                              n,
                                              short_deadline,
                                              rng,
                                              max_iters=100_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final permutation] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
