def _adam_optimize(start_vec, n, deadline, beta_sched, iters_per_beta):
   - initialize logits = np.log(start_vec + eps)
   - m = zeros_like(logits); v = zeros_like(logits)
   - t = 0
   - for beta in beta_sched:
        for i in range(iters_per_beta):
            t += 1
            if time.time()>deadline: break loops
            a = _softmax(logits)
            conv = _conv_fft(a,a)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

"""
propose_candidate
-----------------

A self‑contained, gradient‑based optimiser that works directly on the
ℓ₁‑normalised (sum = 1) step‑function.  The algorithm

1. builds a modest collection of symmetric analytic families together with
   many random Dirichlet / exponential seeds,
2. evaluates a cheap surrogate  J(f)=2·n·max(conv(f,f))  and keeps the
   best few,
3. refines the best seed with a continuation Adam optimiser that minimises
   a smooth log‑sum‑exp surrogate of the max‑convolution.
   The optimisation is performed in log‑softmax (logits) space, guaranteeing
   non‑negativity and the simplex constraint, and the logits are forced to
   stay symmetric at every iteration.
4. optionally perturbs the obtained solution a few times and re‑optimises,
   keeping the best result,
5. finally calls the provided ``evaluate_sequence`` to obtain the true
   score and stores the solution for later runs.

The whole pipeline respects the 1000 s wall‑time limit, never uses the
memory‑intensive linear‑programming step of the previous implementation,
and therefore can afford many more optimisation iterations, which typically
pushes the objective below the previous 1.5085 value.
"""

# ----------------------------------------------------------------------
# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(L log L))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _symmetrize(vec: np.ndarray) -> np.ndarray:
    """Make a vector symmetric (a[i] = a[-i-1]) and renormalise to sum=1."""
    vec = (vec + vec[::-1]) / 2.0
    s = vec.sum()
    if s <= 0.0:
        return np.full_like(vec, 1.0 / len(vec))
    return vec / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (all return ℓ₁‑normalised symmetric vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)               # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return _symmetrize(rng.dirichlet(np.full(n, conc)).astype(np.float64))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _symmetrize(a.astype(np.float64))


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _symmetrize(a)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, keep non‑negative, re‑normalise (preserves symmetry)."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    noisy = _symmetrize(noisy)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1000000.0]

    eps = 1e-30
    logits = np.log(start_vec + eps)          # initial logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    t_iter = 0
    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break

            # 1) obtain current probability vector (ℓ₁‑normalised)
            a = _softmax(logits)
            # enforce symmetry (helps flatten the autocorrelation)
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()

            # 2) convolution and smooth‑max computation
            conv = _conv_fft(a, a)                     # length 2n‑1
            max_conv = np.max(conv)
            cur_val = 2.0 * n * max_conv
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

            shift = max_conv
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                      # smoothed weights, sum=1

            # 3) gradient w.r.t. a (see analysis notes)
            grad_a = 2.0 * _conv_fft(a[::-1], w)[n - 1:n - 1 + n]

            # 4) gradient w.r.t. logits (softmax Jacobian)
            grad_logit = a * (grad_a - np.dot(grad_a, a))

            # 5) Adam update in logits space
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_logit
            v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            lr = 0.05 / (np.sqrt(t_iter) + 1.0)   # modest decay
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

            # 6) keep logits symmetric
            logits = (logits + logits[::-1]) / 2.0

    # final projection
    final_a = _softmax(logits)
    final_a = (final_a + final_a[::-1]) / 2.0
    final_a = final_a / final_a.sum()
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_val = final_val
        best_seq = final_a.copy()

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method builds a pool of symmetric analytic
    and random seeds, selects the most promising, and refines it with a
    continuation Adam optimiser that works directly on a smooth surrogate of
    the max‑convolution objective.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # Problem size – keep the same n as in earlier runs (4000)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if present (symmetrised to the current n)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr.copy()
            else:
                # if length differs, just ignore it
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Assemble a diverse set of symmetric seed vectors
    # ------------------------------------------------------------------
    seeds = []

    # Analytic families (coarse grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 14):
        seeds.append(_gaussian(n, sigma))
    for sigma in (0.10, 0.12):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))
    for p in np.linspace(0.0, 2.0, 13):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))
    for alpha in np.linspace(0.1, 0.9, 7):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet / exponential / low‑frequency seeds
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mix a few random low‑frequency curves with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_symmetrize(mix_alpha * _uniform(n) + (1.0 - mix_alpha) * curve))

    # Include the previous best (if any) and a few perturbations of it
    if prev_arr is not None:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(3):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Evaluate the cheap surrogate on all seeds, keep a few best
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed evaluation (surrogate) ---")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  seed #{i+1}: surrogate = {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Optimise the best seed with continuation Adam
    # ------------------------------------------------------------------
    # use the absolute best raw seed as starting point
    init_seq = evaluated[0][0]
    print("\n--- Starting Adam optimisation ---")
    seq_opt, val_opt = _adam_optimize(init_seq, n, deadline,
                                      iters_per_beta=3000)

    # ------------------------------------------------------------------
    # 4. Optional perturb‑and‑re‑optimise phases (if time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        # small random perturbation of current best
        pert = _perturb(seq_opt, rng, sigma=0.0015)
        seq_new, val_new = _adam_optimize(pert, n, deadline,
                                          iters_per_beta=1500)
        if val_new < val_opt:
            print(f"  Improved after perturbation: {val_new:.6f} < {val_opt:.6f}")
            seq_opt, val_opt = seq_new, val_new
        else:
            break   # no further gain found

    # ------------------------------------------------------------------
    # 5. Official evaluation with the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq_opt.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_opt:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run
    globals()["height_sequence_1"] = seq_opt.tolist()
    return seq_opt.tolist()
