"""
optimize_step_function – fast refinement of a known best step‑function

The routine assumes that a (possibly good) sequence is already stored in the
global variable ``height_sequence_1`` (from a previous run).  It concentrates
on *re‑ordering* the multiset of values (permutation annealing) and then
performs a final exponentiated‑gradient (mirror‑descent) polishing of the
exact max‑convolution objective.

If no previous solution exists a simple exponential‑decay shape is used as a
fallback start.  The algorithm respects the 1000‑second budget, keeps the
globally best sequence, and finally returns it as a plain ``list[float]``.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (used when the ℓ₁‑norm equals 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families (needed only when no previous solution)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution (sub‑gradient on the
    active index).  Returns the best sequence seen and its surrogate value."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation annealing (swap + segment‑reverse moves)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 rng: np.random.Generator,
                 max_iters: int = 300_000):
    """
    Simulated‑annealing style search that only re‑orders the entries.
    Swaps and short segment reversals are proposed; the algorithm keeps the
    best sequence found.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (log‑linear)
    T_start = 1e-3 * cur_val
    T_end = 1e-8 * cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.6:          # swap move (most frequent)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                           # short segment reversal
            i = rng.integers(0, n)
            # choose a segment length with bias toward short segments
            max_len = min(50, n - i)          # at most 50 elements
            seg_len = rng.integers(1, max_len + 1)
            j = i + seg_len - 1
            ns = cur_seq.copy()
            ns[i:j + 1] = ns[i:j + 1][::-1]

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # metropolis acceptance
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    Strategy:
      1. Load the previously best sequence (global ``height_sequence_1``).  If it
         does not exist, initialise with a simple exponential‑decay shape.
      2. Run a permutation‑annealing phase that only re‑orders the values.
         Swaps and short segment reversals are used; a temperature schedule
         permits occasional uphill moves.
      3. Polish the resulting sequence with exponentiated‑gradient mirror‑descent
         on the exact max‑convolution sub‑gradient.
      4. (If time permits) a smoothing L‑BFGS optimisation as a final touch.
      5. Store and return the best sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – keep the same dimension as the previous best.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 1. Load previous solution or initialise a fallback.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            seq0 = np.asarray(prev, dtype=np.float64)
            if seq0.shape != (n,):
                # If the stored sequence has a different length, fall back.
                raise ValueError
        except Exception:
            seq0 = None
    else:
        seq0 = None

    if seq0 is None:
        # Simple exponential decay gives a reasonable starting point.
        seq0 = _exp_decay(n, rate=0.003)

    # Normalise to ℓ₁‑norm = 1, clip negatives (should not happen).
    seq0 = np.maximum(seq0, 0.0)
    total = seq0.sum()
    if total == 0.0:
        seq0 = _uniform(n)
    else:
        seq0 = seq0 / total

    print(f"[Init] surrogate of starting sequence = {_objective_surrogate(seq0):.6f}")

    # --------------------------------------------------------------
    # 2. Permutation annealing (re‑ordering only)
    # --------------------------------------------------------------
    perm_deadline = deadline - 150.0          # keep ~150 s for polishing
    seq_perm, val_perm = _perm_anneal(seq0, n, perm_deadline, rng,
                                      max_iters=300_000)
    print(f"[PermAnneal] best surrogate after re‑ordering = {val_perm:.6f}")

    # --------------------------------------------------------------
    # 3. Exponentiated‑gradient polishing (exact max‑convolution)
    # --------------------------------------------------------------
    seq_eg, val_eg = _exp_gradient_opt(seq_perm, n, deadline,
                                       max_iters=2_200_000, eta0=5.0)
    print(f"[EG] surrogate after mirror‑descent = {val_eg:.6f}")

    # --------------------------------------------------------------
    # 4. Optional L‑BFGS smooth optimisation (if time remains)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(seq_eg, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate after smoothing = {val_lb:.6f}")
        if val_lb < val_eg:
            best_seq = seq_lb
            best_val = val_lb
        else:
            best_seq = seq_eg
            best_val = val_eg
    else:
        best_seq = seq_eg
        best_val = val_eg

    # --------------------------------------------------------------
    # 5. Final evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next run (facilitates incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
