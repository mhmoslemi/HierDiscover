Algorithm
---------

   (uniform, triangular, Gaussian, beta, exponential, sine, semi‑circle,
   polynomial tent, power‑law, random Dirichlet, sparse, low‑frequency sums).
   The best few seeds are kept as *elite*.

2. **Smooth optimisation – Adam** – for every elite seed we minimise a
   log‑sum‑exp smooth surrogate of the max‑convolution using an Adam
   optimiser.  The smoothing parameter β is increased in a continuation
   schedule; for each β we perform many Adam steps (far more than the
   previous L‑BFGS run) so that the iterates approach the true non‑smooth
   optimum more closely.

3. **Mirror‑descent (exponentiated‑gradient)** – a handful of EG restarts
   from random Dirichlet draws and a long EG run from the current best
   sequence.  This directly minimises the exact objective using the
   sub‑gradient of the active lag.

   removes a small amount of mass *simultaneously* from the two indices
   that form the active convolution lag and redistributes it elsewhere.

# ----------------------------------------------------------------------
# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (with ∑a assumed >0)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (active lag only)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g = np.zeros_like(a)
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad) where ``obj`` is the smooth surrogate
    2·n·(shift + log Σ exp(beta·(conv‑shift))/beta) and ``grad`` is the
    gradient w.r.t. the logits ``v`` (so that a = softmax(v)).
    """
    a = _softmax(v)                     # ℓ₁‑normalised (sum = 1)
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # soft‑max over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule: list[float] | None = None,
                   steps_per_beta: int = 2000) -> np.ndarray:
    """
    Continuation Adam on the smooth surrogate.
    ``start_vec`` must be ℓ₁‑normalised (sum = 1).
    Returns a probability vector (ℓ₁‑normalised).
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0, 50_000_000.0, 100_000_000.0,
            500_000_000.0, 1_000_000_000.0, 5_000_000_000.0,
            10_000_000_000.0,
        ]

    # initialise Adam state on logits
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    lr_base = 0.05  # base learning rate for logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            t += 1
            obj, grad = _smooth_objective(v, n, beta)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad * grad)

            m_corr = m / (1.0 - beta1 ** t)
            v_corr = v_hat / (1.0 - beta2 ** t)

            lr = lr_base / np.sqrt(beta)  # modest decay with β
            v -= lr * m_corr / (np.sqrt(v_corr) + eps)

    return _softmax(v)

# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns (best_sequence, best_value).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    rng = np.random.default_rng(12345)
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val

def _local_search(start_a: np.ndarray,
                  n: int,
                  deadline: float,
                  iters: int = 100_000,
                  step_frac: float = 0.012) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb (single‑index moves)."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(98765)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

def _pairwise_local_search(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           iters: int = 60_000,
                           step_frac: float = 0.008) -> tuple[np.ndarray, float]:
    """
    Pairwise hill‑climb: take a small amount of mass from the two
    indices that form the active lag (i, rev = idx-i) and relocate the
    total mass elsewhere.  This reduces the product contributing to the
    current maximum more quickly than single‑index moves.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(192837)

    for _ in range(iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i = rng.integers(i_start, i_end + 1)
        rev = idx - i
        # ensure we have positive mass at both ends
        if a[i] <= 0.0 or a[rev] <= 0.0:
            continue
        # amount we can safely take (avoid negativity)
        delta = step_frac * min(a[i], a[rev])
        if delta <= 0.0:
            continue
        # pick a destination far from the active interval (helps not to
        # create a new peak immediately)
        j = rng.integers(0, n)
        if j == i or j == rev:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[rev] -= delta
        cand[j] += 2.0 * delta
        # safety – clip any tiny negative due to numeric issues
        np.clip(cand, 0.0, None, out=cand)
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _semicircle_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (i - c) / c
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _poly_tent(n: int, power: float = 1.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) * (n - i)) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse pool of ℓ₁‑normalised vectors of length ``n``.
    """
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_sin_shape(n))
    seeds.append(_semicircle_shape(n))
    seeds.append(_poly_tent(n, power=1.2))
    seeds.append(_poly_tent(n, power=2.0))

    # power‑law sweep
    p_vals = np.linspace(0.2, 0.9, 15)
    shifts = [0.0, 5.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # random Dirichlet (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency sums
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixture of existing seeds
    base = np.arange(len(seeds))
    for _ in range(150):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of a random subset
    for _ in range(12):
        idx = rng.integers(len(seeds))
        sh = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], sh))

    # previous best, if any
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final sanitisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned
