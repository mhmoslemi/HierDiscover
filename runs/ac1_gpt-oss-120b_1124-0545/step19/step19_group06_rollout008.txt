"""
propose_candidate
=================

A self‑contained, time‑bounded optimiser for the “max‑convolution” bound.
The optimiser builds a very diverse seed pool (analytic families, Dirichlet
samples, mixtures and rotations), evaluates them exactly and keeps the elite
vectors.  Each elite is polished by a continuation L‑BFGS run on a
log‑sum‑exp smooth surrogate of the true objective, then a long
exponentiated‑gradient (mirror‑descent) phase, followed by a high‑temperature
simulated‑annealing mass‑transfer hill‑climb and a small active‑set linear‑
programming refinement.  The schedule uses the full 1000 s wall‑clock budget,
and the best sequence found is stored globally as ``height_sequence_1`` for
future warm‑starts.

All helper functions are defined at top‑level, no closures or lambdas are
used, and only ``numpy``, ``scipy`` and the provided ``evaluate_sequence``
are imported.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# FFT‑based convolution (real, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective and sub‑gradient (for mirror descent)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """A sub‑gradient of max(conv(a,a)) w.r.t. a (active lag)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth surrogate of the max‑convolution objective.
    Returns (value, gradient w.r.t. v).
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full convolution, length 2n‑1
    shift = np.max(conv)                 # avoid overflow
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # surrogate objective = 2 n (shift + log Σ exp / β)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lag indices
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]  # slice of length n

    # back‑prop through softmax (log‑sum‑exp trick)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1200,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    ``start_vec`` must be ℓ₁‑normalised.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # unconstrained logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(_):
            # permits early abort
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it + 1.0)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if (it & 1023) == 0:  # every 1024 iterations
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer hill‑climb
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 300_000,
    init_temp: float = 2.0,
) -> tuple[np.ndarray, float]:
    """Stochastic local search with exponential cooling."""
    a = start_a.copy()
    cur_val = _objective_exact(a)
    best_a = a.copy()
    best_val = cur_val

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        # transfer a random fraction (up to 5 %) from i to j
        frac = rng.random() * 0.05
        delta = frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = _objective_exact(cand)

        if cand_val < cur_val:
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_a = cand.copy()
                best_val = cand_val
        else:
            progress = it / max_iters
            T = init_temp * max(0.05, 1.0 - progress)  # linear decay, floor 5%
            if rng.random() < np.exp(-(cand_val - cur_val) / T):
                a = cand
                cur_val = cand_val

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming active‑set refinement (high‑β bound)
# ----------------------------------------------------------------------
def _lp_refine(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_steps: int = 5,
    top_k: int = 250,
    mix_ratio: float = 0.12,
) -> tuple[np.ndarray, float]:
    """One‑shot LP that maximises Σg_j under the active convolution bound."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for step in range(max_steps):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = np.max(conv)

        # select the top‑k active lags
        if top_k >= len(conv):
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(-conv, top_k)[:top_k]

        rows = []
        cols = []
        data = []
        for r, k in enumerate(idxs):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                rows.append(r)
                cols.append(j)
                data.append(a[i])

        A = _sp.csr_matrix((data, (rows, cols)), shape=(len(idxs), n))
        b_ub = np.full(len(idxs), M, dtype=np.float64)
        bounds = [(0, None)] * n
        c = np.full(n, -1.0, dtype=np.float64)   # maximise Σg → minimise −Σg

        time_left = deadline - time.time()
        lp_opts = {"presolve": True, "time_limit": max(1.0, time_left * 0.2)}

        try:
            res = _opt.linprog(
                c, A_ub=A, b_ub=b_ub, bounds=bounds,
                method="highs", options=lp_opts,
            )
        except Exception:  # pragma: no cover
            break

        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break
        g_norm = g / s

        new = (1.0 - mix_ratio) * a + mix_ratio * g_norm
        new_val = _objective_exact(new)

        if new_val < best_val:
            best_a = new.copy()
            best_val = new_val

        a = new

    return best_a, best_val


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int, center: float | None = None) -> np.ndarray:
    if center is None:
        center = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - center) / center
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta_: float) -> np.ndarray:
    if alpha <= 0.0 or beta_ <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta_ - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int, width: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.maximum(0.0, np.cos(np.pi * (x - center) / width))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinusoidal(n: int, power: float = 1.0) -> np.ndarray:
    """a_i ∝ sin(π (i+1)/(n+1))^power."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1)) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Seed‑pool construction – richer families & more systematic grids
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic classical families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_triangular(n, center=n * 0.3))
    seeds.append(_triangular(n, center=n * 0.7))

    # exponential grid
    for lam in np.logspace(-5, -2, 15):
        seeds.append(_exponential_decay(n, float(lam)))

    # power‑law grid
    for p in np.linspace(0.25, 1.2, 20):
        seeds.append(_power_law(n, float(p)))

    # beta‑shape grid
    for a in (0.4, 0.8, 1.2, 2.0):
        for b_ in (0.4, 0.8, 1.2, 2.0):
            seeds.append(_beta_shape(n, float(a), float(b_)))

    # raised‑cosine widths
    for w in (n * 0.4, n * 0.6, n * 0.9):
        seeds.append(_raised_cosine(n, width=float(w)))

    # sinusoidal powers
    for p in (0.5, 1.0, 2.0, 4.0):
        seeds.append(_sinusoidal(n, power=float(p)))

    # low‑frequency cosine mixtures
    for nf in (2, 3, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random Dirichlet families (various concentrations)
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.01, 0.03, 0.07, 0.12):
        for _ in range(3):
            seeds.append(_random_sparse(n, frac, rng))

    # pairwise mixtures of existing seeds
    base = np.arange(len(seeds))
    for _ in range(120):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(20):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (warm‑start)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # clean‑up – non‑negative and ℓ₁‑normalised
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimises a non‑negative step‑function (list of heights) that minimise
    ``evaluate_sequence`` under the given time budget (≈1000 s).
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed pool (including warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Evaluate seeds (exact objective) and keep elite
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # keep top‑k for intensive polishing
    seeds.sort(key=_objective_exact)
    top_k = min(30, len(seeds))
    elite = seeds[:top_k]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate (elite seeds)
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before completing L‑BFGS runs.")
            break
        per_beta = 2000
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β={per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – multiple exponentiated‑gradient restarts (random Dirichlet)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 16
        iters_per_restart = int(min(400_000, remaining * 0.07 / restarts))
        print("\n--- Exponentiated‑gradient restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.01, 1.5)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.0,
            )
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long EG run from current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(8_000_000, remaining * 0.30 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=2.5,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 180))
        if max_iters > 0:
            print("\n--- Simulated‑annealing local search ---")
            seq_sa, val_sa = _simulated_annealing(
                global_best_seq,
                n,
                deadline,
                rng,
                max_iters=max_iters,
                init_temp=2.5,
            )
            print(f"[SA] best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[Update] New best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – LP‑based active‑set refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        steps = 6
        print("\n--- LP refinement (active‑set) ---")
        seq_lp, val_lp = _lp_refine(
            global_best_seq,
            n,
            deadline,
            rng,
            max_steps=steps,
            top_k=300,
            mix_ratio=0.10,
        )
        print(f"[LP] best = {val_lp:.6f}")
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[Update] New best after LP refinement = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – final high‑β polishing L‑BFGS (very many iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1000, int(min(4000, remaining * 0.07 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0, 10_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for future warm‑starts
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
