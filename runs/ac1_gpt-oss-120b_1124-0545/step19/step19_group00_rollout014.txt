"""
Propose a candidate step‑function (list of heights) that minimises
`evaluate_sequence`.  The strategy is a lightweight multi‑start search:

1.  Generate a rich set of analytic and stochastic seeds (Beta shapes,
    power‑law, Dirichlet, low‑frequency curves, etc.).  For every raw
    weight vector we also try a *paired* permutation that couples large
    coefficients with small ones (large ↔ small → a_i·a_{n‑1‑i} stays low).

2.  Keep the few best seeds (according to the exact surrogate
    `2·n·max(conv(seq,seq))`).  Each seed is then locally optimised by
    a very cheap "hill‑climb": repeatedly swap two random positions and
    keep the swap if the surrogate improves.  This explores the huge
    permutation space without ever building the 16 M‑element LP matrix
    that caused memory problems before.

3.  The best sequence from the hill‑climb is refined with an L‑BFGS
    optimiser that smooths the max‑convolution by a log‑sum‑exp surrogate
    (the same continuation schedule used previously).  The variables live
    in the log‑softmax space, guaranteeing non‑negativity and unit ℓ₁‑norm.

The whole routine respects a 1000 s wall‑time budget and stores the
result in the global ``height_sequence_1`` for the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used only for the final L‑BFGS smoothing)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:   # pragma: no cover
    _opt = None
    _sp = None

# ----------------------------------------------------------------------
# Numeric utilities (FFT‑based convolution, softmax, projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) – exact when sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Helper: create a permutation that pairs large with small values
# ----------------------------------------------------------------------
def _pair_large_small(v: np.ndarray) -> np.ndarray:
    """
    Return a new vector where the first half contains the largest values
    and the second half the smallest ones (large ↔ small in opposite
    positions → a_i·a_{n‑1‑i} stays low).
    """
    n = v.shape[0]
    desc = np.sort(v)[::-1]               # descending
    asc = np.sort(v)                      # ascending
    half = n // 2
    out = np.empty_like(v)
    out[:half] = desc[:half]              # large values left side
    out[half:] = asc[: n - half]          # small values right side
    # If n is odd the centre element (index half) gets the median value
    if n % 2:
        out[half] = desc[half]
    return out / out.sum()


# ----------------------------------------------------------------------
# Simple hill‑climb that swaps two positions if it improves the surrogate
# ----------------------------------------------------------------------
def _hill_climb(start: np.ndarray,
                rng: np.random.Generator,
                deadline: float,
                max_swaps: int = 30_000) -> tuple[np.ndarray, float]:
    """Random pairwise swaps, keep improvement – cheap local permutation search."""
    a = start.copy()
    best_val = _objective_surrogate(a)
    n = a.shape[0]

    swaps = 0
    while swaps < max_swaps and time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # perform swap
        a[i], a[j] = a[j], a[i]
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            # keep the swap
        else:
            # revert
            a[i], a[j] = a[j], a[i]
        swaps += 1
    return a, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0]

    # work in log‑softmax space so we stay on the simplex automatically
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (softmax space)
            w = exp_beta / sum_exp                # weights for each entry of conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)
