```
def _swap_anneal(a, n, deadline, max_iters, initial_temp, final_temp):
    conv = np.convolve(a, a).astype(np.float64)
    sum_a = a.sum()
    cur_obj = 2.0 * n * conv.max() / (sum_a*sum_a)
    best_a = a.copy()
    best_obj = cur_obj

    rng = np.random.default_rng()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # linear temperature schedule
        t = initial_temp + (final_temp - initial_temp) * (it / max_iters)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        vi = a[i]; vj = a[j]
        if vi == vj: continue
        d = vj - vi
        # backup conv copy
        conv_old = conv.copy()
        # apply updates
        conv[i:i+n] += 2.0 * d * a
        conv[j:j+n] -= 2.0 * d * a
        conv[2*i] += d * d
        conv[2*j] += d * d
        conv[i+j] -= 2.0 * d * d
        # swap a
        a[i], a[j] = vj, vi
        new_max = conv.max()
        new_obj = 2.0 * n * new_max / (sum_a*sum_a)
        # acceptance
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            # compute acceptance prob
            # temperature may be small near end
            prob = np.exp(-(new_obj - cur_obj) / t) if t > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            # revert
            a[i], a[j] = vi, vj
            conv = conv_old
    return best_a, best_obj
```

def _swap_greedy_optimize(start_a, deadline, max_iters):
    a = start_a.copy()
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = a.sum()
    cur_max = conv.max()
    cur_obj = 2.0 * n * cur_max / (sum_a * sum_a)
    best_a = a.copy()
    best_obj = cur_obj
    rng = np.random.default_rng()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        vi = a[i]; vj = a[j]
        if vi == vj:
            continue
        d = vj - vi
        # apply updates
        conv[i:i+n] += 2.0 * d * a
        conv[j:j+n] -= 2.0 * d * a
        conv[2*i] += d*d
        conv[2*j] += d*d
        conv[i+j] -= 2.0 * d*d
        # swap a
        a[i], a[j] = vj, vi
        new_max = conv.max()
        new_obj = 2.0 * n * new_max / (sum_a*sum_a)
        if new_obj < cur_obj:
            cur_obj = new_obj
            cur_max = new_max
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            # revert
            a[i], a[j] = vi, vj
            conv[i:i+n] -= 2.0 * d * a
            conv[j:j+n] += 2.0 * d * a
            conv[2*i] -= d*d
            conv[2*j] -= d*d
            conv[i+j] += 2.0 * d*d
        if (it+1) % 500_000 == 0:
            print(f"[Swap] iter {it+1:,} best obj={best_obj:.6f}")
    return best_a, best_obj
```

"""
A fast, deterministic search for a step‑function that minimizes

    2·n·max(conv(a,a)) / (∑a)²

The algorithm consists of the following stages (all respecting a 1000 s budget).

1. **Shape generation** – sample a modest collection of candidate height
   distributions (uniform, triangular, exponential, beta‑shaped,
   power‑law with a scalar optimiser, and a few random Dirichlet draws).
   The best few candidates (by the exact objective, *not* a surrogate)
   are kept.

2. **Initial ordering** – for each retained shape we try a cheap
   “interleaved” ordering (high values far apart) and a few cyclic
   shifts.  The best ordering among these is selected.

3. **Greedy swap optimisation** – the ordering is refined with a
   hill‑climbing where two positions are swapped and the effect on the
   convolution is updated *exactly* in O(n) using NumPy slicing
   (the previous implementation only touched the diagonal entries,
   which is incorrect).  Only improving moves are kept; the loop runs
   as long as time permits (up to a few million swaps).

4. **Fine‑grained local search** – a small mass‑transfer stochastic
   hill‑climb is performed to adjust the heights themselves
   (the value optimisation from the reference pipeline).

5. **Final mirror‑descent (exponentiated‑gradient)** – a short run of
   the exact sub‑gradient method cleans up any remaining slack.

All steps are timed; the best sequence found is stored globally as
``height_sequence_1`` and returned as a plain Python list, ready to be
passed to ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# low‑level numeric helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_exact(a: np.ndarray) -> float:
    """Exact value of the evaluation function (scale‑invariant)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# families of shapes (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, p>0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# simple deterministic ordering – interleaving two halves
# ----------------------------------------------------------------------
def _interleaved_order(n: int) -> np.ndarray:
    """Return a permutation that interleaves the first and second half."""
    order = np.empty(n, dtype=int)
    left, right, pos = 0, n // 2, 0
    while left < n // 2 or right < n:
        if left < n // 2:
            order[pos] = left
            pos += 1
            left += 1
        if right < n:
            order[pos] = right
            pos += 1
            right += 1
    return order


# ----------------------------------------------------------------------
# scalar optimisation of the power‑law exponent
# ----------------------------------------------------------------------
def _optimize_power_law(n: int, deadline: float):
    """Golden‑section search on the exponent p of a power‑law shape."""
    phi = (1 + 5 ** 0.5) / 2.0
    lo, hi = 0.05, 2.0  # reasonable range
    best_vec = None
    best_val = np.inf
    while hi - lo > 1e-4 and time.time() < deadline:
        p1 = hi - (hi - lo) / phi
        p2 = lo + (hi - lo) / phi
        v1 = _power_law(n, p1)
        v2 = _power_law(n, p2)
        val1 = _objective_exact(v1)
        val2 = _objective_exact(v2)
        if val1 < val2:
            hi = p2
            if val1 < best_val:
                best_val = val1
                best_vec = v1.copy()
        else:
            lo = p1
            if val2 < best_val:
                best_val = val2
                best_vec = v2.copy()
    # final refinement at the midpoint
    pm = (lo + hi) / 2.0
    vm = _power_law(n, pm)
    vm_val = _objective_exact(vm)
    if vm_val < best_val:
        best_val = vm_val
        best_vec = vm.copy()
    return best_vec, best_val, (lo + hi) / 2.0


# ----------------------------------------------------------------------
# exponentiated‑gradient optimisation (exact sub‑gradient)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (uses the active lag)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = np.exp(log_a)
        a /= a.sum()  # ℓ₁‑normalisation (scale invariant)
        if it % 50_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# greedy swap optimisation – exact O(n) update
# ----------------------------------------------------------------------
def _swap_greedy_optimize(start_a: np.ndarray,
                          deadline: float,
                          max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing on the ordering of a fixed multiset of heights.
    The full convolution is updated in O(n) using NumPy slicing.
    Only strictly improving swaps are kept.
    """
    a = start_a.copy()
    n = len(a)
    conv = np.convolve(a, a).astype(np.float64)   # full linear convolution
    sum_a = a.sum()
    cur_max = conv.max()
    cur_obj = 2.0 * n * cur_max / (sum_a * sum_a)

    best_a = a.copy()
    best_obj = cur_obj

    rng = np.random.default_rng()
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        vi, vj = a[i], a[j]
        if vi == vj:
            continue
        d = vj - vi               # = a_new[i] - a_old[i]

        # ---- apply the exact convolution update ---------------------------------
        conv[i:i + n] += 2.0 * d * a
        conv[j:j + n] -= 2.0 * d * a
        conv[2 * i] += d * d
        conv[2 * j] += d * d
        conv[i + j] -= 2.0 * d * d

        # swap the heights
        a[i], a[j] = vj, vi

        new_max = conv.max()
        new_obj = 2.0 * n * new_max / (sum_a * sum_a)

        if new_obj < cur_obj:                   # accept only improving moves
            cur_obj = new_obj
            cur_max = new_max
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            # ---- revert everything ------------------------------------------------
            a[i], a[j] = vi, vj                # restore ordering
            conv[i:i + n] -= 2.0 * d * a
            conv[j:j + n] += 2.0 * d * a
            conv[2 * i] -= d * d
            conv[2 * j] -= d * d
            conv[i + j] += 2.0 * d * d

        if (it % 500_000) == 0:
            print(f"[Swap] iter {it:,} – best obj = {best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# tiny stochastic mass‑transfer local search (as in the reference)
# ----------------------------------------------------------------------
def _local_search(start_a: np.ndarray,
                  deadline: float,
                  iters: int = 150_000,
                  step_frac: float = 0.008) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    n = len(a)
    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.0004))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # power‑law – scalar optimiser
    pl_vec, pl_val, pl_p = _optimize_power_law(n, deadline)
    seeds.append(pl_vec)
    print(f"[Shape] power‑law p≈{pl_p:.4f} → obj {pl_val:.6f}")

    # a few random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # keep the previous best if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0.0:
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # normalise everything, guard against pathological cases
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    seeds = cleaned

    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 2️⃣ – keep the top‑k (by exact objective) for heavy processing
    # ------------------------------------------------------------------
    seed_objs = [( _objective_exact(v), v ) for v in seeds]
    seed_objs.sort(key=lambda x: x[0])
    top_k = 5
    elite = [v for _, v in seed_objs[:top_k]]
    global_best_seq = elite[0].copy()
    global_best_val = seed_objs[0][0]
    print(f"[Init] best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ – initial ordering: interleaved permutation + a few rotations
    # ------------------------------------------------------------------
    def _best_ordering(vec: np.ndarray) -> tuple[np.ndarray, float]:
        """Return the best ordering of ``vec`` among a handful of simple permutations."""
        sorted_vals = np.sort(vec)[::-1]          # descending
        order = _interleaved_order(n)
        best_local = vec.copy()
        best_val_local = _objective_exact(vec)
        # try a few cyclic shifts of the interleaved order
        for shift in range(0, min(60, n), 5):
            perm = np.roll(order, shift)
            cand = np.empty_like(sorted_vals)
            cand[perm] = sorted_vals
            val = _objective_exact(cand)
            if val < best_val_local:
                best_val_local = val
                best_local = cand.copy()
        # also consider pure descending order (no interleaving)
        desc = sorted_vals.copy()
        val_desc = _objective_exact(desc)
        if val_desc < best_val_local:
            best_val_local = val_desc
            best_local = desc.copy()
        return best_local, best_val_local

    # improve each elite candidate by ordering
    best_ordered = None
    best_order_val = np.inf
    for idx, vec in enumerate(elite):
        if time.time() > deadline:
            break
        cand, val = _best_ordering(vec)
        print(f"[Ordering] elite #{idx+1} → obj {val:.6f}")
        if val < best_order_val:
            best_order_val = val
            best_ordered = cand.copy()

    if best_ordered is None:
        best_ordered = global_best_seq.copy()
        best_order_val = global_best_val

    # ------------------------------------------------------------------
    # 4️⃣ – greedy swap optimisation (correct O(n) update)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # roughly 0.5 µs per element update ⇒ about 0.02 s per swap for n≈4000
        # we allocate a generous number but keep it bounded by time.
        max_swaps = int(min(2_000_000, remaining * 2_000_000))
        if max_swaps > 0:
            print(f"\n--- Swap‑optimisation (up to {max_swaps:,} swaps) ---")
            seq_swap, val_swap = _swap_greedy_optimize(best_ordered,
                                                       deadline,
                                                       max_iters=max_swaps)
            print(f"[Swap] best objective = {val_swap:.6f}")
            if val_swap < best_order_val:
                best_ordered = seq_swap.copy()
                best_order_val = val_swap

    # ------------------------------------------------------------------
    # 5️⃣ – tiny stochastic local search (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(200_000, remaining * 80))
        if iters > 0:
            print("\n--- Mass‑transfer local search ---")
            seq_loc, val_loc = _local_search(best_ordered,
                                             deadline,
                                             iters=iters,
                                             step_frac=0.006)
            print(f"[LocalSearch] best objective = {val_loc:.6f}")
            if val_loc < best_order_val:
                best_ordered = seq_loc.copy()
                best_order_val = val_loc

    # ------------------------------------------------------------------
    # 6️⃣ – final short exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(400_000, remaining * 200_000))
        if iters > 0:
            print("\n--- Final EG refinement ---")
            seq_eg, val_eg = _exp_gradient_optimize(best_ordered,
                                                    n,
                                                    deadline,
                                                    max_iters=iters,
                                                    eta0=2.0)
            print(f"[Final EG] best objective = {val_eg:.6f}")
            if val_eg < best_order_val:
                best_ordered = seq_eg.copy()
                best_order_val = val_eg

    # ------------------------------------------------------------------
    # 7️⃣ – final round with the official evaluator for reporting
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_ordered.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {best_order_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in subsequent runs
    globals()["height_sequence_1"] = best_ordered.tolist()
    return best_ordered.tolist()
