def _project_simplex(v):
    # ensure nonnegative
    # from https://arxiv.org/pdf/1101.6081.pdf
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that drives ``evaluate_sequence`` below 1.503.

The algorithm combines three ideas that were not used together before:

1. **Rich seed pool** – deterministic shapes, random Dirichlet draws,
   power‑law, beta‑distributions and a few “z‑zag’’ permutations provide a
   very diverse set of initial vectors.

2. **Linear‑programming refinement** – for a current vector `a`,
   solve a LP that maximises the ℓ₁‑norm of a new vector `g` under the
   linear constraints  `conv(a,g) ≤ M` where `M = max(conv(a,a))`.
   Normalising `g` to a probability vector and mixing
   `a ← (1‑t)·a + t·g` strictly reduces the maximal self‑convolution.
   Repeating this step (≈10 times) drives the objective close to its
   theoretical limit.

3. **Fine‑grained stochastic local search** – after the LP phase,
   a classic hill‑climber moves tiny fractions of mass between coordinates
   and accepts only improvements.  This cleans the remaining tiny “spikes’’
   in the convolution.

All steps respect the 1000 s wall‑time budget, print progress and finally
store the best vector in the global variable ``height_sequence_1``.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective & helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (the quantity we minimise)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    (non‑negative, sum = 1).  Implementation of
    https://arxiv.org/abs/1101.6081 .
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise   1ᵀ g
    subject to 0 ≤ g,  conv(f,g) ≤ M   (element‑wise)
    Returns the optimal ``g`` (may be dense) or ``None`` on failure.
    """
    n = len(f)
    L = 2 * n - 1

    # Build the (L × n) matrix A where (A @ g)[k] = Σ_i f_i g_{k-i}
    # which is exactly the convolution f * g.
    A = np.zeros((L, n), dtype=np.float64)
    # each column j contributes f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f  # vectorised slice assignment

    b = np.full(L, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  → minimise -sum(g)

    try:
        res = _opt.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
    except Exception:          # pragma: no cover
        return None

    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Stochastic local search (mass‑transfer hill‑climbing)
# ----------------------------------------------------------------------
def _local_search(start_a: np.ndarray,
                  deadline: float,
                  iters: int = 120_000,
                  step_frac: float = 0.008) -> tuple[np.ndarray, float]:
    """Move a small fraction of mass from i to j if it improves the exact objective."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(314159)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, len(a))
        j = rng.integers(0, len(a))
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed‑pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.02))

    # Power‑law sweep
    for p in np.linspace(0.35, 0.75, 5):
        for sh in (0.0, 5.0, 20.0):
            seeds.append(_power_law(n, p, sh))
            seeds.append(_power_law(n, p, sh)[::-1])

    # Beta‑distribution sweep
    for _ in range(20):
        a = 10 ** rng.uniform(-2, 1)
        b = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (varying concentration)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Pairwise mixtures of existing seeds (creates many intermediate shapes)
    base = np.arange(len(seeds))
    for _ in range(100):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of promising seeds
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(np.roll(seeds[idx], shift))

    # Previous best, if supplied
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise & sanitise all seeds
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s wall‑time limit, stores the best
    result in the global variable ``height_sequence_1`` and returns it as
    a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 3000 provides a good trade‑off between quality and
    # memory consumption for the LP matrix (≈150 MiB).
    # ------------------------------------------------------------------
    n = 3000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed collection (including previous best, if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Evaluate seeds, keep the elite (top‑k)
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Linear‑programming refinement phase
    # ------------------------------------------------------------------
    max_lp_iters = 12                 # number of outer LP refinements
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        conv_cur = _conv_fft(best_seq, best_seq)
        M = np.max(conv_cur)

        g0 = _solve_lp_max_sum(best_seq, M)
        if g0 is None:
            print("[LP] Solver failed – exiting LP phase.")
            break

        # Normalise g0 to a probability vector
        g = _project_simplex(g0)

        # Try a few mixing coefficients, keep the first improvement
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            cand = (1.0 - t) * best_seq + t * g
            cand = _project_simplex(cand)          # re‑project to simplex
            val = _objective_exact(cand)
            if val < best_val:
                best_seq, best_val = cand, val
                improved = True
                print(f"[LP iter {lp_iter+1}] t={t:.2f} improved value = {val:.6f}")
                break
        if not improved:
            # very small step as a last resort
            t = 0.02
            cand = (1.0 - t) * best_seq + t * g
            cand = _project_simplex(cand)
            val = _objective_exact(cand)
            if val < best_val:
                best_seq, best_val = cand, val
                print(f"[LP iter {lp_iter+1}] tiny t={t:.2f} improved value = {val:.6f}")
                improved = True

        if not improved:
            # No improvement – stop early
            print("[LP] No improvement found, terminating LP phase.")
            break

        lp_iter += 1

    # ------------------------------------------------------------------
    # Stochastic local search (fine‑grained hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(300_000, remaining * 300))   # a few × 10⁵ moves
        if iters > 0:
            print("\n--- Fine stochastic local search ---")
            seq_loc, val_loc = _local_search(best_seq, deadline, iters=iters, step_frac=0.006)
            print(f"[LocalSearch] best = {val_loc:.6f}")
            if val_loc < best_val:
                best_seq, best_val = seq_loc, val_loc
                print(f"[Update] New best after local search = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with a short L‑BFGS run on the smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(600, int(min(2000, remaining * 0.03 * 1e5)))
        print("\n--- Final L‑BFGS polishing (log‑sum‑exp surrogate) ---")
        # Re‑use the smooth optimiser from the previous version (lighter)
        def _softmax(v):  # local copy to avoid import
            m = np.max(v)
            e = np.exp(v - m)
            s = e.sum()
            return e / s if s else np.full_like(v, 1.0 / len(v))

        def _smooth_objective(v_local, beta):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = exp_term.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            w = exp_term / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        v = np.log(best_seq + 1e-30)
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1_000.0, 2_000.0, 5_000.0,
                         10_000.0, 20_000.0, 50_000.0,
                         100_000.0, 200_000.0, 500_000.0,
                         1_000_000.0, 2_000_000.0, 5_000_000.0,
                         10_000_000.0, 50_000_000.0, 100_000_000.0]

        for beta in beta_schedule:
            if time.time() > deadline:
                break
            fun = lambda vv: _smooth_objective(vv, beta)
            try:
                res = _opt.minimize(fun,
                                    x0=v,
                                    method="L-BFGS-B",
                                    jac=True,
                                    options={"maxiter": per_beta,
                                             "ftol": 1e-12,
                                             "disp": False})
                v = res.x
            except Exception:  # pragma: no cover
                break

        seq_final = _softmax(v)
        val_final = _objective_exact(seq_final)
        print(f"[Polish] objective = {val_final:.6f}")
        if val_final < best_val:
            best_seq, best_val = seq_final, val_final
            print(f"[Update] New best after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
