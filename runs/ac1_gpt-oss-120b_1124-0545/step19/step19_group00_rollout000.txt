Propose a candidate step‑function (list of non‑negative heights) that
minimises the objective defined by ``evaluate_sequence``.

The algorithm:

1. **Generate a diverse pool of seeds** – analytic windows, random
   Dirichlet / exponential vectors, low‑frequency cosine curves and
   perturbations of the previous best solution.

2. **Select the most promising seeds** using the cheap surrogate
   ``2·n·max(conv(a,a))`` (exact for ℓ₁‑normalised vectors).

3. **Refine each seed** with a two‑phase optimiser:
   * **Phase A – smoothed‑max gradient descent**.
     The non‑smooth max is approximated by a log‑sum‑exp with an
     increasing “β’’ schedule.  The variables live in the log‑softmax
     space (unconstrained) and are updated with Adam.  This guarantees
     positivity and ℓ₁‑normalisation.

     the final vector with the true max‑convolution objective.

   The optimiser stops when the wall‑time budget (≈ 1000 s) is exhausted.

4. **Return the best sequence** (and store it globally as
   ``height_sequence_1``) so that the next run can start from it.

The implementation avoids large dense matrices (no LP step) and is
fully self‑contained; it runs comfortably within the 1000 s limit on a
single CPU core.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via a real FFT (O(N log N)).
    Returns a vector of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(a: np.ndarray) -> float:
    """
    Exact surrogate objective for a vector that already satisfies
    sum(a)=1:  2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)               # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:

def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

# Optimiser – smooth‑max + exact‑max polishing
# ----------------------------------------------------------------------
def _smooth_max_opt(start_a: np.ndarray,
                    n: int,
                    deadline: float) -> np.ndarray:
    """
    Adam optimisation of the smoothed max (log‑sum‑exp) objective.
    Parameters live in log‑softmax space to respect positivity and the simplex.
    """
    eps = 1e-30
    # initialise logits
    v = np.log(start_a + eps)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    t = 0

    # β schedule – start smooth, end very sharp (≈ exact max)
    betas = [0.5, 2, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]

    # limit steps per β to keep wall‑time reasonable
    steps_per_beta = 2000

    for beta in betas:
        if time.time() > deadline:
            break
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(v)

            # ---- smoothed max ----
            conv = _conv_fft(a, a)                         # length 2n‑1
            shift = np.max(conv)                           # numeric stabiliser
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            if sum_exp == 0.0:
                w = np.full_like(exp_beta, 1.0 / exp_beta.size)
            else:
                w = exp_beta / sum_exp                     # weights for each entry

            # ---- gradient wrt a (exact formula for smoothed max) ----
            rev_a = a[::-1]
            conv_rev_full = _conv_fft(rev_a, w)            # length 2n‑1
            slice_start = n - 1
            slice_end = slice_start + n
            slice_part = conv_rev_full[slice_start:slice_end]
            grad_a = 4.0 * n * slice_part                  # ∂ / ∂a of 2·n·s(β)

            # ---- chain rule to logits v (softmax) ----
            grad_v = a * (grad_a - np.dot(grad_a, a))

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)
            lr = 0.12 / math.sqrt(t)          # modest decaying learning rate
            v = v - lr * m_hat / (np.sqrt(v2_hat) + epsilon)

    # final softmax vector
    a = _softmax(v)

    # ---- Phase B: exact‑max EG polishing ---------------------------------
    log_a = np.log(a + eps)
    eta0 = 5.0
    max_eg_iters = 2500
    for it in range(1, max_eg_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

    return a


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator, prev: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a reasonably large and diverse collection of ℓ₁‑normalised
    candidate vectors.
    """
    seeds = []

    # Analytic families (moderately dense grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.07, 0.30, 20):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 2.5, 15):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet / exponential
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Alternating Dirichlet
    seeds.append(_alternating_dirichlet(n, rng))

    # Perturbations of the previous best (if available)
    if prev is not None and prev.size == n and prev.sum() > 0.0:
        base = prev / prev.sum()
        seeds.append(base)
        for _ in range(4):
            seeds.append(_perturb(base, rng, sigma=0.003))

    # Simple mixtures of uniform with a low‑freq curve – sometimes useful
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    return seeds

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method builds a pool of analytic and random
    seeds, keeps the most promising ones and refines each of them with a
    two‑phase optimiser (smooth‑max Adam + exact‑max EG).  The best vector
    found within the allotted wall‑time is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    safety_margin = 3.0                     # keep a few seconds for final eval
    deadline = start_time + budget_s - safety_margin

    # ------------------------------------------------------------------
    # Problem size (kept consistent with the previous run)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution if it exists
    # ------------------------------------------------------------------
    prev = None
    if 'height_sequence_1' in globals():
        try:
            tmp = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if tmp.shape == (n,):
                prev = tmp.copy()
        except Exception:
            prev = None

    # ------------------------------------------------------------------
    # 1. Broad seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, prev)

    # ------------------------------------------------------------------
    # 2. Cheap surrogate evaluation and selection of top seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))
    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for (vec, _) in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Optimise each top seed individually
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimising seed #{idx+1} (surrogate={_objective_surrogate(seed_vec):.6f}) ---")
        seq_opt = _smooth_max_opt(seed_vec, n, deadline)
        val_opt = _objective_surrogate(seq_opt)
        print(f"    Optimised surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"    *** New best overall surrogate! ***")

    # ------------------------------------------------------------------
    # 4. Official evaluation with the provided function
    # ------------------------------------------------------------------
    true_val = None
    if best_seq is None:
        # fallback – use the best raw seed
        best_seq = evaluated[0][0]
        best_val = evaluated[0][1]
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the best sequence globally for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
