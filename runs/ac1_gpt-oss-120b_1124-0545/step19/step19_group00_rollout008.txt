def _local_search(seq, timeout, step_scale=0.01):
    best = seq.copy()
    best_val = _objective_surrogate(best)
    n = len(seq)
    rng = np.random.default_rng()
    while time.time() < timeout:
        eps = step_scale * rng.random()
        i, j = rng.integers(0,n, size=2)
        if i == j:
            continue
        # Transfer epsilon from i to j
        # ensure we don't go negative
        delta = min(eps, best[i])
        new_seq = best.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = new_seq / new_seq.sum() # renormalize (since sum preserved)
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best = new_seq
    return best, best_val

"""
propose_candidate
-----------------

A multi‑stage search for a probability vector (non‑negative step heights) that
minimises

    2·n·max(conv(p, p))          (equivalently ``evaluate_sequence``)

The algorithm:

1. **Extensive parametric seeding** – sweep a wide range of analytic
   families (sine, Kaiser, Gaussian, power‑law, beta‑shaped, Tukey, etc.),
   plus many random Dirichlet / exponential / low‑frequency curves.
   The cheap surrogate objective (2·n·max(conv)) is evaluated for each
   candidate and the best few are kept.

2. **Mirror‑descent (exponentiated‑gradient) refinement** – starting from each
   top seed we run a long EG optimisation in log‑softmax space, using the
   exact sub‑gradient of the active convolution entry.  The routine stops
   automatically when the wall‑time budget is exhausted.

3. **Continuation L‑BFGS smoothing** – the non‑smooth max is approximated
   by a log‑sum‑exp surrogate with an increasing temperature schedule.
   L‑BFGS‑B (when SciPy is available) performs a few thousand iterations per
   temperature, always respecting the deadline.

4. **Simple stochastic local search** – a few thousand random mass‑transfer
   moves are tried; any improvement (as measured by the surrogate) is kept.

5. **Final evaluation** – the best vector found is scored with the provided
   ``evaluate_sequence``.  The result is stored in the global variable
   ``height_sequence_1`` for possible reuse in a subsequent run.

All steps respect a total wall‑time of roughly ``budget_s`` seconds (default
1000 s).  The code is completely self‑contained and uses only NumPy (and SciPy
if available for the L‑BFGS step)."""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector ℓ₁‑normalised to 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ------------------------------------------------------------
# Random families
# ------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ------------------------------------------------------------
# Mixing / projection helpers
# ------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _ternary_mix(a: np.ndarray, b: np.ndarray, deadline: float,
                 steps: int = 30):
    """Best convex combination of a and b by ternary search on α∈[0,1]."""
    lo, hi = 0.0, 1.0
    best_seq, best_val = None, float('inf')
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if val1 < best_val:
                best_val, best_seq = val1, seq1
        else:
            lo = m1
            if val2 < best_val:
                best_val, best_seq = val2, seq2
    mid = (lo + hi) / 2.0
    seq_mid = _mix_shapes(a, b, mid)
    val_mid = _objective_surrogate(seq_mid)
    if val_mid < best_val:
        best_seq, best_val = seq_mid, val_mid
    return best_seq, best_val


# ------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     eta0: float = 5.0):
    """Mirror‑descent on exact max‑convolution.  Returns best sequence seen."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp   # noqa: F401  (kept for compatibility)
except Exception:  # pragma: no cover
    _opt = None

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (softmax space)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ------------------------------------------------------------
# Very cheap stochastic local search
# ------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  deadline: float,
                  moves: int = 20000,
                  step_scale: float = 0.02):
    """
    Simple random mass‑transfer hill‑climbing.
    At each move a small amount of probability is moved from one index to another.
    """
    rng = np.random.default_rng()
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(seq)

    for _ in range(moves):
        if time.time() > deadline:
            break
        eps = step_scale * rng.random()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(eps, best_seq[i])
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # sum preserved – only renormalise for numerical safety
        new_seq = new_seq / new_seq.sum()
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.5          # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # 1. Broad seed generation (large parametric sweep)
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds (n={n}) ---")
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    seeds.append(_alternating_dirichlet(n, rng))

    # sine window – dense grid
    for gamma in np.linspace(0.1, 5.0, 120):
        seeds.append(_sine_window(n, gamma))

    # Kaiser window – fairly wide range
    for beta in np.linspace(0.1, 30.0, 150):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian – moderate range
    for sigma in np.linspace(0.04, 0.35, 80):
        seeds.append(_gaussian(n, sigma))

    # Double‑Gaussian – a few useful offsets
    for sigma in (0.08, 0.10, 0.12, 0.15):
        for shift in (0.15, 0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma,
                                          shift_frac=shift))

    # Power‑law
    for p in np.linspace(0.0, 2.5, 130):
        seeds.append(_power(n, p))

    # Beta‑shape – moderate grid (≈ 2000 candidates)
    alphas = np.linspace(0.2, 5.0, 45)
    betas  = np.linspace(0.2, 5.0, 45)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # Tukey window
    for a_ in np.linspace(0.0, 1.0, 51):
        seeds.append(_tukey_window(n, a_))

    # Random Dirichlet (various concentrations)
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential
    for _ in range(15):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mix a few low‑freq curves with uniform (to diversify shapes)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous best (if present)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                arr = arr / arr.sum()
                seeds.append(arr)
                for _ in range(5):
                    seeds.append(_perturb(arr, rng, sigma=0.003))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Evaluate surrogate for all seeds (cheap)
    # ------------------------------------------------------------------
    print("\n--- Evaluating seeds (surrogate) ---")
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values after seeding:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine top seeds with EG + L‑BFGS + local search
    # ------------------------------------------------------------------
    best_overall_seq = top_seeds[0].copy()
    best_overall_val = _objective_surrogate(best_overall_seq)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Refinement for seed #{idx+1} ---")
        # allocate a slice of the remaining time for EG
        remaining = deadline - time.time()
        eg_deadline = time.time() + max(5.0, 0.4 * remaining)
        seq_eg, val_eg = _exp_gradient_opt(seed_seq.copy(),
                                           n,
                                           eg_deadline,
                                           max_iters=2_000_000,
                                           eta0=5.0)
        if val_eg < best_overall_val:
            best_overall_val, best_overall_seq = val_eg, seq_eg.copy()
            print(f"  EG improvement: {val_eg:.6f}")

        if time.time() > deadline:
            break

        # L‑BFGS smoothing (if SciPy is present)
        lbfgs_deadline = deadline - 2.0  # keep a tiny buffer
        seq_lb = _lbfgs_smooth_optimize(best_overall_seq,
                                         n,
                                         lbfgs_deadline,
                                         beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                        200.0, 500.0, 1000.0,
                                                        2000.0, 5000.0,
                                                        20000.0, 50000.0,
                                                        100000.0, 200000.0,
                                                        500000.0, 1000000.0],
                                         max_iter_per_beta=4000)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_overall_val:
            best_overall_val, best_overall_seq = val_lb, seq_lb.copy()
            print(f"  L‑BFGS improvement: {val_lb:.6f}")

        if time.time() > deadline:
            break

        # Simple ternary mixing with uniform (can sometimes lower the peak)
        try:
            mixed_seq, mixed_val = _ternary_mix(best_overall_seq,
                                                _uniform(n),
                                                deadline,
                                                steps=25)
            if mixed_val < best_overall_val:
                best_overall_val, best_overall_seq = mixed_val, mixed_seq.copy()
                print(f"  Mixing improvement: {mixed_val:.6f}")
        except Exception:
            pass

        if time.time() > deadline:
            break

        # Random local search for a few seconds
        local_deadline = deadline - 1.0
        seq_loc, val_loc = _local_search(best_overall_seq,
                                         local_deadline,
                                         moves=15000,
                                         step_scale=0.015)
        if val_loc < best_overall_val:
            best_overall_val, best_overall_seq = val_loc, seq_loc.copy()
            print(f"  Local‑search improvement: {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 4. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)

    print(f"\n=== Finished ===\nBest surrogate ≈ {best_overall_val:.6f}"
          f", true evaluate_sequence = {true_val}\nLength = {len(best_overall_seq)}")

    # Store for possible reuse
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
