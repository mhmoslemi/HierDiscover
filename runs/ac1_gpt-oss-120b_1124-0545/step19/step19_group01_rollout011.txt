def _adam_subgrad_opt(start_seq, n, deadline, max_iters=500_000, eta0=8.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # subgradient g
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # convert to log gradient
        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log * grad_log)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = eta0 / np.sqrt(it)   # base lr schedule
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # occasional random permutation attempt
        if it % 50000 == 0:
            # try a random swap
            i, j = np.random.randint(0, n, size=2)
            ns = a.copy()
            ns[i], ns[j] = ns[j], ns[i]
            val_swap = _objective_surrogate(ns)
            if val_swap < cur_val:
                a = ns
                log_a = np.log(a + 1e-30)  # reset log due to swap
                # Reset moments? maybe zero them
                m.fill(0)
                v.fill(0)
                cur_val = val_swap
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
    return best_seq, best_val

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    n = 4000

    # 1. seed generation
    seeds = []

    # fixed shapes
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    # Gaussian sweeps
    for sigma_frac in np.linspace(0.05, 0.35, 10):
        seeds.append(_gaussian(n, sigma_frac))
    # exponential decays
    for rate in np.logspace(-5, -0.3, 15):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))
    # Dirichlet draws
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))
    # random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))
    # low frequency curves
    for nf in (2,3,4,5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # include previous solution if any
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev)==n:
        prev_arr = np.asarray(prev, dtype=np.float64)
        prev_arr = prev_arr / prev_arr.sum()
        seeds.append(prev_arr)

    # 2. evaluate seeds fast
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        v = np.maximum(vec, 0.0)
        s = v.sum()
        if s == 0.0:
            continue
        v = v / s
        val = _objective_surrogate(v)
        evaluated.append((v, val))

    if not evaluated:
        raise RuntimeError("No valid seeds were generated")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # 3. global best init
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\nInitial global best surrogate = {global_best_val:.6f}")

"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²).

The algorithm is deliberately simple but exploits the fact that the
objective depends only on the *shape* of the vector (scaling is irrelevant).

1. **Seed generation** – a large pool of analytically‑parameterised
   families (uniform, triangular, Gaussian, exponential, low‑frequency
   cosine) and many Dirichlet draws.  All seeds are normalised to sum to 1.

2. **Fast surrogate selection** – the exact surrogate
   2·n·max(conv) is computed via a real FFT for each seed and the
   best few (default 5) are kept.

3. **Adam mirror‑descent on the sub‑gradient of the *max*** –
   the optimisation is performed in the log‑softmax (entropy‑mirror)
   domain, guaranteeing non‑negativity and unit ℓ₁‑norm.
   The sub‑gradient of the current maximal convolution entry is obtained
   analytically (O(n) work) and an Adam‑style step‑size is used.
   Wherever the time budget permits, each elite seed receives an
   independent optimisation run.

4. **Final smoothing polish** – the remaining time is spent on a
   smoothed version of the max (log‑sum‑exp) with a schedule of increasing
   β.  This helps to reduce the few remaining “spikes’’ that the pure
   sub‑gradient descent may miss.

5. The best sequence found is evaluated with the provided *evaluate_sequence*
   function and stored globally as ``height_sequence_1`` for possible warm‑starts.

The whole routine respects the 1000 s budget, prints progress
information and returns the final list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact, for a vector with sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine components, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Adam mirror‑descent on the *sub‑gradient* of the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_subgrad_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 1_000_000,
                      eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (entropy) on the exact max‑convolution.
    Uses Adam‑style adaptive steps on the log‑softmax parameters.
    Returns the best sequence encountered and its surrogate value.
    """
    # initialise log‑probabilities
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)                     # current probabilities, sum = 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # transform to gradient w.r.t. log‑parameters
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam update on log‑variables
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = eta0 / math.sqrt(it)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam mirror‑descent on a *smoothed* max (log‑sum‑exp) – final polish
# ----------------------------------------------------------------------
def _smooth_adam_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters_per_beta: int = 60_000) -> tuple[np.ndarray, float]:
    """
    Optimises the smoothed max‑convolution  f_β(a)=2·n·log‑sum‑exp(β·conv)/β
    using Adam in the log‑softmax space.
    A geometric schedule of β values is used.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    # β‑schedule (increasing – smoother approximation → true max)
    beta_schedule = [
        10.0, 30.0, 100.0, 300.0, 1_000.0, 3_000.0,
        10_000.0, 30_000.0, 100_000.0, 300_000.0, 1_000_000.0
    ]

    t = 0  # global iteration counter for bias‑correction
    for beta in beta_schedule:
        it = 0
        while it < max_iters_per_beta and time.time() < deadline:
            t += 1
            a = _softmax(log_a)

            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # stable soft‑max over convolution entries
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / np.sum(exp_beta)

            # gradient wrt a of Σ w·conv  (see analysis)
            grad_a = 2.0 * _conv_fft(a[::-1], w)[n - 1:n - 1 + n]

            # transform to gradient wr.t. log‑parameters
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = 0.5 / math.sqrt(t)          # modest learning rate
            log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

            it += 1
        if time.time() >= deadline:
            break
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm mixes a rich seed pool, Adam mirror‑descent on the exact
    max‑convolution, and a final high‑β smoothing polish.
    The best sequence is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # Problem dimension – 4000 offers a good trade‑off
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build a large pool of candidate seeds
    # ------------------------------------------------------------------
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for sigma_frac in np.linspace(0.05, 0.35, 12):
        seeds.append(_gaussian(n, sigma_frac))

    for rate in np.logspace(-5, -0.3, 18):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Random Dirichlet draws with diversified concentration
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Warm‑start from a previous run, if present
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 0.0:
                seeds.append(arr / s)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        v = np.maximum(vec, 0.0)
        s = v.sum()
        if s == 0.0:
            continue
        v = v / s
        val = _objective_surrogate(v)          # exact surrogate, fast
        evaluated.append((v, val))

    if not evaluated:
        raise RuntimeError("Failed to generate any valid seed.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise the global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each elite seed with Adam sub‑gradient optimisation
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # compute a fair time share for the remaining seeds
        remaining = deadline - time.time()
        share = remaining * 0.85 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] Adam sub‑gradient optimisation (budget ≈ {share:.1f}s)")
        seq_opt, val_opt = _adam_subgrad_opt(seed_vec, n, sub_deadline,
                                             max_iters=1_500_000, eta0=8.0)
        print(f"[Seed {si+1}] best surrogate after sub‑grad = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final smoothing polish (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final smoothing Adam polish ---")
        final_seq, final_val = _smooth_adam_opt(global_best_seq, n, deadline,
                                                max_iters_per_beta=80_000)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final polish] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
