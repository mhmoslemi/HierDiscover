"""Optimization pipeline to find step heights minimizing evaluate_sequence.

Uses:
- A seed (previous best or power‑law)
- Continuation L‑BFGS on a smooth log‑sum‑exp surrogate (high β)
- Exact greedy swap optimisation that updates the full convolution
  in O(n) per swap (the previous code missed a correction term)
- Final polishing L‑BFGS
All stages respect the 1000 s time budget.
"""
import time
import numpy as np
import scipy.optimize as _opt

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, p>0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)

def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)

def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate of the max‑convolution objective."""
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)   # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)

def _swap_optimize_correct(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact greedy hill‑climbing on the ordering of a fixed multiset of heights.
    Uses O(n) updates for each swap and corrects the missing quadratic term
    at index i+j.  Only improving moves are accepted.
    """
    a = start_a.copy()
    conv = np.convolve(a, a)                     # full convolution (size 2n‑1)
    sum_a = a.sum()
    cur_max = conv.max()
    cur_obj = 2.0 * n * cur_max / (sum_a * sum_a)

    best_a = a.copy()
    best_val = cur_obj

    rng = np.random.default_rng(987654321)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        vi = a[i]
        vj = a[j]
        if vi == vj:
            continue
        d = vj - vi                # change applied at position i (new - old)

        # vector delta for the two slices
        delta = 2.0 * d * a

        # apply updates
        conv[i:i + n] += delta
        conv[j:j + n] -= delta
        conv[2 * i] += d * d
        conv[2 * j] += d * d
        conv[i + j] -= 2.0 * d * d   # correction for the (i,j) ordered pair

            cur_obj = new_obj
            if new_obj < best_val:

"""
Optimization pipeline to minimise ``evaluate_sequence``.

The algorithm proceeds in three phases:
  1. Load the previous best sequence (if any) or start from a simple power‑law.
  2. Continuation L‑BFGS on a smooth log‑sum‑exp surrogate with a long
     β‑schedule (high β → tighter approximation of the max).
  3. Exact greedy swap optimisation.  A swap of two heights changes the
     self‑convolution only at the two “self” entries and at the index i+j.
     The implementation updates the full convolution in O(n) per swap,
     correcting the missing quadratic term that the reference implementation
     omitted.
  4. A final high‑β L‑BFGS polishing pass.

All work respects a 1000 s time budget; the best sequence found is stored in
the global ``height_sequence_1`` variable and returned as a plain list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# families of seed shapes
# ----------------------------------------------------------------------
def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, p>0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# exact objective (used for reporting)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate of the max‑convolution objective."""
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient wrt a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Correct swap optimiser (exact O(n) updates)
# ----------------------------------------------------------------------
def _swap_optimize_correct(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing on the ordering of a fixed multiset of heights.
    The convolution is updated in O(n) per swap, using the exact formula
    (including the missing 2·d² correction at index i+j).
    Only strictly improving moves are accepted.
    """
    a = start_a.copy()
    conv = np.convolve(a, a)                     # length 2n‑1
    sum_a = a.sum()
    cur_max = conv.max()
    cur_obj = 2.0 * n * cur_max / (sum_a * sum_a)

    best_a = a.copy()
    best_val = cur_obj

    rng = np.random.default_rng(987654321)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        vi = a[i]
        vj = a[j]
        if vi == vj:
            continue
        d = vj - vi                # replacement amount at position i

        # delta that must be added to the two convolution windows
        delta = 2.0 * d * a

        # apply updates
        conv[i:i + n] += delta
        conv[j:j + n] -= delta
        conv[2 * i] += d * d
        conv[2 * j] += d * d
        conv[i + j] -= 2.0 * d * d   # correction for the (i,j) ordered pair

        # swap the heights
        a[i], a[j] = vj, vi

        new_max = conv.max()
        new_obj = 2.0 * n * new_max / (sum_a * sum_a)

        if new_obj < cur_obj:
            cur_obj = new_obj
            if new_obj < best_val:
                best_val = new_obj
                best_a = a.copy()
        else:
            # revert everything
            a[i], a[j] = vi, vj
            conv[i:i + n] -= delta
            conv[j:j + n] += delta
            conv[2 * i] -= d * d
            conv[2 * j] -= d * d
            conv[i + j] += 2.0 * d * d

        if it % 500_000 == 0:
            print(f"[Swap] iter {it:,} – best obj = {best_val:.6f}")

    return best_a, best_val
