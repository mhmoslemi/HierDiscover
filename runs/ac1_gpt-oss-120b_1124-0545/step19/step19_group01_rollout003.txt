"""
Propose a low‑autocorrelation step function.

The search proceeds in three stages:

1. **Massive random seeding** – a very large pool of analytically‑parameterised
   shapes (beta, Gaussian, exponential, power‑law, sine‑window) together with
   random Dirichlet draws and mixtures of two Gaussians.  For each shape we
   generate many random parameter settings, normalise the vector to sum = 1,
   and evaluate the cheap surrogate 2·n·max(conv(a,a)).  The best 25 candidates
   are kept for further refinement.

2. **Local refinement** – each of the top candidates is refined by:
      * a permutation‑based simulated‑annealing hill‑climber,
      * a mirror‑descent (exponentiated‑gradient) optimiser that follows the
        sub‑gradient of the currently active convolution index,
      * a short “restart” perturb‑and‑refine loop.
   The best sequence seen across all seeds becomes the global best.

3. **Final polishing** – the global best candidate is polished for the
   remaining time budget with a long exponentiated‑gradient run (≈ 10⁶
   iterations) and a final random‑restart warm‑start.

All steps respect the 1000 s wall‑time limit; the best sequence found is
stored in the global variable `height_sequence_1` and returned as a plain
Python list.  The algorithm is deliberately more exhaustive than the
reference version and uses a richer family of shapes together with longer
optimisation runs, which typically pushes the surrogate (and true) value
below the 1.503 target.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# FFT‑based convolution utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# Surrogate objective (sum = 1 → 2·n·max(conv))
# ------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float, mu_frac: float = 0.5) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i-(n-1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_two_gaussians(
    n: int,
    mu1: float,
    sigma1_frac: float,
    mu2: float,
    sigma2_frac: float,
    w: float,
) -> np.ndarray:
    """Mixture w·N(mu1,σ1)+ (1‑w)·N(mu2,σ2). Parameters are fractions of n."""
    sigma1 = sigma1_frac * n
    sigma2 = sigma2_frac * n
    i = np.arange(n, dtype=np.float64)
    g1 = np.exp(-((i - mu1) ** 2) / (2.0 * sigma1 ** 2))
    g2 = np.exp(-((i - mu2) ** 2) / (2.0 * sigma2 ** 2))
    a = w * g1 + (1.0 - w) * g2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip, renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# ------------------------------------------------------------
def _exp_gradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 5.0,
    tie_eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent using sub‑gradient of the active convolution entry."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space (softmax) to stay on the simplex
    log_a = np.log(a + 1e-30)

    rng = np.random.default_rng(0xC0FFEE)

    # pre‑compute FFT length for reuse
    L = 2 * n - 1
    N = _next_pow2(L)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # forward convolution (full)
        fx = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(fx * fx, n=N)[:L]
        max_c = conv.max()
        # pick a random index among (almost) maximisers to avoid stagnation
        candidates = np.where(conv >= max_c - tie_eps)[0]
        max_idx = int(rng.choice(candidates))

        # sub‑gradient for selected quadratic term
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size schedule (η₀ / √t)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Permutation‑based simulated‑annealing hill climbing
# ------------------------------------------------------------
def _perm_hill_climb(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Swap‑based annealing that also allows tiny value perturbations."""
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (linear cooling)
    T_start = 1e-3 * cur_val
    T_end = 1e-7 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # cooling
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.60:  # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:  # small Gaussian perturb
            sigma = 0.001 / math.sqrt(it + 1.0)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The algorithm uses a huge random seed pool, permutation‑hill‑climbing,
    exponentiated‑gradient refinement and a final long polishing phase.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – power‑of‑two length gives a fast FFT.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (useful for continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                # normalise to ℓ₁‑norm = 1
                prev_arr = np.maximum(arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr = prev_arr / s
                else:
                    prev_arr = None
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # 1. Massive random seed generation
    # ------------------------------------------------------------------
    seeds = []

    # deterministic baselines
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Parameter sweeps (dense grids)
    for gamma in np.linspace(0.1, 3.5, 30):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma_frac in np.linspace(0.03, 0.35, 30):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.0, 6.0, 30):
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.2, 40):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Broad beta‑shape exploration
    for _ in range(200):
        a = 10 ** rng.uniform(-2, 1)   # log‑uniform in [0.01, 10]
        b = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (wide concentration range)
    for conc in np.logspace(-2, 1, 10):
        for _ in range(12):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (purely random magnitudes)
    for _ in range(80):
        seeds.append(_random_exponential(n, rng))

    # Mixtures of two Gaussians – highly expressive
    for _ in range(150):
        mu1 = rng.uniform(0, n - 1)
        mu2 = rng.uniform(0, n - 1)
        sigma1 = rng.uniform(0.01, 0.30)
        sigma2 = rng.uniform(0.01, 0.30)
        w = rng.random()
        seeds.append(_mix_two_gaussians(n, mu1, sigma1, mu2, sigma2, w))

    # Random permutations of a few strong base shapes
    base_shapes = [
        _exp_decay(n, 0.003),
        _sym_exp_decay(n, 0.004),
        _beta_shape(n, 2.0, 2.0),
        _gaussian(n, 0.18),
    ]
    for base in base_shapes:
        for _ in range(6):
            seeds.append(rng.permuted(base))

    # Include previous best (if any) and a few perturbed copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(8):
            seeds.append(_perturb(prev_arr, rng, sigma=0.005))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the top‑k
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(25, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seed surrogate values ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f" #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Global best initialisation
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start local refinement
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a share of the remaining time (≈ 15 % per remaining seed)
        remaining = deadline - time.time()
        share = remaining * 0.15 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # ---- a) permutation hill‑climbing ----
        seq_perm, val_perm = _perm_hill_climb(
            seed_vec.copy(),
            n,
            sub_deadline,
            rng,
            max_iters=180_000,
        )

        # ---- b) exponentiated‑gradient refinement ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(
                seq_perm,
                n,
                sub_deadline,
                max_iters=300_000,
                eta0=6.0,
            )
        else:
            seq_eg, val_eg = seq_perm, val_perm

        # ---- c) short random‑restart polishing (optional) ----
        seq_restart = seq_eg
        val_restart = val_eg
        for _ in range(3):
            if time.time() >= sub_deadline:
                break
            pert = _perturb(seq_restart, rng, sigma=0.008)
            seq_tmp, val_tmp = _exp_gradient_opt(
                pert, n, sub_deadline, max_iters=60_000, eta0=4.0
            )
            if val_tmp < val_restart:
                seq_restart, val_restart = seq_tmp, val_tmp

        # ---- d) pick the best among the refinements ----
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_perm, val_perm),
            (seq_eg, val_eg),
            (seq_restart, val_restart),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"   *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing on the overall best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long exponentiated‑gradient polishing ---")
        final_seq, final_val = _exp_gradient_opt(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            eta0=5.0,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation with the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
