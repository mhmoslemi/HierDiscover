"""
Propose a step‑function (list of non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is an evolutionary search with a large pool of analytically
parameterised seeds, random Dirichlet draws and convex mixing.  The best
candidate is polished with a fast exponentiated‑gradient (mirror‑descent)
on the exact max‑convolution.  The whole search respects a 1000 s budget
and stores the result in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoother (fallback ok)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective used for internal optimisation – i.e. the same
    quantity that ``evaluate_sequence`` returns, but without the input
    validation overhead.  The sequence can be arbitrary; the function
    normalises internally.
    """
    n = len(seq)
    s = float(seq.sum())
    # Guard against division‑by‑zero – return a large value.
    if s < 1e-12:
        return 1e12
    conv = _conv_fft(seq, seq)
    m = float(conv.max())
    return 2.0 * n * m / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all return *unnormalised* vectors; we normalise later)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a[a < 0] = 0.0
    return a


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    # alpha ∈ [0,1]; 0 → rectangular, 1 → Hann
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)               # modified Bessel I0
    return a


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    return a


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a


def _double_gaussian(n: int,
                     sigma_frac: float = 0.15,
                     shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.exponential(scale=1.0, size=n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.001) -> np.ndarray:
    """Add zero‑mean Gaussian noise, clip, and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the *exact* max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    Returns the best sequence seen and its surrogate (objective) value.
    """
    a = start_seq.copy()
    # work in log‑space to keep us on the simplex automatically
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # compute full convolution once per iteration
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the entry conv[k]
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size decreasing like 1/√t
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Evolutionary / mixing driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) and stores it in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # keep a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any) as an extra seed
    # ------------------------------------------------------------------
    prev_arr = None
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if arr.size == n:
                prev_arr = arr / arr.sum() if arr.sum() > 0 else None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1.  Build a large pool of initial seeds
    # ------------------------------------------------------------------
    print("\n--- Generating initial seeds ---")
    seed_pool = []

    # deterministic analytic families (coarse grids)
    seed_pool.append(_uniform(n))
    seed_pool.append(_triangular(n))

    for gamma in np.linspace(0.2, 4.0, 16):
        seed_pool.append(_sine_window(n, gamma=gamma))

    for beta in np.linspace(0.5, 15.0, 14):
        seed_pool.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.07, 0.30, 16):
        seed_pool.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 3.0, 19):
        seed_pool.append(_power(n, p))

    for a_ in np.linspace(0.2, 5.0, 12):
        for b_ in np.linspace(0.2, 5.0, 12):
            seed_pool.append(_beta_shape(n, a_, b_))

    for alpha in np.linspace(0.05, 0.9, 12):
        seed_pool.append(_tukey_window(n, alpha))

    for gamma in np.linspace(0.05, 10.0, 20):
        seed_pool.append(_exp_decay(n, gamma))

    for steep in np.linspace(0.5, 30.0, 12):
        for shift in np.linspace(0.1, 0.9, 9):
            seed_pool.append(_logistic_decay(n, steep, shift))

    # random Dirichlet draws with a wide range of concentrations
    for conc in (0.01, 0.02, 0.05, 0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        for _ in range(3):
            seed_pool.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seed_pool.append(_random_exponential(n, rng))

    # perturbations of the previous best (if any)
    if prev_arr is not None:
        seed_pool.append(prev_arr)
        for _ in range(6):
            seed_pool.append(_perturb(prev_arr, rng, sigma=0.002))

    # ------------------------------------------------------------------
    # 2.  Normalise everything (L1 = 1) and evaluate the surrogate
    # ------------------------------------------------------------------
    normed = []
    for vec in seed_pool:
        if time.time() > deadline:
            break
        vec = np.asarray(vec, dtype=np.float64)
        s = vec.sum()
        if s <= 0.0:
            continue
        vec = vec / s                     # L1‑normalise
        val = _objective_surrogate(vec)   # surrogate = 2·n·max(conv)
        normed.append((vec, val))

    # keep the top few as an elite pool
    normed.sort(key=lambda x: x[1])
    elite_size = 30
    elite = normed[:elite_size]
    best_seq, best_val = elite[0]
    print(f"Initial best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Evolutionary loop – mixing + mutation
    # ------------------------------------------------------------------
    mix_interval = 0.02          # seconds between status prints
    next_print = time.time() + mix_interval
    generation = 0

    while time.time() < deadline:
        generation += 1

        # pick two parents with bias towards the better half
        parent1 = elite[rng.integers(0, elite_size // 2)]
        parent2 = elite[rng.integers(0, elite_size // 2)]

        a, _ = parent1
        b, _ = parent2
        alpha = rng.random()                     # convex weight
        child = alpha * a + (1.0 - alpha) * b
        child = child / child.sum()

        # occasional perturbation to increase diversity
        if rng.random() < 0.15:
            child = _perturb(child, rng, sigma=0.001)

        child_val = _objective_surrogate(child)

        # If we are better than the worst elite member, insert
        if child_val < elite[-1][1]:
            elite[-1] = (child, child_val)
            elite.sort(key=lambda x: x[1])
            if child_val < best_val:
                best_seq, best_val = child, child_val

        # Verbose progress
        if time.time() >= next_print:
            elapsed = time.time() - start_time
            print(f"[Gen {generation:5d}] best surrogate = {best_val:.6f} (elapsed {elapsed:.1f}s)")
            next_print = time.time() + mix_interval

    # ------------------------------------------------------------------
    # 4.  Local polishing of the current best with exponentiated‑gradient
    # ------------------------------------------------------------------
    print("\n--- Polishing best candidate with EG ---")
    polish_deadline = deadline - 0.5   # tiny head‑room for final steps
    best_seq, best_val = _exp_gradient_opt(best_seq,
                                           n,
                                           polish_deadline,
                                           max_iters=2_000_000,
                                           eta0=6.0)
    print(f"Polish surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final official evaluation (no warm‑up) and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
