"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Seed pool** – Deterministic analytic families (uniform,
   triangular, Gaussian, beta, exponential, sinusoidal, semi‑circle,
   polynomial‑tent) plus a massive sweep of random Dirichlet / sparse /
   low‑frequency shapes.  All seeds are ℓ₁‑normalised.

2. **Elite selection** – Keep the 50 best seeds (exact objective).

3. **Continuation on a smooth surrogate** – For every elite we minimise
   a log‑sum‑exp smooth approximation of ``max(conv(a,a))`` using
   L‑BFGS‑B on the logits of a soft‑max (so the simplex constraint is
   satisfied automatically).  The smoothing parameter β is increased
   geometrically (1 → … → 1e9).  A fairly large iteration budget
   (≈8 000 per β) is used while respecting the global time limit.

4. **Mirror‑descent (exponentiated‑gradient)** – Several restarts from
   random Dirichlet draws.  The exact sub‑gradient of the active lag
   of the (non‑smooth) max‑convolution is used; step‑size ηₜ=η₀/√t.
   Long runs are performed from the current global best.

5. **Permutation hill‑climb** – The multiset of values is kept, but
   their ordering (and thus the diagonal sums) is permuted.  Random
   permutations are sampled; the best improvement is kept.

6. **Stochastic mass‑transfer hill‑climb** – Two complementary local
   searches (plain random transfers and transfers away from the
   currently maximal diagonal) are executed for many iterations.

7. **Final high‑β polishing** – One more L‑BFGS run with very large β
   to clean residual irregularities.

Throughout the run the current best sequence and its objective value
are printed.  The final ``height_sequence_1`` global variable contains
the best list found.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (active‑lag only)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (active diagonal only)."""
    n = len(a)
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))                     # active diagonal index
    # i runs over all positions that contribute to that diagonal
    i_lo = max(0, k - (n - 1))
    i_hi = min(n - 1, k)
    i = np.arange(i_lo, i_hi + 1, dtype=int)
    j = k - i                                   # partner index
    g = np.zeros_like(a)
    g[i] = a[j]
    g[j] += a[i]                                # each product appears twice except when i=j
    # The expression above adds a[i] * a[j] + a[j] * a[i] = 2 a[i] a[j];
    # the diagonal (i=j) contributes only a[i] (once) which is correct for d_k = Σ a_i a_j.
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad) where `v` are logits of the simplex.
    The surrogate approximates max(conv(a,a)) with a log‑sum‑exp.
    """
    a = _softmax(v)                     # ℓ₁‑normalised probabilities
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # smooth max
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (derivation from the log‑sum‑exp)
    w = exp_term / sum_exp               # weights for each diagonal
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]  # slice back to length n

    # back‑prop through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float],
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    `start_vec` must be ℓ₁‑normalised.
    """
    v = np.log(start_vec + 1e-30)  # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(
                fun=_smooth_objective,
                x0=v,
                args=(n, float(beta)),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_beta,
                         "ftol": 1e-12,
                         "disp": False}
            )
            v = res.x
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exact sub‑gradient."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 250_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic local searches
# ----------------------------------------------------------------------
def _mass_transfer_local_search(start_a: np.ndarray,
                               n: int,
                               deadline: float,
                               iters: int = 200_000,
                               step_frac: float = 0.012) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfers, accept only improvements."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


def _targeted_local_search(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           iters: int = 120_000,
                           step_frac: float = 0.010) -> tuple[np.ndarray, float]:
    """Bias moves away from indices that belong to the currently maximal diagonal."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for _ in range(iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        i = rng.integers(i_lo, i_hi + 1)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

# ----------------------------------------------------------------------
def _permutation_hill_climb(start_a: np.ndarray,
                            n: int,
                            deadline: float,
                            trials: int = 5_000) -> tuple[np.ndarray, float]:
    """
    Keep the multiset of values but permute the ordering.
    Accept only strictly improving permutations.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(314159265)

    for _ in range(trials):
        if time.time() > deadline:
            break
        perm = rng.permutation(n)
        cand = a[perm]
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _semicircle_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (i - c) / c
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _poly_tent(n: int, power: float = 1.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) * (n - i)) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Seed construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_exponential_decay(n, 0.001))

    # additional analytic families
    seeds.append(_sin_shape(n))
    seeds.append(_semicircle_shape(n))
    seeds.append(_poly_tent(n, power=1.2))
    seeds.append(_poly_tent(n, power=2.0))

    # power‑law sweep (including steep decay)
    p_vals = np.linspace(0.2, 1.2, 12)
    shifts = [0.0, 2.0, 5.0, 20.0, 50.0, 200.0]
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # random Dirichlet (wide concentration range)
    for conc in (0.01, 0.03, 0.07, 0.12, 0.25, 0.5, 0.9):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse selections
    for frac in (0.01, 0.02, 0.05, 0.1, 0.15):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine mixtures
    for nf in (2, 3, 4, 5):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures
    base = np.arange(len(seeds))
    for _ in range(300):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising candidates
    for _ in range(18):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # warm‑start from previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

                                                    max_iters=iters_per_restart,
                                                    eta0=2.0)
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:

            seq_rand, val_rand = _mass_transfer_local_search(global_best_seq, n,
                                                             deadline,
                                                             iters=iters_rand,
                                                             step_frac=0.011)
            print(f"[RandomLocal] best = {val_rand:.6f}")
            if val_rand < global_best_val:
                global_best_seq = seq_rand.copy()
                global_best_val = val_rand
                print(f"[Update] New best after random local = {global_best_val:.6f}")

        remaining = deadline - time.time()
        iters_tgt = int(min(250_000, remaining * 30))
        if iters_tgt > 0:
            print("\n--- Targeted local search (active diagonal) ---")
            seq_tgt, val_tgt = _targeted_local_search(global_best_seq, n,
                                                      deadline,
                                                      iters=iters_tgt,
                                                      step_frac=0.010)
            print(f"[TargetedLocal] best = {val_tgt:.6f}")
            if val_tgt < global_best_val:
                global_best_seq = seq_tgt.copy()
                global_best_val = val_tgt
                print(f"[Update] New best after targeted local = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1500, int(min(3500, remaining * 0.05 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq, n, deadline,
                                           beta_schedule=[5e6, 1e7, 5e7,
                                                          1e8, 5e8, 1e9],
                                           max_iter_per_beta=per_beta)
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
