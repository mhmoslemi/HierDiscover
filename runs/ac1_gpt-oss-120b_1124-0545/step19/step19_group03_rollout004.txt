def _adam_optimize(initial_params, n, cos_mat, sin_mat, deadline, beta_schedule, iters_per_beta=2000, lr=0.05):
    D = initial_params.shape[0]
    params = initial_params.copy()
    m = np.zeros_like(params)
    v = np.zeros_like(params)
    best_seq = None
    best_val = np.inf
    t = 0
    eps = 1e-8
    beta1 = 0.9
    beta2 = 0.999
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for it in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            # compute log_a and a
            log_a = params[0] + cos_mat.T @ params[1::2] + sin_mat.T @ params[2::2]   # careful shapes
            # shapes: cos_mat is (K,n). Params[1::2] is length K. So compute sum_{k} param[2k-1] * cos_mat[k]
            # Actually indexing: after constant param[0], for k from 0 to K-1:
            #   c_k = params[1+2*k]
            #   d_k = params[2+2*k]
            # Compute log_a = param[0] + Σ c_k * cos_mat[k] + d_k * sin_mat[k]
            # Use broadcasting:
            log_a = params[0] + np.dot(params[1::2], cos_mat) + np.dot(params[2::2], sin_mat)
            # Then a = softmax(log_a)
            a = _softmax(log_a)
            # conv using _conv_fft
            conv = _conv_fft(a, a)
            max_conv = conv.max()
            shift = max_conv
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            max_smooth = shift + np.log(sum_exp) / beta
            obj = 2.0 * n * max_smooth
            # gradient w.r.t a:
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_log_a = a * (grad_a - dot)
            # gradient wrt params:
            grad_params = np.empty_like(params)
            grad_params[0] = np.sum(grad_log_a)
            # compute for each k:
            # dot with cos and sin vectors
            grad_params[1::2] = cos_mat @ grad_log_a   # each row cos_mat[k] dot grad_log_a yields gradient for c_k
            grad_params[2::2] = sin_mat @ grad_log_a
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_params
            v = beta2 * v + (1 - beta2) * (grad_params ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            params = params - lr * m_hat / (np.sqrt(v_hat) + eps)
            # track best exact objective value:
            exact_val = _objective_surrogate(a)
            if exact_val < best_val:
                best_val = exact_val
                best_seq = a.copy()
    return best_seq, best_val

"""
Improved step‑function optimisation.

The routine builds a rich pool of analytic seeds (uniform, triangular,
Gaussian, Beta‑shapes, an arcsine‑style “U‑shape”, low‑frequency cosine
curves, Dirichlet draws, etc.) and evaluates them with the exact
objective.  The best few seeds are turned into a low‑dimensional
parameterisation of the log‑density:

    log a_i = c0 + Σ_{k=1}^K (c_k cos(2π k i / n) + d_k sin(2π k i / n))

The parameters are optimised with a continuation Adam optimiser that
minimises a smooth approximation of the max‑convolution (log‑sum‑exp
with a schedule of increasing β).  The soft‑max mapping guarantees
non‑negativity and ℓ₁‑normalisation.  A few random initialisations are
also tried.  After the Adam phase a long exponentiated‑gradient (mirror
descent) polishing is performed.  The best sequence found is stored
in the global variable ``height_sequence_1`` and returned.

The algorithm respects a user‑supplied time budget (default 1000 s) and
includes many printouts for progress monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector, sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Random sum of low‑frequency cosine waves, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """U‑shaped distribution with density ∝ 1/√(x(1‑x))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active set (all entries within a tiny tolerance of the max)
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient contributions from every active index
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimisation of a log‑density (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _fit_params_to_seq(seq: np.ndarray,
                       cos_mat: np.ndarray,
                       sin_mat: np.ndarray) -> np.ndarray:
    """
    Least‑squares fit of the log‑density to the Fourier basis.
    Returns the parameter vector (c0, c₁, d₁, …, c_K, d_K).
    """
    log_a = np.log(seq + 1e-30)
    B = np.vstack([np.ones_like(log_a), cos_mat, sin_mat])   # shape (D, n)
    G = B @ B.T
    h = B @ log_a
    try:
        params = np.linalg.solve(G, h)
    except np.linalg.LinAlgError:
        params, *_ = np.linalg.lstsq(G, h, rcond=None)
    return params.astype(np.float64)


def _adam_optimize(initial_params: np.ndarray,
                   n: int,
                   cos_mat: np.ndarray,
                   sin_mat: np.ndarray,
                   deadline: float,
                   beta_schedule,
                   iters_per_beta: int = 1500,
                   lr: float = 0.05):
    """
    Adam optimiser on the smooth max‑convolution objective.
    Parameters are the Fourier coefficients of the log‑density.
    Returns the best exact sequence (according to the true max) found
    before the deadline.
    """
    K = cos_mat.shape[0]
    D = 1 + 2 * K
    params = initial_params.copy()
    m = np.zeros_like(params)
    v = np.zeros_like(params)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for it in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1

            # ----- compose log‑density and softmax -----
            # log_a = c0 + Σ c_k cos_k + d_k sin_k
            log_a = params[0] + np.dot(params[1::2], cos_mat) + np.dot(params[2::2], sin_mat)
            a = _softmax(log_a)

            # ----- smooth max of the convolution -----
            conv = _conv_fft(a, a)
            max_conv = conv.max()
            shift = max_conv                # for numerical stability
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth value (unused for convergence, but useful for logging)
            # smooth_max = shift + np.log(sum_exp) / beta
            # obj_smooth = 2.0 * n * smooth_max

            # ----- exact objective (for tracking best) -----
            exact_val = 2.0 * n * max_conv
            if exact_val < best_val:
                best_val = exact_val
                best_seq = a.copy()

            # ----- gradient w.r.t. a (from the smooth max) -----
            w = exp_beta / sum_exp               # shape = len(conv)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # slice back to length n

            # ----- back‑prop through the soft‑max -----
            dot = np.dot(grad_a, a)
            grad_log_a = a * (grad_a - dot)      # dL/d(log a)

            # ----- gradient w.r.t. parameters -----
            grad_params = np.empty_like(params)
            grad_params[0] = np.sum(grad_log_a)                # c0
            # cosine coefficients
            grad_params[1::2] = cos_mat @ grad_log_a
            # sine coefficients
            grad_params[2::2] = sin_mat @ grad_log_a

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad_params
            v = beta2 * v + (1.0 - beta2) * (grad_params ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            params = params - lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_seq, best_val

        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Low‑frequency random cosine curves
    for nf in (2, 4, 6, 8):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Exponential draws
    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    # Random beta‑shapes (including U‑shaped cases)
    for _ in range(30):
        a = rng.uniform(0.1, 3.0)
        b = rng.uniform(0.1, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # Previous best – if it exists, keep it and perturb a little
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                seeds.append(prev_arr / prev_arr.sum())
                for _ in range(6):
                    noisy = prev_arr + 0.003 * rng.normal(size=n)
                    noisy = np.maximum(noisy, 0.0)
                    s = noisy.sum()
                    if s > 0:
                        seeds.append(noisy / s)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2. Quick exact evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (exact surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Adam optimisation on the top seeds (Fourier‑parameterised)
    # --------------------------------------------------------------
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1_000.0, 2_000.0, 5_000.0,
                  20_000.0, 50_000.0, 100_000.0, 200_000.0]
    iters_per_beta = 1200      # a few thousand updates per seed

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # Fit Fourier parameters to the seed (least‑squares)
        init_params = _fit_params_to_seq(seed_vec, cos_mat, sin_mat)

        # Small budget for each seed – proportion of the remaining time
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(remaining * 0.25, 45.0)   # at most ~45 s

        seq, val = _adam_optimize(init_params, n, cos_mat, sin_mat,
                                  sub_deadline, beta_sched,
                                  iters_per_beta=iters_per_beta,
                                  lr=0.05)

        if seq is not None and val < global_best_val:
            global_best_seq = seq
            global_best_val = val
            print(f"[Seed {si+1}/{top_k}] new global best via Adam = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. A few random Fourier initialisations
    # --------------------------------------------------------------
    for rnd_i in range(5):
        if time.time() >= deadline:
            break
        rnd_params = rng.normal(scale=0.1, size=1 + 2 * K).astype(np.float64)
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(remaining * 0.2, 30.0)

        seq, val = _adam_optimize(rnd_params, n, cos_mat, sin_mat,
                                  sub_deadline, beta_sched,
                                  iters_per_beta=iters_per_beta,
                                  lr=0.05)

        if seq is not None and val < global_best_val:
            global_best_seq = seq
            global_best_val = val
            print(f"[Random {rnd_i+1}] new global best via Adam = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final long exponentiated‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing ---")
        seq, val = _exp_gradient_opt_multi(global_best_seq,
                                           n,
                                           deadline,
                                           max_iters=800_000,
                                           eta0=5.0)
        if val < global_best_val:
            global_best_seq = seq
            global_best_val = val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
