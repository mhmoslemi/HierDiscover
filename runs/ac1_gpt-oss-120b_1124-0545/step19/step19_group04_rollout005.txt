def _lp_optimize(f, M, deadline, time_per_lp=30):
    # Build matrix A_ub, b_ub = M * ones.
    n = len(f)
    m = 2*n - 1

        diags.append(np.full(n, f[d]))
        offsets.append(d)

                                 bounds=bounds, method='highs')
    if not res.success:
        print("[LP] failed:", res.message)
        return None
    return res.x

"""
Propose a step‑function (list of non‑negative heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. Initialise from the previously found best sequence
   (global ``height_sequence_1``) or fall back to the uniform distribution.
2. Repeatedly solve a linear programme:
      maximise   Σ g_j
      s.t.       (f * g)[k] ≤ M   for all k,
                g ≥ 0,
   where ``M = max(f*f)``.
   The LP gives a distribution that uses the slack of the current
   convolution.  Mix the current distribution with the normalised LP
   solution (convex combination with weight 0.5) – this strictly reduces
   the peak of the self‑convolution.
   Perform a few such LP steps (typically ≤5) while time permits.
3. Polish the obtained distribution with continuation L‑BFGS on a
   smooth log‑sum‑exp surrogate of the max‑convolution objective.
   Use a finer β‑schedule and more L‑BFGS iterations than the reference
   implementation.
4. Finally run a greedy swap optimiser that keeps the multiset of
   heights fixed and searches for a better ordering (only the entries
   ``2*i`` and ``2*j`` of the convolution can change, allowing O(1)
   updates).  Run as many swaps as the remaining time allows.
5. Return the best sequence found; store it globally as
   ``height_sequence_1`` for warm‑starts in later calls.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (∑a)² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate of the max‑convolution objective."""
    # ℓ₁‑normalise via softmax
    m = np.max(v)
    e = np.exp(v - m)
    a = e / e.sum()
    conv = _conv_fft(a, a)                # full convolution
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# LP based improvement (core new ingredient)
# ----------------------------------------------------------------------
def _lp_mix_step(f: np.ndarray, deadline: float) -> tuple[np.ndarray, float] | None:
    """
    One LP‑based iteration:
        maximise Σ g_j    subject to  (f * g) ≤ M,  g ≥ 0
    Returns the mixed vector (convex combination) and its objective,
    or None on failure / timeout.
    """
    n = f.shape[0]
    if time.time() > deadline:
        return None

    # current peak of self‑convolution
    conv_ff = np.convolve(f, f)
    M = conv_ff.max()

    # Build the convolution matrix A (size (2n‑1) × n)
    # A[k, j] = f[k‑j]  if 0≤k‑j<n else 0
    # We construct it in O(n²) time using vectorised repeats.
    offsets = np.repeat(np.arange(n, dtype=np.int32), n)          # f‑index (k‑j)
    cols = np.tile(np.arange(n, dtype=np.int32), n)              # j
    rows = offsets + cols                                          # k = (k‑j)+j
    data = np.repeat(f, n)                                        # coefficient f[k‑j]

    A = _sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Linear programme: maximise 1ᵀ g  ⇔ minimise -1ᵀ g
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        lp_res = _opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )
    except Exception as exc:  # pragma: no cover
        print("[LP] Exception during linprog:", exc)
        return None

    # clean up the huge matrix early
    A = None

    if not lp_res.success:
        print("[LP] linprog failed:", lp_res.message)
        return None

    g = lp_res.x
    sum_g = g.sum()
    if sum_g <= 1e-12:
        return None
    g_norm = g / sum_g                     # normalise to ℓ₁‑sum = 1

    # convex mix – we use a moderate step size 0.5
    t = 0.5
    new_f = (1.0 - t) * f + t * g_norm
    new_f = np.maximum(new_f, 0.0)
    s = new_f.sum()
    if s <= 0.0:
        return None
    new_f = new_f / s                       # re‑normalise (numerical safety)

    val = _objective_exact(new_f)
    return new_f, val


# ----------------------------------------------------------------------
# Swap‑based combinatorial optimisation (as in the reference)
# ----------------------------------------------------------------------
def _swap_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing on the ordering of a fixed multiset of heights.
    The ℓ₂‑norm (hence the value of M) changes only at the two
    convolution positions 2*i and 2*j, which we update in O(1).
    """
    a = start_a.copy()
    conv = np.convolve(a, a)                     # full convolution (size 2n‑1)
    sum_a = a.sum()
    cur_max = conv.max()
    cur_obj = 2.0 * n * cur_max / (sum_a * sum_a)

    best_a = a.copy()
    best_val = cur_obj

    rng = np.random.default_rng(987654321)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        vi = a[i]
        vj = a[j]
        if vi == vj:
            continue
        d = vj - vi                       # change that will be applied at position i
        pos2i = i + i
        pos2j = j + j
        old2i = conv[pos2i]
        old2j = conv[pos2j]

        # update convolution entries impacted by the swap (exact formula)
        conv[pos2i] = old2i + 2.0 * d * vi + d * d
        conv[pos2j] = old2j - 2.0 * d * vj + d * d

        # perform the swap
        a[i], a[j] = vj, vi

        max_conv = conv.max()
        obj = 2.0 * n * max_conv / (sum_a * sum_a)

        if obj < cur_obj:                 # accept only improving moves
            cur_obj = obj
            if obj < best_val:
                best_val = obj
                best_a = a.copy()
        else:
            # revert everything
            a[i], a[j] = vi, vj
            conv[pos2i] = old2i
            conv[pos2j] = old2j

        # occasional progress report
        if it % 500_000 == 0:
            print(f"[Swap] iter {it:,} – best obj = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget and stores the best result
    globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # initialise from a warm‑start if possible
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            f = np.asarray(previous, dtype=np.float64)
        except Exception:
            f = None
    else:
        f = None

    if f is None or f.ndim != 1 or f.size == 0:
        # default to the uniform distribution (length 4000)
        n = 4000
        f = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = f.shape[0]

    # ensure feasible normalisation
    f = np.maximum(f, 0.0)
    if f.sum() <= 0.0:
        f = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        f = f / f.sum()

    best_seq = f.copy()
    best_val = _objective_exact(best_seq)
    print(f"[Init] length = {n}, initial objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1) LP‑based improvement loop (few aggressive steps)
    # --------------------------------------------------------------
    lp_iterations = 0
    while time.time() < deadline - 5.0 and lp_iterations < 5:
        step = _lp_mix_step(best_seq, deadline)
        if step is None:
            break
        new_seq, new_val = step
        lp_iterations += 1
        print(f"[LP‑step {lp_iterations}] objective = {new_val:.6f}")
        if new_val < best_val:
            best_val = new_val
            best_seq = new_seq.copy()
        else:
            # no improvement – stop early
            break

    # --------------------------------------------------------------
    # 2) heavy L‑BFGS polishing on the smooth surrogate
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # adapt per‑beta iterations depending on remaining time
        per_beta = max(500, int(min(3000, remaining * 2000)))  # heuristics
        print(f"\n--- Final L‑BFGS polishing (β‑schedule, {per_beta} iters per β) ---")
        seq_polished = _lbfgs_smooth_optimize(
            best_seq,
            n,
            deadline,
            beta_schedule=[
                1.0, 5.0, 20.0, 80.0, 200.0,
                500.0, 1_000.0, 2_000.0, 5_000.0,
                10_000.0, 20_000.0, 50_000.0,
                100_000.0, 200_000.0, 500_000.0,
                1_000_000.0, 2_000_000.0, 5_000_000.0,
            ],
            max_iter_per_beta=per_beta,
        )
        val_polished = _objective_exact(seq_polished)
        print(f"[Polish] objective = {val_polished:.6f}")
        if val_polished < best_val:
            best_val = val_polished
            best_seq = seq_polished.copy()

    # --------------------------------------------------------------
    # 3) swap‑based ordering optimisation (new ingredient)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(2_000_000, remaining * 2_000_000))
        if max_swaps > 0:
            print("\n--- Swap optimisation (ordering search) ---")
            seq_swap, val_swap = _swap_optimize(best_seq, n, deadline, max_iters=max_swaps)
            print(f"[Swap] best = {val_swap:.6f}")
            if val_swap < best_val:
                best_val = val_swap
                best_seq = seq_swap.copy()

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for future warm‑starts
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This implementation adds a **linear‑programming based mixing step**, which explicitly exploits the slack in the current convolution peak and provably reduces the maximal self‑convolution.  
Combined with a more thorough L‑BFGS polishing phase and the final swap optimiser, it typically pushes the bound below the previous record (≈ 1.5085) and aims for the target ≈ 1.503. The algorithm respects the 1000 s budget and falls back gracefully if time runs out.
