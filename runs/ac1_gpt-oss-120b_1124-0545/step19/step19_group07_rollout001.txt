def _perm_hill_climb(start_a: np.ndarray, n: int, deadline: float, max_iters: int) -> tuple[np.ndarray, float]:
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng()
    it = 0
    while it < max_iters and time.time() < deadline:
        # Use targeted or random based on probability
        if rng.random() < 0.1:
            # target
            conv = _conv_fft(a, a)
            k = int(np.argmax(conv))
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idxs = np.arange(i_start, i_end + 1)
            js = k - idxs
            # compute products
            products = a[idxs] * a[js]
            max_idx = int(np.argmax(products))
            i = int(idxs[max_idx])
            j = int(js[max_idx])
            # find index m with smallest value not i/j
            m_candidates = np.where((np.arange(n) != i) & (np.arange(n) != j))[0]
            if len(m_candidates)==0:
                continue
            m = int(m_candidates[a[m_candidates].argmin()])
            # swap i with m
            cand = a.copy()
            cand[i], cand[m] = cand[m], cand[i]
            val = _objective_exact(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                a = cand
                it += 1
                continue
            # swap j with m
            cand = a.copy()
            cand[j], cand[m] = cand[m], cand[j]
            val = _objective_exact(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                a = cand
                it += 1
                continue
        else:
            # random swap
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
            val = _objective_exact(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                a = cand
        it += 1
    return best_a, best_val
```

* Finish with a dedicated permutation‑hill‑climbing phase that swaps
  indices to break large product pairs on the dominant lag.  This phase
  is cheap (single FFT per swap) and is very effective for shaving the
  last few thousandths off the bound.
* All steps respect the 1000 s wall‑time limit and continuously keep the
  best vector in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → ℓ₁‑normalised (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) – only the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogates (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp approx. of max(conv).  Returns (obj, grad) w.r.t. logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # length 2n‑1
    shift = np.max(conv)                 # for numeric stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Continuation L‑BFGS on the log‑sum‑exp surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta,
                         "disp": False,
                         "ftol": 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 2.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    a = np.maximum(a, 1e-30)          # avoid zeros in log
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


def _perm_hill_climb(start_a: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Greedy permutation search.
    Swaps two indices if the move lowers the exact objective.
    10 % of the moves are *targeted* (swap a high‑product pair with a
    smallest element) to focus on the dominant lag.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng()
    it = 0
    # pre‑compute index array for fast exclusion
    all_idx = np.arange(n, dtype=np.int64)

    while it < max_iters and time.time() < deadline:
        it += 1
        if rng.random() < 0.1:
            # ---- targeted move ----
            conv = _conv_fft(a, a)
            k = int(np.argmax(conv))          # dominant lag
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            products = a[i_range] * a[j_range]
            max_pair = int(np.argmax(products))
            i = int(i_range[max_pair])
            j = int(j_range[max_pair])

            # pick an index with the smallest value (excluding i,j)
            mask = (all_idx != i) & (all_idx != j)
            candidates = all_idx[mask]
            if candidates.size == 0:
                continue
            m = int(candidates[a[candidates].argmin()])

            # try swapping i↔m
            cand = a.copy()
            cand[i], cand[m] = cand[m], cand[i]
            val = _objective_exact(cand)
            if val < best_val:
                best_val, best_a, a = val, cand.copy(), cand
                continue

            # try swapping j↔m
            cand = a.copy()
            cand[j], cand[m] = cand[m], cand[j]
            val = _objective_exact(cand)
            if val < best_val:
                best_val, best_a, a = val, cand.copy(), cand
                continue
        else:
            # ---- random swap ----
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
            val = _objective_exact(cand)
            if val < best_val:
                best_val, best_a, a = val, cand.copy(), cand

    return best_a, best_val


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _zigzag(v: np.ndarray) -> np.ndarray:
    """Interleave largest / smallest values to obtain a ‘zig‑zag’ ordering."""
    s = np.sort(v)[::-1]               # descending
    out = np.empty_like(s)
    left, right = 0, len(s) - 1
    i = 0
    while left <= right:
        out[i] = s[left]
        i += 1
        left += 1
        if left <= right:
            out[i] = s[right]
            i += 1
            right -= 1
    return out / out.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    n = 4000                              # problem size (good trade‑off)

    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # 1️⃣  Collect a modest pool of seeds (including any previous best)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = [
        _uniform(n),
        _triangular(n),
        _exponential_decay(n, 0.001),
        _exponential_decay(n, 0.01),
        _exponential_decay(n, 0.05),
        _exponential_decay(n, 0.1),
    ]

    # a few random Dirichlet draws
    for conc in (0.02, 0.1, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random beta shapes
    for _ in range(6):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, alpha, beta))

    # zig‑zag reshuffle of a Dirichlet draw
    v = _random_dirichlet(n, 0.5, rng)
    seeds.append(_zigzag(v))

    if prev_arr is not None and prev_arr.size == n and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())

    # enforce non‑negativity and ℓ₁‑normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        clean.append(v / s if s > 0 else _uniform(n))
    seeds = clean

    # ------------------------------------------------------------------
    # 2️⃣  Keep the elite few seeds (lowest exact objective)
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # keep top‑K for heavy L‑BFGS refinement
    seeds.sort(key=_objective_exact)
    elite = seeds[:min(5, len(seeds))]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # 3️⃣  L‑BFGS on the smooth surrogate (light schedule)
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec, n, deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=2000,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Exact mirror‑descent (exponentiated‑gradient) – long run
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(6_000_000, remaining * 2500))  # ~0.4 ms per it ≈ 2 k it/s
        print("\n--- EG optimisation (exact max) ---")
        seq_eg, val_eg = _exp_gradient_optimize(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            eta0=2.5,
        )
        print(f"[EG] best = {val_eg:.6f}")
        if val_eg < global_best_val:
            global_best_seq, global_best_val = seq_eg.copy(), val_eg
            print(f"[Update] New best after EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Permutation hill‑climbing – the final polishing stage
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(500_000, remaining * 5000))  # aim for several hundred‑k swaps
        print("\n--- Permutation hill‑climbing ---")
        seq_perm, val_perm = _perm_hill_climb(
            global_best_seq, n, deadline,
            max_iters=max_swaps,
        )
        print(f"[PermHC] best = {val_perm:.6f}")
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[Update] New best after PermHC = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final evaluation (via the provided evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible warm‑starts in the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
