"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

The core idea is to start from the *arcsine* distribution
(Beta(½,½)).  This distribution has the remarkable property that the
distribution of the sum of two independent copies is (almost) uniform,
so its discrete self‑convolution is extremely flat – the maximum
coefficient is close to the theoretical lower bound 1/(2·n‑1).  On
top of that we run a cheap exponentiated‑gradient (mirror‑descent)
optimiser that directly uses the exact sub‑gradient of the max‑convolution,
and finish with a short stochastic local‑search.

The whole routine respects the 1000 s wall‑time limit, keeps the best
vector in the global variable ``height_sequence_1`` and returns it as a
plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable ℓ₁‑normalisation via softmax (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Only the anti‑diagonal(s) attaining the maximum are active.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))          # the lag with maximal value
    g = np.zeros_like(a)

    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range                 # partners on the opposite side
    g[i_range] = 2.0 * a[rev]           # ∂(a_i a_rev)/∂a_i = a_rev (twice)
    return g


# ----------------------------------------------------------------------
# Simple exponentiated‑gradient optimiser (mirrored descent)
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact objective.
    The iterate stays ℓ₁‑normalised via the softmax re‑parameterisation.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # sub‑gradient of the max‑convolution term
        g = _subgradient_maxconv(a)

        # decreasing step‑size
        eta = eta0 / np.sqrt(it)

        # update in log‑space (multiplicative mirror map)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        # occasional bookkeeping
        if it % 150_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive stochastic local search (mass transfers)
# ----------------------------------------------------------------------
def _local_search(start_a: np.ndarray,
                  n: int,
                  deadline: float,
                  iters: int = 50_000,
                  step_frac: float = 0.012) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# The arcsine (Beta(½,½)) seed – the analytical starting point
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """
    Discrete approximation of the Beta(½,½) (arcsine) density on {0,…,n‑1}.
    Uses the midpoint rule for each bin and normalises to sum = 1.
    """
    # bin centre in (0,1)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid division by zero – the endpoints are never exactly 0 or 1
    w = 1.0 / np.sqrt(x * (1.0 - x))
    w_sum = w.sum()
    return w / w_sum

    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size – 4000 gives an attractive trade‑off between quality
    # and speed.  Larger n would push the bound closer to 1 but costs more.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Assemble a tiny but powerful seed set (arcsine + a few generic seeds)
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    seeds.append(_arcsine_distribution(n))

    # 2. A few cheap generic baselines (uniform, triangular, exponential)
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))                     # uniform
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    tri = 1.0 - np.abs(i - c) / c
    tri = np.maximum(tri, 0.0)
    seeds.append(tri / tri.sum())
    lam = 0.001
    expd = np.exp(-lam * i)
    seeds.append(expd / expd.sum())

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None

        if val < best_val:

    print(f"[Init] best seed objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    try:

        true_val = None

    return best_seq.tolist()
