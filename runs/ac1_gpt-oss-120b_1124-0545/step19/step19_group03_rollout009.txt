Optimise a non‑negative step function to minimise
    2·n·max(conv(a, a))/(sum(a)^2)

The routine follows a three‑stage strategy:

   (b) exponentiated‑gradient mirror descent on the exact max‑convolution,

   perturbations on the simplex.

All stages respect the overall time budget and the final sequence is

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:  # should not happen, safeguard
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors of length n)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _windows is None:
        return _uniform(n)
    a = _windows.kaiser(n, beta).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _chebwin_window(n: int, attenuation: float) -> np.ndarray:
    if _windows is None:
        return _uniform(n)
    a = _windows.chebwin(n, at=attenuation).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, NW: float) -> np.ndarray:
    if _windows is None:
        return _uniform(n)
    a = _windows.dpss(n, NW, Kmax=1, sym=True)[0].astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float, sep_frac: float) -> np.ndarray:
    """Two Gaussian bumps of equal mass, separated by `sep_frac` of the length."""
    sigma = sigma_frac * n
    center1 = (n - 1) / 2.0 - sep_frac * n / 2.0
    center2 = (n - 1) / 2.0 + sep_frac * n / 2.0
    i = np.arange(n, dtype=np.float64)
    a1 = np.exp(-((i - center1) ** 2) / (2.0 * sigma ** 2))
    a2 = np.exp(-((i - center2) ** 2) / (2.0 * sigma ** 2))
    a = a1 + a2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _two_blocks(n: int, block_frac: float) -> np.ndarray:
    """Two uniform blocks at the ends, each carrying half the total mass."""
    block_len = int(round(block_frac * n / 2.0))
    if block_len <= 0:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = 1.0
    a[-block_len:] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalize."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()

def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 5.0):
    """
    Exponentiated‑gradient (mirror descent) optimiser on the exact max‑convolution.
    Uses a sub‑gradient built from all active maximal indices.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # Work in log‑space (softmax) – positivity and ℓ₁‑norm are enforced automatically.
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # Active set: indices whose value is within a tiny relative tolerance.
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # Sub‑gradient accumulation.
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            # derivative of max is average of gradients from each active index.
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Continuation L‑BFGS optimiser on a smooth log‑sum‑exp surrogate of max.
    Operates in log‑softmax space to stay on the simplex.
    """
    if _opt is None:
        return start_vec  # SciPy not available.

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t a (smooth max derivative)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)  # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _hill_climb(seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_trials: int = 100_000,
                move_frac: float = 0.001):
    """
    Very light stochastic hill‑climbing: transfer a tiny amount of mass
    between two random coordinates and keep the move if it reduces the
    exact max‑convolution objective.  Operates on the simplex.
    Returns the best sequence seen.
    """
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    for attempt in range(max_trials):
        if time.time() > deadline:
            break

        # Choose donor (i) with non‑negligible mass.
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        # Choose receiver j != i.
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # Amount to move – proportional to current donor mass.
        delta = move_frac * a[i] * rng.random()
        # Ensure we stay non‑negative.
        a[i] -= delta
        a[j] += delta

        val = _objective_surrogate(a)
        if val < cur_val:
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = a.copy()
        else:
            # reject – revert the move.
            a[i] += delta
            a[j] -= delta

    return best_seq, best_val

"""
Optimise a non‑negative step function to minimise

    2·n·max(conv(a, a)) / (sum(a)**2)

The routine works inside a 1000 s budget and stores the best sequence in the
global variable ``height_sequence_1``.  It follows three stages:

1️⃣  **Rich seeding** – a large variety of analytic windows (Kaiser, Chebyshev,
    DPSS, double‑Gaussian, two‑block, etc.) together with many random
    Dirichlet draws.  All candidates are ℓ₁‑normalised and quickly screened
    with the exact max‑convolution objective.

2️⃣  **Multi‑start refinement** – the *k* best seeds are refined by
    (a) continuation L‑BFGS on a log‑sum‑exp surrogate of the max,
    (b) exponentiated‑gradient mirror descent on the exact max‑convolution,
    (c) a very light stochastic hill‑climbing that moves tiny amounts of mass
    on the simplex.

3️⃣  **Final polishing** – the global best undergoes a longer L‑BFGS run,
    an extended exponentiated‑gradient descent and a final low‑temperature
    hill‑climbing pass.  The final sequence is evaluated with the supplied
    ``evaluate_sequence`` function.

All stages respect the overall time budget.  The best sequence is returned
as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for sophisticated windows and L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:                     # pragma: no cover
    _opt = None

try:
    from scipy.signal import windows as _windows
except Exception:                     # pragma: no cover
    _windows = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors of length n)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _windows is None:
        return _uniform(n)
    a = _windows.kaiser(n, beta).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _chebwin_window(n: int, attenuation: float) -> np.ndarray:
    if _windows is None:
        return _uniform(n)
    a = _windows.chebwin(n, at=attenuation).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, NW: float) -> np.ndarray:
    if _windows is None:
        return _uniform(n)
    a = _windows.dpss(n, NW, Kmax=1, sym=True)[0].astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float, sep_frac: float) -> np.ndarray:
    """Two equal‑mass Gaussian bumps separated by `sep_frac` of the length."""
    sigma = sigma_frac * n
    c = (n - 1) / 2.0
    d = sep_frac * n / 2.0
    i = np.arange(n, dtype=np.float64)
    a1 = np.exp(-((i - (c - d)) ** 2) / (2.0 * sigma ** 2))
    a2 = np.exp(-((i - (c + d)) ** 2) / (2.0 * sigma ** 2))
    a = a1 + a2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _two_blocks(n: int, block_frac: float) -> np.ndarray:
    """Two uniform blocks at the extremes, each carries half the total mass."""
    block_len = int(round(block_frac * n / 2.0))
    if block_len <= 0:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = 1.0
    a[-block_len:] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()

# Optimisers
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 5.0):
    """
    Exponentiated‑gradient (mirror descent) optimiser on the exact
    max‑convolution.  Works in log‑softmax space, therefore positivity and
    ℓ₁‑normalisation are automatic.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Continuation L‑BFGS optimiser on a smooth log‑sum‑exp surrogate of the
    max.  Optimisation is performed in log‑softmax space to stay on the
    probability simplex.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _hill_climb(seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_trials: int = 100_000,
                move_frac: float = 0.001):
    """
    Very lightweight stochastic hill‑climbing on the simplex.
    Tiny mass is transferred between two random coordinates; a move is kept
    iff it lowers the exact max‑convolution surrogate.
    Returns the best sequence seen.
    """
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    for attempt in range(max_trials):
        if time.time() > deadline:
            break

        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = move_frac * a[i] * rng.random()
        a[i] -= delta
        a[j] += delta

        val = _objective_surrogate(a)
        if val < cur_val:
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function within the allotted budget.  The best sequence is stored in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Problem size – a balance between resolution and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load any previously discovered solution (must match current n).
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_tukey_window(n, 0.75))
    seeds.append(_sine_window(n, 1.0))

    # Parameter sweeps over well‑behaved families
    for gamma in np.linspace(0.2, 4.0, 20):
        seeds.append(_sine_window(n, gamma))
    for sigma_frac in np.linspace(0.04, 0.40, 20):
        seeds.append(_gaussian(n, sigma_frac))
    for p in np.linspace(0.0, 5.0, 20):
        seeds.append(_power(n, p))
    for rate in np.logspace(-5, -0.3, 16):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))
    for alpha in np.linspace(0.2, 3.0, 12):
        for beta in np.linspace(0.2, 3.0, 12):
            seeds.append(_beta_shape(n, alpha, beta))

    # Advanced window families (if SciPy is available)
    if _windows is not None:
        for beta in np.linspace(2.0, 14.0, 12):
            seeds.append(_kaiser_window(n, beta))
        for att in np.linspace(30.0, 120.0, 12):
            seeds.append(_chebwin_window(n, att))
        for NW in np.linspace(2.0, 8.0, 7):
            seeds.append(_dpss_window(n, NW))

    # Two‑bump profiles
    for sigma_frac in np.linspace(0.02, 0.15, 12):
        for sep_frac in np.linspace(0.4, 0.9, 5):
            seeds.append(_double_gaussian(n, sigma_frac, sep_frac))

    # Two‑block profiles
    for block_frac in np.linspace(0.05, 0.4, 8):
        seeds.append(_two_blocks(n, block_frac))

    # Low‑frequency random cosine curves
    for nf in (2, 3, 4, 6, 8):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random Dirichlet draws (covering extreme sparsity)
    for conc in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(20):
        seeds.append(_random_exponential(n, rng))

    # Perturbations of the previous best (if available)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(10):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Light surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.12 / max(1, (top_k - si))

        # (a) continuation L‑BFGS on smooth surrogate
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec, n, sub_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1_000.0,
                                                          2_000.0, 5_000.0,
                                                          20_000.0, 50_000.0,
                                                          100_000.0, 200_000.0,
                                                          500_000.0, 1_000_000.0],
                                           max_iter_per_beta=1500)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        # (b) exponentiated‑gradient on the exact max‑convolution
        seq_eg, val_eg = _exp_gradient_opt_multi(seq_lbfgs, n,
                                                sub_deadline,
                                                max_iters=150_000,
                                                eta0=6.0)

        # (c) very light hill‑climbing around the EG result
        seq_hc, val_hc = _hill_climb(seq_eg, n, sub_deadline, rng,
                                    max_trials=30_000, move_frac=0.0008)

        # Choose the best among the three refinements plus the raw seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_lbfgs, val_lbfgs),
            (seq_eg, val_eg),
            (seq_hc, val_hc)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing on the global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (L‑BFGS → EG → Hill‑climb) ---")
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq, n, deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1_000.0,
                                                          2_000.0, 5_000.0,
                                                          20_000.0, 50_000.0,
                                                          100_000.0, 200_000.0,
                                                          500_000.0, 1_000_000.0,
                                                          2_000_000.0],
                                           max_iter_per_beta=2500)
        seq_eg, val_eg = _exp_gradient_opt_multi(seq_lbfgs, n, deadline,
                                                max_iters=800_000,
                                                eta0=5.0)
        seq_hc, val_hc = _hill_climb(seq_eg, n, deadline, rng,
                                     max_trials=100_000, move_frac=0.0005)

        final_candidates = [
            (seq_lbfgs, _objective_surrogate(seq_lbfgs)),
            (seq_eg, val_eg),
            (seq_hc, val_hc)
        ]
        global_best_seq, global_best_val = min(final_candidates, key=lambda x: x[1])
        print(f"[Final] polished surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
