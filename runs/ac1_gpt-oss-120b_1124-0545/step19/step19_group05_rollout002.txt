"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
The objective  2·n·max(conv(a,a)) / (∑a)²  is convex (max of convex quadratics)
and scaling‐invariant.  We therefore optimise a *probability vector*
 a≥0, ∑a=1.

1. **Rich seed pool** – Analytic families (uniform, linear, power‑law,
   exponential, sinusoid, semi‑circle, polynomial tent, skewed β‑shapes)
   and many random Dirichlet / sparse draws.  A few rotations and mixtures
   are added to enlarge the convex hull.  The previous best sequence is also
   injected.

2. **Smooth continuation (log‑sum‑exp)** – For each elite seed we run
   L‑BFGS on a smoothed surrogate
   `β·log Σ exp(conv/β)` with a schedule of β up to 1e12.
   The number of L‑BFGS iterations per β is chosen adaptively from the
   remaining wall‑time.

3. **Accelerated exponentiated‑gradient (EG)** – Starting from the
   current best we perform a long EG run on the *exact* max‑convolution
   objective.  The step‑size decays as η₀ / (t+1)^α (α≈0.55) which gives
   a faster O(1/t)‑type decay while preserving the projection onto the
   simplex via the soft‑max trick.

4. **Intensive stochastic local searches** – Two complementary hill‑climbs:
   a) plain random mass‑transfer;
   b) targeted moves away from the active lag (the lag attaining the
      current max).  Both use a *cooling* step‑fraction schedule and run
   until the time budget is exhausted.

5. **Final polishing** – A short high‑β L‑BFGS run (β up to 5·10¹²) to
   clean any remaining irregularities.

All stages respect the 1000 s wall‑time budget.  The best sequence found
(ℓ₁‑normalised) is stored in the global variable ``height_sequence_1`` and
returned as a plain list.
"""

import time
import numpy as np
from scipy import optimize as _opt

# ------------------------------------------------------------
# Low‑level helpers
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# Exact objective & sub‑gradient (non‑smooth)
# ------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact value of the evaluation function (sum(a) is assumed >0)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (active lag only)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))               # lag attaining the max
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g = np.zeros_like(a)
    g[i_range] = 2.0 * a[rev]
    return g


# ------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used for continuation L‑BFGS
# ------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of the exact objective.
    Returns (obj, grad) where the gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                     # points on the simplex
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # smoothed max
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see derivation)
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through the simplex soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

                           deadline: float,
                           beta_schedule,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Continuation L‑BFGS on the smoothed surrogate.
    ``start_vec`` must be ℓ₁‑normalised.
    """
    v = np.log(start_vec + 1e-30)      # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(
                fun=_smooth_objective,
                x0=v,
                args=(n, beta),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_beta,
                         "ftol": 1e-12,
                         "disp": False}
            )
            v = res.x
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact objective
# ------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 5_000_000,
                           eta0: float = 2.0,
                           decay: float = 0.55) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
    objective.  Step‑size η_t = η₀ / (t+1)^decay.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / ((it + 1) ** decay)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 250_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[EG] iter {it:,d}  best = {best_val:.6f}")

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ------------------------------------------------------------
# Stochastic local searches
# ------------------------------------------------------------
def _random_mass_transfer(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         iters: int = 300_000,
                         init_frac: float = 0.02,
                         decay: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain random mass‑transfer hill‑climb with a decaying step‑fraction."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(1122334455)

    for k in range(iters):
        if time.time() > deadline:
            break
        # geometrically decay the transferred fraction
        step_frac = init_frac * ((iters - k) / iters) ** decay
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


def _targeted_mass_transfer(start_a: np.ndarray,
                            n: int,
                            deadline: float,
                            iters: int = 250_000,
                            init_frac: float = 0.015,
                            decay: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass *away* from the active lag.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(9988776655)

    for k in range(iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        step_frac = init_frac * ((iters - k) / iters) ** decay

        i = rng.integers(i_start, i_end + 1)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = step_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         iters: int = 400_000,
                         init_temp: float = 1e-3,
                         decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Very simple Metropolis–Hastings style annealer.
    Perturb the log‑probabilities with Gaussian noise.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    loga = np.log(a + 1e-30)
    rng = np.random.default_rng(42424242)

    for k in range(iters):
        if time.time() > deadline:
            break
        temp = init_temp * ((iters - k) / iters) ** decay
        # propose a Gaussian perturbation of the logits
        proposal = loga + rng.normal(scale=temp, size=n)
        cand = _softmax(proposal)
        val = _objective_exact(cand)
        if val < best_val or rng.random() < np.exp(-(val - best_val) / max(temp, 1e-12)):
            # accept
            loga = proposal
            a = cand
            if val < best_val:
                best_val = val
                best_a = cand.copy()
    return best_a, best_val


# ------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i)           # descending integers
    s = a.sum()
    return a / s


def _linear_increasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = i + 1.0
    s = a.sum()
    return a / s


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _semicircle_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (i - c) / c
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _poly_tent(n: int, power: float = 1.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) * (n - i)) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ------------------------------------------------------------
# Seed‑pool construction
# ------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))
    seeds.append(_linear_increasing(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_sin_shape(n))
    seeds.append(_semicircle_shape(n))
    seeds.append(_poly_tent(n, power=1.2))
    seeds.append(_poly_tent(n, power=2.0))

    # power‑law sweep – wide exponent / shift grid
    p_vals = np.linspace(0.2, 1.0, 13)
    shifts = [0.0, 5.0, 20.0, 50.0, 100.0, 250.0, 500.0]
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # random Dirichlet (wide concentration range)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse random
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(3):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine mixtures
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise convex combinations
    base = list(range(len(seeds)))
    for _ in range(150):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of a few promising candidates
    for _ in range(10):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # inject previous best if available
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final sanity – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    The routine respects the 1000 s wall‑time budget, stores the best
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0    # safety margin

    # --------------------------------------------------------
    # Problem size – n=4000 gives a good trade‑off between quality
    # and per‑iteration cost.
    # --------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # --------------------------------------------------------
    # Seed collection (including warm‑start)
    # --------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # --------------------------------------------------------
    # Evaluate seeds, keep elite pool
    # --------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    seeds.sort(key=_objective_exact)
    elite = seeds[:min(20, len(seeds))]
    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # --------------------------------------------------------
    # Stage 1 – Continuation L‑BFGS on a smooth surrogate
    # --------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        1e7, 5e7, 1e8, 5e8, 1e9, 5e9, 1e10, 5e10, 1e11, 5e11, 1e12
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break

        # allocate a modest per‑β budget; we cap it to keep total time low
        per_beta = max(1000, int(min(3000, (deadline - time.time()) * 0.04 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β≈{per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=per_beta)
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # --------------------------------------------------------
    # Stage 2 – Exponentiated‑gradient (long run from the current best)
    # --------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(10_000_000, remaining * 0.30 * 1e5))
        print("\n--- Long EG optimisation (starting from best) ---")
        seq_eg, val_eg = _exp_gradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=max_iters,
            eta0=2.2,
            decay=0.55,
        )
        print(f"[Long EG] best = {val_eg:.6f}")
        if val_eg < global_best_val:
            global_best_seq = seq_eg.copy()
            global_best_val = val_eg
            print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # --------------------------------------------------------
    # Stage 3 – Random mass‑transfer local search
    # --------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(600_000, remaining * 45))
        if iters > 0:
            print("\n--- Random mass‑transfer local search ---")
            seq_loc, val_loc = _random_mass_transfer(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                init_frac=0.025,
                decay=0.6,
            )
            print(f"[RandomLocal] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after random local search = {global_best_val:.6f}")

    # --------------------------------------------------------
    # Stage 4 – Targeted mass‑transfer local search
    # --------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters_tgt = int(min(500_000, remaining * 30))
        if iters_tgt > 0:
            print("\n--- Targeted local search (active‑lag focus) ---")
            seq_tgt, val_tgt = _targeted_mass_transfer(
                global_best_seq,
                n,
                deadline,
                iters=iters_tgt,
                init_frac=0.018,
                decay=0.55,
            )
            print(f"[TargetedLocal] best = {val_tgt:.6f}")
            if val_tgt < global_best_val:
                global_best_seq = seq_tgt.copy()
                global_best_val = val_tgt
                print(f"[Update] New best after targeted local search = {global_best_val:.6f}")

    # --------------------------------------------------------
    # Stage 5 – Simulated annealing (fine‑grained perturbations)
    # --------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters_sa = int(min(300_000, remaining * 25))
        if iters_sa > 0:
            print("\n--- Simulated annealing fine‑tuning ---")
            seq_sa, val_sa = _simulated_annealing(
                global_best_seq,
                n,
                deadline,
                iters=iters_sa,
                init_temp=2e-3,
                decay=0.65,
            )
            print(f"[SA] best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[Update] New best after SA = {global_best_val:.6f}")

    # --------------------------------------------------------
    # Stage 6 – Final high‑β polishing L‑BFGS (very high β values)
    # --------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1000, int(min(3000, remaining * 0.04 * 1e5)))
        high_beta_sched = [5e9, 1e10, 5e10, 1e11, 5e11, 1e12]
        print("\n--- Final high‑β polishing L‑BFGS ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=high_beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # --------------------------------------------------------
    # Final official evaluation via the provided security‑checked function
    # --------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
