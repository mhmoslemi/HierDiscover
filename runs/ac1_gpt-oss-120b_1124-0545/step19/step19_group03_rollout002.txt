Propose a step‑function (non‑negative sequence) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The algorithm consists of three phases:

1. **Broad seeding** – analytic families (uniform, triangular, windows,
   beta‑shapes, Dirichlet draws, low‑frequency cosine curves) are generated
   and ℓ₁‑normalised.  The surrogate objective
   ``2·n·max(conv(seq,seq))`` (identical to ``evaluate_sequence`` for a normalised
   sum) is evaluated quickly and the best *k* seeds are retained.

2. **Refinement** – each top seed is optimised with an Adam‑style
   exponentiated‑gradient method that follows the exact sub‑gradient of the
   max‑convolution (gradient derived from the *single* maximal index).  The
   optimisation works in log‑softmax space, guaranteeing non‑negativity and
   unit ℓ₁‑norm.  A short simulated‑annealing pass can escape shallow minima.

3. **Final polishing** – a longer Adam run from the best sequence found,
   followed by a very low‑temperature simulated‑annealing refinement.
   The final sequence is stored in the global variable ``height_sequence_1``
   and returned.

All work respects the supplied time budget (default 1000 s).  The code
contains many safety checks and prints progress information.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    a = np.hamming(n).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (single‑max sub‑gradient)
# ----------------------------------------------------------------------
def _adam_max_opt(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 300_000,
                  lr: float = 0.07,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient on the exact max‑convolution.
    The sub‑gradient is taken from the *single* maximal index.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # pre‑compute powers for bias correction (to avoid pow in loop)
    beta1_pow = beta1
    beta2_pow = beta2

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(conv.argmax())

        # gradient of max‑convolution at index k:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_idx = np.arange(start_i, end_i + 1, dtype=int)
        rev_idx = k - i_idx
        g = np.zeros_like(a)
        g[i_idx] = 2.0 * a[rev_idx]

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a = log_a - step
        a = _softmax(log_a)

        # update running powers
        beta1_pow *= beta1
        beta2_pow *= beta2

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing (lightweight)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.01,
                         T_end: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbations with linearly decreasing temperature.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and stores
    the best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for final evaluation & printing
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 4000 has proven a good compromise of quality vs speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if available (same length)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                # Normalise to unit ℓ₁ for consistency
                s = arr.sum()
                if s > 0:
                    prev_arr = arr / s
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation – a very broad and diverse pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_tukey_window(n, 0.75))

    # Parameter sweeps for sine, Gaussian, power, exponential
    for gamma in np.linspace(0.2, 3.0, 20):
        seeds.append(_sine_window(n, gamma))
    for sigma_frac in np.linspace(0.07, 0.35, 20):
        seeds.append(_gaussian(n, sigma_frac))
    for p in np.linspace(0.0, 5.0, 20):
        seeds.append(_power(n, p))
    for rate in np.logspace(-4.5, -0.3, 12):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Random beta‑shape parameters
    for _ in range(80):
        a = rng.uniform(0.1, 6.0)
        b = rng.uniform(0.1, 6.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws covering extreme sparsity to near‑uniform
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Exponential draws
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 6):
        for _ in range(8):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Perturbations of the previous best, if any
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(10):
            seeds.append(_perturb(prev_arr, rng, sigma=0.0025))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement with Adam optimiser
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # Preserve ~20 % of remaining time for later polishing; split the rest equally
        slice_time = remaining * (0.8 / max(1, (top_k - si)))
        sub_deadline = time.time() + slice_time

        # Adam optimisation on the exact max‑convolution
        seq_opt, val_opt = _adam_max_opt(seed_vec.copy(),
                                         n,
                                         sub_deadline,
                                         max_iters=300_000,
                                         lr=0.07)

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"[Seed {si+1}/{top_k}] new global best (Adam) = {global_best_val:.6f}")

        # A brief simulated‑annealing pass can escape shallow plateaus
        if time.time() < sub_deadline - 0.5:
            seq_sa, val_sa = _simulated_annealing(seq_opt,
                                                  n,
                                                  sub_deadline - 0.5,
                                                  rng,
                                                  max_attempts=30_000,
                                                  start_sigma=0.003,
                                                  T_start=0.01,
                                                  T_end=1e-6)
            if val_sa < global_best_val:
                global_best_seq = seq_sa
                global_best_val = val_sa
                print(f"[Seed {si+1}/{top_k}] improved after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing (long Adam + fine SA)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Final polishing ---")
        # Longer Adam run from the current best
        final_seq, final_val = _adam_max_opt(global_best_seq,
                                             n,
                                             deadline - 3.0,
                                             max_iters=1_500_000,
                                             lr=0.05)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

        # Very low‑temperature SA for final fine‑tuning
        if time.time() < deadline - 1.0:
            final_sa_seq, final_sa_val = _simulated_annealing(
                global_best_seq,
                n,
                deadline,
                rng,
                max_attempts=60_000,
                start_sigma=0.001,
                T_start=0.005,
                T_end=1e-8)
            if final_sa_val < global_best_val:
                global_best_seq = final_sa_seq
                global_best_val = final_sa_val
                print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
