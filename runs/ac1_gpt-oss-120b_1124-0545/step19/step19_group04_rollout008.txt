"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Seed generation** – a large, diverse pool of ℓ₁‑normalised vectors.
   Families include uniform, triangular, power‑law, beta‑shaped,
   Dirichlet draws, sparse random masks and low‑frequency cosine curves.
2. **Quick elite selection** – evaluate every seed with the exact objective
   (FFT convolution) and keep the best few.
3. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp surrogate of the
   max‑convolution.  The “temperature” β is increased on a schedule;
   for each β we run a generous number of L‑BFGS iterations.
4. **Adam fine‑tuning** – starting from the best L‑BFGS result, continue the
   optimisation of the smooth surrogate with the Adam optimiser (much more
   iterations, decreasing step‑size with β).
5. **Stochastic mass‑transfer local search** – hill‑climbing in the exact
   objective by moving a small fraction of mass from one entry to another.
6. **Final evaluation** – compute the official ``evaluate_sequence`` score.
"""

import time
import numpy as np
import scipy.optimize as _opt

# --------------------------------------------------------------
# basic numeric utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# --------------------------------------------------------------
# analytic families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, p>0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# --------------------------------------------------------------
# exact objective & sub‑gradient helpers
# --------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# --------------------------------------------------------------
# smooth surrogate (log‑sum‑exp) for continuation L‑BFGS
# --------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate of the max‑convolution objective."""
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2000,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={
                    "maxiter": max_iter_per_beta,
                    "disp": False,
                    "ftol": 1e-12,
                },
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# --------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# --------------------------------------------------------------
def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    iters_per_beta: int = 3000,
) -> np.ndarray:
    """Adam on the smooth surrogate, annealing the β‑parameter."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    vhat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # step‑size scaled with β (larger β ⇒ smaller steps)
        lr = 0.5 / (np.sqrt(beta) + 1.0)

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            obj, grad = _smooth_objective(v, n, beta)

            m = beta1 * m + (1 - beta1) * grad
            vhat = beta2 * vhat + (1 - beta2) * (grad ** 2)

            m_hat = m / (1 - beta1 ** t)
            vhat_hat = vhat / (1 - beta2 ** t)

            v -= lr * m_hat / (np.sqrt(vhat_hat) + eps)

    return _softmax(v)


# --------------------------------------------------------------
# Aggressive stochastic local search (mass transfers)
# --------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 200_000,
    step_frac: float = 0.006,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing by moving a tiny fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# --------------------------------------------------------------
# Seed‑pool construction
# --------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.0004))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # dense grid of power‑law exponents
    for p in np.linspace(0.30, 0.80, 25):
        seeds.append(_power_law(n, float(p)))

    # random Dirichlet families
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures
    base = np.arange(len(seeds))
    for _ in range(100):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(15):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final normalisation and sanity checks
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s time budget and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # problem size – the reference implementation used 4000
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed pool (including possible warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Quick exact evaluation of seeds, keep elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    # keep the top‑k seeds for the heavy L‑BFGS stage
    seeds.sort(key=_objective_exact)
    top_k = min(12, len(seeds))
    elite = seeds[:top_k]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS (more iterations per β)
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break
        per_beta = 1800  # a bit more than the previous 1200
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β={per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam fine‑tuning on the smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam fine‑tuning (smooth surrogate) ---")
        seq_adam = _adam_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=beta_sched,
            iters_per_beta=2500,
        )
        val_adam = _objective_exact(seq_adam)
        print(f"[Adam] objective = {val_adam:.6f}")
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Update] New global best after Adam = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – aggressive stochastic local search (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(400_000, remaining * 100))  # allocate time proportionally
        if iters > 0:
            print("\n--- Stochastic local search (mass transfers) ---")
            seq_loc, val_loc = _local_search(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.006,
            )
            print(f"[LocalSearch] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for future warm‑starts
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
