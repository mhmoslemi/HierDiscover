# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient of the max‑convolution term
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) with respect to a.
    Returns d max(conv)/da evaluated at the active lag.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))                     # active lag
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g = np.zeros_like(a)
    g[i_range] = 2.0 * a[rev]                      #  ∂/∂a_i Σ a_i a_{idx-i}
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – gradient w.r.t. logits v
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of the exact objective.
    Returns (obj, grad) where the gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                                 # ℓ₁‑normalised
    conv = _conv_fft(a, a)
    shift = np.max(conv)                            # for numeric stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # smooth max = shift + log(sum_exp)/beta
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp                          # soft‑weights over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                     # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]           # chain‑rule for max term

    # back‑prop through softmax: ∂obj/∂v = a * (grad_a - <grad_a,a>)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    ``start_vec`` must be ℓ₁‑normalised (sum = 1).
    Returns a new ℓ₁‑normalised vector.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0, 50_000_000.0, 100_000_000.0,
            500_000_000.0, 1_000_000_000.0,
        ]

    # initialise logits (add a tiny epsilon to avoid log(0))
    v = np.log(start_vec + 1e-30)

        if time.time() > deadline:
            break
        try:

                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-12, "disp": False},
            )
            v = res.x
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact objective.
    Returns (best_vector, best_value).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    eps = 1e-30

    it = 0
    # pre‑allocate log vector for speed
    log_a = np.log(a + eps)
    while it < max_iters and time.time() < deadline:
        it += 1

        eta = eta0 / np.sqrt(it)

        if it % 200_000 == 0:

            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val

def _targeted_local_search(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           iters: int = 80_000,
                           step_frac: float = 0.010) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that moves a small fraction of mass away from the indices
    that contribute to the currently active (maximal) convolution lag.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    rng = np.random.default_rng(987654321)

    for _ in range(iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        # pick a source inside the active interval
        i = rng.integers(i_start, i_end + 1)
        if a[i] <= 0.0:
            continue
        # pick any destination outside the interval (or anywhere)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

# Analytic shape families (all ℓ₁‑normalised)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    return a / a.sum()


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    return a / a.sum()


def _semicircle_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (i - c) / c
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, 1.0))
    return a / a.sum()


def _poly_tent(n: int, power: float = 1.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) * (n - i)) ** power
    return a / a.sum()


def _binary_spaced(n: int, k: int) -> np.ndarray:
    """
    k equally‑spaced spikes (non‑zero entries).  Mass is shared equally.
    """
    if k <= 0:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    positions = np.linspace(0, n - 1, k, dtype=int)
    a[positions] = 1.0 / k
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a dense random point on the simplex."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse vector with ~frac·n non‑zero entries, each equal weight."""
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sum of a few low‑frequency cosines, clipped to ≥0.
    """
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)

def _collect_seeds(n: int,

    """
    Generate a diverse collection of ℓ₁‑normalised vectors (length n).
    """
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_sin_shape(n))
    seeds.append(_semicircle_shape(n))
    seeds.append(_poly_tent(n, power=1.2))
    seeds.append(_poly_tent(n, power=2.0))

    # power‑law sweep
    for p in np.linspace(0.2, 0.9, 13):
        for sh in (0.0, 5.0, 20.0, 50.0, 100.0):
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # binary‑spaced spikes – try a few different counts
    for k in (int(sqrt(n)), int(sqrt(n) * 1.5), int(sqrt(n) * 0.75)):
        seeds.append(_binary_spaced(n, k))

    # random Dirichlet draws (dense)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):

    # random sparse draws
    for frac in (0.01, 0.02, 0.05, 0.1):
        for _ in range(3):

    # low‑frequency curves
    for nf in (2, 3, 4, 5):

    # pairwise mixtures
    base = np.arange(len(seeds))
    for _ in range(250):
        i, j = rng.choice(base, size=2, replace=False)

        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

                print(f"[Update] New best after targeted local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(2500, remaining * 0.04 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000.0, 10_000_000.0, 50_000_000.0,
                           100_000_000.0, 500_000_000.0, 1_000_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation via the provided security‑checked function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
