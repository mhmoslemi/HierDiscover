"""
Propose a step function (non‑negative heights) that minimises the bound
computed by ``evaluate_sequence``.

The algorithm proceeds in several stages while respecting a 1000 s
wall‑clock budget.

1. **Rich seed pool** – Analytic families (uniform, triangular,
   exponential, power‑law, beta, Gaussian, raised‑cosine,
   DPSS, Kaiser) with dense parameter sweeps, plus random Dirichlet,
   sparse and low‑frequency cosine seeds.  All seeds are ℓ₁‑normalised
   (sum = 1), because the objective is scale‑invariant.

2. **Exact evaluation of seeds** – Compute the exact objective
   ``2·n·max(conv(a,a))`` (the denominator cancels for ℓ₁‑normalised
   vectors) and keep the best few (elite) candidates.

3. **Continuation L‑BFGS on a log‑sum‑exp surrogate** – Each elite seed
   is refined by a sequence of L‑BFGS runs with increasing smoothness
   parameter β.  The smooth surrogate approximates the max‑convolution
   and provides a high‑quality gradient; the result is projected back
   to the simplex using a softmax.

4. **Exponentiated‑gradient mirror descent on the exact objective** –
   Starting from the best vector found so far, we run many iterations
   of mirror descent (a.k.a. exponentiated gradient).  The sub‑gradient
   of the max‑convolution is obtained analytically; the update stays
   in the simplex automatically.

5. **Simulated‑annealing mass‑transfer hill‑climb** – A stochastic
   local‑search phase that moves a small fraction of mass from one
   index to another, accepting occasional uphill moves with a linear
   cooling schedule.

6. **Final polishing** – One more short L‑BFGS pass with very large
   β values to fine‑tune the solution.

All stages stop as soon as the global deadline is reached.  The best
sequence found is stored in the global variable ``height_sequence_1``
for warm‑starts in later calls and returned as a plain ``list[float]``.
"""

import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, p>0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‐shape: a_i ∝ i^{α-1} (n‑i+1)^{β-1}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a /= a.sum()
    return a


def _gaussian(n: int, mu: float, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def _raised_cosine(n: int, width: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.maximum(0.0, np.cos(np.pi * (x - center) / width))
    a /= a.sum()
    return a


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """DPSS (Slepian) window, sym=False gives a causal shape."""
    a = _windows.dpss(n, nw, Kmax=1, sym=False)[0]
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    a = _windows.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a /= a.sum()
    return a


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    mixed /= mixed.sum()
    return mixed


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(conv(a,a)) for ℓ₁‑normalised a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv   # because sum(a)=1 ⇒ denominator =1


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution.
    Returns ∂max(conv(a,a))/∂a (a vector of length n).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    # active lag = idx
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range               # partner index
    g[i_range] = 2.0 * a[rev]         # derivative of a_i * a_rev + a_rev * a_i
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) for continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth surrogate of max‑convolution:
      f_β(a) = 2·n·(shift + log∑_k exp(β·(conv_k‑shift))/β)
    where a = softmax(v) (ℓ₁‑normalised).
    Returns (objective, gradient w.r.t. v).
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # Gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags (weights)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length 3n‑2, convolution of a_rev with w
    grad_a = 2.0 * full[n - 1:n - 1 + n]  # pick the central n entries

    # Back‑prop through softmax (v → a)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
        ]

    # Work in the log‑space (unconstrained)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        # L‑BFGS‑B with callback to monitor timeout
        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on the exact max‑conv objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)                 # sub‑gradient of max(conv)
        eta = eta0 / np.sqrt(it)                    # diminishing step
        log_a = np.log(a + 1e-30) - eta * g         # mirror step in log‑space
        a = _softmax(log_a)                         # back to simplex

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing stochastic local search
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 300_000,
    init_temp: float = 2.0,
) -> tuple[np.ndarray, float]:
    """Mass‑transfer hill‑climb with occasional uphill moves."""
    a = start_a.copy()
    cur_val = _objective_exact(a)
    best_a = a.copy()
    best_val = cur_val

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        frac = rng.random() * 0.02          # move at most 2 % of source mass
        delta = frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = _objective_exact(cand)

        if cand_val < cur_val:
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_a = cand.copy()
                best_val = cand_val
        else:
            progress = it / max_iters
            T = init_temp * max(0.05, 1.0 - progress)  # linear cooling, floor 5 %
            if rng.random() < np.exp(-(cand_val - cur_val) / T):
                a = cand
                cur_val = cand_val

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed‑pool construction – now enriched with more analytic families
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_raised_cosine(n, width=n * 0.8))

    # Exponential decay – extensive grid
    for lam in np.geomspace(5e-5, 5e-2, num=30):
        seeds.append(_exponential_decay(n, float(lam)))

    # Power‑law – dense grid covering gentle to steep decays
    for p in np.linspace(0.2, 3.0, 40):
        seeds.append(_power_law(n, float(p)))

    # Beta‑shape – coarse grid over (α,β)
    for alpha in (0.5, 1.0, 2.0, 4.0):
        for beta in (0.5, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, float(alpha), float(beta)))

    # Gaussian – several bandwidths centred in the middle
    for sigma in (n * 0.1, n * 0.2, n * 0.33):
        seeds.append(_gaussian(n, mu=n / 2, sigma=float(sigma)))

    # DPSS windows (Slepian) – narrow‑band shapes
    for nw in (2.0, 3.0, 4.0, 5.0):
        try:
            seeds.append(_dpss_window(n, float(nw)))
        except Exception:
            pass

    # Kaiser windows – a few beta values
    for bet in (2.0, 4.0, 6.0, 8.0, 12.0):
        seeds.append(_kaiser_window(n, float(bet)))

    # Random Dirichlet families (different concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse families (different fill fractions)
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # Low‑frequency cosine curves (different numbers of modes)
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixtures of existing seeds
    base = np.arange(len(seeds))
    for _ in range(80):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], float(alpha)))

    # Rotations of promising seeds (helps break symmetry)
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], int(shift)))

    # Warm‑start from previous best, if available
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        prev_norm = previous / previous.sum()
        seeds.append(prev_norm)

    # Clean‑up – enforce non‑negativity & ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimizes
    ``evaluate_sequence`` within the given time budget.

    The global variable ``height_sequence_1`` is updated with the best found
    sequence for future warm‑starts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # ------------------------------------------------------------------
    # Problem size – a slightly larger support often yields a tighter bound.
    # ------------------------------------------------------------------
    n = 5000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed pool (including warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Quick exact evaluation of seeds – keep elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    # Sort all seeds by objective, keep the top‑k for intensive refinement
    seeds.sort(key=_objective_exact)
    elite_k = min(15, len(seeds))
    elite = seeds[:elite_k]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – L‑BFGS continuation on the smooth surrogate
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS refinement #{idx + 1}/{len(elite)} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=1800,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Exponentiated‑gradient mirror descent (multiple restarts)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 9
        iters_per_restart = int(min(400_000, remaining * 0.08 / restarts))
        print("\n--- Exponentiated‑gradient restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            # Random Dirichlet start with a modest concentration
            start_vec = _random_dirichlet(n, rng.uniform(0.02, 1.5), rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.0,
            )
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Long EG optimisation starting from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(6_000_000, remaining * 0.35 * 1e5))
        max_iters = max(max_iters, 0)
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=2.5,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 200))
        if max_iters > 0:
            print("\n--- Simulated‑annealing local search ---")
            seq_sa, val_sa = _simulated_annealing(
                global_best_seq,
                n,
                deadline,
                rng,
                max_iters=max_iters,
                init_temp=2.0,
            )
            print(f"[SA] best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[Update] New best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Final high‑β L‑BFGS polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(3000, remaining * 0.07 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000.0, 1_000_000_0.0, 2_000_000_0.0, 5_000_000_0.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for warm‑starts in future invocations
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
