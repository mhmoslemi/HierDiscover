```python
import time
import math
import numpy as np

try:
    import scipy.optimize as _opt
except Exception:
    _opt = None

def _next_pow2(x: int) -> int:
    """Smallest power of 2 ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0/len(v))
    return e / s

def _apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Apply a mass transfer ``delta`` from position i to j.
    Returns new (a, conv) copies.
    """
    n = len(a)
    a_new = a.copy()
    conv_new = conv.copy()
    a_new[i] -= delta
    a_new[j] += delta

    # linear contribution of the change
    conv_new[i:i+n] += -2.0 * delta * a
    conv_new[j:j+n] +=  2.0 * delta * a

    # quadratic δ² terms
    conv_new[2*i] += delta * delta
    conv_new[2*j] += delta * delta
    conv_new[i + j] -= 2.0 * delta * delta
    return a_new, conv_new

def _rev_vector(a: np.ndarray, k: int):
    """
    Vector ``r`` of length n where r[i] = a[k-i] if 0≤k-i<n else 0.
    Used for the gradient of the active convolution index k.
    """
    n = len(a)
    rev = np.zeros_like(a)
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    if low <= high:
        idx = np.arange(low, high + 1, dtype=int)
        rev[idx] = a[k - idx]
    return rev

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000):
    """
    Optimize the smooth max‑convolution surrogate by L‑BFGS‑B.
    Operates in log‑softmax (mirror) space so the simplex constraint is
    automatically satisfied.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0,
            5_000_000.0, 20_000_000.0, 100_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max derivative)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            # full convolution of rev_a with w has length 3n‑2,
            # we only need the central n entries.
            full = _conv_fft(rev_a, w)   # length = 2n-1 + n -1 = 3n-2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.

    The routine starts from the globally stored best sequence (if any) and
    performs a fast, O(n) per‑iteration *mass‑transfer* local optimisation.
    At each step a small amount of probability mass is moved from a location
    that contributes most to the current maximal convolution value towards a
    location that contributes the least.  The convolution is updated
    incrementally, enabling millions of iterations inside the time budget.
    When the greedy phase ends a short L‑BFGS refinement on a smooth
    surrogate is attempted (if SciPy is present).  The final sequence is
    stored in ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    safety = 2.0
    deadline = start_time + budget_s - safety

    # ------------------------------------------------------------------
    # Load previous best if it exists, otherwise fall back to a simple
    # triangular guess of length 4000.
    # ------------------------------------------------------------------
    n = 4000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.shape[0] > 0:
                n = arr.shape[0]
                a = arr.copy()
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # triangular shape as a reasonable start
        i = np.arange(n, dtype=np.float64)
        c = (n - 1) / 2.0
        a = 1.0 - np.abs(i - c) / c
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure normalisation to ℓ₁‑norm = 1
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a_sum

    # Initial full convolution (exact, via FFT)
    conv = _conv_fft(a, a)
    best_val = float(2.0 * n * conv.max())
    best_a = a.copy()
    best_conv = conv.copy()
    print(f"[Init] n={n}, initial surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Greedy mass‑transfer loop
    # ------------------------------------------------------------------
    attempts = 0
    successes = 0
    no_improve = 0
    while time.time() < deadline:
        # fetch the current maximal convolution index
        k_max = int(np.argmax(conv))

        # gradient‑like vector for that index
        rev = _rev_vector(a, k_max)          # rev[i] = a[k_max-i] when valid

        # contribution of each coordinate to the active peak
        contrib = a * rev
        i_max = int(np.argmax(contrib))

        # choose a destination with the smallest rev entry (ideally zero)
        min_rev = rev.min()
        # indices with rev very close to the minimum (tolerance for floating noise)
        cand_j = np.where(np.isclose(rev, min_rev, atol=1e-16))[0]
        # ensure i_max not selected; if it is, drop it from the candidates
        if cand_j.size == 0 or (cand_j == i_max).all():
            # fallback to the global minimum (could be i_max itself)
            cand_j = np.arange(n)
        # remove i_max if present
        cand_j = cand_j[cand_j != i_max]
        if cand_j.size == 0:
            # as a last resort, move to a random index
            j_min = int(rng.integers(0, n))
        else:
            j_min = int(rng.choice(cand_j))

        # step size – a modest fraction of the source weight, decaying with time
        remaining = deadline - time.time()
        total_budget = budget_s - safety
        ratio = max(0.0, remaining / total_budget)
        # start with up to 50 % of the source, shrink to ≈10 % as we approach the deadline
        frac = 0.1 + 0.4 * ratio
        delta = min(a[i_max], frac * a[i_max])
        if delta <= 0.0:
            # nothing to move – abort this iteration
            attempts += 1
            continue

        # apply the move (using the current a for the linear part)
        a_new, conv_new = _apply_move(a, conv, i_max, j_min, delta)

        new_val = float(2.0 * n * conv_new.max())

        # Accept only if we obtain a strict improvement, otherwise use a tiny
        # probabilistic acceptance to escape possible plateaus.
        if new_val < best_val - 1e-12:
            a = a_new
            conv = conv_new
            best_val = new_val
            best_a = a.copy()
            best_conv = conv.copy()
            successes += 1
            no_improve = 0
        else:

            temp = 1e-5 * ratio
            if rng.random() < math.exp(-(new_val - best_val) / (temp + 1e-12)):

                    best_conv = conv.copy()
            no_improve = 0

# -*- coding: utf-8 -*-
"""
Fast greedy refinement + optional L‑BFGS polish for the step‑function problem.

The routine starts from the globally stored best sequence (if any) – otherwise it
creates a simple triangular guess of length 4000 – and repeatedly moves a small
amount of mass from a position that contributes most to the current maximal
convolution value to a position that contributes the least.  The convolution
vector is updated incrementally in O(n) time, allowing millions of iterations
inside the 1000 s budget.  When the greedy phase finishes a short L‑BFGS
refinement on a smooth log‑sum‑exp surrogate is performed (if SciPy is
available).  The final sequence is stored in the global variable
``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used only for the final L‑BFGS polish)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (O(n log n)).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Helper functions for the greedy mass‑transfer optimiser
# ----------------------------------------------------------------------
def _rev_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Return a vector r of length n where r[i] = a[k‑i] if 0 ≤ k‑i < n,
    otherwise 0.  Used as the gradient of the active convolution index.
    """
    n = len(a)
    r = np.zeros_like(a)
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    if low <= high:
        idx = np.arange(low, high + 1, dtype=int)
        r[idx] = a[k - idx]
    return r


def _apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Transfer a mass ``delta`` from coordinate i to coordinate j.
    Returns copies of the updated (a, conv) pair.
    The convolution update is performed in O(n) using the exact linearised
    formula:
        conv' = conv - 2·δ·(a shifted by i) + 2·δ·(a shifted by j)
                + δ²·e_{2i} + δ²·e_{2j} – 2·δ²·e_{i+j}
    """
    n = len(a)
    a_new = a.copy()
    conv_new = conv.copy()
    a_new[i] -= delta
    a_new[j] += delta

    # linear part (uses the *old* a)
    conv_new[i:i + n] += -2.0 * delta * a
    conv_new[j:j + n] +=  2.0 * delta * a

    # quadratic δ² terms
    conv_new[2 * i] += delta * delta
    conv_new[2 * j] += delta * delta
    conv_new[i + j] -= 2.0 * delta * delta
    return a_new, conv_new


# ----------------------------------------------------------------------
# L‑BFGS optimisation of a smooth log‑sum‑exp surrogate (optional)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000):
    """
    Minimise a smooth approximation of the max‑convolution objective
    with L‑BFGS‑B in log‑soft‑max space (the simplex constraint is built‑in).
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0,
            5_000_000.0, 20_000_000.0, 100_000_000.0
        ]

    # work in log‑soft‑max (mirror‑descent) space
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule for the log‑soft‑max parametrisation
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the time budget and stores the best found
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    safety = 2.0  # reserve a couple of seconds for final evaluation
    deadline = start_time + budget_s - safety

    # --------------------------------------------------------------
    # 1️⃣ Load previous solution (if any) or start from a triangular guess
    # --------------------------------------------------------------
    n = 4000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.shape[0] > 0:
                n = arr.shape[0]
                a = arr.copy()
            else:
                a = None
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # triangular shape – a simple, reasonably good analytic start
        i = np.arange(n, dtype=np.float64)
        c = (n - 1) / 2.0
        a = 1.0 - np.abs(i - c) / c
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

    # ℓ₁‑normalise (the objective is scale‑invariant)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    # --------------------------------------------------------------
    # 2️⃣ Initial exact convolution (via FFT) and objective
    # --------------------------------------------------------------
    conv = _conv_fft(a, a)
    best_val = float(2.0 * n * conv.max())
    best_a = a.copy()
    best_conv = conv.copy()
    print(f"[Init] n={n}, initial surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Greedy mass‑transfer optimisation (millions of cheap iterations)
    # --------------------------------------------------------------
    attempts = 0
    successes = 0
    no_improve = 0

    while time.time() < deadline:
        # active shift where the convolution attains its maximum
        k_max = int(np.argmax(conv))

        # gradient‑like vector for this shift
        rev = _rev_vector(a, k_max)          # rev[i] = a[k_max-i] (or 0)

        # contribution of each coordinate to the active peak
        contrib = a * rev
        i_max = int(np.argmax(contrib))

        # destination – pick a coordinate with the *smallest* rev entry
        min_rev = rev.min()
        cand_j = np.where(np.isclose(rev, min_rev, atol=1e-16))[0]
        # ensure i_max is not selected as destination
        cand_j = cand_j[cand_j != i_max]
        if cand_j.size == 0:
            # fallback: any index different from i_max
            j_min = int(rng.integers(0, n - 1))
            if j_min >= i_max:
                j_min += 1
        else:
            j_min = int(rng.choice(cand_j))

        # step size – a decaying fraction of the source mass
        remaining = deadline - time.time()
        total_budget = budget_s - safety
        ratio = max(0.0, remaining / total_budget)   # 1 → early, 0 → late
        frac = 0.1 + 0.4 * ratio                      # between 0.1 and 0.5
        delta = min(a[i_max], frac * a[i_max])
        if delta <= 0.0:
            attempts += 1
            continue

        # propose the move
        a_new, conv_new = _apply_move(a, conv, i_max, j_min, delta)
        new_val = float(2.0 * n * conv_new.max())

        # accept only strict improvements, otherwise very occasional
        # simulated‑annealing acceptance to escape shallow plateaus
        if new_val < best_val - 1e-12:
            a = a_new
            conv = conv_new
            best_val = new_val
            best_a = a.copy()
            best_conv = conv.copy()
            successes += 1
            no_improve = 0
        else:
            temp = 1e-5 * ratio
            if rng.random() < math.exp(-(new_val - best_val) / (temp + 1e-12)):
                a = a_new
                conv = conv_new
            no_improve += 1

        attempts += 1

        # occasional full recomputation to keep numerical error in check
        if attempts % 5000 == 0:
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            conv = _conv_fft(a, a)

        # if we have been stuck for a long time, inject a small random shake
        if no_improve >= 100_000:
            sigma = 0.01 * np.mean(a)
            pert = a + sigma * rng.normal(size=n)
            pert = np.clip(pert, 0.0, None)
            s = pert.sum()
            if s > 0.0:
                a = pert / s
                conv = _conv_fft(a, a)
                cur_val = float(2.0 * n * conv.max())
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    best_conv = conv.copy()
            no_improve = 0

        # light progress report
        if attempts % 20000 == 0:
            elapsed = time.time() - start_time
            print(f"[{elapsed:.1f}s] attempts={attempts}, best surrogate={best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Optional L‑BFGS polishing (if SciPy is available and time permits)
    # --------------------------------------------------------------
    if _opt is not None and time.time() < deadline:
        print("\n--- L‑BFGS smooth refinement ---")
        # leave a tiny margin for the final call to evaluate_sequence
        lbfgs_deadline = deadline - 0.5
        refined = _lbfgs_smooth_optimize(best_a, n, lbfgs_deadline,
                                         beta_schedule=None,
                                         max_iter_per_beta=2500)
        refined_val = float(2.0 * n * _conv_fft(refined, refined).max())
        if refined_val < best_val:
            best_a = refined
            best_val = refined_val
            print(f"[L‑BFGS] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣ Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_a.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next run
    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
