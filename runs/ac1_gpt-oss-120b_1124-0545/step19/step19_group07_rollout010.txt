"""
Propose a step‑function that (hopefully) improves the current best bound.
The strategy is deliberately simple and robust:

1.  Start from the uniform distribution, which is the known ℓ₁‑optimal
    shape for the exact objective (its value is 2.0 for any length).

2.  Perform a fast stochastic hill‑climbing / simulated‑annealing loop:
    repeatedly move a small fraction of mass from one index to another,
    always re‑normalising the vector to keep the ℓ₁‑norm equal to 1.
    After each move we evaluate the *exact* objective with the provided
    ``evaluate_sequence`` function.  Improvements are kept.

3.  Because the evaluation routine is cheap (FFT‑based convolution) we
    can afford many millions of cheap trials inside the 1000 s budget.

4.  The best sequence found is stored in the global variable
    ``height_sequence_1`` (as required by the surrounding framework) and
    returned as a plain list.

The algorithm respects the non‑negativity requirement, never produces
NaNs or infinities, and stops well before the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: exact objective (mirrors ``evaluate_sequence`` but works on
# ``np.ndarray`` for speed).  This is *only* used for quick comparisons
# inside the optimiser; the final score is always obtained via
# ``evaluate_sequence``.
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact evaluation of the objective for a non‑negative vector."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so we never overshoot the wall‑time limit
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start – n={n}, time budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Initial candidate: the uniform distribution (theoretical ℓ₁‑optimum)
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = a.copy()
    best_val = evaluate_sequence(best_seq.tolist())
    print(f"[Init] uniform objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Helper to occasionally re‑normalise (numerical drift)
    # ------------------------------------------------------------------
    def renorm(v: np.ndarray) -> np.ndarray:
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    # ------------------------------------------------------------------
    # Stochastic hill‑climbing: move a fraction of mass from i → j.
    # ---------------------------------------------------------------
    step_frac = 0.015          # fraction of mass moved each trial
    max_iters = 30_000_000     # large upper bound – actual stop is time‑driven
    it = 0
    last_report = start_time

    while time.time() < deadline and it < max_iters:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue

        # transfer a small chunk of mass
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # keep non‑negativity (tiny negatives can appear due to rounding)
        if cand[i] < 0.0:
            continue

        # re‑normalise to avoid drift (does not change the objective
        # analytically, but keeps the vector well‑behaved numerically)
        cand = renorm(cand)

        # fast exact evaluation (identical to ``evaluate_sequence``)
        cur_val = _objective_raw(cand)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = cand.copy()
            a = cand  # continue hill‑climbing from the improved point
            now = time.time()
            print(f"[Iter {it:>8}] new best = {best_val:.6f} (t={now - start_time:.1f}s)")
            # keep the report interval reasonable
            last_report = now

        it += 1

        # Occasionally inject a completely new random direction to avoid
        # getting trapped in a tiny local basin.
        if it % 1_000_000 == 0:
            # replace a fraction of the entries with a fresh Dirichlet draw
            mask = rng.random(n) < 0.03           # ~3 % of coordinates
            if mask.any():
                new_part = rng.dirichlet(np.full(mask.sum(), 0.5))
                cand = a.copy()
                cand[mask] = new_part
                cand = renorm(cand)
                cur_val = _objective_raw(cand)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cand.copy()
                    a = cand
                    print(f"[Refresh {it}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_val = evaluate_sequence(best_seq.tolist())
    print(
        f"\n=== Finished ===\nIterations performed: {it}\n"
        f"Best surrogate ≈ {best_val:.6f}, evaluate_sequence = {final_val:.6f}"
    )

    # store in the global namespace as required by the framework
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
