"""
Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the full linear convolution.  The algorithm works in three
phases:

1. **Broad seeding** – generate a diverse pool of ℓ1‑normalised vectors
   (analytic families, Dirichlet draws, low‑frequency cosine curves,
   perturbed previous best).

2. **Smooth optimisation** – for each promising seed run a continuation
   L‑BFGS optimiser on the log‑sum‑exp surrogate of the maximum
   convolution (the surrogate is exact when the smoothing parameter β→∞).
   A cheap exponentiated‑gradient (mirror descent) step follows.

3. **Local refinement** – a simple hill‑climbing move that transfers a
   tiny amount of mass between two coordinates and accepts the move if the
   surrogate improves.  Random convex mixtures of the best vectors are also
   tried.

All vectors are kept on the probability simplex (sum = 1); the scaling cancels
out of the objective, so we never have to worry about the clipping that
`evaluate_sequence` performs.  The best vector found is stored globally as
`height_sequence_1` and returned as a plain Python list.
"""

import time
import math
import numpy as np

# optional SciPy – used only for the L‑BFGS optimiser
try:
    import scipy.optimize as _opt
except Exception:   # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Basic numerical helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2 * n * max(conv(a, a)).  Assumes ∑ a = 1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(a: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip, renormalise."""
    noisy = a + sigma * rng.normal(size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (both already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_seq, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space, guaranteeing the simplex constraint.
    """
    if _opt is None:                     # pragma: no cover
        # SciPy not available – return the unchanged vector.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0]

    # work in log‑softmax space so the simplex is automatic
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            # log‑sum‑exp approximation of the max
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (via chain rule through softmax)
            w = exp_beta / sum_exp                     # weights for each entry of conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                # d obj / d v
            return obj, grad_v

        # allow early exit if time runs out
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple hill‑climbing local search
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                cur_val: float,
                deadline: float,
                steps: int = 5000,
                step_scale: float = 0.01):
    """
    Transfer a tiny amount of mass between two random coordinates.
    Accept the move if the surrogate improves.
    Returns (best_seq, best_surrogate).
    """
    a = seq.copy()
    n = a.shape[0]
    best_seq = a
    best = cur_val
    rng = np.random.default_rng()

    for _ in range(steps):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a proportion of a[i] to a[j]
        delta = step_scale * rng.random() * a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalise (numerical safety)
        a_new /= a_new.sum()
        new_val = _objective_surrogate(a_new)
        if new_val < best:
            a = a_new
            best = new_val
            best_seq = a_new
            # slowly increase step size when we are improving
            step_scale = min(step_scale * 1.1, 0.2)
        else:
            step_scale = max(step_scale * 0.95, 1e-5)

    return best_seq, best

        seq_mix, val_mix = _hill_climb(mixed,
                                       _objective_surrogate(mixed),
                                       deadline,
                                       steps=2000,
                                       step_scale=0.008)
        if val_mix < best_val:
            best_val, best_seq = val_mix, seq_mix.copy()
            print(f"[Mix] new best surrogate = {best_val:.6f}")
            top_vectors.append(best_seq.copy())
            top_vectors.sort(key=lambda v: _objective_surrogate(v))
            top_vectors = top_vectors[:15]

    # ------------------------------------------------------------------
    # 5. Final long‑run L‑BFGS with a richer β‑schedule (if time permits).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long L‑BFGS run ---")
        seq_final = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                           beta_schedule=[1, 5, 20, 80, 200, 500,
                                                          1000, 2000, 5000,
                                                          10000, 20000, 50000,
                                                          100000, 200000, 500000,
                                                          1000000],
                                           max_iter_per_beta=3000)
        val_final = _objective_surrogate(seq_final)
        print(f"  final surrogate = {val_final:.6f}")
        if val_final < best_val:
            best_val, best_seq = val_final, seq_final.copy()

    # ------------------------------------------------------------------
    # 6. Final official evaluation (uses the provided `evaluate_sequence`).
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:         # pragma: no cover
        print("Warning: evaluate_sequence raised", exc)

    # Store the best sequence globally for the next run.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
