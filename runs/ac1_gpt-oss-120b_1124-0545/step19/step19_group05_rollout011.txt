def _simulated_annealing(start_a, n, deadline, max_iters=1_000_000, init_temp=1e-5, final_temp=1e-8):
    a = start_a.copy()
    cur_val = _objective_exact(a)
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng(12345)
    # Temperature schedule linear in log-space:
    temp = init_temp
    for it in range(max_iters):
        if time.time() > deadline: break
        # linear cooling: temp = init_temp * (final_temp / init_temp) ** (it / max_iters)
        # compute fractional progress:
        frac = it / max_iters
        temp = init_temp * ((final_temp / init_temp) ** frac)
        # propose move:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0: continue
        # delta random fraction between min_frac and max_frac times a[i]
        # choose delta = a[i] * rng.uniform(0.001, 0.02) maybe adjust out of bounds
        delta = a[i] * rng.uniform(0.001, 0.02)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = _objective_exact(cand)
        delta_val = cand_val - cur_val
        if delta_val < 0 or np.exp(-delta_val / temp) > rng.random():
            a = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        # Optional: every 100k steps compute active lag and bias moves...
    return best_a, best_val

Define:

```
def _simulated_annealing(start_a: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    a = start_a.copy()
    cur_val = _objective_exact(a)
    conv_cur = _conv_fft(a, a)
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng(987654321)
    it = 0
    # parameters:
    init_temp = 1e-3
    final_temp = 1e-8
    max_iters = 2_000_000   # or while time remains, break.
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute temperature:
        frac = it / max_iters
        temp = init_temp * ((final_temp / init_temp) ** frac)
        # decide move type:
        if rng.random() < 0.6:
            # targeted move
            idx = int(np.argmax(conv_cur))
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = rng.integers(i_start, i_end + 1)
        else:
            i = rng.integers(0, n)
        if a[i] <= 0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # delta fraction
        frac_move = rng.uniform(0.005, 0.03) # 0.5% to 3%
        delta = a[i] * frac_move
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = _objective_exact(cand)
        delta_val = cand_val - cur_val
        if delta_val < 0 or np.exp(-delta_val / max(temp, 1e-12)) > rng.random():
            # accept
            a = cand
            cur_val = cand_val
            conv_cur = _conv_fft(a, a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
            # optional: print progress occasionally
            if it % 50000 == 0:
                print(f"[SA] iter {it}, cur_val={cur_val:.6f}, best_val={best_val:.6f}")
    return best_a, best_val
```

"""
Simple yet powerful simulated‑annealing optimiser for the step‑function
objective ``evaluate_sequence``.

The idea is to start from the best known sequence (or a random Dirichlet
seed) and repeatedly transfer a small amount of mass from one index to
another.  The convolution is updated incrementally in O(n) using the
exact algebraic formula, so millions of moves fit comfortably into the
1000 s budget.  An exponential temperature schedule enables occasional
up‑hill moves, helping the search escape shallow local minima.
After the annealing phase a brief L‑BFGS polishing (smooth surrogate)
is performed.

The final sequence is stored in the global ``height_sequence_1`` variable
and returned as a plain list.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum ≡ 1)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)².  Assumes ∑a≈1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    # sum is normalised to 1 → denominator = 1
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Smooth surrogate used by the polishing L‑BFGS stage
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth approximation of the exact objective. Returns (obj, grad)."""
    a = _softmax(v)                     # ℓ₁‑normalised (sum = 1)
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 800) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1e3, 1e4, 1e5, 1e6, 1e7,
            1e8, 1e9, 1e10,
        ]

    v = np.log(start_vec + 1e-30)   # logits of the simplex point

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(
                fun=_smooth_objective,
                x0=v,
                args=(n, float(beta)),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": max_iter_per_beta,
                    "ftol": 1e-12,
                    "disp": False,
                },
            )
            v = res.x
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑Annealing optimiser (incremental convolution updates)
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         init_temp: float = 0.01,
                         final_temp: float = 1e-9,
                         p_targeted: float = 0.6,
                         min_frac: float = 0.001,
                         max_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    SA on the exact max‑convolution objective.
    - Moves mass from a source index to a destination index.
    - The convolution is updated incrementally (O(n)).
    - Temperature decays exponentially with elapsed time.
    - Returns the best sequence seen.
    """
    a = start_a.copy()
    conv = _conv_fft(a, a)                # current full convolution
    cur_val = 2.0 * n * np.max(conv)      # current objective (sum ≈ 1)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(987654321)

    start_time = time.time()
    total_alloc = max(0.0, deadline - start_time)

    it = 0
    # Loop until time runs out
    while time.time() < deadline:
        it += 1
        # temperature based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / total_alloc if total_alloc > 0 else 1.0
        temp = init_temp * ((final_temp / init_temp) ** frac)

        # ---- pick source -------------------------------------------------
        if rng.random() < p_targeted:
            # target the active lag interval
            idx = int(np.argmax(conv))
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = rng.integers(i_start, i_end + 1)
        else:
            i = rng.integers(0, n)

        if a[i] <= 0.0:
            continue

        # ---- pick destination --------------------------------------------
        j = rng.integers(0, n)
        if i == j:
            continue

        # ---- amount to move ------------------------------------------------
        frac_move = rng.random() * (max_frac - min_frac) + min_frac
        delta = a[i] * frac_move
        if delta <= 0.0:
            continue

        # ---- incremental convolution update (candidate) -------------------
        conv_cand = conv.copy()
        # slice updates using the *old* a (before modification)
        conv_cand[i:i + n] -= 2.0 * delta * a
        conv_cand[j:j + n] += 2.0 * delta * a
        conv_cand[i + i] += delta * delta
        conv_cand[j + j] += delta * delta
        conv_cand[i + j] -= 2.0 * delta * delta

        max_conv_cand = np.max(conv_cand)
        cand_val = 2.0 * n * max_conv_cand

        # ---- Metropolis acceptance -----------------------------------------
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            # uphill move: accept with probability exp(-Δ/ T)
            prob = math.exp(-(cand_val - cur_val) / max(temp, 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move
            a[i] -= delta
            a[j] += delta
            conv = conv_cand
            cur_val = cand_val

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

        # occasional progress report
        if it % 250_000 == 0:
            print(f"[SA] iter {it:,}, cur = {cur_val:.6f}, best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size and starting point
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) > 0:
        n = len(prev)
        a0 = np.asarray(prev, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        print(f"[Init] using previous sequence, n={n}")
    else:
        n = 4000
        a0 = rng.dirichlet(np.ones(n)).astype(np.float64)   # random simplex point
        print(f"[Init] random start, n={n}")

    # quick sanity check
    cur_val = _objective_exact(a0)
    print(f"[Init] objective = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # 1) Simulated‑Annealing phase (majority of the time budget)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        print("\n--- Simulated Annealing ---")
        sa_seq, sa_val = _simulated_annealing(
            a0,
            n,
            deadline,
            init_temp=0.02,
            final_temp=1e-9,
            p_targeted=0.65,
            min_frac=0.001,
            max_frac=0.025,
        )
        print(f"[SA] best value = {sa_val:.6f}")
    else:
        sa_seq, sa_val = a0, cur_val

    # ------------------------------------------------------------------
    # 2) Short L‑BFGS polishing (smooth surrogate, very high β)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (L‑BFGS) ---")
        seq_final = _lbfgs_smooth_optimize(

        print(f"[Polish] objective = {val_final:.6f}")

        if val_final < sa_val:
            best_seq = seq_final
            best_val = val_final
        else:
            best_seq = sa_seq
            best_val = sa_val
    else:
        best_seq = sa_seq
        best_val = sa_val

    # ------------------------------------------------------------------
    # 3) Official evaluation via the provided security‑checked function

    try:

            f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
