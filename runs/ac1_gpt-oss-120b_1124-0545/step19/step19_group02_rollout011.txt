"""
Propose a candidate step‑function (list of non‑negative heights) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The algorithm works in three phases:

1) **Diverse seed generation** – analytic windows (Hamming, Kaiser, Gaussian,
   exponential, logistic, etc.) plus a new *double‑exponential* family and many
   random Dirichlet draws.  All seeds are ℓ₁‑normalised (sum = 1).

2) **Local refinement** – the best seed (and a few perturbed copies) are
   refined with a fast *Adam* mirror‑descent on the exact max‑convolution.
   The iterate lives in log‑softmax space, guaranteeing feasibility.
   The optimizer runs as long as time permits (tens of thousands of
   iterations).

3) **Final polishing** – a long exponentiated‑gradient (EG) run (the same
   sub‑gradient used by the original code) finishes the optimisation and
   pushes the surrogate value a little further.

The final sequence is stored in the global variable ``height_sequence_1`` and
returned as a plain Python list.  The whole routine respects a 1000 s wall‑time
budget.
"""

import time

# --------------------------------------------------------------
# Basic numeric helpers
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# --------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a vector with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exp(n: int, gamma_left: float, gamma_right: float) -> np.ndarray:
    """Sum of a left‑hand and a right‑hand exponential decay."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    left = np.exp(-gamma_left * i)
    right = np.exp(-gamma_right * (1.0 - i))
    a = left + right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                 # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

# Optimisers

def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam on the exact max‑convolution.
    The variables are maintained in log‑softmax space (log_a).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]          # 2·a_{k-i}

        # gradient w.r.t. log‑softmax variables
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # back to the simplex
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient (mirror‑descent) on the exact max‑conv.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    for gl in (0.5, 1.0, 2.0, 4.0, 8.0):
        for gr in (0.5, 1.0, 2.0, 4.0, 8.0):
            seeds.append(_double_exp(n, gl, gr))

    # beta‑shape grid
    alphas = np.linspace(0.2, 5.0, 6)
    betas = np.linspace(0.2, 5.0, 6)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 6):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – various concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # Perturb the previous best (if any)
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n:
                seeds.append(prev_arr / prev_arr.sum())
                for _ in range(3):
                    seeds.append(_perturb(prev_arr, rng, sigma=0.003))
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # Keep the best seed as a starting point
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement with Adam (fast, many iterations)
    # ------------------------------------------------------------------
    # Allocate roughly half the remaining budget to Adam optimisation
    remaining_time = deadline - time.time()
    adam_deadline = time.time() + remaining_time * 0.45

    print("\n--- Adam optimisation (first pass) ---")
    seq_adam, val_adam = _adam_optimize(best_seq,
                                        n,
                                        adam_deadline,
                                        max_iters=300_000,
                                        lr=0.7)
    if val_adam < best_val:
        best_seq, best_val = seq_adam, val_adam
        print(f"[Adam] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Small random perturbations + short Adam runs (to escape local minima)
    # ------------------------------------------------------------------
    perturbed_candidates = []
    for i in range(4):
        if time.time() > deadline:
            break
        pert = _perturb(best_seq, rng, sigma=0.0015)
        pert_dead = time.time() + (deadline - time.time()) * 0.2 / (4 - i)
        seq_p, val_p = _adam_optimize(pert,
                                      n,
                                      pert_dead,
                                      max_iters=120_000,
                                      lr=0.7)
        perturbed_candidates.append((seq_p, val_p))
        if val_p < best_val:
            best_seq, best_val = seq_p, val_p
            print(f"[Perturb #{i+1}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long EG polishing (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=6.0)
        if val_final < best_val:
            best_seq, best_val = seq_final, val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional final L‑BFGS smoothing (if time permits)
    # ------------------------------------------------------------------
    try:
        import scipy.optimize as _opt
    except Exception:  # pragma: no cover
        _opt = None

    if _opt is not None and time.time() < deadline:
        # very short L‑BFGS run just to polish any remaining roughness
        print("\n--- Final L‑BFGS polishing ---")
        def _lbfgs_smooth_opt(start_vec: np.ndarray,
                               n: int,
                               deadline_local: float) -> np.ndarray:
            v = np.log(start_vec + 1e-30)
            beta_sched = [1.0, 5.0, 20.0, 80.0,
                          200.0, 500.0, 1000.0,
                          2000.0, 5000.0]
            for beta in beta_sched:
                if time.time() > deadline_local:
                    break
                beta_val = float(beta)

                def fun(v_local):
                    a = _softmax(v_local)
                    conv = _conv_fft(a, a)
                    max_c = np.max(conv)
                    shift = max_c
                    expb = np.exp(beta_val * (conv - shift))
                    sumexp = expb.sum()
                    obj = 2.0 * n * (shift + np.log(sumexp) / beta_val)

                    w = expb / sumexp
                    rev_a = a[::-1]
                    full = _conv_fft(rev_a, w)
                    grad_a = 2.0 * full[n - 1:n - 1 + n]
                    dot = np.dot(grad_a, a)
                    grad_v = a * (grad_a - dot)
                    return obj, grad_v

                def callback(_):
                    if time.time() > deadline_local:
                        raise KeyboardInterrupt()

                try:
                    res = _opt.minimize(fun=fun,
                                        x0=v,
                                        method='L-BFGS-B',
                                        jac=True,
                                        callback=callback,
                                        options={'maxiter': 4000,
                                                 'disp': False,
                                                 'ftol': 1e-12})
                    v = res.x
                except KeyboardInterrupt:
                    break
                except Exception as e:  # pragma: no cover
                    print(f"[L‑BFGS] exception {e}")
                    break
            return _softmax(v)

        seq_lbfgs = _lbfgs_smooth_opt(best_seq, n, deadline)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_seq, best_val = seq_lbfgs, val_lbfgs
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")
    else:
        print("\n--- L‑BFGS not available or time exhausted ---")

    # ------------------------------------------------------------------
    # 7. Official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
