"""
step‑function optimiser

Goal
----
Find a non‑negative list of heights ``a`` (length *n*) minimizing

    2·n·max(conv(a,a)) / (sum(a)**2)

as evaluated by the supplied ``evaluate_sequence`` function.

Method
------
1. Build a diverse pool of *seed* vectors (uniform, triangular,
   exponential‑decay, power‑law, beta, Dirichlet, etc.).  Each seed is
   ℓ₁‑normalised (∑a = 1) and non‑negative.
2. Run a continuation L‑BFGS optimiser on a smooth max‑approximation
   (log‑sum‑exp) for every seed.  The optimiser works on the logits
   (softmax‑parametrisation) and gradually raises the smoothing factor
   ``β`` to focus on the true peak.
3. After each L‑BFGS run try two deterministic orderings of the height
   multiset – (i) sorted ascending, (ii) a “zig‑zag’’ ordering where
   large and small entries alternate.  Keep the best ordering.
4. Starting from the globally best vector, perform a long exponentiated‑
   gradient (mirror‑descent) run directly on the exact objective.
5. Refine further with a stochastic mass‑transfer local search.
6. One final cheap L‑BFGS polishing stage.
7. Return the best vector found (list of ``float``).  The vector is also
   stored globally as ``height_sequence_1`` for warm‑starts.

All heavy work respects the 1000 s budget; intermediate progress is
printed.  The code uses only NumPy, SciPy and the provided
``evaluate_sequence`` – no file or network access.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → ℓ₁‑simplex (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# families of ℓ₁‑normalised shapes (all return a 1‑norm‑1 vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# ordering utilities
# ----------------------------------------------------------------------
def _ascending_order(a: np.ndarray) -> np.ndarray:
    """Return a copy sorted increasingly (small → large)."""
    return np.sort(a)


def _zigzag_order(a: np.ndarray) -> np.ndarray:
    """
    Zig‑zag ordering: largest, smallest, 2nd‑largest, 2nd‑smallest, …
    This spreads extreme values across the sequence.
    """
    desc = np.sort(a)[::-1]          # descending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    toggle = True
    for v in desc:
        if toggle:
            out[lo] = v
            lo += 1
        else:
            out[hi] = v
            hi -= 1
        toggle = not toggle
    return out


# ----------------------------------------------------------------------
# exact objective & sub‑gradient (for EG)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (∑a is usually 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth approx. of max‑conv objective (value, gradient w.r.t. logits)."""
    a = _softmax(v)                     # ℓ₁‑normalised vector
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # numeric stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # weights over all lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # back‑prop through softmax (logit → a)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 1200) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 800.0,
            3200.0, 12800.0, 51200.0, 200_000.0,
            1_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        # interrupt callback
        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 800_000,
                           eta0: float = 1.5) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# aggressive stochastic local search (mass transfers)
# ----------------------------------------------------------------------
def _local_search(start_a: np.ndarray,
                  n: int,
                  deadline: float,
                  iters: int = 300_000,
                  step_frac: float = 0.006) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# seed construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.0004))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # power‑law grid (including steeper decays)
    for p in np.linspace(0.15, 1.20, 25):
        seeds.append(_power_law(n, float(p)))

    # exponential decays with a broader range
    for lam in (0.0002, 0.0006, 0.0012):
        seeds.append(_exponential_decay(n, lam))

    # diverse Dirichlet draws
    for conc in (0.005, 0.01, 0.02, 0.05, 0.2):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # beta shapes with varied parameters
    for a, b in [(0.5, 5.0), (5.0, 0.5), (0.8, 0.8), (3.0, 1.0), (1.0, 3.0)]:
        seeds.append(_beta_shape(n, a, b))

    # rotated versions of a few promising seeds
    rot_candidates = seeds[:5]
    for base in rot_candidates:
        for shift in (n // 4, n // 2, 3 * n // 4):
            seeds.append(np.roll(base, shift))

    # deterministic orderings of each seed (ascending + zig‑zag)
    enriched: list[np.ndarray] = []
    for v in seeds:
        enriched.append(v)                               # original
        enriched.append(_ascending_order(v))            # sorted ascending
        enriched.append(_zigzag_order(v))               # zig‑zag
    seeds = enriched

    # previous best, if any
    if previous is not None and previous.shape == (n,) and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final normalisation & safety
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The best sequence (list of floats) is returned and stored globally
    as ``height_sequence_1`` for warm‑starts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # problem size – a modest size that still gives many degrees of freedom
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # 1) seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors (including orderings).")

    # ------------------------------------------------------------------
    # 2) quick elite selection
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    # keep top‑k seeds for the costly L‑BFGS phase
    seeds.sort(key=_objective_exact)
    elite = seeds[:min(12, len(seeds))]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # 3) continuation L‑BFGS on the smooth surrogate (each elite seed)
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        800.0, 3200.0, 12800.0, 51200.0,
        200_000.0, 1_000_000.0, 5_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=1500,
        )
        # try deterministic reorderings (ascending / zig‑zag)
        candidates = [seq_opt,
                      _ascending_order(seq_opt),
                      _zigzag_order(seq_opt)]
        for cand in candidates:
            val = _objective_exact(cand)
            if val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = val
                print(f"[Update] New global best (L‑BFGS) = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) long exponentiated‑gradient run from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(1_200_000, remaining * 800))
        print("\n--- Long EG optimisation (starting from best) ---")
        seq_eg, val_eg = _exp_gradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=max_iters,
            eta0=2.2,
        )
        # re‑order after EG as well
        for cand in [seq_eg, _ascending_order(seq_eg), _zigzag_order(seq_eg)]:
            val = _objective_exact(cand)
            if val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = val
                print(f"[Update] New best after EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) aggressive stochastic local search (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(400_000, remaining * 120))
        print("\n--- Stochastic local search (mass transfers) ---")
        seq_loc, val_loc = _local_search(
            global_best_seq,
            n,
            deadline,
            iters=iters,
            step_frac=0.005,
        )
        for cand in [seq_loc, _ascending_order(seq_loc), _zigzag_order(seq_loc)]:
            val = _objective_exact(cand)
            if val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = val
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) final polishing L‑BFGS (very high β, few iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = int(min(2000, remaining * 2_000))
        print("\n--- Final polishing L‑BFGS (high β) ---")
        seq_fin = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000.0, 10_000_000.0, 20_000_000.0],
            max_iter_per_beta=per_beta,
        )
        for cand in [seq_fin, _ascending_order(seq_fin), _zigzag_order(seq_fin)]:
            val = _objective_exact(cand)
            if val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = val
                print(f"[Update] New best after final L‑BFGS = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7) final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for warm‑starts in future calls
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
