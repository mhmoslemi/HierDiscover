"""
Improved search for a step‐function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Very diverse seeding** – analytic families, a huge number of
   Dirichlet draws spanning many concentration parameters, random
   exponentials, and a few Fourier‑phase constructions.  All seeds are
   ℓ₁‑normalised.

2. **Adaptive mirror‑descent (Adam) on the exact max‑convolution** – a
   mirror‑descent update in log‑space (the exponential‑gradient trick) with
   Adam moments.  The sub‑gradient is obtained from *all* active indices
   where the convolution attains its current maximum.

3. **Long simulated‑annealing polishing** – Gaussian mass transfers
   between random coordinates, with a slowly decreasing temperature.
   The best sequence found during the whole run is stored into the global
   ``height_sequence_1`` variable.

The optimiser respects the 1000 s budget, uses FFT‑based convolutions for
speed, and prints a short log of progress.  The returned list can be fed
directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for an optional L‑BFGS refinement)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector that sums to one."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_cosine(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, projected to the simplex."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(num_freq):
        freq = rng.uniform(0.0, 0.5)          # normalised frequency ∈ (0, 0.5)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.001) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return seq.copy()
    return noisy / s


def _mix(seq_a: np.ndarray, seq_b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * seq_a + alpha * seq_b
    s = mixed.sum()
    return mixed / s if s > 0 else seq_a.copy()


# ----------------------------------------------------------------------
# Mirror‑descent with Adam moments (exact max convolution)
# ----------------------------------------------------------------------
def _adam_exp_gradient(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000,
                       lr: float = 0.5,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) using Adam–style adaptive steps.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active set – indices whose value is (almost) the maximal one
        eps_rel = max_val * 1e-12
        active = np.where(conv >= max_val - eps_rel)[0]

        # sub‑gradient of the exact max‑convolution
        if active.size:
            g_tot = np.zeros_like(a)
            for k in active:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx_i = np.arange(i_start, i_end + 1, dtype=int)
                rev_idx = k - idx_i
                g_tot[idx_i] += a[rev_idx]
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a = log_a - step

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (gaussian mass transfer)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 300_000,
                         t_start: float = 0.02,
                         t_end: float = 1e-8,
                         sigma_start: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Gaussian mass‑transfer SA.  A mass `delta` is taken from a random index
    and added to another random index (both kept non‑negative).  Temperature
    and perturbation magnitude are cooled linearly.
    Returns the best sequence observed.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear schedule
        T = t_start * (1.0 - attempts / max_attempts) + t_end * (attempts / max_attempts)
        sigma = sigma_start * (T / t_start)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            attempts += 1
            continue

        # propose to move a small amount of mass from i to j
        delta = sigma * rng.normal()
        # ensure we stay non‑negative
        delta = np.clip(delta, -a[i], a[j])

        ns = a.copy()
        ns[i] -= delta
        ns[j] += delta

        val = _objective_surrogate(ns)
        delta_obj = val - cur_val
        if delta_obj < 0 or rng.random() < math.exp(-delta_obj / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the time budget
    and stores the best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0           # a tiny safety margin

    # --------------------------------------------------------------
    # Problem size (feel free to change – 4000 works well with the
    # current time budget)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best if it exists
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. SEED GENERATION – very wide variety
    # --------------------------------------------------------------
    seeds = []

    # Analytic families (coarse parameter grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.25))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_tukey_window(n, 0.75))

    for gamma in np.linspace(0.2, 3.0, 18):
        seeds.append(_sine_window(n, gamma))

    for sigma_frac in np.linspace(0.05, 0.4, 18):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.0, 5.0, 18):
        seeds.append(_power(n, p))

    for rate in np.logspace(-4, -0.5, 16):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Random Dirichlet draws – concentration covering many orders of magnitude
    conc_vals = np.logspace(-4, 1, 15)
    for conc in conc_vals:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(40):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random cosine sums
    for nf in (2, 3, 4, 5, 7, 9):
        for _ in range(6):
            seeds.append(_lowfreq_cosine(n, nf, rng))

    # Fourier‑phase constructions: random complex Gaussian in frequency domain,
    # enforce Hermitian symmetry, iFFT, keep non‑negative part.
    for _ in range(12):
        # build a random real signal with random phases
        freq = rng.normal(size=n) + 1j * rng.normal(size=n)
        # enforce hermitian symmetry for real output
        freq = (freq + np.conj(freq[::-1])) / 2.0
        real_signal = np.fft.ifft(freq).real
        a = np.maximum(real_signal, 0.0)
        if a.sum() == 0.0:
            a = _uniform(n)
        a /= a.sum()
        seeds.append(a)

    # Perturb previous best if we have one
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(6):
            seeds.append(_perturb(prev_arr, rng, sigma=0.004))

    # --------------------------------------------------------------
    # 2. QUICK surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n=== Seed evaluation ===")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  Rank #{i+1:2d}: surrogate = {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start refinement (Adam‑EG + SA)
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining time for this seed
        remaining = deadline - time.time()
        # keep ≈12 % of the remaining budget for this seed, the rest for later steps
        sub_deadline = time.time() + remaining * 0.12 / max(1, (top_k - si))

        # ---- (a) Adam‑accelerated exponentiated‑gradient on exact max ----
        seq_adam, val_adam = _adam_exp_gradient(seed_vec.copy(),
                                                n,
                                                sub_deadline,
                                                max_iters=150_000,
                                                lr=0.8)

        # ---- (b) short SA polishing ----
        if time.time() < sub_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_adam,
                                                  n,
                                                  sub_deadline,
                                                  rng,
                                                  max_attempts=80_000,
                                                  t_start=0.015,
                                                  sigma_start=0.003)
        else:
            seq_sa, val_sa = seq_adam, val_adam

        # pick best of the two refinements (plus original seed)
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_adam, val_adam),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long polishing (Adam‑EG + SA)
    # --------------------------------------------------------------

        final_seq, final_val = _adam_exp_gradient(global_best_seq,
                                                  n,

                                                  max_iters=1_800_000,
                                                  lr=0.6)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

        if time.time() < deadline:
            print("\n--- Final SA polishing ---")
            final_sa_seq, final_sa_val = _simulated_annealing(
                global_best_seq,
                n,
                deadline,
                rng,
                max_attempts=60_000,
                t_start=0.008,
                sigma_start=0.001
            )
            if final_sa_val < global_best_val:
                global_best_seq = final_sa_seq
                global_best_val = final_sa_val
                print(f"[Final SA] improved to {global_best_val:.6f}")

    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store the best sequence globally for the next run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
