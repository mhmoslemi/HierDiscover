def smooth_optimize_adam(start_a, n, deadline):
    # start with log probabilities
    u = np.log(start_a + 1e-30)
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 20000.0, 50000.0, 100000.0, 200000.0, 500000.0, 1000000.0]
    # We could also go up to 5e6.
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    for beta in beta_schedule:
        if time.time() > deadline: break
        # number of inner steps depends on remaining time and maybe fixed
        steps = 2000
        for _ in range(steps):
            if time.time() > deadline: break
            t += 1
            a = _softmax(u)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_u = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_u
            v = beta2 * v + (1 - beta2) * (grad_u * grad_u)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step_size = 0.5 / (np.sqrt(v_hat) + eps)  # tune step size
            u = u - step_size * m_hat
```

        if time.time() > deadline: break
        # Choose number of steps based on remaining time
        remaining = deadline - time.time()

        if steps < 10: steps = 10

            conv = _conv_fft(a, a)
            maxc = np.max(conv)

            v_ = beta2 * v_ + (1 - beta2) * (grad_u * grad_u)

        a_current = _softmax(u)
        val = _objective_surrogate(a_current)
        if val < best_val:
            best_val = val
            best_seq = a_current.copy()
    return best_seq, best_val

"""
Algorithm
---------

1.  Build a large, diverse set of seed vectors (analytic windows,
    beta‑shapes with tiny parameters, arcsine‑type “U‑shaped” vectors,
    random Dirichlet draws, etc.).  All vectors are ℓ₁‑normalised
    (they sum to 1) so that the surrogate objective is simply
    ``2·n·max(conv(a,a))``.

2.  Evaluate the surrogate for every seed and keep the best few (default 12).

3.  For each of those seeds run an intensive local optimisation:
    * a)  L‑BFGS‑B on the smooth‐max (log‑sum‑exp) surrogate using a
        long schedule of β values (up to 10⁶) and many iterations
        per β; this gives a high‑quality descent direction.
    * b)  A few thousand steps of Adam on the same surrogate (starting
        from the L‑BFGS result) to explore any remaining directions.
    * c)  A final exponentiated‑gradient (mirror‑descent) pass on the
        exact max‑convolution to polish the solution.

    The best result among all refined candidates is kept.

4.  As a cheap post‑processing step, convexly mix the best vector with
    the arcsine‑type vector using a ternary search – sometimes a tiny
    mixture improves the max‑convolution by a few 10⁻⁶.

5.  Return the best sequence (as a plain ``list[float]``) and store it
    in the global variable ``height_sequence_1`` for the next run.

The whole search respects a global time budget (default 1000 s); the
deadline is checked frequently so that the function always returns a
valid solution before the timeout.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:

    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:

    a = (a0

def _kaiser_window(n: int, beta: float) -> np.ndarray:

    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete grid."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_window(n: int) -> np.ndarray:
    """U‑shaped arcsine‑like distribution:
       a_i ∝ 1/√((i+½)*(n‑½‑i)).
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1) + 1/√(n‑i).  Symmetric heavy‑end shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    """Ternary search for the best convex combination of a and b."""
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if best_val is None or val1 < best_val:
                best_val = val1
                best_seq = seq1
        else:
            lo = m1
            if best_val is None or val2 < best_val:
                best_val = val2
                best_seq = seq2
    # final refinement
    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective_surrogate(best_seq)
    else:
        mid = (lo + hi) / 2.0
        cand = _mix_shapes(a, b, mid)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq = cand
            best_val = cand_val
    return best_seq, best_val

# L‑BFGS smoother (if SciPy is available)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000):
    """Smooth‑max optimisation using L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0,
                         1000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = np.sum(exp_beta)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

        if time.time() > deadline:
            break
        # number of Adam steps for this β (adapts to remaining time)
        remaining = deadline - time.time()
        max_steps = int(min(steps_per_beta, max(5, remaining / 0.003)))
        for _ in range(max_steps):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(u)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]
            dot = np.dot(grad_a, a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u * grad_u)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.05 / (np.sqrt(v_hat) + eps)
            u = u - step * m_hat

        # evaluate surrogate after this β
        a_cur = _softmax(u)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    if best_seq is None:
        best_seq = _softmax(u)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step‑function minimising ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin
    n = 4000                                        # length of the step function

    # ------------------------------------------------------------------
    # 1.  Build a rich seed family
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # classic windows
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))
    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 14):
        seeds.append(_gaussian(n, sigma))
    for sigma in (0.10, 0.12):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))
    for p in np.linspace(0.0, 2.0, 13):
        seeds.append(_power(n, p))

    # beta‑shape – include very small parameters (U‑shaped)
    for a_ in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for b_ in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
            seeds.append(_beta_shape(n, a_, b_))

    # arcsine‑like heavy‑edge shape
    seeds.append(_arcsine_window(n))
    seeds.append(_inverse_sqrt(n))

    # exponential / logistic families
    for gamma in np.linspace(0.1, 8.0, 16):
        seeds.append(_exp_decay(n, gamma))
    for steep in np.linspace(0.5, 30.0, 10):
        for shift in np.linspace(0.1, 0.9, 9):
            seeds.append(_logistic_decay(n, steep, shift))

    # random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_dirichlet(n, 1.0, rng) * rng.exponential(size=n))

    # Low‑frequency random cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            # reuse the low‑freq generator from the original script
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            s = a.sum()
            seeds.append(a / s if s > 0 else _uniform(n))

    # perturbation of previous best (if any)
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None
    if prev_arr is not None and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(4):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                seeds.append(noisy / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Pairwise mixing of the best seeds (cheap improvement)
    # ------------------------------------------------------------------
    mix_deadline = deadline - 2.0
    mixture_candidates = []
    for i in range(len(top_seeds)):
        for j in range(i + 1, len(top_seeds)):
            if time.time() > mix_deadline:
                break
            seq_mix, val_mix = _best_mixed_pair(top_seeds[i], top_seeds[j],
                                                mix_deadline, steps=20)
            mixture_candidates.append((seq_mix, val_mix))

    all_candidates = evaluated[:top_k] + mixture_candidates
    all_candidates.sort(key=lambda x: x[1])
    best_seq = all_candidates[0][0].copy()
    best_val = all_candidates[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Intensive local refinement of the best few candidates
    # ------------------------------------------------------------------
    refined = []
    num_refine = min(4, len(all_candidates))
    for idx in range(num_refine):
        if time.time() >= deadline:
            break
        seed_vec = all_candidates[idx][0]
        # allocate a fraction of remaining budget to this candidate
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.45 / (num_refine - idx)

        print(f"\n[Refine #{idx+1}] start surrogate = "
              f"{_objective_surrogate(seed_vec):.6f}")

        # L‑BFGS (if available)
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec, n, sub_deadline,
                                           max_iter_per_beta=8000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        # Adam fine‑tuning
        if time.time() < sub_deadline:
            seq_adam, val_adam = _adam_optimize(seq_lbfgs, n, sub_deadline)
        else:
            seq_adam, val_adam = seq_lbfgs, val_lbfgs

        # final EG polishing
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam, n, sub_deadline,
                                               max_iters=200_000, eta0=6.0)
        else:
            seq_eg, val_eg = seq_adam, _objective_surrogate(seq_adam)

        local_best_seq, local_best_val = min(
            [(seed_vec, _objective_surrogate(seed_vec)),
             (seq_lbfgs, val_lbfgs),
             (seq_adam, val_adam),
             (seq_eg, val_eg)],
            key=lambda x: x[1])

        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")
        refined.append((local_best_seq, local_best_val))

        if local_best_val < best_val:
            best_seq = local_best_seq
            best_val = local_best_val
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional post‑mix with arcsine shape (tiny improvement sometimes)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        arcsine_vec = _arcsine_window(n)
        best_seq, best_val = _best_mixed_pair(best_seq, arcsine_vec,
                                              deadline, steps=25)

    # ------------------------------------------------------------------
    # 6. Long EG polishing (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(best_seq, n, deadline,
                                                max_iters=2_000_000,
                                                eta0=5.0)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final L‑BFGS smoothing (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing ---")
        seq_final_lbfgs = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                                max_iter_per_beta=8000)
        val_final_lbfgs = _objective_surrogate(seq_final_lbfgs)
        if val_final_lbfgs < best_val:
            best_seq = seq_final_lbfgs
            best_val = val_final_lbfgs
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Official evaluation with the provided evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
