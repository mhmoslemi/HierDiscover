"""
Propose a low‑autocorrelation step‐function.

The algorithm consists of three stages:

1. **Rich seed pool** – a handful of analytic families (power law,
   beta‑shape, exponential decay) together with a bunch of random
   Dirichlet draws.  All seeds are ℓ₁‑normalised (sum = 1) and their
   surrogate value 2·n·max(conv) is computed.  The best `top_k`
   seeds are kept.

2. **Smooth Adam optimisation** – each top seed is refined by minimising
   a smooth approximation of the true objective:
        2·n·log‑sum‑exp(β·conv(seq,seq))/β .
   The optimisation is performed on the log‑softmax parametrisation
   (mirrored‑descent) using the Adam optimiser.  A schedule of increasing
   β (1 → … → 10⁵) makes the surrogate gradually approach the true
   max‑convolution.  The best sequence seen during the run (according to
   the exact surrogate) is retained.

3. **Permutation hill‑climbing + final EG polishing** – the best
   sequence from step 2 is locally reordered (swap moves) and lightly
   perturbed.  Accepted moves follow a simple simulated‑annealing rule.
   Finally a short exponentiated‑gradient (sub‑gradient of the exact max)
   polish is applied.

All work respects the 1000 s time budget and the globally best sequence
is stored in ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric helpers (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                         # should never happen
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p, i = 0..n‑1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shape a_i ∝ i^{α‑1} (n‑i+1)^{β‑1} ."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet( conc,…,conc )."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule=None,
                     max_iters_per_beta: int = 30_000,
                     eta0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth surrogate
        2·n·log‑sum‑exp(β·conv)/β
    using Adam.  Returns the best sequence (according to the exact
    surrogate) seen during the run.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         20_000.0, 50_000.0, 100_000.0,
                         200_000.0, 500_000.0, 1_000_000.0]

    # initialise log‑probabilities
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # iterations for this β – limited by remaining time
        remaining_time = deadline - time.time()
        # cheap estimate: 5 µs per iter is ~200 k iters per sec
        # we cap by the user‑provided budget
        iters = min(max_iters_per_beta,
                    int(remaining_time / 0.005))   # rough safety

        for _ in range(iters):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(v)                   # probabilities, sum=1

            # exact convolution (full length 2n‑1)
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            beta_conv = beta * conv
            shift = np.max(beta_conv)          # numerical stabilization
            exp_term = np.exp(beta_conv - shift)
            w = exp_term / exp_term.sum()      # softmax weights over k

            # gradient wrt a  (see analysis)
            # g = 4·n·( w * a ) convolved with reversed a
            a_rev = a[::-1]
            g_full = _conv_fft(w, a_rev)       # length = 3n‑2
            grad_a = 4.0 * n * g_full[n - 1:n - 1 + n]

            # mirror‑descent gradient: ∂/∂v = a·(grad_a - ⟨grad_a,a⟩)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat_hat = v_hat / (1.0 - beta2 ** t)

            v = v - eta0 * m_hat / (np.sqrt(v_hat_hat) + epsilon)

            # occasional exact evaluation
            if t % 5_000 == 0:
                cur_seq = _softmax(v)
                cur_val = _objective_surrogate(cur_seq)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()

    # final exact check
    final_seq = _softmax(v)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_seq, best_val = final_seq, final_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill‑climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two positions or perturbs the vector.
    Accepts worsening moves with a simple annealing schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing schedule parameters
    T_start = 1e-3 * cur_val          # temperature proportional to objective magnitude
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # linear temperature cooling
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # small Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # accept worse move with probability exp(−Δ/T)
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple exponentiated‑gradient (sub‑gradient of exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution (active index sub‑gradient)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    # Load a previous solution if it exists (continuation runs)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Seed generation
    # --------------------------------------------------------------
    seeds = []

    # Analytic families – a compact but expressive set
    seeds.append(_uniform(n))
    # power‑law exponents spanning the interesting region
    for p in np.linspace(0.1, 3.0, 12):
        seeds.append(_power(n, p))

    # exponential decays – dense sweep
    for rate in np.logspace(-5, -0.3, 30):
        seeds.append(_exp_decay(n, rate))

    # beta‑shape – a few focused points (including the arcsine case)
    for (a, b) in [(0.5, 0.5), (0.7, 0.7), (1.5, 1.5), (2.0, 2.0),
                   (0.3, 1.0), (1.0, 0.3)]:
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (various concentrations)
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Include previous best (if any) and some perturbations of it
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            # small Gaussian jitter then renormalise
            noisy = prev_arr + 0.004 * rng.normal(size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0.0:
                seeds.append(noisy / s)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the leading few seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Refine each top seed with Adam (smooth surrogate) +
    #    permutation hill‑climb + final EG polish
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fair share of remaining time for this seed
        remaining = deadline - time.time()
        share = remaining * 0.20 / max(1, (top_k - si))   # 20 % of remaining per seed
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] optimisation budget ≈ {share:.1f}s")

        # ---- (a) Adam on smooth surrogate ----
        seq_adam, val_adam = _adam_smooth_opt(seed_vec, n, sub_deadline,
                                              max_iters_per_beta=25_000,
                                              eta0=0.07)

        # ---- (b) permutation hill‑climb on the Adam result ----
        if time.time() < sub_deadline:
            seq_hc, val_hc = _perm_hill_climb(seq_adam, n, sub_deadline, rng,
                                              max_iters=150_000)
        else:
            seq_hc, val_hc = seq_adam, val_adam

        # ---- (c) final exact‑max exponentiated‑gradient polish ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_hc, n, sub_deadline,
                                               max_iters=250_000,
                                               eta0=5.0)
        else:
            seq_eg, val_eg = seq_hc, val_hc

        # pick best of the three refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_adam, val_adam),
            (seq_hc, val_hc),
            (seq_eg, val_eg)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final global polishing (short Adam run with very high β)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final global Adam polishing ---")
        final_seq, final_val = _adam_smooth_opt(global_best_seq, n,
                                                deadline,
                                                max_iters_per_beta=30_000,
                                                eta0=0.04,
                                                beta_schedule=[1e5, 5e5, 1e6])
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
