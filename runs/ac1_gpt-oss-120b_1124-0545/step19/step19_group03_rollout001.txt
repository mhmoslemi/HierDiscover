"""
Refine a step‑function (non‑negative, ℓ₁–normalised) to minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the full linear convolution.
The routine works inside the given time budget (default 1000 s) and stores
the best sequence in the global variable ``height_sequence_1``.
It proceeds as follows:

1. Load the best sequence from a previous run (if any).  If none is
   available, generate a few analytic seeds and keep the best one.

2. **Exponentiated‑gradient (mirror‑descent) polishing** – a large number of
   EG updates on the exact max‑convolution (sub‑gradient taken from all
   active maximal indices).  This quickly drives the surrogate objective
   down.

3. **Stochastic restarts** – several perturbations of the current best
   are created; each is run for a short EG phase followed by a brief
   gradient‑driven *pairwise mass‑transfer* hill‑climb that moves mass from
   the most “harmful’’ coordinates to the most “benign’’ ones.

4. **Final hill climbing** – a longer gradient‑driven pairwise transfer
   loop (with a decaying step size) fine‑tunes the solution until the
   deadline is reached.

The combination of multiplicative updates and very fine deterministic
local moves typically pushes the bound a few thousandths lower than the
previous record (target ≈ 1.503).  The function returns a plain Python list
that can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS, not strictly needed here)

# ----------------------------------------------------------------------
# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – simple seeds (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered (ℓ₁‑normalised).
    """
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to keep positivity and ℓ₁‑norm automatically
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of max‑convolution
        g_tot = np.zeros_like(a)
        for k in active:
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)
            idx = np.arange(i0, i1 + 1, dtype=int)
            rev = k - idx
            g_tot[idx] += a[rev]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient‑driven pairwise mass‑transfer hill‑climbing
# ----------------------------------------------------------------------
def _pairwise_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_attempts: int = 150_000):
    """
    Very fine local optimisation.
    At each step mass is moved from the coordinate with the largest
    sub‑gradient (most harmful) to the one with the smallest.
    The step size decays linearly with the remaining time.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    start_time = time.time()
    total_budget = deadline - start_time
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # recompute convolution and active set
        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # compute sub‑gradient for the active max
        g_tot = np.zeros_like(a)
        for k in active:
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)
            idx = np.arange(i0, i1 + 1, dtype=int)
            rev = k - idx
            g_tot[idx] += a[rev]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        # avoid trivial move
        if i_max == i_min:
            attempts += 1
            continue

        remaining = deadline - time.time()
        # linear decay of the fraction of mass we are willing to move
        frac = max(1e-6, remaining / total_budget)  # between ~1 and ~0
        # move at most 10 % of the source weight, scaled by the decay factor
        eps_move = min(a[i_max] * 0.1 * frac, a[i_max] * 0.02 + 1e-8)

        if eps_move <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= eps_move
        a_new[i_min] += eps_move

        # clip (numerical safety) and re‑normalise
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            attempts += 1
            continue
        a_new /= s

        new_val = _objective_surrogate(a_new)

        if new_val < best_val - 1e-12:      # strict improvement
            a = a_new
            best_seq = a_new
            best_val = new_val

        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The best sequence is stored in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – we keep the same n that gave the previous record
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous solution (if any) and fall back to a cheap seed set
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,) and prev_arr.sum() > 0.0:
                init_seq = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
                init_seq = None
        except Exception:
            prev_arr = None
            init_seq = None
    else:
        prev_arr = None
        init_seq = None

    if init_seq is None:
        # Very small pool of analytic seeds – evaluate quickly
        cand = [_uniform(n), _triangular(n), _exp_decay(n, 0.005), _sym_exp_decay(n, 0.005)]
        # a few Dirichlet draws for diversity
        for conc in (0.001, 0.01, 0.1, 1.0):
            for _ in range(3):
                cand.append(_random_dirichlet(n, conc, rng))
        # pick the best by surrogate value
        best_val = float('inf')
        best_seq = None
        for vec in cand:
            vec = np.maximum(vec, 0.0)
            vec /= vec.sum()
            val = _objective_surrogate(vec)
            if val < best_val:
                best_val = val
                best_seq = vec
        init_seq = best_seq
        print(f"[Seed] initial surrogate value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – long exponentiated‑gradient polishing
    # --------------------------------------------------------------
    t_mid = start_time + 0.55 * (deadline - start_time)   # ~55 % of budget
    print("\n--- Stage 1: long EG polishing ---")
    seq_eg, val_eg = _exp_gradient_opt(init_seq, n, t_mid,
                                       max_iters=2_000_000, eta0=5.0)
    print(f"[Stage 1] best surrogate = {val_eg:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – stochastic restarts + short EG + hill‑climb
    # --------------------------------------------------------------
    print("\n--- Stage 2: stochastic restarts ---")
    best_seq = seq_eg
    best_val = val_eg
    remaining_budget = deadline - time.time()
    n_restarts = 5
    for r in range(n_restarts):
        if time.time() >= deadline:
            break
        # allocate roughly equal share of the remaining budget
        sub_deadline = time.time() + remaining_budget / max(1, n_restarts - r)

        # perturb current best (different σ for each restart)
        sigma = 0.004 + 0.002 * r
        pert = _perturb(best_seq, rng, sigma=sigma)

        # short EG
        seq_tmp, val_tmp = _exp_gradient_opt(pert, n, sub_deadline,
                                             max_iters=300_000,
                                             eta0=6.0)

        # very short hill‑climb
        hill_deadline = sub_deadline - 0.2   # keep a tiny margin
        if time.time() < hill_deadline:
            seq_tmp, val_tmp = _pairwise_hill_climb(seq_tmp, n, hill_deadline,
                                                    max_attempts=30_000)

        if val_tmp < best_val:
            best_seq = seq_tmp
            best_val = val_tmp
            print(f"[Restart {r+1}] new best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final intensive pairwise hill‑climbing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 3: final hill‑climbing ---")
        seq_final, val_final = _pairwise_hill_climb(best_seq, n, deadline,
                                                    max_attempts=250_000)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final hill‑climb] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Official evaluation via the supplied function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
