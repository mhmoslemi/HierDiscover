"""
Improved search for a step‑function that minimises ``evaluate_sequence``.
-----------------------------------------------------------------------
The objective is 2·n·max(conv(a,a))/sum(a)².  Because the sum can be fixed
(e.g. by a softmax parametrisation) the problem reduces to flattening the
full linear convolution as much as possible – i.e. making all entries close
to their average value S²/(2n‑1).  The new algorithm adds a *quadratic‑error*
stage that directly minimises the L² distance of the convolution from its
average.  After a few L‑BFGS runs on this smooth surrogate we polish the
solution with a continuation on a *log‑sum‑exp* smooth max (as in the
original code) and finish with cheap stochastic hill‑climbs.

Main stages
-----------
1.  Seed pool – analytic windows (Hann, Hamming, Blackman, Kaiser) and the
    families already used by the reference implementation.
2.  L²‑flattening – minimise ½‖conv(a,a)–mean(conv)‖² using L‑BFGS on the
    simplex (logits → softmax).  This quickly drives the convolution towards
    a constant profile.
3.  High‑β polishing – continuation L‑BFGS on a smooth‐max surrogate to
    shrink the true ∞‑norm.
4.  Targeted stochastic local search – mass transfers away from the
    currently active lag.
5.  Final evaluation and return.

The whole routine respects the 1000 s wall‑time budget.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Smooth surrogates
# ----------------------------------------------------------------------
def _smooth_max_objective(v: np.ndarray, n: int, beta: float):
    """Log‑sum‑exp smooth max of the convolution (β → ∞ → max)."""
    a = _softmax(v)                     # ℓ₁‑normalised (∑a = 1)
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _smooth_l2_objective(v: np.ndarray, n: int):
    """
    L²‑error surrogate: minimise ‖conv(a,a) – mean(conv)‖².
    Returns (obj, grad) where the gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised (∑a = 1)
    conv = _conv_fft(a, a)
    avg = conv.mean()                   # = 1/(2n‑1) for ∑a = 1
    r = conv - avg                       # residual
    # gradient of ½‖r‖² w.r.t a : 2·conv(r, a_rev)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, r)           # length = 3n‑2
    grad_a = 4.0 * full[n - 1: n - 1 + n]

    obj = np.sum(r * r)

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    surrogate,
                    beta_schedule: list[float] | None = None,
                    max_iter_per_beta: int = 1500) -> np.ndarray:
    """
    Generic continuation L‑BFGS.
    ``surrogate`` – a callable (v, n, beta) → (obj, grad) for a given β.
    If ``beta_schedule`` is supplied the surrogate is called with the
    current β, otherwise beta is ignored (e.g. the L² surrogate).
    """
    if beta_schedule is None:
        beta_schedule = [None]          # dummy list – only one pass

    v = np.log(start_vec + 1e-30)        # logits of the probability vector

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        args = (n, beta) if beta is not None else (n,)
        try:
            res = _opt.minimize(
                fun=lambda x, *a: surrogate(x, *a),
                x0=v,
                args=args,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_beta,
                         "ftol": 1e-12,
                         "disp": False},
            )
            v = res.x
        except Exception:   # pragma: no cover
            break

    return _softmax(v)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    return a / a.sum()


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    return a / a.sum()


def _semicircle_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (i - c) / c
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, 1.0))
    return a / a.sum()


def _poly_tent(n: int, power: float = 1.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) * (n - i)) ** power
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    # numpy provides the Kaiser window directly
    a = np.kaiser(n, beta)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)

# Seed‑pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_exponential_decay(n, 0.001))

    # new analytic families
    seeds.append(_sin_shape(n))
    seeds.append(_semicircle_shape(n))
    seeds.append(_poly_tent(n, power=1.2))
    seeds.append(_poly_tent(n, power=2.0))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_kaiser_window(n, beta=8.0))

    # power‑law sweep
    p_vals = np.linspace(0.2, 0.9, 12)
    shifts = [0.0, 5.0, 20.0, 50.0, 100.0]
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # random Dirichlet families (very sparse)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(3):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures
    base = np.arange(len(seeds))
    for _ in range(100):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(8):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Stochastic local searches
# ----------------------------------------------------------------------
def _targeted_local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 100_000,
    step_frac: float = 0.010,
) -> tuple[np.ndarray, float]:
    """Hill‑climb biased towards the active (maximal) convolution lag."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(99112233)

    for _ in range(iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i = rng.integers(i_start, i_end + 1)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (list of floats) and stores it in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed collection (including warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Evaluate seeds, retain elite pool (by exact objective)
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # keep the top‑k seeds for heavy optimisation
    seeds.sort(key=_objective_exact)
    elite = seeds[:min(20, len(seeds))]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – L²‑flattening (quadratic error surrogate)
    # ------------------------------------------------------------------
    print("\n--- Stage 1: L²‑flattening (conv → constant) ---")
    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            break
        per_beta = max(800, int(min(1800, (deadline - time.time()) * 0.06 * 1e5)))
        seq_opt = _lbfgs_optimize(
            seed_vec,
            n,
            deadline,
            surrogate=_smooth_l2_objective,
            beta_schedule=[None],          # single pass – β not used
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L² {idx + 1}/{len(elite)}] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS on smooth max (aggressive polishing)
    # ------------------------------------------------------------------
    print("\n--- Stage 2: high‑β smooth‑max polishing ---")
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 50_000_000.0, 100_000_000.0,
        500_000_000.0, 1_000_000_000.0,
        5_000_000_000.0,
    ]
    seq_opt = _lbfgs_optimize(
        global_best_seq,
        n,
        deadline,
        surrogate=_smooth_max_objective,
        beta_schedule=beta_sched,
        max_iter_per_beta=1200,
    )
    val_opt = _objective_exact(seq_opt)
    print(f"[Polish] objective = {val_opt:.6f}")
    if val_opt < global_best_val:
        global_best_seq = seq_opt.copy()
        global_best_val = val_opt
        print(f"[Update] New best after polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – targeted stochastic local search (active‑lag focus)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(200_000, remaining * 45))
        print("\n--- Stage 3: targeted local search ---")
        seq_loc, val_loc = _targeted_local_search(
            global_best_seq,
            n,
            deadline,
            iters=iters,
            step_frac=0.011,
        )
        print(f"[TargetedLocal] best = {val_loc:.6f}")
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation via the provided security‑checked function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
