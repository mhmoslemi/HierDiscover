def _smooth_objective_and_grad(seq, n, beta):
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp
    rev_seq = seq[::-1]
    full = _conv_fft(rev_seq, w)
    grad = 2.0 * full[n-1:n-1+n]
    return obj, grad

def _optimize_weights(seeds, n, deadline, beta_schedule, max_iter_per_beta=2000):
    m = len(seeds)
    seeds_arr = np.stack(seeds)  # shape (m,n)
    # initialize v = zeros
    v = np.zeros(m, dtype=np.float64)

    # Use SciPy L-BFGS if available, else simple gradient descent.
    if _opt is not None:
        for beta in beta_schedule:
            if time.time() > deadline:
                break

            def fun(v_local):
                w = _softmax(v_local)
                mix = np.dot(w, seeds_arr)  # shape (n,)
                obj, grad_mix = _smooth_objective_and_grad(mix, n, beta)
                # grad w.r.t seeds weights:
                grad_w = seeds_arr @ grad_mix   # (m,) dot each seed
                grad_v = w * (grad_w - np.dot(w, grad_w))
                return obj, grad_v

            def callback(_):
                if time.time() > deadline:
                    raise KeyboardInterrupt()

            try:
                res = _opt.minimize(fun=fun,
                                    x0=v,
                                    method='L-BFGS-B',
                                    jac=True,
                                    callback=callback,
                                    options={'maxiter': max_iter_per_beta, 'ftol': 1e-12, 'disp': False})
                v = res.x
            except KeyboardInterrupt:
                break
            except Exception as e:
                print(f"[Weight L-BFGS] exception {e}, beta={beta}")
                break

        final_w = _softmax(v)
        final_mix = np.dot(final_w, seeds_arr)
        return final_mix
    else:
        # fallback: simple projected gradient descent (or uniform)
        w = _softmax(v)
        for beta in beta_schedule:
            if time.time() > deadline:

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(a)=1) – smooth version
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad_a) where
        obj = 2·n·smooth_max(conv(a,a))
    with smooth_max implemented by log‑sum‑exp with parameter ``beta``.
    The gradient is w.r.t. *a*.
    """
    conv = _conv_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # smooth max value
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights for each convolution entry (∂/∂conv)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice back to length n
    return obj, grad_a


def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate for a ℓ1‑normalised vector: 2·n·max(conv)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2.0 * np.pi * i / (n - 1))
         + a2 * np.cos(4.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                 # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best sequence (and its surrogate value) seen during the run.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Low‑dimensional mixture optimisation (weights over a small seed set)
# ----------------------------------------------------------------------
def _optimize_weights(seeds, n, deadline,
                      beta_schedule=None,
                      max_iter_per_beta: int = 2000):
    """
    Finds a convex combination of *seeds* (all ℓ1‑normalised) that minimises
    the smoothed max‑convolution using L‑BFGS‑B in the soft‑max parameters.
    Returns the resulting mixture vector (also ℓ1‑normalised).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 20000.0, 50000.0]

    seeds_arr = np.stack(seeds)          # shape (m, n)
    m = seeds_arr.shape[0]

    # start from uniform weights → soft‑max parameters = 0
    v = np.zeros(m, dtype=np.float64)

    if _opt is None:                     # pragma: no cover
        # fall‑back: plain projected gradient descent (very slow)
        w = _softmax(v)
        for beta in beta_schedule:
            if time.time() > deadline:
                break
            for _ in range(max_iter_per_beta):
                mix = np.dot(w, seeds_arr)
                _, grad_mix = _smooth_obj_and_grad(mix, n, beta)
                grad_w = seeds_arr @ grad_mix
                grad_v = w * (grad_w - np.dot(w, grad_w))
                step = 0.1
                v -= step * grad_v
                w = _softmax(v)
                if time.time() > deadline:
                    break
        final_mix = np.dot(_softmax(v), seeds_arr)
        return final_mix

    # Use L‑BFGS‑B
    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            w_local = _softmax(v_local)                # weights on the simplex
            mix = np.dot(w_local, seeds_arr)            # candidate vector
            obj, grad_mix = _smooth_obj_and_grad(mix, n, beta)

            # gradient w.r.t. the weights
            grad_w = seeds_arr @ grad_mix               # shape (m,)
            # chain rule through soft‑max
            grad_v = w_local * (grad_w - np.dot(w_local, grad_w))
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[Weight L‑BFGS] exception {e} at β={beta}")
            break

    final_w = _softmax(v)
    final_mix = np.dot(final_w, seeds_arr)
    return final_mix


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found (list of floats)
    and stores it globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0                # safety margin

    # -------------------------------------------------
    # Problem size (the original solution used n = 4000)
    # -------------------------------------------------
    n = 4000

    # -------------------------------------------------
    # 1. Seed generation (a modest but diverse set)
    # -------------------------------------------------
    print(f"\n--- Generating {n}‑point seeds ---")
    seeds = []

    # analytic families (a few representative parameter choices)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    for gamma in np.linspace(0.2, 3.0, 6):
        seeds.append(_sine_window(n, gamma=gamma))

    for beta in np.linspace(1.0, 12.0, 6):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.09, 0.22, 7):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.4, 4.0, 5)
    betas = np.linspace(0.4, 4.0, 5)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    for alpha in np.linspace(0.2, 0.8, 5):
        seeds.append(_tukey_window(n, alpha))

    for gamma in np.linspace(0.1, 5.0, 6):
        seeds.append(_exp_decay(n, gamma))

    for steep in np.linspace(0.5, 20.0, 5):
        for shift in np.linspace(0.2, 0.8, 4):
            seeds.append(_logistic_decay(n, steep, shift))

    # Random Dirichlet / exponential draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random cosine curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Warm‑start from the previous run (if available)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n:
                seeds.append(prev_arr / prev_arr.sum())
                for _ in range(3):
                    seeds.append(_perturb(prev_arr, rng, sigma=0.0025))
        except Exception:
            pass

    # -------------------------------------------------
    # 2. Quick surrogate evaluation and selection of promising seeds
    # -------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:02d}: {v:.6f}")

    # -------------------------------------------------
    # 3. Optimise a convex combination of the top seeds (low‑dimensional L‑BFGS)
    # -------------------------------------------------
    print("\n--- Optimising mixture of top seeds ---")
    mix_deadline = deadline - 2.0
    mixture = _optimize_weights(top_seeds,
                                n,
                                mix_deadline,
                                beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                               200.0, 500.0, 1000.0,
                                               2000.0, 5000.0],
                                max_iter_per_beta=2500)

    best_seq = mixture
    best_val = _objective_surrogate(best_seq)
    print(f"[Mixture] surrogate = {best_val:.6f}")

    # -------------------------------------------------
    # 4. Final exact polishing with exponentiated‑gradient
    # -------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient ---")
        eg_seq, eg_val = _exp_gradient_opt(best_seq,
                                           n,
                                           deadline,
                                           max_iters=400_000,
                                           eta0=6.0)
        if eg_val < best_val:
            best_seq = eg_seq
            best_val = eg_val
            print(f"[EG] improved surrogate = {best_val:.6f}")

    # -------------------------------------------------
    # 5. Official evaluation (may be a little slower – call once)
    # -------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:    # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
