"""
Improved search for a step‐function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The strategy expands the seed pool (extra analytic families, finer
parameter sweeps, cheap random draws) and uses a more aggressive
exponentiated‑gradient descent that follows the *single* worst convolution
index (instead of averaging over all active indices).  After a short
refinement each of the best seeds we run a long L‑BFGS optimisation on a
log‑sum‑exp surrogate with a richer β‑schedule, a final long EG polishing,
and a tiny simulated‑annealing pass.  All work respects the global time
budget and stores the best found sequence in the global variable
``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    a = np.hamming(n).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Standard Tukey window, α∈[0,1]."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.empty(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window (numpy implementation)."""
    a = np.kaiser(n, beta).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """(i+1)^{α‑1}·(n‑i)^{β‑1}  on  i=0..n‑1 ."""
    i = np.arange(1, n + 1, dtype=np.float64)   # 1‑based to avoid zero⁰
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalize."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_one_max(start_seq: np.ndarray,
                              n: int,
                              deadline: float,
                              max_iters: int = 150_000,
                              eta0: float = 8.0):
    """
    Mirror‑descent (exponentiated‑gradient) where the sub‑gradient is taken
    *only* from the single maximal convolution index (arg‑max).
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)          # stay on the simplex automatically

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(conv.argmax())        # index of the current worst term

        # build sub‑gradient for conv[k]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        g = np.zeros_like(a)
        g[idx_i] += a[rev_idx]        # a_{k-i}
        # the derivative contains a_i as second part – we add it via symmetry:
        g[rev_idx] += a[idx_i]        # a_{i}
        # for i == rev_idx this adds the same term twice = 2*a_i which is correct

        # learning‑rate schedule
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay on the simplex.
    """
    if _opt is None:
        return start_vec  # scipy unavailable – return unchanged

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 3.0, 8.0, 20.0, 50.0,
            150.0, 500.0, 1500.0, 5000.0,
            20_000.0, 80_000.0, 300_000.0,
            1_200_000.0, 5_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smooth‑max w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 80_000,
                         start_sigma: float = 0.004,
                         T_start: float = 0.02,
                         T_end: float = 1e-8):
    """
    Lightweight SA: Gaussian perturbations with a linearly decreasing temperature.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

    seeds = []

    # Analytic families (coarse grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))

    # Tukey – fine sweep
    for alpha in np.linspace(0.1, 0.9, 12):
        seeds.append(_tukey_window(n, alpha))

    # Kaiser – sweep of beta
    for beta in np.linspace(0.5, 12.0, 12):
        seeds.append(_kaiser_window(n, beta))

    # Sine window – gamma sweep
    for gamma in np.linspace(0.2, 4.0, 15):
        seeds.append(_sine_window(n, gamma))

    # Gaussian – sigma fraction sweep
    for sigma_frac in np.linspace(0.06, 0.40, 15):
        seeds.append(_gaussian(n, sigma_frac))

    # Power law – p sweep
    for p in np.linspace(0.0, 5.0, 16):
        seeds.append(_power(n, p))

    # Exponential decay – rate sweep
    for rate in np.logspace(-4.0, -0.3, 18):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Beta shape – fine grid on small exponents (U‑shaped) and asymmetrics
    alphas = np.logspace(-2.5, -0.1, 12)   # 0.003–0.79
    betas  = np.logspace(-2.5,  0.0, 12)   # 0.003–1
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Asymmetric beta (monotone) – one side very small, the other moderate
    for a in [0.02, 0.05, 0.1]:
        for b in [1.0, 2.0, 4.0]:
            seeds.append(_beta_shape(n, a, b))
            seeds.append(_beta_shape(n, b, a))

    # Random Dirichlet draws (include extreme sparsity)
    conc_vals = (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0)
    for conc in conc_vals:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6, 8):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Perturbations of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        prev_norm = prev_arr / prev_arr.sum()
        seeds.append(prev_norm)
        for _ in range(6):
            seeds.append(_perturb(prev_norm, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds (no optimisation yet)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(15, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best with the absolute best seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (short EG + L‑BFGS)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of remaining time for this seed
        remaining = deadline - time.time()
        # keep a fraction for later stages (≈12 % per remaining seed)
        sub_deadline = time.time() + remaining * 0.12 / max(1, (top_k - si))

        # ----- a) short exponentiated‑gradient (single‑max)
        seq_eg, val_eg = _exp_gradient_opt_one_max(seed_vec.copy(),
                                                  n,
                                                  sub_deadline,
                                                  max_iters=100_000,
                                                  eta0=9.0)

        # ----- b) L‑BFGS smoothing (more aggressive β‑schedule)
        if time.time() < sub_deadline:
            seq_lb = _lbfgs_smooth_optimize(seq_eg,
                                            n,
                                            sub_deadline,
                                            beta_schedule=[
                                                0.5, 1.0, 3.0, 8.0, 20.0,
                                                50.0, 150.0, 500.0, 1500.0,
                                                5000.0, 20_000.0, 80_000.0,
                                                300_000.0, 1_200_000.0,
                                                5_000_000.0],
                                            max_iter_per_beta=2500)
            val_lb = _objective_surrogate(seq_lb)
        else:
            seq_lb, val_lb = seq_eg, val_eg

        # ----- c) tiny simulated‑annealing to escape possible shallow minima
        hill_deadline = sub_deadline - 0.2
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_lb,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=30_000,
                                                  start_sigma=0.003,
                                                  T_start=0.015,
                                                  T_end=5e-8)
        else:
            seq_sa, val_sa = seq_lb, val_lb

        # pick the best among the three refinements (+ original seed)
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lb, val_lb),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"   *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing (large EG → L‑BFGS → final SA)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt_one_max(global_best_seq,
                                                         n,
                                                         deadline,
                                                         max_iters=800_000,
                                                         eta0=7.0)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- L‑BFGS refinement (high‑β) ---")
        refined_seq = _lbfgs_smooth_optimize(global_best_seq,
                                             n,
                                             deadline,
                                             beta_schedule=[
                                                 0.5, 1.0, 3.0, 8.0, 20.0,
                                                 50.0, 150.0, 500.0, 1500.0,
                                                 5000.0, 20_000.0, 80_000.0,
                                                 300_000.0, 1_200_000.0,
                                                 5_000_000.0, 2.0e7],
                                             max_iter_per_beta=3500)
        refined_val = _objective_surrogate(refined_seq)
        if refined_val < global_best_val:
            global_best_seq, global_best_val = refined_seq, refined_val
            print(f"[L‑BFGS] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final short SA pass ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=40_000,
                                                          start_sigma=0.001,
                                                          T_start=8e-3,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq, global_best_val = final_sa_seq, final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
