def _adam_opt(start_seq, n, deadline, max_iters=500000, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(log_a)
        # compute conv, gradient
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # gradient w.r.t a
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        grad_a = np.zeros_like(a)
        grad_a[idx_i] = 2 * a[rev_idx]  # as before

        # gradient w.r.t log a: a * (grad_a - dot(grad_a, a))
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate val
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute a from log_a
        a = _softmax(log_a)
        # compute convection and gradient
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        grad_a = np.zeros_like(a)
        grad_a[idx_i] = 2 * a[rev_idx]

import time, math
import numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())
def _conv_fft(x,y):
    L = len(x)+len(y)-1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx*fy, n=N)[:L]
    return conv

def _softmax(v):
    m = np.max(v)
    e = np.exp(v-m)
    s = e.sum()
    if s==0:
        return np.full_like(v, 1.0/len(v))
    return e/s

def _objective_surrogate(seq):
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0*n*np.max(conv))

# families definitions: similar to above.

def _uniform(n): return np.full(n, 1.0/n)
def _triangular(n):
    i = np.arange(n, dtype=np.float64)
    c = (n-1)/2.0
    a = 1.0 - np.abs(i - c)/c
    a[a<0]=0
    s=a.sum()
    return a/s if s>0 else _uniform(n)

def _sine_window(n,gamma=1.0):
    i=np.arange(n,dtype=np.float64)
    a = np.sin(np.pi*(i+1)/(n+1))**gamma
    s=a.sum()
    return a/s if s>0 else _uniform(n)

def _gaussian(n,sigma_frac):
    sigma = sigma_frac * n
    mu = (n-1)/2.0
    i = np.arange(n,dtype=np.float64)
    a = np.exp(-((i-mu)**2)/(2*sigma**2))
    s=a.sum()
    return a/s if s>0 else _uniform(n)

def _power(n,p):
    idx = np.arange(1,n+1,dtype=np.float64)
    a=1.0/(idx**p)
    s=a.sum()
    return a/s if s>0 else _uniform(n)

def _beta_shape(n,alpha,beta):
    i=np.arange(1,n+1,dtype=np.float64)
    a = (i**(alpha-1.0))*((n-i+1)**(beta-1.0))
    s=a.sum()
    return a/s if s>0 else _uniform(n)

def _exp_decay(n, rate):
    i=np.arange(n,dtype=np.float64)
    a=np.exp(-rate*i)
    s=a.sum()
    return a/s if s>0 else _uniform(n)

def _sym_exp_decay(n, rate):
    i=np.arange(n,dtype=np.float64)
    centre = (n-1)/2.0
    a=np.exp(-rate*np.abs(i-centre))
    s=a.sum()
    return a/s if s>0 else _uniform(n)

def _random_dirichlet(n, conc, rng):
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _random_exponential(n, rng):
    a = rng.exponential(scale=1.0,size=n)
    s=a.sum()
    return a/s if s>0 else _uniform(n)

def _lowfreq_curve(n, num_freq, rng):
    t = np.arange(n,dtype=np.float64)
    a = np.zeros(n,dtype=np.float64)
    for k in range(1,num_freq+1):
        phase = rng.uniform(0.0,2.0*math.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0*math.pi*k*t/n + phase)
    a = np.maximum(a,0.0)
    s=a.sum()
    return a/s if s>0 else _uniform(n)

def _mix_shapes(a,b,alpha):
    mixed = (1.0-alpha)*a + alpha*b
    s=mixed.sum()
    return mixed/s if s>0 else a.copy()

def _perturb(seq, rng, sigma=0.002):
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy,0.0)
    s = noisy.sum()
    if s <= 0.0: return seq.copy()
    return noisy/s

# Permutation hill climb
def _perm_hill_climb(start_seq, n, deadline, rng, max_iters=200_000):
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it=0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)
        if rng.random() < 0.55:
            i,j = rng.integers(0, n, size=2)
            if i==j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:
            sigma = 0.001 / math.sqrt(it+1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns,0.0)
            s = ns.sum()
            if s == 0.0: continue
            ns = ns / s
        val = _objective_surrogate(ns)
        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val
    return best_seq, best_val

# Adam optimizer
def _adam_opt(start_seq, n, deadline, max_iters=300_000, lr0=5.0,
              beta1=0.9, beta2=0.999, eps=1e-8):
    # start in log domain
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    best_seq = None
    best_val = float('inf')
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(log_a)
        # Compute max convolution and its gradient
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        grad_a = np.zeros_like(a)
        grad_a[idx_i] = 2.0 * a[rev_idx]

        # Gradient w.r.t. log a
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update
        lr = lr0 / math.sqrt(t)
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate new point
        a_new = _softmax(log_a)
        cur_val = _objective_surrogate(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
    if best_seq is None:
        best_seq = _softmax(log_a)
        best_val = _objective_surrogate(best_seq)
    return best_seq, best_val

# optional L-BFGS smoothing
try:
    import scipy.optimize as _opt
except Exception:
    _opt = None

def _lbfgs_smooth_optimize(start_vec, n, deadline,
                           beta_schedule=None, max_iter_per_beta=2000):
    if _opt is None:
        return start_vec
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
                         2_000.0, 5_000.0, 20_000.0, 50_000.0,
                         100_000.0, 200_000.0, 500_000.0, 1_000_000.0]
    v = np.log(start_vec + 1e-30)
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()
        try:
            res = _opt.minimize(fun=fun, x0=v,
                                method='L-BFGS-B', jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break
    return _softmax(v)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (vector) that minimises the
    given ``evaluate_sequence`` objective.  The algorithm proceeds in three
    stages:
        1. Generate a diverse pool of analytic and random seed vectors and
           keep the few best according to the surrogate objective
           2·n·max(conv(a,a)).
        2. For each top seed run a permutation‑based simulated‑annealing
           hill‑climb followed by a momentum‑based Adam mirror‑descent
           optimisation in log‑probability space.  Optionally a smoothing
           L‑BFGS pass refines the result further.
        3. A final long‑run Adam polishing uses the remaining time budget.
    The best sequence found is stored in the global variable
    ``height_sequence_1`` and returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    n = 4000

    # Load previous best if present
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        if np.any(arr < 0):
            prev_arr = None
        else:
            total = arr.sum()
            if total > 0:
                prev_arr = arr / total

    # ---------------------------------------------------------------
    # 1. Massive seed generation
    # ---------------------------------------------------------------
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.1, 3.5, 20):
        seeds.append(_sine_window(n, gamma))

    for sigma_frac in np.linspace(0.05, 0.35, 20):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.0, 4.0, 20):
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.3, 40):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Beta shape random draws
    for _ in range(80):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws with a wide range of concentration parameters
    conc_vals = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(12):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Exponential draws
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(7):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures
    mix_alphas = (0.2, 0.5, 0.8)
    for alpha in mix_alphas:
        for rate in (0.001, 0.005, 0.02):
            curve = _exp_decay(n, rate)
            seeds.append(_mix_shapes(_uniform(n), curve, alpha))

    # Random permutations of a few promising bases
    base_shapes = [_exp_decay(n, 0.003), _sym_exp_decay(n, 0.004),
                   _beta_shape(n, 2.0, 2.0), _gaussian(n, 0.18)]
    for base in base_shapes:
        for _ in range(5):
            seeds.append(np.random.permutation(base))

    # Include previous best if any
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            seeds.append(_perturb(prev_arr, rng, sigma=0.004))

    # ---------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep top‑k
    # ---------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total == 0.0:
            continue
        vec = vec / total
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ---------------------------------------------------------------
    # 3. Initialise global best
    # ---------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ---------------------------------------------------------------
    # 4. Multi‑start refinement
    # ---------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # Allocate roughly equal share for each remaining seed + a buffer for final polishing
        share = remaining * 0.15 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] start refinement, budget ≈ {share:.1f}s")

        # a) permutation hill climbing
        seq_perm, val_perm = _perm_hill_climb(seed_vec.copy(),
                                              n,
                                              sub_deadline,
                                              rng,
                                              max_iters=180_000)

        # b) Adam optimisation
        if time.time() < sub_deadline:
            seq_adam, val_adam = _adam_opt(seq_perm,
                                           n,
                                           sub_deadline,
                                           max_iters=300_000,
                                           lr0=6.0)
        else:
            seq_adam, val_adam = seq_perm, val_perm

        # c) optional L‑BFGS smoothing
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_adam,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1_000.0,
                                                              2_000.0, 5_000.0,
                                                              20_000.0, 50_000.0,
                                                              100_000.0, 200_000.0,
                                                              500_000.0, 1_000_000.0],
                                               max_iter_per_beta=2000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_adam, val_adam

        # Pick best among candidates
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_perm, val_perm),
            (seq_adam, val_adam),
            (seq_lbfgs, val_lbfgs)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ---------------------------------------------------------------
    # 5. Final long Adam polishing
    # ---------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long Adam polishing ---")
        final_seq, final_val = _adam_opt(global_best_seq,
                                         n,
                                         deadline,
                                         max_iters=2_500_000,
                                         lr0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved to {global_best_val:.6f}")

    # ---------------------------------------------------------------
    # 6. Official evaluation
    # ---------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

"""
Improved search for a low‑autocorrelation step function.

The algorithm works in three stages:

1. **Diverse seed pool** – a large collection of analytic families (uniform,
   triangular, sin‑window, Gaussian, power‑law, exponential, β‑shape, low‑frequency
   cosine) together with many random Dirichlet draws and random permutations.
   All seeds are ℓ₁‑normalised (sum = 1) and the best few are kept.

2. **Refinement of each top seed** – a simulated‑annealing permutation hill‑climb
   followed by a momentum‑based Adam mirror‑descent optimiser in log‑probability
   space (exact sub‑gradient of the active convolution entry).  An optional
   L‑BFGS pass on the smooth log‑sum‑exp surrogate can improve the result further.

3. **Final polishing** – the globally best sequence is run through a long Adam
   optimisation using the remaining time budget.

The best sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet distribution with symmetric concentration."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine components, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * math.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * math.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Permutation‑based hill climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two positions or perturbs the vector.
    Accepts worsening moves with a simple annealing schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # small Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam‑style mirror‑descent on the exact max‑convolution.
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              lr0: float = 5.0,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Optimise the surrogate objective 2·n·max(conv(a,a)) via Adam
    in log‑probability space (mirror‑descent).  Returns the best vector
    encountered and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = None
    best_val = float('inf')

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # max‑convolution and its sub‑gradient
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        grad_a = np.zeros_like(a)
        grad_a[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t. log‑probabilities
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update (learning‑rate decays with √t)
        lr = lr0 / math.sqrt(t)
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate new point
        a_new = _softmax(log_a)
        cur_val = _objective_surrogate(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()

    if best_seq is None:          # fallback (should never happen)
        best_seq = _softmax(log_a)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS smoothing (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt                     # pragma: no cover
except Exception:                                   # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)
