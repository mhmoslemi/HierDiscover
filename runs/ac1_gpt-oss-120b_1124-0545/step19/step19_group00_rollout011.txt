    best_seq = seq.copy()

                    best_val = new_val

def propose_candidate(seed=123, budget_s=1000):
    '''
    Simple hill‑climbing local search starting from the best known candidate.
    The algorithm repeatedly transfers a small amount of ℓ₁‑mass between two
    positions and keeps improvements.  By allocating the time budget among a
    few diverse seed vectors we avoid being trapped in a local minimum.
    '''
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety

    # problem size
    n = 4000

    # get previous best if any
    prev = globals().get("height_sequence_1")
    if prev is not None and len(prev) == n:
        base = np.asarray(prev, dtype=np.float64)
        base_sum = base.sum()
        if base_sum > 0:
            base = base / base_sum
        else:
            base = None
    else:
        base = None

    # define seed generators
    seeds = []

    if base is not None:
        seeds.append(base.copy())

    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))  # uniform

    # random Dirichlet seeds
    for conc in (0.05, 0.2, 0.5, 1.0):
        seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # keep only n-length seeds
    # evaluate surrogate and keep top few
    evaluated = []
    for s in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(s)  # we'll define similar
        evaluated.append((s, val))
    evaluated.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in evaluated[:min(4, len(evaluated))]]

    best_seq = None
    best_val = float('inf')

    # allocate time among seeds
    per_seed_deadline = deadline - 0.5
    # For each seed run hill climbing within remaining time divided.

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate part of remaining proportionally
        # simple: each gets roughly remaining/(remaining_seeds)
        # We'll compute a deadline for this run:
        run_deadline = time.time() + remaining / (len(top_seeds) - idx)

        seq, val = _hill_climb(seed_vec, run_deadline, start_sigma=0.4)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
        print(f"[seed {idx}] best surrogate after hill‑climb = {val:.6f}")

    # final evaluation
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate = {best_val:.6f}, true = {true_val:.6f}")
    except Exception as exc:
        print("evaluate_sequence raised", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

    n = len(seq)
    # ensure L1 normalized
    seq = seq.copy()
    s = seq.sum()
    if s <= 0:
        seq = np.full(n, 1.0/n)
    else:
        seq = seq / s

    best_seq = seq.copy()
    best_conv = _conv_fft(seq, seq)

    cur_seq = seq.copy()
    cur_val = best_val

    sigma = start_sigma
    rng = np.random.default_rng()
    no_improve = 0
    start_time = time.time()
    while time.time() < deadline:
        # pick i,j
        i = rng.integers(n)
        if cur_seq[i] == 0:

        j = rng.integers(n-1)

"""
Local‑search optimiser for the step‑function upper‑bound problem.

The algorithm proceeds as follows:

1. Gather a few seed vectors – the previously best known solution (if any),
   the uniform distribution, and a handful of random Dirichlet draws.
2. For each seed run a simple hill‑climbing / simulated‑annealing loop.
   A small amount of ℓ₁‑mass is transferred from one coordinate to another,
   the new vector is re‑normalised to sum = 1 and its exact max‑convolution
   is evaluated (via an FFT – O(n log n)).  Improvements are kept; if the
   search stalls a random perturbation of the current best is performed.
3. When time permits a final smooth optimisation using SciPy’s L‑BFGS‑B
   optimiser is carried out on a log‑sum‑exp surrogate of the max‑convolution.
4. The best sequence found is stored in the global variable
   ``height_sequence_1`` and returned as a plain list of floats.

All helpers are top‑level functions; no closures or lambda expressions are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used only for the final L‑BFGS polishing step)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = None
    _sp = None


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (O(n log n)).
    The result has length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum=1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                deadline: float,
                start_sigma: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Transfer a small amount of mass between two random coordinates,
    keep the move if it reduces the surrogate objective or with a tiny
    probability (simulated‑annealing).  When the search stalls a random
    perturbation of the best‑so‑far vector is performed.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    # ensure exact ℓ₁‑normalisation
    cur = start_seq.copy()
    s = cur.sum()
    if s <= 0.0:
        cur = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur = cur / s

    # current best
    best = cur.copy()
    best_val = _objective_surrogate(best)

    cur_val = best_val
    sigma = start_sigma
    no_improve = 0
    start_time = time.time()

    while time.time() < deadline:
        # pick source index (must have some mass)
        i = rng.integers(n)
        if cur[i] <= 0.0:
            continue
        # pick a different destination index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # amount to move – a random fraction of the source mass
        max_eps = cur[i]
        eps = rng.random() * max_eps * sigma
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        # clip numerical noise and re‑normalise (should stay at 1, but we keep safety)
        cand = np.clip(cand, 0.0, None)
        s2 = cand.sum()
        if s2 == 0.0:
            continue
        cand /= s2

        val_cand = _objective_surrogate(cand)

        if val_cand < cur_val - 1e-12:
            # deterministic improvement
            cur = cand
            cur_val = val_cand
            no_improve = 0
            if val_cand < best_val - 1e-12:
                best = cand.copy()
                best_val = val_cand
                sigma = max(1e-8, sigma * 0.97)      # tighten step size gradually
        else:
            # occasional uphill move – temperature decays linearly with time
            prog = (deadline - time.time()) / max(1e-12, deadline - start_time)
            temp = 1e-5 * prog
            if rng.random() < math.exp(-(val_cand - cur_val) / (temp + 1e-12)):
                cur = cand
                cur_val = val_cand
            no_improve += 1

        # if we have gone a long time without improvement, make a bigger random jump
        if no_improve >= 3000:
            perturb = rng.normal(scale=sigma, size=n)
            cand2 = best + perturb
            cand2 = np.clip(cand2, 0.0, None)
            s3 = cand2.sum()
            if s3 > 0.0:
                cand2 /= s3
                cur = cand2
                cur_val = _objective_surrogate(cand2)
            sigma = start_sigma
            no_improve = 0

    return best, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       beta_schedule=None,
                       max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0, 20000.0,
                         50000.0, 100000.0, 200000.0,
                         500000.0, 1000000.0]

    # work in log‑softmax space so the simplex is automatic
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numeric stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (softmax space)
            w = exp_beta / sum_exp            # weights for each entry of conv
            rev_a = a[::-1]
            # convolution of rev_a with w gives the derivative of each a_i
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The pipeline:

    1. Build a handful of seed vectors (previous best, uniform,
       several Dirichlet draws).
    2. Run a fast hill‑climbing / simulated‑annealing optimiser on each
       seed, respecting the overall wall‑time budget.
    3. If SciPy is available, finish with a continuation L‑BFGS polish.
    4. Return the best sequence found and store it globally.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 1.0

    n = 4000                                 # fixed problem size

    # --------------------------------------------------------------
    # 1. Gather seed vectors
    # --------------------------------------------------------------
    seeds = []

    # previous best (if any)
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a few random Dirichlet draws (different concentration parameters)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            d = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            seeds.append(d)                    # already sums to 1

    # --------------------------------------------------------------
    # 2. Hill‑climbing on each seed (time‑sliced)
    # --------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    num_seeds = len(seeds)

    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        # allocate a fair share of the remaining time to this seed
        remaining = deadline - time.time()
        remaining_seeds = num_seeds - idx
        seed_deadline = time.time() + remaining / max(1, remaining_seeds)

        seq, val = _hill_climb(seed_vec, seed_deadline, start_sigma=0.3)

        print(f"[seed {idx}] hill‑climb surrogate = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # --------------------------------------------------------------
    # 3. Optional L‑BFGS polishing (uses SciPy if present)
    # --------------------------------------------------------------
    if best_seq is not None and _opt is not None and time.time() < deadline:
        print("\n--- L‑BFGS polishing ---")
        seq_lb = _lbfgs_smooth_opt(best_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_val:
            best_seq = seq_lb.copy()
            best_val = val_lb
            print(f"[L‑BFGS] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final (official) evaluation and return
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform (should never happen)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
