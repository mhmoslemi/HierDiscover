    log_a = np.log(start_seq + 1e-30)  # keep log values.

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(start_seq)

    it_total = 0

    for beta in beta_schedule:
        for it_in in range(max_iters_per_beta):
            if time.time() > deadline: break
            it_total += 1
            a = _softmax(log_a)
            conv = _conv_fft(a, a)
            max_conv = conv.max()
            # smoothed weights
            exp_term = np.exp(beta * (conv - max_conv))
            sum_exp = exp_term.sum()
            if sum_exp == 0.0:
                w = np.full_like(conv, 1.0/len(conv))
            else:
                w = exp_term / sum_exp

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)  # length = len(rev_a)+len(w)-1 = n + (2n-1) -1 = 3n-2
            grad_a = 2.0 * full[n-1:n-1+n]

            # gradient wrt log
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            eta = eta0 / math.sqrt(it_total)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        if time.time() > deadline:
            break

    return best_seq, best_val

"""
A fast multi‑stage optimiser for the “max‑convolution” problem.

The goal is to find a non‑negative sequence a (sum(a)=1) that minimises

    2·n·max_k (a * a)[k]

where * denotes the linear convolution.  The optimum lies just above
2·n/(2·n‑1)≈1, while a uniform sequence gives 2.  The algorithm follows
three orthogonal ideas:

1. **Rich seed pool** – many analytic families (exponential,
   Gaussian, beta‑shape, sine‑windows, random Dirichlet, low‑frequency
   cosine sums, etc.) together with a few random permutations.

2. **Smooth exponentiated‑gradient descent** – instead of a hard sub‑
   gradient of the active index we minimise a smooth approximation
   `log‑sum‑exp` of the maximum.  The gradient is computed analytically
   in the soft‑max (log‑space) representation, and an Adam–style update
   is applied to the logits.  A schedule of increasing β values drives
   the optimisation from a very smooth landscape to the true max.

3. **Permutation hill‑climbing** – once a good shape has been found,
   a simple simulated‑annealing style local search swaps positions and
   perturbs values.  This exploits the fact that the convolution depends
   on the ordering of the heights.

The whole workflow respects the 1000 s budget, always keeping the
best sequence in the global variable ``height_sequence_1``.  The final
value returned is the raw list that can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_log_normal(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.lognormal(mean=0.0, sigma=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine components, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _permute(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a random permutation of the array."""
    p = np.arange(len(arr))
    rng.shuffle(p)
    return arr[p]


# ----------------------------------------------------------------------
# Smooth exponentiated‑gradient optimiser (Adam‑style)
# ----------------------------------------------------------------------
def _smooth_expgd_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule,
                     iters_per_beta: int = 1500,
                     eta0: float = 5.0,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Optimises the exact max‑convolution using a smooth‐max surrogate.
    The variables are stored as logits (log‑softmax).  An Adam‑like
    update is applied to these logits.
    """
    # initialise logits
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)          # first‑moment
    v = np.zeros_like(log_a)          # second‑moment
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(start_seq)

    it_total = 0

    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            it_total += 1

            a = _softmax(log_a)
            conv = _conv_fft(a, a)               # length = 2n‑1
            max_c = np.max(conv)

            # smooth‑max weights (softmax over β·conv)
            exp_term = np.exp(beta * (conv - max_c))
            sum_exp = exp_term.sum()
            if sum_exp == 0.0:
                w = np.full_like(conv, 1.0 / len(conv))
            else:
                w = exp_term / sum_exp

            # gradient w.r.t. a : 2·∑_k w_k·a shifted
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. logits (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update in log‑space
            eta = eta0 / math.sqrt(it_total)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            # bias‑corrected moments
            m_hat = m / (1.0 - beta1)
            v_hat = v / (1.0 - beta2)

            step = eta / (np.sqrt(v_hat) + eps)
            log_a = log_a - step * m_hat

            # keep track of the best surrogate seen so far
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

        if time.time() > deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two positions or perturbs the vector.
    Accepts worsening moves with a simple annealing schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (linear)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move – most common
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur_seq.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue
        else:                                   # tiny gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # accept worse move with probability exp(−Δ/T)
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine builds a large pool of seeded
    shapes, selects the most promising few, refines each by a smooth
    exponentiated‑gradient optimiser followed by permutation hill‑climbing,
    and finally carries out a long polishing run on the best candidate.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – a compromise between expressivity and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (helps continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Massive seed generation (analytic families + random draws)
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.1, 3.5, 20):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma_frac in np.linspace(0.05, 0.35, 20):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.0, 4.0, 20):
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.3, 40):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # random beta shapes
    for _ in range(80):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws – wide range of concentrations
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(12):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random log‑normal draws
    for _ in range(15):
        seeds.append(_random_log_normal(n, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(7):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform and exponential decays
    for alpha in (0.2, 0.5, 0.8):
        for rate in (0.001, 0.005, 0.02):
            curve = _exp_decay(n, rate)
            mixed = (1.0 - alpha) * _uniform(n) + alpha * curve
            mixed /= mixed.sum()
            seeds.append(mixed)

    # random permutations of a few promising bases
    base_shapes = [_exp_decay(n, 0.003), _sym_exp_decay(n, 0.004),
                   _beta_shape(n, 2.0, 2.0), _gaussian(n, 0.18)]
    for base in base_shapes:
        for _ in range(5):
            seeds.append(_permute(base, rng))

    # include previous best if available
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            # small perturbation of the previous best
            noisy = prev_arr + 0.004 * rng.normal(size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0.0:
                seeds.append(noisy / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds – keep the top‑k
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (smooth EG + permutation hill‑climb)
    # ------------------------------------------------------------------
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 1_000.0, 2_000.0,
                     5_000.0, 20_000.0, 50_000.0,
                     100_000.0, 200_000.0, 500_000.0,
                     1_000_000.0]

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # allocate roughly 15 % of remaining time per remaining seed
        share = remaining * 0.15 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # ---- (a) smooth exponentiated‑gradient optimisation ----
        seq_smooth, val_smooth = _smooth_expgd_opt(seed_vec,
                                                   n,
                                                   sub_deadline,
                                                   beta_schedule,
                                                   iters_per_beta=1800,
                                                   eta0=6.0)

        # ---- (b) permutation‑based hill‑climbing ----
        if time.time() < sub_deadline:
            seq_hill, val_hill = _perm_hill_climb(seq_smooth,
                                                  n,
                                                  sub_deadline,
                                                  rng,
                                                  max_iters=180_000)
        else:
            seq_hill, val_hill = seq_smooth, val_smooth

        # Choose the best of the two refinements (plus original seed)
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_smooth, val_smooth),
            (seq_hill, val_hill)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing on the global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long smooth EG polishing ---")
        final_seq, final_val = _smooth_expgd_opt(global_best_seq,
                                                n,
                                                deadline,
                                                beta_schedule,
                                                iters_per_beta=3000,
                                                eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
