def _adam_optimize(start_vec, n, deadline, beta_schedule=None, init_lr=0.01, max_iter_per_beta=2000):
    # start in log space
    log_a = np.log(start_vec + 1e-30)
    m_t = np.zeros_like(log_a)
    v_t = np.zeros_like(log_a)
    t = 0
    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)

    if beta_schedule is None:
        beta_schedule = [1,5,20,80,200,500,1000,2000,5000,20000,50000,100000,200000,500000,1000000]

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        for i in range(max_iter_per_beta):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(log_a)
            # conv via FFT
            conv = _conv_fft(a, a)
            max_val = conv.max()  # shift
            exp_beta = np.exp(beta_val * (conv - max_val))
            sum_exp = exp_beta.sum()
            # smooth max value
            smooth_max = max_val + np.log(sum_exp) / beta_val
            # compute gradient
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)  # length = 2n-1 ?
            grad_a = 2.0 * full[n-1:n-1+n]   # n elements

            # chain rule to log space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            m_t = 0.9 * m_t + 0.1 * grad_log
            v_t = 0.999 * v_t + 0.001 * (grad_log ** 2)
            m_hat = m_t / (1 - 0.9**t)
            v_hat = v_t / (1 - 0.999**t)

            # compute exact surrogate objective using max conv
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

"""
Improved search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the full linear convolution.  The routine

* builds a large, very diverse pool of seeds (analytic windows,
  Dirichlet draws, exponential draws, low‑frequency cosine sums **and**
  *binary sparse* patterns with many exact zeros),
* evaluates them cheaply with an FFT‑based surrogate,
* keeps the best few,
* refines each of them with a custom Adam optimiser that works in
  log‑softmax space on a smooth log‑sum‑exp approximation of the
  ``max``‑convolution,
* finally runs a long polishing phase and returns the best sequence
  found.

All work respects the global time budget (default 1000 s).  The best
sequence is stored in the global variable ``height_sequence_1`` for
future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    a = np.hamming(n).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _binary_sparse_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Create a vector with exactly `m` equal non‑zero entries (value 1/m)."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=m, replace=False)
    a[idx] = 1.0 / m
    return a


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalize."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    return noisy / s if s > 0 else seq.copy()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Adam optimiser on a smooth log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   init_lr: float = 0.02,
                   max_iter_per_beta: int = 2000) -> tuple[np.ndarray, float]:
    """
    Optimise the smoothed max‑convolution objective using Adam in
    log‑softmax space.  ``start_vec`` must be ℓ₁‑normalised.
    Returns the best vector seen (ℓ₁‑normalised) and its exact surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80, 200, 500, 1000,
                         2000, 5000, 20000, 50000, 100000,
                         200000, 500000, 1_000_000]

    # initialise log‑softmax parameters
    log_a = np.log(start_vec + 1e-30)
    m_t = np.zeros_like(log_a)
    v_t = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(max_iter_per_beta):
            if time.time() > deadline:
                break

            t_iter += 1
            a = _softmax(log_a)

            # --- smoothed max of the convolution ---------------------------------
            conv = _conv_fft(a, a)                 # length 2n‑1
            shift = np.max(conv)                   # = exact max (for stability)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth max (not needed for the gradient but kept for completeness)
            _ = shift + np.log(sum_exp) / beta_val

            # --- gradient w.r.t. a (log‑sum‑exp) ---------------------------------
            w = exp_beta / sum_exp                 # soft‑weights for each index of conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)             # length 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]    # slice of length n

            # chain‑rule to log‑softmax parameters
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # --- Adam update ------------------------------------------------------
            m_t = beta1 * m_t + (1 - beta1) * grad_log
            v_t = beta2 * v_t + (1 - beta2) * (grad_log ** 2)
            m_hat = m_t / (1 - beta1 ** t_iter)
            v_hat = v_t / (1 - beta2 ** t_iter)

            # learning‑rate scaling (smaller steps for large beta)
            lr = init_lr / math.sqrt(beta_val)

            log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # --- evaluate exact surrogate objective --------------------------------
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a small safety margin

    # --------------------------------------------------------------
    # Problem size – a power of two makes the FFT cheap
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Load a previously found solution, if any
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr.copy()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.   Very diverse seeding
    # --------------------------------------------------------------
    print(f"\n--- Generating seed pool (n={n}) ---")
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_tukey_window(n, 0.75))

    for gamma in np.linspace(0.1, 4.0, 20):
        seeds.append(_sine_window(n, gamma))
    for sigma_frac in np.linspace(0.05, 0.45, 20):
        seeds.append(_gaussian(n, sigma_frac))
    for p in np.linspace(0.0, 5.0, 20):
        seeds.append(_power(n, p))
    for rate in np.logspace(-4, -0.4, 15):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Beta‑shape random sampling
    for _ in range(30):
        a = rng.uniform(0.05, 8.0)
        b = rng.uniform(0.05, 8.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws (very sparse for small concentration)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(20):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6, 8):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse binary seeds – the key new ingredient
    print("--- Adding binary sparse seeds ---")
    for m in (50, 70, 90, 110, 130, 150, 170, 190):
        for _ in range(5):
            seeds.append(_binary_sparse_seed(n, m, rng))

    # Perturbations of the previously best sequence (if present)
    if prev_arr is not None:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(6):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # --------------------------------------------------------------
    # 2.   Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3.   Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4.   Multi‑start refinement (Adam optimiser)
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a percentile of the remaining time for this seed
        remaining = deadline - time.time()
        # keep ~10 % for each of the remaining seeds + final polish
        sub_deadline = time.time() + remaining * 0.1 / max(1, (top_k - si))

        seq_opt, val_opt = _adam_optimize(seed_vec,
                                           n,
                                           sub_deadline,
                                           beta_schedule=[1, 5, 20, 80,
                                                          200, 500, 1000,
                                                          2000, 5000, 20000,
                                                          50000, 100000,
                                                          200000, 500000,
                                                          1_000_000],
                                           init_lr=0.02,
                                           max_iter_per_beta=1500)

        # keep the best result from this seed
        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"[Seed {si+1}/{top_k}] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5.   Final long polishing (more Adam iterations)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing with Adam ---")
        final_seq, final_val = _adam_optimize(global_best_seq,
                                               n,
                                               deadline,
                                               init_lr=0.015,
                                               max_iter_per_beta=3000)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Polish] improved to surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6.   Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
