"""
Propose a step‑function of non‑negative heights that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation follows a multi‑stage strategy:

1.  Very diverse seed generation (analytic families, Dirichlet draws,
    exponential draws, low‑frequency cosine curves, permutations and
    perturbations of a previous solution).

2.  Fast surrogate evaluation (the objective with the ℓ₁‑norm forced to 1)
    to keep the best *k* seeds.

3.  Stochastic exponentiated‑gradient (mirror‑descent) on the **exact**
    max‑convolution, selecting a *single* active index per iteration.
    This allows many more iterations inside the time budget.

4.  A long simulated‑annealing phase that mixes global Gaussian perturbations
    with targeted mass‑transfer moves between two random coordinates.

5.  Optional short smoothing passes (moving‑average kernels) and a final
    short EG polish.

All heavy work respects the global deadline (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1`` and
returned as a list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i), i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    a = np.hamming(n).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Stochastic exponentiated‑gradient optimisation (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt_stochastic(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 rng: np.random.Generator,
                                 max_iters: int = 1_500_000,
                                 eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    At each iteration a *single* active index (where the convolution attains
    its maximum) is chosen uniformly at random, which yields a cheap
    stochastic sub‑gradient and permits many more updates inside the time
    budget.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]
        # random active index
        k = int(active[rng.integers(len(active))])

        # build gradient for this k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # slice assignment
        grad = np.zeros_like(a)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        grad[idx] = a[rev]
        grad *= 2.0  # factor 2 from symmetry

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing with two move types
# ----------------------------------------------------------------------
def _simulated_annealing_extended(start_seq: np.ndarray,
                                  n: int,
                                  deadline: float,
                                  rng: np.random.Generator,
                                  max_attempts: int = 5_000_000,
                                  start_sigma: float = 0.004,
                                  T_start: float = 0.01,
                                  T_end: float = 1e-6):
    """
    Lightweight SA that combines:
      * global Gaussian perturbations,
      * directed mass‑transfer between two random coordinates.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    for attempt in range(max_attempts):
        if time.time() >= deadline:
            break

        # linear cooling
        T = T_start * (1.0 - attempt / max_attempts) + T_end * (attempt / max_attempts)

        if rng.random() < 0.5:
            # Gaussian perturbation of the whole vector
            sigma = start_sigma * (T / T_start)
            ns = a + sigma * rng.normal(size=n)
        else:
            # Transfer a small fraction of mass from i to j
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            # move up to 5 % of a[i]
            delta = a[i] * rng.uniform(0.0, 0.05)
            ns = a.copy()
            ns[i] -= delta
            ns[j] += delta

        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta_val = val - cur_val
        if delta_val < 0.0 or rng.random() < math.exp(-delta_val / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
    return best_seq, best_val

    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_tukey_window(n, 0.75))

    # Sine windows
    for gamma in np.linspace(0.05, 5.0, 30):
        seeds.append(_sine_window(n, gamma))

    # Gaussian windows
    for sigma_frac in np.linspace(0.02, 0.5, 30):
        seeds.append(_gaussian(n, sigma_frac))

    # Power‑law families
    for p in np.linspace(0.0, 6.0, 30):
        seeds.append(_power(n, p))

    # Exponential decays
    for rate in np.logspace(-5, -0.3, 30):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Beta‑shape draws (broad range)
    for a in np.linspace(0.1, 10.0, 10):
        for b in np.linspace(0.1, 10.0, 10):
            seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws (including very sparse)
    for conc in (0.0005, 0.001, 0.005, 0.01, 0.05, 0.2, 1.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Exponential draws
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random cosine curves
    for nf in (2, 3, 5, 7, 9):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixes & permutations of a few bases
    bases = [_gaussian(n, 0.22), _sine_window(n, 1.2), _tukey_window(n, 0.6)]
    for base in bases:
        for _ in range(4):
            seeds.append(np.random.permutation(base))
        for mix_alpha in (0.2, 0.5, 0.8):
            curve = _lowfreq_curve(n, 4, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Perturb previous best if we have one
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # Normalise & prune seeds (fast surrogate evaluation)
    # ------------------------------------------------------------------
    norm_seeds = []
    for vec in seeds:
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        norm_seeds.append(vec / s)

    evaluated = []
    for vec in norm_seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 2. Global best initialisation
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Multi‑start stochastic EG refinement
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining time for this seed
        remaining = deadline - time.time()
        # keep ~15 % of the remaining budget for each upcoming seed
        sub_deadline = time.time() + min(remaining * 0.15 / max(1, (top_k - si)),
                                         20.0)

        seq_opt, val_opt = _exp_gradient_opt_stochastic(seed_vec.copy(),
                                                        n,
                                                        sub_deadline,
                                                        rng,
                                                        max_iters=1_500_000,
                                                        eta0=6.0)

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"[Seed {si+1}/{top_k}] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long simulated‑annealing pass
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Extensive Simulated Annealing ---")
        seq_sa, val_sa = _simulated_annealing_extended(global_best_seq.copy(),
                                                       n,
                                                       deadline,
                                                       rng,
                                                       max_attempts=5_000_000,
                                                       start_sigma=0.004,
                                                       T_start=0.01,
                                                       T_end=1e-6)
        if val_sa < global_best_val:
            global_best_seq = seq_sa
            global_best_val = val_sa
            print(f"[SA] Improved global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional short smoothing passes
    # ------------------------------------------------------------------
    for ksize in range(2, 5):
        if time.time() >= deadline:
            break
        kernel = np.ones(ksize, dtype=np.float64) / ksize
        smooth = np.convolve(global_best_seq, kernel, mode='same')
        smooth = np.maximum(smooth, 0.0)
        sm_sum = smooth.sum()
        if sm_sum == 0.0:
            continue
        smooth /= sm_sum
        val_smooth = _objective_surrogate(smooth)
        if val_smooth < global_best_val:
            global_best_seq = smooth
            global_best_val = val_smooth
            print(f"[Smooth k={ksize}] Improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final short EG polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_final, val_final = _exp_gradient_opt_stochastic(global_best_seq.copy(),
                                                            n,
                                                            deadline,
                                                            rng,
                                                            max_iters=300_000,
                                                            eta0=5.0)
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final
            print(f"[Final EG] Improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
