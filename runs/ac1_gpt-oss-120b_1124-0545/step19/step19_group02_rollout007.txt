def _adam_optimize(start_seq, n, deadline, max_iters=300000, lr=0.05):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(log_a)
        # compute subgradient at max conv
        grad_a, _, _ = _subgrad(a)
        # compute gradient w.r.t log_a
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log * grad_log)

        log_a -= step
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

    log_a = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = start_val

                best_seq = cand_seq.copy()

"""
Propose a candidate step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm:

1. Generate a rich set of seeds (analytic windows, Dirichlet draws,
   exponential / logistic decays, etc.) and keep the best few
   according to the surrogate objective 2·n·max(conv) (sum is forced to 1).

2. Refine each selected seed with a custom Adam optimiser that works on
   the log‑softmax representation and uses the exact sub‑gradient of the
   max‑convolution (the active convolution entry).  Adam provides an
   adaptive learning‑rate while staying on the probability simplex.

3. Finish with a simulated‑annealing pass on the log‑probability vector
   to escape possible local minima.  Proposals are Gaussian perturbations
   in the log‑space; acceptance follows the Metropolis rule with a
   linearly decaying temperature.

4. The best sequence found is evaluated with the official
   ``evaluate_sequence`` function and stored globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (probability vector, sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate objective (sum(seq)==1): 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                 # NumPy provides the modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative, renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    return noisy / s if s > 0 else seq.copy()


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution
# ----------------------------------------------------------------------
def _subgrad(seq: np.ndarray):
    """
    Returns (g, k, conv_k) where g = ∇_a max(conv) at the active index k.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    k = int(np.argmax(conv))
    grad = np.zeros_like(seq)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=int)
    grad[i] = 2.0 * seq[k - i]
    return grad, k, conv[k]


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (log‑softmax variables)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 500_000,
                   lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Adam on log‑softmax variables using the exact sub‑gradient of max(conv).
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    best_seq = _softmax(log_a)
    best_val = _objective(best_seq)
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(log_a)

        # exact sub‑gradient at the current active convolution entry
        grad_a, _, _ = _subgrad(a)

        # gradient wrt log‑softmax variables
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log * grad_log)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # evaluate new point
        a_new = _softmax(log_a)
        cur_val = _objective(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing on log‑probability vector (exploratory polishing)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         start_val: float,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple SA in log‑space; proposals are Gaussian perturbations.
    Temperature and step‑size decay linearly with the fraction of time spent.
    """
    rng = np.random.default_rng()
    log_a = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = start_val
    best_seq = cur_seq.copy()
    best_val = cur_val
    start_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        if total <= 0:
            break
        prog = elapsed / total                     # 0 → 1
        T = max(1e-9, 1.0 - prog)                  # linear cooling
        sigma = 0.5 * (1.0 - prog) + 0.001          # step‑size shrink

        cand_log = log_a + sigma * rng.standard_normal(size=log_a.shape)
        cand_seq = _softmax(cand_log)
        cand_val = _objective(cand_seq)

        # Metropolis acceptance
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / T):
            log_a = cand_log
            cur_seq = cand_seq
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand_seq.copy()

    return best_seq, best_val

        seeds.append(prev_arr)
        for _ in range(3):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2️⃣  Quick surrogate evaluation and keep the top few seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective(vec)
        evaluated.append((vec, val))
    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective(best_seq)
    print(f"[Init] Best surrogate among {len(evaluated)} seeds: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Refine the best seeds with Adam + Simulated Annealing
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds[:3]):
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining budget to this seed
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.35 / (3 - idx)

        print(f"\n[Refine #{idx+1}] start surrogate { _objective(seed_vec):.6f}")

        # Adam optimisation on the exact objective
        seq_adam, val_adam = _adam_optimize(seed_vec, n, sub_deadline,
                                            max_iters=800_000, lr=0.07)

        # SA polishing from the Adam result
        seq_sa, val_sa = _simulated_annealing(seq_adam, val_adam, n, sub_deadline)

        # pick the best among raw seed, Adam, SA
        candidates = [
            (seed_vec, _objective(seed_vec)),
            (seq_adam, val_adam),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < best_val:
            best_val = local_best_val
            best_seq = local_best_seq.copy()
            print(f"  >>> Updated global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final long simulated‑annealing polishing (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final simulated‑annealing polishing ---")
        best_seq, best_val = _simulated_annealing(best_seq, best_val, n, deadline)
        print(f"[Final SA] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
