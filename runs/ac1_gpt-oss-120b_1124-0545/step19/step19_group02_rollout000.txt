"""
Propose a candidate step‑function that (hopefully) improves the current best
value ~1.5085 towards the target 1.503.

Algorithm
---------

1. **Initial seeds** – a tiny pool of simple shapes (uniform, previous best,
   a few Dirichlet draws and tiny perturbations).  The best of these seeds
   gives a starting point.

2. **Global search in a low‑dimensional parametric family** – we model a
   distribution as a mixture of a few Gaussian windows

        a_i ∝ Σ_k w_k · exp( – (i/(n‑1) – μ_k)² / (2·σ_k²) ) ,

   with 5 components (15 free parameters).  ``scipy.optimize.differential_evolution``
   (or a pure random search if SciPy is unavailable) minimises the surrogate
   objective ``2·n·max(conv(a,a))`` on the simplex.  The best mixture is
   added to the candidate pool.

3. **Smooth optimisation** – the surrogate is smoothed with a log‑sum‑exp
   approximation and optimised by L‑BFGS‑B in the log‑softmax space.
   A generous schedule of β values (1 → 10⁶) is used.

4. **Exact mirror‑descent polishing** – an exponentiated‑gradient (mirror‑descent)
   method minimises the true max‑convolution objective.  Up to a few
   million iterations are performed while respecting the global time budget.

5. **Optional final L‑BFGS fine‑tuning** – after the exact polishing we run a
   short L‑BFGS pass to remove any remaining smoothing bias.

All stages respect the overall 1000 s limit; the best sequence seen so far
is continuously updated.  The final sequence is stored in the global variable
``height_sequence_1`` and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the global DE optimiser)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a probability vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space.
    """
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # Abort if we run out of time.
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Parametric mixture (Gaussian components)
# ----------------------------------------------------------------------
def _mixture_gaussian(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a probability vector from ``params``.
    params layout (K components):
        w_0 … w_{K-1}      (non‑negative, unrestricted)
        μ_0 … μ_{K-1}      (in [0,1])
        σ_0 … σ_{K-1}      (positive)
    The resulting vector is normalised to sum to 1.
    """
    K = 5
    w = np.maximum(params[0:K], 0.0)               # enforce non‑negativity
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    # Normalise the weights – they only control relative contribution.
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)   # normalised coordinate in [0,1]
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s
