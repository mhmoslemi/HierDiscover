"""
Propose a candidate step‐function (non‑negative heights) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
1.  Gather a starting point: either the previous best global variable
    ``height_sequence_1`` (if it exists) or the uniform distribution.
2.  Define the *target* convolution – the uniform probability vector of
    length ``2*n-1`` (the average possible value of the auto‑convolution).
    Achieving a convolution close to uniform forces the maximum entry to be
    close to the average, which lowers the final objective.
3.  Perform a large‑scale mirror‑descent / Adam optimisation on the *L2*
    distance between the current convolution and the target.  The gradient
    is obtained analytically via two FFT‑based convolutions, and each iterate
    is projected back onto the probability simplex (ℓ₁‑norm = 1,
    non‑negative).
4.  Keep the best sequence seen (lowest surrogate ``2*n*max(conv)``) and,
    after the time budget, return it.  The true objective is printed for
    reference and stored globally as ``height_sequence_1`` for the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast and accurate)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """
    Surrogate objective for a ℓ₁‑normalised vector (sum = 1):
    2·n·max(conv(seq, seq)).
    """
    conv = _conv_full_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on L2 distance to the uniform convolution
# ----------------------------------------------------------------------
def _adam_optimize(init: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 200_000,
                    lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    minimise ‖conv(a,a) – u‖₂²  (u = uniform vector) with Adam,
    keeping a on the probability simplex.

    Returns the best sequence seen and its surrogate value.
    """
    # target: uniform distribution over 2n‑1 entries (the average of the convolution)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    a = init.copy()
    # safety: avoid exact zeros which can lead to numerical stagnation
    a = np.clip(a, 1e-30, None)
    a /= a.sum()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- convolution and error ---
        conv = _conv_full_fft(a, a)          # length 2n‑1
        err = conv - target                  # same length

        # --- gradient of ½‖err‖² w.r.t. a ---
        # ∂/∂a (a*a) = 2 * (a convolved with reversed a)
        # for the L2 error the factor becomes 4.
        rev_a = a[::-1]
        grad_full = _conv_full_fft(err, rev_a)      # length 3n‑2
        start = n - 1
        grad = 4.0 * grad_full[start:start + n]

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _proj_simplex(a)                 # ensure non‑negativity & ℓ₁‑norm = 1

        # --- keep track of the best surrogate value ---
        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional progress report
        if it % 5000 == 0:
            print(f"[Adam] iter {it:6d}  best surrogate ≈ {best_val:.6f}")

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method builds a distribution whose
    self‑convolution is as close as possible to a uniform vector, which
    drives the maximum convolution entry towards its theoretical minimum.
    """
    rng = np.random.default_rng(seed)
    t_start = time.time()
    # keep a small safety margin for the final evaluation
    deadline = t_start + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – we stay with 4000 (as used in previous attempts)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Retrieve the previous solution (if any) and normalise it.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # 2. Build a small pool of initial candidates.
    # ------------------------------------------------------------------
    init_pool = []

    # a) reuse the previous best if it exists
    if prev_arr is not None and prev_arr.size == n:
        init_pool.append(prev_arr / prev_arr.sum())

    # b) the uniform distribution (a known baseline)
    init_pool.append(np.full(n, 1.0 / n, dtype=np.float64))

    # c) a few random Dirichlet draws – these give diverse starting points
    for _ in range(3):
        init_pool.append(rng.dirichlet(np.full(n, 1.0)).astype(np.float64))

    # ------------------------------------------------------------------
    # 3. Quick surrogate evaluation of the pool – keep the best one.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for cand in init_pool:
        cand = cand / cand.sum()               # enforce ℓ₁‑norm = 1
        val = _objective_surrogate(cand, n)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
    print(f"Initial surrogate best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation: Adam on the L2 distance to the uniform convolution.
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 10.0:                 # ensure we have enough budget for a pass
        seq_opt, val_opt = _adam_optimize(best_seq, n,
                                          deadline=deadline,
                                          max_iters=300_000,
                                          lr=0.02)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
        print(f"After Adam optimisation best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Official evaluation with the provided ``evaluate_sequence``.
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the best sequence globally for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
