"""
Propose a step‑function (non‑negative height sequence) that minimises
evaluate_sequence(sequence) within the 1000 s budget.

The algorithm follows three complementary optimisation phases:
    1️⃣  A large, diverse seed pool (deterministic families + random Dirichlet
        / log‑normal draws) is built and the best few are refined with
        continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the max‑convolution.
    2️⃣  Mirror‑descent (exponentiated‑gradient) on the *exact* objective is run
        from many random restarts as well as from the current best point.
    3️⃣  A final polishing stage (high‑β L‑BFGS) and a short simulated‑annealing
        mass‑transfer hill‑climb are performed.

All steps respect the wall‑clock deadline and the best sequence found is stored
in the global variable ``height_sequence_1`` for warm‑starts in later calls.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.fft as _fft
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# FFT‑based convolution (full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = _fft.rfft(x, n=N)
    fy = _fft.rfft(y, n=N)
    conv = _fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Problem‑specific helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²  (scale‑invariant)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    The active lag is the index where the convolution attains its maximum.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))
    # Build gradient: for each i participating in the active lag,
    #   ∂/∂a_i  Σ a_i a_{k-i} = a_{k-i} + a_{k-i} (if i≠k‑i) or 2 a_i (if i=j)
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    j_range = k - i_range
    # i_range and j_range are the two indices that multiply.
    # For i==j we add 2*a_i, else we add a_j (once, because the sum already
    # contains both i,j pairs – this matches the derivative of the *sum* form).
    same = i_range == j_range
    g[i_range] += a[j_range]
    g[j_range] += a[i_range]
    # correct double‑count for the diagonal terms
    g[same] = 2.0 * a[i_range[same]]
    return g


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – should not happen for non‑negative inputs
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _lognormal(n: int, rng: np.random.Generator) -> np.ndarray:
    a = np.exp(rng.normal(0.0, 1.0, size=n))
    return a / a.sum()


def _truncated_normal(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.normal(0.0, 1.0, size=n)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(logits: np.ndarray, n: int, beta: float):
    """Log‑sum‑exp surrogate of max‑convolution."""
    a = _softmax(logits)                 # ℓ₁‑normalised vector
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # subtraction for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (from the smooth max)
    w = exp_term / sum_exp               # soft weights over lags
    rev_a = a[::-1]
    # convolution rev_a * w gives the gradient before the softmax back‑prop
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through the softmax
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return obj, grad_logits


def _lbfgs_continuation(start_vec: np.ndarray,
                        n: int,
                        deadline: float,
                        beta_sched: list[float] | None = None,
                        iters_per_beta: int = 3000) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_sched is None:
        beta_sched = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0,
        ]

    logits = np.log(start_vec + 1e-30)   # unrestricted parameters

    for beta in beta_sched:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v):
            return _smooth_objective(v, n, beta_val)

        def cb(v):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun,
                                x0=logits,
                                method="L-BFGS-B",
                                jac=True,
                                callback=cb,
                                options={"maxiter": iters_per_beta,
                                         "disp": False,
                                         "ftol": 1e-12})
            logits = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break
    return _softmax(logits)


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) on the true objective
# ----------------------------------------------------------------------
def _mirror_descent(start_a: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 300_000,
                    eta0: float = 1.5) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        # mirror‑step in log‑space
        log_a = np.log(a + 1e-30) - eta * g
        a = np.exp(log_a)
        a /= a.sum()

        if it % 50_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer hill‑climb
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 250_000,
                         init_temp: float = 2.0) -> tuple[np.ndarray, float]:
    a = start_a.copy()
    cur_val = _objective_exact(a)
    best_a = a.copy()
    best_val = cur_val

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        # move a small fraction (up to 2 %) from i to j
        frac = rng.random() * 0.02
        delta = frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = _objective_exact(cand)
        if cand_val < cur_val:
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_a = cand.copy()
                best_val = cand_val
        else:
            progress = it / max_iters
            T = init_temp * max(0.05, 1.0 - progress)
            if rng.random() < np.exp(-(cand_val - cur_val) / T):
                a = cand
                cur_val = cand_val
    return best_a, best_val

# Seed‑pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:

    # ----- deterministic families -------------------------------------------------
    seeds.append(_uniform(n))

    seeds.append(_truncated_normal(n, rng))

    # ----- power‑law sweep --------------------------------------------------------
    for p in np.linspace(0.30, 0.80, 31):
        seeds.append(_power_law(n, float(p)))

    # ----- random Dirichlet -------------------------------------------------------
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ----- random sparse ----------------------------------------------------------
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # ----- mixtures & rotations ---------------------------------------------------
    base = np.arange(len(seeds))
    for _ in range(120):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    for _ in range(20):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # ----- warm‑start -------------------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # ----- final normalisation ----------------------------------------------------
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function obeys the 1000 s wall‑clock limit and stores the best sequence
    in the global ``height_sequence_1`` for warm‑starts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # ------------------------------------------------------------------
    # problem size (can be tuned – larger n gives a tighter theoretical lower bound)
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # seed pool (includes warm‑start if available)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # evaluate seeds, keep the elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    seeds.sort(key=_objective_exact)
    elite = seeds[:min(30, len(seeds))]   # a slightly larger elite set

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0,
    ]

    print("\n--- Phase 1: L‑BFGS smooth optimisation ---")
    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing Phase 1.")
            break
        print(f" L‑BFGS run {idx + 1}/{len(elite)} ...")
        seq_opt = _lbfgs_continuation(seed_vec,
                                      n,
                                      deadline,
                                      beta_sched=beta_schedule,
                                      iters_per_beta=3000)
        val_opt = _objective_exact(seq_opt)
        print(f"  → objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"   ** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multiple mirror‑descent (exponentiated‑gradient) restarts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 16
        # allocate roughly a constant budget per restart but never exceed the remaining time
        iters_per_restart = int(min(350_000, remaining * 0.07 / restarts))
        print("\n--- Phase 2: Mirror‑descent restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            # random Dirichlet start
            conc = rng.uniform(0.01, 1.2)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_md, val_md = _mirror_descent(start_vec,
                                             n,
                                             deadline,
                                             max_iters=iters_per_restart,
                                             eta0=1.8)
            print(f" MD {r + 1}/{restarts}: best = {val_md:.6f}")
            if val_md < global_best_val:
                global_best_seq = seq_md.copy()
                global_best_val = val_md
                print(f"   ** Updated global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long mirror‑descent from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(7_000_000, remaining * 0.30 * 1e5))
        if max_iters > 0:
            print("\n--- Phase 3: Long mirror‑descent from best ---")
            seq_md, val_md = _mirror_descent(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_iters,
                                             eta0=2.2)
            print(f" Long MD best = {val_md:.6f}")
            if val_md < global_best_val:
                global_best_seq = seq_md.copy()
                global_best_val = val_md
                print(f"   ** New global best after long MD = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(350_000, remaining * 150))
        if max_iters > 0:
            print("\n--- Phase 4: Simulated‑annealing hill‑climb ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                                  n,
                                                  deadline,
                                                  rng,
                                                  max_iters=max_iters,
                                                  init_temp=2.0)
            print(f" SA best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"   ** New global best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Phase 5: Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_continuation(global_best_seq,
                                        n,
                                        deadline,
                                        beta_sched=[500_000.0,
                                                    1_000_000.0,
                                                    2_000_000.0,
                                                    5_000_000.0,
                                                    10_000_000.0],
                                        iters_per_beta=2000)
        val_final = _objective_exact(seq_final)
        print(f" Final L‑BFGS objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"   ** New global best after final polish = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for future warm‑starts
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
