def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, numpy as np, scipy.optimize as _opt, scipy.sparse as _sp

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    n = 4000

    # Build seed list
    seeds = []
    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, mu=n/2, sigma=n/5))
    seeds.append(_raised_cosine(n, width=n*0.8))
    # exponential decays
    for lam in [0.0005, 0.001, 0.002, 0.005]:
        seeds.append(_exponential_decay(n, lam))
    # power law exponents
    for p in np.linspace(0.3, 1.0, 15):
        seeds.append(_power_law(n, p))
    # beta shapes grid
    for alpha in [0.3,0.5,0.8,1.0,1.5,2.0,3.0]:
        for beta in [0.3,0.5,0.8,1.0,1.5,2.0,3.0]:
            seeds.append(_beta_shape(n, alpha, beta))
    # sinusoidal and arcsine
    seeds.append(_sinusoidal(n))
    seeds.append(_arcsine(n))

    # random Dirichlet
    for conc in [0.02,0.05,0.1,0.2,0.5,1.0]:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse
    for frac in [0.01,0.02,0.05,0.1]:
        for _ in range(3):
            seeds.append(_random_sparse(n, frac, rng))

    # low freq cos
    for nf in [2,3,4]:
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mix random seeds
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotate some seeds
    for _ in range(30):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # Normalise and clean
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    seeds = cleaned

    # Evaluate seeds
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] Best seed objective = {best_val:.6f}")

    # Keep elite top_k
    top_k = min(30, len(seeds))
    seeds.sort(key=_objective_exact)
    elite = seeds[:top_k]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # Stage L-BFGS on elite
    beta_sched = [1.0,5.0,20.0,80.0,200.0,500.0,1000.0,2000.0,5000.0,1e4,2e4,5e4,1e5,2e5,5e5,1e6]
    # we may add higher later
    per_beta = 800
    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] Deadline before completing LBFGS on elite.")
            break
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline, beta_schedule=beta_sched, max_iter_per_beta=per_beta)
        val_opt = _objective_exact(seq_opt)
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[LBFGS {idx+1}/{top_k}] New global best = {global_best_val:.6f}")

    # Stage 2: exponenial gradient restarts (few)
    remaining = deadline - time.time()
    if remaining > 5.0:
        restarts = 6
        iters_per_restart = int(min(300_000, remaining * 0.1 / restarts))
        for r in range(restarts):
            if time.time() > deadline:
                break
            start_vec = _random_dirichlet(n, rng.uniform(0.01, 1.5), rng)
            seq_eg, val_eg = _exp_gradient_optimize(start_vec, n, deadline, max_iters=iters_per_restart, eta0=2.0)
            if val_eg < global_best_val:
                global_best_val = val_eg
                global_best_seq = seq_eg.copy()
                print(f"[EG restart {r+1}/{restarts}] New global best = {global_best_val:.6f}")

    # Stage 3: Long EG from current best
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 0.4 * 1e5))  # adapt
        seq_eg, val_eg = _exp_gradient_optimize(global_best_seq, n, deadline, max_iters=max_iters, eta0=2.5)
        if val_eg < global_best_val:
            global_best_val = val_eg
            global_best_seq = seq_eg.copy()
            print(f"[Long EG] New global best = {global_best_val:.6f}")

    # Stage 4: Simple hill climbing mass transfer
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(200_000, remaining * 100))
        seq_hc, val_hc = _mass_transfer_local_search(global_best_seq, n, deadline, rng, max_iters=max_iters, step_frac=0.01)
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[HC] New global best = {global_best_val:.6f}")

    # Stage 5: LP refinement
    if time.time() < deadline:
        seq_lp, val_lp = _lp_refine(global_best_seq, n, deadline, rng, max_steps=5, top_k=250, mix_ratio=0.12)
        if val_lp < global_best_val:
            global_best_val = val_lp
            global_best_seq = seq_lp.copy()
            print(f"[LP] New global best = {global_best_val:.6f}")

    # Stage 6: Final polishing high beta L-BFGS
    if time.time() < deadline:
        per_beta = max(800, int(min(3000, (deadline - time.time()) * 0.07 * 1e5)))
        seq_final = _lbfgs_smooth_optimize(global_best_seq, n, deadline,
                                            beta_schedule=[5e5,1e6,2e6,5e6,1e7],
                                            max_iter_per_beta=per_beta)
        val_final = _objective_exact(seq_final)
        if val_final < global_best_val:
            global_best_val = val_final
            global_best_seq = seq_final.copy()
            print(f"[Final L-BFGS] New global best = {global_best_val:.6f}")

    # Evaluate final using official evaluate_sequence
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {global_best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as e:
        print("Warning evaluate_sequence error:", e)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

def _mass_transfer_local_search(a, n, deadline, rng, max_iters=100000, step_frac=0.02):
    a = a.copy()
    cur_val = _objective_exact(a)
    best_a = a.copy()
    best_val = cur_val
    # Precompute indices sorted by a to choose high and low quickly? But update each iteration.
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Choose i weighted by a (more mass)
        i = rng.choice(n, p=a)  # probability proportional to a
        # Choose j uniformly at random
        j = rng.integers(0, n)
        if i == j: continue
        # Determine delta
        delta = step_frac * a[i]
        if delta <= 0: continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # Ensure nonneg
        if cand[i] < 0: continue
        cand_val = _objective_exact(cand)
        if cand_val < cur_val:
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
        # else maybe accept with small probability? Not needed.
    return best_a, best_val

a cheap mass‑transfer hill‑climb and a small LP‑based active‑set refinement.
The whole routine respects a 1000 s wall‑clock budget and stores the best
sequence in the global variable ``height_sequence_1`` for warm‑starts.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, p>0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑density on a discrete grid."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, mu: float, sigma: float) -> np.ndarray:
    """Discretised Gaussian (truncated, normalised)."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int, width: float) -> np.ndarray:
    """Half‑cosine bump, zero outside a central window."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.maximum(0.0, np.cos(np.pi * (x - center) / width))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinusoidal(n: int) -> np.ndarray:
    """Positive half‑sine on [0,n‑1]."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    """U‑shaped distribution ∝ 1/√(x(1‑x))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)). Returns a vector of size len(a)."""
    n = len(a)
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))          # active lag
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    # for i in [i_start, i_end] we have j = k-i
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = k - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) for continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate of the max‑convolution objective."""
    a = _softmax(v)                    # ℓ₁‑normalised
    conv = _conv_fft(a, a)             # full convolution
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see derivation in original code)
    w = exp_term / sum_exp              # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 800,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0, 10_000_000.0,
        ]

    # work in log‑space (logits) – unconstrained
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple mass‑transfer hill‑climb (greedy local search)
# ----------------------------------------------------------------------
def _mass_transfer_local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 200_000,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Move a small fraction of mass from a heavy index to a random index
    whenever the move improves the exact objective."""
    a = start_a.copy()
    cur_val = _objective_exact(a)
    best_a = a.copy()
    best_val = cur_val

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # pick donor proportional to its mass (heavier atoms are tried more)
        i = rng.choice(n, p=a)
        # pick a random recipient (could be the same – then we skip)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # candidate stays non‑negative by construction
        cand_val = _objective_exact(cand)
        if cand_val < cur_val:
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set)
# ----------------------------------------------------------------------
def _lp_refine(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_steps: int = 5,
    top_k: int = 250,
    mix_ratio: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    One‑shot LP that maximises Σg_j under the active convolution bound.
    The solution g is mixed back into the current a.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for step in range(max_steps):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = np.max(conv)

        # pick top‑k lags (including the maximum)
        if top_k >= len(conv):
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(-conv, top_k)[:top_k]

        rows = []
        cols = []
        data = []
        for r, k in enumerate(idxs):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                rows.append(r)
                cols.append(j)
                data.append(a[i])

        A = _sp.csr_matrix((data, (rows, cols)), shape=(len(idxs), n))
        b_ub = np.full(len(idxs), M, dtype=np.float64)
        bounds = [(0, None)] * n
        c = np.full(n, -1.0, dtype=np.float64)  # maximise Σg → minimise −Σg

        time_left = deadline - time.time()
        lp_options = {"presolve": True, "time_limit": max(1.0, time_left * 0.2)}

        try:
            res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                               method="highs", options=lp_options)
        except Exception:  # pragma: no cover
            break

        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break
        g_norm = g / s

        # convex combination (mix_ratio controlled)
        new = (1.0 - mix_ratio) * a + mix_ratio * g_norm
        new_val = _objective_exact(new)

        if new_val < best_val:
            best_val = new_val
            best_a = new.copy()

        a = new

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence`` within the provided time budget.
    The best found sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # problem size – 4000 has proved a good trade‑off
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Build a large, diverse seed pool
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, mu=n / 2, sigma=n / 5))
    seeds.append(_raised_cosine(n, width=n * 0.8))
    seeds.append(_sinusoidal(n))
    seeds.append(_arcsine(n))

    # exponential decay (few λ values)
    for lam in (0.0005, 0.001, 0.002, 0.004, 0.008):
        seeds.append(_exponential_decay(n, lam))

    # power‑law exponents
    for p in np.linspace(0.30, 1.00, 15):
        seeds.append(_power_law(n, float(p)))

    # beta‑shape grid
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        for beta in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
            seeds.append(_beta_shape(n, float(alpha), float(beta)))

    # random Dirichlet families
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.01, 0.02, 0.05, 0.1):
        for _ in range(3):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures of existing seeds
    base = np.arange(len(seeds))
    for _ in range(150):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(40):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # warm‑start from previous global best, if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,) and arr.sum() > 0.0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # Normalise and enforce non‑negativity
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    seeds = cleaned

    # ------------------------------------------------------------------
    # Quick exact evaluation of seeds, keep the elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] Best raw seed objective = {best_val:.6f}")

    # keep the top‑k for intensive optimisation
    top_k = min(30, len(seeds))
    seeds.sort(key=_objective_exact)
    elite = seeds[:top_k]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate (elite seeds)
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing elite L‑BFGS.")
            break
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=800,
        )
        val_opt = _objective_exact(seq_opt)
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[LBFGS {idx+1}/{top_k}] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – few exponentiated‑gradient restarts (exploratory)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        restarts = 6
        iters_per_restart = int(min(300_000, remaining * 0.1 / restarts))
        for r in range(restarts):
            if time.time() > deadline:
                break
            start_vec = _random_dirichlet(n, rng.uniform(0.01, 1.5), rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.0,
            )
            if val_eg < global_best_val:
                global_best_val = val_eg
                global_best_seq = seq_eg.copy()
                print(f"[EG restart {r+1}/{restarts}] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long EG optimisation from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(1_500_000, remaining * 0.4 * 1e5))
        seq_eg, val_eg = _exp_gradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=max_iters,
            eta0=2.5,
        )
        if val_eg < global_best_val:
            global_best_val = val_eg
            global_best_seq = seq_eg.copy()
            print(f"[Long EG] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – cheap mass‑transfer hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 120))
        seq_hc, val_hc = _mass_transfer_local_search(
            global_best_seq,
            n,
            deadline,
            rng,
            max_iters=max_iters,
            step_frac=0.015,
        )
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[Hill‑Climb] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – LP‑based active‑set refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_lp, val_lp = _lp_refine(
            global_best_seq,
            n,
            deadline,
            rng,
            max_steps=5,
            top_k=250,
            mix_ratio=0.12,
        )
        if val_lp < global_best_val:
            global_best_val = val_lp
            global_best_seq = seq_lp.copy()
            print(f"[LP] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(3000, remaining * 0.07 * 1e5)))
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000.0, 10_000_000.0, 20_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        if val_final < global_best_val:
            global_best_val = val_final
            global_best_seq = seq_final.copy()
            print(f"[Final L‑BFGS] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in future invocations
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
