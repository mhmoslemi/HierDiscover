def hill_climb(a, conv, max_iters, time_deadline):
    n = len(a)
    L = 2*n-1
    best_val = 2*n*conv.max()
    best_a = a.copy()
    for iteration in range(max_iters):
        if time.time() > time_deadline: break
        # pick random i, j
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # choose delta
        # maybe proportion of a_j, within [0, a_j]
        max_delta = a[j]  # cannot move more than source
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * 0.5  # move at most half of source
        # compute new conv via update
        conv_new = conv.copy()
        # linear updates
        conv_new[i:i+n] += 2*delta*a
        conv_new[j:j+n] -= 2*delta*a
        # quadratic updates
        idx2i = 2*i
        idx2j = 2*j
        idxij = i + j
        conv_new[idx2i] += delta*delta
        conv_new[idx2j] += delta*delta
        conv_new[idxij] -= 2*delta*delta

        # compute max
        new_max = conv_new.max()
        # compute objective
        new_val = 2*n*new_max

        if new_val < best_val - 1e-12: # improvement
            # accept move
            a[i] += delta
            a[j] -= delta
            conv = conv_new
            best_val = new_val
            best_a = a.copy()
        else:
            # maybe accept with probability (simulated annealing)
            pass
```

def hill_climbing(a, time_deadline):
    n = len(a)
    conv = _conv_full(a)
    best_a = a.copy()
    best_val = 2*n * conv.max()
    iters = 0
    no_improve = 0
    while time.time() < time_deadline:
        # choose random indices
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if i == j: continue
        if a[j] <= 0: continue
        # choose delta fraction
        max_delta = a[j]
        # sample factor. Could use small value or random uniform
        # maybe use exponential distribution to get varying sizes
        factor = rng.random()  # uniform 0-1
        delta = max_delta * factor * 0.5  # up to 0.5 of a_j
        if delta <= 0:
            continue
        # compute updates
        conv_new = conv.copy()
        delta2a = 2.0 * delta * a
        conv_new[i:i+n] += delta2a
        conv_new[j:j+n] -= delta2a
        conv_new[2*i] += delta*delta
        conv_new[2*j] += delta*delta
        conv_new[i+j] -= 2.0 * delta * delta

"""
Propose a candidate step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.  The algorithm starts from the best known solution (if any)
or from a uniform distribution, then performs a large number of cheap local
mass‑transfer moves that directly update the full convolution.  Accepted moves
strictly improve the surrogate objective 2·n·max(conv).  After the time budget
(≈1000 s) is almost exhausted a short mirror‑descent (exponentiated‑gradient)
phase is run to polish the result.  The final sequence is returned and stored
in the global ``height_sequence_1`` for possible warm‑starts on the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Length of the result is 2·len(a) - 1.
    """
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (seq is assumed ℓ¹‑normalised)."""
    n = seq.shape[0]
    conv = _conv_full(seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Hill‑climbing local search (mass transfer moves)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                report_every: int = 5000) -> tuple[np.ndarray, float]:
    """
    Perform greedy hill‑climbing on the exact max‑convolution objective.
    At each iteration a random amount δ is moved from a source index j to a
    target index i (i≠j).  The convolution is updated in O(n) thanks to the
    explicit formula for the effect of a mass transfer.
    Only strictly improving moves are accepted.
    """
    n = a.shape[0]
    conv = _conv_full(a)
    best_val = _objective(a)
    best_a = a.copy()
    it = 0
    no_improve = 0
    # pre‑allocate a buffer for the updated convolution to avoid repeated allocation
    while time.time() < deadline:
        # --- pick a random move ------------------------------------------------
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if a[j] <= 0.0:
            continue

        # pick a delta: up to 50 % of the source mass, uniformly random
        delta = rng.random() * a[j] * 0.5
        if delta <= 0.0:
            continue

        # --- compute the updated convolution ----------------------------------
        conv_new = conv.copy()
        # linear part: 2·δ·a shifted to the two positions
        delta2a = 2.0 * delta * a
        conv_new[i:i + n] += delta2a
        conv_new[j:j + n] -= delta2a
        # quadratic correction (three isolated entries)
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = float(conv_new.max())
        new_val = 2.0 * n * new_max

        # --- accept if strictly better ----------------------------------------
        if new_val < best_val - 1e-12:      # tiny tolerance to avoid noise
            # apply the move
            a[i] += delta
            a[j] -= delta
            conv = conv_new
            best_val = new_val
            best_a = a.copy()
            no_improve = 0
            if it % report_every == 0:
                print(f"[HillClimb] iter {it:8d}, val {best_val:.9f}")
        else:
            no_improve += 1

        it += 1

        # occasional restart if we are stuck for a long time
        if no_improve > 300_000:
            # a gentle perturbation to shake out of a plateau
            eps = 1e-4
            perturb = rng.normal(scale=eps, size=n)
            a = np.maximum(a + perturb, 0.0)
            a_sum = a.sum()
            if a_sum <= 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a /= a_sum
            conv = _conv_full(a)
            best_val = _objective(a)
            best_a = a.copy()
            no_improve = 0
            print(f"[HillClimb] restart after {it} iterations, new val {best_val:.9f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Variables live in log‑space and are projected by the softmax.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)
        conv = _conv_full(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (as in the reference code)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g

        cur_val = _objective(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The pipeline consists of:

    1. Warm‑start from the previous best (global variable ``height_sequence_1``)
       or a uniform distribution.
    2. Aggressive hill‑climbing using cheap O(n) convolution updates.
    3. A short exponentiated‑gradient polishing phase.
    4. Final official evaluation and storage of the best sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size (fixed, same as in the reference run)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) == n:
        a0 = np.asarray(prev, dtype=np.float64)
        total = a0.sum()
        if total > 0.0:
            a0 = a0 / total
        else:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        print("[Init] Warm‑start from previous best.")
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        print("[Init] No previous solution – start from uniform.")

    # ------------------------------------------------------------------
    # Phase 1 – Hill‑climbing
    # ------------------------------------------------------------------
    hc_deadline = deadline - 15.0       # reserve a few seconds for polishing
    print("[Phase 1] Hill‑climbing start.")
    best_a, best_val = _hill_climb(a0.copy(), hc_deadline, rng)
    print(f"[Phase 1] Finished. Best surrogate value = {best_val:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 2.0
        print("[Phase 2] Exponentiated‑gradient polishing.")
        seq_eg, val_eg = _exp_gradient_opt(best_a.copy(),
                                          n,
                                          eg_deadline,
                                          max_iters=2_000_000,
                                          eta0=5.0)
        if val_eg < best_val:
            best_a, best_val = seq_eg, val_eg
            print(f"[Phase 2] Improvement found: {best_val:.9f}")
        else:
            print(f"[Phase 2] No improvement (best still {best_val:.9f})")

    # Final evaluation
    # ------------------------------------------------------------------
    # Normalise one last time (tiny drift may have occurred)
    best_a /= best_a.sum()
    try:

              f"evaluate_sequence = {true_val:.9f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run
    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
