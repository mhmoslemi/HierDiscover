"""
Hybrid seed + continuation L‑BFGS + Simulated Annealing optimizer.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a))² ,   a_i ≥ 0

where ``conv`` is the full linear convolution.
The algorithm proceeds in three stages:

1.  Generate a diverse pool of ℓ₁‑normalised candidates (uniform,
    triangular, power‑law, Dirichlet draws, low‑frequency mixtures,
    etc.) and keep the elite few.
2.  For each elite run a continuation L‑BFGS on a smooth log‑sum‑exp
    surrogate of the max‑convolution.  The continuation steps through
    an increasing schedule of the smoothing parameter β, ending at a
    very large value (≈10¹⁰) so that the surrogate is almost the exact
    max.
3.  Starting from the best surrogate, perform a simulated‑annealing
    hill‑climb that repeatedly moves a small amount of mass from a
    randomly chosen source index to a random destination.  Moves that
    increase the exact objective are occasionally accepted according
    to an exponential temperature schedule, allowing the search to
    escape shallow local minima.  After the annealing phase a final
    short L‑BFGS polishing run (very high β) cleans up any remaining
    irregularities.

All stages respect a total wall‑time budget (default 1000 s).  The best
sequence found is stored in the global ``height_sequence_1`` variable
and returned as a plain Python list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (used by the annealer)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (a is assumed non‑negative)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of the exact objective.
    Returns (obj, grad) where the gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised (sum = 1)
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # smooth max of the convolution values
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (derived analytically)
    w = exp_term / sum_exp               # softmax over lag indices
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through the simplex softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    ``start_vec`` must already be ℓ₁‑normalised.
    Returns a new ℓ₁‑normalised vector.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0, 50_000_000.0, 100_000_000.0,
            500_000_000.0, 1_000_000_000.0, 5_000_000_000.0,
            1_000_000_000_0.0,
        ]

    # start from logits of the initial probability vector
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(
                fun=_smooth_objective,
                x0=v,
                args=(n, float(beta)),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_beta,
                         "ftol": 1e-12,
                         "disp": False},
            )
            v = res.x
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _semicircle_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (i - c) / c
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """Create a vector with only a small fraction of non‑zero entries."""
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mix of two ℓ₁‑normalised vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    """Cyclic shift (roll) of a vector."""
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised seed vectors (length n)."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_sin_shape(n))
    seeds.append(_semicircle_shape(n))

    # Power‑law sweep – wide range of exponents and shifts
    for p in np.linspace(0.2, 1.4, 12):
        for sh in (0.0, 5.0, 20.0, 100.0):
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # Random Dirichlet draws (sparse to dense)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.6, 1.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # Pairwise mixtures to explore convex hull
    base = np.arange(len(seeds))
    for _ in range(150):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of promising seeds
    for _ in range(10):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Warm‑start from previous best if available
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Final sanitisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Simulated annealing mover
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000,
                         init_temp: float = 1e-3,
                         final_temp: float = 1e-7,
                         step_frac: float = 0.012) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly moves a fraction of mass from a source index
    to a destination index.  Moves that increase the exact objective are
    accepted with probability exp(-Δ/T).  The temperature decays exponentially.
    """
    a = start_a.copy()
    best_a = a.copy()
    cur_val = _objective_exact(a)

    rng = np.random.default_rng(123456789)
    # pre‑compute logarithmic temperature schedule
    if max_iters <= 0:
        return best_a, cur_val
    log_T0 = np.log(init_temp)
    log_Tf = np.log(final_temp)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # exponential cooling
        T = np.exp(log_T0 + (log_Tf - log_T0) * (it / max_iters))

        # propose a move
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # re‑normalise to eliminate tiny drift (keeps sum exactly 1)
        cand_sum = cand.sum()
        if cand_sum != 1.0:
            cand /= cand_sum

        cand_val = _objective_exact(cand)
        diff = cand_val - cur_val
        if diff < 0.0 or rng.random() < np.exp(-diff / T):
            a = cand
            cur_val = cand_val
            if cur_val < _objective_exact(best_a):
                best_a = a.copy()
        # occasional reporting (optional)
        if it % 50_000 == 0:
            print(f"[SA] iter {it}/{max_iters}, T={T:.2e}, cur={cur_val:.6f}, best={_objective_exact(best_a):.6f}")

    return best_a, _objective_exact(best_a)
