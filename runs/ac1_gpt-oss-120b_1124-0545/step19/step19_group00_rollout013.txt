"""
Simple local‑search optimizer for the max–convolution objective.

The algorithm starts from the best sequence known so far (global variable
`height_sequence_1`) or from a random Dirichlet vector if none exists.
It repeatedly transfers a small amount of mass from a source index to a
target index, updating the full convolution exactly by using the
property

    (a+Δ) * (a+Δ) = a*a + 2·a*Δ + Δ*Δ

where Δ is a sparse vector with only two non‑zero entries.
The convolution is therefore updated with O(n) slice operations instead of
recomputing an FFT at every step.  Moves are accepted if they improve the
current surrogate value, otherwise they may be accepted with a small
temperature‑controlled probability (simulated annealing).  The best
sequence found during the run is finally evaluated with the provided
`evaluate_sequence` function and stored back to `height_sequence_1`.

The whole search respects the 1000 s wall‑time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (length = len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))  (seq assumed sum=1)."""
    n = len(seq)
    conv = _conv_full(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Local‑search core (exact convolution update)
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray,
                conv: np.ndarray,
                i: int,
                j: int,
                delta: float) -> None:
    """
    Apply the transfer of `delta` mass from index i to j.
    The convolution array `conv` (full length 2n‑1) is updated in‑place
    using the exact formula.
    """
    n = a.shape[0]
    #   conv ← conv - 2·δ·a[i‑shift] + 2·δ·a[j‑shift]  (slice updates)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    #   add the quadratic part Δ*Δ
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    conv[i + j] -= 2.0 * delta * delta

    #   finally modify the vector a itself
    a[i] -= delta
    a[j] += delta


def _local_search(start_seq: np.ndarray,
                  budget_s: float,
                  start_seed: int = 0) -> tuple[np.ndarray, float]:
    """
    Perform a time‑bounded stochastic hill‑climbing / simulated‑annealing
    on the surrogate objective.  Returns the best sequence encountered
    and its surrogate value.
    """
    rng = np.random.default_rng(start_seed)
    a = start_seq.copy()
    n = a.shape[0]
    conv = _conv_full(a, a)
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)   # 0 → 1

        #--- schedule -------------------------------------------------
        # move size shrinks from ~0.4·a[i] to ~0.01·a[i]
        max_move_factor = max(0.01, 0.4 * (1.0 - frac) + 0.01)
        # temperature for uphill moves (very low, just to escape tiny plateaus)
        temp = max(1e-8, 5e-4 * (1.0 - frac))

        #--- choose source index (biased towards large entries) ----------
        i = rng.choice(n, p=a)                # source distribution ∝ a_i
        if a[i] < 1e-12:
            continue

        #--- choose target index (uniform) -------------------------------
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        #--- pick a transfer size ----------------------------------------
        delta = max_move_factor * a[i] * rng.random()
        if delta <= 0.0:
            continue

        #--- keep copies for possible rollback -----------------------------
        a_prev = a.copy()
        conv_prev = conv.copy()
        cur_val_prev = cur_val

        #--- apply move ----------------------------------------------------
        _apply_move(a, conv, i, j, delta)

        # new surrogate value
        cur_val = 2.0 * n * np.max(conv)

        #--- acceptance test (simulated annealing) -----------------------
        improved = cur_val < cur_val_prev
        if improved or rng.random() < math.exp(-(cur_val - cur_val_prev) / temp):
            # keep the move, update best if needed
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
            # continue with new state
        else:
            # reject: roll back
            a[:] = a_prev
            conv[:] = conv_prev
            cur_val = cur_val_prev

        # occasional progress report (without affecting time limit)
        if it % 50000 == 0:
            print(f"[search] it={it:,} best={best_val:.6f} cur={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The search consists of:

    1. Load the previously best known vector (global ``height_sequence_1``)
       – if unavailable fall back to a random Dirichlet seed.
    2. Run a fast O(n) local‑search that repeatedly moves a small amount of
       mass between two positions while updating the full convolution exactly.
    3. Return the best vector found (stored back to ``height_sequence_1``).
    """
    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    n = 4000  # the size used in the previous runs; we stay consistent

    # ------------------------------------------------------------------
    # Load previous best if present, otherwise initialise randomly
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and np.all(arr >= 0):
                start_vec = arr / arr.sum()
                print("[init] Loaded previous best sequence.")
            else:
                raise ValueError
        except Exception:
            start_vec = None
    else:
        start_vec = None

    if start_vec is None:
        rng = np.random.default_rng(seed)
        start_vec = rng.dirichlet(np.full(n, 0.5)).astype(np.float64)
        print("[init] No previous sequence – using random Dirichlet start.")

    # ------------------------------------------------------------------
    # Run the stochastic local search
    # ------------------------------------------------------------------
    start = time.time()
    remaining_budget = budget_s - (time.time() - start)
    if remaining_budget < 1.0:
        raise RuntimeError("Not enough time budget after initialisation.")

    print(f"[search] Starting local search (n={n}) with budget ≈{remaining_budget:.1f}s")
    best_seq, best_surrogate = _local_search(start_vec, remaining_budget, seed)

    # ------------------------------------------------------------------
    # Final official evaluation (exact objective)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store the best sequence globally for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
