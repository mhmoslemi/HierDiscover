def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0

    n=4000

    ### Seed generation
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = []
    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n,0.25))
    seeds.append(_linear_decrease(n))
    seeds.append(_quadratic_decrease(n))
    # Exponential fast
    for lam in np.logspace(-4, -0.5, 7):
        seeds.append(_exponential_decay(n, lam))
        seeds.append(_exponential_decay_rev(n, lam))
    # Beta shape random 50
    for _ in range(50):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, alpha, beta))
    # Random Dirichlet draws
    for conc in (0.02,0.05,0.1,0.2,0.5,1.0,2.0):
        for _ in range(10):
            seeds.append(_random_dirichlet(n, conc, rng))
    # Zigzag permutations of Dirichlet draws
    for _ in range(15):
        v = _random_dirichlet(n, 0.5, rng)
        seeds.append(_zigzag(v))
    # previous best
    if prev_arr is not None and prev_arr.size==n:
        seeds.append(prev_arr/prev_arr.sum())
    # Clean and normalize
    cleaned=[]
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v,0.0)
        s=v.sum()
        if s==0.0:
            v=_uniform(n)
            s=v.sum()
        cleaned.append(v/s)
    seeds = cleaned

    best_val = np.inf
    best_seq = None
    for v in seeds:
        val=_objective_exact(v)
        if val<best_val:
            best_val=val
            best_seq=v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # keep elite top 15
    seeds.sort(key=_objective_exact)
    elite = seeds[:15]
    global_best_seq=best_seq.copy()
    global_best_val=best_val

    # Stage L-BFGS
    beta_sched = [1,5,20,80,200,500,1e3,2e3,5e3,1e4,2e4,5e4,1e5,2e5,5e5,1e6,2e6,5e6,1e7,5e7,1e8,5e9,1e12]
    for idx, seed_vec in enumerate(elite):
        if time.time()>deadline: break
        print(f"\n[L-BFGS] run {idx+1}/{len(elite)}")

        val_opt=_objective_exact(seq_opt)
        print(f"    L-BFGS obj = {val_opt:.6f}")
        if val_opt<global_best_val:
            global_best_seq=seq_opt.copy()
            global_best_val=val_opt
            print(f"    New best = {global_best_val:.6f}")

    # Stage EG restarts

    # Determine iteration budget per restart based on remaining time
    while time.time()<deadline and restarts>0:

        if iters_per_restart<=0: break

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → ℓ₁‑normalised (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) – only the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g

def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp approximation of max(conv). Returns (obj, grad) where the
    gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)
    shift = np.max(conv)                 # numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # weights over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # size = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _pnorm_objective(v: np.ndarray, n: int, p: float):
    """
    p‑norm surrogate (‖conv(a,a)‖ₚ). Returns (obj, grad) w.r.t. logits v.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    conv_pow = conv ** (p - 1.0)
    obj = np.sum(conv ** p) ** (1.0 / p)

    w = p * conv_pow
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay_rev(n: int, lam: float) -> np.ndarray:
    return _exponential_decay(n, lam)[::-1]


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    a = (n - i) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:

def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)

    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _zigzag(v: np.ndarray) -> np.ndarray:
    """Interleave largest / smallest values to obtain a ‘zig‑zag’ ordering."""
    s = np.sort(v)[::-1]
    out = np.empty_like(s)
    left, right = 0, len(s) - 1
    i = 0
    while left <= right:
        out[i] = s[left]
        i += 1
        left += 1
        if left <= right:
            out[i] = s[right]
            i += 1
            right -= 1
    return out / out.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ---------- problem size ----------
    n = 4000
    print(f"\n=== Optimisation start — n={n}, budget≈{budget_s}s ===")

    # ---------- seed pool ----------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ---------- initialise best ----------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # keep elite for heavy optimisation
    seeds.sort(key=_objective_exact)
    elite = seeds[:min(15, len(seeds))]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ---------- Stage 1 – L‑BFGS on smooth surrogate ----------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 50_000_000.0, 100_000_000.0,
        5_000_000_000.0, 1_000_000_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=6000,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ---------- Stage 2 – EG restarts ----------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient restarts (exact max) ---")
        restarts = 12
        while restarts > 0 and time.time() < deadline:
            remaining = deadline - time.time()
            iters_per_restart = int(min(2_000_000,
                                        remaining * 0.15 / restarts))
            if iters_per_restart <= 0:
                break
            conc = rng.uniform(0.02, 1.5)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.5,
            )
            print(f"[EG restart] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New global best after EG = {global_best_val:.6f}")
            restarts -= 1

    # ---------- Stage 3 – permutation search ----------
    if time.time() < deadline:
        print("\n--- Permutation search on best vector ---")
        a = global_best_seq.copy()
        v_sorted = np.sort(a)[::-1]                 # descending
        candidates = [
            v_sorted,
            v_sorted[::-1],
            _zigzag(v_sorted),
        ]
        # a few circular shifts of the descending order
        step = max(1, n // 40)
        for shift in range(0, n, step):
            candidates.append(np.roll(v_sorted, shift))

        for idx, cand in enumerate(candidates):
            if time.time() > deadline:
                break
            val = _objective_exact(cand)
            if val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = val
                print(f"[Permute {idx}] improved to {global_best_val:.6f}")

    # ---------- Stage 4 – aggressive local search ----------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(800_000, remaining * 80))
        print(f"\n--- Mass‑transfer local search (iters={iters}) ---")
        seq_loc, val_loc = _local_search(
            global_best_seq,
            n,
            deadline,
            iters=iters,
            step_frac=0.008,
        )
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[Local] improved to {global_best_val:.6f}")

        # swap‑based hill climbing
        remaining = deadline - time.time()
        if remaining > 10.0:
            swap_iters = int(min(200_000, remaining * 30))
            print(f"\n--- Swap hill climbing (iters={swap_iters}) ---")
            seq_swap, val_swap = _swap_hill_climb(
                global_best_seq,
                n,
                deadline,
                iters=swap_iters,
            )
            if val_swap < global_best_val:
                global_best_seq = seq_swap.copy()
                global_best_val = val_swap
                print(f"[Swap] improved to {global_best_val:.6f}")

    # ---------- Stage 5 – final polishing L‑BFGS ----------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = int(min(4000, remaining * 0.04 * 1e5))
        print("\n--- Final polishing L‑BFGS (ultra‑high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000_000.0, 1_000_000_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Final] improved to {global_best_val:.6f}")

    # ---------- final evaluation ----------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
