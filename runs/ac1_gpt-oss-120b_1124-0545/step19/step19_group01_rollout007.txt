def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 4000
    # param sampling
    best_seq = None
    best_val = float('inf')
    sample_count = 0
    max_samples = 20000  # high number earlier but break earlier with time

    while time.time() < deadline and sample_count < max_samples:
        # sample rates
        log_r1 = rng.uniform(-12.0, -0.2)   # rates from e-12 to ~0.82
        log_r2 = rng.uniform(-12.0, -0.2)
        # sample weights from Dirichlet
        weights = rng.dirichlet([1.0,1.0,1.0])
        w1, w2, w3 = weights
        # Build vector
        i = np.arange(n, dtype=np.float64)
        comp1 = np.exp(-np.exp(log_r1) * i)
        comp2 = np.exp(-np.exp(log_r2) * i)
        vec = w1 * comp1 + w2 * comp2 + w3 * np.ones(n)
        vec_sum = vec.sum()
        if vec_sum == 0.0:
            continue
        vec /= vec_sum
        # evaluate surrogate
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
        sample_count += 1
        # occasionally log progress
        if sample_count % 2000 == 0:
            print(f"[Sampling] {sample_count} samples, best surrogate {best_val:.6f}")

    if best_seq is None:
        # fall back to uniform
        best_seq = np.full(n, 1.0/n)

"""
Propose a low‑autocorrelation (small max‑convolution) step function.

Algorithm
---------

1. **Rich parametric seed generation** – we generate vectors of length *n*
   (default 4000) as mixtures of two exponential decays and a uniform
   background.  The three mixture weights are drawn from a Dirichlet
   distribution, the decay rates from a log‑uniform range.  All vectors are
   renormalised to L¹‑norm = 1.  This family includes the previously used
   exponential / uniform shapes but also many new intermediate forms.

2. **Fast stochastic search** – 10‑20 k random seeds are sampled while we are
   still far from the time limit.  The best (lowest surrogate value) is kept.

3. **Ordering refinement** – a short simulated‑annealing‑style hill‑climb
   swaps two positions at a time; because the convolution depends on the
   ordering, a better permutation can lower the max‑convolution peak.

4. **Exponentiated‑gradient polishing** – the best vector is then optimised
   with mirror‑descent on the exact max‑convolution (sub‑gradient on the
   active index).  The step‑size schedule is long (up to a few million
   iterations) to drive the objective down further.

5. **Optional permutation hill‑climb** – after the gradient phase we try a
   few more random swaps; sometimes a simple re‑ordering yields a tiny
   improvement.

6. **Final smoothed‑max optimisation** – we finish with the L‑BFGS
   optimiser applied to a log‑sum‑exp smooth approximation of the max
   convolution.  A much larger `max_iter_per_beta` than the baseline is
   used (5 000) to allow deeper convergence.

All steps respect the global deadline (default 1000 s).  The globally best
sequence is stored in ``height_sequence_1`` and also returned as a list of
floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# FFT‑based full convolution (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a vector with L1 = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient on the active index).
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5_000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple permutation hill‑climb (swap moves only)
# ----------------------------------------------------------------------
def _perm_hill_climb(seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions, accept only improving moves.
    Returns best sequence seen and its surrogate value.
    """
    cur_seq = seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]
        val = _objective_surrogate(ns)
        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found (list of floats) and stores it as
    ``height_sequence_1`` for possible continuation runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem dimension – 4000 works well for the time budget.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load a previous solution if it exists (helps continuation).
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        prev_arr = np.asarray(prev, dtype=np.float64)
        prev_arr = np.maximum(prev_arr, 0.0)
        if prev_arr.sum() > 0:
            prev_arr = prev_arr / prev_arr.sum()
    else:
        prev_arr = None

    # --------------------------------------------------------------
    # 1. Random mixture‑of‑exponential seed generation
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    sample_cnt = 0
    max_samples = 30_000   # upper bound – actual number limited by time

    print("\n--- Parameter‑space sampling (mixture of exponentials) ---")
    while time.time() < deadline and sample_cnt < max_samples:
        # draw decay rates on a log scale
        log_r1 = rng.uniform(-12.0, -0.2)   # r = exp(log_r) ∈ [~6e‑6, 0.82]
        log_r2 = rng.uniform(-12.0, -0.2)

        # mixture weights from Dirichlet (expo1, expo2, uniform)
        w1, w2, w3 = rng.dirichlet([1.0, 1.0, 1.0])

        # build the raw (unnormalised) vector
        i = np.arange(n, dtype=np.float64)
        r1 = np.exp(log_r1)
        r2 = np.exp(log_r2)
        comp1 = np.exp(-r1 * i)
        comp2 = np.exp(-r2 * i)
        vec = w1 * comp1 + w2 * comp2 + w3 * np.ones(n, dtype=np.float64)

        total = vec.sum()
        if total == 0.0:
            continue
        vec /= total   # L¹‑normalise

        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
        sample_cnt += 1

        if sample_cnt % 5000 == 0:
            elapsed = time.time() - start_time
            print(f"  Sample #{sample_cnt:5d}, best surrogate = {best_val:.6f} (elapsed {elapsed:.1f}s)")

    # Incorporate previous best if any
    if prev_arr is not None:
        val_prev = _objective_surrogate(prev_arr)
        if val_prev < best_val:
            best_val = val_prev
            best_seq = prev_arr.copy()
        print(f"  Included previous solution: surrogate = {val_prev:.6f}")

    # --------------------------------------------------------------
    # 2. Short permutation hill‑climb to improve ordering
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Ordering hill‑climb (swap moves) ---")
        order_deadline = time.time() + remaining * 0.2
        seq_perm, val_perm = _perm_hill_climb(best_seq, n, order_deadline, rng, max_iters=150_000)
        if val_perm < best_val:
            best_seq = seq_perm
            best_val = val_perm
            print(f"  Ordering improvement: new best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Long exponentiated‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing ---")
        eg_seq, eg_val = _exp_gradient_opt(best_seq, n, deadline,
                                           max_iters=3_000_000, eta0=6.0)
        if eg_val < best_val:
            best_seq = eg_seq
            best_val = eg_val
            print(f"  EG improvement: best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Optional additional swap hill‑climb (post‑EG)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Post‑EG ordering refinement ---")
        post_deadline = time.time() + remaining * 0.15
        seq_post, val_post = _perm_hill_climb(best_seq, n, post_deadline, rng, max_iters=80_000)
        if val_post < best_val:
            best_seq = seq_post
            best_val = val_post
            print(f"  Post‑EG ordering improvement: best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final smoothed‑max L‑BFGS refinement (more iterations per β)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing refinement ---")
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                           beta_schedule=[
                                               1.0, 5.0, 20.0, 80.0,
                                               200.0, 500.0, 1_000.0,
                                               2_000.0, 5_000.0, 20_000.0,
                                               50_000.0, 100_000.0, 200_000.0,
                                               500_000.0, 1_000_000.0
                                           ],
                                           max_iter_per_beta=5_000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_seq = seq_lbfgs
            best_val = val_lbfgs
            print(f"  L‑BFGS improvement: best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
