"""
propose_candidate – enhanced search for a low‑autocorrelation step function

The scoring function is

    2·n·max( convolution( a, a ) ) / ( sum(a) ** 2 )
    (the implementation clips negatives and limits each entry to 1000)

For a non‑negative vector the convolution’s maximum is attained either
at the two borders (a_i²) or at the centre where a is paired with its
reverse.  The optimum therefore trades off a small maximal entry against a
small “cross‑correlation’’ with the reversed copy.

The algorithm works in three phases:

1. **Massive diversified seeding** – analytic families (exponential,
   power‑law, beta, Gaussian, sine‑window), Dirichlet draws, low‑frequency
   cosine curves and random block‑wise constructions.  All seeds are
   ℓ₁‑normalised (∑a_i = 1) and stored together with a cheap surrogate
   objective (FFT‑based convolution).

2. **Permutation‑based simulated‑annealing hill‑climbing** – each of the
   best‐k seeds undergoes a fast local search that swaps positions,
   perturbs values and (optionally) re‑sorts the vector to stay monotone.
   A simple linear cooling schedule and occasional acceptance of
   worsening moves help escape shallow local minima.

3. **Mirror‑descent polishing** – from the locally‑optimised vectors we
   run a sub‑gradient mirror‑descent step on the exact max‑convolution.
   The update is performed in log‑softmax space (adds up to 1 automatically)
   and accelerated with Adam‑style moment estimates.  The best result of
   all seeds is finally polished for the remaining time budget.

The whole search respects the 1000 s limit, always keeping the globally
best sequence in the variable ``height_sequence_1`` and returning it as a
list of floats.
"""

import time
import math
import numpy as np
from concurrent.futures import ProcessPoolExecutor, as_completed

# ----------------------------------------------------------------------
# Utility for FFT‑based full convolution (O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate / exact objective (the surrogate is the exact objective
# because we keep the sum normalised to 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(seq,seq)).  ``seq`` is assumed non‑negative."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine components, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _block_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piece‑wise constant sequence (random number of blocks)."""
    blocks = rng.integers(2, 15)            # 2‑14 blocks
    # random lengths that sum to n
    lengths = rng.dirichlet(np.ones(blocks)) * n
    lengths = np.round(lengths).astype(int)
    # fix rounding issues
    diff = n - lengths.sum()
    if diff != 0:
        lengths[0] += diff
    heights = rng.exponential(scale=1.0, size=blocks)
    seq = np.repeat(heights, lengths)[:n]
    return seq / seq.sum()


def _sorted_desc(seq: np.ndarray) -> np.ndarray:
    """Return a descending‑sorted copy (keeps ℓ₁‑norm = 1)."""
    s = np.sort(seq)[::-1]
    return s / s.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Mirror‑descent on max‑convolution (sub‑gradient at the active index)
# ----------------------------------------------------------------------
def _mirror_descent(seq: np.ndarray, n: int, deadline: float,
                    max_iters: int = 300_000,
                    eta0: float = 5.0,
                    beta1: float = 0.9,
                    beta2: float = 0.999,
                    eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the exact max‑convolution.
    Operates in log‑softmax space to keep the vector feasible.
    Returns the best sequence seen and its surrogate value.
    """
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    # log‑space variables (softmax guarantees sum = 1)
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active term
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - i_range
        g[i_range] = 2.0 * a[rev]

        # Adam update in log‑space
        grad_log = -g + np.dot(g, a)  # gradient of log‑softmax objective
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = np.exp(log_a)
        a /= a.sum()                     # softmax normalisation

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style permutation hill‑climbing
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray, n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that swaps two positions, perturbs values or re‑sorts.
    Accepts worsening moves with a temperature schedule.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    # temperature schedule (linear cooling)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        move_type = rng.random()
        if move_type < 0.45:                     # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue
        elif move_type < 0.75:                   # perturb values
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s
        else:                                    # re‑sort descending (helps monotone shape)
            ns = np.sort(cur)[::-1]

        val = _objective(ns, n)

        if val < cur_val:
            cur, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # accept worse move with probability exp(−Δ/T)
            prob = math.exp(-(val - cur_val) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur, cur_val = ns, val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The best sequence discovered is stored in the global variable
    ``height_sequence_1`` and also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a short safety margin

    # --------------------------------------------------------------
    # Problem dimension – 4000 was a good trade‑off in earlier runs.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load a previous solution if it exists for warm restarts.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Massive seed generation
    # --------------------------------------------------------------
    seeds = []

    # Analytic families – dense grids
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.1, 3.5, 20):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma_frac in np.linspace(0.05, 0.35, 20):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.0, 4.0, 20):
        seeds.append(_power(n, p))

    # Exponential / symmetric exponential
    for rate in np.logspace(-5, -0.3, 40):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Beta‑shape random draws
    for _ in range(80):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(12):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(7):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random blockwise constructions
    for _ in range(30):
        seeds.append(_block_sequence(n, rng))

    # Descending‑sorted random vectors (often give low central cross‑term)
    for _ in range(25):
        rand = rng.random(n)
        seeds.append(_sorted_desc(rand))

    # Mixtures of uniform & exponential (adds a bit of flatness)
    mix_alphas = (0.2, 0.5, 0.8)
    for alpha in mix_alphas:
        for rate in (0.001, 0.005, 0.02):
            curve = _exp_decay(n, rate)
            mixed = (1.0 - alpha) * _uniform(n) + alpha * curve
            seeds.append(mixed / mixed.sum())

    # Permutations of a few strong bases
    base_shapes = [_exp_decay(n, 0.003),
                   _sym_exp_decay(n, 0.004),
                   _beta_shape(n, 2.0, 2.0),
                   _gaussian(n, 0.18)]
    for base in base_shapes:
        for _ in range(5):
            seeds.append(rng.permutation(base))

    # Include previous best if available and perturbations of it
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            seeds.append(_perturb(prev_arr, rng, sigma=0.004))

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds – keep the top‑k
    # --------------------------------------------------------------
    def eval_one(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            return np.inf, None
        v = v / s
        val = _objective(v, n)
        return val, v

    # Parallel evaluation (2 workers – respects the 2‑CPU limit)
    evaluated = []
    with ProcessPoolExecutor(max_workers=2) as pool:
        futures = [pool.submit(eval_one, vec) for vec in seeds]
        for fut in as_completed(futures):
            val, vec = fut.result()
            if math.isfinite(val):
                evaluated.append((vec, val))
            if time.time() > deadline:
                break

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq, n)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start refinement (hill‑climb + mirror‑descent)
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # give each seed roughly equal remaining time, keeping 15 % for final polish
        remaining = deadline - time.time()
        share = remaining * 0.15 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] budget ≈ {share:.1f}s")

        # ---- (a) permutation‑based hill‑climbing ----
        seq_hc, val_hc = _hill_climb(seed_vec.copy(), n, sub_deadline, rng,
                                     max_iters=180_000)

        # ---- (b) mirror‑descent polishing from hill‑climb result ----
        if time.time() < sub_deadline:
            seq_md, val_md = _mirror_descent(seq_hc, n, sub_deadline,
                                             max_iters=300_000, eta0=6.0)
        else:
            seq_md, val_md = seq_hc, val_hc

        # ---- pick best of the three (seed, hill‑climb, mirror) ----
        candidates = [
            (seed_vec, _objective(seed_vec, n)),
            (seq_hc, val_hc),
            (seq_md, val_md)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq, global_best_val = local_best_seq, local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long mirror‑descent polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long mirror‑descent polishing ---")
        final_seq, final_val = _mirror_descent(global_best_seq, n,
                                               deadline,
                                               max_iters=2_200_000,
                                               eta0=5.0)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
