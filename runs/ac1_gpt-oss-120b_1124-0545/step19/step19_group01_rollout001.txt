def compute_grad_logits(v, beta):
   a = softmax(v)
   conv = _conv_fft(a, a)
   shift = conv.max()
   exp_beta = np.exp(beta * (conv - shift))
   sum_exp = exp_beta.sum()
   w = exp_beta / sum_exp
   rev_a = a[::-1]
   full = _conv_fft(rev_a, w)
   grad_a = 2.0 * full[n-1 : n-1+n]   # same as in code
   # gradient w.r.t logits: grad_v = a * (grad_a - (grad_a @ a))
   dot = np.dot(grad_a, a)
   grad_v = a * (grad_a - dot)
   return grad_v, a, conv.max()  # conv max maybe return for monitoring

"""
propose_candidate – rich seed pool + Adam‑based continuation on a smooth max‑convolution

The algorithm works in three phases:

1. **Seed generation** – a large and very diverse collection of analytic
   families (uniform, triangular, sine‑window, Gaussian, beta‑shaped,
   exponential, symmetric‑exponential, power‑law, inverse‑power) together
   with many random Dirichlet draws and random permutations.  All seeds are
   ℓ₁‑normalised (sum = 1).

2. **Coarse selection + local refinement** – the seeds are evaluated with the
   exact surrogate 2·n·max(conv) and the best *k* are kept.  Each of those
   seeds is refined by a **continuation Adam optimiser** that minimises a
   smooth approximation of the max‑convolution (log‑sum‑exp with an increasing
   β‑parameter).  The optimiser works directly in the log‑softmax space,
   guaranteeing feasibility (non‑negativity and unit sum) without any
   projection.

3. **Final polishing** – the globally best sequence found so far is fed into a
   long Adam run with a finer β‑schedule to obtain the final answer.

The whole routine respects a 1000‑second budget, always keeping the best
sequence in the global variable ``height_sequence_1`` and returning it as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Numerics
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·|i‑(n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inv_power_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped family a_i ∝ ((i+1)*(n‑i))^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = n - i + 1
    denom = i * j
    a = denom ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _permute(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a random permutation of a 1‑D array."""
    return rng.permuted(arr)


# ----------------------------------------------------------------------
# Gradient of the smooth max‑convolution (log‑softmax variables)
# ----------------------------------------------------------------------
def _grad_logits(v: np.ndarray, beta: float, n: int):
    """
    Returns (grad_v, a, cur_max) where:
      a = softmax(v)                     (feasible probability vector)
      grad_v = ∂/∂v  ( 2·n·log‑sum‑exp_{k}[β·conv_k] )
      cur_max = max(conv)
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)                     # length = 2n‑1
    shift = conv.max()
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp                     # softmax over convolution entries

    rev_a = a[::-1]
    # convolution of rev_a with w gives the gradient of each a_i
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # slice of length n

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)                # chain rule through softmax

    return grad_v, a, shift

# Adam‑based continuation optimiser (smooth max → true max)
# ----------------------------------------------------------------------

    Returns the best sequence encountered and its exact surrogate value.
    """

            0.5, 1.0, 2.0, 5.0, 10.0,

    t_iter = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break

        # how many iterations can we afford for this beta?
        remaining = deadline - time.time()
        # rough cost per iteration ≈ 2 FFTs → ≈ 1e‑3 s, be conservative:
        est_iter = max(1, int(remaining / 0.0012))
        iters = min(steps_per_beta, est_iter)

        for _ in range(iters):
            if time.time() >= deadline:
                break

            grad_v, a_cur, cur_max = _grad_logits(v, beta, n)

            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_state = beta2 * v_state + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v_state / (1.0 - beta2 ** t_iter)

            # learning rate slowly decays with β (larger β → sharper landscape)
            lr = lr_base / math.sqrt(beta + 1.0)
            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation (every 30 steps)
            if t_iter % 30 == 0:
                cur_seq = _softmax(v)
                cur_val = _objective_surrogate(cur_seq)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()

        # end of this β – a short “cool‑down” evaluation
        cur_seq = _softmax(v)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    # final safeguard evaluation
    cur_seq = _softmax(v)
    cur_val = _objective_surrogate(cur_seq)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.2, 3.5, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma in np.linspace(0.05, 0.35, 12):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 4.0, 12):
        seeds.append(_power(n, p))

    # exponential families (dense sweep)
    for rate in np.logspace(-5, -0.3, 30):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # inverse‑power U‑shapes
    for p in np.linspace(0.3, 2.5, 9):
        seeds.append(_inv_power_shape(n, p))

    # Beta shape grid (focus on U‑shapes)
    alphas = np.geomspace(0.05, 2.0, 9)   # 0.05,0.07,…,2.0
    for a in alphas:
        for b in alphas:
            seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws (various concentrations)
    for conc in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential vectors
    for _ in range(15):
        seeds.append(_random_exponential(n, rng))

    # Random permutations of a few promising bases
    base_shapes = [_exp_decay(n, 0.002), _sym_exp_decay(n, 0.003),
                   _beta_shape(n, 0.5, 0.5), _gaussian(n, 0.12)]
    for base in base_shapes:
        for _ in range(6):
            seeds.append(_permute(base, rng))

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered within the allotted time.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # problem dimension – 5000 provides a fine balance between quality and speed
    n = 5000

    # ------------------------------------------------------------------
    # Load previous solution if present (enables continuation across runs)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_arr = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Massive seed generation
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)

    # optionally seed with the previous best and a few perturbations
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(8):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                seeds.append(noisy / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the top‑k seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() >= deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed screening (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1:02d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Global best initialisation
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (Adam continuation)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fair slice of the remaining budget
        remaining = deadline - time.time()
        # keep about 12 % for each remaining seed + final polish
        share = remaining * 0.12 / max(1, (top_k - idx))
        sub_deadline = time.time() + share

        print(f"\n[Seed {idx + 1}] refinement budget ≈ {share:.1f}s")

        refined_seq, refined_val = _adam_continuation(seed_vec,
                                                      n,
                                                      sub_deadline,
                                                      steps_per_beta=800,
                                                      lr_base=0.07)

        print(f"[Seed {idx + 1}] refined surrogate = {refined_val:.6f}")

        if refined_val < global_best_val:
            global_best_seq = refined_seq
            global_best_val = refined_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long Adam polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (long Adam run) ---")
        final_seq, final_val = _adam_continuation(global_best_seq,
                                                  n,
                                                  deadline,
                                                  steps_per_beta=1500,
                                                  lr_base=0.05)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:            # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
