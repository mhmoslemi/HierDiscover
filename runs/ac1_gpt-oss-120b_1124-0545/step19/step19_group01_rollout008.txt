def _lp_refine(f, n, deadline, max_iter=50, t_start=0.2, t_decay=0.97):
    a = f.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = t_start
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_fft(a, a)
        M = np.max(conv)
        # Build A
        data = [np.full(n - k, a[k]) for k in range(n)]
        offsets = np.arange(n)
        A = scipy.sparse.diags(data, offsets, shape=(2*n-1, n), format='csc')
        b = np.full(2*n-1, M)

        c = -np.ones(n)  # maximize sum(g)
        # bounds for g: (0, None)
        bounds = [(0, None)] * n

        # solve
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                    method='highs', options={'presolve': True, 'time_limit': max(1, deadline - time.time())})
        if not res.success:
            print("LP failed at iteration", it, res.message)
            break
        g = res.x
        sum_g = np.sum(g)
        if sum_g <= 0:
            break
        g_scaled = g * (1.0 / sum_g)  # ensure sum = 1
        # mix
        a = (1 - t) * a + t * g_scaled
        # renormalize to sum =1 (just in case)
        a = a / np.sum(a)
        # update best
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # decay t

"""
Propose a new step‑function sequence that improves the upper bound on
``evaluate_sequence``.

The algorithm works in three stages:

1. **Wide seed generation** – many analytic families (uniform, triangular,
   Gaussian, exponential, beta‑shape, Dirichlet draws, low‑frequency cosine
   curves) plus a novel “random‑phase” Fourier seed are created and
   normalised to sum = 1.  A cheap surrogate ``2·n·max(conv(a,a))`` is used
   to keep the ten best seeds.

2. **Linear‑programming refinement** – starting from each top seed we solve a
   small linear program:
       maximise   Σ g_j
       subject to  f * g ≤ M   (element‑wise),   g ≥ 0
   where * is the linear convolution, ``f`` is the current sequence and
   ``M = max(conv(f,f))``.  The LP moves mass towards positions that keep the
   convolution peak under the current bound.  After solving we normalise and
   mix the new vector with the old one (convex combination with a shrinking
   step‑size).  This step quickly drives the peak down.

3. **Exponential‑gradient polishing** – the best candidate from the LP
   stage is polished by the original mirror‑descent (sub‑gradient on the
   active convolution index).  The polishing runs until the time budget is
   exhausted.

The whole routine respects the 1000 s budget, keeps the best sequence in the
global variable ``height_sequence_1`` and finally evaluates it with the
provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports – used for the linear‑programming refinement.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _linopt
    import scipy.sparse as _sp
except Exception:       # pragma: no cover
    _linopt = None
    _sp = None

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (used during the search)
# Sequence must already be normalised (sum = 1).
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine components, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_phase_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a non‑negative sequence by taking the inverse FFT of a random‑phase
    spectrum with constant magnitude.  The resulting real vector is clipped to
    non‑negative and normalised.
    """
    # Build a Hermitian‑symmetric spectrum so that the IFFT is real.
    freqs = np.zeros(n, dtype=np.complex128)
    # DC component – random positive real value
    freqs[0] = rng.uniform(0.0, 2.0)
    half = n // 2
    # positive frequencies (excluding Nyquist when n even)
    for k in range(1, half):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        freqs[k] = np.exp(1j * phase)               # unit magnitude
        freqs[-k] = np.conj(freqs[k])               # Hermitian symmetry
    if n % 2 == 0:
        # Nyquist frequency must be real
        freqs[half] = rng.uniform(0.0, 2.0)
    seq = np.fft.ifft(freqs).real
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0.0:
        return _uniform(n)
    return seq / s


# ----------------------------------------------------------------------
# Linear‑programming refinement (if SciPy is available)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> _sp.csc_matrix:
    """
    Build the sparse matrix A such that A @ g == np.convolve(f, g, mode='full').
    Size of A is (2·n‑1) × n, where n = len(f).  Each diagonal k (0 ≤ k < n)
    carries the constant value f[k].
    """
    n = f.shape[0]
    # diag k appears on row i = j + k  (offset = k)
    data = [np.full(n - k, f[k], dtype=np.float64) for k in range(n)]
    offsets = np.arange(n, dtype=np.int64)
    A = _sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _lp_refine(start_seq: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 12,
               t_start: float = 0.2,
               t_decay: float = 0.93) -> tuple[np.ndarray, float]:
    """
    Linear‑programming based refinement.
    Repeatedly solves   maximise Σ g_j   s.t.   f * g ≤ M, g ≥ 0,
    where M = max(conv(f,f)).  The solution g is normalised to sum = 1 and
    mixed with the current f using a decreasing step size t.
    """
    if _linopt is None or _sp is None:
        # SciPy not present – skip LP refinement.
        return start_seq, _objective_surrogate(start_seq)

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = t_start

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current bound
        conv = _conv_fft(a, a)
        M = np.max(conv)

        # build A @ g ≤ M
        A = _build_conv_matrix(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise Σ g  → minimise −Σ g
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        # Solve the LP (HiGHS is fast and deterministic)
        res = _linopt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                              method='highs', options={'presolve': True})
        if not res.success:
            # If LP fails we abort the refinement.
            print(f"[LP] Iteration {it+1}: linprog failed – {res.message}")
            break

        g = res.x
        s_g = np.sum(g)
        if s_g <= 0.0:
            break
        # Normalise to unit sum.
        g = g / s_g

        # Convex mixing
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        s_a = np.sum(a)
        if s_a == 0.0:
            a = _uniform(n)
        else:
            a = a / s_a

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # decrease step size
        t *= t_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) polishing
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 250_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient at the active
    index).  Works in the log‑space of the simplex to keep iterates feasible.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (exact formula for linear conv)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = np.exp(log_a)
        a = a / a.sum()                       # simplex projection (softmax)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

                                    n,
                                    sub_deadline,
                                    max_iters=12,
                                    t_start=0.25,
                                    t_decay=0.92)

        # Update global best if LP already improves.
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"      LP improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=500_000,
                                                 eta0=6.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
