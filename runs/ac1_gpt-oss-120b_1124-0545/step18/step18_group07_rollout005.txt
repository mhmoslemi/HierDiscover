def simulated_annealing(a_init, n, deadline, max_iters=200_000):
    # start with current a
    cur = a_init.copy()
    cur_val = 2 * n * np.max(_conv_fft(cur, cur))
    best = cur.copy()
    best_val = cur_val
    # precompute random generator
    rng = np.random.default_rng()
    # temperature schedule: linearly decreasing from T0 to 0
    T0 = 0.01 * cur_val  # starting tolerance
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # choose source i with probability proportional to cur[i]
        i = rng.choice(n, p=cur)
        # choose target j uniformly random (different)
        j = rng.integers(0, n-1)
        if j == i:
            continue
        # transfer epsilon: random fraction of cur[i]
        eps = cur[i] * rng.uniform(0.001, 0.05)  # 0.1% to 5% of source mass
        # propose new distribution
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        # ensure non-negative (should be)
        # compute conv
        new_val = 2 * n * np.max(_conv_fft(new, new))
        delta = new_val - cur_val
        if delta < 0:
            accept = True
        else:
            # temperature
            frac = 1.0 - (time.time() - start_time) / (deadline - start_time)
            T = T0 * frac
            accept = rng.random() < math.exp(-delta / (T + 1e-12))
        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()
        # optionally occasionally perform random reset to best
        if it % 10000 == 0:
            # attempt to escape local minima by small random jitter
            cur = best.copy()
            cur_val = best_val
    return best, best_val

"""
Propose a better step‑function for ``evaluate_sequence``.

Approach
--------
1. Build a diverse pool of ℓ₁‑normalised “seeds’’ (analytic windows,
   Dirichlet draws, edge‑biased shapes and the previous solution if any).
2. Keep the few seeds with the smallest surrogate ``2·n·max(conv)``.
3. For each promising seed run a *continuation* L‑BFGS optimiser:
   we smooth the non‑smooth ``max`` by a log‑sum‑exp surrogate and
   optimise the resulting smooth objective in the log‑probability
   domain (soft‑max → simplex).  A generous β‑schedule and a large
   per‑β iteration budget are used.
4. Polishing: exact mirror‑descent (exponentiated‑gradient) on the true
   max‑convolution.  The step‑size follows a 1/√t schedule with a
   modest over‑relaxation factor.
5. Global stochastic refinement: a simple simulated‑annealing move set
   that transfers a random fraction of mass between two positions.
   Only improving moves are accepted, but with a slowly decreasing
   temperature we occasionally accept uphill moves to escape local minima.
6. Final short mirror‑descent pass.
7. Return the best sequence found and store it in the global variable
   ``height_sequence_1`` for the next round.

All steps respect the global 1000 s wall‑clock budget, use only
numpy/scipy and the provided ``evaluate_sequence`` function, and never
return negative numbers.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    # guard against all‑zero (should not happen)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _max_conv(seq: np.ndarray) -> float:
    """Maximum of the full auto‑convolution of ``seq`` (seq ≥ 0, sum = 1)."""
    return float(np.max(_conv_fft(seq, seq)))


def _surrogate(seq: np.ndarray) -> float:
    """Surrogate objective = 2·n·max(conv)  (n = len(seq), sum(seq)=1)."""
    n = len(seq)
    return 2.0 * n * _max_conv(seq)


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_bias(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))
    for sigma in np.linspace(0.10, 0.45, 5):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 7):
        seeds.append(_power(n, p))
    for a in (0.05, 0.1, 0.3, 1.0, 2.0):
        for b in (0.05, 0.1, 0.3, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))
    for p in (0.25, 0.5, 0.75, 1.0):
        seeds.append(_edge_bias(n, p))

    # random draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))
    for bf in (0.05, 0.15, 0.30):
        seeds.append(_block_uniform(n, bf, rng))

    # previous solution (resampled if length differs)
    if previous is not None:
        if previous.size == n:
            seeds.append(previous / previous.sum())
        else:
            # simple linear interpolation / truncation to the new size
            x_old = np.linspace(0, 1, previous.size)
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, previous)
            if interp.sum() > 0:
                seeds.append(interp / interp.sum())
    # sanitise
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned

# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       beta_schedule=None,
                       max_iter_per_beta: int = 12000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a probability vector (ℓ₁‑normalised).
    """
    if _opt is None:
        # SciPy unavailable – fall back to the start vector
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6
        ]

    # work in log‑probability space (stay on simplex via softmax)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # abort if we are out of time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000,
                     overrelax: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size  (classical 1/√t schedule, slightly enlarged)
        eta = overrelax * 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report
        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best_surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of probability mass between two positions.
    Accept uphill moves with a decreasing temperature to escape local minima.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    best = cur.copy()
    best_val = cur_val
    start_time = time.time()

    # initial temperature as a small fraction of the current objective
    T0 = max(1e-8, 0.02 * cur_val)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # pick source proportional to its mass (more likely to move mass from larger bins)
        i = rng.choice(n, p=cur)
        # pick a distinct target uniformly
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the source mass
        eps = cur[i] * rng.uniform(0.001, 0.04)   # 0.1 % – 4 % of source
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps

        cand_val = _surrogate(cand)
        delta = cand_val - cur_val

        # annealing acceptance
        if delta < 0.0:
            accept = True
        else:
            # linear cooling schedule
            elapsed = time.time() - start_time
            remaining = max(1e-9, deadline - start_time - elapsed)
            frac = remaining / (deadline - start_time)
            T = T0 * frac
            accept = rng.random() < math.exp(-delta / (T + 1e-12))

        if accept:
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()

        # every 20 k steps we re‑centre the walk on the current best
        if it % 20_000 == 0:
            cur = best.copy()
            cur_val = best_val

    return best, best_val
