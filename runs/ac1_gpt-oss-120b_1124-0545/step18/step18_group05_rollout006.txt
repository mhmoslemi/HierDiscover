"""
Propose a non‑negative step function that minimizes

        2·n·max(conv(a, a)) / (sum(a)²)

The algorithm is a multi‑stage optimiser:

1️⃣  Build a diverse seed pool (analytic families + Dirichlet draws) and keep the
    best seed w.r.t. the exact surrogate 2·n·max(conv).

2️⃣  Optimise the surrogate with an Adam‑style mirror‑descent on the logit
    (soft‑max) parametrisation.  The sub‑gradient is taken w.r.t. the currently
    active convolution index.  Adam’s adaptive step‑size gives a much faster
    descent than plain √t‑decay.

3️⃣  Refine the result with L‑BFGS‑B on a smooth log‑sum‑exp approximation of the
    max‑convolution.  The β‑schedule is continued up to 5·10⁶ so the surrogate
    is virtually the true max.

4️⃣  A short stochastic‑annealing walk in logit space finalises the solution
    (small temperature, many iterations).  The best vector seen throughout the
    run is stored globally as ``height_sequence_1`` for the next invocation.

All stages respect the overall time budget (default 1000 s).  The code works
with or without SciPy – if SciPy is missing the L‑BFGS stage is skipped.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used only for the L‑BFGS polish)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:               # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector, sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return a / a.sum()


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    return a / a.sum()


# ----------------------------------------------------------------------
# Exact surrogate (the objective when the vector is ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# L‑BFGS optimiser on a smooth max‑convolution surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 3000) -> np.ndarray:
    """Polish ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:                                   # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0, 2_000_000.0,
            5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # ---- smooth max via log‑sum‑exp ----
            shifted = conv - max_c               # max becomes 0
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            # ---- gradient w.r.t. a (smooth) ----
            w = exp_beta / sum_exp               # soft weights, sum = 1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # central slice

            # ---- chain rule through soft‑max ----
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                     # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing walk in logit space (final polishing)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Short SA walk (logits) – returns best vector and its surrogate value."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6           # cooling
        sigma = 0.15 * (1.0 - progress) + 1e-4       # step size

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam‑style mirror‑descent on the exact surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the exact non‑smooth surrogate.
    Returns the best vector seen and its surrogate value.
    """
    rng = np.random.default_rng()
    v = np.log(start_seq + 1e-30)          # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(v)

        # full convolution
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t. logits (mirror‑descent)
        grad_v = a * (g - np.dot(g, a))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)

        # a slowly decaying learning‑rate
        alpha = 0.3 / (1.0 + 0.0005 * t)

        v -= alpha * m_hat / (np.sqrt(v_hat_corr) + eps)

        # evaluate surrogate and keep the best
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report
        if t % 20000 == 0:
            elapsed = time.time() - (deadline - (budget_s - 2.0))
            print(f"[Adam] it={t:,} best surrogate={best_val:.6f} elapsed={elapsed:.1f}s")

    return best_seq, best_val

def propose_candidate(seed: int = 123,

    """

    """

        if s <= 0.0:

    # ---------------------------------------------------------------

    # ---------------------------------------------------------------
    if time.time() < deadline and _opt is not None:

        # give L‑BFGS a reasonable slice of the remaining budget

        print("\n--- L‑BFGS polish (smooth surrogate) ---")
        seq_lbfgs = _lbfgs_optimize(best_seq, n, deadline,

                                                   50_000, 100_000,
                                                   200_000, 500_000,
                                                   1_000_000, 2_000_000,

                                    max_iter_per_beta=max_iter_per_beta)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:

            print(f"[Update] new global best after L‑BFGS = {best_val:.6f}")

    # ---------------------------------------------------------------

    return best_seq.tolist()
