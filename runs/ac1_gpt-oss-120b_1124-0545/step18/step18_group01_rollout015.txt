"""
Propose a candidate step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the full linear convolution.  The search works in three
phases:

1. **Broad cheap sampling** – analytical families (uniform, linear,
   exponential, power‑law) and a large random‑search over mixtures of two
   Gaussian “bumps”.  All candidates are ℓ₁‑normalised (∑a = 1) and
   evaluated with the exact surrogate 2·n·max(conv).

2. **Local refinement** – the best few seeds are polished with a smooth
   log‑sum‑exp surrogate of the max‑convolution (continuation L‑BFGS‑B)
   inside the log‑softmax space so the positivity constraint is
   automatically satisfied.

3. **Final polishing** – an exponentiated‑gradient (mirror‑descent)
   pass on the exact max‑convolution from the best refined candidate.

The routine respects the supplied time budget (default 1000 s) and stores
the best sequence found in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_exact(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decrease(n: int) -> np.ndarray:
    """a_i ∝ (n‑i).  Decreasing linear shape."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i).  Normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (decreasing power‑law)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _gaussian_mixture(n: int,
                      mu1: float, sigma1: float,
                      mu2: float, sigma2: float,
                      w1: float) -> np.ndarray:
    """Two‑Gaussian mixture, normalised (w2 = 1‑w1)."""
    i = np.arange(n, dtype=np.float64)
    g1 = np.exp(-((i - mu1) ** 2) / (2.0 * sigma1 ** 2))
    g2 = np.exp(-((i - mu2) ** 2) / (2.0 * sigma2 ** 2))
    a = w1 * g1 + (1.0 - w1) * g2
    s = a.sum()
    return a / s


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and L‑BFGS optimisation
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Optimise a smooth approximation of max‑conv using L‑BFGS‑B."""
    if _opt is None:
        return start_vec  # SciPy missing – return unchanged.

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    # work in log‑softmax space to keep a ≥ 0, Σa = 1
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            # smooth max = shift + log‑sum‑exp / β
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                        # shape (2n‑1,)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                    # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # project onto simplex (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_exact(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        # active index (the one attaining the max)
        k = int(np.argmax(conv))

        # Sub‑gradient of the active entry.
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_exact(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
