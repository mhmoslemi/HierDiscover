"""
Improved search for the step‑function that minimises ``evaluate_sequence``.

The optimiser works in several layers:

1. **Diversified seed pool** – classic families (uniform, triangular,
   Gaussian, beta, exponential, power‑law, random Dirichlet, sparse,
   low‑frequency) plus many random mixtures.  All seeds are ℓ₁‑normalised
   to the required mass ``sqrt(2·n)``.

2. **Global random‑parameter search** – a fast Monte‑Carlo sweep over
   five analytic families (power‑law, exponential, Gaussian, beta,
   Dirichlet) with wide parameter ranges.  The best candidates are kept.

3. **Fine‑grained hill‑climbing on the power‑law parameters** – starting
   from the best power‑law found, the exponent and shift are locally
   optimised (alternating step reductions) while the shape is kept
   ℓ₁‑normalised.

4. **Mirror‑descent on the exact nonsmooth objective** – an exponentiated‑gradient
   (a.k.a. mirror descent) run for as many iterations as time permits.
   It uses the true sub‑gradient of ``max(conv(a,a))`` and thus can
   escape the small basin left by the previous stages.

5. **Aggressive stochastic mass‑transfer local search** – a simple hill‑climb
   that repeatedly moves a tiny fraction of mass from one index to another.
   Even a handful of thousand moves can tighten the final bound.

All stages constantly update the global best sequence.  The routine respects
the 1000 s wall‑time limit, stores the best result in the global variable
``height_sequence_1`` and finally returns it as a plain list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective & helpers (max‑convolution value)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)².  For our normalisation this equals max(conv)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (active lag only)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Mirror‑descent on the exact (nonsmooth) objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) minimisation of the exact max‑conv."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = np.exp(log_a)
        a = a / a.sum() * np.sum(start_a)          # renormalise to the same ℓ₁‑mass

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val

# Stochastic mass‑transfer hill‑climb
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 120_000,
    step_frac: float = 0.011,
) -> tuple[np.ndarray, float]:
    """Simple hill‑climb: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)

def _collect_seeds(

    seeds: list[np.ndarray] = []

    # Deterministic families

    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))

    # Power‑law sweep – extended ranges

    for p in p_vals:

            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # Random Dirichlet / sparse families
    for conc in (0.02, 0.05, 0.2, 0.5, 1.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixtures (helps explore convex hull)
    base = np.arange(len(seeds))
    for _ in range(150):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of promising seeds
    for _ in range(15):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Include previous best if available
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean‑up: non‑negativity + ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

        log_i = np.log(idx[mask] + 1.0)
        # Linear regression: log a ≈ -p * log(i+shift)  → we just fit slope
        # Using least squares on log–log gives an estimate for p.
        p_est = -np.polyfit(log_i, log_a, 1)[0]
        p_est = max(0.05, min(2.0, p_est))
        # shift is trickier; we set it to zero for now
        return p_est, 0.0

    # If the best shape is roughly monotone, try a focused refinement
    if time.time() < deadline - 5.0:
        # Derive a crude initial guess from the best sequence
        init_p, init_sh = _fit_powerlaw_to(global_best_seq)
        p, sh = init_p, init_sh
        step_p = 0.1
        step_sh = 10.0
        print(f"[Refine] starting power‑law refinement from p={p:.4f}, shift={sh:.1f}")
        while (step_p > 1e-4 or step_sh > 0.5) and time.time() < deadline - 2.0:
            improved = False
            # try a small grid around current (p,sh)
            for dp in (-step_p, 0.0, step_p):
                for dsh in (-step_sh, 0.0, step_sh):
                    if dp == 0.0 and dsh == 0.0:
                        continue
                    p2 = max(0.01, p + dp)
                    sh2 = max(0.0, sh + dsh)
                    cand = _power_law(n, p2, sh2) * target_sum
                    val = _objective_exact(cand)
                    if val < global_best_val:
                        global_best_val = val
                        global_best_seq = cand.copy()
                        p, sh = p2, sh2
                        improved = True
                        print(f"  [Refine] improved to {global_best_val:.6f} (p={p:.4f}, sh={sh:.1f})")
            if not improved:
                step_p *= 0.5
                step_sh *= 0.5

    # ------------------------------------------------------------------
    # Stage 4 – mirror‑descent (exponentiated gradient) on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # allocate as many iterations as we can afford (~0.2 µs per iteration in practice)
        max_iters = int(min(8_000_000, remaining * 4_000))
        print(f"\n--- Mirror‑descent (exp‑gradient) start – up to {max_iters:,} iters ---")
        seq_eg, val_eg = _exp_gradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=max_iters,
            eta0=1.8,
        )
        print(f"[Mirror] best value = {val_eg:.6f}")
        if val_eg < global_best_val:
            global_best_val = val_eg
            global_best_seq = seq_eg.copy()
            print(f"[Update] New global best after mirror descent = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – aggressive stochastic local search (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        remaining = deadline - time.time()
        iters = int(min(250_000, remaining * 40))
        print(f"\n--- Stochastic local search – {iters:,} moves ---")
        seq_loc, val_loc = _local_search(
            global_best_seq,
            n,
            deadline,
            iters=iters,
            step_frac=0.012,
        )
        print(f"[LocalSearch] best = {val_loc:.6f}")
        if val_loc < global_best_val:
            global_best_val = val_loc
            global_best_seq = seq_loc.copy()
            print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
