Improved step‑function search for minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of:

1. **Diverse seeding** – deterministic analytic families, a systematic
   sweep over a few parametric families (U‑shaped “edge‑block”,
   generalized arcsine, beta‑shape, etc.) and a large set of random

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Exact surrogate (used while searching – sum = 1 assumed)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """Exact objective for a probability vector (∑ vec = 1):
       2·n·max(conv(vec, vec))
    """
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _double_power(n: int, p: float) -> np.ndarray:
    """Symmetric sum of two power laws – large at both ends."""
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a = np.where(np.isfinite(a), a, 0.0)
    return _normalize(a)


def _arcsine_window(n: int) -> np.ndarray:
    """U‑shaped arcsine‑like density  (≈1/√(x·(1‑x)))"""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _gen_arcsine_power(n: int, gamma: float) -> np.ndarray:
    """Generalised arcsine: (i·(n+1‑i))⁻ᵍᵃᵐᵐᵃ."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i * (n + 0.5 - i)) ** (-gamma)
    a = np.where(np.isfinite(a), a, 0.0)
    return _normalize(a)


def _edge_block_uniform_with_interior(n: int, block_ratio: float, edge_weight: float) -> np.ndarray:
    """
    Two small uniform blocks on the two ends (total edge weight = edge_weight)
    and a uniform interior for the remaining mass.
    block_ratio – fraction of total length occupied by each end block.
    edge_weight – total mass placed on both edge blocks (0 < edge_weight < 1).
    """
    L = max(1, int(block_ratio * n))
    interior_len = n - 2 * L
    if interior_len <= 0:
        interior_len = 0
    a = np.zeros(n, dtype=np.float64)
    # edge mass equally split left/right
    edge_per_block = (edge_weight / 2.0) / L if L > 0 else 0.0
    a[:L] = edge_per_block
    a[-L:] = edge_per_block
    # interior mass
    interior_mass = 1.0 - edge_weight
    if interior_len > 0:
        a[L:-L] = interior_mass / interior_len
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with all concentration parameters = conc."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves; cut to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose a few disjoint blocks and put uniform mass on each."""
    nb = rng.integers(2, 6)                 # number of blocks
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for _ in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()      # share of the remaining mass
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth surrogate via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective with L‑BFGS‑B.
    Operates in log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        # a rather aggressive schedule that pushes the smooth max very close
        beta_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 1000.0,
                         5000.0, 20000.0, 1e5, 5e5,
                         1e6, 5e6, 1e7]

    # initialise logits (log‑softmax variables)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)           # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            w = exp_beta / sum_exp          # soft‑max weights over conv entries
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb (mass transfer based on sub‑gradient)
# ----------------------------------------------------------------------
def _gradient_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Move a (rather large) fraction of mass from a coordinate with a large
    sub‑gradient (i.e. contributing strongly to the current maximal
    convolution entry) to a coordinate with a tiny sub‑gradient.
    Only improvements (exact surrogate) are accepted.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        # current maximum index and sub‑gradient
        conv = _conv_fft(cur, cur)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a (only the active index matters)
        g = np.zeros_like(cur)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * cur[k_max - idx]

        # choose source (large g·a) and destination (small g)
        src = int(np.argmax(g * cur))
        dst = int(np.argmin(g))

        if src == dst or cur[src] <= 0.0:
            # fallback to a random move if the gradient gives a degenerate pair
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            src, dst = i, j

        # transfer a random *large* fraction of source mass
        frac = rng.uniform(0.1, 0.5)          # 10 %‑50 %
        eps = cur[src] * frac
        if eps <= 0.0:
            continue

        new = cur.copy()
        new[src] -= eps
        new[dst] += eps
        new_val = _objective_surrogate(new)

        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
                # occasional debug print (commented out for speed)
                # print(f"[Hill‑climb] new best {best_val:.6f}")
        else:
            # occasional tiny random perturbation to escape tiny plateaus
            if rng.random() < 0.001:
                i, j = rng.integers(n, size=2)
                if i == j:
                    continue
                eps2 = cur[i] * rng.uniform(0.001, 0.01)
                cand = cur.copy()
                cand[i] -= eps2
                cand[j] += eps2
                cand_val = _objective_surrogate(cand)
                if cand_val < cur_val:
                    cur = cand
                    cur_val = cand_val
                    if cand_val < best_val:
                        best = cand
                        best_val = cand_val

    return best, best_val


# ----------------------------------------------------------------------
# Main proposal function
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The routine respects the time budget,
    prints progress information, and stores the best vector in the global
    variable ``height_sequence_1`` for the next invocation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # safety margin for the final call

    # ------------------------------------------------------------------
    # 1. Problem size (trade‑off quality vs speed)
    # ------------------------------------------------------------------
    n = 4000
    # ------------------------------------------------------------------
    # 2. Keep the best ``top_k`` seeds (exact surrogate)
    # ------------------------------------------------------------------
    top_k = 30
    best_heap = []          # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert a candidate into the heap while keeping only the best ``top_k``."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # 3. Deterministic analytical seeds
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    _push_candidate(_arcsine_window(n))
    for gamma in np.linspace(0.5, 4.0, 9):
        _push_candidate(_gen_arcsine_power(n, gamma))
    for sigma in np.linspace(0.07, 0.25, 12):
        _push_candidate(_gaussian(n, sigma))
    for p in np.linspace(0.05, 3.0, 12):
        _push_candidate(_power(n, p))
        _push_candidate(_double_power(n, p))
    for beta in np.linspace(1.0, 15.0, 8):
        # Kaiser window (beta≈2..15 gives a variety of tapers)
        try:
            _push_candidate(_kaiser_window(n, beta))
        except Exception:  # pragma: no cover
            pass
    # Edge‑block with interior mass – systematic sweep
    for blk in np.linspace(0.01, 0.15, 7):
        for ew in np.linspace(0.2, 0.8, 7):
            _push_candidate(_edge_block_uniform_with_interior(n, blk, ew))

    # ------------------------------------------------------------------
    # 4. Random seeding (Dirichlet draws with a wide span of concentrations)
    # ------------------------------------------------------------------
    conc_choices = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_seed_budget = 4000
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # occasional cyclic shift – changes the correlation pattern
        if rng.random() < 0.25:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # a few exponential draws (heavier tails)
    for _ in range(500):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        if rng.random() < 0.25:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # low‑frequency random curves
    for nf in (2, 3, 5, 8):
        for _ in range(200):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            if rng.random() < 0.25:
                shift = rng.integers(n)
                if shift:
                    vec = np.roll(vec, shift)
            _push_candidate(vec)

    # mixed block structures
    for _ in range(300):
        if time.time() > deadline:
            break
        _push_candidate(_random_mixed_blocks(n, rng))

    # ------------------------------------------------------------------
    # 5. Re‑use the previous solution (if compatible)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and np.sum(prev_arr) > 0:
                _push_candidate(_normalize(prev_arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] retained {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 7. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Local refinements – continuation L‑BFGS on the top few seeds
    # ------------------------------------------------------------------
    max_refinements = min(5, len(seeds))
    for idx in range(max_refinements):
        if time.time() > deadline - 8.0:
            break
        seed_vec = seeds[idx]
        # allocate a share of the remaining time (≈55 % for refinements)
        remaining = deadline - time.time()
        refine_deadline = time.time() + max(6.0, remaining * 0.55 / max_refinements)

        try:
            refined = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             refine_deadline,
                                             beta_schedule=[0.5, 2.0, 10.0, 50.0,
                                                            200.0, 1e3,
                                                            5e3, 2e4,
                                                            1e5, 5e5,
                                                            1e6, 5e6,
                                                            1e7],
                                             max_iter_per_beta=8000)
        except RuntimeError:
            # SciPy missing – fall back to keeping the seed unchanged
            refined = seed_vec
        refined_val = _objective_surrogate(refined)
        if refined_val < global_best_val:
            global_best_seq = refined
            global_best_val = refined_val
            print(f"[Refine {idx + 1}] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Gradient‑guided hill‑climb (final local search)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n--- Gradient‑guided hill‑climb ---")
        seq_hc, val_hc = _gradient_hill_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
