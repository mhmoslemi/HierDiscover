Improved search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm expands on the previous version by:

* keeping a larger pool of promising seeds (top 30 instead of 20);
* using a longer and more aggressive L‑BFGS continuation with a finer
  beta‑schedule and a higher ``max_iter_per_beta``;

* a small amount of random mutation/re‑optimisation on the current best

All helpers are top‑level functions – no closures or lambdas – and the

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _objective_surrogate(vec: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv(vec, vec)) for a probability vector."""
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))

# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Optimise the smoothed max‑convolution objective with L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 10000.0, 20000.0,
                         50000.0, 100000.0, 200000.0, 500000.0,
                         1_000_000.0, 2_000_000.0]

    v = np.log(start_vec + 1e-30)      # logits (soft‑max space)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                     # soft‑max weights over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # ∂obj/∂a_i

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------
def _subgrad_optimize(start_vec: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 20000) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on 2·n·max(conv(a,a)).
    The gradient of a single convolution entry (the max) is
        g_i = 2 * a_{k - i}     (if 0 ≤ k−i < n, else 0)
    where k is a shift attaining the maximum.
    """
    a = _normalize(start_vec.copy())
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        ks = np.where(conv == max_val)[0]
        # pick a random arg‑max to avoid cycling
        k = ks[np.random.randint(len(ks))]

        # compute gradient g_i = 2 * a_{k-i} where defined
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # step size – classic 1/√t schedule
        step = 0.15 / np.sqrt(it)

        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        a = _normalize(a_new)

        # occasional early‑stop check
        if it % 1000 == 0:
            # if the objective hasn't improved noticeably, we may stop
            cur_val = _objective_surrogate(a)
            # (nothing special – just keep going until time runs out)
            pass
    return a


# ----------------------------------------------------------------------
# Greedy hill‑climb (tiny + medium mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Randomised hill‑climb that moves a fraction of mass between two positions."""
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # Choose a transfer size – sometimes tiny, sometimes moderate
        eps_frac = rng.uniform(0.001, 0.02)      # 0.1 % … 2 % of source mass
        eps = cur[i] * eps_frac
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Helpers for seeding
# ----------------------------------------------------------------------
def _push_candidate(vec: np.ndarray, heap: list, top_k: int):
    """Insert a candidate into the heap while keeping only the ``top_k`` best."""
    val = _objective_surrogate(vec)
    if len(heap) < top_k:
        heapq.heappush(heap, (-val, vec))
    else:
        if val < -heap[0][0]:
            heapq.heappushpop(heap, (-val, vec))


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, prints progress information,
    and stores the best vector in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation call and a tiny buffer
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size – a compromise between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Heap for the best seeds (larger pool than the previous version)
    # ------------------------------------------------------------------
    top_k = 30
    best_heap: list[tuple[float, np.ndarray]] = []   # stores (-value, vector)

    # ------------------------------------------------------------------
    # 3. Analytic families (wide variety)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n), best_heap, top_k)
    _push_candidate(_triangular(n), best_heap, top_k)
    for gamma in np.linspace(0.2, 3.0, 12):
        _push_candidate(_sine_window(n, gamma), best_heap, top_k)
    _push_candidate(_hamming_window(n), best_heap, top_k)
    _push_candidate(_blackman_window(n), best_heap, top_k)
    _push_candidate(_nuttall_window(n), best_heap, top_k)
    for beta in np.linspace(1.0, 15.0, 10):
        _push_candidate(_kaiser_window(n, beta), best_heap, top_k)
    for sigma in np.linspace(0.08, 0.30, 15):
        _push_candidate(_gaussian(n, sigma), best_heap, top_k)
    for p in np.linspace(0.1, 3.0, 15):
        _push_candidate(_power(n, p), best_heap, top_k)
        _push_candidate(_double_power(n, p), best_heap, top_k)
    # beta‑shape – include many heavy‑tail variants
    for _ in range(30):
        a = 10 ** rng.uniform(-2, 0)
        b = 10 ** rng.uniform(-2, 0)
        _push_candidate(_beta_shape(n, a, b), best_heap, top_k)
    _push_candidate(_arcsine_window(n), best_heap, top_k)
    for gamma in np.linspace(0.5, 4.0, 9):
        _push_candidate(_cosine_edge_window(n, gamma), best_heap, top_k)
    for _ in range(20):
        blk = rng.uniform(0.05, 0.3)
        lw = rng.random()
        _push_candidate(_two_block_uniform(n, block_ratio=blk, left_weight=lw),
                        best_heap, top_k)

    # ------------------------------------------------------------------
    # 4. Massive random seeding
    # ------------------------------------------------------------------
    conc_choices = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_seed_budget = 6000
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # occasional cyclic shift – changes the correlation pattern
        if rng.random() < 0.25:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec, best_heap, top_k)

    # a few exponential draws (heavier tails)
    for _ in range(800):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        if rng.random() < 0.25:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec, best_heap, top_k)

    # low‑frequency random curves
    for nf in (2, 3, 5, 8):
        for _ in range(250):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            if rng.random() < 0.2:
                shift = rng.integers(n)
                if shift:
                    vec = np.roll(vec, shift)
            _push_candidate(vec, best_heap, top_k)

    # mixed block structures
    for _ in range(300):
        if time.time() > deadline:
            break
        _push_candidate(_random_mixed_blocks(n, rng), best_heap, top_k)

    # ------------------------------------------------------------------
    # 5. Re‑use the previous solution if it matches the current size
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and np.sum(prev_arr) > 0:
                _push_candidate(_normalize(prev_arr), best_heap, top_k)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] kept {len(seeds)} candidates, best surrogate = "
          f"{_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 7. Global best initialisation
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Local refinements – L‑BFGS + sub‑gradient polishing
    # ------------------------------------------------------------------
    max_refinements = min(8, len(seeds))   # a few top seeds get full attention
    for idx in range(max_refinements):
        if time.time() > deadline - 5.0:
            break
        seed_vec = seeds[idx]

        # allocate a slice of remaining time for this seed
        remaining = deadline - time.time()
        # keep a small safety margin for the final hill‑climb
        refine_deadline = time.time() + max(8.0, remaining * 0.5 / max_refinements)

        # ---- L‑BFGS continuation ----
        try:
            refined = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             refine_deadline,
                                             beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                            200.0, 500.0,
                                                            1000.0, 2000.0,
                                                            5000.0, 10000.0,
                                                            20000.0, 50000.0,
                                                            100000.0, 200000.0,
                                                            500000.0,
                                                            1_000_000.0],
                                             max_iter_per_beta=8000)
        except RuntimeError:
            refined = seed_vec

        # ---- sub‑gradient polishing ----
        refined = _subgrad_optimize(refined, n, refine_deadline)

        refined_val = _objective_surrogate(refined)
        if refined_val < global_best_val:
            global_best_seq = refined
            global_best_val = refined_val
            print(f"[Refine {idx+1}] improved to {global_best_val:.6f}")

        # ---- occasional random mutation on the current best ----
        if idx == max_refinements - 1 and time.time() < deadline - 2.0:
            # add a tiny Gaussian perturbation, re‑normalise and polish again
            noise = rng.normal(loc=0.0, scale=0.01, size=n)
            mutated = _normalize(np.maximum(global_best_seq + noise, 0.0))
            mutated = _lbfgs_smooth_optimize(mutated, n,
                                            deadline - 1.0,
                                            beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                           200.0, 500.0,
                                                           1000.0, 2000.0,
                                                           5000.0, 10000.0],
                                            max_iter_per_beta=4000)
            mutated = _subgrad_optimize(mutated, n, deadline - 0.5)
            mutated_val = _objective_surrogate(mutated)
            if mutated_val < global_best_val:
                global_best_seq = mutated
                global_best_val = mutated_val
                print(f"[Mutation] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Final greedy hill‑climb (use any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final hill‑climb ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
