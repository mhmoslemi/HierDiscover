def _collect_seeds_extended(n, rng, previous):
    seeds = []

    # basic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    # sine
    for g in (0.2,0.5,1.0,2.0,4.0,8.0):
        seeds.append(_sine_window(n, gamma=g))
    # windows: hamming, blackman, nuttall
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    # kaiser
    for beta in (1,2,4,6,10,12,16,20,30,40):
        seeds.append(_kaiser_window(n,beta))
    # gaussian
    for sigma in np.linspace(0.05, 0.5, 9):
        seeds.append(_gaussian(n, sigma))
    # double gaussian
    for sigma in (0.05, 0.1, 0.12):
        for shift in (0.15,0.2,0.25,0.3):
            seeds.append(_double_gaussian(n, sigma, shift))
    # power
    for p in np.linspace(-1.0,5.0,13):
        seeds.append(_power(n, p))
    # beta shape extremes
    alphas = (0.01,0.05,0.1,0.3,0.5,1,2,5)
    betas = (0.01,0.05,0.1,0.3,0.5,1,2,5)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    # tukey windows
    for alpha in np.linspace(0.05, 0.95, 10):
        seeds.append(_tukey_window(n, alpha))
    # edge bias
    for p in np.linspace(0.1, 3.0, 15):
        seeds.append(_edge_bias(n, p))
    # lowfreq curves
    for nf in (2,3,4,5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))
    # random Dirichlet
    for conc in (0.01,0.05,0.1,0.2,0.5,1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    # random exponential
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    # sparse random
    for k in (max(1, int(0.001*n)), max(1, int(0.005*n))):
        for _ in range(2):
            seeds.append(_sparse_random(n,k,rng))
    # block uniform
    for bf in (0.02,0.05,0.1,0.2):
        for _ in range(2):
            seeds.append(_block_uniform(n,bf,rng))
    # mixtures with uniform
    for shape_func in [_edge_bias, _beta_shape, _gaussian]:
        if shape_func is _beta_shape:
            for a in (0.05,0.5,1.0):
                for b in (0.05,0.5,1.0):
                    shape = shape_func(n,a,b)
                    for alpha in np.linspace(0.1,0.9,5):
                        mix = _mix_shapes(_uniform(n), shape, alpha)
                        seeds.append(mix)
        else:
            for param in (0.2,0.5,1.0):
                shape = shape_func(n, param) if shape_func is not _beta_shape else shape_func(n,param,param)
                for alpha in np.linspace(0.1,0.9,5):
                    seeds.append(_mix_shapes(_uniform(n), shape, alpha))
    # alternating Dirichlet
    seeds.append(_alternating_dirichlet(n, rng))
    # previous solution
    if previous is not None and previous.shape[0]==n and previous.sum()>0:
        seeds.append(previous / previous.sum())

    # Normalize and clip any invalid
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:

        2·n·max(conv(seq, seq)).

    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.12,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _edge_bias(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution concentrating mass near the ends."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # --- Simple analytic families ------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for g in (0.2, 0.5, 1.0, 2.0, 4.0, 8.0):
        seeds.append(_sine_window(n, gamma=g))

    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    for beta in (1, 2, 4, 6, 10, 12, 16, 20, 30, 40):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.05, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    # a few double‑gaussians with varied shift / width
    for sigma in (0.05, 0.1, 0.12):
        for shift in (0.15, 0.2, 0.25, 0.3):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))

    # power law – include negative exponents (heavy centre) and large p (U‑shape)
    for p in np.linspace(-1.0, 5.0, 13):
        seeds.append(_power(n, p))

    # beta‑shape grid – go deep into the “U” region
    for a in (0.01, 0.05, 0.1, 0.3, 1.0, 2.0, 5.0):
        for b in (0.01, 0.05, 0.1, 0.3, 1.0, 2.0, 5.0):
            seeds.append(_beta_shape(n, a, b))

    # tukey windows – many alphas
    for alpha in np.linspace(0.05, 0.95, 10):
        seeds.append(_tukey_window(n, alpha))

    # edge‑biased families (U‑shaped)
    for p in np.linspace(0.1, 3.0, 15):
        seeds.append(_edge_bias(n, p))

    # low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Sparse random vectors (very few spikes)
    for k in (max(1, int(0.001 * n)), max(1, int(0.005 * n))):
        for _ in range(2):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors
    for bf in (0.02, 0.05, 0.1, 0.2):
        for _ in range(2):
            seeds.append(_block_uniform(n, bf, rng))

    # Mixtures with uniform (uniform + a structured shape)
    for shape_func in [_edge_bias, _gaussian]:
        for param in (0.2, 0.5, 1.0):
            sh = shape_func(n, param)
            for alpha in np.linspace(0.1, 0.9, 5):
                seeds.append(_mix_shapes(_uniform(n), sh, alpha))

    # Mix beta‑shapes with uniform
    for a in (0.05, 0.5, 1.0):
        for b in (0.05, 0.5, 1.0):
            sh = _beta_shape(n, a, b)
            for alpha in np.linspace(0.1, 0.9, 5):
                seeds.append(_mix_shapes(_uniform(n), sh, alpha))

    # Alternating Dirichlet (helps later permutation search)
    seeds.append(_alternating_dirichlet(n, rng))

    # Previous best (if any)
    if previous is not None and previous.shape[0] == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # ----------------------------------------------------------------------
    # Normalise sanity‑check (very rarely needed)
    # ----------------------------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised probability vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6,
        ]

    # work in log‑probability space (stay on simplex via softmax)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (see derivation in original code)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact max‑convolution mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # classic 1/√t schedule (t = it)
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 200_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Swap‑based permutation hill‑climber (exact objective)
# ----------------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap only if it reduces the exact
    max‑convolution.  Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 200_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val

                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Heavy permutation‑swap local search (final fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(2_000_000, remaining * 0.20))
        max_swaps = max(500_000, max_swaps)
        print("\n--- Permutation swap local search ---")
        seq_perm, val_perm = _perm_swap_opt(global_best_seq,
                                            n,
                                            deadline,
                                            max_swaps=max_swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[PermSwap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final short exponentiated‑gradient refinement (after reordering)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp2 = int(min(400_000, remaining * 0.04))
        if max_exp2 > 100_000:
            print(f"[ExpGrad‑final] polishing with {max_exp2:,} iterations")
            seq_final, val_final = _exp_gradient_opt(global_best_seq,
                                                     n,
                                                     deadline,
                                                     max_iters=max_exp2)
            if val_final < global_best_val:
                global_best_seq = seq_final.copy()
                global_best_val = val_final
                print(f"[ExpGrad‑final] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
