def _subgradient_descent(a0, n, deadline):
    a = a0.copy()
    best = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    step0 = 0.5  # initial step size
    while time.time() < deadline:
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        if max_val * 2 * n < best_val:  # direct check earlier? not needed.
            pass
        # compute subgradient
        max_idxs = np.flatnonzero(conv == max_val)
        grad = np.zeros_like(a)
        for t in max_idxs:
            # compute slice indices
            start = max(0, t - (n - 1))
            end = min(n - 1, t)
            # a[t - j] for j in start..end
            j = np.arange(start, end+1, dtype=int)
            grad[j] += 2.0 * a[t - j]
        grad /= len(max_idxs)
        # step size schedule
        step = step0 / np.sqrt(it + 1)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0.0:
            a_new = _uniform(n)
        else:
            a_new /= s
        a = a_new
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            # Optionally print improvements
            print(f"[Subgrad] iter {it} improved to {best_val:.6f}")
        it += 1
    return best, best_val

"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------
1. **Broad seeding** – deterministic analytic windows, a large
   random Dirichlet/Exponential pool and a few hand‑crafted families.
   All candidates are normalised to sum = 1 and the best ``top_k``
   vectors (by the exact surrogate 2·n·max(conv)) are kept.

2. **Local polishing** – each of the top seeds is refined by a
   continuation L‑BFGS‑B optimiser that works on the softmax
   representation of the vector (smooth log‑sum‑exp surrogate for the
   max).  The optimiser runs a long β‑schedule and many iterations.

3. **Convex mixing** – all refined vectors are convexly mixed pair‑wise.
   For each pair we scan α∈[0,1] (step 0.05) and keep the best mixture.

4. **Projected sub‑gradient descent** – starting from the current best
   vector we perform a simple sub‑gradient descent on the true
   non‑smooth objective.  The sub‑gradient of ``max(conv)`` is obtained
   analytically (O(n)) and each iteration projects the iterate back onto
   the probability simplex by truncation and renormalisation.

5. The best vector found is evaluated with the official ``evaluate_sequence``
   and stored globally as ``height_sequence_1`` for the next call.

The whole routine respects a user supplied time budget (default 1000 s)
and prints progress information.
"""

import time
import heapq
import numpy as np

# optional SciPy import – required only for the L‑BFGS phase
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# FFT‑based convolution (linear, zero‑padded)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (renormalise)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Exact surrogate (exact objective assuming sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """2·n·max(conv(vec, vec)) – the quantity we minimise during search."""
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _double_power(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _arcsine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


def _edge_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    """Mass concentrated at the two ends, decaying exponentially inward."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet sample with all concentration parameters = conc."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, cut to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    nb = rng.integers(2, 6)
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for _ in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)

# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The optimisation works in log‑softmax space (vector v) guaranteeing
    feasibility (non‑negative, sum = 1) throughout.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0]

    # initialise logits (log‑softmax variables)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)               # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smoothed max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a
            rev_a = a[::-1]
            w = exp_beta / sum_exp
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(start_vec: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Tiny sub‑gradient steps on ``max(conv(a,a))``.
    After each step we project onto the probability simplex by
    truncating negatives and renormalising.
    """
    a = start_vec.copy()
    best = a.copy()
    best_val = _objective_surrogate(best)
    iteration = 0
    step0 = 0.5

    while time.time() < deadline:
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        # indices attaining the maximum (could be several)
        max_idx = np.flatnonzero(conv == max_val)

        # compute sub‑gradient (average over all active shifts)
        grad = np.zeros_like(a)
        for t in max_idx:
            j_start = max(0, t - (n - 1))
            j_end = min(n - 1, t)
            j = np.arange(j_start, j_end + 1, dtype=int)
            # grad_j contribution = 2 * a[t - j]
            grad[j] += 2.0 * a[t - j]
        grad /= max_idx.size

        # simple diminishing step size
        step = step0 / np.sqrt(iteration + 1)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0.0:
            a_new = _uniform(n)
        else:
            a_new /= s

        a = a_new
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[Subgrad] iter {iteration:4d} improved to {best_val:.6f}")
        iteration += 1

    return best, best_val


# ----------------------------------------------------------------------
# Convex mixing of refined candidates
# ----------------------------------------------------------------------
def _mix_candidates(seeds: list[np.ndarray],
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Try convex combinations of each pair of seeds (α from 0…1, step 0.05)."""
    best_seq = None
    best_val = float('inf')
    alphas = np.linspace(0.0, 1.0, 21)

    m = len(seeds)
    for i in range(m):
        for j in range(i + 1, m):
            if time.time() > deadline:
                return best_seq, best_val
            a_i = seeds[i]
            a_j = seeds[j]
            for alpha in alphas:
                cand = alpha * a_i + (1.0 - alpha) * a_j
                val = _objective_surrogate(cand)
                if val < best_val:
                    best_val = val
                    best_seq = cand.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    prints progress information, and stores the best vector in the
    global variable ``height_sequence_1`` for the next call.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Choose problem size (4000 works well with the FFT implementation)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Heap that keeps the best ``top_k`` seeds (by surrogate)
    # ------------------------------------------------------------------
    top_k = 30
    best_heap = []          # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert a candidate into the heap while keeping only the best ``top_k``."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # 3. Deterministic analytical seeds (wide variety)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        _push_candidate(_sine_window(n, gamma))
    _push_candidate(_edge_exponential(n, lam=0.003))
    _push_candidate(_edge_exponential(n, lam=0.008))
    _push_candidate(_arcsine_window(n))

    for gamma in np.linspace(0.5, 4.0, 9):
        _push_candidate(_cosine_edge_window(n, gamma))

    for sigma in np.linspace(0.07, 0.32, 12):
        _push_candidate(_gaussian(n, sigma))

    for p in np.linspace(0.05, 4.0, 15):
        _push_candidate(_power(n, p))
        _push_candidate(_double_power(n, p))

    # beta‑shape (U‑shaped) – explore a wide log‑space grid
    for _ in range(40):
        a = 10 ** rng.uniform(-2.5, 0)   # 0.003 … 1
        b = 10 ** rng.uniform(-2.5, 0)
        _push_candidate(_beta_shape(n, a, b))

    # two‑block uniform variants
    for _ in range(20):
        blk = rng.uniform(0.04, 0.35)
        lw = rng.random()
        _push_candidate(_two_block_uniform(n, block_ratio=blk, left_weight=lw))

    # ------------------------------------------------------------------
    # 4. Massive random seeding
    # ------------------------------------------------------------------
    conc_choices = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_seed_budget = 5000
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # occasional cyclic shift – changes correlation structure
        if rng.random() < 0.3:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # exponential draws (heavier tail)
    for _ in range(600):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        if rng.random() < 0.3:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # low‑frequency random curves
    for nf in (2, 3, 5, 8):
        for _ in range(250):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            if rng.random() < 0.3:
                shift = rng.integers(n)
                if shift:
                    vec = np.roll(vec, shift)
            _push_candidate(vec)

    # mixed block structures
    for _ in range(250):
        if time.time() > deadline:
            break
        _push_candidate(_random_mixed_blocks(n, rng))

    # ------------------------------------------------------------------
    # 5. Re‑use previous solution if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                _push_candidate(_normalize(prev_arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] retained {len(seeds)} candidates, best surrogate = "
          f"{_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 7. Initialise the global best
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Local refinements – continuation L‑BFGS on the top few seeds
    # ------------------------------------------------------------------
    max_refinements = min(8, len(seeds))
    refined = []
    for idx in range(max_refinements):
        if time.time() > deadline - 5.0:
            break
        seed_vec = seeds[idx]
        remaining = deadline - time.time()
        # give each refinement a fair slice of the remaining time
        refine_deadline = time.time() + max(5.0, remaining * 0.55 / max_refinements)

        try:
            refined_vec = _lbfgs_smooth_optimize(seed_vec,
                                                 n,
                                                 refine_deadline,
                                                 beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                                200.0, 500.0,
                                                                1000.0, 2000.0,
                                                                5000.0, 10000.0,
                                                                20000.0, 50000.0,
                                                                100000.0, 200000.0,
                                                                500000.0],
                                                 max_iter_per_beta=4000)
        except RuntimeError:
            refined_vec = seed_vec  # SciPy not available – fall back

        refined_val = _objective_surrogate(refined_vec)
        refined.append(refined_vec)
        if refined_val < global_best_val:
            global_best_seq = refined_vec.copy()
            global_best_val = refined_val
            print(f"[Refine {idx+1}] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Convex mixing of refined candidates
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0 and refined:
        print("\n--- Convex mixing phase ---")
        mix_seq, mix_val = _mix_candidates(refined, n, deadline - 0.5)
        if mix_seq is not None and mix_val < global_best_val:
            global_best_seq = mix_seq.copy()
            global_best_val = mix_val
            print(f"[Mix] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Projected sub‑gradient descent from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Sub‑gradient descent phase ---")
        sub_seq, sub_val = _subgradient_descent(global_best_seq, n, deadline - 0.2)
        if sub_val < global_best_val:
            global_best_seq = sub_seq.copy()
            global_best_val = sub_val
            print(f"[Sub‑grad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 11. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
