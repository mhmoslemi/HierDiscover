            rows.extend(range(start, end))

        g = res.x

propose_candidate
-----------------

A new search strategy that blends a broad family‑seed stage (as before) with a
convex‑programming refinement step.  The refinement solves, for the current
candidate `f`, the linear program

    max   Σ g_i
    s.t.  (f * g)[k] ≤ M      ∀k            (convolution ≤ current max)
          g_i ≥ 0

where `M = max(conv(f,f))`.  If the LP can increase the ℓ₁‑norm of `g` beyond
that of `f`, scaling `g` back to the original ℓ₁‑norm yields a vector that
strictly reduces the max‑convolution when mixed with `f`.  A short ternary
search is used to pick the optimal mixing weight.  A few LP iterations are
performed, after which the result is polished with the exponentiated‑gradient
(EG) sub‑gradient method and a continuation L‑BFGS optimiser that smooths the
max with a log‑sum‑exp surrogate.

The whole pipeline respects a 1000 s wall‑time budget; the best sequence found

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

    2·n·max(conv(seq, seq)).

    conv = _conv_fft(seq, seq)

def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on exact max‑convolution.
    Returns best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        # SciPy not available – fall back to the input vector unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1000000.0]

    # work in log‑softmax space so we stay on the simplex automatically
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (softmax space)
            w = exp_beta / sum_exp          # weights for each entry of conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)     # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Linear‑programming “boost” step
# ----------------------------------------------------------------------
def _build_conv_ub_matrix(f: np.ndarray) -> _sp.csr_matrix:
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that A @ g = conv(f, g)
    (full linear convolution).  The matrix is used for the LP constraint
    conv(f, g) ≤ M.
    """
    n = f.shape[0]
    # Rows: i + j for all i,j ∈ [0, n-1]
    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel().astype(np.int32)
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    data = np.repeat(f, n)
    A = _sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_max_sum(given_f: np.ndarray, M: float, deadline: float):
    """
    Solve   max Σ g_i   s.t. conv(given_f, g) ≤ M, g ≥ 0.
    Returns the optimal vector g (may be dense but sparse is fine).
    """
    if _opt is None or _sp is None:
        raise RuntimeError("SciPy required for LP step")

    n = given_f.shape[0]
    A = _build_conv_ub_matrix(given_f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimise -Σg_i  ⇔ maximise Σg_i
    bounds = [(0, None)] * n

    # HiGHS is fast and handles sparse matrices well
    res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                       method='highs',
                       options={'presolve': True, 'time_limit': max(1.0, deadline - time.time())})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def _ternary_mix(a: np.ndarray, b: np.ndarray, deadline: float, steps: int = 30):
    """Best convex combination of a and b by ternary search on α∈[0,1]."""
    lo, hi = 0.0, 1.0
    best_seq = None
    best_val = float('inf')
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if val1 < best_val:
                best_val, best_seq = val1, seq1
        else:
            lo = m1
            if val2 < best_val:
                best_val, best_seq = val2, seq2
    # final candidate at midpoint
    mid = (lo + hi) / 2.0
    seq_mid = _mix_shapes(a, b, mid)
    val_mid = _objective_surrogate(seq_mid)
    if val_mid < best_val:
        best_seq, best_val = seq_mid, val_mid
    return best_seq, best_val


def _lp_refine(initial_f: np.ndarray,
               deadline: float,
               max_lp_iters: int = 6,
               mix_steps: int = 30,
               min_improve: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based refinement iterations starting from `initial_f`.
    Returns the best sequence (ℓ₁‑normalised) and its surrogate value.
    """
    f = initial_f.copy()
    n = f.shape[0]
    best_val = _objective_surrogate(f)
    best_f = f.copy()
    print(f"[LP‑refine] start surrogate = {best_val:.6f}")

    for it in range(max_lp_iters):
        if time.time() > deadline:
            break
        conv_ff = _conv_fft(f, f)
        M = float(np.max(conv_ff))
        try:
            g = _lp_max_sum(f, M, deadline)
        except Exception as exc:
            print(f"  LP iteration {it} failed: {exc}")
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            print(f"  LP iteration {it}: no ℓ₁‑norm increase (sum_g={sum_g:.6f})")
            break
        # Scale g back to ℓ₁‑norm = 1 and mix
        scale = 1.0 / sum_g
        g_scaled = g * scale
        # mix with ternary search for best α
        f_new, val_new = _ternary_mix(f, g_scaled, deadline, steps=mix_steps)
        # renormalise (numerical safety)
        f_new = f_new / f_new.sum()
        val_new = _objective_surrogate(f_new)
        print(f"  LP iteration {it}: new surrogate = {val_new:.6f} (sum_g={sum_g:.6f})")
        if val_new < best_val - min_improve:
            best_val, best_f = val_new, f_new.copy()
            f = f_new
        else:
            print(f"  LP iteration {it}: improvement too small, stopping.")
            break
    return best_f, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The pipeline consists of:

    1. Broad seed generation (analytic families, random Dirichlet/exp,
       low‑frequency curves, perturbations of the previous best).
    2. Quick surrogate evaluation → keep the top‑few raw seeds.
    3. Linear‑programming refinement (up to a few iterations) that
       injects extra ℓ₁‑mass where the convolution slack permits.
    4. Final polishing with exponentiated‑gradient sub‑gradient descent
       followed by continuation L‑BFGS smoothing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size (fixed as in the previous run)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Broad seed generation (same families as before, but a bit more
    #    variation to give the LP a richer starting point)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic families (coarse grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 14):
        seeds.append(_gaussian(n, sigma))
    for sigma in (0.10, 0.12):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma,
                                          shift_frac=shift))
    for p in np.linspace(0.0, 2.0, 13):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))
    for alpha in np.linspace(0.1, 0.9, 7):
        seeds.append(_tukey_window(n, alpha))
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))
    seeds.append(_alternating_dirichlet(n, rng))

    # Add perturbed copies of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(5):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Surrogate evaluation of all seeds (cheap)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))
    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Linear‑programming based refinement of the best raw seed
    # ------------------------------------------------------------------
    # Start from the absolute best raw seed
    init_seq = evaluated[0][0]
    lp_deadline = deadline - 10.0   # keep a buffer for final polishing
    refined_seq, refined_val = _lp_refine(init_seq, lp_deadline,
                                         max_lp_iters=6,
                                         mix_steps=25)

    # ------------------------------------------------------------------
    # 4. Final polishing (EG + L‑BFGS) on the LP‑refined sequence
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing ---")
        eg_deadline = deadline - 5.0
        seq_eg, val_eg = _exp_gradient_opt(refined_seq.copy(),
                                           n,
                                           eg_deadline,
                                           max_iters=2_500_000,
                                           eta0=5.0)
        if val_eg < refined_val:
            refined_seq, refined_val = seq_eg, val_eg

    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing ---")
        lbfgs_deadline = deadline
        seq_lbfgs = _lbfgs_smooth_optimize(refined_seq,
                                           n,
                                           lbfgs_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1000.0,
                                                          2000.0, 5000.0,
                                                          20000.0, 50000.0,
                                                          100000.0, 200000.0,
                                                          500000.0, 1000000.0],
                                           max_iter_per_beta=4000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < refined_val:
            refined_seq, refined_val = seq_lbfgs, val_lbfgs

    # ------------------------------------------------------------------
    # 5. Official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(refined_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {refined_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the best sequence for the next run
    globals()["height_sequence_1"] = refined_seq.tolist()
    return refined_seq.tolist()
