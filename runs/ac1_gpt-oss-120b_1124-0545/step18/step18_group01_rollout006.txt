def hill_climb(seq, n, deadline, rng, max_steps=2000):
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    steps = 0
    while steps < max_steps and time.time() < deadline:
        # propose new neighbor
        # choose move type: shift mass from i to j.
        i = rng.choice(n, p=a / a.sum())  # choose index weighted by mass
        # Or choose any nonzero index:
        nonzero_idx = np.where(a > 1e-12)[0]
        if len(nonzero_idx)==0:
            break
        i = rng.choice(nonzero_idx)
        # choose j (maybe any index)
        j = rng.integers(0, n)
        # delta up to fraction of a[i]
        delta = a[i] * rng.uniform(0.0, 0.5)  # small move
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        # ensure non-neg, sum unchanged.
        # compute val
        val = _objective_surrogate(a)
        if val < best_val:

"""
Propose a step‑function (list of non‑negative heights) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The search proceeds in three stages, all respecting a global time budget
(default 1000 s).

1. **Broad seed generation** – a large collection of dense (Dirichlet,
   analytic windows, low‑frequency cosine sums) and sparse (few non‑zero
   spikes) candidates are sampled and evaluated using the exact
   surrogate `2·n·max(conv)`.  Only the best `top_k` seeds are retained.

2. **Local refinement** – each retained seed is improved by a cascade of
   optimisation methods:
        * exponentiated‑gradient (mirror‑descent) on the exact max‑conv,
        * continuation L‑BFGS on a log‑sum‑exp smooth surrogate,
        * a cheap hill‑climbing / simulated‑annealing phase.
   The best local result updates the global optimum.

3. **Final polishing** – a long exponentiated‑gradient run from the
   current global best attempts a final reduction.

All phases stop as soon as the deadline is reached.  The best sequence
found is stored in the global variable ``height_sequence_1`` and returned
as a plain Python list.
"""

import time
import math
import heapq
import numpy as np

# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------

# ----------------------------------------------------------------------
# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Randomised families
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_spike(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Create a vector with exactly k positive spikes (positions random)."""
    if k <= 0:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    w = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s

def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise a smoothed max‑convolution objective using L‑BFGS‑B.
    Variables are kept in log‑softmax space to guarantee feasibility.
    """
    if _opt is None:
        # SciPy not available – fall back to the input vector unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _hill_climbing(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   max_attempts: int = 200_000,
                   sigma_start: float = 0.005,
                   sigma_decay: float = 0.9995):
    """
    Simple stochastic hill‑climbing / simulated‑annealing.
    Random Gaussian perturbations are applied in log‑space (softmax)
    and accepted only if they improve the surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    sigma = sigma_start
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # perturb in log‑space – this keeps positivity naturally
        log_a = np.log(a + 1e-30)
        log_a += sigma * rng.normal(size=n)
        ns = _softmax(log_a)

        val = _objective_surrogate(ns)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns  # move current point
        sigma *= sigma_decay
        attempts += 1

    return best_seq, best_val

def _generate_dense_seeds(n: int, rng: np.random.Generator):
    """Yield a variety of dense (full‑length) candidates."""
    # Analytic family sweeps
    yield _uniform(n)
    yield _triangular(n)

    for gamma in np.linspace(0.1, 3.5, 20):
        yield _sine_window(n, gamma)

    for sigma_frac in np.linspace(0.05, 0.35, 20):
        yield _gaussian(n, sigma_frac)

    for p in np.linspace(0.0, 4.0, 20):
        yield _power(n, p)

    for rate in np.logspace(-4, -0.5, 12):
        yield _exp_decay(n, rate)
        yield _sym_exp_decay(n, rate)

    # Random Dirichlet / exponential draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            yield _random_dirichlet(n, conc, rng)

    for _ in range(15):
        yield _random_exponential(n, rng)

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(5):
            yield _lowfreq_curve(n, nf, rng)

    # Random permutations of a few promising shapes
    base_shapes = [_gaussian(n, 0.18),
                   _sine_window(n, 1.2),
                   _beta_shape(n, 2.0, 2.0)]
    for base in base_shapes:
        for _ in range(3):
            yield np.random.permutation(base)

    # Mixtures of uniform and low‑freq curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * curve
            mixed /= mixed.sum()
            yield mixed


def _generate_sparse_seeds(n: int, rng: np.random.Generator):
    """Yield sparse spike candidates with a few non‑zero entries."""
    # Few spikes, equal weights
    for k in (8, 12, 16, 20, 32, 48, 64):
        for _ in range(5):
            yield _sparse_spike(n, k, rng)

    # Random number of spikes, exponential weights
    for _ in range(30):
        k = rng.integers(5, 150)
        yield _sparse_spike(n, k, rng)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a small safety margin – leave a couple of seconds for the final
    # evaluate_sequence call (and printing)
    deadline = start_time + budget_s - 2.0

    # Problem size – 4000 proved a good trade‑off between quality and speed
    n = 4000

    # ------------------------------------------------------------------
    # Prepare a heap that keeps the best *top_k* candidates seen so far
    # (max‑heap on the negative surrogate value)
    # ------------------------------------------------------------------
    top_k = 15
    heap = []  # contains tuples (-value, ndarray)

    def _push_candidate(vec: np.ndarray):
        """Normalize ``vec`` to ℓ₁ = 1, evaluate and possibly keep it."""
        v = np.maximum(vec, 0.0)
        s = v.sum()
        if s == 0.0:
            return
        v = v / s
        val = _objective_surrogate(v)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, v))
        else:
            if -heap[0][0] > val:
                heapq.heapreplace(heap, (-val, v))

    # ------------------------------------------------------------------
    # Load a previous solution (if any) – gives a strong starting point
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                _push_candidate(arr)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1: generate a large pool of seeds (dense + sparse)
    # ------------------------------------------------------------------
    print("\n--- Phase 1: Seed generation ---")
    # dense seeds
    for vec in _generate_dense_seeds(n, rng):
        if time.time() > deadline:
            break
        _push_candidate(vec)

    # sparse seeds
    for vec in _generate_sparse_seeds(n, rng):
        if time.time() > deadline:
            break
        _push_candidate(vec)

    # Extract the current top seeds sorted by true surrogate value
    current_top = sorted([(-neg, vec) for neg, vec in heap], key=lambda x: x[0])
    top_seeds = [vec for _, vec in current_top]

    print("\nTop seeds after Phase 1 (surrogate values):")
    for i, (val, _) in enumerate(current_top):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Initialise global best from the seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2: local refinement of each top seed
    # ------------------------------------------------------------------
    print("\n--- Phase 2: Local refinement ---")
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # Allocate a slice of the remaining budget for this seed.
        # Roughly split the remaining time evenly among the seeds that are
        # still to be processed, keeping a small reserve for the final polish.
        remaining = deadline - time.time()
        seeds_left = len(top_seeds) - idx
        sub_deadline = time.time() + max(0.0, remaining * 0.85 / seeds_left)

        # (a) exponentiated‑gradient descent
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=400_000,
                                           eta0=6.0)

        # (b) continuation L‑BFGS on a smoothed surrogate
        if time.time() < deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               deadline,
                                               max_iter_per_beta=5000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs = seq_eg
            val_lbfgs = val_eg

        # (c) cheap hill‑climbing / simulated‑annealing
        hill_deadline = deadline - 0.5  # reserve a tiny margin
        if time.time() < hill_deadline:
            seq_hc, val_hc = _hill_climbing(seq_lbfgs,
                                            n,
                                            hill_deadline,
                                            rng,
                                            max_attempts=250_000,
                                            sigma_start=0.004,
                                            sigma_decay=0.9996)
        else:
            seq_hc, val_hc = seq_lbfgs, val_lbfgs

        # Pick the best among the three refinements and the original seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_hc, val_hc)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {idx+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3: final long exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: Final EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
