def _smooth_max_and_grad(a, beta):
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    # Use shift for numerical stability
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    obj = shift + np.log(sum_exp) / beta  # this is smooth max

    w = exp_beta / sum_exp   # weights sum to 1
    # compute gradient wrt a:
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[len(a)-1: len(a)-1 + len(a)]

    # gradient w.r.t v: a * (grad_a - dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v
```

def _adam_optimize(start_seq, n, deadline, beta_schedule, steps_per_beta=2000, lr=0.05, eps=1e-8, beta1=0.9, beta2=0.999):
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    t = 0
    best_seq = None
    best_val = np.inf
    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            a = _softmax(v)
            # compute smooth max and gradient w.r.t v
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            smooth_max = shift + np.log(sum_exp) / beta
            obj = 2 * n * smooth_max
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1**t)
            v_hat_corr = v_hat / (1 - beta2**t)
            v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # evaluate surrogate exact object for best tracking
            if obj < best_val:
                best_val = obj
                best_seq = a.copy()
        # maybe after each beta we can decay lr or adjust.

    return best_seq if best_seq is not None else _softmax(v)
```

def _optimize_seq(start_seq, n, deadline):
    # returns best_seq and best_val
    best_seq = start_seq
    # compute initial best exact value
    conv = _conv_fft(start_seq, start_seq)
    best_val = 2*n*np.max(conv)

    # Beta schedule
    beta_schedule = [0.5, 1.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    # Adam hyperparameters
    lr = 0.05
    eps = 1e-8
    beta1 = 0.9
    beta2 = 0.999

    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    t = 0

    for beta in beta_schedule:
        # Determine iterations per beta based on remaining time and uniform schedule
        # Roughly aim for 5000 per beta but ensure not exceed time.
        max_iters = 5000
        for _ in range(max_iters):
            if time.time() > deadline:
                return best_seq, best_val
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # update best exact
            val_exact = 2 * n * max_c
            if val_exact < best_val:
                best_val = val_exact
                best_seq = a.copy()

            # smooth max and gradient
            shift = max_c  # shift for stability
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            # smooth_max = shift + np.log(sum_exp) / beta  # not used directly
            w = exp_beta / sum_exp

            # gradient w.r.t a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad_v
            v_sq = beta2 * v_sq + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_sq_hat = v_sq / (1 - beta2 ** t)
            v = v - lr * m_hat / (np.sqrt(v_sq_hat) + eps)

        # optionally reduce learning rate at higher beta
        lr *= 0.9
    # final exact EG refinement maybe
    # optional: run some exponentiated gradient iterations for exact max
    # Use _exp_gradient_opt defined earlier with maybe max_iters=500_000

def _exp_gradient_opt(start_seq, n, deadline, max_iters=500_000, eta0=5.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2 * n * np.max(_conv_fft(a, a))
    log_a = np.log(a + 1e-30)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # compute subgradient
        g = np.zeros_like(a)
        # For each i, j with i + j = max_idx range:
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        # indices i_start .. i_end inclusive
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)
        # compute new value
        cur_val = 2 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Propose a new step‑function (list of non‑negative heights).

The algorithm consists of three layers:

1. **Broad seed pool** – analytic windows, Dirichlet / exponential draws,
   low‑frequency cosine sums and a few mixtures.  All seeds are ℓ1‑normalised.

2. **Continuation Adam on a smooth max‑convolution** – a log‑softmax
   parametrisation guarantees positivity and unit ℓ1‑norm.
   For a schedule of β values the smooth approximation
   `log‑sum‑exp(β·conv)` is minimised by Adam.  The exact max‑convolution
   is monitored and the best sequence is kept.

3. **Exact exponentiated‑gradient polishing** – after the smoothed phase
   we run a mirror‑descent (exponentiated‑gradient) pass on the true
   max‑convolution to fine‑tune the active constraints.

All steps respect a 1000 s wall‑clock budget and store the best sequence
in the global variable ``height_sequence_1`` for future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ1‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·|i-(n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) on the true max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its exact objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation Adam optimiser on a smoothed max‑convolution.
# ----------------------------------------------------------------------
def _smooth_adam_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth log‑sum‑exp surrogate of max(conv(a,a))
    using Adam in log‑softmax space.
    Returns the best sequence (exact objective) found before deadline.
    """
    # initialise Adam state in log‑space
    v = np.log(start_seq + 1e-30)          # logits
    m = np.zeros_like(v)                  # 1st moment
    v_sq = np.zeros_like(v)               # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    t = 0

    # best tracker (exact objective)
    best_seq = start_seq.copy()
    best_val = 2.0 * n * np.max(_conv_fft(start_seq, start_seq))

    # schedule of smoothing parameters β
    beta_schedule = [
        0.5, 1.0, 5.0, 10.0, 20.0,
        50.0, 100.0, 200.0, 500.0,
        1000.0, 2000.0, 5000.0, 10000.0
    ]

    for beta in beta_schedule:
        # iteration budget per β (more for small β)
        if beta <= 20:
            max_iter = 4000
        elif beta <= 200:
            max_iter = 3000
        else:
            max_iter = 2000

        for _ in range(max_iter):
            if time.time() > deadline:
                return best_seq, best_val

            a = _softmax(v)                     # current probabilities (sum = 1)

            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            exact_val = 2.0 * n * max_c
            if exact_val < best_val:
                best_val = exact_val
                best_seq = a.copy()

            # ----- smooth max (log‑sum‑exp) -----
            shift = max_c                         # stabiliser
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                # soft‑weights, ∑w = 1

            # gradient w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # size n

            # gradient w.r.t. logits v (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ----- Adam update -----
            t += 1
            m = beta1 * m + (1 - beta1) * grad_v
            v_sq = beta2 * v_sq + (1 - beta2) * (grad_v ** 2)

            m_hat = m / (1 - beta1 ** t)
            v_sq_hat = v_sq / (1 - beta2 ** t)

            v = v - lr * m_hat / (np.sqrt(v_sq_hat) + eps)

        # slowly decay learning‑rate as we get closer to the true max
        lr *= 0.9

    # ------------------------------------------------------------------
    # Final exact‑max polishing with exponentiated‑gradient (shorter run)
    # ------------------------------------------------------------------
    seq_eg, val_eg = _exp_gradient_opt(best_seq, n, deadline,
                                       max_iters=200_000,
                                       eta0=5.0)
    if val_eg < best_val:
        best_val = val_eg
        best_seq = seq_eg.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and stores the best sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – a historically‑good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous solution if present
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum() if arr.sum() > 0 else None
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Seed generation – a wide and diverse pool
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic families (coarse grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_tukey_window(n, 0.8))

    for sigma_frac in np.linspace(0.05, 0.30, 12):
        seeds.append(_gaussian(n, sigma_frac))

    for rate in np.logspace(-4, -0.5, 10):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    for _ in range(25):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (focus on sparse and dense regimes)
    for conc in (0.01, 0.05, 0.2, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(20):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Perturbations of previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start optimisation (continuation Adam + final EG)
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fraction of the remaining budget for this seed
        remaining = deadline - time.time()
        # keep roughly 15 % per upcoming seed + final polishing
        sub_deadline = time.time() + remaining * 0.15 / max(1, (top_k - si))

        print(f"\n[Seed {si+1}/{top_k}] optimisation start (time left ≈ {remaining:.1f}s)")

        opt_seq, opt_val = _smooth_adam_optimize(seed_vec, n, sub_deadline)

        print(f"  -> seed {si+1} best surrogate = {opt_val:.6f}")

        if opt_val < global_best_val:
            global_best_val = opt_val
            global_best_seq = opt_seq.copy()
            print(f"     *** new global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final polishing – short random hill‑climbing + EG
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (random perturbations + EG) ---")
        seq = global_best_seq.copy()
        best_seq = seq.copy()
        best_val = global_best_val

        # simple hill‑climbing with decreasing step size
        sigma = 0.004
        decay = 0.9996
        attempts = 0
        max_attempts = 250_000

        while time.time() < deadline and attempts < max_attempts:
            ns = seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                attempts += 1
                continue
            ns = ns / s
            val = _objective_surrogate(ns)
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
                seq = ns
            sigma *= decay
            attempts += 1

        # finish with a short exact‑max EG run
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, deadline,
                                           max_iters=300_000,
                                           eta0=4.0)
        if val_eg < best_val:
            best_val = val_eg
            best_seq = seq_eg.copy()

        global_best_seq = best_seq
        global_best_val = best_val
        print(f"  Polishing result surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
