"""
propose_candidate
-----------------
Searches for a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of:

1.  Broad seed generation (analytic families, Dirichlet / exponential draws,
    low‑frequency cosine curves, monotone exponentials, Beta‑shapes, and
    the previous best if available).  All seeds are ℓ₁‑normalised.

2.  Quick surrogate evaluation of all seeds (exact max‑convolution, i.e.
    2·n·max(conv) because the sum is forced to 1).

3.  Multi‑restart optimisation of the best few seeds using a
    **log‑sum‑exp smooth max** objective and **Adam** in the
    simplex‑parameterisation (log‑softmax).  The smoothing parameter β is
    increased gradually (the schedule is the same as in the reference
    code).  Adam’s adaptive step‑size together with bias‑correction
    dramatically improves convergence compared with plain exponentiated
    gradient.

4.  A short random‑perturbation hill‑climbing phase to escape possible
    local minima after Adam.

5.  A final long Adam polishing run (large β, many iterations).

All steps are time‑aware; the routine stops before the supplied time budget
and returns the best sequence found.  The best sequence is also stored in the
global variable ``height_sequence_1`` for reuse in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) for a sequence with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves; clipped and normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture of two normalised shapes."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   max_iters_per_beta: int = 2000,
                   lr0: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Optimises the smooth max (log‑sum‑exp) of the convolution using Adam.
    The parameters are kept in log‑softmax space (v) guaranteeing that the
    iterate stays in the probability simplex.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    # Initialise log‑softmax parameters
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)          # first moment
    v_hat = np.zeros_like(v)      # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    best_seq = None
    best_val = float('inf')
    lr = lr0

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(max_iters_per_beta):
            if time.time() > deadline:
                break
            t += 1

            # a = softmax(v)
            a = _softmax(v)

            # full convolution a * a
            conv = _conv_fft(a, a)

            # smooth‑max weights w = softmax(beta * conv)
            conv_max = np.max(conv)
            exp_term = np.exp(beta * (conv - conv_max))
            w = exp_term / exp_term.sum()

            # gradient w.r.t. a : 2 * (rev_a * w)  (slice of length n)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                     # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]           # keep the central n entries

            # gradient w.r.t. v (softmax Jacobian trick)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update (bias‑corrected)
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)

            m_hat = m / (1 - beta1 ** t)
            v_hat_corr = v_hat / (1 - beta2 ** t)

            v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # Evaluate the exact surrogate (no smoothing) to track progress
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

        # modest learning‑rate decay after each β‑stage
        lr *= 0.8

    # fallback if nothing was ever accepted
    if best_seq is None:
        best_seq = _softmax(v)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val

def _hill_climbing(start_seq: np.ndarray,

    """
    Very lightweight random‑perturbation search.  Returns the best
    sequence discovered before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    sigma = sigma_start
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s
        val = _objective_surrogate(ns)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns      # move the current centre
        sigma *= sigma_decay
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – the same size as in the reference implementation.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if present (helps when the script is run
    # repeatedly).
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,) and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()      # ensure exact ℓ₁‑normalisation
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation – a deliberately diverse pool.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))

    # parameter sweeps for a few families
    for gamma in np.linspace(0.5, 3.0, 15):
        seeds.append(_sine_window(n, gamma=gamma))
    for sigma_frac in np.linspace(0.07, 0.30, 15):
        seeds.append(_gaussian(n, sigma_frac))
    for p in np.linspace(0.1, 3.0, 15):
        seeds.append(_power(n, p))
    for rate in np.logspace(-4, -0.8, 12):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # beta‑shape grid (asymmetric shapes are crucial)
    for a in np.linspace(0.2, 4.0, 8):
        for b in np.linspace(0.2, 4.0, 8):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet / exponential
    for conc in (0.01, 0.05, 0.1, 0.3, 0.7, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of decreasing exponential with its reverse (creates anti‑correlation)
    for rate in (0.001, 0.003, 0.01):
        dec = _exp_decay(n, rate)
        rev = dec[::-1]
        seeds.append(_mix_shapes(dec, rev, alpha=0.3))
        seeds.append(_mix_shapes(dec, rev, alpha=0.7))

    # include the previous best if we have one
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(4):
            seeds.append(prev_arr + 0.002 * rng.normal(size=n))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of seeds
    # ------------------------------------------------------------------
    evaluated = []
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise the global best from the very best seed.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑restart refinement using Adam on the smoothed objective.
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget (≈15 % per remaining seed)
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.15 / max(1, (top_k - si))

        seq_adam, val_adam = _adam_optimize(seed_vec.copy(),
                                            n,
                                            sub_deadline,
                                            max_iters_per_beta=2500,
                                            lr0=0.12)

        # short hill‑climbing refinement
        hill_deadline = sub_deadline - 0.5        # keep a tiny margin
        if time.time() < hill_deadline:
            seq_hc, val_hc = _hill_climbing(seq_adam,
                                            n,
                                            hill_deadline,
                                            rng,
                                            max_attempts=120_000,
                                            sigma_start=0.003,
                                            sigma_decay=0.9996)
        else:
            seq_hc, val_hc = seq_adam, val_adam

        # keep the best of this branch
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_adam, val_adam),
            (seq_hc, val_hc)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing – a longer Adam run from the overall best.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (Adam) ---")
        final_seq, final_val = _adam_optimize(global_best_seq,
                                               n,
                                               deadline,
                                               max_iters_per_beta=4000,
                                               lr0=0.08)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional extra hill‑climbing (quick random tweaks) – can squeeze
    #    a little more out of the solution if time permits.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random perturbation hill‑climbing ---")
        seq_hc2, val_hc2 = _hill_climbing(global_best_seq,
                                          n,
                                          deadline,
                                          rng,
                                          max_attempts=200_000,
                                          sigma_start=0.002,
                                          sigma_decay=0.9997)
        if val_hc2 < global_best_val:
            global_best_seq = seq_hc2
            global_best_val = val_hc2
            print(f"[Final HC] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
