"""
propose_candidate
-----------------
A fast two‑phase optimiser for the max‑auto‑convolution objective.

Phase 1: a long mirror‑descent (exponentiated‑gradient) run starting from the
previous best sequence (or a diverse set of analytic seeds).  The method works
directly on the exact objective – it updates the log‑probabilities with a
1/√t step‑size, always staying on the ℓ₁‑simplex.

Phase 2: a cheap targeted mass‑transfer local search that updates the full
convolution incrementally (O(n) per trial, no FFT).  At each trial the current
lag with maximal autocorrelation is identified, a little mass is moved from a
position participating in that lag to a random other position, and the move is
kept only if it strictly lowers the peak.  The convolution is updated via the
closed‑form formula
    conv' = conv + 2·δ·(shift_target – shift_source)
          + δ²·e_{2·target} + δ²·e_{2·source} – 2·δ²·e_{source+target}.
Periodically we recompute the convolution from scratch to avoid drift.

Both phases respect a global 1000 s wall‑clock budget and keep the best
sequence seen.  The final sequence is stored in the global variable
``height_sequence_1`` for the next call.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for ℓ₁‑normalised vectors:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _edge_bias(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution concentrating mass near the ends."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a large, diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for g in (0.3, 0.5, 1.0, 2.0, 4.0, 8.0):
        seeds.append(_sine_window(n, gamma=g))

    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    for beta in (2, 4, 8, 12, 16, 20, 30):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))

    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))

    # Beta‑shape grid – extend into the very‑U region
    for a in (0.05, 0.1, 0.3, 1.0, 2.0, 4.0):
        for b in (0.05, 0.1, 0.3, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random vectors
    for k in (max(1, int(0.005 * n)), max(1, int(0.01 * n))):
        for _ in range(2):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors
    for bf in (0.05, 0.15, 0.30):
        for _ in range(2):
            seeds.append(_block_uniform(n, bf, rng))

    # Edge‑biased families (strong U‑shapes)
    for p in (0.25, 0.5, 0.75, 1.0, 1.5, 2.0):
        seeds.append(_edge_bias(n, p))

    # Mixtures
    seeds.append(_mix_shapes(_uniform(n), _edge_bias(n, 0.5), 0.3))

    # Alternating Dirichlet
    seeds.append(_alternating_dirichlet(n, rng))

    # Permuted heavy‑tailed seed
    heavy = _random_dirichlet(n, 0.05, rng)
    for _ in range(3):
        perm = heavy[rng.permutation(n)]
        seeds.append(perm)

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity‑check
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.7 / math.sqrt(it)          # slightly larger than the classic 0.5
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional log
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:,} new best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase 2 – targeted mass‑transfer local search (incremental convolution)
# ----------------------------------------------------------------------
def _targeted_mass_transfer_opt(start_a: np.ndarray,
                                deadline: float,
                                max_moves: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Randomly moves a small amount of mass from a position that participates
    in the current maximal lag to a random other position.  The convolution
    is updated incrementally (O(n) per trial).  Returns (best_sequence,
    best_surrogate_value).
    """
    n = len(start_a)
    a = start_a.copy()
    conv = _conv_fft(a, a)
    cur_max = conv.max()
    best_a = a.copy()
    best_val = 2.0 * n * cur_max

    # padded view for O(1) shift extraction: [0..n-2] zeros, a, zeros
    pad = np.concatenate([np.zeros(n - 1, dtype=np.float64),
                          a,
                          np.zeros(n - 1, dtype=np.float64)])
    rng = np.random.default_rng()

    accepted = 0
    for mv in range(1, max_moves + 1):
        if time.time() > deadline:
            break

        # identify the lag with maximal autocorrelation
        kmax = int(np.argmax(conv))

        # choose a donor among the indices that contribute to that lag
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        donor = rng.integers(i_start, i_end + 1)
        # partner = kmax - donor  # not used directly, just defines the lag

        # pick a distinct target index
        target = rng.integers(0, n)
        while target == donor:
            target = rng.integers(0, n)

        # choose a modest amount of mass to move
        max_delta = a[donor] * 0.5   # at most half of the donor's mass
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()   # uniform in [0, max_delta)

        if delta <= 0.0:
            continue

        # incremental update of the convolution
        shift_donor = pad[donor: donor + 2 * n - 1]
        shift_target = pad[target: target + 2 * n - 1]

        # conv' = conv + 2*delta*(shift_target - shift_donor)
        conv_candidate = conv + 2.0 * delta * (shift_target - shift_donor)

        # add the quadratic (delta^2) corrections
        idx_2t = 2 * target
        idx_2d = 2 * donor
        idx_td = target + donor
        conv_candidate[idx_2t] += delta * delta
        conv_candidate[idx_2d] += delta * delta
        conv_candidate[idx_td] -= 2.0 * delta * delta

        new_max = conv_candidate.max()
        # accept only if we strictly improve the peak
        if new_max + 1e-15 < cur_max:
            # apply the move
            a[donor] -= delta
            a[target] += delta
            pad[n - 1 + donor] = a[donor]
            pad[n - 1 + target] = a[target]

            conv = conv_candidate
            cur_max = new_max
            accepted += 1

            if cur_max < best_val:
                best_val = 2.0 * n * cur_max
                best_a = a.copy()
                # occasional progress report
                if accepted % 100_000 == 0:
                    print(f"[MassTransfer] accepted={accepted:,} best={best_val:.6f}")

            # periodic full recompute to avoid drift
            if accepted % 250_000 == 0:
                conv = _conv_fft(a, a)
                cur_max = conv.max()
        # otherwise reject – nothing changes

    print(f"[MassTransfer] finished with {accepted} accepted moves.")
    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # Initialise from previous run if possible
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.sum() > 0:
                init_seq = prev_arr / prev_arr.sum()
                print("[Init] using previously stored sequence.")
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # Otherwise generate a diversified seed pool
    # ------------------------------------------------------------------
    if prev_arr is None:
        print(f"[Init] generating seed pool (n={n})")
        seeds = _collect_seeds_extended(n, rng, None)
        evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
        evals.sort(key=lambda x: x[1])
        init_seq = evals[0][0]
        print(f"[Init] best seed surrogate = {evals[0][1]:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long exponentiated‑gradient optimisation
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        max_iters = int(min(5_000_000, remaining * 5.0))   # crude estimate
        max_iters = max(250_000, max_iters)
        print(f"[Phase 1] exponentiated‑gradient up to {max_iters:,} iterations")
        seq1, val1 = _exp_gradient_opt(init_seq, n, deadline, max_iters=max_iters)
        print(f"[Phase 1] best surrogate = {val1:.6f}")
    else:
        seq1 = init_seq
        val1 = _objective_surrogate(seq1)

    # ------------------------------------------------------------------
    # Phase 2 – targeted mass‑transfer local search
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_moves = int(min(2_000_000, remaining * 2000.0))   # aim for ~0.5 ms / move
        max_moves = max(200_000, max_moves)
        print(f"[Phase 2] mass‑transfer local search, up to {max_moves:,} moves")
        seq2, val2 = _targeted_mass_transfer_opt(seq1, deadline, max_moves=max_moves)
        print(f"[Phase 2] best surrogate = {val2:.6f}")
    else:
        seq2, val2 = seq1, val1

    # ------------------------------------------------------------------
    # Final tidy‑up – evaluate with the official function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq2.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {2.0 * n * np.max(_conv_fft(seq2, seq2)):.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next round
    globals()["height_sequence_1"] = seq2.tolist()
    return seq2.tolist()
