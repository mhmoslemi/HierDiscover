def _permute_hill_climb(start_seq, n, deadline, max_iters=200000):
    a = start_seq.copy()
    best_val = _objective_exact(a)
    best_seq = a.copy()
    rng = np.random.default_rng(12345)
    # indices sorted by value
    idx_desc = np.argsort(-a)   # descending
    # precompute sets
    top_frac = 0.2
    bottom_frac = 0.2
    top_n = max(2, int(top_frac * n))
    bot_n = max(2, int(bottom_frac * n))
    top_idx = idx_desc[:top_n]
    bot_idx = idx_desc[-bot_n:]

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # pick i from top_idx, j from bot_idx (or random if all zeros)
        i = rng.choice(top_idx)
        j = rng.choice(bot_idx)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]
        val = _objective_exact(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            # update idx_desc maybe recalc each time? we can skip to reduce cost.
            # If improvement, recompute top and bot indices maybe every few steps.
            if it % 1000 == 0:
                idx_desc = np.argsort(-a)
                top_idx = idx_desc[:top_n]
                bot_idx = idx_desc[-bot_n:]
        else:
            # revert
            a[i], a[j] = a[j], a[i]
    return best_seq, best_val
```

def _subgradient_maxconv_all(a, tol=1e-12):
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    for idx in active:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end+1)
        rev = idx - i_range
        g[i_range] += 2.0 * a[rev]
    if len(active) > 0:
        g /= len(active)
    return g
```

def _project_onto_simplex(v, z=1.0):
    if z <= 0:
        raise ValueError(...)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - z) / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - z) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s budget)
--------------------------------
1. **Seed creation** – Build a diverse pool of ℓ₁‑normalised vectors:
   uniform, power‑law decays with a wide range of exponents and shifts,
   exponential decays, and low‑concentration Dirichlet draws.  The pool
   is deliberately larger than the previous runner (more exponents,
   larger shifts).

   perform many iterations of sub‑gradient descent on the exact
   ``max(conv)`` objective, projecting after each step onto the simplex
   (ℓ₁ = 1, non‑negative).  This method can drive entries to exact zero,
   which the soft‑max based mirror‑descent could not.

   the ordering of the values.  Swapping two entries never changes the
   ℓ₁‑mass, but can strongly affect the convolution peak.  A greedy
   improvement loop (accept only if the objective decreases) is run for
   as many swaps as time permits.

4. **Final polishing** – A short continuation L‑BFGS on the smooth
   log‑sum‑exp surrogate (very high β) refines the continuous values.

5. **Return** – The best sequence found is stored in the global
   ``height_sequence_1`` variable and returned as a plain list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradients
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for a single active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _subgradient_maxconv_all(a: np.ndarray, atol: float = 1e-12) -> np.ndarray:
    """Average sub‑gradient over all lags that are (almost) maximal."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = np.max(conv)
    # treat all entries within atol of the max as active
    active = np.where(conv >= max_val - atol)[0]
    g = np.zeros_like(a)
    for idx in active:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g[i_range] += 2.0 * a[rev]
    if len(active) > 0:
        g /= len(active)
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of 2·n·max(conv) / (∑a)².
    Returns (obj, grad) where grad is w.r.t. logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised (sum = 1)
    conv = _conv_fft(a, a)
    shift = np.max(conv)                 # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 800) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)   # returns a vector that sums to 1


# ----------------------------------------------------------------------
# Projection onto the ℓ₁‑simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}.
    Returns the projected vector.
    """
    if z <= 0:
        raise ValueError("z must be positive")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - z) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - z) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (allows exact zeros)
# ----------------------------------------------------------------------
def _projected_subgradient_optimize(start_a: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    max_iters: int = 300_000,
                                    eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective with projection onto the simplex.
    Returns the best vector seen and its objective value.
    """
    a = start_a.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum                     # enforce ℓ₁ = 1

    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv_all(a)   # more balanced direction
        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_onto_simplex(a, 1.0)  # keep ℓ₁ = 1, non‑negative

        if it % 20000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation hill‑climbing (mass‑preserving swaps)
# ----------------------------------------------------------------------
def _permute_hill_climb(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Greedy pairwise swaps that never change ℓ₁‑mass.
    Accept only if the objective improves.
    """
    a = start_seq.copy()
    best_val = _objective_exact(a)
    best_seq = a.copy()
    rng = np.random.default_rng(314159)

    # periodically recompute top / bottom index sets
    recompute_every = 5000
    top_frac = 0.20
    bot_frac = 0.20
    top_n = max(2, int(top_frac * n))
    bot_n = max(2, int(bot_frac * n))

    idx_desc = np.argsort(-a)
    top_idx = idx_desc[:top_n]
    bot_idx = idx_desc[-bot_n:]

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i = int(rng.choice(top_idx))
        j = int(rng.choice(bot_idx))
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]
        val = _objective_exact(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            # update the ordering after an improvement
            if it % recompute_every == 0:
                idx_desc = np.argsort(-a)
                top_idx = idx_desc[:top_n]
                bot_idx = idx_desc[-bot_n:]
        else:
            # revert
            a[i], a[j] = a[j], a[i]

    return best_seq, best_val


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw (always ℓ₁‑normalised)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Seed‑pool construction (broader sweep)
# ----------------------------------------------------------------------
def _collect_seeds_custom(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (sum = 1)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))

    # power‑law sweep (broader than before)
    p_vals = np.linspace(0.20, 1.20, 21)          # 21 exponents
    shifts = [0.0, 5.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # exponential family
    for lam in np.logspace(-4, -1, 12):           # 0.0001 … 0.1
        seeds.append(_exponential_decay(n, float(lam)))

    # sparse Dirichlet draws (favoring extreme values)
    for conc in (0.01, 0.05, 0.1, 0.3):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # clean‑up – enforce non‑negativity & ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    n = 6000                                # chosen to balance quality / speed
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # 1️⃣ Initialise a pool of seeds
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0.0:
                # re‑normalise to ℓ₁ = 1
                prev_arr = prev_arr / prev_arr.sum()
                seeds = _collect_seeds_custom(n, rng) + [prev_arr]
            else:
                seeds = _collect_seeds_custom(n, rng)
        except Exception:
            seeds = _collect_seeds_custom(n, rng)
    else:
        seeds = _collect_seeds_custom(n, rng)

    print(f"Generated {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # 2️⃣ Evaluate seeds, keep the best elite
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # 3️⃣ Projected sub‑gradient optimisation (allows exact zeros)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly 30 % of remaining time to this stage
        max_iters = int(min(400_000, remaining * 30_000))
        print(f"\n--- Projected sub‑gradient descent (max_iters={max_iters}) ---")
        seq_proj, val_proj = _projected_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=max_iters,
            eta0=1.5,
        )
        print(f"[Sub‑grad] best = {val_proj:.6f}")
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[Update] New global best after sub‑gradient = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Permutation hill‑climbing (mass‑preserving swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(300_000, remaining * 200))
        print(f"\n--- Permutation hill‑climb (max_swaps={max_swaps}) ---")
        seq_perm, val_perm = _permute_hill_climb(
            global_best_seq,
            n,
            deadline,
            max_iters=max_swaps,
        )
        print(f"[Permute] best = {val_perm:.6f}")
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[Update] New global best after permutation = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final polishing with continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(400, int(min(1500, remaining * 0.07 * 1e5)))
        beta_sched = [1e6, 5e6, 1e7, 5e7, 1e8]
        print("\n--- Final L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New global best after final polish = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the provided checker
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
