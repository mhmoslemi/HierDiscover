"""
Propose a step‐function sequence that minimises the
evaluate_sequence() objective.

Algorithm
---------
1. Build a large but cheap pool of ℓ1‑normalised candidate
   distributions (analytic families, fine grids, random Dirichlet,
   previous solution).
2. Evaluate a fast surrogate 2·n·max(conv) for every candidate with
   FFT‑based convolution and keep the best few.
3. For each of the top candidates run a long exponentiated‑gradient
   (mirror‑descent) optimisation on the exact max‑convolution objective.
   The updates are performed in log‑probability space (softmax) and
   stop as soon as the time budget expires.
4. Finish with a modest swap‑based hill‑climber that tries random
   permutations of the entries without ever increasing the objective.
5. Return the best sequence found (also stored in the global variable
   ``height_sequence_1`` for the next round) and report the official
   evaluate_sequence() value.

All heavy work respects the 1000 s wall‑clock budget; the programme
uses only NumPy and the provided evaluate_sequence().
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                     # pragma: no cover
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for ℓ₁‑normalised vectors:
    2·n·max(conv(seq, seq)).
    The return equals evaluate_sequence(seq) whenever sum(seq)=1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _edge_bias(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution concentrating mass near the ends."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float,
                     rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex combination (1‑α)·a + α·b, renormalised to sum 1."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # classic 1/√t schedule
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report (optional, cheap)
        if it % 200_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ------------------------------------------------------------
# Swap‑based permutation hill‑climber
# ------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap only if it reduces the exact
    max‑convolution.  Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 200_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # c) Beta‑shape grid (U‑shaped when α,β<1)
    alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    betas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    for a in alphas:
        for b in betas:
            candidates.append(_beta_shape(n, a, b))

    # d) Two‑Gaussian mixtures (different spreads / shifts)
    for sigma_frac in np.linspace(0.08, 0.35, 7):
        for shift_frac in np.linspace(0.15, 0.35, 5):
            candidates.append(_double_gaussian(n,
                                                sigma_frac=sigma_frac,
                                                shift_frac=shift_frac))

    # e) Random Dirichlet draws (occasionally very spiky)
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(3):
            candidates.append(_random_dirichlet(n, conc, rng))

    # f) Previous best (if available)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                candidates.append(prev_arr / prev_arr.sum())
        except Exception:      # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 3. Quick surrogate evaluation, keep the best few
    # ------------------------------------------------------------------
    print(f"\n--- Evaluating {len(candidates)} candidates (surrogate) ---")
    evaluated = [(vec, _objective_surrogate(vec)) for vec in candidates]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_candidates = [vec for vec, _ in evaluated[:top_k]]
    print("Best surrogate among raw candidates:",
          f"{evaluated[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4. Polishing each top candidate with exponentiated‑gradient
    # ------------------------------------------------------------------
    global_best_seq = top_candidates[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    for idx, cand in enumerate(top_candidates):
        if time.time() > deadline:
            print("[Time] deadline reached before polishing all seeds.")
            break

        # allocate remaining time proportionally
        remaining = deadline - time.time()
        # very rough estimate: one iteration ≈ 0.00018 s → ≈ 5600 it / sec
        est_iters = int(min(900_000, remaining * 5600))
        max_iters = max(300_000, min(800_000, est_iters))

        print(f"\n--- Exponentiated‑gradient polishing #{idx+1} "
              f"(max_iters={max_iters:,}) ---")
        seq_opt, val_opt = _exp_gradient_opt(cand, n, deadline,
                                             max_iters=max_iters)

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Light permutation‑swap local search (final fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(800_000, remaining * 3000))
        max_swaps = max(300_000, max_swaps)
        print("\n--- Permutation‑swap hill‑climber ---")
        seq_perm, val_perm = _perm_swap_opt(global_best_seq,
                                            n,
                                            deadline,
                                            max_swaps=max_swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[Swap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final short exponentiated‑gradient refinement (after reordering)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 5600))
        if max_iters > 100_000:
            print(f"\n--- Final short exponentiated‑gradient pass (iters={max_iters:,}) ---")
            seq_final, val_final = _exp_gradient_opt(global_best_seq,
                                                     n,
                                                     deadline,
                                                     max_iters=max_iters)
            if val_final < global_best_val:
                global_best_seq = seq_final.copy()
                global_best_val = val_final
                print(f"[Final] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next round
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
