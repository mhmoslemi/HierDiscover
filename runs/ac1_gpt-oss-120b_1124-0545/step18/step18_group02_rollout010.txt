def _targeted_local_search(start_a, n, deadline, iters=50000, step_frac=0.01):
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)
    for _ in range(iters):
        if time.time() > deadline:
            break
        # compute convolution and identify active lag
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        # active indices i where i and rev = idx-i are within range
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        # choose i from active range with relatively large a[i]
        i = rng.integers(i_start, i_end+1)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm overview
------------------
1. **Seed generation** – Assemble a diverse pool of ℓ₁‑normalised vectors.
   In addition to the classic families (uniform, triangular, Gaussian, beta,
   Dirichlet, sparse, low‑frequency), we sweep several new families:
   * power‑law with a broad exponent range,
   * a sine‑shaped bump,
   * a semi‑circular (quarter‑ellipse) profile,
   * a polynomial “tent” shape,
   * and many random mixtures / rotations of the above.
   The best few (by the exact objective) are retained as elite seeds.

2. **Continuation L‑BFGS on a smooth surrogate** – For each elite seed we
   minimise a log‑sum‑exp smooth approximation of the max‑convolution.
   The optimisation is a continuation in the smoothing parameter ``β``,
   increasing up to 1e12, with a modest per‑β iteration budget.
   The result of each run is projected back to the simplex (softmax).

3. **Exponentiated‑gradient (mirror‑descent) restarts** – Starting from a
   handful of random Dirichlet draws we run a long EG optimiser that
   follows the exact (non‑smooth) max‑convolution objective using the
   sub‑gradient from the active lag.  The step‑size decays as η₀/√t.

4. **Hybrid stochastic local search** – Two complementary hill‑climbs are
   performed on the current best sequence:
   * a plain random mass‑transfer search,
   * a *targeted* search that preferentially moves mass away from the
     indices contributing to the currently active (maximal) convolution lag.
   Both use a decreasing step‑fraction schedule.

5. **Final high‑β polishing** – One last L‑BFGS run with very large β values
   (up to 1e13) cleans up any remaining irregularities.

All stages respect the global 1000 s wall‑time budget.  The best sequence
found (ℓ₁‑normalised, i.e. sum = 1) is stored in the global variable
``height_sequence_1`` and returned as a plain list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (active lag only)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    # indices i that contribute to lag idx
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g = np.zeros_like(a)
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of the exact objective.
    Returns (obj, grad) where the gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised (sum = 1)
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # smooth max
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see derivation in the original code)
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through the simplex softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    ``start_vec`` must already be ℓ₁‑normalised.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0, 50_000_000.0, 100_000_000.0,
            500_000_000.0, 1_000_000_000.0,
        ]

    # start from logits of the initial probability vector
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        try:
            res = _opt.minimize(
                fun=_smooth_objective,
                x0=v,
                args=(n, beta_val),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-12, "disp": False},
            )
            v = res.x
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        # exponentiated‑gradient step on the simplex
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic mass‑transfer hill‑climb
# ----------------------------------------------------------------------
def _local_search(start_a: np.ndarray,
                  n: int,
                  deadline: float,
                  iters: int = 100_000,
                  step_frac: float = 0.012) -> tuple[np.ndarray, float]:
    """Random mass transfers, accept only improving moves."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


def _targeted_local_search(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           iters: int = 80_000,
                           step_frac: float = 0.010) -> tuple[np.ndarray, float]:
    """
    Hill‑climb biased towards the indices that form the active (maximal)
    convolution lag.  Mass is moved away from that active region.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for _ in range(iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        # draw source from the active interval, destination anywhere else
        i = rng.integers(i_start, i_end + 1)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_shape(n: int) -> np.ndarray:
    """a_i ∝ sin(π * i / (n‑1)), zero at the ends, positive inside."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _semicircle_shape(n: int) -> np.ndarray:
    """Upper half of a unit circle (semi‑ellipse) on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (i - c) / c
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _poly_tent(n: int, power: float = 1.5) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{power}, symmetric tent‑like shape."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) * (n - i)) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Seed‑pool construction (augmented with new families)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_exponential_decay(n, 0.001))

    # new analytic families
    seeds.append(_sin_shape(n))
    seeds.append(_semicircle_shape(n))
    seeds.append(_poly_tent(n, power=1.2))
    seeds.append(_poly_tent(n, power=2.0))

    # power‑law sweep – extended range
    p_vals = np.linspace(0.2, 0.9, 15)          # more exponents
    shifts = [0.0, 5.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # random Dirichlet families (very sparse)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures to explore convex hull
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final cleanup – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main entry point

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # problem size – a modest length gives a good trade‑off
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed collection (including warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Evaluate seeds, retain elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # keep the top‑k seeds for heavy optimisation
    seeds.sort(key=_objective_exact)
    elite = seeds[:min(30, len(seeds))]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on smooth surrogate (aggressive)
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 50_000_000.0, 100_000_000.0,
        500_000_000.0, 1_000_000_000.0, 5_000_000_000.0,
        1_000_000_000_0.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break

        per_beta = max(800, int(min(2500, (deadline - time.time()) * 0.07 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β≈{per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient restarts (more restarts, longer iters)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 10
        iters_per_restart = int(min(600_000, remaining * 0.10 / restarts))
        print("\n--- EG restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.01, 1.3)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.2,
            )
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long EG run from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(8_000_000, remaining * 0.30 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=2.5,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – stochastic local searches
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(200_000, remaining * 45))
        if iters > 0:
            print("\n--- Random mass‑transfer local search ---")
            seq_loc, val_loc = _local_search(
                global_best_seq,

            print(f"[RandomLocal] best = {val_loc:.6f}")

                global_best_val = val_loc

            seq_tgt, val_tgt = _targeted_local_search(

                deadline,
                iters=iters_tgt,

                global_best_val = val_tgt
                print(f"[Update] New best after targeted local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(2500, remaining * 0.04 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000.0, 10_000_000.0, 50_000_000.0,
                           100_000_000.0, 500_000_000.0, 1_000_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation via the provided security‑checked function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
