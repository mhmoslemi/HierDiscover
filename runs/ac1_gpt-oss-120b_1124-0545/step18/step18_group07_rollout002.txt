"""
Propose a step‑function (non‑negative vector) minimizing
evaluate_sequence(sequence).

Algorithm
---------

* A moderate length n (≈3000) is used – large enough to give a good
  search space but still cheap for many iterations.
* A large, diverse pool of ℓ¹‑normalised “seed” vectors is built from
  analytic windows (Gaussian, beta, Tukey, power, edge‑biased …), random
  Dirichlet draws and mixes of them.  An extra fine grid for the most
  promising families (Gaussian sigma, power‑law p, beta a/b) is added.
* The seeds are sorted by a cheap surrogate
          S(a)=2·n·max(conv(a,a))
  and only the best few (default 5) are kept.
* For each seed we run a long **exponentiated‑gradient (mirror‑descent)**
  optimiser on the *exact* max‑convolution objective.
  The update is
        log_a ← log_a – η·g,
        a      ← softmax(log_a)                (projection onto the simplex)
  where g is a sub‑gradient of the active convolution lag.
  The step–size η = c / √t (c≈0.7) and the number of iterations is
  chosen dynamically from the remaining wall‑clock time – up to a few
  million iterations.
* The best sequence found so far is further refined by a **swap‑based
  permutation hill‑climber** that accepts any swap that does not increase
  the exact objective.
* Finally the official ``evaluate_sequence`` is called, the global
  variable ``height_sequence_1`` is updated and the best vector is
  returned.

All heavy work (convolution, max‑convolution) is performed with a
single FFT per evaluation, which is much faster than the double‑FFT
used in the reference implementation and thus leaves more budget for
the expensive mirror‑descent iterations.

The code is completely self‑contained, uses only NumPy (SciPy is not
required) and respects the 1000 s wall‑clock limit.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Full linear convolution a * a using a *single* FFT (real input).
    Returns a 1‑D array of length len(a)*2-1.
    """
    ft = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(ft * ft, n=nfft)[: (2 * len(a) - 1)]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(a: np.ndarray, n: int, nfft: int) -> float:
    """Exact surrogate used during the search: 2·n·max(conv(a,a))."""
    conv = _conv_self_fft(a, nfft)
    return float(2.0 * n * np.max(conv))

# Analytic families – all return ℓ¹‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_bias(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_all_seeds(n: int, rng: np.random.Generator,
                      previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Very diverse pool of ℓ¹‑normalised candidates.
    Includes analytic families, fine‑gridded parameter sweeps and random draws.
    """
    seeds: list[np.ndarray] = []

    # Basic analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))

    # Gaussian – dense sweep of sigma
    for sigma in np.linspace(0.07, 0.30, 15):
        seeds.append(_gaussian(n, sigma))

    # Power‑law – fine sweep of exponent
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta‑shape – grid – extended to strong U‑shapes
    for a in (0.05, 0.1, 0.3, 1.0, 2.0, 4.0):
        for b in (0.05, 0.1, 0.3, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Edge‑biased U‑shapes
    for p in (0.25, 0.5, 0.75, 1.0):
        seeds.append(_edge_bias(n, p))

    # Random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Alternating Dirichlet (helps permutation later)
    seeds.append(_alternating_dirichlet(n, rng))

    # Mix uniform with edge‑biased
    seeds.append(_mix_shapes(_uniform(n), _edge_bias(n, 0.5), 0.3))

    # Previous best, if compatible
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity check (very rarely needed)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     nfft: int,
                     deadline: float,
                     max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n, nfft)

    # work in log‑probability space (softmax = ℓ¹‑projection)
    log_a = np.log(a + 1e-30)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute the full self‑convolution and locate the active lag
        conv = _conv_self_fft(a, nfft)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx
        g[idx] = 2.0 * a[rev_idx]

        # step‑size schedule – a bit more aggressive than 0.5/√t
        eta = 0.7 / math.sqrt(t)

        # mirror‑descent update in log‑space
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n, nfft)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report
        if t % 200_000 == 0:
            print(f"[ExpGrad] iter={t:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Swap‑based permutation hill‑climber (exact objective)
# ----------------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   nfft: int,
                   deadline: float,
                   max_swaps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap only if it does not increase
    the exact surrogate value.  Returns the best sequence found.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq, n, nfft)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # execute swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective_surrogate(cur_seq, n, nfft)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 200_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val

    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_all_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seeds.")

    # ------------------------------------------------------------------
    # 3. Quick surrogate evaluation, keep the best few
    # ------------------------------------------------------------------
    print("\n--- Evaluating seeds (surrogate) ---")
    evaluated = [(vec, _objective_surrogate(vec, n, nfft)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (exponentiated gradient)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq, n, nfft)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all seeds.")
            break

        # allocate iterations based on remaining time
        remaining = deadline - time.time()
        # Rough estimate: we can afford about 2000 iterations per second
        max_iters = int(min(2_500_000, remaining * 2000))
        max_iters = max(300_000, max_iters)   # be generous

        print(f"\n--- Exp‑Gradient on seed #{idx+1} (max_iters={max_iters}) ---")
        seq_opt, val_opt = _exp_gradient_opt(seed_vec, n, nfft,
                                             deadline, max_iters=max_iters)

        print(f"[ExpGrad] best surrogate after seed #{idx+1} = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # Small random perturbation + brief extra EG (helps escape shallow minima)
        if time.time() < deadline:
            perturb = rng.normal(scale=0.02, size=n)
            perturbed = np.maximum(global_best_seq + perturb, 0.0)
            perturbed /= perturbed.sum()
            extra_iters = int(min(150_000, (deadline - time.time()) * 1500))
            if extra_iters > 20_000:
                print(f"[Perturb] extra EG with {extra_iters:,} iters")
                seq_pert, val_pert = _exp_gradient_opt(perturbed, n, nfft,
                                                      deadline, max_iters=extra_iters)
                if val_pert < global_best_val:
                    global_best_seq = seq_pert.copy()
                    global_best_val = val_pert
                    print(f"[Perturb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Permutation‑swap local search (final fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(1_200_000, remaining * 1000))
        max_swaps = max(300_000, max_swaps)
        print("\n--- Permutation swap local search ---")
        seq_perm, val_perm = _perm_swap_opt(global_best_seq, n, nfft,
                                            deadline, max_swaps=max_swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[PermSwap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next round
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
