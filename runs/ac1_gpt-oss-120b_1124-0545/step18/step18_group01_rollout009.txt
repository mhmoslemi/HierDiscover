Algorithm

   uniform, 𝛽‑shapes, exponential decay (including very slow rates),

   (the “surrogate” 2·n·max(conv(a,a)) for Σa=1) and keep the best few.

3. **Smooth‑max Adam optimisation** – for each top seed run a
   continuation optimiser that minimises a *smoothed* max‑convolution
   (log‑sum‑exp) using Adam in the soft‑max (log‑probability) space.
   The smoothing parameter β is increased from 0.1 up to 1e8 so that the

   the current global best.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# Exact surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Objective for a vector with Σa = 1: 2·n·max(conv(a,a))."""
    n = len(seq)
    conv = _conv_full_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    # α∈[0,1]; α=0 → uniform, α=1 → Hann
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·|i-(n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip, renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()

        if time.time() > deadline:
            break

        conv = _conv_full_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)  # classic diminishing stepsize
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Hill‑climbing (random local perturbations)
# ----------------------------------------------------------------------
def _hill_climbing(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   max_attempts: int = 200_000,
                   sigma_start: float = 0.005,
                   sigma_decay: float = 0.9995):
    """Lightweight random‑perturbation search."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    sigma = sigma_start
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s
        val = _objective_surrogate(ns)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns                     # move current point
        sigma *= sigma_decay
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser for the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   max_iters_per_beta: int = 1500):
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) version of the objective.
    The optimisation is performed in the log‑softmax space (guaranteed
    positivity and unit ℓ₁‑norm).  A continuation over increasing β values
    gradually tightens the surrogate to the exact max.
    """
    if beta_schedule is None:
        beta_schedule = [
            0.1, 1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 20_000.0, 50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0, 5_000_000.0, 20_000_000.0, 1e8
        ]

    # initialise Adam state in log‑softmax space
    v = np.log(start_seq + 1e-30)      # shape (n,)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)           # second‑moment accumulator
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t_iter = 0

    best_seq = None
    best_val = math.inf

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(max_iters_per_beta):
            t_iter += 1
            if time.time() > deadline:
                break

            a = _softmax(v)                         # current simplex point
            conv = _conv_full_fft(a, a)              # full linear convolution
            shift = conv.max()                       # numerical stabiliser
            # smooth max via log‑sum‑exp
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                    # ∂smooth_max/∂conv

            # gradient wrt a (∂smooth_max/∂a)  – 2·n factor comes from the overall objective
            rev_a = a[::-1]
            c1 = _conv_full_fft(w, rev_a)            # length = 3n‑2
            g1 = c1[n - 1:n - 1 + n]                 # slice of length n
            c2 = _conv_full_fft(w, a)
            g2 = c2[:n]
            grad_a = 2.0 * n * (g1 + g2)              # ∂obj/∂a

            # chain rule through soft‑max
            grad_v = a * (grad_a - np.dot(grad_a, a))

            # Adam update (scalar step‑size α = 0.1 works well)
            alpha = 0.1
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v * grad_v)
            m_hat = m / (1 - beta1 ** t_iter)
            v_hat_hat = v_hat / (1 - beta2 ** t_iter)
            v = v - alpha * m_hat / (np.sqrt(v_hat_hat) + eps)

            # keep best ever seen (exact surrogate)
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    if best_seq is None:
        best_seq = _softmax(v)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    keeps the best sequence in the global variable ``height_sequence_1``
    and finally returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # Problem size – 4000 works well for the time budget
    n = 4000

    # ------------------------------------------------------------------
    # 1. Seed generation – broad diverse pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # deterministic analytic families (coarse grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, alpha=0.5))

    # beta‑shape sweeps (wide range of alphas/betas)
    for a in [0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        for b in [0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
            seeds.append(_beta_shape(n, a, b))

    # exponential decays – include *very* slow rates
    for rate in np.logspace(-5, -0.5, 12):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # random Dirichlet draws (covering spiky to smooth)
    for conc in [0.001, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 5, 7):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # previous best (if available) – also add perturbed copies
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) == n:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            seeds.append(prev_arr / prev_arr.sum())
            for _ in range(5):
                seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (Adam → EG → Hill‑climbing)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget for this seed
        remaining = deadline - time.time()
        # keep ~15 % of the remaining budget for each upcoming seed + final polish
        sub_deadline = time.time() + remaining * 0.15 / max(1, (top_k - si))

        # ---- Adam on smoothed max ----
        seq_adam, val_adam = _adam_optimize(seed_vec.copy(),
                                            n,
                                            sub_deadline,
                                            beta_schedule=None,
                                            max_iters_per_beta=1500)

        # ---- Exact‑max exponentiated‑gradient polishing ----
        if time.time() < deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam,
                                               n,
                                               deadline,
                                               max_iters=250_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # ---- Short hill‑climbing ----
        hill_deadline = deadline - 0.5      # reserve a little margin
        if time.time() < hill_deadline:
            seq_hc, val_hc = _hill_climbing(seq_eg,
                                            n,
                                            hill_deadline,
                                            rng,
                                            max_attempts=150_000,
                                            sigma_start=0.003,
                                            sigma_decay=0.9997)
        else:
            seq_hc, val_hc = seq_eg, val_eg

        # pick the best among all refinements for this seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_adam, val_adam),
            (seq_eg, val_eg),
            (seq_hc, val_hc)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long EG polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
