"""
Propose a step‑function (non‑negative vector) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm is a three‑stage refinement:

1. **Warm‑start** – use the previous solution (if any) or a uniform vector.
2. **Smooth‑max L‑BFGS** – minimise a log‑sum‑exp surrogate of the
   max‑convolution while staying on the probability simplex (softmax
   parametrisation).  The β‑schedule is extended up to 1e9 and a generous
   iteration budget is allocated.
3. **Exact‑max exponentiated‑gradient** – mirror‑descent on the true
   max‑convolution (sub‑gradient w.r.t. the active lag).  A long run
   (≈ 5·10⁵ – 1·10⁶ updates) drives the solution close to a local optimum.
4. **Fine‑grained annealed mass‑transfer** – repeated random transfers of a
   small amount of mass between two positions, accepting downhill moves
   always and uphill moves with a linearly‑cooling Metropolis rule.
   The convolution is recomputed exactly after each accepted move.

All stages are time‑aware: they stop as soon as the global deadline
( budget seconds − a small safety margin ) is reached.  The best sequence
found is stored in ``height_sequence_1`` and returned as a plain list.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a generous pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 8, 12, 16, 20):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for a in (0.3, 1.0, 2.0, 4.0):
        for b in (0.3, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random vectors
    for k in (max(1, int(0.01 * n)), max(1, int(0.03 * n))):
        for _ in range(2):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors
    for bf in (0.05, 0.15, 0.30):
        for _ in range(2):
            seeds.append(_block_uniform(n, bf, rng))

    # Heavy‑tailed alternating Dirichlet
    seeds.append(_alternating_dirichlet(n, rng))

    # Permuted heavy‑tailed seed (helps permutation search)
    heavy = _random_dirichlet(n, 0.05, rng)
    for _ in range(3):
        seeds.append(heavy[rng.permutation(n)])

    # Previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity‑check – every vector ends up ℓ₁‑normalised.
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """
    Optimise a log‑sum‑exp smoothing of max‑convolution using L‑BFGS‑B.
    Parameters
    ----------
    start_vec : ndarray
        ℓ₁‑normalised start point.
    n : int
        Length of the vector.
    deadline : float
        Unix timestamp when we must stop.
    beta_schedule : list[float] | None
        Increasing smoothing parameters.
    max_iter_per_beta : int
        Upper bound on L‑BFGS iterations for each beta.
    Returns
    -------
    ndarray
        Optimised ℓ₁‑normalised vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy is required for the L‑BFGS stage.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6,
            1e7, 5e7, 1e8, 5e8, 1e9
        ]

    # work in log‑probability space – softmax guarantees the simplex constraint
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (derived from the smoothed max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient optimiser (mirror‑descent)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_objective_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate – classic 1/√t schedule
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 200_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple symmetric averaging (helps a lot for the convolution peak)
# ----------------------------------------------------------------------
def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Average a vector with its reversal and re‑normalise onto the simplex."""
    mixed = (seq + seq[::-1]) * 0.5
    s = mixed.sum()
    return mixed / s if s > 0 else seq.copy()


# ----------------------------------------------------------------------
# Continuous annealed local search (mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealed_mass_transfer(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_moves: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a small amount of mass between two positions.
    Metropolis acceptance with a linearly cooling temperature.
    Returns best sequence seen and its max‑convolution value.
    """
    rng = np.random.default_rng()
    cur_a = start_seq.copy()
    cur_conv = _conv_fft(cur_a, cur_a)
    cur_max = cur_conv.max()
    best_a = cur_a.copy()
    best_max = cur_max

    # Initial temperature – a small fraction of the current peak
    T0 = max(0.02 * cur_max, 1e-12)

    move = 0
    # adaptive step size – starts relatively large and decays
    while move < max_moves and time.time() < deadline:
        move += 1

        i = rng.integers(0, n)
        if cur_a[i] <= 1e-14:
            continue
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n)

        # fraction of mass at i to move; decays exponentially with move count
        frac = max(0.005, math.exp(-move / 200_000))
        delta = frac * cur_a[i]       # ensure we never make a[i] negative
        if delta <= 0.0:
            continue

        a_candidate = cur_a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # Exact recomputation of the convolution (still cheap for n≈4000)
        conv_candidate = _conv_fft(a_candidate, a_candidate)
        max_candidate = conv_candidate.max()

        # Temperature schedule (linear cooling)
        T = max(T0 * (1.0 - move / max_moves), 1e-12)

        accept = (max_candidate < cur_max) or (rng.random() < math.exp(-(max_candidate - cur_max) / T))
        if accept:
            cur_a = a_candidate
            cur_conv = conv_candidate
            cur_max = max_candidate
            if cur_max < best_max:
                best_max = cur_max
                best_a = cur_a.copy()

        if move % 100_000 == 0:
            print(f"[MassTransfer] moves={move:,} best_max={best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final official evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Warm‑start (previous solution or uniform)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is not None and prev_arr.sum() > 0:
        start_vec = prev_arr / prev_arr.sum()
        print("[Init] Using previous solution as warm‑start.")
    else:
        start_vec = _uniform(n)
        print("[Init] No valid previous solution – starting from uniform.")

    # ------------------------------------------------------------------
    # 3. L‑BFGS smoothing stage
    # ------------------------------------------------------------------
    if time.time() < deadline and _opt is not None:
        # allocate roughly 30 % of the remaining budget to this phase
        remaining = deadline - time.time()
        max_iter_per_beta = int(min(20000, remaining * 0.06))
        max_iter_per_beta = max(8000, max_iter_per_beta)
        print(f"\n--- L‑BFGS (smooth max) --- max_iter_per_beta={max_iter_per_beta}")
        seq_lbfgs = _lbfgs_smooth_optimize(start_vec,
                                           n,
                                           deadline,
                                           max_iter_per_beta=max_iter_per_beta)
        val_lbfgs = _objective(seq_lbfgs)
        print(f"[L‑BFGS] objective = {val_lbfgs:.6f}")
    else:
        seq_lbfgs = start_vec
        val_lbfgs = _objective(seq_lbfgs)
        print("[L‑BFGS] Skipped (no time or SciPy missing).")

    # ------------------------------------------------------------------
    # 4. Exact‑max exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # roughly 40 % of the remaining time for the mirror‑descent
        max_iters = int(min(1_200_000, remaining * 500))
        max_iters = max(300_000, max_iters)
        print(f"\n--- Exponentiated‑gradient (exact max) --- max_iters={max_iters}")
        seq_exp, val_exp = _exp_gradient_opt(seq_lbfgs,
                                             n,
                                             deadline,
                                             max_iters=max_iters)
        if val_exp < val_lbfgs:
            seq_lbfgs = seq_exp
            val_lbfgs = val_exp
            print(f"[ExpGrad] improved to {val_exp:.6f}")
        else:
            print("[ExpGrad] no improvement over L‑BFGS result.")
    else:
        print("[ExpGrad] Skipped (deadline reached).")

    # ------------------------------------------------------------------
    # 5. Optional symmetrisation (often beneficial)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sym = _symmetrize(seq_lbfgs)
        val_sym = _objective(seq_sym)
        if val_sym < val_lbfgs:
            seq_lbfgs = seq_sym
            val_lbfgs = val_sym
            print(f"[Symmetrize] improved to {val_sym:.6f}")

    # ------------------------------------------------------------------
    # 6. Fine‑grained annealed mass‑transfer search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_moves = int(min(800_000, remaining * 800))
        max_moves = max(200_000, max_moves)
        print(f"\n--- Annealed continuous local search --- max_moves={max_moves}")
        seq_final, max_conv = _annealed_mass_transfer(seq_lbfgs,
                                                      n,
                                                      deadline,
                                                      max_moves=max_moves)
        val_final = 2.0 * n * max_conv
        if val_final < val_lbfgs:
            seq_lbfgs = seq_final
            val_lbfgs = val_final
            print(f"[MassTransfer] improved to {val_final:.6f}")
        else:
            print("[MassTransfer] no improvement.")
    else:
        print("[MassTransfer] Skipped (deadline reached).")

    # ------------------------------------------------------------------
    # 7. Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq_lbfgs.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_lbfgs:.6f}, official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store globally for the next round
    globals()["height_sequence_1"] = seq_lbfgs.tolist()
    return seq_lbfgs.tolist()
