def _adam_opt(start_a, n, deadline, beta_schedule, steps_per_beta, lr=0.1):
    v = np.log(start_a + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1=0.9; beta2=0.999; eps=1e-8; t=0

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)  # maybe for shift in smoothing. But we need conv for gradient.
            # shift to avoid overflow:
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp  # weights
            # compute grad_a = 2 * conv_rev(w)
            rev_a = a[::-1]
            # convolution of rev_a and w:
            grad_a = 2.0 * _conv_fft(rev_a, w)[n-1:n-1+n]
            # compute gradient w.r.t v: grad_v = a * (grad_a - dot(grad_a, a))
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam step:
            t += 1
            m = beta1 * m + (1-beta1) * grad_v
            v_hat = beta2 * v_hat + (1-beta2) * (grad_v**2)
            m_hat = m / (1 - beta1**t)
            v_hat_corr = v_hat / (1 - beta2**t)
            v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

    return _softmax(v)

def _adam_optimize(start_a, n, deadline, beta_schedule, steps_per_beta, lr0=0.1):
    v = np.log(start_a + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    best_seq = None
    best_val = float('inf')
    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smoothing for stability, but compute weights as exp(beta*(conv - max_c))
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat_corr = v_hat / (1 - beta2 ** t)

            # optionally track best with surrogate
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    if best_seq is None:

    return best_seq

"""
Propose candidate – multi‑start Adam + L‑BFGS refinement.

The search proceeds in three phases:

1. **Seed generation** – a diverse collection of analytic windows,
   Dirichlet / exponential draws, low‑frequency cosine curves and
   sparse random supports.  All vectors are ℓ₁‑normalised (sum = 1).

2. **Select & mix** – keep the best few seeds (according to the cheap
   surrogate 2 · n · max(conv)) and improve each of them by a
   two‑stage optimisation:
        • Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
        • A final continuation L‑BFGS pass (the routine already used in
          the reference implementation) for fine‑grained polishing.

3. **Global polishing** – a long exponentiated‑gradient (mirror‑descent)
   run on the exact max‑convolution, starting from the best sequence
   found in phase 2.

All work stops automatically when the overall time budget (default 1000 s)
is exhausted.  The best sequence discovered is stored in the global
variable ``height_sequence_1`` and returned as a plain list of floats.
"""

import math
import numpy as np

# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a ℓ₁‑normalised vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                                   # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """m distinct positions get random positive weights (then normalise)."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = rng.random(m) + 0.1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Convex mixing of two shapes (convex combination remains ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    """Ternary search for the best convex combination of a and b."""
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        v1 = _objective_surrogate(seq1)
        v2 = _objective_surrogate(seq2)
        if v1 < v2:
            hi = m2
            if best_val is None or v1 < best_val:
                best_val = v1
                best_seq = seq1
        else:
            lo = m1
            if best_val is None or v2 < best_val:
                best_val = v2
                best_seq = seq2
    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective_surrogate(best_seq)
    else:
        mid = (lo + hi) / 2.0
        cand = _mix_shapes(a, b, mid)
        v = _objective_surrogate(cand)
        if v < best_val:
            best_seq = cand
            best_val = v
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """Optimise the smoothed max‑convolution objective using L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 20000.0,
                         50000.0, 100000.0, 200000.0, 500000.0,
                         1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimisation on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   steps_per_beta: int = 2000,
                   lr0: float = 0.1):
    """
    Adam optimiser on the smoothed max‑convolution defined by a
    log‑sum‑exp with a temperature β.
    Returns the best sequence observed (ℓ₁‑normalised).
    """
    v = np.log(start_a + 1e-30)                # log‑softmax parameters
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_seq = None
    best_val = float('inf')

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            a = _softmax(v)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                        # numerical stability
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp               # soft‑max weights over conv entries

            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]   # shape (n,)

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule through softmax

            t += 1
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)

            m_hat = m / (1 - beta1 ** t)
            v_hat_corr = v_hat / (1 - beta2 ** t)

            lr = lr0 / math.sqrt(t)             # simple decay
            v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    if best_seq is None:
        best_seq = _softmax(v)
    return best_seq

    # Sparse random supports (different cardinalities)
    for m in (int(0.04 * n), int(0.06 * n), int(0.08 * n)):
        for _ in range(4):
            seeds.append(_random_sparse(n, m, rng))
    # Alternating Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # Inject a perturbed copy of the previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Pairwise mixing of top seeds (optional – cheap and often fruitful)
    # ------------------------------------------------------------------
    print("\n--- Pairwise mixing of top seeds ---")
    mixture_candidates = []
    mix_deadline = deadline - 1.0
    for i in range(len(top_seeds)):
        for j in range(i + 1, len(top_seeds)):
            if time.time() > mix_deadline:
                break
            seq_mix, val_mix = _best_mixed_pair(top_seeds[i],
                                                top_seeds[j],
                                                mix_deadline,
                                                steps=25)
            mixture_candidates.append((seq_mix, val_mix))

    # Combine raw seeds and mixtures
    all_candidates = evaluated[:top_k] + mixture_candidates
    all_candidates.sort(key=lambda x: x[1])
    best_seq = all_candidates[0][0].copy()
    best_val = all_candidates[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (Adam + L‑BFGS)
    # ------------------------------------------------------------------
    print("\n--- Refinement of top candidates ---")
    refined_candidates = []

    # We'll refine up to 8 of the most promising candidates
    max_refine = min(8, len(all_candidates))
    for idx in range(max_refine):
        if time.time() >= deadline:
            break
        seed_vec, _ = all_candidates[idx]

        # allocate a slice of the remaining budget for this candidate
        now = time.time()
        remaining = deadline - now
        # give roughly equal share, but keep a little for final polishing
        sub_deadline = now + max(1.0, remaining * 0.45 / (max_refine - idx))

        print(f"\n[Refine #{idx+1}] start surrogate = {_objective_surrogate(seed_vec):.6f}")

        # ---- Adam optimisation on smoothed max ----
        beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                      1000.0, 2000.0, 5000.0, 20000.0,
                      50000.0, 100000.0, 200000.0]
        seq_adam = _adam_optimize(seed_vec,
                                  n,
                                  sub_deadline,
                                  beta_schedule=beta_sched,
                                  steps_per_beta=2000,
                                  lr0=0.12)
        val_adam = _objective_surrogate(seq_adam)

        # ---- L‑BFGS continuation for extra polishing ----
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_adam,
                                               n,
                                               sub_deadline,
                                               beta_schedule=beta_sched,
                                               max_iter_per_beta=2500)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs = seq_adam
            val_lbfgs = val_adam

        # Keep the best of raw seed / Adam / L‑BFGS
        candidates = [(seed_vec, _objective_surrogate(seed_vec)),
                      (seq_adam, val_adam),
                      (seq_lbfgs, val_lbfgs)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")

        refined_candidates.append((local_best_seq, local_best_val))

        if local_best_val < best_val:
            best_val = local_best_val
            best_seq = local_best_seq.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long EG polishing (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (exact max) ---")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=5.0)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final L‑BFGS smoothing (optional, may lower the max slightly)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing ---")
        seq_final_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                                n,
                                                deadline,
                                                beta_schedule=beta_sched,
                                                max_iter_per_beta=2500)
        val_final_lbfgs = _objective_surrogate(seq_final_lbfgs)
        if val_final_lbfgs < best_val:
            best_seq = seq_final_lbfgs
            best_val = val_final_lbfgs
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
