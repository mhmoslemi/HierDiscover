def _smooth_obj_and_grad(logits, beta, n):
    a = _softmax(logits)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shifted = conv - max_c

    sum_exp = np.sum(exp_beta)

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

The algorithm differs from the previous baseline:

* builds a diverse set of analytic and random Dirichlet seeds;
* picks the best seed (surrogate = 2·n·max(conv));
* refines that seed with an Adam‑style mirror‑descent on a **smooth**
  max‑convolution (log‑sum‑exp surrogate) in logit space;
* the smoothness parameter β is increased gradually (1 → 1e6) so that
  early iterations receive a well‑behaved gradient and later iterations
  focus on the true max;
* learning‑rates decay as 1/√t and are additionally scaled ∝1/√β;
* the best vector found is finally evaluated with the official
  ``evaluate_sequence`` routine and stored globally for the next run.

No external files are accessed; only ``numpy`` (and optionally ``scipy`` for an
optional L‑BFGS polish if time permits) are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for L‑BFGS polishing (used only if available)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via a real FFT (O(n log n)).
    The returned array has length len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exact surrogate (objective when the sum is normalised to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth max‑convolution objective (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(logits: np.ndarray, beta: float, n: int):
    """
    Given logits (log‑probabilities), compute

        obj   = 2·n·[ max(conv) + log‑sum‑exp(β·(conv‑max))/β ]
        grad  = ∂obj/∂logits   (shape = logits)

    Returns (obj, grad, max_conv_entry).
    """
    a = _softmax(logits)

    # ---------- convolution a*a ----------
    N1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=N1)
    conv = np.fft.irfft(a_fft * a_fft, n=N1)[: 2 * n - 1]

    max_c = np.max(conv)
    shifted = conv - max_c

    # ---------- smooth max (log‑sum‑exp) ----------
    exp_beta = np.exp(beta * shifted)          # max_c is subtracted → safe
    sum_exp = np.sum(exp_beta)
    smooth_max = max_c + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max

    # ---------- weights for the gradient ----------
    w = exp_beta / sum_exp                     # length 2n‑1, sums to 1

    # ---------- gradient wrt a (via correlation) ----------
    N2 = _next_pow2(3 * n - 2)
    rev_a = a[::-1]
    rev_fft = np.fft.rfft(rev_a, n=N2)
    w_fft = np.fft.rfft(w, n=N2)
    grad_full = np.fft.irfft(rev_fft * w_fft, n=N2)[: 3 * n - 2]
    grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]

    # ---------- chain rule through the softmax ----------
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return obj, grad_logits, max_c


# ----------------------------------------------------------------------
# Adam‑style mirror‑descent in logit space
# ----------------------------------------------------------------------
def _adam_optimize(init_logits: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   iters_per_beta: int,
                   lr0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by minimising the smooth‑max objective.
    Returns the best probability vector encountered and its surrogate value.
    """
    v = init_logits.copy()
    m = np.zeros_like(v)          # first moment
    v2 = np.zeros_like(v)         # second moment

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t = 0                         # iteration counter
    pow1 = 1.0
    pow2 = 1.0

    best_seq = None
    best_val = np.inf
    finished_early = False

    for beta in beta_schedule:
        # a simple scaling of the learning rate for larger β
        beta_lr_factor = 1.0 / math.sqrt(beta) if beta > 0 else 1.0

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                finished_early = True
                break

            t += 1
            pow1 *= beta1
            pow2 *= beta2

            obj, grad, max_c = _smooth_obj_and_grad(v, beta, n)
            cur_val = 2.0 * n * max_c
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()

            # Adam update (learning‑rate decays like 1/√t)
            eta = lr0 * beta_lr_factor / math.sqrt(t)

            m = beta1 * m + (1.0 - beta1) * grad
            v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - pow1)
            v_hat = v2 / (1.0 - pow2)

            v = v - eta * m_hat / (np.sqrt(v_hat) + eps)

        print(f"[Adam] β={beta:.1e}, best surrogate so far = {best_val:.6f}")
        if finished_early:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS polish (smooth surrogate) – identical to the earlier code
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 2000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:  # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

def propose_candidate(seed: int = 123,

    """

    metric.  The routine builds many analytic/random seeds, selects the best,
    and then refines it with a smooth‑max Adam optimiser (plus an optional
    L‑BFGS polish if time permits).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep the same size as the previous best (4000)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seeds (analytic + Dirichlet)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _sin2(n),
        _hann_window(n),
        _arcsine(n),
    ]

    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # random Dirichlet draws with a wide range of concentration parameters
    conc_params = np.logspace(-2, 1, 12)  # 0.01 … 10, a few values
    for conc in conc_params:
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # include the previous best if it exists and has the right length
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # normalise everything onto the probability simplex
    def _norm(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / len(v))
        return v / s

    seeds = [_norm(np.asarray(v, dtype=np.float64)) for v in seeds]

    # ------------------------------------------------------------------
    # 2. Quick surrogate ranking – pick the best seed
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for seq in seeds:
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"\n[Init] best surrogate among seeds: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam‑style refinement on the smooth max‑convolution
    # ------------------------------------------------------------------
    init_logits = np.log(best_seq + 1e-30)

    # β schedule – start smooth and finish extremely sharp
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        20_000.0, 50_000.0, 100_000.0,
        200_000.0, 500_000.0, 1_000_000.0,
    ]

    # iterations per β – tuned to stay well within the time budget
    iters_per_beta = 4000

    seq_adam, val_adam = _adam_optimize(init_logits, n, deadline,
                                        beta_schedule, iters_per_beta,
                                        lr0=0.25)

    # In case the Adam optimiser never updated (highly unlikely)
    if seq_adam is None:
        seq_adam = best_seq
        val_adam = best_val
    else:
        # compute the exact surrogate for the final Adam output
        val_adam = _objective_surrogate(seq_adam)

    if val_adam < best_val:
        best_val = val_adam
        best_seq = seq_adam.copy()
        print(f"[Adam] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional short L‑BFGS polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline and _opt is not None:
        remaining = deadline - time.time()
        # give at most a few seconds to L‑BFGS; limit iterations per β
        max_iter_per_beta = max(300, int(min(1500, remaining * 0.02 * 1e5)))
        print("\n--- L‑BFGS polishing (short) ---")
        seq_polished = _lbfgs_optimize(best_seq, n, deadline,
                                       beta_schedule=[1, 5, 20, 80, 200, 500,
                                                      1_000, 2_000, 5_000],
                                       max_iter_per_beta=max_iter_per_beta)
        val_polished = _objective_surrogate(seq_polished)
        print(f"[L‑BFGS] polished surrogate = {val_polished:.6f}")
        if val_polished < best_val:
            best_val = val_polished
            best_seq = seq_polished.copy()
            print(f"[Update] new global best after L‑BFGS = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for the next round
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
