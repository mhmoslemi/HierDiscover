"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1.  Build a diverse pool of analytically‑parameterised seeds:
    – uniform, exponentials (fine λ‑grid), power‑laws (fine p‑grid),
      random mixtures of the two, random Dirichlet vectors and a few
      sparse vectors.
    – optionally add the previous best sequence (if it has the same
      length).

2.  Normalise every seed to ℓ₁‑sum = 1 and evaluate the *exact* objective
    (2 n·max(conv)/‖a‖₁²).  Keep the best few (the “elite”) for the
    heavy optimisation stage.

3.  Continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the
    max‑convolution, starting from each elite seed.  The β‑schedule is
    very aggressive (up to 5 · 10⁶) and each β gets a modest number of
    iterations – this quickly drives the surrogate towards the true
    max‑convolution.

4.  Refinement by exponentiated‑gradient (mirror descent) on the *exact*
    nonsmooth objective.  The sub‑gradient of the active lag is computed
    analytically; a decaying step‑size  ηₜ = 1/√t  is used.  The loop
    stops when the time budget is exhausted.

5.  The best sequence found is stored globally as ``height_sequence_1``
    (for warm‑starts in future calls) and returned as a plain list.

The code respects the 1000 s budget, prints progress, and avoids any
closures or lambda expressions as required.
"""
import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex (stable)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# exact objective & sub‑gradient helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² .  a may be any non‑negative vector."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the active lag.
    Returns a vector g such that g_i = 2·a_{ℓ−i} for indices i that
    participate in the maximal lag ℓ (0‑based).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))                     # active lag
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i).  lam > 0."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p},  p > 0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_expo_power(n: int, lam: float, p: float, alpha: float) -> np.ndarray:
    """α·exp(‑lam·i) + (1‑α)·(i+1)^{‑p}, then ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = alpha * np.exp(-lam * i) + (1.0 - alpha) * (i + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """Place equal mass on a random fraction of the coordinates."""
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# smooth surrogate (log‑sum‑exp) for continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate of the max‑convolution objective."""
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)              # full convolution
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient wrt a
    w = exp_term / sum_exp               # soft‑max over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]  # slice the middle n values

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1000,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:  # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 20_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[EG] iter {it:,} – new best {best_val:.6f}")

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val
