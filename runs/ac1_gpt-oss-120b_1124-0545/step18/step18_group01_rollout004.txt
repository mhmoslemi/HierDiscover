def _adam_flat_opt(start_seq, n, deadline, max_iters=200000, lr=0.1):
    a = start_seq.copy()
    v = np.log(a+1e-30) # or maybe directly a for easier projection but we will use log domain
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    
    target_val = 1.0/(2*n-1)
    # constant target array length 2n-1
    t = np.full(2*n-1, target_val, dtype=np.float64)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(v)  # ensure sum=1
        # compute conv
        conv = _conv_fft(a, a)  # length 2n-1
        w = conv - t
        # gradient w.r.t a: 2 * rev_a conv w
        rev_a = a[::-1]
        grad_a = 2.0 * _conv_fft(rev_a, w)[n-1:n-1+n]
        # gradient wrt v:
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1-beta1) * grad_v
        v2 = beta2 * v2 + (1-beta2) * (grad_v**2)
        m_hat = m / (1 - beta1**it)
        v2_hat = v2 / (1 - beta2**it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # evaluate surrogate (max conv)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    
    return best_seq, best_val

def _proj_simplex(v):
    """Projects v onto probability simplex (non-negative, sum=1)."""
    # https://doi.org/10.1021/ac0710415
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

   self‑convolution and the *flat* target 1/(2n‑1) (the theoretical lower‑bound
   for a constant convolution).  The optimisation is performed in the
   log‑softmax (mirror‑descent) space with an Adam optimiser.  The surrogate
   (max‑convolution) is monitored and the best sequence is kept.

3. **Final polishing** – a long exponentiated‑gradient (sub‑gradient on the
   true max) run on the best candidate found in step 2.

All phases respect the overall time budget (default 1000 s) and the best
sequence is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate (exact) objective for ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (sum=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Projection onto the probability simplex (for optional use)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser that drives the convolution toward a flat target
# ----------------------------------------------------------------------
def _adam_flat_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr: float = 0.10) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in log‑softmax space that minimises
    ‖a*a - 1/(2n‑1)·1‖₂².
    Returns the best sequence (w.r.t. the max‑convolution surrogate)
    encountered before the deadline.
    """
    # initialise log‑probabilities
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    target_val = 1.0 / (2 * n - 1)
    target = np.full(2 * n - 1, target_val, dtype=np.float64)

    best_seq = None
    best_val = np.inf

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)

        # convolution and residual
        conv = _conv_fft(a, a)                     # length 2n‑1
        w = conv - target

        # gradient w.r.t. a : 2 * rev_a * w   (restricted to n entries)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain‑rule to obtain gradient w.r.t. v
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        # parameter update
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # evaluate surrogate (max‑convolution)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    # in the unlikely case no improvement was recorded
    if best_seq is None:
        best_seq = _softmax(v)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing (random perturbations) – used only as a cheap
# fallback after the Adam stage.
# ----------------------------------------------------------------------
def _hill_climbing(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   max_attempts: int = 150_000,
                   sigma_start: float = 0.005,
                   sigma_decay: float = 0.9995) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    sigma = sigma_start
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s
        val = _objective_surrogate(ns)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns
        sigma *= sigma_decay
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0    # small safety margin

    # --------------------------------------------------------------
    # Problem size – a trade‑off between expressiveness and speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load a previous solution if present
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1️⃣  Seed generation – a very wide pool
    # --------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_power(n, p=0.5))
    seeds.append(_power(n, p=2.0))
    seeds.append(_exp_decay(n, rate=0.001))
    seeds.append(_sym_exp_decay(n, rate=0.001))

    # random Dirichlet draws with diverse concentrations
    for conc in (0.02, 0.05, 0.1, 0.3, 0.7, 1.5, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random cosine curves
    for nf in (2, 4, 6):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform and low‑freq curves
    for mix_alpha in (0.2, 0.5, 0.8):
        curve = _lowfreq_curve(n, 4, rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # a few random permutations of promising shapes
    for base in [_gaussian(n, 0.18), _sine_window(n, gamma=1.2), _beta_shape(n, 2.0, 2.0)]:
        for _ in range(2):
            seeds.append(np.random.permutation(base))

    # perturbations of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(3):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # --------------------------------------------------------------
    # 2️⃣  Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seed could be generated.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))          # a modest number – we invest more time per seed
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\nInitial global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Multi‑start optimisation (Adam → flat target)
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fraction of the remaining budget for this seed
        remaining = deadline - time.time()
        # keep a small reserve for the final polishing
        per_seed_budget = remaining * 0.45 / max(1, (top_k - si))
        sub_deadline = time.time() + per_seed_budget

        # ----- (a) Adam optimisation towards a flat convolution -----
        seq_flat, val_flat = _adam_flat_opt(seed_vec.copy(),
                                            n,
                                            sub_deadline,
                                            max_iters=250_000,
                                            lr=0.12)

        # ----- (b) short hill‑climbing to fine‑tune the max ----- 
        hill_deadline = deadline - 2.0          # leave a couple seconds for final stage
        if time.time() < hill_deadline:
            seq_hc, val_hc = _hill_climbing(seq_flat,
                                            n,
                                            hill_deadline,
                                            rng,
                                            max_attempts=120_000,
                                            sigma_start=0.004,
                                            sigma_decay=0.9996)
        else:
            seq_hc, val_hc = seq_flat, val_flat

        # Choose the best among the refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_flat, val_flat),
            (seq_hc, val_hc)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final long exponentiated‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final exponentiated‑gradient polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
