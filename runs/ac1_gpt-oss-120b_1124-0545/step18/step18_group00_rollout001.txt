"""
Propose a step‑function (non‑negative heights) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq,seq)) / (sum(seq)**2)

The evaluation is invariant to scaling, so we enforce ℓ₁‑normalisation
(sum = 1).  The search proceeds in several stages:

1. **Generate a diverse pool of analytic and random seeds** (window
   functions, Dirichlet draws, low‑frequency cosine curves, etc.).
   All vectors are ℓ₁‑normalised.

2. **Random convex mixing of top seeds** – we repeatedly draw a random
   subset of the best few seeds, sample Dirichlet mixing coefficients,
   form the mixture and keep the best mixture (the surrogate objective
   is ``2·n·max(conv)`` for sum‑normalised vectors).

3. **Smooth‑max L‑BFGS refinement** – the surrogate is the
   log‑sum‑exp smooth approximation of the max.  Optimisation runs in
   the log‑softmax (simplex) space, using a schedule of increasing
   smoothing parameters β.  We keep the best iterate seen.

4. **Simple hill‑climbing on the simplex** – random transfers of a tiny
   amount of mass between two coordinates are accepted if they lower
   the exact max‑convolution; a temperature schedule allows occasional
   uphill moves.

The whole pipeline respects a global time budget (default 1000 s).  The
best sequence found is stored in ``height_sequence_1`` for later runs.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# Optional SciPy import for L‑BFGS
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimizer
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: List[float] = None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        # A dense schedule that slowly sharpens the smooth‑max.
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1e6]

    # work in log‑softmax space
    v = np.log(start_vec + 1e-30)

    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth surrogate of max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (softmax space)
            w = exp_beta / sum_exp                      # weights for each entry of conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                  # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
            a = _softmax(v)
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return best_seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 6.0) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on exact max‑convolution.
    Returns best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (see derivation in comments)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random convex mixing of a pool of seeds
# ----------------------------------------------------------------------
def _random_mixing(seeds: List[np.ndarray],
                   top_ids: List[int],
                   n: int,
                   deadline: float,
                   mixes: int = 3000,
                   max_group: int = 4) -> Tuple[np.ndarray, float]:
    """
    Perform random convex mixes of a selected subset of seeds.
    Returns the best mixture and its surrogate value.
    """
    best_seq = seeds[top_ids[0]].copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng(12345)

    for _ in range(mixes):
        if time.time() > deadline:
            break
        # pick a random group size (2…max_group)
        group_size = rng.integers(2, max_group + 1)
        # choose distinct seed indices from the top pool
        ids = rng.choice(top_ids, size=group_size, replace=False)
        # Dirichlet weights
        w = rng.dirichlet(np.ones(group_size))
        mix = np.zeros(n, dtype=np.float64)
        for idx, weight in zip(ids, w):
            mix += weight * seeds[idx]
        # numerical drift
        mix = mix / mix.sum() if mix.sum() > 0 else best_seq.copy()
        val = _objective_surrogate(mix)
        if val < best_val:
            best_val = val
            best_seq = mix.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing on the simplex
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 200_000,
                init_step: float = 0.01) -> Tuple[np.ndarray, float]:
    """
    Randomly move a tiny amount of mass between two coordinates.
    Accepts improvements; occasional uphill moves via a temperature schedule.
    """
    rng = np.random.default_rng(987654)
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    step = init_step
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # exponential decay of step size
        step = init_step * (0.9995 ** it)
        # select two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # amount to move (bounded by current mass at i)
        epsilon = step * rng.random()
        eps = min(epsilon, cur[i])
        if eps <= 0:
            continue
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        # renormalise not needed (mass conserved)
        cand_val = _objective_surrogate(cand)
        # accept if better or with temperature (basic simulated annealing)
        if cand_val < cur_val or rng.random() < 0.001:
            cur = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur.copy()
    return best_seq, best_val

    n = 8000

    # ------------------------------------------------------------------
    # Load previous solution (if any) – used only for occasional perturbation.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Broad seed generation (analytic families + random draws)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds: List[np.ndarray] = []

    # Analytic families (coarse grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 10):
        seeds.append(_sine_window(n, gamma=gamma))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in np.linspace(1.0, 12.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 12):
        seeds.append(_gaussian(n, sigma))
    for sigma in (0.10, 0.12):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))
    for p in np.linspace(0.0, 2.0, 12):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.3, 5.0, 6)
    betas = np.linspace(0.3, 5.0, 6)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))
    for alpha in np.linspace(0.1, 0.9, 6):
        seeds.append(_tukey_window(n, alpha))
    # Random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))
    # Alternating Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # Add perturbed copies of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(4):
            seeds.append(_perturb(prev_arr, rng, sigma=0.004))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated: List[Tuple[np.ndarray, float]] = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_ids = list(range(top_k))  # indices in evaluated list
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"\nTop {top_k} surrogate values among seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Random convex mixing of top seeds
    # ------------------------------------------------------------------
    mix_deadline = deadline - 5.0
    print("\n--- Random mixing of top seeds ---")
    mix_seq, mix_val = _random_mixing(
        seeds=[vec for vec, _ in evaluated],
        top_ids=top_ids,
        n=n,
        deadline=mix_deadline,
        mixes=5000,
        max_group=4
    )
    best_seq = mix_seq
    best_val = mix_val
    print(f"[Mix] best surrogate after random mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start L‑BFGS (smooth‑max) refinement on a few candidates
    # ------------------------------------------------------------------
    refined_candidates: List[Tuple[np.ndarray, float]] = []

    # Take the best few (up to 5) seeds+mixes for refinement
    start_candidates = [best_seq] + [vec for vec, _ in evaluated[:4]]
    for idx, cand in enumerate(start_candidates):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + (deadline - time.time()) * 0.45 / max(1, (len(start_candidates) - idx))
        print(f"\n[Refine #{idx+1}] L‑BFGS start with surrogate {_objective_surrogate(cand):.6f}")

        cand_lbfgs = _lbfgs_smooth_optimize(
            start_vec=cand,
            n=n,
            deadline=sub_deadline,
            beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0,
                           500.0, 1000.0, 2500.0, 5000.0,
                           20000.0, 50000.0, 100000.0,
                           200000.0, 500000.0, 1e6],
            max_iter_per_beta=4000
        )
        cand_val = _objective_surrogate(cand_lbfgs)
        refined_candidates.append((cand_lbfgs, cand_val))

        if cand_val < best_val:
            best_seq = cand_lbfgs
            best_val = cand_val
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient polishing (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing ---")
        seq_eg, val_eg = _exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=800_000,
            eta0=7.0
        )
        if val_eg < best_val:
            best_seq = seq_eg
            best_val = val_eg
            print(f"[EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Simple hill‑climbing (mass transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Hill climbing phase ---")
        seq_hc, val_hc = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=300_000,
            init_step=0.02
        )
        if val_hc < best_val:
            best_seq = seq_hc
            best_val = val_hc
            print(f"[Hill] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
