"""
propose_candidate
-----------------
A fast multi‑start exponentiated‑gradient + cheap‑parameter search
for the max‑auto‑convolution problem.

The algorithm works in three phases:

1. **Broad seed generation**  
   - Randomised β‑shape families, edge‑biased families, Dirichlet draws,
     simple analytic windows and the previously‑found solution (if any) are
     turned into ℓ₁‑normalised vectors.
   - All seeds are evaluated with the cheap surrogate  
     `2·n·max(conv(a,a)) / (sum(a)**2)` and the best few (default 10) are kept.

2. **Short Exp‑Gradient polishing**  
   For each of the kept seeds a limited mirror‑descent (exponentiated‑gradient)
   run is performed (≈20 k–200 k iterations, limited by the remaining time).
   The best result of these runs becomes the *current best*.

3. **Long final optimisation**  
   – A long Exp‑Gradient run (up to a few million iterations) refines the
     current best.  
   – A random‑swap hill‑climber is then applied (up to a few hundred‑thousand
     swaps) while never increasing the exact objective.  
   – A short final Exp‑Gradient polish finishes the search.

All steps are time‑aware: the global 1000 s budget (minus a small safety
margin) is respected, and the function always returns the best sequence
found before the deadline.  The returned list can be passed directly to the
provided ``evaluate_sequence`` function.

The implementation uses only ``numpy`` and, if available, ``scipy`` for the
optional L‑BFGS smooth optimiser (not required for the core logic).
"""

import time
import math
import numpy as np

# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                     # should never happen
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    conv = _conv_fft(seq, seq)
    maxb = np.max(conv)
    s = np.sum(seq)
    if s == 0.0:
        return float("inf")
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _edge_bias(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution concentrating mass near the ends."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration parameter ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mix of two ℓ₁‑normalised vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning‑rate schedule (1/√t)
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple swap‑based permutation hill‑climber
# ----------------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap only if it reduces the exact
    max‑convolution.  Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq, n)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_val = _objective_surrogate(cur_seq, n)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 100_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val
