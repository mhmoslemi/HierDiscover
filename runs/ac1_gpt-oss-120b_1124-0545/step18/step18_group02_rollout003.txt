"""
Propose a step‑function that minimises ``evaluate_sequence``.

The search is completely autonomous and runs for up to 1000 s.
It proceeds in several layers:

1. **Broad seed generation** – deterministic families (uniform, triangular,
   Gaussian, β‑shape, exponential, logarithmic, low‑frequency sum‑of‑cosines)
   together with an extensive power‑law sweep (exponent 0.1…1.5,
   shift up to half the length).  Random Dirichlet, sparse and mixtures
   are also added.  All seeds are ℓ₁‑normalised to the required mass
   √(2·n).

2. **Continuation L‑BFGS** – each elite seed (top‑30) is refined on a
   log‑sum‑exp smooth surrogate of the max‑convolution objective.
   The β‑schedule is long (up to 1e12) and each β receives many L‑BFGS
   iterations, allowing the optimiser to approach the true max.

3. **Exponentiated‑gradient (mirror‑descent)** – many restarts with
   aggressive step‑size are run on the exact non‑smooth objective.
   The best result of this phase becomes the new global incumbent.

4. **Simulated‑annealing mass‑transfer** – a stochastic hill‑climber
   that occasionally accepts worsening moves.  The temperature decays
   exponentially, enabling escape from shallow local minima.

5. **Final high‑β polishing** – a short L‑BFGS run with ultra‑large β
   (≈1e13) cleans up any remaining roughness.

The routine respects the wall‑time limit, updates the global variable
``height_sequence_1`` with the best found list and returns it.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (max‑convolution)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    # contributions are 2*a[idx-i] for i where both indices lie in [0,n)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approx of 2·n·max(conv) / (∑a)².
    Returns (obj, grad) – gradient w.r.t. logits v.
    """
    a = _softmax(v)               # ℓ₁‑normalised (sum = 1)
    conv = _conv_fft(a, a)        # full convolution, length = 2n‑1
    shift = np.max(conv)          # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp                 # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)             # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]   # slice the central n entries

    # back‑propagation through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float],
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    ``start_vec`` is already ℓ₁‑normalised (sum = 1).
    """
    # work in logits space
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "disp": False,
                                         "ftol": 1e-13})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)          # still sum‑1


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 1_500_000,
                           eta0: float = 2.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing stochastic local search
# ----------------------------------------------------------------------
def _annealing_search(start_a: np.ndarray,
                      n: int,
                      deadline: float,
                      iters: int = 80_000,
                      init_temp: float = 0.02,
                      final_temp: float = 1e-6,
                      step_frac: float = 0.025) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb with temperature.
    ``step_frac`` controls the maximal fraction of mass moved in a single move.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    rng = np.random.default_rng(123456789)

    # geometric temperature schedule
    temp = init_temp
    decay = (final_temp / init_temp) ** (1.0 / max(1, iters - 1))

    for _ in range(iters):
        if time.time() > deadline:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = step_frac * a[i] * rng.random()
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = _objective_exact(cand)
        diff = cand_val - best_val

        # accept if improvement or with Metropolis probability
        if diff < 0 or rng.random() < np.exp(-diff / temp):
            a = cand
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()

        temp *= decay
    return best_a, best_val

# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _log_decay(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ 1 / (log(i + shift + e))^{p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (np.log(i + shift + np.e) ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)

# Seed‑pool construction (augmented)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # Wider power‑law sweep
    p_vals = np.linspace(0.10, 1.50, 15)          # 15 exponents
    shifts = [0.0, 2.0, 5.0, 20.0, 50.0, 150.0, 300.0]
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # Log‑decay family
    for p in (0.5, 1.0, 1.5):
        for sh in (0.0, 5.0, 20.0, 100.0):
            seeds.append(_log_decay(n, float(p), float(sh)))
            seeds.append(_log_decay(n, float(p), float(sh))[::-1])

    # Random Dirichlet families (favoring sparsity)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.6):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse families
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixtures (convex hull exploration)
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of promising seeds
    for _ in range(20):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Cleanup – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned
