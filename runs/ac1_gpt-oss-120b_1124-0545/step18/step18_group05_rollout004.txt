def _upsample_sequence(seq_small, size_large):
    x_small = np.arange(len(seq_small))
    x_large = np.linspace(0, len(seq_small)-1, size_large)
    seq_large = np.interp(x_large, x_small, seq_small)

    seq_large_sum = seq_large.sum()
    if seq_large_sum == 0:
        seq_large = np.full(size_large, 1.0/size_large)
    else:
        seq_large = seq_large / seq_large_sum
    return seq_large

def _adam_optimize(start_vec, n, deadline, beta=200, max_iters=20000, lr=0.1):
    v = np.log(start_vec + eps)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    for t in range(1, max_iters+1):
        if time.time()>deadline: break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shifted = conv - max_c
        exp_beta = np.exp(beta * shifted)
        sum_exp = exp_beta.sum()
        obj = 2.0 * n * (max_c + np.log(sum_exp)/beta)
        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w) # length 3n-2
        grad_a = 2.0 * full[n-1:n-1+n]
        # grad w.r.t logits (v)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat_corr = v_hat / (1 - beta2**t)
        v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps_adam)
        # optionally monitor obj
        if t % 1000 == 0:
            val = _objective_surrogate(_softmax(v))
            if val < best_val:
                best_val=val; best_seq=_softmax(v).copy()
    return _softmax(v)
```

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

Algorithm
---------

1. **Small‑scale exploration** (n_small = 2000) –  
   Build a diverse pool of analytic families (uniform, triangular,
   Gaussian, power‑law, Hann, arcsine, exponential, beta) and many
   random Dirichlet draws.  Keep the 10 best seeds according to the
   exact surrogate `2·n·max(conv)`.

2. **Heavy mirror‑descent** on the probability simplex –  
   For each of the best seeds run exponentiated‑gradient (mirror
   descent) on the non‑smooth objective (sub‑gradient of the active
   convolution entry).  The update stays in log‑probability space.
   The cheap iteration (FFT‑based convolution) allows tens of
   millions of updates in the allocated time, yielding a very good
   candidate on the small problem.

3. **Upsample** the best small‑scale vector to the target size
   `n_target = 4000` by linear interpolation (preserves shape) and
   renormalise to a probability vector.

4. **Smooth‑surrogate polishing** –  
   Optimise the smooth log‑sum‑exp surrogate of the max‑convolution
   with L‑BFGS‑B (in logit space).  A schedule of increasing
   temperature `β` drives the surrogate towards the true max.
   The number of L‑BFGS iterations is chosen adaptively from the
   remaining budget.

5. **Optional simulated‑annealing** –  
   A final short SA walk in logit space helps to escape any
   remaining shallow local minima.

6. **Return the best vector** (global variable `height_sequence_1`
   for reuse) and report the true `evaluate_sequence` value.

The whole pipeline respects the 1000 s wall‑time limit, works
without filesystem/network access and only uses NumPy + SciPy
(if available).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS‑B polishing)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised probability vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, lam: float) -> np.ndarray:
    """Geometric law truncated to length n, normalised."""
    if lam <= 0.0:
        return _uniform(n)
    # a_i ∝ lam**i
    i = np.arange(n, dtype=np.float64)
    a = lam ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta pdf on {0,…,n‑1}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _norm_seq(v: np.ndarray) -> np.ndarray:
    """Project onto probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Exact surrogate for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Returns 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# L‑BFGS polish of the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 1500) -> np.ndarray:
    """Refine a probability vector with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:                               # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                # weights, sum = 1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # derivative of max_c term + smooth term

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                         # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (logit space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper: up‑sample a probability vector by linear interpolation
# ----------------------------------------------------------------------
def _upsample(seq_small: np.ndarray, size_large: int) -> np.ndarray:
    """Interpolate a small probability vector to a larger size."""
    x_small = np.arange(len(seq_small), dtype=np.float64)
    x_large = np.linspace(0, len(seq_small) - 1, size_large)
    seq_large = np.interp(x_large, x_small, seq_small)
    seq_large = np.maximum(seq_large, 0.0)
    total = seq_large.sum()
    if total <= 0.0:
        return np.full(size_large, 1.0 / size_large, dtype=np.float64)
    return seq_large / total

# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation metric.  The routine performs a fast small‑scale
    mirror‑descent search, up‑samples the best candidate, and then
    refines it with L‑BFGS (smooth surrogate) and a short simulated‑
    annealing walk.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin of a couple of seconds for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem sizes
    # ------------------------------------------------------------------
    n_small = 2000          # fast exploratory grid
    n_target = 4000         # final size (used for reporting)

    # ------------------------------------------------------------------
    # 1. Build a rich seed pool for the small problem
    # ------------------------------------------------------------------
    seeds_small = []

    # Analytic families
    seeds_small.append(_uniform(n_small))
    seeds_small.append(_triangular(n_small))
    seeds_small.append(_sin2(n_small))
    seeds_small.append(_hann_window(n_small))
    seeds_small.append(_arcsine(n_small))

    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds_small.append(_gaussian(n_small, sigma))

    for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds_small.append(_power(n_small, p))

    for lam in (0.3, 0.5, 0.7, 0.85, 0.95):
        seeds_small.append(_exponential(n_small, lam))

    for a, b in ((0.5, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)):
        seeds_small.append(_beta_shape(n_small, a, b))

    # Random Dirichlet draws with a wide range of concentration parameters
    conc_params = np.logspace(-2, 1, 12)          # 0.01 … 10
    for conc in conc_params:
        for _ in range(3):
            seeds_small.append(rng.dirichlet(np.full(n_small, conc)))

    # Include the previous best if present (down‑sample it)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size >= 2 * n_small:
                # down‑sample by averaging consecutive pairs
                arr_small = arr.reshape(-1, arr.size // n_small).mean(axis=1)
                seeds_small.append(_norm_seq(arr_small))
        except Exception:
            pass

    # Normalise everything onto the probability simplex
    seeds_small = [_norm_seq(np.asarray(v, dtype=np.float64)) for v in seeds_small]

    # ------------------------------------------------------------------
    # 2. Quick ranking of seeds (surrogate on the small problem)
    # ------------------------------------------------------------------
    best_small_seq = None
    best_small_val = np.inf
    for seq in seeds_small:
        val = _objective_surrogate(seq)
        if val < best_small_val:
            best_small_val = val
            best_small_seq = seq.copy()
    print(f"\n[Init] best surrogate among {len(seeds_small)} seeds (n={n_small}): {best_small_val:.6f}")

    # Keep the 10 best seeds for the intensive mirror‑descent phase
    seeds_small_sorted = sorted(seeds_small,
                                key=lambda s: _objective_surrogate(s))
    top_seeds = seeds_small_sorted[:10]

    # ------------------------------------------------------------------
    # 3. Mirror‑descent (exponentiated‑gradient) on the small problem
    # ------------------------------------------------------------------
    # Allocate ~60 % of the total time for this phase
    stage_a_deadline = start_time + 0.60 * (deadline - start_time)

    eps = 1e-30
    for idx, seed_seq in enumerate(top_seeds, 1):
        cur_seq = seed_seq.copy()
        log_a = np.log(cur_seq + eps)

        it = 0
        # a reasonable cap – we break earlier if the global deadline is reached
        max_iters = 300_000
        while it < max_iters and time.time() < stage_a_deadline:
            it += 1
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))

            # sub‑gradient of the active max‑convolution entry
            g = np.zeros_like(cur_seq)
            i_start = max(0, max_idx - (n_small - 1))
            i_end = min(n_small - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            g[idx_i] = 2.0 * cur_seq[rev_idx]

            # step‑size schedule (≈1/√t)
            eta = 0.5 / math.sqrt(it)
            log_a -= eta * g
            cur_seq = _softmax(log_a)

            if it % 50_000 == 0:
                cur_val = _objective_surrogate(cur_seq)
                if cur_val < best_small_val:
                    best_small_val = cur_val
                    best_small_seq = cur_seq.copy()
                    print(f"[Stage A] seed {idx:<2d} it={it:,} new best (n={n_small}) {best_small_val:.6f}")

    print(f"[Stage A] finished (best small surrogate = {best_small_val:.6f})")

    # ------------------------------------------------------------------
    # 4. Upsample to the target size and normalise
    # ------------------------------------------------------------------
    seq_target = _upsample(best_small_seq, n_target)

    # ------------------------------------------------------------------
    # 5. L‑BFGS polishing on the target problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:   # keep a little head‑room for SA & final eval
        remaining = deadline - time.time()
        # Roughly split the remaining budget: half for L‑BFGS, half for SA
        lbfgs_budget = remaining * 0.5
        lbfgs_deadline = time.time() + lbfgs_budget

        # adapt the per‑β iteration budget to the remaining time
        max_iter_per_beta = max(1500, int(min(4000, lbfgs_budget * 25)))
        print("\n--- L‑BFGS polishing (target n) ---")
        seq_polished = _lbfgs_optimize(seq_target,
                                        n_target,

                                        beta_schedule=[1, 5, 20, 80, 200,
                                                       500, 1_000, 2_000,
                                                       5_000, 20_000],
                                        max_iter_per_beta=max_iter_per_beta)
        val_polished = _objective_surrogate(seq_polished)
        print(f"[L‑BFGS] polished surrogate = {val_polished:.6f}")
        if val_polished < best_small_val:
            best_small_val = val_polished
            seq_target = seq_polished.copy()
            print(f"[Update] new global best (after L‑BFGS) = {best_small_val:.6f}")
        else:
            seq_target = seq_target  # keep the previous best

    # 6. Short simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_sa_iters = int(min(400_000, remaining * 2000))
        max_sa_iters = max(200_000, max_sa_iters)

        seq_sa, val_sa = _simulated_annealing_opt(seq_target,

        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = seq_target.tolist()
    return seq_target.tolist()
