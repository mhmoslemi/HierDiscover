"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimises
the evaluation function

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is deliberately different from the previous version
and consists of the following stages (all respect a global time budget,
default 1000 s):

1. **Rich seed generation** – analytic windows, many Dirichlet / exponential
   draws, low‑frequency cosine curves and a few *structured* permutations
   (alternating‑ends, high‑low interleaving).  All seeds are ℓ₁‑normalised.

2. **Fast surrogate screening** – for each seed we compute the cheap
   surrogate `2·n·max(conv)` using an FFT based convolution and keep the
   best K (K≈12).

3. **Pairwise mixing** – the K best seeds are mixed pair‑wise.  For each
   pair a ternary search on the mixing coefficient α∈[0,1] is performed
   to find the convex combination with the lowest surrogate value.

4. **Continuous refinement** – the top few (≤4) candidates are improved
   by (a) exponentiated‑gradient (mirror‑descent) on the *exact* max‑convolution
   and (b) a continuation L‑BFGS optimisation of a log‑sum‑exp smoothed
   max (the gradient is obtained analytically).  Both steps are run for as
   long as time permits.

5. **Discrete local search** – starting from the best refined vector we
   perform a simple stochastic swap‑descent: two random indices are swapped,
   the surrogate is re‑evaluated and the swap is kept only if it improves
   the surrogate.  This inexpensive hill‑climber often finds a slightly
   better ordering of the same multiset of heights.

6. **Final polishing** – an additional long EG pass followed by another
   L‑BFGS smoothing pass.

7. **True evaluation** – the final sequence is handed to `evaluate_sequence`
   and stored in the global variable ``height_sequence_1`` for the next
   run.

The whole pipeline adapts to the remaining time budget at every stage,
so it always returns a valid sequence before the 1000 s deadline.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0

         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

# Structured permutations of a multiset of values

def _alternating_ends(arr: np.ndarray) -> np.ndarray:
    """
    Place the sorted values alternatingly at the left‑most and right‑most
    positions (largest at the left, second largest at the right, …).
    This spreads big entries far apart.
    """
    n = arr.shape[0]
    order = np.argsort(arr)[::-1]                # descending indices
    out = np.empty_like(arr)
    left, right = 0, n - 1
    toggle = True
    for idx in order:
        if toggle:
            out[left] = arr[idx]
            left += 1
        else:
            out[right] = arr[idx]
            right -= 1
        toggle = not toggle
    out /= out.sum()
    return out


def _high_low_interleave(arr: np.ndarray) -> np.ndarray:
    """
    Interleave high and low values: high, low, high, low, …
    """
    n = arr.shape[0]
    sorted_desc = np.sort(arr)[::-1]
    sorted_asc = np.sort(arr)
    out = np.empty_like(arr)
    i_h, i_l = 0, 0
    for pos in range(n):
        if pos % 2 == 0:
            out[pos] = sorted_desc[i_h]
            i_h += 1
        else:
            out[pos] = sorted_asc[i_l]
            i_l += 1
    out /= out.sum()
    return out


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Gaussian noise, non‑negative projection, re‑normalisation."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best visited sequence and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (see comment in original code)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space, guaranteeing the simplex constraint.
    """
    if _opt is None:
        # SciPy not available – fall back to the input vector unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1_000_000.0,
                         5_000_000.0, 20_000_000.0]

    v = np.log(start_vec + 1e-30)   # work in log‑softmax space

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Pairwise mixing optimisation (simple ternary search)
# ----------------------------------------------------------------------
def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    """Return (mixed_seq, surrogate) for the best convex combination
    of a and b found by a ternary‑search on α∈[0,1].
    """
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None

    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0

        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)

        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)

        if val1 < val2:
            hi = m2
            if best_val is None or val1 < best_val:
                best_val = val1
                best_seq = seq1
        else:
            lo = m1
            if best_val is None or val2 < best_val:
                best_val = val2
                best_seq = seq2

    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective_surrogate(best_seq)
    else:
        mid = (lo + hi) / 2.0
        cand = _mix_shapes(a, b, mid)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq = cand
            best_val = cand_val

    return best_seq, best_val


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Discrete swap‑descent (local re‑ordering)
# ----------------------------------------------------------------------
def _swap_descent(start_seq: np.ndarray,
                  rng: np.random.Generator,
                  deadline: float,
                  max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climber that swaps two random entries and
    keeps the swap only if the surrogate improves.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    n = a.shape[0]

    for it in range(max_iters):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform the swap
        a[i], a[j] = a[j], a[i]
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found (list of floats)
    and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size (fixed – 4000 gave good trade‑off)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Diverse seed generation (analytic + stochastic + structured)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic windows & families (coarse grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 14):
        seeds.append(_gaussian(n, sigma))
    for sigma in (0.10, 0.12):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))
    for p in np.linspace(0.0, 2.0, 13):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))
    for alpha in np.linspace(0.1, 0.9, 7):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – wide range of concentrations
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random Exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 8):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Structured permutations of random Dirichlet draws
    for conc in (0.05, 0.1, 0.2):
        raw = _random_dirichlet(n, conc, rng)
        seeds.append(_alternating_ends(raw))
        seeds.append(_high_low_interleave(raw))

    # Include the previous best (if available) and a few perturbations
    if prev_arr is not None and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(5):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"\nTop {top_k} surrogate values among seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Pairwise mixing of the best seeds
    # ------------------------------------------------------------------
    print("\n--- Pairwise mixing of top seeds ---")
    mixture_candidates = []
    mix_deadline = deadline - 2.0

    for i in range(len(top_seeds)):
        for j in range(i + 1, len(top_seeds)):
            if time.time() > mix_deadline:
                break
            seq_mix, val_mix = _best_mixed_pair(top_seeds[i],
                                                top_seeds[j],
                                                mix_deadline,
                                                steps=35)
            mixture_candidates.append((seq_mix, val_mix))

    all_candidates = evaluated[:top_k] + mixture_candidates
    all_candidates.sort(key=lambda x: x[1])
    best_seq = all_candidates[0][0].copy()
    best_val = all_candidates[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (EG + L‑BFGS) on a few promising candidates
    # ------------------------------------------------------------------
    refined_candidates = []
    for idx, (seed_vec, _) in enumerate(all_candidates[:4]):
        if time.time() >= deadline:
            break
        # allocate a fraction of the remaining budget to this refinement
        remaining = deadline - time.time()
        # each refinement gets roughly equal share
        sub_deadline = time.time() + remaining * 0.45 / (4 - idx)

        print(f"\n[Refine #{idx+1}] surrogate start = {_objective_surrogate(seed_vec):.6f}")

        # (a) short EG refinement
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                          n,
                                          sub_deadline,
                                          max_iters=250_000,
                                          eta0=6.0)

        # (b) L‑BFGS smoothing (if time permits)
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1000.0,
                                                              2000.0, 5000.0,
                                                              20000.0, 50000.0,
                                                              100000.0, 200000.0,
                                                              500000.0, 1_000_000.0,
                                                              5_000_000.0],
                                               max_iter_per_beta=3000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs = seq_eg
            val_lbfgs = val_eg

        # keep the best among raw, EG, L‑BFGS
        candidates = [(seed_vec, _objective_surrogate(seed_vec)),
                      (seq_eg, val_eg),
                      (seq_lbfgs, val_lbfgs)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")

        refined_candidates.append((local_best_seq, local_best_val))

        if local_best_val < best_val:
            best_seq = local_best_seq
            best_val = local_best_val
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Stochastic swap‑descent on the current best vector
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Swap‑descent local re‑ordering ---")
        seq_swapped, val_swapped = _swap_descent(best_seq,
                                                rng,
                                                deadline,
                                                max_iters=8000)
        if val_swapped < best_val:
            best_seq = seq_swapped
            best_val = val_swapped
            print(f"[Swap‑descent] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final long EG polishing (full remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=5.0)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final L‑BFGS smoothing (to possibly lower the max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing ---")
        seq_final_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                                n,
                                                deadline,
                                                beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                               200.0, 500.0, 1000.0,
                                                               2000.0, 5000.0,
                                                               20000.0, 50000.0,
                                                               100000.0, 200000.0,
                                                               500000.0, 1_000_000.0,
                                                               5_000_000.0, 20_000_000.0],
                                                max_iter_per_beta=4000)
        val_final_lbfgs = _objective_surrogate(seq_final_lbfgs)
        if val_final_lbfgs < best_val:
            best_seq = seq_final_lbfgs
            best_val = val_final_lbfgs
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
