"""
Hybrid optimisation for the step‑function attack:

* 1️⃣  Build a very diverse pool of ℓ₁‑normalised seeds – analytic families
    (uniform, triangular, windows, Gaussians, …) together with a large set
    of random‑parameter instances (random γ for the sine‑window, random σ
    for Gaussians, random powers, random β‑values for the Kaiser‑window,
    random (α,β) for the beta‑shape, random block‑uniform, …) and a few
    Dirichlet draws.

* 2️⃣  Keep the best few seeds (by the exact surrogate
    `2·n·max(conv(seq,seq))`) as starting points.

* 3️⃣  Perform a long **exponentiated‑gradient** (mirror‑descent with KL
    geometry) on the *true* max‑convolution.  The update is
        a_i ← a_i·exp(‑η·g_i) ,   g = sub‑gradient of the active lag,
    followed by renormalisation.  The learning‑rate η=1/√t is used and
    the loop runs for as many iterations as the remaining budget permits
    (up to ≈2 M).  The best vector seen during the run is retained.

* 4️⃣  A final **simulated‑annealing fine‑tuner** moves a small amount of
    mass from one position to another (preserving the ℓ₁‑norm).  Swaps that
    improve the objective are always accepted; worsening moves are accepted
    with Metropolis probability using a linearly cooling temperature.
    This stage performs a few‑hundred‑thousand cheap moves and often
    squeezes the remaining 10⁻⁶‑level improvement.

* 5️⃣  The best vector is stored in the global variable ``height_sequence_1``
    and returned as a plain list of floats – ready for ``evaluate_sequence``.

All steps respect a global 1000 s budget and constantly check the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities – same as the reference implementation
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n·log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate `2·n·max(conv(seq,seq))` (ℓ₁‑normalised input)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact‑max)
# ----------------------------------------------------------------------
def _exp_gradient_opt_long(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with multiplicative updates on the *exact* max‑convolution.
    Runs for ``max_iters`` iterations or until ``deadline`` is reached.
    Returns the best sequence and the best surrogate value seen.
    """
    a = np.maximum(start_seq, 1e-30)
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a)

    # For stability we recompute `a` from the log‑space only once per iteration.
    # The update rule is a_i ← a_i·exp(‑η·g_i) with η = 1/√t.
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = kmax - idx
        g[idx] = 2.0 * a[rev]

        eta = 1.0 / math.sqrt(it)          # a slightly larger step than 0.5/√t
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            # numerical disaster – restart from the best known point
            a = best_seq.copy()
            s = a.sum()
        a = a / s

        cur_val = _objective(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()

        if it % 200_000 == 0:
            print(f"[ExpGrad‑long] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (mass‑transfer moves)
# ----------------------------------------------------------------------
def _anneal_fine_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int) -> tuple[np.ndarray, float]:
    """
    Randomly transfers a small amount of mass from one position to another.
    Accepts worsening moves with a temperature that linearly cools to 0.
    Returns the best sequence encountered.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T0 = cur_val * 0.02 if cur_val > 0 else 1.0   # modest initial temperature

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] <= 0.0:
            continue

        # transfer a random *fraction* of the mass at i to j
        delta = cur_seq[i] * rng.random() * 0.30   # up to 30 % of a_i
        cur_seq[i] -= delta
        cur_seq[j] += delta

        new_val = _objective(cur_seq)
        delta_obj = new_val - cur_val
        T = max(T0 * (1.0 - it / max_swaps), 1e-12)
        accept = (delta_obj < 0.0) or (rng.random() < math.exp(-delta_obj / T))

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert the move
            cur_seq[i] += delta
            cur_seq[j] -= delta

        if it % 100_000 == 0:
            print(f"[AnnealFine] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the global time budget,
    builds a huge set of diverse initial vectors, refines the best one
    with a long exponentiated‑gradient run and finishes with a tiny
    simulated‑annealing fine‑tuning phase.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG / timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for bookkeeping

    # ------------------------------------------------------------------
    # 1. Problem size (the benchmark uses n = 4000)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Seed generation (reuse previous solution if present)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            tmp = np.asarray(previous, dtype=np.float64)
            if tmp.size == n and tmp.sum() > 0:
                prev_arr = tmp
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 3. Quick evaluation of all seeds – keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Initialise optimisation from the absolute best seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long exponentiated‑gradient optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # aim for roughly 1 – 2 million iterations (adjusted to remaining time)
        max_iters = int(min(2_000_000, remaining * 2.0))
        max_iters = max(400_000, max_iters)          # enforce a meaningful budget
        print(f"\n--- Exponentiated‑gradient (max_iters={max_iters}) ---")
        global_best_seq, global_best_val = _exp_gradient_opt_long(
            global_best_seq, n, deadline, max_iters=max_iters)
        print(f"[ExpGrad] final best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Fine‑grained simulated‑annealing (mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(800_000, remaining * 5.0))
        max_swaps = max(200_000, max_swaps)
        print("\n--- Simulated‑annealing fine‑tuner ---")
        global_best_seq, global_best_val = _anneal_fine_opt(
            global_best_seq, n, deadline, max_swaps=max_swaps)
        print(f"[AnnealFine] final best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation with the provided ``evaluate_sequence`` function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"Warning: evaluate_sequence raised an exception: {exc}")
        true_val = None

    # Store for possible reuse in a subsequent call
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
