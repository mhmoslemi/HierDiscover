"""
Propose a sequence of non‑negative heights that (hopefully) beats the current
best bound ~1.5085 for the evaluation function

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a hybrid of three ideas:

1. **Initialisation** – start from the previously best sequence (if any) or a
   uniform distribution, all ℓ₁‑normalised to sum = 1.

2. **Simple Frank‑Wolfe style step** – given the current f,
   compute M = max(conv(f,f)).  Solve a tiny linear programme

        maximise   Σ b_j
        subject to  b ≥ 0,  b ≤ 1,
                    (f * b)[k] ≤ M   for the (few) indices k where f*f attains M

   The LP pushes mass into positions that do not increase the convolution
   peak.  The solution b is ℓ₁‑normalised and mixed with f using a 1‑D
   ternary search (convex combination) to obtain the best reduction of the
   surrogate 2·n·max(conv).

3. **Refinement** – a short exponentiated‑gradient (mirror‑descent) pass on the
   exact max‑convolution followed by an L‑BFGS continuation on a smooth
   log‑sum‑exp surrogate.  The whole loop is repeated a few times (or until the
   time budget expires).  Finally a longer EG polishing and a last L‑BFGS
   smoothing are performed.

The LP is tiny because only the *active* convolution indices (those attaining
the current maximum) are kept as constraints.  To keep the LP bounded we also
impose a trivial upper bound `b ≤ 1`.  The whole search respects the global
budget (default 1000 s).  The best sequence found is stored in the global
variable ``height_sequence_1`` for the next run.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on exact max‑convolution.
    Returns best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    try:
        import scipy.optimize as _optlb
    except Exception:  # pragma: no cover
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp               # weights for each entry of conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _optlb.minimize(fun=fun,
                                  x0=v,
                                  method='L-BFGS-B',
                                  jac=True,
                                  callback=callback,
                                  options={'maxiter': max_iter_per_beta,
                                           'disp': False,
                                           'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Pairwise mixing optimisation (simple ternary search)
# ----------------------------------------------------------------------
def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    """Return (mixed_seq, surrogate) for the best convex combination
    of a and b found by a ternary‑search on α∈[0,1].
    """
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None

    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0

        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)

        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)

        if val1 < val2:
            hi = m2
            if best_val is None or val1 < best_val:
                best_val = val1
                best_seq = seq1
        else:
            lo = m1
            if best_val is None or val2 < best_val:
                best_val = val2
                best_seq = seq2

    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective_surrogate(best_seq)
    else:
        mid = (lo + hi) / 2.0
        cand = _mix_shapes(a, b, mid)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq = cand
            best_val = cand_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small linear‑programming step (maximise ℓ₁‑norm of b)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, deadline: float):
    """
    Solve  max Σ b_j  subject to 0 ≤ b_j ≤ 1,
    and (f * b)[k] ≤ M for all indices k where the current max M is attained
    (plus a few neighbouring peaks for robustness).
    Returns a feasible ℓ₁‑normalised vector b (or None on failure).
    """
    n = len(f)
    conv_f_f = _conv_fft(f, f)
    # active indices – those within a tiny tolerance of the current max
    tol = max(1e-12 * M, 1e-14)
    active = np.where(conv_f_f >= M - tol)[0]

    # make sure we have at least a handful of constraints
    if active.size < 5:
        active = np.unique(np.concatenate([active,
                                           np.argsort(conv_f_f)[-5:]]))

    rows = []
    for k in active:
        row = np.zeros(n, dtype=np.float64)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_range = np.arange(i_min, i_max + 1, dtype=int)
        j_range = k - i_range
        row[j_range] = f[i_range]          # (f * b)[k] = Σ_i f[i]·b[k−i]
        rows.append(row)

    A_ub = np.vstack(rows) if rows else np.empty((0, n))
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)

    # bounds: 0 ≤ b_j ≤ 1   (the upper bound makes the LP bounded even with few rows)
    bounds = [(0.0, 1.0)] * n

    # Linear objective: maximise Σ b_j  → minimise −Σ b_j
    c = -np.ones(n, dtype=np.float64)

    # Solve with SciPy's high‑performance HiGHS interior‑point solver
    try:
        res = _opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                           bounds=bounds, method='highs')
    except Exception as e:  # pragma: no cover
        print("[LP] linprog exception:", e)
        return None

    if not res.success:
        # LP could be infeasible or unbounded – we treat it as a failure
        print("[LP] linprog failed:", res.message)
        return None

    b = res.x
    s = b.sum()
    if s <= 0.0:
        return None
    return b / s   # ℓ₁‑normalise


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine follows the hybrid scheme described
    in the module docstring and stores the best sequence in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # Warm‑start from previous solution if it exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            f = np.asarray(prev, dtype=np.float64)
            if f.size != n:
                f = None
        except Exception:
            f = None
    else:
        f = None

    if f is None or np.sum(f) <= 0:
        f = _uniform(n)            # ℓ₁‑normalised uniform start
    else:
        f = np.maximum(f, 0.0)
        f = f / f.sum()             # ensure exact ℓ₁‑normalisation

    best_seq = f.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"Initial surrogate (ℓ₁=1) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Main Frank‑Wolfe‑like loop (LP + mixing + short EG)
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline and iteration < 12:
        iteration += 1
        # current maximal convolution value
        conv_ff = _conv_fft(f, f)
        M = float(np.max(conv_ff))

        # LP step: find a direction b that respects the current max
        lp_deadline = deadline - 0.5   # keep a little room for mixing
        b = _lp_step(f, M, lp_deadline)
        if b is None:
            print(f"[Iter {iteration}] LP step failed – aborting loop.")
            break

        # best convex combination of f and the new direction b
        mix_deadline = deadline - 0.5
        mixed_seq, mixed_val = _best_mixed_pair(f, b, mix_deadline, steps=30)

        if mixed_val < best_val:
            best_val = mixed_val
            best_seq = mixed_seq.copy()
            print(f"[Iter {iteration}] improved surrogate to {best_val:.6f}")

        # update f for the next iteration (keep the promising mixed point)
        f = mixed_seq

        # a brief exponentiated‑gradient polishing (≈30 s max)
        eg_deadline = min(deadline, time.time() + 30.0)
        if eg_deadline > time.time():
            seq_eg, val_eg = _exp_gradient_opt(f.copy(), n,
                                               eg_deadline,
                                               max_iters=150_000,
                                               eta0=5.0)
            if val_eg < best_val:
                best_val = val_eg
                best_seq = seq_eg.copy()
                f = seq_eg
                print(f"[Iter {iteration}] post‑EG improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Random‑restart hill‑climbing (use up remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # generate a fresh Dirichlet seed (low concentration → spiky)
        seed_vec = _random_dirichlet(n, conc=0.05, rng=rng)
        # short EG refinement
        eg_deadline = min(deadline, time.time() + 15.0)
        if eg_deadline <= time.time():
            break
        seq_eg, val_eg = _exp_gradient_opt(seed_vec, n,
                                           eg_deadline,
                                           max_iters=80_000,
                                           eta0=5.0)
        if val_eg < best_val:
            best_val = val_eg
            best_seq = seq_eg.copy()
            f = seq_eg
            print(f"[Random] new best surrogate {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final long EG polishing (use the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=5.0)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final L‑BFGS smoothing (still respecting the deadline)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing ---")
        seq_final_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                                n,
                                                deadline,
                                                beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                               200.0, 500.0, 1000.0,
                                                               2000.0, 5000.0,
                                                               20000.0, 50000.0,
                                                               100000.0, 200000.0,
                                                               500000.0, 1000000.0],
                                                max_iter_per_beta=2500)
        val_final_lbfgs = _objective_surrogate(seq_final_lbfgs)
        if val_final_lbfgs < best_val:
            best_seq = seq_final_lbfgs
            best_val = val_final_lbfgs
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (the true, security‑checked function)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
