# ----------------------------------------------------------------------
# FFT‑based full linear convolution (zero‑padded)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (∑v = 1)."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Exact surrogate (assumes ∑vec = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """Exact objective for a probability vector:
       2·n·max(conv(vec, vec))
    """
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return normalised vectors (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _double_power(n: int, p: float) -> np.ndarray:
    """Symmetric sum of two power laws – heavy at both ends, low centre."""
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return _normalize(a)
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return _normalize(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """|cos|‑shaped – large at ends, small in middle."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)


def _arcsine_like(n: int, gamma: float = 0.5) -> np.ndarray:
    """U‑shaped density a_i ∝ (i·(n‑i))^{‑γ} (γ=0.5 gives classic arcsine)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i * (n + 1 - i)) ** (-gamma)
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    """Two uniform blocks at the far left/right ends."""
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators – return normalised vectors (sum = 1)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet sample with all concentration parameters = conc."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, cut at ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random disjoint blocks with uniform mass on each."""
    nb = rng.integers(2, 6)
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for _ in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)


def _random_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Convex combination of 2‑3 random analytic families."""
    families = [
        lambda: _uniform(n),
        lambda: _triangular(n),
        lambda: _sine_window(n, rng.uniform(0.2, 3.0)),
        lambda: _gaussian(n, rng.uniform(0.05, 0.35)),
        lambda: _power(n, rng.uniform(0.1, 4.0)),
        lambda: _double_power(n, rng.uniform(0.1, 4.0)),
        lambda: _beta_shape(n,
                            10 ** rng.uniform(-2, 1),
                            10 ** rng.uniform(-2, 1)),
        lambda: _cosine_edge_window(n, rng.uniform(0.5, 4.0)),
        lambda: _arcsine_like(n, rng.uniform(0.2, 2.0)),
        lambda: _two_block_uniform(n,
                                   block_ratio=rng.uniform(0.05, 0.4),
                                   left_weight=rng.random())
    ]
    k = rng.integers(2, 5)                         # number of components
    chosen = rng.choice(len(families), size=k, replace=False)
    weights = _normalize(rng.dirichlet(np.ones(k)))
    mix = np.zeros(n, dtype=np.float64)
    for w, idx in zip(weights, chosen):
        mix += w * families[idx]()
    return _normalize(mix)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Optimise the smoothed max‑convolution objective using L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1000.0, 2000.0,
            5000.0, 10000.0, 20000.0, 50000.0,
            100000.0, 200000.0, 500000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)           # logits for soft‑max

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)         # probability vector (sum = 1)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = np.sum(exp_beta)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            w = exp_beta / sum_exp
            full = _conv_fft(rev_a, w)    # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-13})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:          # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Greedy hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer hill‑climb on the simplex."""
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # transfer a random *fraction* of mass from i to j
        eps_frac = rng.uniform(0.001, 0.05)          # 0.1 % … 5 %
        eps = cur[i] * eps_frac
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val

            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and np.sum(prev_arr) > 0:
                _push_candidate(_normalize(prev_arr))
        except Exception:
            pass

    # -------------------------------------------------------------
    # 8. Extract best seeds (ordered from best to worst)
    # -------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] retained {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # -------------------------------------------------------------
    # 9. Initialise global best
    # -------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # -------------------------------------------------------------
    # 10. Local refinements – continuation L‑BFGS on the top seeds
    # -------------------------------------------------------------
    max_refinements = min(5, len(seeds))
    for idx in range(max_refinements):
        if time.time() > deadline - 5.0:
            break
        seed_vec = seeds[idx]
        remaining = deadline - time.time()
        # reserve ~55 % of remaining time for each refinement (shared)
        refine_deadline = time.time() + max(5.0, remaining * 0.55 / max_refinements)

        try:
            refined = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             refine_deadline,
                                             beta_schedule=[
                                                 1.0, 5.0, 20.0, 80.0,
                                                 200.0, 500.0, 1000.0,
                                                 2000.0, 5000.0, 10000.0,
                                                 20000.0, 50000.0, 100000.0,
                                                 200000.0, 500000.0, 1_000_000.0
                                             ],
                                             max_iter_per_beta=8000)
        except RuntimeError:
            refined = seed_vec   # fallback when SciPy missing

        refined_val = _objective_surrogate(refined)
        if refined_val < global_best_val:
            global_best_seq = refined
            global_best_val = refined_val
            print(f"[Refine {idx+1}] improved to {global_best_val:.6f}")

    # -------------------------------------------------------------
    # 11. Final greedy hill‑climb using any leftover time
    # -------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final hill‑climb ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # -------------------------------------------------------------
    # 12. Evaluate with the official evaluator (exact objective)
    # -------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
