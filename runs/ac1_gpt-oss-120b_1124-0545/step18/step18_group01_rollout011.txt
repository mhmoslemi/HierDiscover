def _adam_opt(start_seq, n, deadline, beta_schedule, iters_per_beta=500, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(start_seq)

    it_total = 0
    for beta in beta_schedule:
        if time.time()>deadline:
            break
        beta_val = float(beta)
        for _ in range(iters_per_beta):
            if time.time()>deadline:
                break
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n-1:n-1+n]  # length n
            # gradient wrt v
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam updates
            it_total += 1
            m = beta1 * m + (1 - beta1) * grad_v
            v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** it_total)
            v2_hat = v2 / (1 - beta2 ** it_total)
            v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
            # evaluate surrogate
            a_cur = _softmax(v)
            cur_val = _objective_surrogate(a_cur)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a_cur.copy()
    return best_seq, best_val

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alt_pattern(n: int, ratio: float) -> np.ndarray:
    """alternating pattern; `ratio` fraction of the total mass on even indices."""
    even_cnt = (n + 1) // 2
    odd_cnt = n // 2
    w_even = ratio / even_cnt
    w_odd = (1.0 - ratio) / odd_cnt
    a = np.empty(n, dtype=np.float64)
    a[::2] = w_even
    a[1::2] = w_odd
    return a


def _random_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """m randomly selected positions with equal weight."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see original implementation)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              beta_schedule=None,
              iters_per_beta: int = 800,
              lr: float = 0.03,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8):
    """
    Adam optimisation on the log‑sum‑exp smoothed max objective.
    Operates in log‑softmax space (guaranteeing non‑negativity & ℓ₁‑norm 1).
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 10_000_000.0, 20_000_000.0
        ]

    theta = np.log(start_seq + 1e-30)               # parameters in ℝⁿ
    m = np.zeros_like(theta)
    v = np.zeros_like(theta)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(start_seq)

    it_total = 0

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(theta)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth approximation (log‑sum‑exp)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                     # weights ∈ Δ_{2n‑1}

            # gradient w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # the part that aligns with a

            # gradient w.r.t. theta (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_theta = a * (grad_a - dot)

            # Adam update
            it_total += 1
            m = beta1 * m + (1.0 - beta1) * grad_theta
            v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)

            m_hat = m / (1.0 - beta1 ** it_total)
            v_hat = v / (1.0 - beta2 ** it_total)

            theta -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # evaluate surrogate objective
            a_cur = _softmax(theta)
            cur_val = _objective_surrogate(a_cur)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing (random local perturbations)
# ----------------------------------------------------------------------
def _hill_climbing(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   max_attempts: int = 200_000,
                   sigma_start: float = 0.005,
                   sigma_decay: float = 0.9995):
    """
    Very lightweight random‑perturbation search.
    Returns the best sequence discovered before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    sigma = sigma_start
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s
        val = _objective_surrogate(ns)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns
        sigma *= sigma_decay
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # Problem size – a trade‑off between flexibility and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load a previous solution if present (helps continuity).
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Seed generation – a very broad and random pool.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic families – coarse grids
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, 1.0))

    # Parameter sweeps
    for gamma in np.linspace(0.2, 4.0, 25):
        seeds.append(_sine_window(n, gamma))
    for sigma_frac in np.linspace(0.04, 0.4, 25):
        seeds.append(_gaussian(n, sigma_frac))
    for p in np.linspace(0.0, 5.0, 25):
        seeds.append(_power(n, p))
    for rate in np.logspace(-5, -0.3, 20):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Beta‑shape families
    for _ in range(40):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet / exponential draws (wide range of concentrations)
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5, 6, 7):
        for _ in range(8):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Alternating patterns
    for r in (0.1, 0.25, 0.5, 0.75, 0.9):
        seeds.append(_alt_pattern(n, r))

    # Random supports (sparse vectors)
    for m in (30, 60, 90, 120, 150, 180, 220):
        seeds.append(_random_support(n, m, rng))

    # Mixtures of uniform and low‑freq curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Include perturbed copies of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(10):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds.
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best.
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start refinement on the top seeds.
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # Time budget for this seed – keep ~12 % for each remaining seed.
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.12 / max(1, (top_k - si))

        # ----- (a) medium‑length exponentiated‑gradient -----
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=400_000,
                                           eta0=6.0)

        # ----- (b) Adam smoothing (more aggressive β‑schedule) -----
        if time.time() < deadline:
            seq_adam, val_adam = _adam_opt(seq_eg,
                                           n,
                                           deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1_000.0,
                                                          2_000.0, 5_000.0, 20_000.0,
                                                          50_000.0, 100_000.0,
                                                          200_000.0, 500_000.0,
                                                          1_000_000.0, 2_000_000.0,
                                                          5_000_000.0, 1_000_000_0.0],
                                           iters_per_beta=1000,
                                           lr=0.025)
        else:
            seq_adam = seq_eg
            val_adam = val_eg

        # ----- (c) short hill‑climbing -----
        hill_deadline = deadline - 0.5
        if time.time() < hill_deadline:
            seq_hc, val_hc = _hill_climbing(seq_adam,
                                            n,
                                            hill_deadline,
                                            rng,
                                            max_attempts=180_000,
                                            sigma_start=0.004,
                                            sigma_decay=0.9996)
        else:
            seq_hc, val_hc = seq_adam, val_adam

        # Pick the best among all refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_adam, val_adam),
            (seq_hc, val_hc)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")
        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long EG polishing (exponentiated‑gradient).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function.
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
