def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A
```

"""
Optimisation of a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The code builds a good starting vector (either a cached result or a
hand‑crafted analytic seed), then applies a *linear‑programming* polishing
phase derived from the classic Frank–Wolfe scheme described in the prompt.
For a current vector `f` we solve

    maximise   Σ b_j
    subject to b_j ≥ 0  and  (f * b)_k ≤ M   (∀k)

where `M = max(conv(f,f))`.  The LP tries to increase the ℓ₁‑norm of a
vector that does not increase the worst convolution entry.  By mixing the
solution `b` back into `f` we obtain a strictly smaller `M`.  After a few
LP steps we finish with a short simulated‑annealing walk in logit space.
The whole routine respects the 1000 s time budget and returns the best
vector found.

import time

# ----------------------------------------------------------------------
# Fast linear convolution (real FFT) for the surrogate objective
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))   (seq is already ℓ₁‑normalised)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seeds (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_bad(n: int) -> np.ndarray:
    """The original implementation (singular at the ends).  Kept for completeness."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Build the (f * b) ≤ M linear constraints as a sparse matrix
# (row k, column j) entry = f_{k‑j} if 0 ≤ k‑j < n  else 0.
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> _sp.csr_matrix:
    n = len(f)
    # For each offset d = 0 … n‑1 we place a diagonal of length n
    # filled with f[d].  Offsets correspond to row‑col = d.
    diagonals = [np.full(n, val, dtype=np.float64) for val in f]
    offsets = np.arange(n, dtype=np.int32)           # 0 … n‑1
    mat = _sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')
    return mat

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s != 0 else np.full_like(v, 1.0 / len(v))

    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)

    best = cur.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logit = logit + sigma * rng.normal(size=len(cur))
        proposal = _softmax(proposal_logit)
        val = _objective_surrogate(proposal)

        if val < cur_val or math.exp(-(val - cur_val) / T) > rng.random():
            logit = proposal_logit
            cur = proposal
            cur_val = val
            if cur_val < best_val:
                best = cur.copy()
                best_val = cur_val

        if it % 100_000 == 0:
            print(f"[SA] it={it:,} best surrogate={best_val:.6f}")

    return best

# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    The routine

      1. loads the previous best vector if available,
      2. builds a small pool of analytic seeds and keeps the best,
      3. applies a few LP‑based polishing steps (Frank–Wolfe style),
      4. finishes with a short simulated‑annealing walk.

    The whole computation respects the given time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand works well for the LP (quadratic in n)
    # ------------------------------------------------------------------
    n = 2000                     # chosen to keep LP matrix moderate

    # ------------------------------------------------------------------
    # 1. Starting point – previous best if it exists, otherwise a seed pool
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            cur_seq = np.asarray(prev, dtype=np.float64)
            if cur_seq.size == n and cur_seq.sum() > 0:
                cur_seq = np.maximum(cur_seq, 0.0)
                cur_seq = cur_seq / cur_seq.sum()
                print("[Init] loaded previous best")
                # continue with this vector
                # (if shape mismatches, fall back to seeds)
                use_prev = True
            else:
                use_prev = False
        except Exception:
            use_prev = False
    else:
        use_prev = False

    if not use_prev:
        # build a small pool of analytical candidates
        seeds = [
            _uniform(n),
            _triangular(n),
            _sin2(n),
            _hann_window(n),
            _arcsine_bad(n),
        ]
        for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
            seeds.append(_gaussian(n, sigma))
        for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
            seeds.append(_power(n, p))

        # random Dirichlet draws (wide range of concentration)
        conc_params = np.logspace(-2, 1, 12)          # 0.01 … 10
        for conc in conc_params:
            for _ in range(3):
                seeds.append(rng.dirichlet(np.full(n, conc)))

        # pick the best seed by the exact surrogate
        best_val = np.inf
        best_seq = None
        for s in seeds:
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            val = _objective_surrogate(s)
            if val < best_val:
                best_val = val
                best_seq = s.copy()
        cur_seq = best_seq
        print(f"[Init] best surrogate from seeds: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. LP‑based polishing (Frank–Wolfe style)
    # ------------------------------------------------------------------
    best_seq = cur_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # number of LP refinements – we keep it low to respect the time budget
    max_lp_iters = 6
    t = 0.5                     # mixing weight

    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        # current worst convolution entry
        conv = _conv_fft(cur_seq, cur_seq)
        M = float(np.max(conv))

        # build the linear matrix for (cur_seq * b) ≤ M
        A = _build_conv_matrix(cur_seq)

        # solve maximise Σ b_j  (i.e. minimise –Σ b_j)  subject to A b ≤ M, b ≥ 0
        c = -np.ones(n, dtype=np.float64)               # minimise –sum(b)
        bounds = [(0, None)] * n
        # a tiny tolerance on the right‑hand side helps the solver
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)

        res = _opt.linprog(c, A_ub=A, b_ub=b_ub,
                           bounds=bounds, method='highs',
                           options={'presolve': True, 'time_limit': max(deadline - time.time(), 1.0)})

        if not res.success:
            print(f"[LP] iteration {lp_it+1} failed: {res.message}")
            break

        b_opt = res.x
        b_opt_sum = b_opt.sum()
        if b_opt_sum <= 0:
            break

        # normalise the LP solution to stay on the probability simplex
        b_norm = b_opt / b_opt_sum

        # convex combination (mirror‑descent style)
        cur_seq = (1.0 - t) * cur_seq + t * b_norm
        cur_seq = np.maximum(cur_seq, 0.0)
        cur_seq /= cur_seq.sum()

        cur_val = _objective_surrogate(cur_seq)
        print(f"[LP] it={lp_it+1:2d} surrogate={cur_val:.6f}")

        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    # ------------------------------------------------------------------
    # 3. Short simulated‑annealing walk (logit space)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa_iters = int(min(400_000, remaining * 1500))
        max_sa_iters = max(150_000, max_sa_iters)
        print(f"\n--- Simulated‑Annealing (≈{max_sa_iters:,} iters) ---")
        seq_sa = _simulated_annealing(best_seq, deadline, max_iters=max_sa_iters)
        val_sa = _objective_surrogate(seq_sa)
        print(f"[SA] final surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()

    # ------------------------------------------------------------------
    # 4. Final evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                         # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
