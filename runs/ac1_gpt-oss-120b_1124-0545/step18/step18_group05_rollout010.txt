def improve_via_lp(a, max_iters, time_deadline):

    n = len(a); M = np.max(conv(a,a))

    for iter in range(max_iters):
        # Build A matrix for inequalities conv(a,b) <= M.

        L = 2*n-1

        A = np.empty((L, n), dtype=np.float64)

        for k in range(L):
            # compute starting index for b columns
            i_start = max(0, k - (n-1))
            i_end = min(k, n-1)
            # length = i_end - i_start + 1
            # For columns i_start..i_end, A[k,i] = a[k - i]
            A[k, i_start:i_end+1] = a[k - np.arange(i_start, i_end+1)]

        # Add RHS: M repeated L

        # Use scipy.optimize.linprog

        # Objective: maximize sum(b) => minimize -sum(b)
        c = -np.ones(n)

        bounds = [(0, None) for _ in range(n)]

        res = _opt.linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=bounds, method='highs', options={'disp': False, 'presolve': True})  # need minimize

        if not res.success:
            print('LP_failed', iter)
            break

        b = res.x

        sum_b = b.sum()
        if sum_b <= a.sum():
            # no improvement probable
            break
        # scale to have same sum = a.sum()
        g = b * (a.sum() / sum_b)
        # convex combine
        a = (1 - t) * a + t * g
        # recompute M
        conv_a = _conv_fft(a, a)
        M = np.max(conv_a)

        val = 2 * n * M / (a.sum()**2)
        print(f"LP iter {iter+1} objective {val:.6f}")

        # stop if time near deadline
        if time.time() > time_deadline:
            break

    return a

def _lbfgs_optimize_with_schedule(start_seq, n, deadline, beta_schedule, max_iter_per_beta):

def _adam_optimize(start_seq, n, deadline, max_iters=200000):
    rng = np.random.default_rng()
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        # Use max (subgradient) rather than smooth.
        # Choose single argmax; if multiple, decrement or average.
        max_idx = int(np.argmax(conv))

        # compute subgradient g (same as earlier)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]  # subgradient

        # gradient wrt v
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)
        lr = 0.3 / (np.sqrt(t) + 1)  # decaying learning rate
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        if t % 5000 == 0:
            cur_val = _objective_surrogate(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()
                print(f"[Adam] it={t} new best {best_val:.6f}")

    # final best
    return best_seq, best_val

"""
Propose a non‑negative step‑function (probability vector) that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines three stages:

1. **Diverse seeding** – analytic families (uniform, triangular, Gaussian,
   power‑law, sinusoid, Hann, arcsine) and many Dirichlet draws.
   All candidates are ℓ₁‑normalised and the best few are selected by the exact
   surrogate 2·n·max(conv) (i.e. sum(a)=1).

2. **Continuation L‑BFGS** – a smooth log‑sum‑exp surrogate of the max‑convolution
   is minimised for a schedule of “inverse‑temperature’’ β values.  The optimiser
   works in log‑it space (soft‑max parametrisation) and respects the time budget
   via a callback that raises ``KeyboardInterrupt`` when the deadline is passed.
   Each seed is polished; the best result is retained.

3. **Adam fine‑tuning** – the final vector is refined with Adam on the true
   non‑smooth objective (sub‑gradient of the active max‑convolution entry) for
   the remaining time.

The final vector is stored in the global ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS optimisation)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.linalg as _linalg
except Exception:            # pragma: no cover
    _opt = None
    _linalg = None

# ----------------------------------------------------------------------
# Global deadline used by the L‑BFGS callback (avoids closures)
# ----------------------------------------------------------------------
_DEADLINE_GLOBAL: float | None = None


def _deadline_callback(xk):
    """Called by ``scipy.optimize.minimize`` – aborts when budget expires."""
    if _DEADLINE_GLOBAL is not None and time.time() > _DEADLINE_GLOBAL:
        raise KeyboardInterrupt()


# ----------------------------------------------------------------------
# Fundamental numeric utilities (FFT‑based convolution, softmax, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exact surrogate (the objective when sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth‐max surrogate used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj(v: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the β‑smoothed max‑convolution.
    Variables are logits v; a = softmax(v) is a probability vector.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)

    # shifted version for numerical stability
    shifted = conv - max_c
    exp_beta = np.exp(beta * shifted)
    sum_exp = exp_beta.sum()

    # smooth approximation of the max
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)

    # gradient w.r.t. a (Eq. from the docstring)
    w = exp_beta / sum_exp                     # soft weights, sum = 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # slice the relevant part

    # back‑prop through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# L‑BFGS continuation optimiser (smooth surrogate)
# ----------------------------------------------------------------------
def _lbfgs_optimize_with_schedule(start_seq: np.ndarray,
                                  n: int,
                                  deadline: float,
                                  beta_schedule,
                                  max_iter_per_beta: int):
    """Run L‑BFGS for a series of β values; keep the best result."""
    global _DEADLINE_GLOBAL
    _DEADLINE_GLOBAL = deadline

    v = np.log(start_seq + 1e-30)   # logits
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    if _opt is None:  # pragma: no cover
        return best_seq, best_val

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        try:
            res = _opt.minimize(fun=_smooth_obj,
                                x0=v,
                                args=(n, beta),
                                method='L-BFGS-B',
                                jac=True,
                                callback=_deadline_callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta}: {e}")
            break

        a = _softmax(v)
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.0e} new best surrogate {best_val:.6f}")

    _DEADLINE_GLOBAL = None
    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the *true* max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000):
    """Adam optimiser in logit space using the exact max‑convolution sub‑gradient."""
    rng = np.random.default_rng()
    v = np.log(start_seq + 1e-30)   # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active max‑convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Chain rule through softmax
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update (learning‑rate decays roughly as 1/√t)
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)

        lr = 0.3 / (math.sqrt(t) + 1.0)   # simple decay
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        if t % 5000 == 0:
            cur_val = _objective_surrogate(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()
                print(f"[Adam] it={t:,} new best surrogate {best_val:.6f}")

    return best_seq, best_val

    n = 2500

            pass

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    def _norm(v):

        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / len(v))
        return v / s

    seeds = [_norm(np.asarray(s, dtype=np.float64)) for s in seeds]

    # ------------------------------------------------------------------

        if val < best_val:

    # ------------------------------------------------------------------
    # 5. Final official evaluation (the true metric)
    # ------------------------------------------------------------------
    try:

        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    return best_seq.tolist()
