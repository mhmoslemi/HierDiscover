"""
Propose a non‑negative step‑function (list of heights) that minimises
evaluate_sequence.  The algorithm proceeds in three phases:

1. **Seed generation & quick screening** – a large, diverse pool of
   ℓ₁‑normalised sequences (analytic windows, random Dirichlet draws,
   low‑frequency curves, previous solution) is built and evaluated with a
   cheap surrogate 2·n·max(conv).  The best few seeds are kept.

2. **Smooth optimisation** – each promising seed is refined with a
   continuation L‑BFGS optimiser that smooths the max‑convolution using
   a log‑sum‑exp surrogate.  This yields a high‑quality starting point.

3. **Local search (simulated annealing)** – from the best L‑BFGS result
   we run a fast mass‑transfer simulated‑annealing routine that updates
   the convolution incrementally (O(n) slice operations) and accepts
   uphill moves with a temperature schedule.  A short exponentiated‑
   gradient polishing pass finishes the optimisation.

All work respects the 1000 s wall‑clock budget; the best sequence found
is stored in the global variable ``height_sequence_1`` and returned as a
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – required only for the L‑BFGS stage
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution (real FFT, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                     # should not happen
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for ℓ₁‑normalised vectors:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_bias(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution concentrating mass near the ends."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw, then ℓ₁‑normalise."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised probability vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6,
        ]

    # work in log‑probability space (stay on simplex via softmax)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # classic 1/√t schedule
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local search (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_steps: int,
                         max_move_frac: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Fast SA that moves a small amount of mass from i→j using an
    incremental convolution update.
    Returns (best_sequence, best_surrogate_value).
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    # Temperature schedule – exponential decay from 20 % of cur_max to almost 0
    T0 = cur_max * 0.20
    T_end = cur_max * 1e-6
    decay = (T_end / T0) ** (1.0 / max_steps) if max_steps > 0 else 1.0
    T = T0

    recompute_interval = max(5_000, n)      # periodic exact recompute

    for step in range(1, max_steps + 1):
        if time.time() > deadline:
            print(f"[SA] deadline reached after {step} steps.")
            break

        T *= decay

        # pick i proportional to its mass (larger entries move more often)
        i = rng.choice(n, p=a / a.sum())
        j = rng.integers(0, n)
        if i == j:
            continue

        max_delta = a[i]
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * max_move_frac
        if delta <= 0.0:
            continue

        # --- incremental update of convolution --------------------------------
        conv_new = conv.copy()
        # linear part (uses the old a)
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a

        # second‑order correction
        idx_2i = 2 * i
        if idx_2i < conv_new.size:
            conv_new[idx_2i] += delta * delta
        idx_2j = 2 * j
        if idx_2j < conv_new.size:
            conv_new[idx_2j] += delta * delta
        idx_ij = i + j
        if idx_ij < conv_new.size:
            conv_new[idx_ij] -= 2.0 * delta * delta

        # update the vector
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0:               # numerical safety
            a_new[i] = 0.0

        new_max = conv_new.max()

        # Metropolis acceptance
        if new_max < cur_max or rng.random() < math.exp(-(new_max - cur_max) / (T + 1e-20)):
            a = a_new
            conv = conv_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()

        # occasional exact recompute to curb drift
        if step % recompute_interval == 0:
            conv = _conv_fft(a, a)
            cur_max = conv.max()

    return best_a, 2.0 * n * best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence found (list of floats).
    """
    # ------------------------------------------------------------------
    # 0. RNG / timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size – a power of two makes FFT a bit faster
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 2. Build a diverse seed pool
    # ------------------------------------------------------------------
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 1.0, 2.0, 4.0, 8.0):
        seeds.append(_sine_window(n, gamma=g))
    for a in (0.05, 0.1, 0.3, 1.0, 2.0):
        for b in (0.05, 0.1, 0.3, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))
    for p in (0.25, 0.5, 0.75, 1.0):
        seeds.append(_edge_bias(n, p))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Previous best solution, if any
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    print(f"\n--- Seed generation (n={n}) ---")
    print(f"Collected {len(seeds)} seeds.")

    # ------------------------------------------------------------------
    # 3. Quick surrogate evaluation, keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth L‑BFGS refinement on the selected seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    if _opt is not None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6,
        ]

        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                print("[Time] deadline reached before finishing all seed optimisations.")
                break

            remaining = deadline - time.time()
            max_iter_per_beta = int(min(25000, remaining * 0.05))
            max_iter_per_beta = max(8000, max_iter_per_beta)

            print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter_per_beta={max_iter_per_beta}) ---")
            seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             deadline,
                                             beta_schedule=beta_schedule,
                                             max_iter_per_beta=max_iter_per_beta)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient polishing (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(800_000, remaining * 0.08))
        max_exp = max(400_000, max_exp)
        print(f"\n[ExpGrad] polishing with up to {max_exp:,} iterations")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing mass‑transfer local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Rough estimate: aim for a few hundred thousand moves per remaining minute
        max_steps = int(min(3_000_000, remaining * 3000))
        max_steps = max(300_000, max_steps)
        print(f"\n--- Simulated Annealing (up to {max_steps:,} moves) ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                              n,
                                              deadline,
                                              max_steps=max_steps,
                                              max_move_frac=0.2)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final short exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp2 = int(min(200_000, remaining * 0.04))
        if max_exp2 > 50_000:
            print(f"[ExpGrad‑final] polishing with {max_exp2:,} iterations")
            seq_final, val_final = _exp_gradient_opt(global_best_seq,
                                                     n,
                                                     deadline,
                                                     max_iters=max_exp2)
            if val_final < global_best_val:
                global_best_seq = seq_final.copy()
                global_best_val = val_final
                print(f"[ExpGrad‑final] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # ------------------------------------------------------------------
    # 9. Store globally for the next round and return
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
