def _exp_gradient_opt(start_seq, n, deadline, max_iters=300_000, eta0=5.0, eps=1e-12):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # identify active indices within tolerance
        act_idxs = np.where(conv >= max_val - eps)[0]
        # compute gradient as sum over active indices
        g = np.zeros_like(a)
        for k in act_idxs:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            idx_i = np.arange(i_start, i_end+1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        # if multiple active, average gradient
        if len(act_idxs) > 1:
            g = g / len(act_idxs)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _blockwise_constant(n, num_blocks, rng):
    """Create piecewise constant step function: randomly assign mass to each block (nonnegative)."""
    if num_blocks < 1: num_blocks = 1
    # generate block probabilities from Dirichlet
    probs = rng.dirichlet(np.ones(num_blocks))
    # assign each block equal length approx n/num_blocks; last block may have remainder
    block_lengths = [n // num_blocks] * num_blocks
    remainder = n - sum(block_lengths)
    for i in range(remainder):
        block_lengths[i] += 1
    # construct sequence
    a = np.empty(n, dtype=np.float64)
    pos = 0
    for bl, length in zip(probs, block_lengths):
        a[pos:pos+length] = bl / length
        pos += length
    # Normalize sum to 1
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = _uniform(n)
    return a

"""
Propose a step‑function (non‑negative heights) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The evaluation is homogeneous → we work with ℓ₁‑normalised vectors (sum = 1).
The algorithm consists of

1. **Rich seed generation** – analytic windows, Dirichlet / exponential draws,
   low‑frequency cosine curves, block‑wise constant “step” shapes and several
   perturbations of the previous best solution.

2. **Intensive mixing** – the best K seeds are mixed pair‑wise (ternary search)
   and also combined at random as three‑way convex combos.

3. **Refinement** – each promising candidate is polished with
   *exponentiated‑gradient* (mirror descent on the exact max‑convolution) and
   subsequently with a *smooth‑max* L‑BFGS optimisation (log‑sum‑exp surrogate)
   on an exponentially increasing β‑schedule.  The best of the three
   (raw, EG‑polished, L‑BFGS‑polished) is kept.

4. **Final polishing** – a long EG run followed by a high‑β L‑BFGS pass.

The whole pipeline respects a global time budget (default 1000 s) and stores
the best sequence in the global variable ``height_sequence_1`` for the next run.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # Modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blockwise_constant(n: int, num_blocks: int, rng: np.random.Generator) -> np.ndarray:
    """Piecewise‑constant step shape: random mass per block, uniform inside."""
    if num_blocks <= 0:
        num_blocks = 1
    # random masses for the blocks (Dirichlet → sum = 1)
    block_masses = rng.dirichlet(np.ones(num_blocks))
    # block sizes (as equal as possible)
    base_len = n // num_blocks
    lengths = np.full(num_blocks, base_len, dtype=int)
    remainder = n - base_len * num_blocks
    lengths[:remainder] += 1
    a = np.empty(n, dtype=np.float64)
    pos = 0
    for mass, L in zip(block_masses, lengths):
        if L > 0:
            a[pos:pos + L] = mass / float(L)
        pos += L
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0,
                     eps_active: float = 1e-12) -> tuple[np.ndarray, float]:
    """Mirror‑descent on exact max‑convolution (sub‑gradient = avg of
    all active indices). Returns best sequence seen and its surrogate value."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set (indices within eps of the max)
        active = np.where(conv >= max_val - eps_active)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if active.size > 1:
            g /= float(active.size)

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space guaranteeing the simplex.
    """
    if _opt is None:
        # SciPy not available – return input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 2e4, 5e4, 1e5, 2e5,
                         5e5, 1e6, 2e6, 5e6, 1e7]

    # work in log‑softmax space
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            # stable log‑sum‑exp of (beta * (conv - shift))
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (in the original space)
            w = exp_beta / sum_exp                           # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]           # restrict to length n

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Pairwise mixing optimisation (simple ternary search)
# ----------------------------------------------------------------------
def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30) -> tuple[np.ndarray, float]:
    """Return (mixed_seq, surrogate) for the best convex combination
    of a and b found by a ternary‑search on α∈[0,1]."""
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None

    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0

        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)

        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)

        if val1 < val2:
            hi = m2
            if best_val is None or val1 < best_val:
                best_val = val1
                best_seq = seq1
        else:
            lo = m1
            if best_val is None or val2 < best_val:
                best_val = val2
                best_seq = seq2

    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective_surrogate(best_seq)
    else:
        mid = (lo + hi) / 2.0
        cand = _mix_shapes(a, b, mid)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq = cand
            best_val = cand_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``. Returns the best sequence (list of floats) and
    stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size (fixed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and np.isfinite(arr).all():
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Broad seed generation (diverse families)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # include previous best if available
    if prev_vec is not None:
        seeds.append(prev_vec)

    # Analytic windows (a modest grid)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 8):
        seeds.append(_sine_window(n, gamma=gamma))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in np.linspace(1.0, 12.0, 6):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.22, 6):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 2.0, 6):
        seeds.append(_power(n, p))
    # Beta‑shape grid (moderate)
    alphas = np.linspace(0.2, 5.0, 5)
    betas = np.linspace(0.2, 5.0, 5)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))
    for alpha in np.linspace(0.1, 0.9, 5):
        seeds.append(_tukey_window(n, alpha))

    # Block‑wise constant shapes
    for nb in (5, 10, 20):
        seeds.append(_blockwise_constant(n, nb, rng))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Alternating Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # Perturb previous best (if any)
    if prev_vec is not None:
        for sigma in (0.001, 0.003, 0.005):
            for _ in range(3):
                seeds.append(_perturb(prev_vec, rng, sigma=sigma))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"\nTop {top_k} surrogate values among seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Pairwise mixing of top seeds + random three‑way mixtures
    # ------------------------------------------------------------------
    print("\n--- Mixing phase (pairwise + triple) ---")
    mixture_candidates = []

    mix_deadline = deadline - 2.0

    # pairwise ternary search
    for i in range(len(top_seeds)):
        for j in range(i + 1, len(top_seeds)):
            if time.time() > mix_deadline:
                break
            seq_mix, val_mix = _best_mixed_pair(top_seeds[i],
                                                top_seeds[j],
                                                mix_deadline,
                                                steps=25)
            mixture_candidates.append((seq_mix, val_mix))

    # random three‑way convex combos
    for _ in range(30):
        if time.time() > mix_deadline:
            break
        idxs = rng.choice(len(top_seeds), size=3, replace=False)
        weights = rng.dirichlet(np.ones(3))
        combo = (weights[0] * top_seeds[idxs[0]] +
                 weights[1] * top_seeds[idxs[1]] +
                 weights[2] * top_seeds[idxs[2]])
        combo = combo / combo.sum()
        mixture_candidates.append((combo, _objective_surrogate(combo)))

    # combine raw top seeds and mixtures
    all_candidates = evaluated[:top_k] + mixture_candidates
    all_candidates.sort(key=lambda x: x[1])
    best_seq = all_candidates[0][0].copy()
    best_val = all_candidates[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (EG + L‑BFGS) on a few promising candidates
    # ------------------------------------------------------------------
    refined_candidates = []
    # we will refine at most the top 5 candidates
    for idx, (seed_vec, _) in enumerate(all_candidates[:5]):
        if time.time() >= deadline:
            break
        # allocate a slice of remaining time (more for early candidates)
        remaining = deadline - time.time()
        share = (5 - idx) / sum(range(1, 6))   # decreasing share
        sub_deadline = time.time() + remaining * share

        print(f"\n[Refine #{idx+1}] starting surrogate = {_objective_surrogate(seed_vec):.6f}")

        # short EG refinement
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                          n,
                                          sub_deadline,
                                          max_iters=800_000,
                                          eta0=6.0)

        # L‑BFGS fine‑tuning (if time permits)
        if time.time() < sub_deadline:
            seq_lb = _lbfgs_smooth_optimize(seq_eg,
                                            n,
                                            sub_deadline,
                                            beta_schedule=[0.5, 1.0, 5.0, 20.0, 80.0,
                                                           200.0, 500.0, 1000.0,
                                                           2000.0, 5000.0, 2e4,
                                                           5e4, 1e5, 2e5, 5e5,
                                                           1e6, 2e6, 5e6, 1e7],
                                            max_iter_per_beta=2000)
            val_lb = _objective_surrogate(seq_lb)
        else:
            seq_lb = seq_eg
            val_lb = val_eg

        # pick the best among raw, EG and L‑BFGS
        local_best_seq, local_best_val = min(
            [(seed_vec, _objective_surrogate(seed_vec)),
             (seq_eg, val_eg),
             (seq_lb, val_lb)],
            key=lambda x: x[1])

        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")

        refined_candidates.append((local_best_seq, local_best_val))

        if local_best_val < best_val:
            best_seq = local_best_seq
            best_val = local_best_val
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long EG polishing (full remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=5.0)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final L‑BFGS smoothing with a very high β‑schedule
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing (high β) ---")
        seq_final_lb = _lbfgs_smooth_optimize(best_seq,
                                              n,
                                              deadline,
                                              beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                             200.0, 500.0, 1000.0,
                                                             2000.0, 5000.0,
                                                             2e4, 5e4, 1e5,
                                                             2e5, 5e5, 1e6,
                                                             2e6, 5e6, 1e7,
                                                             2e7, 5e7],
                                              max_iter_per_beta=2500)
        val_final_lb = _objective_surrogate(seq_final_lb)
        if val_final_lb < best_val:
            best_seq = seq_final_lb
            best_val = val_final_lb
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
