def adam_optimize(start_a, n, deadline, max_iters):
    v = np.log(start_a + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v) # second moment
    beta1=0.9, beta2=0.999, eps=1e-8
    best_a = start_a.copy()
    best_val = objective_exact(start_a)
    for t in range(1, max_iters+1):
        if time.time() > deadline: break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        # compute subgradient g_max (size n) using formula from _subgradient_maxconv
        # we need to compute g_max for this idx.
        # use i_range logic.
        g_max = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        i_range = np.arange(i_start, i_end+1)
        rev = max_idx - i_range
        g_max[i_range] = 2.0 * a[rev]
        # compute gradient for objective
        s = np.sum(a)
        obj = 2.0 * n * max_val / (s*s)
        # gradient w.r.t a
        grad_a = (2.0 * n / (s*s)) * g_max - (4.0 * n * max_val / (s**3)) * np.ones(n)
        # backprop to v
        grad_v = a * (grad_a - np.dot(grad_a, a))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v**2)
        m_hat = m / (1 - beta1**t)
        v_hat_hat = v_hat / (1 - beta2**t)
        lr = 0.1 # choose
        v = v - lr * m_hat / (np.sqrt(v_hat_hat) + eps)
        # Evaluate best checkpoint maybe every some iterations
        if t % 1000 == 0:
            cur_a = _softmax(v)
            cur_val = _objective_exact(cur_a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
    # final
    cur_a = _softmax(v)
    cur_val = _objective_exact(cur_a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = cur_a.copy()
    return best_a, best_val

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.
=======================================================================
The strategy combines three complementary phases:

1️⃣  **Rich seed pool** – a collection of analytic families (uniform,
    linear‑decrease, exponential, power‑law, logarithmic, sorted‑Dirichlet
    etc.), random Dirichlet draws and mixtures thereof.  All seeds are
    ℓ₁‑normalised.

2️⃣  **Adam on a soft‑max parametrisation** – each seed is used as a start
    for a mirror‑descent (exponentiated‑gradient) optimisation where the
    exact objective

            f(a) = 2·n·max(conv(a,a)) / (∑a)²

    is differentiable almost everywhere.  The gradient w.r.t. the
    logits‑vector `v` (where `a = softmax(v)`) is obtained analytically via
    the sub‑gradient of the active convolution lag.  An Adam update with
    bias‑correction provides fast, stable descent.  A small number of
    iterations (≈ 3 000) is performed for many seeds and the best result
    is kept.

3️⃣  **Simulated annealing fine‑tuning** – starting from the current global
    best we execute a stochastic mass‑transfer walk that occasionally
    accepts worsening moves according to a cooling schedule.  This helps
    escaping shallow local minima that the deterministic Adam step may
    miss.

The whole pipeline respects the 1000 s budget, prints progress and stores
the final sequence in the global variable ``height_sequence_1`` for warm‑starts.

The result is a list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decrease(n: int) -> np.ndarray:
    """a_i ∝ (n-i) (strictly decreasing)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, p>0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int) -> np.ndarray:
    """a_i ∝ 1/(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector and sort it decreasingly (more “head‑heavy”)."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]  # descending
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# exact objective & sub‑gradient helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of the active convolution entry `conv[idx]` w.r.t. `a`.
    """
    n = len(a)
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Adam optimizer on the exact objective (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3000,
    lr: float = 0.12,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam on the exact max‑convolution objective.
    """
    # initialise logits
    v = np.log(start_a + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)          # second moment estimate
    best_a = start_a.copy()
    best_val = _objective_exact(start_a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        a = _softmax(v)                     # ℓ₁‑normalised sequence
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        max_conv = conv[idx]

        # sub‑gradient of the active max‑convolution entry
        g_max = _subgradient_maxconv(a, idx)

        # gradient of the scalar objective w.r.t. a
        s = np.sum(a)
        grad_a = (2.0 * n / (s * s)) * g_max - (4.0 * n * max_conv / (s ** 3)) * np.ones_like(a)

        # back‑propagation through the soft‑max
        grad_v = a * (grad_a - np.dot(grad_a, a))

        # Adam update with bias correction
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # occasional checkpoint
        if t % 250 == 0:
            cur_a = _softmax(v)
            cur_val = _objective_exact(cur_a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
    # final evaluation
    cur_a = _softmax(v)
    cur_val = _objective_exact(cur_a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = cur_a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer) refinement
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    init_temp: float = 0.02,
    final_temp: float = 1e-6,
    step_frac: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a small fraction of mass from i → j.
    Accepts uphill moves with probability exp(-Δ / T).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    cur_val = best_val
    rng = np.random.default_rng(987654321)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear cooling schedule
        temp = init_temp * ((final_temp / init_temp) ** (it / max_iters))

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = _objective_exact(cand)
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-(cand_val - cur_val) / max(temp, 1e-12)):
                accept = True

        if accept:
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
    return best_a, best_val

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decrease(n))
    seeds.append(_logarithmic(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.0004))
    seeds.append(_power_law(n, 0.35))
    seeds.append(_power_law(n, 0.55))

    # power‑law grid (finer than the reference)
    for p in np.linspace(0.30, 0.80, 21):
        seeds.append(_power_law(n, float(p)))

    # random Dirichlet families (different concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(3):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures of existing seeds
    base = np.arange(len(seeds))
    for _ in range(70):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # warm‑start from previous run
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final sanitisation – ensure non‑negative & ℓ₁‑normalised
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

    # ------------------------------------------------------------------
    # Seed pool (including possible warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Evaluate all seeds, keep the elite
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # Sort seeds by objective for the Adam stage
    seeds.sort(key=_objective_exact)
    elite = seeds[:15]                     # a modest elite size
    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation from elite seeds
    # ------------------------------------------------------------------
    print("\n--- Phase 1: Adam optimisation of elite seeds ---")
    for idx, seed_vec in enumerate(elite, 1):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing Adam runs.")
            break
        iters = 3000
        seq_opt, val_opt = _adam_optimize(
            seed_vec,
            n,
            deadline,
            max_iters=iters,
            lr=0.12,
        )
        print(f"[Adam #{idx}] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Multiple Adam restarts with random Dirichlet starts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 10
        iters_per_restart = min(2500, int(remaining * 0.12))
        print("\n--- Phase 2: Adam restarts from random Dirichlet draws ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.01, 1.5)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_adam, val_adam = _adam_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                lr=0.12,
            )
            print(f"[Restart {r+1}/{restarts}] objective = {val_adam:.6f}")
            if val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Update] New global best after restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing fine‑tuning from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 0.30))
        if max_iters > 0:
            print("\n--- Phase 3: Simulated annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                init_temp=0.02,
                final_temp=1e-6,
                step_frac=0.012,
            )
            print(f"[SA] best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[Update] New global best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in future invocations
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
