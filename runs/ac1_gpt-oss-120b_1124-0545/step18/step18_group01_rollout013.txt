assistantfinal**Solution Overview**

The goal is to minimise  

\[
J(a)=\frac{2\,n\;\max_k (a*a)[k]}{(\sum_i a_i)^2},
\]

where `a` is a non‑negative step‑function (list of heights) and `*` denotes the
full linear convolution.
The evaluation routine is scale‑invariant, therefore we may enforce
the normalisation `∑a_i = 1` and simply minimise  

\[
\tilde J(a)=2\,n\;\max_k (a*a)[k].
\]

The programme below

1. **Generates a very large and diverse set of candidate shapes**  
   – analytic windows (`uniform`, `triangular`, `sine`, `gaussian`,
     `kaiser`, `chebwin`, `tukey`, exponential families, beta‑shapes, …)  
   – random Dirichlet / exponential draws, low‑frequency cosine curves, and
     many random mixtures.  
   All candidates are ℓ₁‑normalised.

2. **Selects the best few (default 10) according to the cheap surrogate**
   `2·n·max(conv)`.  This step is fast because the convolution is performed
   with a real FFT.

3. **Refines each elite candidate** by a two‑stage optimiser  
   * **Continuation L‑BFGS** on a smooth approximation of the max‑convolution
     (log‑sum‑exp with a growing temperature β).  The optimisation is carried
     out in the log‑softmax space, guaranteeing the constraints `a_i ≥ 0`,
     `∑a_i = 1`.  The schedule uses very large β (up to 10⁸) and a generous
     iteration budget (`max_iter_per_beta = 5 000`).  
   * **Exponentiated‑gradient (mirror descent)** on the exact max‑convolution,
     starting from the L‑BFGS output, for a final polishing pass.

4. **Keeps the best sequence found** (according to the true
   `evaluate_sequence` function) and stores it in the global variable
   `height_sequence_1` for possible reuse in later runs.

All steps respect the global deadline (`budget_s` seconds).  The programme
prints progress information, and finally returns a list of heights ready for
`evaluate_sequence`.

"""

evaluate_sequence().  The algorithm builds a huge pool of analytic and random
candidates, keeps the best few according to a fast surrogate (2·n·max(conv)),
and refines them with a continuation L‑BFGS optimiser (smooth max via
log‑sum‑exp) followed by an exponentiated‑gradient pass.  The whole search
honours a user‑provided time budget (default 1000 s) and stores the best
sequence in the global variable ``height_sequence_1``.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ i^{α‑1} (n‑i+1)^{β‑1},  i=1..n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _kaiser_win is None:
        return _uniform(n)
    w = _kaiser_win(n, beta)
    w = np.asarray(w, dtype=np.float64)
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


def _cheb_window(n: int, attenuation: float) -> np.ndarray:
    if _chebwin_win is None:
        return _uniform(n)
    w = _chebwin_win(n, at=attenuation)
    w = np.asarray(w, dtype=np.float64)
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0):
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5_000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The variables are kept in log‑softmax space, guaranteeing non‑negativity
    and unit ℓ₁‑norm.
    """
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 10_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp            # soft‑max weights
            rev_a = a[::-1]
            # gradient of the smoothed max w.r.t. a (size n)
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # map gradient to log‑softmax space
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-13})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple hill‑climbing (random local perturbations)
# ----------------------------------------------------------------------
def _hill_climbing(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   max_attempts: int = 200_000,
                   sigma_start: float = 0.003,
                   sigma_decay: float = 0.9997):
    """
    Lightweight random‑perturbation search.  Returns the best sequence
    discovered before deadline or attempt limit.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    sigma = sigma_start
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s
        val = _objective_surrogate(ns)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns
        sigma *= sigma_decay
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The routine respects the time budget and stores the best found sequence in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 offers a good trade‑off between expressiveness
    # and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Massive seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic “classical” families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_gaussian(n, sigma_frac=0.2))
    seeds.append(_tukey_window(n, alpha=0.5))

    # Wide parameter sweeps
    for gamma in np.linspace(0.2, 4.0, 30):
        seeds.append(_sine_window(n, gamma))

    for sigma_frac in np.linspace(0.05, 0.35, 25):
        seeds.append(_gaussian(n, sigma_frac))

    for rate in np.logspace(-5, -0.3, 30):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Kaiser & Chebyshev windows
    for beta in np.linspace(0.0, 20.0, 21):
        seeds.append(_kaiser_window(n, beta))

    for att in np.linspace(20, 200, 19):
        seeds.append(_cheb_window(n, att))

    # Tukey window sweep
    for alpha in np.linspace(0.1, 0.9, 17):
        seeds.append(_tukey_window(n, alpha))

    # Beta‑shape random sampling (very wide range)
    for _ in range(250):
        a = 10 ** rng.uniform(-2, 2)          # range ≈ 0.01 … 100
        b = 10 ** rng.uniform(-2, 2)
        seeds.append(_beta_shape(n, a, b))
        seeds.append(_beta_shape(n, b, a))    # swapped → opposite skew

    # Random Dirichlet / exponential draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(20):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(8):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform – exploring different blend ratios
    for mix_alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        curve = _lowfreq_curve(n, 3, rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Perturbed copies of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(8):
            seeds.append(_perturb(prev_arr, rng, sigma=0.0025))

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation of every seed
    # ------------------------------------------------------------------
    evaluated = []
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement of the elite candidates
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a portion of the remaining time to this seed
        remaining = deadline - time.time()
        # keep ~20 % of the remaining time for each future seed + final polish
        sub_deadline = time.time() + remaining * 0.20 / max(1, (top_k - si))

        # ----- (a) continuation L‑BFGS on smoothed max -----
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec,
                                           n,
                                           sub_deadline,
                                           beta_schedule=None,
                                           max_iter_per_beta=5_000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        # ----- (b) a medium‑length exponentiated‑gradient pass -----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                               n,
                                               sub_deadline,
                                               max_iters=600_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # ----- (c) short hill‑climbing for a quick local tweak -----
        hill_deadline = sub_deadline - 0.4   # keep a little safety margin
        if time.time() < hill_deadline:
            seq_hc, val_hc = _hill_climbing(seq_eg,
                                            n,
                                            hill_deadline,
                                            rng,
                                            max_attempts=120_000,
                                            sigma_start=0.004,
                                            sigma_decay=0.9996)
        else:
            seq_hc, val_hc = seq_eg, val_eg

        # pick the best among the three refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_lbfgs, val_lbfgs),
            (seq_eg, val_eg),
            (seq_hc, val_hc)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing (exponentiated‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the best sequence for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
