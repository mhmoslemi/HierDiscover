def _fourier_log_shape(n, K, amp_scale, rng):
    i = np.arange(n, dtype=np.float64)
    loga = np.zeros(n)
    for k in range(1, K+1):
        amp = rng.uniform(-amp_scale, amp_scale)
        phi = rng.uniform(0.0, 2.0*np.pi)
        loga += amp * np.cos(2.0*np.pi * k * i / n + phi)
    a = np.exp(loga)
    s = a.sum()
    return a / s

"""
Propose a step‑function that minimises ``evaluate_sequence``.
----------------------------------------------------------------
The optimiser works in five stages:

1. **Seed generation** – a large and diverse seed pool (uniform,
   triangular, Gaussian, beta, power‑law, random Dirichlet,
   low‑frequency cosine sums, random Fourier‑log shapes, sparse and
   mixed vectors).  All seeds are ℓ₁‑normalised.

2. **Continuation L‑BFGS** – each elite seed is refined with a
   log‑sum‑exp smooth surrogate of the max‑convolution.  The
   continuation walks a flexible β‑schedule (up to 1e9) while
   respecting the remaining wall‑time.

3. **Mirror‑descent (exponentiated‑gradient)** – a series of restarts
   from random Dirichlet points followed by a long run starting from
   the best vector found so far.  The exact sub‑gradient of the
   max‑convolution is used.

4. **Stochastic local search** – mass‑transfer hill‑climbing together
   with occasional Gaussian perturbations of the log‑probabilities.

5. **Final polishing** – a short, very high‑β L‑BFGS run.

The routine respects the 1000 s wall‑time limit, stores the best
result in the global variable ``height_sequence_1`` and returns it as
a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    # each pair contributes twice except the centre when i==rev
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of 2·n·max(conv) / (∑a)².
    Returns (obj, grad) – gradient w.r.t. logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised probabilities
    conv = _conv_fft(a, a)               # full linear convolution
    shift = np.max(conv)                 # numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t a
    w = exp_term / sum_exp               # softmax over lags (weights)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through the softmax that maps v → a
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1200,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            1e7, 5e7, 1e8, 5e8, 1e9,
        ]

    # initialise logits (unconstrained)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[EG] iteration {it:,} – new best {best_val:.6f}")

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic mass‑transfer local search (plus Gaussian perturbation)
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 80_000,
    step_frac: float = 0.010,
    rng: np.random.Generator | None = None,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass between coordinates.
    Occasionally adds a tiny Gaussian perturbation of the log‑vector."""
    if rng is None:
        rng = np.random.default_rng(12345)

    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    log_a = np.log(a + 1e-30)

    for it in range(iters):
        if time.time() > deadline:
            break

        # 1️⃣ mass transfer move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j and a[i] > 0.0:
            delta = step_frac * a[i]
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            val = _objective_exact(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                a = cand
                log_a = np.log(a + 1e-30)

        # 2️⃣ occasional log‑perturbation (≈5 % of iterations)
        if (it & 31) == 0:                     # roughly every 32 steps
            pert = rng.normal(scale=0.02, size=n)
            cand_log = log_a + pert
            cand = _softmax(cand_log)
            val = _objective_exact(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                a = cand
                log_a = cand_log

    return best_a, best_val

# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _fourier_log_shape(
    n: int,
    K: int,
    amp_scale: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """a_i ∝ exp( Σ_{k=1}^K c_k·cos(2πk·i/n + φ_k) ), ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    loga = np.zeros(n, dtype=np.float64)
    for k in range(1, K + 1):
        c = rng.uniform(-amp_scale, amp_scale)
        phi = rng.uniform(0.0, 2.0 * np.pi)
        loga += c * np.cos(2.0 * np.pi * k * i / n + phi)
    a = np.exp(loga)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Seed‑pool construction (augmented with Fourier‑log shapes)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diversified list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # 1️⃣ deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # 2️⃣ power‑law sweep (expanded)
    p_vals = np.linspace(0.3, 0.9, 13)          # more exponents
    shifts = [0.0, 5.0, 20.0, 50.0, 100.0, 200.0, 400.0]
    for p in p_vals:
        for sh in shifts:
            v = _power_law(n, float(p), float(sh))
            seeds.append(v)
            seeds.append(v[::-1])               # mirrored version

    # 3️⃣ random Dirichlet (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # 4️⃣ random sparse patterns
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # 5️⃣ low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # 6️⃣ Fourier‑log shapes (new family)
    for _ in range(30):
        K = rng.integers(3, 11)                      # 3‑10 frequencies
        amp_scale = rng.uniform(0.03, 0.5)           # modest amplitude
        seeds.append(_fourier_log_shape(n, K, amp_scale, rng))

    # 7️⃣ pairwise mixtures (helps to explore convex hull)
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # 8️⃣ rotations of promising seeds
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # 9️⃣ previous best (warm‑start)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce ℓ₁‑normalisation and non‑negativity
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        restarts = min(10, int(remaining // 40))      # roughly 40 s per restart
        iters_per_restart = int(min(500_000, remaining * 0.25 / restarts))
        print("\n--- EG restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.02, 1.0)
            start_vec = _random_dirichlet(n, conc, rng)   # sum = 1
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.5,
            )
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Stage 3 – long EG run from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(8_000_000, remaining * 0.30 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=3.0,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Stage 4 – aggressive stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(250_000, remaining * 70))
        if iters > 0:
            print("\n--- Stochastic local search (mass transfers + perturb) ---")
            seq_loc, val_loc = _local_search(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.009,
                rng=rng,
            )
            print(f"[LocalSearch] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣ Stage 5 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(400, int(min(1500, remaining * 0.02 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000.0, 1e7, 5e7, 1e8, 5e8, 1e9],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8️⃣ Final evaluation with the provided ``evaluate_sequence``.
    # ------------------------------------------------------------------
    try:
        # Scale to the conventional ℓ₁‑mass √(2·n) before the final call.
        final_scaled = (global_best_seq * target_sum).tolist()
        true_val = evaluate_sequence(final_scaled)
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_scaled = global_best_seq.tolist()
        true_val = None

    # store the (scaled) result globally for possible warm‑starts
    globals()["height_sequence_1"] = final_scaled
    return final_scaled
