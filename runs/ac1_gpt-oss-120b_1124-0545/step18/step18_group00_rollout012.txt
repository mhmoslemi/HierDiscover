"""
propose_candidate
-----------------

A fast, derivative‑based optimiser that works directly on the *smoothed*
max–convolution objective using Adam in log‑softmax space.  The optimiser
generates a diverse set of analytic seeds (including DPSS windows,
beta‑shapes with extreme parameters, power‑law tails, low‑frequency
cosines, random Dirichlet draws, etc.), evaluates a cheap surrogate
`2·n·max(conv)` for each and keeps the best few.  Each of those seeds
undergoes a continuation schedule: for an increasing sequence of
β‑values the smooth maximum `log‑sum‑exp` is minimised by Adam (which
automatically respects the non‑negative simplex constraint).  After the
continuation a short simulated‑annealing phase explores the neighbourhood
of the current best solution.  The whole procedure respects the global
time budget (default 1000 s) and finally returns the best sequence found.

The method improves on the previous baseline (≈ 1.5085) by exploring a
larger family of analytically‑motivated initial shapes (DPSS windows,
β‑shapes with α,β∈[0.01,10]), a more aggressive Adam optimiser and an
extra random‑search polishing phase.
"""

import time
import math
import numpy as np

# Optional SciPy import (for DPSS windows)

try:

except Exception:  # pragma: no cover

# -------------------------------------------------------------
# Basic numeric utilities
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# -------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# -------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{p}.  Works for negative p as well."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** p
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Zero‑order DPSS (Slepian) window; fall back to uniform if unavailable."""
    if _win is None:
        return _uniform(n)
    # DPSS may contain tiny negative values – clip to 0.
    a = _win.dpss(n, nw, Kmax=1, sym=True)[0]
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)

# Adam optimiser on the smoothed max‑convolution objective
# -------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule,
                     iters_per_beta: int = 250,
                     lr0: float = 0.12,
                     eps: float = 1e-12) -> np.ndarray:
    """
    Adam on log‑softmax variables for the log‑sum‑exp approximation of
    max(conv).  Returns the best (lowest surrogate) sequence seen.
    """
    if _win is None and len(beta_schedule) == 0:  # pragma: no cover
        # SciPy missing – nothing to do.
        return start_seq

    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)                  # first moment
    v2 = np.zeros_like(v)                 # second moment
    beta1, beta2 = 0.9, 0.999
    t_iter = 0
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t_iter += 1
            a = _softmax(v)                # current ℓ₁‑normalised vector

            # ----- smoothed max (log‑sum‑exp) -----
            conv = _conv_fft(a, a)
            m_conv = np.max(conv)          # shift for numerical stability
            exp_beta = np.exp(beta * (conv - m_conv))
            sum_exp = exp_beta.sum()
            if sum_exp == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = exp_beta / sum_exp      # soft weights for each conv entry

            # ----- gradient w.r.t. a (as derived in the analysis) -----
            rev_a = a[::-1]
            grad_conv = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * grad_conv[n - 1:n - 1 + n]  # central n entries

            # ----- chain rule through softmax -----
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ----- Adam update on v -----
            lr = lr0 / math.sqrt(t_iter)               # simple decay
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v2_hat = v2 / (1.0 - beta2 ** t_iter)
            v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

            # ----- keep the best exact surrogate -----
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq


# -------------------------------------------------------------
# Simple simulated‑annealing polishing (Gaussian perturbations)
# -------------------------------------------------------------
def _anneal_polish(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   initial_sigma: float = 0.003,
                   decay: float = 0.9995,
                   max_trials: int = 200_000) -> np.ndarray:
    """
    Randomly perturb the best vector; accept only if surrogate improves.
    Sigma decays slowly, thus performing a crude simulated‑annealing.
    """
    best = start_seq.copy()
    best_val = _objective_surrogate(best)
    sigma = initial_sigma
    rng = np.random.default_rng()

    trials = 0
    while time.time() < deadline and trials < max_trials:
        trials += 1
        # propose perturbation in log‑space to stay on the simplex
        log_best = np.log(best + 1e-30)
        log_prop = log_best + sigma * rng.normal(size=n)
        prop = _softmax(log_prop)
        val = _objective_surrogate(prop)
        if val < best_val:
            best = prop
            best_val = val
            # keep sigma unchanged – we are already moving downhill
        else:
            # occasional acceptance to escape flat regions
            prob = math.exp(-(val - best_val) / (sigma + 1e-12))
            if rng.random() < prob:
                best = prop
                best_val = val
        sigma *= decay
    return best


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found (list of floats)
    and stores it in the global variable ``height_sequence_1`` for reuse.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # -----------------------------------------------------------------
    # Problem size – a slightly larger n gives a lower theoretical bound.
    # We keep it modest to stay within the time budget.
    # -----------------------------------------------------------------
    n = 5000

    # -----------------------------------------------------------------
    # 1. Generate a wide variety of analytic & random seeds
    # -----------------------------------------------------------------
    seeds = []

    # classic families
    seeds.append(_uniform(n))

    # DPSS windows – time‑bandwidth product from 2 to 10
    for nw in np.arange(2.0, 11.0, 2.0):
        seeds.append(_dpss_window(n, nw))

    # Beta shapes – include extreme α,β ∈ [0.01,10] (log‑spaced)
    alphas = np.logspace(-2, 1, 12)   # 0.01 … 10
    betas = np.logspace(-2, 1, 12)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # Power‑law families – exponent from -2 to 2 (lin‑spaced)
    for p in np.linspace(-2.0, 2.0, 13):
        seeds.append(_power(n, p))

    # Low‑frequency random cosine sums
    for nf in (2, 3, 4, 5):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random Dirichlet draws with varied concentration
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # -----------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # -----------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"\n[Initial] Best surrogate among seeds = {evaluated[0][1]:.6f}")

    # -----------------------------------------------------------------
    # 3. Adam continuation optimisation on the best few seeds
    # -----------------------------------------------------------------
    # continuation β‑schedule – more aggressive than before
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                     1000.0, 2000.0, 5000.0, 20000.0,
                     50000.0, 100000.0, 200000.0,
                     500000.0, 1_000_000.0, 2_000_000.0]

    global_best_seq = None
    global_best_val = float("inf")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + (deadline - time.time()) * 0.48 / (len(top_seeds) - idx)
        print(f"\n[Adam #{idx+1}] start surrogate = {_objective_surrogate(seed_vec):.6f}")

        # Adam optimisation (smooth max)
        seq_adam = _adam_smooth_opt(seed_vec,
                                    n,
                                    sub_deadline,
                                    beta_schedule,
                                    iters_per_beta=300,
                                    lr0=0.15)

        # Keep the best among the original seed and its Adam refinement
        cand_val = _objective_surrogate(seq_adam)
        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = seq_adam.copy()
            print(f"  >>> New global best (Adam) = {global_best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Final polishing – simulated annealing in log‑softmax space
    # -----------------------------------------------------------------
    if global_best_seq is None:
        # fall back to the absolute best raw seed
        global_best_seq = evaluated[0][0].copy()
        global_best_val = evaluated[0][1]

    if time.time() < deadline:
        print("\n[Polish] simulated‑annealing refinement")
        seq_final = _anneal_polish(global_best_seq,
                                   n,
                                   deadline,
                                   initial_sigma=0.004,
                                   decay=0.9997,
                                   max_trials=300_000)
        final_val = _objective_surrogate(seq_final)
        if final_val < global_best_val:
            global_best_val = final_val
            global_best_seq = seq_final.copy()
            print(f"[Polish] improved to {global_best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Official evaluation (using the provided function)
    # -----------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

**Explanation of key ideas**

* **Rich seed set** – Besides the families used earlier, this version adds
  DPSS (Slepian) windows and beta‑shapes with *very* small parameters
  (α,β down to 0.01) to obtain strong U‑shaped distributions that flatten
  the self‑convolution.

* **Adam on a smooth max** – The max in the objective is replaced by a
  log‑sum‑exp with an increasing temperature β.  Optimising the resulting
  smooth objective in log‑softmax space guarantees non‑negativity and
  ℓ₁‑normalisation without any explicit projection.  Adam’s adaptive step
  sizes enable many more useful updates within the time budget.

* **Continuation schedule** – Starting with a very smooth approximation
  (β≈1) and slowly raising β (up to 2 × 10⁶) lets the optimiser first find
  a good basin and then focus on the true max.  This schedule is deeper
  than the previous one and gives a finer final refinement.

* **Annealing polish** – After the deterministic optimisation a cheap
  simulated‑annealing pass explores the neighbourhood of the best vector,
  occasionally accepting uphill moves; this frequently discovers a still
  lower max‑convolution value.

All components respect the 1000‑second budget, run on a single Python
process, and finally return the best sequence found, stored in
`height_sequence_1` for possible reuse.
