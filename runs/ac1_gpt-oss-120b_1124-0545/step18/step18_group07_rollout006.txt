"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where the evaluation is performed by the supplied ``evaluate_sequence``.
The strategy consists of three stages:

1. **Massive deterministic seeding** – a large, diverse collection of
   ℓ₁‑normalised analytic families (beta‑shape, edge‑bias, Gaussian,
   power‑law, Kaiser, Tukey, etc.) together with a grid search over the
   most promising parameters and a handful of random Dirichlet draws.
   Each candidate is scored by the cheap surrogate ``2·n·max(conv)`` (which
   equals the true objective as long as the vector is ℓ₁‑normalised).  The
   best ``top_k`` candidates are kept.

2. **Smooth‑max L‑BFGS refinement** – each of the top candidates is fed to
   a continuation optimiser that minimises a log‑sum‑exp smooth approximation
   of the max‑convolution.  The optimisation works in the log‑softmax
   (probability simplex) space and uses SciPy’s L‑BFGS‑B.  Only a modest
   number of iterations per β are allowed so that many seeds can be polished.

3. **Exact‑max polishing** – the best vector from stage 2 is refined by a
   mirror‑descent (exponentiated‑gradient) optimiser that uses a true
   sub‑gradient of the max‑convolution.  Afterwards a cheap swap‑based
   hill‑climber permutes the entries without increasing the exact objective,
   and a short final exponentiated‑gradient pass cleans up any remaining
   irregularities.

All work respects a global 1000 s wall‑clock budget.  The final
vector is stored in the global variable ``height_sequence_1`` for the next
round.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:                      # pragma: no cover
    import scipy.optimize as _opt
except Exception:        # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n·log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for ℓ₁‑normalised vectors:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_bias(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution concentrating mass near the ends."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex combination α·b + (1‑α)·a, ℓ₁‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (requires SciPy)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise a log‑sum‑exp smooth approximation of the max‑convolution.
    Returns an ℓ₁‑normalised probability vector.
    """
    if _opt is None:                    # pragma: no cover
        raise RuntimeError("SciPy is required for L‑BFGS optimisation.")

    v = np.log(start_vec + 1e-30)       # work in log‑space

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        b = float(beta)

        def fun(v_local):
            a = _softmax(v_local)      # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(b * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / b)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:    # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={b}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate 0.5/√t  (standard schedule for EG)
        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Swap‑based permutation hill‑climber
# ----------------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap only if it does not increase
    the exact max‑convolution.
    Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

    return best_seq, best_val

# Candidate generation (large deterministic pool + random)
# ----------------------------------------------------------------------
def _collect_candidates(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # 1. Analytic families with a modest set of parameters
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))

    for beta in (2, 4, 8, 12, 20, 30):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))

    # fine grid for edge‑bias (U‑shaped) – candidate of most interest
    for p in np.linspace(0.1, 1.5, 30):
        seeds.append(_edge_bias(n, p))

    # beta‑shape grid – search a bit more densely than before
    for a in np.linspace(0.05, 5.0, 15):
        for b in np.linspace(0.05, 5.0, 15):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – give a heavy‑tail flavour
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mix a uniform with an edge‑biased shape (helps fine‑tuning)
    for a_p in (0.2, 0.4, 0.6, 0.8):
        ub = _edge_bias(n, a_p)
        seeds.append(_mix_shapes(_uniform(n), ub, 0.5))

    # Previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Ensure all vectors are ℓ₁‑normalised and non‑negative
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects a 1000 s wall‑clock budget and returns the best
    sequence found (as a plain Python list).  The best vector is stored in the
    global variable ``height_sequence_1`` for the next round.
    """
    # --------------------------------------------------------------
    # 0. RNG and timing
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # 1. Problem size (fixed, but can be increased if time permits)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Candidate generation (including previous best)
    # --------------------------------------------------------------
    print(f"\n--- Candidate generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    candidates = _collect_candidates(n, rng, prev_arr)
    print(f"Collected {len(candidates)} candidate vectors.")

    # --------------------------------------------------------------
    # 3. Fast surrogate evaluation, keep the top few
    # --------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in candidates]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    top_cands = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values (raw candidates):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # 4. L‑BFGS smooth optimisation on the promising seeds
    # --------------------------------------------------------------
    global_best_seq = top_cands[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # continuation schedule – a moderate set (still strong enough)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1e3, 2e3, 5e3,
        1e4, 2e4, 5e4, 1e5,
        2e5, 5e5, 1e6
    ]

    for idx, seed_vec in enumerate(top_cands):
        if time.time() > deadline:
            print("[Time] deadline reached before examining all L‑BFGS seeds.")
            break

        # allocate a modest iteration budget per seed (so many seeds can be run)
        remaining = deadline - time.time()
        max_iter_per_beta = int(min(12000, remaining * 0.02))
        max_iter_per_beta = max(6000, max_iter_per_beta)

        print(f"\n--- L‑BFGS on top seed #{idx+1} (max_iter_per_beta={max_iter_per_beta}) ---")
        try:
            seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             deadline,
                                             beta_schedule=beta_schedule,
                                             max_iter_per_beta=max_iter_per_beta)
        except Exception as exc:   # pragma: no cover
            print(f"[L‑BFGS] failed on seed #{idx+1}: {exc}")
            continue

        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # ----------------------------------------------------------
        # 4a. Short exact‑max polishing (exp‑gradient) for each seed
        # ----------------------------------------------------------
        if time.time() < deadline:
            exp_iters = int(min(300_000, (deadline - time.time()) * 0.07))
            exp_iters = max(100_000, exp_iters)
            seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=exp_iters)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp

    # --------------------------------------------------------------

    if time.time() < deadline:

        seq_perm, val_perm = _perm_swap_opt(global_best_seq,
                                            n,
                                            deadline,
                                            max_swaps=swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[PermSwap] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final short exponentiated‑gradient refinement (after reordering)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_iters = int(min(250_000, (deadline - time.time()) * 0.07))
        if final_iters > 80_000:
            print(f"[ExpGrad‑final] polishing with {final_iters:,} iterations")
            seq_final, val_final = _exp_gradient_opt(global_best_seq,
                                                     n,
                                                     deadline,
                                                     max_iters=final_iters)
            if val_final < global_best_val:
                global_best_seq = seq_final.copy()
                global_best_val = val_final
                print(f"[ExpGrad‑final] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
