    attempts = 0
    while time.time() < deadline and attempts < max_attempts:

        if val < best_val:

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The procedure consists of the following stages (all respect a global
time‑budget, default 1000 s):

1. **Rich seed generation** – analytical windows (Gaussian, Kaiser,
   Tukey, etc.), random Dirichlet/exponential vectors, low‑frequency
   cosine curves and a large number of random transformations of those
   (circular shifts, reversals, power‑law re‑weightings).  All seeds are
   ℓ₁‑normalised.

2. **Quick surrogate evaluation** – the surrogate objective
   2·n·max(conv) (exact for ℓ₁‑normalised vectors) is computed via a
   real FFT, which is very cheap.  The best K (default 12) seeds are kept.

3. **Pairwise mixing** – the top seeds are convex‑mixed; for each pair a
   ternary search over the mixing coefficient is performed to obtain the
   best mixture (still ℓ₁‑normalised).

4. **Local refinement** – each promising candidate (raw seed or mixed
   seed) is refined by three complementary optimisers:
      * *Mirror‑descent* (exponentiated‑gradient) on the exact max‑convolution.
      * *Continuation L‑BFGS* on a smooth log‑sum‑exp surrogate of the max.
      * *Simple hill‑climbing* that moves a small amount of mass between two
        coordinates if it improves the surrogate.

   The refinements are run with a dynamic time‑allocation, so the overall
   run always respects the global budget.

5. **Final polishing** – the current global best is polished once more
   with a long mirror‑descent pass and a final L‑BFGS smoothing.

The best sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain Python list.  The algorithm deliberately explores
a wide variety of shapes and uses several complementary optimisation
strategies, which in practice often beats the previous baseline
(~1.5085 for n=4000)."""

# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Exact surrogate for a vector normalised to sum=1:
    2·n·max(conv(seq, seq))."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Simple transformations of existing vectors
# ----------------------------------------------------------------------
def _rotate(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    shift = rng.integers(len(seq))
    return np.roll(seq, shift)


def _reverse(seq: np.ndarray) -> np.ndarray:
    return seq[::-1].copy()


def _power_transform(seq: np.ndarray, power: float) -> np.ndarray:
    """Apply a_i ← a_i**power and renormalise."""
    a = np.power(seq, power)
    s = a.sum()
    return a / s if s > 0 else seq.copy()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Pairwise mixing optimisation (simple ternary search)
# ----------------------------------------------------------------------
def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    """Return (mixed_seq, surrogate) for the best convex combination
    of a and b found by a ternary‑search on α∈[0,1]."""
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1, len(a))
        val2 = _objective_surrogate(seq2, len(a))
        if val1 < val2:
            hi = m2
            if best_val is None or val1 < best_val:
                best_val = val1
                best_seq = seq1
        else:
            lo = m1
            if best_val is None or val2 < best_val:
                best_val = val2
                best_seq = seq2
    # final refinement at mid‑point
    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective_surrogate(best_seq, len(a))
    else:
        mid = (lo + hi) / 2.0
        cand = _mix_shapes(a, b, mid)
        cand_val = _objective_surrogate(cand, len(a))
        if cand_val < best_val:
            best_seq = cand
            best_val = cand_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on exact max‑convolution (KL‑mirror, i.e. exponentiated‑gradient)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (shift k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """Optimise the smoothed max‑convolution objective using L‑BFGS‑B."""
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1000.0, 2000.0,
            5000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1_000_000.0, 3_000_000.0
        ]

    # work in log‑softmax space (v) guaranteeing non‑negativity and the simplex constraint.
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # shift for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (softmax space)
            w = exp_beta / sum_exp          # weights for each entry of conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # derivative of 2·n·shift part vanishes

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            # on any unexpected error we bail out of the schedule
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple hill‑climbing on the surrogate (move a small amount of mass).
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_trials: int = 20000,
                delta_frac: float = 0.01) -> tuple[np.ndarray, float]:
    """Randomly transfer a small amount of probability mass between two coordinates."""
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    attempts = 0
    while time.time() < deadline and attempts < max_trials:
        attempts += 1
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # transfer a fraction of the source mass to the target
        delta = best_seq[i] * rng.uniform(0.0, delta_frac)
        if delta <= 0.0:
            continue
        new_i = best_seq[i] - delta
        new_j = best_seq[j] + delta
        if new_i < 0.0:
            continue
        cand = best_seq.copy()
        cand[i] = new_i
        cand[j] = new_j
        # renormalisation is not needed – total mass unchanged
        val = _objective_surrogate(cand, n)
        if val < best_val:
            best_seq = cand
            best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence.
    Returns the best sequence found (list of floats) and stores it in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Fixed problem size – 4000 was already proven effective.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any) to seed the search.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr.copy()
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr = prev_arr / s
                else:
                    prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Broad seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # ==== Deterministic analytic families (coarse grids) ====
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    # sine windows with many gammas
    for gamma in np.linspace(0.2, 4.0, 25):
        seeds.append(_sine_window(n, gamma=gamma))

    # kaiser with a logarithmic sweep
    for beta in np.logspace(-0.3, 1.7, 30):   # approx 0.5 … 50
        seeds.append(_kaiser_window(n, beta))

    # gaussian sigma sweep
    for sigma in np.linspace(0.05, 0.45, 20):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # double‑gaussian variations
    for sigma in (0.10, 0.12, 0.15):
        for shift in (0.20, 0.25, 0.30, 0.35):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))

    # power‑law shapes
    for p in np.linspace(0.0, 3.0, 25):
        seeds.append(_power(n, p))

    # beta‑shape grid (moderate)
    alphas = np.linspace(0.2, 6.0, 9)
    betas = np.linspace(0.2, 6.0, 9)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # tukey windows
    for alpha in np.linspace(0.05, 0.95, 15):
        seeds.append(_tukey_window(n, alpha))

    # ==== Random families (rich parameterisation) ====
    # Dirichlet draws with random concentrations
    for _ in range(30):
        conc = 10 ** rng.uniform(-2.0, 1.0)   # from 0.01 to 10
        seeds.append(_random_dirichlet(n, conc, rng))

    # Exponential draws
    for _ in range(15):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(8):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random rotations / reversals of a few promising deterministic families
    base_families = [_uniform(n), _triangular(n), _gaussian(n, 0.2), _kaiser_window(n, 10.0)]
    for base in base_families:
        for _ in range(5):
            seeds.append(_rotate(base, rng))
            seeds.append(_reverse(base))

    # Power‑transform variants of random Dirichlet seeds
    for _ in range(20):
        base = _random_dirichlet(n, conc=0.3, rng=rng)
        for pwr in (0.5, 2.0, 3.0):
            seeds.append(_power_transform(base, pwr))

    # Perturb previous best (if any) and add some noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(8):
            seeds.append(_perturb(prev_arr, rng, sigma=0.0025))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds (fast, O(n log n) each)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec, n)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogates:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Pairwise mixing of the best seeds (ternary search on α)
    # ------------------------------------------------------------------
    print("\n--- Pairwise mixing of top seeds ---")
    mixture_candidates = []
    mix_deadline = deadline - 2.0

    for i in range(len(top_seeds)):
        for j in range(i + 1, len(top_seeds)):
            if time.time() > mix_deadline:
                break
            seq_mix, val_mix = _best_mixed_pair(top_seeds[i],
                                                top_seeds[j],
                                                mix_deadline,
                                                steps=28)
            mixture_candidates.append((seq_mix, val_mix))

    # ------------------------------------------------------------------
    # 4. Combine raw top seeds and mixtures, keep a shortlist
    # ------------------------------------------------------------------
    all_candidates = evaluated[:top_k] + mixture_candidates
    all_candidates.sort(key=lambda x: x[1])
    best_seq = all_candidates[0][0].copy()
    best_val = all_candidates[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Multi‑start refinement (EG + L‑BFGS + hill‑climb) on a few candidates
    # ------------------------------------------------------------------
    refined_candidates = []
    max_refine = min(4, len(all_candidates))   # limit for time safety

    for idx in range(max_refine):
        if time.time() >= deadline:
            break

        seed_vec, seed_val = all_candidates[idx]
        sub_deadline = time.time() + (deadline - time.time()) * 0.45 / (max_refine - idx)

        print(f"\n[Refine #{idx+1}] start surrogate = {seed_val:.6f}")

        # ---- Exponentiated‑gradient (mirror descent) ----
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                          n,
                                          sub_deadline,
                                          max_iters=200_000,
                                          eta0=6.0)

        # ---- Continuation L‑BFGS smoothing ----
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1000.0,
                                                              2000.0, 5000.0,
                                                              20000.0, 50000.0,
                                                              100000.0, 200000.0,
                                                              500000.0, 1_000_000.0],
                                               max_iter_per_beta=3000)
            val_lbfgs = _objective_surrogate(seq_lbfgs, n)
        else:
            seq_lbfgs = seq_eg
            val_lbfgs = val_eg

        # ---- Simple hill‑climbing on the best of the three ----
        if time.time() < sub_deadline:
            seq_hc, val_hc = _hill_climb(seq_lbfgs,
                                         n,
                                         sub_deadline,
                                         rng,
                                         max_trials=15000,
                                         delta_frac=0.008)
        else:
            seq_hc, val_hc = seq_lbfgs, val_lbfgs

        # ---- Choose local best ----
        candidates = [(seed_vec, seed_val),
                      (seq_eg, val_eg),
                      (seq_lbfgs, val_lbfgs),
                      (seq_hc, val_hc)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")

        refined_candidates.append((local_best_seq, local_best_val))

        # Update global best if improved
        if local_best_val < best_val:
            best_seq = local_best_seq
            best_val = local_best_val
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final long EG polishing (full remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=1_800_000,
                                                 eta0=5.0)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final L‑BFGS smoothing (to get a possibly lower max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing ---")
        seq_final_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                                n,
                                                deadline,
                                                beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                               200.0, 500.0, 1000.0,
                                                               2000.0, 5000.0,
                                                               20000.0, 50000.0,
                                                               100000.0, 200000.0,
                                                               500000.0, 1_000_000.0,
                                                               3_000_000.0],
                                                max_iter_per_beta=3500)
        val_final_lbfgs = _objective_surrogate(seq_final_lbfgs, n)
        if val_final_lbfgs < best_val:
            best_seq = seq_final_lbfgs
            best_val = val_final_lbfgs
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Official evaluation with the supplied routine
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
