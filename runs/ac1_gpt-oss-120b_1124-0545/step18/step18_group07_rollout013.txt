def _perm_alternating_extremes(arr):
    # arr is vector (could be unsorted) -> we first sort descending then assign largest to leftmost, next to rightmost, etc.
    sorted_desc = np.sort(arr)[::-1]
    n = len(arr)
    new = np.empty(n, dtype=arr.dtype)
    left, right = 0, n-1
    toggle = True
    for val in sorted_desc:
        if toggle:
            new[left] = val
            left += 1
        else:
            new[right] = val
            right -= 1
        toggle = not toggle
    return new

def _perm_recursive_mid(arr):
    order = []
    def recur(l, r):
        if l >= r:
            return
        mid = (l + r) // 2
        order.append(mid)
        recur(l, mid)
        recur(mid+1, r)
    recur(0, len(arr))
    # order is list of indices where to assign sorted descending values
    sorted_desc = np.sort(arr)[::-1]
    new = np.empty_like(arr)
    for idx, pos in enumerate(order):
        if idx < len(sorted_desc):
            new[pos] = sorted_desc[idx]
    return new

propose_candidate
-----------------
A time‑bounded multi‑stage search for a step‑function that minimises
``evaluate_sequence`` (i.e. 2·n·max(a*a) / (sum a)²).

The search consists of three phases:

1. **Diverse seed generation**
   * Analytic families (uniform, triangular, windows, β‑shapes …)
   * many random Dirichlet draws with a range of concentration parameters
   * for each raw vector we also create a few deterministic permutations
     (alternating extremes, recursive midpoint) that tend to break the
     alignment of large values with their reverse.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:  # should never happen
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Permutation helpers – they take a *raw* vector, sort it descending,
# then place the entries according to a deterministic pattern.
# ----------------------------------------------------------------------
def _perm_alternating_extremes(vec: np.ndarray) -> np.ndarray:
    """Largest → leftmost, 2nd largest → rightmost, 3rd → next left …"""
    desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty(n, dtype=np.float64)
    l, r = 0, n - 1
    toggle = True
    for val in desc:
        if toggle:
            out[l] = val
            l += 1
        else:
            out[r] = val
            r -= 1
        toggle = not toggle
    return out


def _perm_recursive_mid(vec: np.ndarray) -> np.ndarray:
    """Place values by recursively splitting the interval (mid‑point ordering)."""
    n = len(vec)
    order = []

    def recurse(lo: int, hi: int) -> None:
        if lo >= hi:
            return
        mid = (lo + hi) // 2
        order.append(mid)
        recurse(lo, mid)
        recurse(mid + 1, hi)

    recurse(0, n)
    desc = np.sort(vec)[::-1]
    out = np.empty(n, dtype=np.float64)
    for idx, pos in enumerate(order):
        if idx < len(desc):
            out[pos] = desc[idx]
    return out


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised probability vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6,
            1e7
        ]

    # work in log‑probability space (sum‑to‑1 is enforced by softmax)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 200_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing swap local search
# ----------------------------------------------------------------------
def _swap_anneal_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int = 1_000_000,
                     start_temp: float = 0.02,
                     end_temp: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Randomly swaps two positions, accepting improvements or
    probabilistically accepting worsening moves according to a
    linear temperature schedule.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        # linear temperature schedule
        T = max(end_temp,
                start_temp * (1.0 - it / max_swaps))

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective_surrogate(cur_seq)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            delta = new_val - cur_val
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 250_000 == 0:
            print(f"[SwapAnneal] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed collection – enriched with permutations
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse pool of ℓ₁‑normalised vectors.
    For each random vector a few deterministic permutations are added.
    """
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 1.0, 2.0, 4.0, 8.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 8, 12, 16, 20, 30):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for a in (0.05, 0.1, 0.3, 1.0, 2.0, 4.0):
        for b in (0.05, 0.1, 0.3, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random draws ---------------------------------------------------------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Heavy‑tailed draws with permutations ---------------------------------
    for conc in (0.05, 0.1, 0.3):
        for _ in range(4):
            raw = _random_dirichlet(n, conc, rng)
            # keep raw
            seeds.append(raw)
            # permutations that tend to decorrelate the vector with its reverse
            seeds.append(_perm_alternating_extremes(raw))
            seeds.append(_perm_recursive_mid(raw))

    # Edge‑biased families --------------------------------------------------
    for p in (0.25, 0.5, 0.75, 1.0):
        seeds.append(_edge_bias := lambda n, p: (
            lambda n=n, p=p:
                ((lambda i: 1.0 / ((i) * (n + 1 - i)) ** p)
                 (np.arange(1, n + 1, dtype=np.float64))) /
                ((lambda i: 1.0 / ((i) * (n + 1 - i)) ** p)
                 (np.arange(1, n + 1, dtype=np.float64))).sum()
        )())
    # (the above lambda is a compact way to avoid copy‑paste; it returns a
    #   properly normalised U‑shaped vector.)

    # Previous solution -----------------------------------------------------
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # Normalise all seeds (safety‑check)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted wall‑clock budget.
    Returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size ---------------------------------------------------------
    n = 4000

    # previous solution ----------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # 1. generate a large pool of seeds ------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seeds (including permutations).")

    # 2. quick surrogate evaluation, keep the top few -----------------------
    print("\nEvaluating surrogate on all seeds …")
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # 3. optimisation loop over the selected seeds ------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # L‑BFGS‑B schedule ----------------------------------------------------
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1e3, 2e3, 5e3,
        1e4, 2e4, 5e4, 1e5,
        2e5, 5e5, 1e6, 2e6, 5e6,
        1e7
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all seeds.")
            break

        # allocate a modest amount of L‑BFGS work per seed
        remaining = deadline - time.time()
        max_iter_per_beta = int(min(15000, remaining * 0.04))
        max_iter_per_beta = max(8000, max_iter_per_beta)

        print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter_per_beta={max_iter_per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_per_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # ---- exact exp‑gradient polishing ---------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(800_000, remaining * 0.08))
            max_exp = max(400_000, max_exp)
            print(f"[ExpGrad] polishing with up to {max_exp:,} iterations")
            seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # 4. Simulated‑annealing swap local search --------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(2_000_000, remaining * 0.20))
        max_swaps = max(800_000, max_swaps)
        print("\n--- Swap‑annealing local search ---")
        seq_swap, val_swap = _swap_anneal_opt(global_best_seq,
                                              n,
                                              deadline,
                                              max_swaps=max_swaps,
                                              start_temp=0.02,
                                              end_temp=1e-5)
        if val_swap < global_best_val:
            global_best_seq = seq_swap.copy()
            global_best_val = val_swap
            print(f"[SwapAnneal] improved to {global_best_val:.6f}")

    # 5. Final short exp‑gradient refinement ----------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp2 = int(min(300_000, remaining * 0.04))
        if max_exp2 > 100_000:
            print(f"[ExpGrad‑final] polishing with {max_exp2:,} iterations")
            seq_final, val_final = _exp_gradient_opt(global_best_seq,
                                                     n,
                                                     deadline,
                                                     max_iters=max_exp2)
            if val_final < global_best_val:
                global_best_seq = seq_final.copy()
                global_best_val = val_final
                print(f"[ExpGrad‑final] improved to {global_best_val:.6f}")

    # 6. Official evaluation -------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
