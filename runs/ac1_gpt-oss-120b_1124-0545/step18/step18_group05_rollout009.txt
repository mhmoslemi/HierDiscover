"""
A multi‑phase optimiser for the step‑function problem.

Phase 1 – diverse analytic / random seeds.
Phase 2 – Adam mirror‑descent on the exact L∞ objective
          (sub‑gradient w.r.t. the active convolution entry).
Phase 3 – Continuation L‑BFGS on a smooth log‑sum‑exp surrogate.
Phase 4 – Fast local “mass‑transfer” hill‑climbing that updates the
          convolution incrementally.
Phase 5 – (optional) a short simulated‑annealing walk for fine
          polishing.

All helper functions are defined at top level – no closures or lambdas.
The routine respects the 1000 s budget and returns the best sequence it
finds.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for L‑BFGS‑B (used only for polishing)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:                     # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exact surrogate (objective for a probability vector)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# L‑BFGS helper (smooth log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_obj(v_local: np.ndarray, n: int, beta: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of 2·n·max(conv(a,a)) where a = softmax(v).

    Returns (objective, gradient wrt v).
    """
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)

    shifted = conv - max_c
    exp_beta = np.exp(beta * shifted)
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)

    # gradient of the smooth surrogate
    w = exp_beta / sum_exp                     # soft weights, sum = 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)               # chain rule through softmax
    return obj, grad_v


def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 2000) -> np.ndarray:
    """Polish a probability vector with L‑BFGS‑B on the smooth surrogate."""
    if _opt is None:                                 # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # minimise the smooth surrogate for the current β
        try:
            res = _opt.minimize(fun=_lbfgs_obj,
                                x0=v,
                                args=(n, beta),
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except Exception as e:                         # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break
    return _softmax(v)


# ----------------------------------------------------------------------
# Adam mirror‑descent on the exact L∞ objective
# ----------------------------------------------------------------------
def _adam_mirror_descent(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the sub‑gradient of max(conv(a,a)).
    Returns best sequence seen and its surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        dot = np.dot(g, a)
        grad_v = a * (g - dot)                     # chain‑rule through softmax

        # Adam update on the log‑its
        m = beta1 * m + (1 - beta1) * grad_v
        v = beta2 * v + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        eta = 0.5 / math.sqrt(t)
        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)

        a = _softmax(log_a)

        if t % 20000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[AdamMD] it={t:,} new best surrogate {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local hill‑climbing by moving a small chunk of mass between two positions
# ----------------------------------------------------------------------
def _local_hill_climb(seq: np.ndarray,
                      deadline: float,
                      max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Fast coordinate‑wise refinement.
    Moves a fraction of mass from one index to another and accepts the move
    only if it strictly lowers max(conv).
    """
    n = len(seq)
    a = seq.copy()
    L = 2 * n - 1
    conv = _conv_fft(a, a)
    best_max = float(np.max(conv))
    best_seq = a.copy()
    best_val = 2.0 * n * best_max

    rng = np.random.default_rng()
    moves = 0

    while time.time() < deadline and moves < max_moves:
        moves += 1

        # pick i proportionally to its weight (biggest contributors first)
        i = rng.choice(n, p=a)
        # pick a random destination index
        j = rng.integers(0, n)
        if i == j or a[i] < 1e-12:
            continue

        # move at most 10 % of the source mass (but at least a tiny amount)
        frac = rng.uniform(0.01, 0.10)
        delta = min(frac * a[i], a[i])

        # compute tentative new convolution
        new_conv = conv.copy()
        new_conv[i:i + n] -= delta * a
        new_conv[j:j + n] += delta * a
        new_max = float(new_conv.max())

        if new_max + 1e-16 < best_max:          # strict improvement
            # accept the move
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            best_max = new_max
            best_seq = a.copy()
            best_val = 2.0 * n * best_max

        if moves % 20000 == 0:
            print(f"[HillClimb] moves={moves:,} best surrogate {best_val:.6f}")

    return best_seq, best_val

# Simulated‑annealing refinement (logit‑space) – kept from the original code
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val
