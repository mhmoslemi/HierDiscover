"""
Search for a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

The algorithm expands the seed pool (adding Gaussian and offset power‑law
families, a finer grid of beta‑shapes, etc.), keeps the elite seeds,
performs a continuation L‑BFGS optimisation on a smooth log‑sum‑exp
surrogate, refines the result with several exponentiated‑gradient (mirror
descent) runs, and finally applies a cheap simulated‑annealing hill‑climb.
All stages respect a user‑supplied time budget (default 1000 s) and the
best sequence found is stored globally as ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, p>0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law_offset(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i+shift)^{-p} with a shift>0 to soften the head."""
    i = np.arange(0, n, dtype=np.float64) + shift
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle of the array."""
    mu = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    """Cyclic shift."""
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# objective helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# smooth surrogate (log‑sum‑exp) for continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate of the max‑convolution objective."""
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp: float = 0.8,
    cooling: float = 0.99995,
    max_frac: float = 0.03,
) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass from i to j, accepting occasional uphill steps."""
    rng = np.random.default_rng(12345)
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    cur_val = best_val
    temp = init_temp

    for it in range(max_iters):
        if time.time() > deadline:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue

        # random fraction of the mass at i
        frac = rng.random() * max_frac
        delta = frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _objective_exact(cand)
        delta_val = val - cur_val
        accept = False
        if delta_val < 0:
            accept = True
        else:
            # metropolis criterion
            prob = np.exp(-delta_val / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = a.copy()

        # temperature schedule (log‑linear)
        if (it + 1) % 2000 == 0:
            temp *= cooling
            if temp < 1e-12:
                temp = 1e-12

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed‑pool construction – enriched with new families
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # basic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.0002))

    # beta grid – includes very skewed shapes
    for a in (0.2, 0.4, 0.6, 0.8, 1.2, 2.0):
        for b in (0.2, 0.4, 0.6, 0.8, 1.2, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian – a set of sigmas covering very narrow to very wide
    for sigma_frac in np.linspace(0.03, 0.45, 15):   # fraction of n
        sigma = sigma_frac * n
        seeds.append(_gaussian_shape(n, sigma))

    # power‑law grid (dense)
    for p in np.linspace(0.25, 1.00, 30):
        seeds.append(_power_law(n, float(p)))

    # offset power‑law (shift helps to tame the head)
    for p in np.linspace(0.30, 0.80, 12):
        for shift in (0.2, 0.5, 1.0, 2.0):
            seeds.append(_power_law_offset(n, float(p), float(shift)))

    # random Dirichlet families (various concentrations)
    for conc in (0.01, 0.03, 0.07, 0.15, 0.4, 0.8):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.01, 0.03, 0.07, 0.15):
        for _ in range(3):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures of already‑generated seeds
    base = np.arange(len(seeds))
    for _ in range(120):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(20):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean‑up – enforce non‑negativity & ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The function respects a time budget (default 1000 s) and stores the
    best sequence globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety cushion

    # --------------------------------------------------------------
    n = 4000                     # length of the candidate
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # --------------------------------------------------------------
    # Assemble seed pool
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # --------------------------------------------------------------
    # Quick exact evaluation of seeds, keep the elite pool
    # --------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    # keep the top‑k seeds for heavy L‑BFGS stage
    seeds.sort(key=_objective_exact)
    top_k = min(25, len(seeds))
    elite = seeds[:top_k]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # --------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate
    # --------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0,
    ]

    per_beta = 1500
    print("\n--- L‑BFGS refinement of elite seeds ---")
    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break
        print(f"[L‑BFGS] seed #{idx + 1}/{len(elite)}")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"    -> objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"    ** new global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – several exponentiated‑gradient (mirror‑descent) restarts
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 18
        iters_per_restart = int(min(400_000, remaining * 0.12 / restarts))
        print("\n--- EG restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            # random Dirichlet start
            conc = rng.uniform(0.02, 1.5)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.0,
            )
            print(f"[EG {r + 1}/{restarts}]  best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"    ** new global best after EG = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – simulated annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(900_000, remaining * 0.30 * 1e5))
        if max_iters > 0:
            print("\n--- Simulated Annealing fine‑tune ---")
            seq_sa, val_sa = _simulated_annealing_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                init_temp=0.7,
                cooling=0.99996,
                max_frac=0.02,
            )
            print(f"[SA] best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"    ** new global best after SA = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final high‑β L‑BFGS polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1200, int(min(3500, remaining * 0.08 * 1e5)))
        print("\n--- Final high‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0, 1e7],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"    ** new global best after final polishing = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in future invocations
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
