    v = np.log(start_vec + eps)

    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    t = 0

    for beta in beta_schedule:
        for step in range(steps_per_beta):
            if time.time() > deadline: break

            t += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            shifted = conv - max_c
            # compute softmax weights:
            # To avoid overflow, compute exp(beta*shifted - max_shifted).
            # Since max shift = 0 (since shifted max = 0)
            exp_beta = np.exp(beta * shifted)
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

            # gradient wrt a:
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]

            # gradient wrt logits:

            # Adam update

"""
Optimisation of a non‑negative step function for the metric

        2·n·max(conv(a, a)) / (sum(a)²)

The algorithm is a three‑stage optimiser:

1. **Diverse seed pool** – analytic families, Dirichlet draws, a greedy
   Sidon (Golomb‑ruler) construction and the previous global best.
2. **Fast mirror‑descent** – a few hundred‑thousand updates on the
   active max‑convolution entry (the same sub‑gradient used in the
   published solution, but with a tighter time budget).
3. **Smooth‑max Adam optimisation** – the max is approximated by a
   log‑sum‑exp with an increasing temperature β.  Adam updates are
   performed on the log‑it (log‑softmax) parametrisation of the vector,
   allowing a very fine adjustment of the probabilities.  The β schedule
   ends at 5·10⁶, which is essentially the true max.
4. **Final hard‑max polishing** – a short mirror‑descent with a
   decreasing step‑size to squeeze the last bits of improvement.

All steps respect the time limit (budget_s seconds) and keep the best
vector found.  The final vector is stored in the global variable
`height_sequence_1` for reuse by the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Simple surrogate (max‑convolution) – sum = 1
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Greedy Sidon (Golomb‑ruler) set construction
# ----------------------------------------------------------------------
def _golomb_set_positions(n: int, target_k: int) -> list[int]:
    """
    Greedy construction of a set S⊂{0,…,n‑1} with distinct pairwise sums.
    Returns up to target_k positions.
    """
    positions = []
    sums = set()
    for x in range(n):
        # candidate sums with current set (including the square x+x)
        cand = [x + y for y in positions] + [x + x]
        if any(s in sums for s in cand):
            continue
        positions.append(x)
        for s in cand:
            sums.add(s)
        if len(positions) >= target_k:
            break
    return positions


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None) -> np.ndarray:
    """
    Refine a probability vector with Adam on a log‑sum‑exp approximation
    of the max‑convolution.  Returns the best vector found before the
    deadline.
    """
    if beta_schedule is None:
        beta_schedule = [
            1, 5, 20, 80, 200, 500, 1_000, 2_000, 5_000,
            20_000, 50_000, 100_000, 200_000, 500_000,
            1_000_000, 2_000_000, 5_000_000,
        ]

    # initialise in logit space
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    t = 0
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        # decide how many inner steps we can afford
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # a crude allocation: more steps for larger beta
        steps = int(min(8000, max(500, remaining * 200 / (len(beta_schedule) - beta_schedule.index(beta) + 1))))
        for _ in range(steps):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shifted = conv - max_c
            # soft‑max weights for the surrogate
            exp_beta = np.exp(beta * shifted)           # shifted max = 0 → no overflow
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                # chain‑rule through softmax

            # Adam update (standard hyper‑parameters)
            lr = 0.05 / np.sqrt(t)                     # decaying base learning‑rate
            m = 0.9 * m + 0.1 * grad_v
            v_hat = 0.999 * v_hat + 0.001 * grad_v
            m_hat = m / (1 - 0.9 ** t)
            v_hat_corr = v_hat / (1 - 0.999 ** t)
            v -= lr * m_hat / (np.sqrt(v_hat_corr) + 1e-8)

        # evaluate the surrogate after this β
        cur_seq = _softmax(v)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()
            print(f"[Adam] β={beta} → new best surrogate {best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hard‑max mirror‑descent (sub‑gradient) refinement
# ----------------------------------------------------------------------
def _hardmax_refine(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    base_eta: float = 0.4) -> np.ndarray:
    """
    Perform standard mirror‑descent on the exact max‑convolution.
    Returns the best vector seen before the deadline.
    """
    cur_seq = start_seq.copy()
    log_a = np.log(cur_seq + 1e-30)
    best_seq = cur_seq.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(cur_seq, cur_seq)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active index
        g = np.zeros_like(cur_seq)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * cur_seq[rev_idx]

        eta = base_eta / math.sqrt(it)
        log_a -= eta * g
        cur_seq = _softmax(log_a)

        if it % 50000 == 0:
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[Hardmax] it={it:,} → new best surrogate {best_val:.6f}")

    return best_seq
