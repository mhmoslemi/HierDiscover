def _lbfgs_opt(start_vec, n, deadline):

        raise RuntimeError
    v = np.log(start_vec + 1e-30)
    # schedule
    betas = [1,5,20,100,500,2000,10000,50000,200000,1e6] maybe.
    for beta in betas:
        if time.time()>deadline: break
        beta_val = float(beta)
        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a,a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
            # gradient

            rev_a = a[::-1]

            grad_a = 2.0 * full[n-1:n-1+n]  # size n
            # backprop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v
        def callback(xk):
            if time.time() > deadline:
                raise KeyboardInterrupt()

                               callback=callback,

    best_seq = cur_seq.copy()
    best_val = cur_val
    T0 = 0.1 * cur_val if cur_val>0 else 1.0
    for it in range(1, max_swaps+1):
        if time.time() > deadline: break
        i, j = rng.integers(0,n,size=2)
        if i==j:
            continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective(cur_seq)
        delta = new_val - cur_val
        T = max(T0 * (1.0 - it / max_swaps), 1e-12)
        if delta < 0 or rng.random() < math.exp(-delta/T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        if it % 50000 == 0:
            print(f"[AnnealSwap] swaps={it:,} best={best_val:.6f}")

"""
Propose a non‑negative step‑function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The search works in the following stages (all within a 1000 s budget).

1. **Seed generation** – a large pool of analytically‑generated shapes
   (uniform, triangular, cosine‑type windows, beta‑shapes, Gaussians,
   Kaiser, power‑law, random Dirichlet / exponential draws, …).  In
   addition a small parameter grid (beta‑shape, Gaussian σ, Tukey α,
   power‑p, Kaiser β, sine‑window γ) is explored, producing many
   diverse ℓ₁‑normalised candidates.

2. **Coarse evaluation** – each seed is evaluated with the exact
   objective `2·n·max(conv)` (the denominator is 1 after normalisation).
   The best ~20 seeds are kept.

3. **Smooth‑max L‑BFGS** – each top seed is refined by minimising a
   log‑sum‑exp surrogate of the max‑convolution.  The optimisation is
   performed in log‑probability space, so the simplex constraint is
   enforced automatically.  A schedule of increasing smoothness
   parameters β is used; the L‑BFGS routine is stopped early if the
   global time budget is exhausted.

4. **Exact‑max exponentiated‑gradient polishing** – the best sequence
   from L‑BFGS is polished with mirror‑descent on the true max‑convolution
   (sub‑gradient on the active lag, step‑size ηₜ = 0.5/√t).  This phase
   can use several hundred thousand iterations because each iteration
   is cheap (FFT‑based convolution).

5. **Permutation annealing** – a Metropolis‑style swap‑based local‑search
   explores the permutation space of the current best sequence,
   accepting worsening moves with a linearly‑cooled temperature.

6. **Final refinement** – a short L‑BFGS pass is attempted after the
   annealing, then the final sequence is stored in the global variable
   ``height_sequence_1`` and returned as a list of floats.

All stages respect a global deadline (budget_s – a few seconds safety
margin).  The code uses only NumPy and SciPy (for L‑BFGS) and contains
no file or network I/O.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_gaussian(n, 0.20))

    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))

    seeds.append(_power(n, 0.0))

    seeds.append(_kaiser_window(n, beta=4.0))

    seeds.append(_tukey_window(n, alpha=0.3))

    for conc in (0.1, 0.3, 0.7, 1.5):

        s = v.sum()
        if s > 0:

        else:

# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_opt(start_vec: np.ndarray,
               n: int,
               deadline: float) -> np.ndarray:
    """Run L‑BFGS on the smooth max surrogate."""
    if _opt is None:
        raise RuntimeError("SciPy is required for the L‑BFGS optimiser.")

    v = np.log(start_vec + 1e-30)       # work in log‑probability space

    # smoothing schedule – progressively larger β
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 1e3, 2e3, 5e3,
                     1e4, 2e4, 5e4, 1e5,
                     2e5, 5e5, 1e6, 2e6, 5e6, 1e7]

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised automatically
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max via log‑sum‑exp
            shift = max_c                     # improve numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (derived from smoothed max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        # per‑β iteration budget – generous but time‑aware
        max_iter = 4000
        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient optimiser (mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient at the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # classic 1/√t step‑size
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing permutation hill‑climber
# ----------------------------------------------------------------------
def _anneal_swap_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int = 300_000) -> tuple[np.ndarray, float]:
    """Random swaps with Metropolis acceptance."""
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # initial temperature – a small fraction of the current objective
    T0 = 0.1 * cur_val if cur_val > 0 else 1.0

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective(cur_seq)
        delta = new_val - cur_val

        # linear cooling schedule
        T = max(T0 * (1.0 - it / max_swaps), 1e-12)
        accept = (delta < 0) or (rng.random() < math.exp(-delta / T))

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 50_000 == 0:
            print(f"[AnnealSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Symmetrisation helper (optional)
# ----------------------------------------------------------------------
def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Average a vector with its reversal and renormalise."""
    mixed = (seq + seq[::-1]) * 0.5
    s = mixed.sum()
    return mixed / s if s > 0 else seq.copy()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------- problem size -------------------
    n = 4000

    # ------------------- previous solution (if any) -------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list):
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            prev_arr = arr / arr.sum()

    # ------------------- seed generation -------------------
    print(f"\n--- Seed generation (n={n}) ---")
    base_seeds = _collect_base_seeds(n, rng, prev_arr)
    grid_seeds = _grid_search_seeds(n, rng, deadline)
    all_seeds = base_seeds + grid_seeds
    print(f"Collected {len(all_seeds)} seed vectors.")

    # ------------------- quick evaluation of seeds -------------------
    evaluations = [(vec, _objective(vec)) for vec in all_seeds]
    evaluations.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [vec for vec, _ in evaluations[:top_k]]
    print("\nTop seed objective values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluations[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------- optimisation loop over top seeds -------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all seed optimisation.")
            break

        # L‑BFGS stage
        remaining = deadline - time.time()
        max_iter_per_beta = max(2000, int(min(4000, remaining * 0.03)))
        print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter_per_beta≈{max_iter_per_beta}) ---")
        seq_opt = _lbfgs_opt(seed_vec, n, deadline)
        val_opt = _objective(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best after L‑BFGS = {global_best_val:.6f}")

        # Exact‑max exponentiated‑gradient polishing
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_iters = max(200_000, int(min(300_000, remaining * 0.5)))
            seq_exp, val_exp = _exp_gradient_opt(global_best_seq, n, deadline,
                                                 max_iters=max_iters)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------- optional symmetrisation -------------------
    if time.time() < deadline:
        seq_sym = _symmetrize(global_best_seq)
        val_sym = _objective(seq_sym)
        if val_sym < global_best_val:
            global_best_seq = seq_sym
            global_best_val = val_sym
            print(f"[Symmetrize] improved to {global_best_val:.6f}")

    # ------------------- simulated‑annealing permutation hill‑climber -------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = max(200_000, int(min(300_000, remaining * 0.4)))
        print("\n--- Simulated‑annealing swap local search ---")
        seq_perm, val_perm = _anneal_swap_opt(global_best_seq, n, deadline,
                                              max_swaps=max_swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm
            global_best_val = val_perm
            print(f"[AnnealSwap] improved to {global_best_val:.6f}")

    # ------------------- final short L‑BFGS refinement -------------------
    if time.time() < deadline:
        print("\n--- Final short L‑BFGS refinement ---")
        seq_ref = _lbfgs_opt(global_best_seq, n, deadline)
        val_ref = _objective(seq_ref)
        if val_ref < global_best_val:
            global_best_seq = seq_ref
            global_best_val = val_ref
            print(f"[Refine] final objective = {global_best_val:.6f}")

    # ------------------- official evaluation -------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later call
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
