"""
Propose a non‑negative step‑function (vector) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm improves on the previous baseline by

1. **Richer seed pool** – many analytic families (beta, power, Gaussian,
   sine‑window, mixtures) and a large set of random draws with widely
   sampled parameters.
2. **Permutation search on the seed** – for each seed we try a few random
   shuffles (different orderings of the same multiset of values) because the
   convolution peak strongly depends on the ordering.
3. **Smooth‑max L‑BFGS** – minimise a log‑sum‑exp surrogate of the max‑convolution
   using a log‑softmax parametrisation (the ℓ₁‑norm is preserved automatically).
4. **Exact‑max exponentiated‑gradient polish** – mirror‑descent on the true
   max‑convolution with a 1/√t learning‑rate schedule.
5. **Final annealing on permutations** – Metropolis swaps (and occasional
   segment reversals) to fine‑tune the ordering.
6. **Symmetrisation step** – averaging with the reversed copy, because a
   symmetric shape often reduces the autocorrelation peak.

All stages are time‑aware: the remaining budget is split dynamically.
The best sequence found is stored in the global variable ``height_sequence_1``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum(seq)**2)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    s = np.sum(seq)
    # sum is forced ≥ 0.01 by evaluate_sequence – we avoid division by zero.
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # 1) Analytic families with broad parameter sweeps
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.2, 3.0, 7):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma in np.linspace(0.07, 0.45, 8):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 12.0, 13):
        seeds.append(_power(n, p))

    for _ in range(30):
        a = rng.uniform(0.01, 20.0)
        b = rng.uniform(0.01, 20.0)
        seeds.append(_beta_shape(n, a, b))

    # 2) Mixture of two beta shapes (adds extra flexibility)
    for _ in range(10):
        a1 = rng.uniform(0.01, 10.0)
        b1 = rng.uniform(0.01, 10.0)
        a2 = rng.uniform(0.01, 10.0)
        b2 = rng.uniform(0.01, 10.0)
        w = rng.random()
        mix = w * _beta_shape(n, a1, b1) + (1 - w) * _beta_shape(n, a2, b2)
        s = mix.sum()
        if s > 0:
            seeds.append(mix / s)

    # 3) Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # 4) Random Dirichlet & exponential draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # 5) Heavy‑tailed alternating Dirichlet (helps permutation search)
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    seeds.append(a / a.sum())

    # 6) Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity‑check
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Smooth‑max L‑BFGS optimiser
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Optimise a log‑sum‑exp smoothed max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6, 1e7,
        ]

    # work in log‑probability space – softmax respects simplex automatically
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp (shifted by max_c for stability)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (derived from the smoothed max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # len = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Exact‑max exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress output
        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Simulated‑annealing permutation hill‑climber (swap + segment‑reverse)
# ----------------------------------------------------------------------
def _anneal_swap_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int = 300_000) -> tuple[np.ndarray, float]:
    """Random swaps (and occasional segment reversals) with Metropolis acceptance."""
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # initial temperature – a fraction of the current objective
    T0 = max(0.1 * cur_val, 1e-3)

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        # random move: swap (most common) or reverse a short segment (rare)
        if rng.random() < 0.9:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            new_val = _objective(cur_seq)
            delta = new_val - cur_val
            revert = lambda: cur_seq.__setitem__(slice(i, i + 1), cur_seq[j:j + 1])  # dummy (won’t be called)
        else:
            i, j = sorted(rng.integers(0, n, size=2))
            if j - i < 2:
                continue
            # reverse in‑place
            cur_seq[i:j + 1] = cur_seq[i:j + 1][::-1]
            new_val = _objective(cur_seq)
            delta = new_val - cur_val

        # temperature schedule – linear cooling
        T = max(T0 * (1.0 - it / max_swaps), 1e-12)
        accept = delta < 0 or rng.random() < math.exp(-delta / T)

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert move
            if rng.random() < 0.9:
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            else:
                cur_seq[i:j + 1] = cur_seq[i:j + 1][::-1]

        if it % 50_000 == 0:
            print(f"[AnnealSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val

def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Average a vector with its reversal and renormalise."""
    mixed = (seq + seq[::-1]) * 0.5
    s = mixed.sum()
    return mixed / s if s > 0 else seq.copy()

        except RuntimeError:
            # SciPy missing – fall back to simple mirror‑descent
            seq_opt, _ = _exp_gradient_opt(seed_vec, n, deadline, max_iters=200_000)

        val_opt = _objective(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best after L‑BFGS = {global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Stage 2 – Exact‑max exponentiated‑gradient polishing
        # ------------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp_iters = int(min(800_000, remaining * 0.10))
            max_exp_iters = max(400_000, max_exp_iters)
            seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp_iters)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Symmetrisation (may lower the peak further)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sym = _symmetrize(global_best_seq)
        val_sym = _objective(seq_sym)
        if val_sym < global_best_val:
            global_best_seq = seq_sym
            global_best_val = val_sym
            print(f"[Symmetrize] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Permutation annealing (fine‑tune ordering)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(600_000, remaining * 0.12))
        max_swaps = max(300_000, max_swaps)
        print("\n--- Simulated‑annealing permutation fine‑tuner ---")
        seq_perm, val_perm = _anneal_swap_opt(global_best_seq,
                                              n,
                                              deadline,
                                              max_swaps=max_swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[AnnealSwap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
