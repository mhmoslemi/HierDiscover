    v = np.log(start_a + 1e-30)  # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, max_iters+1):
        if time.time()>deadline: break
        a = _softmax(v)
        # compute grad of objective = 2*n*max_conv? Actually we just need gradient w.r.t v of max_conv

        conv = _conv_fft(a, a)

        g = _subgradient_maxconv(a)  # shape n
        # gradient wrt v: a*(g - dot(g,a))

def _adam_optimize(start_a: np.ndarray, n: int, deadline: float, max_iters: int = 200000, lr: float = 0.2):
    a = start_a.copy()
    # ensure sum=1 after any rounding
    a = a / a.sum()
    v = np.log(a + 1e-30)  # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_val = _objective_exact(a)
    for t in range(1, max_iters+1):
        if time.time()>deadline:
            break
        a = _softmax(v)  # ensure sum=1
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        # subgradient of maxconv
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g[i_range] = 2.0 * a[rev]
        # gradient w.r.t. logits:
        grad_v = a * (g - np.dot(g, a))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v * grad_v)
        m_hat = m / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)
        v = v - lr * m_hat / (np.sqrt(v_hat_hat) + eps)
        if t % 10000 == 0:
            cur_val = _objective_exact(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
    # final check
    cur_val = _objective_exact(a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()
    return best_a, best_val

"""Optimise a non‑negative step function for the evaluate_sequence objective.

The search proceeds in several stages within a 1000 s budget:

1. **Seed construction** – a large variety of analytic families (uniform,
   monotone power‑law, exponential, beta, low‑frequency cosine, random
   Dirichlet/sparse) and the previous best solution are generated and
   ℓ₁‑normalised.

2. **Fast evaluation** – each seed is scored by the exact objective
   `2·n·max(conv(a,a))/sum(a)^2`.  The best `top_k` seeds are kept.

3. **Continuation L‑BFGS** – each elite seed is refined with a smooth
   log‑sum‑exp surrogate of the max‑convolution.  A schedule of increasing
   smoothness parameters β is used (β from 1 up to 1e7), each stage limited
   to a few thousand L‑BFGS iterations.

4. **Adam mirror‑descent on the exact objective** – starting from the
   current global best, we optimise the exact max‑convolution using Adam on
   the logits of the probability vector (soft‑max parametrisation).  The
   sub‑gradient of the active lag is used; Adam’s first‑ and second‑moment
   adaptivity yields a faster descent than plain exponentiated‑gradient.

5. **Stochastic mass‑transfer local search** – a simple hill‑climbing step
   that moves a small fraction of mass between random coordinates, accepting
   only improving moves.

6. **Final high‑β polishing** – a last L‑BFGS run with very large β
   values (≈ 10⁶‒10⁷) to fine‑tune the solution.

The best sequence found is stored globally as ``height_sequence_1`` for
warm‑starts in future calls and is returned as a plain list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decrease(n: int) -> np.ndarray:
    """a_i ∝ (n-i)  (strictly decreasing)."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _quadratic_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64) ** 2
    return a / a.sum()


def _power_law(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-p} , p>0."""
    i = np.arange(1, n + 1, dtype=np.float64) + offset
    a = i ** (-p)
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) for continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate of the max‑convolution objective."""
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0,
        ]
    # initialise unconstrained logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        b = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, b)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam mirror‑descent on the exact objective
# ----------------------------------------------------------------------
def _adam_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.2,
) -> tuple[np.ndarray, float]:
    """Adam on logits for the exact max‑convolution objective."""
    a = start_a.copy()
    a = a / a.sum()
    v = np.log(a + 1e-30)               # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_val = _objective_exact(a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        a = _softmax(v)                 # always on the simplex
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))

        # sub‑gradient of max‑convolution at active lag
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g[i_range] = 2.0 * a[rev]

        # gradient w.r.t. logits
        grad_v = a * (g - np.dot(g, a))

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v * grad_v)

        m_hat = m / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v_hat_hat) + eps)

        if t % 10_000 == 0:
            cur_val = _objective_exact(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

    # final check
    cur_val = _objective_exact(a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive stochastic local search (mass transfers)
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 150_000,
    step_frac: float = 0.008,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Seed‑pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decrease(n))
    seeds.append(_quadratic_decrease(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.0004))

    # power‑law grid (exponent & optional offset)
    for p in np.linspace(0.20, 1.50, 27):
        for off in (0.0, 0.5, 1.0, 2.0, 5.0, 10.0):
            seeds.append(_power_law(n, float(p), offset=off))

    # exponential grid
    for lam in np.logspace(-5, -1, 30):
        seeds.append(_exponential_decay(n, float(lam)))

    # beta‑shape grid (skewed to the left)
    alphas = np.linspace(0.2, 1.5, 7)
    betas = np.linspace(1.0, 5.0, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # low‑frequency cosine curves
    for nf in (1, 2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random Dirichlet families
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(3):
            seeds.append(_random_sparse(n, frac, rng))

    # pairwise mixtures of existing seeds
    base = np.arange(len(seeds))
    for _ in range(80):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (warm‑start)
    if previous is not None and previous.shape == (n,) and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – non‑negativity + ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The best sequence is stored globally as ``height_sequence_1`` and also
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    n = 4000  # length of the step function
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # seed construction (including warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # quick exact evaluation of seeds, keep the elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    # keep the top‑k for the heavy L‑BFGS stage
    seeds.sort(key=_objective_exact)
    top_k = min(20, len(seeds))
    elite = seeds[:top_k]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0, 1e7,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break
        per_beta = 2500   # modest budget per β value
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam mirror‑descent on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 1e5))
        print("\n--- Adam optimisation (exact objective) ---")
        seq_adam, val_adam = _adam_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=max_iters,
            lr=0.2,
        )
        print(f"[Adam] best = {val_adam:.6f}")
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Update] New global best after Adam = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – aggressive stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(300_000, remaining * 80))
        if iters > 0:
            print("\n--- Stochastic local search (mass transfers) ---")
            seq_loc, val_loc = _local_search(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.006,
            )
            print(f"[LocalSearch] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New global best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1000, int(min(4000, remaining * 0.07 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5e6, 1e7, 2e7],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
