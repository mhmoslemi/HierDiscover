"""
Propose a candidate step‑function (list of non‑negative heights).

The algorithm proceeds in three stages:

1. **Broad seed generation** (size ≈ 2500) – a large collection of analytic
   families (uniform, triangular, sine windows, Gaussians, power‑law,
   beta‑shapes, exponential decays, low‑frequency cosine curves, Dirichlet /
   exponential draws) plus a few random permutations and mixtures.  All
   seeds are ℓ₁‑normalised.

2. **Multi‑start refinement** – for the best few seeds we run a short
   exponentiated‑gradient (mirror‑descent) optimisation on the exact
   ``max(conv)`` objective, then a continuation L‑BFGS optimiser on the
   smooth log‑sum‑exp surrogate (a schedule of β values).  The best local
   result updates the global optimum.

3. **Final polishing** – the globally‑best sequence is up‑sampled to the
   final length (4000), then refined with a long exponentiated‑gradient pass,
   a fine‑grained L‑BFGS run with very large β (≈ 10⁸) and finally a short
   hill‑climbing phase.  The result is evaluated with the provided
   ``evaluate_sequence`` function and stored in the global ``height_sequence_1``.

The routine respects a 1000 s time budget and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)

except Exception:  # pragma: no cover

# ----------------------------------------------------------------------
# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector with sum = 1:
        2·n·max(conv(seq, seq)).
    The returned value equals the *surrogate* bound that the optimisation
    routine minimises; the true objective is obtained by ``evaluate_sequence``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i), i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to respect positivity
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]                # factor 2 for symmetry

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Parameters are stored in log‑softmax space to guarantee feasibility.
    """
    if _opt is None:
        # SciPy not available – fall back to the input vector unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 10_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # numerical stabiliser
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximation of max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (see derivation in the original code)
            w = exp_beta / sum_exp                 # softmax weights over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)             # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # convert to gradient w.r.t. v (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "disp": False,
                                         "ftol": 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple hill‑climbing (random local perturbations)
# ----------------------------------------------------------------------
def _hill_climbing(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   max_attempts: int = 200_000,
                   sigma_start: float = 0.003,
                   sigma_decay: float = 0.9997) -> tuple[np.ndarray, float]:
    """
    Very lightweight random‑perturbation search. Returns the best sequence
    discovered before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    sigma = sigma_start
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s
        val = _objective_surrogate(ns)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns   # move current point
        sigma *= sigma_decay
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper to up‑sample a probability vector to a larger size
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation up‑sampling followed by ℓ₁‑normalisation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        return _uniform(new_n)
    return up / s


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` objective.  The routine respects the time budget
    and stores the best sequence it discovers in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Problem size – a modest size for the exploratory phase
    # ------------------------------------------------------------------
    base_n = 2500                         # seed generation & first refinement
    final_n = 4000                        # size required by the competition

    # ------------------------------------------------------------------
    # Load a previously found sequence (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (final_n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (base_n)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (base_n={base_n}) ---")
    seeds = []

    # Analytic families – coarse grids
    seeds.append(_uniform(base_n))
    seeds.append(_triangular(base_n))
    seeds.append(_sine_window(base_n, gamma=1.0))

    # Parameter sweeps
    for gamma in np.linspace(0.2, 4.0, 24):
        seeds.append(_sine_window(base_n, gamma))
    for sigma_frac in np.linspace(0.04, 0.30, 24):
        seeds.append(_gaussian(base_n, sigma_frac))
    for p in np.linspace(0.0, 5.0, 25):
        seeds.append(_power(base_n, p))
    for rate in np.logspace(-4, -0.4, 15):
        seeds.append(_exp_decay(base_n, rate))
        seeds.append(_sym_exp_decay(base_n, rate))

    # Beta‐shape random sampling
    for _ in range(40):
        a = rng.uniform(0.05, 5.0)
        b = rng.uniform(0.05, 5.0)
        seeds.append(_beta_shape(base_n, a, b))

    # Random Dirichlet / exponential draws
    for conc in (0.01, 0.03, 0.07, 0.15, 0.4, 1.0, 2.5):
        for _ in range(3):
            seeds.append(_random_dirichlet(base_n, conc, rng))
    for _ in range(20):
        seeds.append(_random_exponential(base_n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(6):
            seeds.append(_lowfreq_curve(base_n, nf, rng))

    # Random permutations of nice bases
    bases = [_gaussian(base_n, 0.18), _sine_window(base_n, 1.2),
             _beta_shape(base_n, 2.0, 2.0)]
    for base in bases:
        for _ in range(4):
            seeds.append(np.random.permutation(base))

    # Mixtures with uniform
    for mix_alpha in (0.15, 0.35, 0.65):
        for nf in (2, 4):
            curve = _lowfreq_curve(base_n, nf, rng)
            seeds.append(_mix_shapes(_uniform(base_n), curve, mix_alpha))

    # Perturbations of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        # down‑sample the previous 4000‑vector to base_n for a fair start
        prev_small = _upsample(prev_arr, base_n)
        seeds.append(prev_small)
        for _ in range(6):
            seeds.append(_perturb(prev_small, rng, sigma=0.004))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values on base_n):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best (still on base_n)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement on the top seeds (still base_n)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget for this seed
        rem = deadline - time.time()
        # keep ~12 % of the remaining budget per remaining seed
        sub_deadline = time.time() + rem * 0.12 / max(1, (top_k - si))

        # ----- (a) medium‑length exponentiated‑gradient -----
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           base_n,
                                           sub_deadline,
                                           max_iters=150_000,
                                           eta0=6.0)

        # ----- (b) continuation L‑BFGS smoothing -----
        if time.time() < deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               base_n,
                                               deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1_000.0,
                                                              2_000.0, 5_000.0,
                                                              20_000.0, 50_000.0,
                                                              100_000.0, 200_000.0,
                                                              500_000.0, 1_000_000.0],
                                               max_iter_per_beta=1500)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # ----- (c) short hill‑climbing -----
        hill_deadline = deadline - 0.5  # reserve a tiny margin for final polish
        if time.time() < hill_deadline:
            seq_hc, val_hc = _hill_climbing(seq_lbfgs,
                                            base_n,
                                            hill_deadline,
                                            rng,
                                            max_attempts=120_000,
                                            sigma_start=0.004,
                                            sigma_decay=0.9996)
        else:
            seq_hc, val_hc = seq_lbfgs, val_lbfgs

        # Choose the best of the local candidates
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_hc, val_hc)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best (base_n) = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Upsample to final_n and a final polish
    # ------------------------------------------------------------------
    print("\n--- Upsampling to final size and final polishing ---")
    global_best_seq = _upsample(global_best_seq, final_n)

    # Final long EG polishing
    if time.time() < deadline:
        final_eg_deadline = deadline - 30.0   # keep some room for L‑BFGS & HC
        final_seq_eg, final_val_eg = _exp_gradient_opt(global_best_seq,
                                                       final_n,
                                                       final_eg_deadline,
                                                       max_iters=2_500_000,
                                                       eta0=5.0)
    else:
        final_seq_eg, final_val_eg = global_best_seq, _objective_surrogate(global_best_seq)

    # Fine L‑BFGS with very large β (smooth max → hard max)
    if time.time() < deadline:
        final_lbfgs_deadline = deadline - 10.0
        final_seq_lbfgs = _lbfgs_smooth_optimize(final_seq_eg,
                                                 final_n,
                                                 final_lbfgs_deadline,
                                                 beta_schedule=[
                                                     1e6, 2e6, 5e6, 1e7,
                                                     2e7, 5e7, 1e8, 2e8,
                                                     5e8, 1e9
                                                 ],
                                                 max_iter_per_beta=2500)
        final_val_lbfgs = _objective_surrogate(final_seq_lbfgs)
    else:
        final_seq_lbfgs, final_val_lbfgs = final_seq_eg, final_val_eg

    # Final hill‑climbing (very fine perturbations)
    if time.time() < deadline:
        final_hc_deadline = deadline - 1.0
        final_seq_hc, final_val_hc = _hill_climbing(final_seq_lbfgs,
                                                    final_n,
                                                    final_hc_deadline,
                                                    rng,
                                                    max_attempts=250_000,
                                                    sigma_start=0.001,
                                                    sigma_decay=0.9999)
    else:
        final_seq_hc, final_val_hc = final_seq_lbfgs, final_val_lbfgs

    # Pick overall best among the final candidates
    final_candidates = [
        (global_best_seq, _objective_surrogate(global_best_seq)),
        (final_seq_eg, final_val_eg),
        (final_seq_lbfgs, final_val_lbfgs),
        (final_seq_hc, final_val_hc)
    ]
    best_seq, best_surrogate = min(final_candidates, key=lambda x: x[1])

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
