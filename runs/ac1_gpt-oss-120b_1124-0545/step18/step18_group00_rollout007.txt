"""
search_step_function.py
-----------------------

A lightweight alternative to the previous search routine.  It builds a rich
library of analytic “window’’ families, augments them with a large set of
random convex mixtures of the best seeds, and then performs a two‑stage
refinement:

1. **Log‑softmax L‑BFGS smoothing** – optimises a smooth approximation of
   the max‑convolution (log‑sum‑exp) with a warm‑up schedule of β values.
2. **Projected sub‑gradient descent** – works directly on the true
   non‑smooth objective using a Euclidean projection onto the probability
   simplex.

The code respects a global time budget (default 1000 s) and stores the
best sequence it has found in the global variable ``height_sequence_1`` for
future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax; returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic window families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0 - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0 from scipy; fallback to 1 if unavailable
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector and return it ℓ₁‑normalised (already normal)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mixing utilities
# ----------------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    return mixed if s <= 0.0 else mixed / s


def _random_convex_mix(seeds: list[np.ndarray],
                       rng: np.random.Generator,
                       budget_deadline: float,
                       max_samples: int = 5000) -> tuple[np.ndarray, float]:
    """
    Draw random convex combinations of the supplied *seeds*.
    Returns the best mixture found (vector, surrogate value).
    """
    k = len(seeds)
    if k == 0:
        raise ValueError("no seeds to mix")
    seeds_mat = np.vstack(seeds)                     # shape (k, n)
    best_seq = None
    best_val = float("inf")
    # Dirichlet concentration: small values give sparse mixes,
    # moderate values give balanced mixes.
    alpha_vec = np.full(k, 0.5, dtype=np.float64)

    for i in range(max_samples):
        if time.time() > budget_deadline:
            break
        w = rng.dirichlet(alpha_vec)
        mixed = w @ seeds_mat                       # vector of length n
        val = _objective_surrogate(mixed)
        if val < best_val:
            best_val = val
            best_seq = mixed.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# L‑BFGS smoothing optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1000000.0]

    # Work in log‑softmax space.
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # log‑sum‑exp smoothing
            shift = max_c                # numeric stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (softmax space)
            w = exp_beta / sum_exp                     # shape (2n‑1,)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # slice n entries
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)               # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (exact max)
# ----------------------------------------------------------------------
def _proj_subgrad_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 20000,
                     eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) and stores it globally as
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a 2‑second safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size (fixed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed creation (analytic + random)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic families (moderate grid)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.2, 3.0, 8):
        seeds.append(_sine_window(n, gamma=gamma))

    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    for beta in np.linspace(2.0, 15.0, 7):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.09, 0.25, 8):
        seeds.append(_gaussian(n, sigma))

    for sigma in (0.10, 0.14):
        for shift in (0.20, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.5, 4.0, 5)
    betas = np.linspace(0.5, 4.0, 5)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    for alpha in np.linspace(0.1, 0.9, 5):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.8, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Alternating Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # Previous best (if available)
    if prev_arr is not None:
        seeds.append(prev_arr)                     # already normalised

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:02d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Random convex mixtures of the best seeds
    # ------------------------------------------------------------------
    mix_deadline = deadline - 5.0  # leave room for refinements
    print("\n--- Random convex mixtures ---")
    mixed_seq, mixed_val = _random_convex_mix(top_seeds, rng, mix_deadline,
                                             max_samples=4000)
    if mixed_seq is not None:
        evaluated.append((mixed_seq, mixed_val))

    # Re‑sort after mixing
    evaluated.sort(key=lambda x: x[1])
    # Choose a handful of promising candidates for heavy polishing
    polish_candidates = evaluated[:5]   # (vec, surrogate)

    print("\nCandidates selected for refinement:")
    for i, (_, v) in enumerate(polish_candidates):
        print(f"  Candidate {i+1}: surrogate {v:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement stage (L‑BFGS smoothing + exact sub‑gradient)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    for idx, (cand_vec, _) in enumerate(polish_candidates):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + (deadline - time.time()) * 0.45 / (len(polish_candidates) - idx)

        print(f"\n[Refine #{idx+1}] L‑BFGS smoothing start")
        seq_lbfgs = _lbfgs_smooth_optimize(cand_vec, n, sub_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1000.0,
                                                          2000.0, 5000.0,
                                                          20000.0, 50000.0,
                                                          100000.0, 200000.0,
                                                          500000.0, 1000000.0],
                                           max_iter_per_beta=3000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[Refine #{idx+1}] after L‑BFGS surrogate = {val_lbfgs:.6f}")

        # Fine‑tune with exact projected sub‑gradient descent
        if time.time() < sub_deadline:
            print(f"[Refine #{idx+1}] Projected sub‑gradient start")
            seq_pg, val_pg = _proj_subgrad_opt(seq_lbfgs, n, sub_deadline,
                                               max_iters=15000, eta0=0.8)
            print(f"[Refine #{idx+1}] after PGD surrogate = {val_pg:.6f}")
        else:
            seq_pg, val_pg = seq_lbfgs, val_lbfgs

        # Determine the best of the three (raw, L‑BFGS, PGD)
        candidates = [(cand_vec, _objective_surrogate(cand_vec)),
                      (seq_lbfgs, val_lbfgs),
                      (seq_pg, val_pg)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < best_val:
            best_val = local_best_val
            best_seq = local_best_seq.copy()
            print(f"  >>> Updated global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polish (extra PGD passes if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Final projected sub‑gradient polish ---")
        # allocate remaining time proportionally
        final_deadline = time.time() + remaining * 0.9
        seq_final, val_final = _proj_subgrad_opt(best_seq, n, final_deadline,
                                                 max_iters=20000, eta0=0.5)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final polish] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation with the provided routine
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for subsequent runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
