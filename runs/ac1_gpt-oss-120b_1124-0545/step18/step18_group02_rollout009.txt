"""
Propose a step‑function that minimises ``evaluate_sequence``.
----------------------------------------------------------------
This version builds on the previous “three‑stage’’ optimiser but makes
several changes that tend to push the objective a little lower:

* a richer seed pool (more power‑law exponents, larger shifts, extra
  low‑frequency curves);
* a more aggressive continuation L‑BFGS stage (larger β‑schedule,
  more iterations per β);
* longer exponentiated‑gradient (mirror‑descent) runs;
* an additional fine‑tuning stage that runs Adam on the smooth surrogate
  with a very high β;
* a final stochastic local‑search phase with a decreasing step size.

All stages respect a global 1000 s wall‑time limit, continually
updating a global variable ``height_sequence_1`` with the best sequence
found so far.
"""
import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradients
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Returns a vector g such that g_i = 2·∑_{j: i+j = k*} a_j,
    where k* is an index attaining the maximum.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)

    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS / Adam
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of 2·n·max(conv) / (∑a)².
    Operates on unconstrained logits v; returns (obj, grad_v).
    """
    a = _softmax(v)                     # ℓ₁‑normalised (sum = 1)
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0, 50_000_000.0, 100_000_000.0,
            500_000_000.0, 1_000_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-13},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


def _adam_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta: float = 1e9,
    lr: float = 0.1,
    max_iters: int = 600_000,
) -> np.ndarray:
    """
    Simple Adam optimiser on the smooth surrogate (fixed β).
    Returns an ℓ₁‑normalised vector.
    """
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        obj, grad = _smooth_objective(v, n, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * (grad ** 2)
        m_corr = m / (1 - beta1 ** t)
        v_corr = v_hat / (1 - beta2 ** t)
        v -= lr * m_corr / (np.sqrt(v_corr) + eps)

        # occasional progress check
        if t % 100_000 == 0:
            cur = _objective_exact(_softmax(v) * np.sqrt(2.0 * n))
            print(f"[Adam] iter {t:,}  obj≈{cur:.6f}")

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[EG] iter {it:,}  new best {best_val:.6f}")

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic mass‑transfer local search
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 150_000,
    step_frac: float = 0.012,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Seed‑pool construction (augmented with richer families)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # classic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.30))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # richer power‑law sweep
    p_vals = np.linspace(0.25, 0.90, 14)          # more exponents
    shifts = [0.0, 5.0, 20.0, 50.0, 120.0, 250.0]  # larger shifts
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])  # mirrored

    # exponential families with diverse rates
    for lam in (0.0003, 0.001, 0.003, 0.01):
        seeds.append(_exponential_decay(n, lam))

    # random Dirichlet draws (including very sparse)
    for conc in (0.02, 0.05, 0.15, 0.4, 0.8):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse vectors
    for frac in (0.02, 0.05, 0.1, 0.25):
        for _ in range(6):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures of existing seeds
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of a few promising seeds
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # problem size – a larger *n* gives more flexibility, 4000 stays fast
    n = 4000
    target_sum = np.sqrt(2.0 * n)

    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Collect seeds (including warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Evaluate seeds – keep the elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v * target_sum)   # ℓ₁‑mass = sqrt(2n)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective ≈ {best_val:.6f}")

    # store top‑k seeds for heavy L‑BFGS stage
    seeds.sort(key=_objective_exact)
    elite = seeds[:min(30, len(seeds))]

    global_best_seq = best_seq.copy() * target_sum
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS (smooth surrogate)
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 50_000_000.0, 100_000_000.0,
        500_000_000.0, 1_000_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break

        seed_vec = seed_vec * target_sum
        per_beta = max(1000, int(min(4000, (deadline - time.time()) * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β≈{per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient restarts (mirror descent)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 5
        iters_per_restart = int(min(600_000, remaining * 0.13 / restarts))
        print("\n--- EG restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.02, 1.0)
            start_vec = _random_dirichlet(n, conc, rng) * target_sum
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.2,
            )
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long EG run from current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(8_000_000, remaining * 0.2 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=2.5,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Adam fine‑tuning on smooth surrogate (very high β)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(700_000, remaining * 0.12 * 1e5))
        if max_iters > 0:
            print("\n--- Adam fine‑tuning (β≈1e9) ---")
            seq_adam = _adam_smooth_optimize(
                global_best_seq,
                n,
                deadline,
                beta=1e9,
                lr=0.07,
                max_iters=max_iters,
            )
            seq_adam = seq_adam * target_sum  # back to correct ℓ₁‑mass
            val_adam = _objective_exact(seq_adam)
            print(f"[Adam] objective = {val_adam:.6f}")
            if val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Update] New best after Adam = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – aggressive stochastic local search (decreasing step)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(200_000, remaining * 45))
        if iters > 0:
            print("\n--- Stochastic local search (mass transfers) ---")
            step = 0.05
            a = global_best_seq.copy()
            best_val = global_best_val
            best_a = a.copy()
            rng = np.random.default_rng(314159)
            for it in range(iters):
                if time.time() > deadline:
                    break
                # gradually shrink step size
                if it % 20_000 == 0 and it > 0:
                    step *= 0.5
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] <= 0.0:
                    continue
                delta = step * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                val = _objective_exact(cand)
                if val < best_val:
                    best_val = val
                    best_a = cand.copy()
                    a = cand
            if best_val < global_best_val:
                global_best_seq = best_a.copy()
                global_best_val = best_val
                print(f"[Update] New best after stochastic local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – final polishing L‑BFGS (very high β, few iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(2500, remaining * 0.04 * 1e5)))
        print("\n--- Final polishing L‑BFGS (high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000.0, 1_000_000_000.0, 5_000_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (using the provided evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
