def _gradient_descent_optimize(a0, n, deadline, beta_schedule, steps_per_beta=2000, init_step=0.5):

   v = np.log(a0 + 1e-30)

   best_a = a0

   best_val = _objective_surrogate(best_a)

   for beta in beta_schedule:

        for iter in range(steps_per_beta):

            if time.time() > deadline: break

            a = _softmax(v)

            conv = _conv_fft(a, a)

            M = np.max(conv)

            # compute smooth objective
            shift = M
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = exp_term.sum()
            smooth_val = shift + np.log(sum_exp) / beta

            # gradient_core
            w = exp_term / sum_exp

            rev_a = a[::-1]

            grad_a = 2.0 * full[n-1:n-1+n]   # shape n

def _hill_climb(seq, n, deadline, rng):
    cur = seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val
    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j: continue
        # compute fraction 0 to cur[i]
        eps_frac = rng.uniform(0.001, 0.05)  # up to 5% of source mass
        eps = cur[i] * eps_frac
        if eps == 0: continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val

"""
Propose a non‑negative step‑function (list of heights) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm works in three phases:

1. **Seeding** – generate a diverse set of normalised vectors using
   analytical windows (uniform, arcsine, two‑block etc.) and many random
   Dirichlet draws.  The best ``top_k`` candidates (according to the exact
   surrogate 2·n·max(conv)) are retained.

2. **Smooth max optimisation** – each retained seed is refined by a fast
   gradient‑descent optimiser that works on the log‑softmax representation.
   The optimiser minimises a smooth approximation of the max‑convolution
   (log‑sum‑exp) with an annealing schedule of the smoothing parameter β.
   This yields a much tighter minimum than the previous L‑BFGS approach
   while allowing thousands of iterations within the time budget.

3. **Final greedy hill‑climb** – a cheap mass‑transfer hill‑climb is run
   on the best vector found to capture any remaining local improvements.

The routine respects the supplied time budget (default 1000 s), stores the
best sequence in the global variable ``height_sequence_1`` for warm‑starts,
and finally returns the best list of heights.
"""

import time
import heapq
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (zero‑padded)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Exact surrogate (the quantity to be minimised when sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑ vec = 1):
        2·n·max(conv(vec, vec))
    """
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine_window(n: int) -> np.ndarray:
    """U‑shaped arcsine‑like density (≈1/√(x·(1‑x)))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    """
    Two uniform blocks placed at the left and right ends.
    block_ratio – fraction of the total length occupied by each block.
    left_weight – total mass placed on the left block (right gets 1‑left).
    """
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """|cos|‑shaped – large at both ends, small in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with all concentration parameters = conc."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


# ----------------------------------------------------------------------
# Smooth max optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_max_optimize(start_vec: np.ndarray,
                         n: int,
                         deadline: float,
                         beta_schedule=None,
                         max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Gradient descent on the log‑sum‑exp approximation of max(conv).

    The optimisation works in the log‑softmax variables ``v`` which
    automatically satisfy the simplex constraints after the softmax.
    """
    if beta_schedule is None:
        beta_schedule = [5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 10000.0,
                         20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    best_vec = start_vec.copy()
    best_val = _objective_surrogate(best_vec)

    for beta in beta_schedule:
        # step size scales down with sqrt(beta) – works well in practice
        step = 0.5 / max(1.0, np.sqrt(beta))

        for it in range(max_iter_per_beta):
            if time.time() > deadline:
                return best_vec

            a = _softmax(v)                     # current probability vector
            conv = _conv_fft(a, a)              # full linear convolution
            M = np.max(conv)                    # exact max for the surrogate

            # --- smooth max via log‑sum‑exp ---------------------------------
            shift = M
            exp_beta = np.exp(beta * (conv - shift))   # ≤ 1, max term = 1
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                       # soft‑max weights

            # --- gradient wrt a --------------------------------------------
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # --- convert to gradient wrt v (log‑softmax variables) ---------
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # --- simple gradient step ---------------------------------------
            v = v - step * grad_v

            # occasional exact evaluation to keep the best candidate
            if (it & 127) == 0:          # every 128 iterations
                a_test = _softmax(v)
                val = _objective_surrogate(a_test)
                if val < best_val:
                    best_val = val
                    best_vec = a_test.copy()

    return best_vec


# ----------------------------------------------------------------------
# Greedy hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny random mass transfers; accepts only improvements.
    Returns the best vector seen and its exact surrogate value.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # move a random fraction (0.1 % – 5 %) of the source mass
        eps = cur[i] * rng.uniform(0.001, 0.05)
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The function respects the allotted time budget, prints progress,
    and stores the best vector in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final call to ``evaluate_sequence``
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # problem size – larger n gives a tighter theoretical bound but costs
    # more per iteration; 4000 is a good compromise.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # keep the best *top_k* seeds (by exact surrogate)
    # ------------------------------------------------------------------
    top_k = 15
    best_heap = []                     # stores (-value, vector)

    def _push(vec: np.ndarray):
        """Insert a candidate into the heap while keeping only the best top_k."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # deterministic analytical seeds (low‑dimensional families)
    # ------------------------------------------------------------------
    _push(_uniform(n))
    _push(_arcsine_window(n))
    _push(_cosine_edge_window(n, gamma=0.8))
    _push(_cosine_edge_window(n, gamma=2.0))
    _push(_two_block_uniform(n, block_ratio=0.15, left_weight=0.6))
    _push(_two_block_uniform(n, block_ratio=0.30, left_weight=0.4))

    # ------------------------------------------------------------------
    # massive random seeding (Dirichlet with a range of concentrations)
    # ------------------------------------------------------------------
    conc_options = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_budget = 3000
    for _ in range(random_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_options)
        vec = _random_dirichlet(n, conc, rng)
        # occasional cyclic shift – changes the correlation pattern
        if rng.random() < 0.25:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push(vec)

    # ------------------------------------------------------------------
    # reuse previous solution if appropriate
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape[0] == n and prev_arr.sum() > 0.0:
                _push(_normalize(prev_arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] retained {len(seeds)} candidates, best surrogate = "
          f"{_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # initialise global best
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # local refinements – smooth‑max gradient descent on the top seeds
    # ------------------------------------------------------------------
    max_refinements = min(6, len(seeds))
    for idx in range(max_refinements):
        if time.time() > deadline - 5.0:
            break
        seed_vec = seeds[idx]
        # allocate a slice of the remaining time to this refinement
        remaining = deadline - time.time()
        # reserve a small portion for later hill‑climb
        per_deadline = time.time() + max(5.0, remaining * 0.6 / max_refinements)

        refined = _smooth_max_optimize(seed_vec, n, per_deadline)
        refined_val = _objective_surrogate(refined)

        if refined_val < global_best_val:
            global_best_seq = refined
            global_best_val = refined_val
            print(f"[Refine {idx+1}] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # final greedy hill‑climb (uses any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final hill‑climb ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # evaluate with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
