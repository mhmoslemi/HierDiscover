def _anneal_swap_and_mass_opt(start_seq,n,deadline,max_swaps):
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    T0 = 0.1 * cur_val if cur_val>0 else 1.0
    for it in range(1,max_swaps+1):
        if time.time() > deadline:
            break
        if rng.random() < 0.7: # do swap
            i,j = rng.integers(0,n,size=2)
            if i==j: continue
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            new_val = _objective(cur_seq)
            delta = new_val - cur_val
            T = max(T0 * (1.0 - it / max_swaps), 1e-12)
            accept = delta < 0 or rng.random() < math.exp(-delta / T)
            if accept:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = cur_seq.copy()
            else:
                # revert
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        else: # mass transfer
            i,j = rng.integers(0,n,size=2)
            if i==j: continue
            # transfer small epsilon from i to j
            if cur_seq[i] < 1e-12: continue
            epsilon = rng.random() * cur_seq[i] * 0.1  # up to 10% of mass at i
            # Move epsilon from i to j
            cur_seq[i] -= epsilon
            cur_seq[j] += epsilon
            new_val = _objective(cur_seq)
            delta = new_val - cur_val
            T = max(T0 * (1.0 - it / max_swaps), 1e-12)
            accept = delta < 0 or rng.random() < math.exp(-delta / T)
            if accept:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = cur_seq.copy()
            else:
                # revert transfer
                cur_seq[i] += epsilon
                cur_seq[j] -= epsilon
        if it % 100_000 == 0:
            print(f"[Anneal] it={it:,} best={best_val:.6f}")
    return best_seq, best_val

   and store it in ``height_sequence_1`` for the next run.

All time‑critical loops frequently check the deadline and abort
gracefully, guaranteeing a result before the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n} (discrete version)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collector – much richer than the previous version
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a large, diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # ------- Analytic families -------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    for beta in (2, 4, 8, 12, 16, 20):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))

    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))

    # ------- Beta‑shape grid (including arcsine) -------
    for a in np.linspace(0.1, 2.0, 20):
        for b in np.linspace(0.1, 2.0, 20):
            seeds.append(_beta_shape(n, float(a), float(b)))
    # explicit arcsine and a few classic exponents
    for a,b in ((0.5,0.5), (0.3,0.3), (0.2,0.2), (0.1,0.1)):
        seeds.append(_beta_shape(n, a, b))

    # ------- Tukey window sweep -------
    for alpha in np.linspace(0.05, 0.95, 19):
        seeds.append(_tukey_window(n, alpha))

    # ------- Random Dirichlet draws (wide conc range) -------
    for conc in (0.01, 0.03, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ------- Random exponential draws -------
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # ------- Low‑frequency cosine curves -------
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # ------- Sparse random vectors -------
    for k in (max(1, int(0.005 * n)), max(1, int(0.02 * n))):
        for _ in range(3):
            seeds.append(_sparse_random(n, k, rng))

    # ------- Block‑uniform vectors -------
    for bf in (0.025, 0.075, 0.15, 0.30):
        for _ in range(3):
            seeds.append(_block_uniform(n, bf, rng))

    # ------- Alternating Dirichlet -------
    seeds.append(_alternating_dirichlet(n, rng))

    # ------- Permuted heavy‑tailed Dirichlet -------
    heavy = _random_dirichlet(n, 0.01, rng)
    for _ in range(4):
        seeds.append(heavy[rng.permutation(n)])

    # ------- Mix of previous solution (if any) -------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity‑check – every vector ends up ℓ₁‑normalised.
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
_DEFAULT_BETA_SCHEDULE = [
    1.0, 5.0, 20.0, 80.0, 200.0,
    500.0, 1e3, 2e3, 5e3,
    1e4, 2e4, 5e4, 1e5,
    2e5, 5e5, 1e6, 2e6, 5e6, 1e7
]


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Optimise a log‑sum‑exp smoothing of max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = _DEFAULT_BETA_SCHEDULE

    # work in log‑probability space – softmax guarantees the simplex constraint
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (derived from the smoothed max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000,
                     eta_factor: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_objective_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # Sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Learning rate – classic 1/√t schedule with tunable factor
        eta = eta_factor / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress output (kept light to avoid overhead)
        if it % 200_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing swap + mass‑transfer optimiser
# ----------------------------------------------------------------------
def _anneal_swap_and_mass_opt(start_seq: np.ndarray,
                              n: int,
                              deadline: float,
                              max_swaps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Random swaps and small mass transfers with Metropolis acceptance.
    Returns the best sequence encountered and its objective value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # Initial temperature – a small fraction of the current objective
    T0 = 0.1 * cur_val if cur_val > 0 else 1.0

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        # With probability 0.7 perform a swap, otherwise a tiny mass transfer
        if rng.random() < 0.7:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            new_val = _objective(cur_seq)
            delta = new_val - cur_val
            T = max(T0 * (1.0 - it / max_swaps), 1e-12)
            accept = (delta < 0.0) or (rng.random() < math.exp(-delta / T))
            if accept:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = cur_seq.copy()
            else:
                # revert
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        else:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_seq[i] < 1e-12:
                continue
            eps = rng.random() * cur_seq[i] * 0.1   # move ≤10 % of mass at i
            cur_seq[i] -= eps
            cur_seq[j] += eps
            new_val = _objective(cur_seq)
            delta = new_val - cur_val
            T = max(T0 * (1.0 - it / max_swaps), 1e-12)
            accept = (delta < 0.0) or (rng.random() < math.exp(-delta / T))
            if accept:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = cur_seq.copy()
            else:
                # revert transfer
                cur_seq[i] += eps
                cur_seq[j] -= eps

        if it % 100_000 == 0:
            print(f"[Anneal] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Misc helper – symmetrisation
# ----------------------------------------------------------------------
def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Average a vector with its reversal and re‑normalise onto the simplex."""
    mixed = (seq + seq[::-1]) * 0.5
    s = mixed.sum()
    return mixed / s if s > 0 else seq.copy()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under a 1000 s budget.

    Returns the best sequence (list of floats) found.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Seed generation (reuse previous solution if available)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list) and len(previous) == n:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 3. Quick evaluation of all seeds – keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed objective values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. L‑BFGS continuation on the very best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # Allocate roughly 30 % of remaining time for L‑BFGS
    remaining = deadline - time.time()
    if remaining > 20 and _opt is not None:
        lbfgs_budget = remaining * 0.35
        per_seed_budget = lbfgs_budget / min(5, len(top_seeds))
        max_iter_per_beta = max(4000, int(per_seed_budget * 1200))

        print("\n--- L‑BFGS optimisation (top seeds) ---")
        for idx, seed_vec in enumerate(top_seeds[:5]):
            if time.time() > deadline:
                break
            print(f"  L‑BFGS on seed #{idx+1}")
            try:
                seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                                 n,
                                                 deadline,
                                                 beta_schedule=_DEFAULT_BETA_SCHEDULE,
                                                 max_iter_per_beta=max_iter_per_beta)
                val_opt = _objective(seq_opt)
                if val_opt < global_best_val:
                    global_best_seq = seq_opt.copy()
                    global_best_val = val_opt
                    print(f"    Improved to {global_best_val:.6f}")
            except Exception as e:  # pragma: no cover
                print(f"    L‑BFGS failed: {e}")

        # optional mixing of the two best seeds
        if len(top_seeds) >= 2 and time.time() < deadline:
            mixed = _mix_shapes(top_seeds[0], top_seeds[1], 0.5)
            print("  L‑BFGS on mixed seed")
            seq_mix = _lbfgs_smooth_optimize(mixed,
                                             n,
                                             deadline,
                                             beta_schedule=_DEFAULT_BETA_SCHEDULE,
                                             max_iter_per_beta=int(max_iter_per_beta * 0.8))
            val_mix = _objective(seq_mix)
            if val_mix < global_best_val:
                global_best_seq = seq_mix.copy()
                global_best_val = val_mix
                print(f"    Mixed seed improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient polishing (exact max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10:
        # aim for ~1500 iterations per second (empirical)
        max_iters = int(min(2_000_000, remaining * 1500))
        print(f"\n--- Exponentiated‑gradient polishing ({max_iters:,} iters) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_iters,
                                             eta_factor=0.7)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"    ExpGrad improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation (averaging with reversal)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sym = _symmetrize(global_best_seq)
        val_sym = _objective(seq_sym)
        if val_sym < global_best_val:
            global_best_seq = seq_sym
            global_best_val = val_sym
            print(f"\n[Symmetrize] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Simulated‑annealing (swap + mass‑transfer) fine‑tuning
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5:
        max_swaps = int(min(800_000, remaining * 300))
        print(f"\n--- Simulated‑annealing fine‑tuning ({max_swaps:,} moves) ---")
        seq_perm, val_perm = _anneal_swap_and_mass_opt(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_swaps=max_swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"    Anneal improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
