def grad_max_conv(a, idx):
    n = len(a)
    grad = np.zeros_like(a, dtype=np.float64)
    # compute overlapping region
    i_start = max(0, idx - (n - 1))
    i_end = min(idx, n - 1)  # inclusive
    # range i_start..i_end inclusive
    # with j = idx - i
    # assign grad[i] = 2 * a[j]
    # we can vectorize: i = np.arange(i_start, i_end+1); j = idx - i
    i = np.arange(i_start, i_end+1, dtype=int)
    j = idx - i
    grad[i] = 2.0 * a[j]
    return grad

"""
search for a step‑function (non‑negative heights) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The algorithm
----------------
* **Problem size** – we use a power‑of‑two length (4096) so that the FFT
  convolution is fast.
* **Seeding** – a large, diverse collection of analytical families
  (uniform, triangular, various windows, Beta‑shapes, power‑laws, etc.)
  and a massive random Dirichlet / exponential pool are generated.
  The best ``top_k`` (≈30) candidates according to the exact surrogate
  ``2·n·max(conv)`` (sum is normalised to 1) are kept in a heap.
* **Local optimisation** – each top seed is refined with a continuation
  L‑BFGS‑B optimiser that minimises a smooth log‑sum‑exp surrogate of the
  max‑convolution.  For robustness we run a few random perturbations of
  each seed (projected back onto the simplex) and keep the best result.
* **Greedy hill‑climb** – with any remaining time a cheap mass‑transfer
  hill‑climber is applied: a tiny fraction of mass is moved between two
  coordinates and only improvements are kept.
* **Final evaluation** – the best vector found is passed to
  ``evaluate_sequence`` and stored in the global variable
  ``height_sequence_1`` for the next round.

The routine respects the supplied time budget (default 1000 s) and prints
progress information.  It returns a plain ``list[float]`` suitable for the
evaluation harness.
"""

# ----------------------------------------------------------------------
# FFT‑based linear convolution (zero‑padded)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Exact surrogate (exact objective for normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑ vec = 1):
        2·n·max(conv(vec, vec))
    """
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _double_power(n: int, p: float) -> np.ndarray:
    """Symmetric sum of two power laws – high at both ends, low in the centre."""
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return _normalize(a)
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return _normalize(a)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return _normalize(a)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return _normalize(a)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    return _normalize(a)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0
    return _normalize(a)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _arcsine_window(n: int) -> np.ndarray:
    """U‑shaped arcsine‑like density (≈1/√(x·(1‑x)))"""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """|cos|‑shaped – large at both ends, small in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    """
    Two uniform blocks placed at the left and right ends.
    block_ratio – fraction of the total length occupied by each block.
    left_weight – total mass placed on the left block (right gets 1‑left).
    """
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with all concentration parameters = conc."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, cut to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose a few disjoint blocks and put uniform mass on each."""
    nb = rng.integers(2, 6)                 # number of blocks
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for _ in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The optimisation runs in log‑softmax space (vector v),
    guaranteeing a feasible probability vector after the softmax.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        # a longer schedule that pushes the smooth approximation very close
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1000.0, 2000.0, 5000.0, 10000.0,
            20000.0, 50000.0, 1e5, 2e5, 5e5,
            1e6, 2e6, 5e6
        ]

    # initialise logits (log‑softmax variables)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp (shifted for stability)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            w = exp_beta / sum_exp            # soft‑max weights over conv entries
            full = _conv_fft(rev_a, w)        # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer between two coordinates)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: move a random small fraction of mass from i to j.
    Returns the best vector found and its surrogate value.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # choose a fraction of the source mass (0.1% … 2%)
        eps = cur[i] * rng.uniform(0.001, 0.02)
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget, prints
    progress, and stores the best vector in the global variable
    ``height_sequence_1`` for the next call.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final ``evaluate_sequence`` call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a problem size
    # ------------------------------------------------------------------
    # Power‑of‑two length is cheap for FFT; 4096 gives about the same
    # discrimination power as the previous 4000 but is a little faster.
    n = 4096

    # ------------------------------------------------------------------
    # Keep a heap with the best ``top_k`` seeds (by exact surrogate)
    # ------------------------------------------------------------------
    top_k = 30
    best_heap = []          # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert a candidate into the heap while keeping only the best ``top_k``."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # 1. Deterministic analytical seeds (wide variety)
    # ------------------------------------------------------------------
    print("[Seeding] generating deterministic families …")
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        _push_candidate(_sine_window(n, gamma))
    _push_candidate(_hamming_window(n))
    _push_candidate(_blackman_window(n))
    _push_candidate(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 10):
        _push_candidate(_kaiser_window(n, beta))
    for sigma in np.linspace(0.08, 0.30, 15):
        _push_candidate(_gaussian(n, sigma))
    for p in np.linspace(0.05, 3.0, 20):
        _push_candidate(_power(n, p))
        _push_candidate(_double_power(n, p))
    # Beta‑shape – sweep log‑space heavily
    for _ in range(80):
        a = 10 ** rng.uniform(-3, 0)          # 0.001 … 1
        b = 10 ** rng.uniform(-3, 0)
        _push_candidate(_beta_shape(n, a, b))
    # arcsine and cosine‑edge windows (U‑shaped)
    _push_candidate(_arcsine_window(n))
    for gamma in np.linspace(0.5, 4.0, 12):
        _push_candidate(_cosine_edge_window(n, gamma))
    # two‑block uniform shapes
    for _ in range(25):
        blk = rng.uniform(0.05, 0.35)
        lw = rng.random()
        _push_candidate(_two_block_uniform(n, block_ratio=blk, left_weight=lw))

    # ------------------------------------------------------------------
    # 2. Massive random seeding
    # ------------------------------------------------------------------
    print("[Seeding] generating random Dirichlet / exponential candidates …")
    conc_choices = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_seed_budget = 6000
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # occasional random cyclic shift – changes the correlation pattern
        if rng.random() < 0.25:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # a few exponential draws (heavier tails)
    for _ in range(800):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        if rng.random() < 0.25:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # low‑frequency random curves
    for nf in (2, 3, 5, 8):
        for _ in range(250):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            if rng.random() < 0.25:
                shift = rng.integers(n)
                if shift:
                    vec = np.roll(vec, shift)
            _push_candidate(vec)

    # mixed block structures
    for _ in range(300):
        if time.time() > deadline:
            break
        _push_candidate(_random_mixed_blocks(n, rng))

    # ------------------------------------------------------------------
    # 3. Re‑use the previous solution if it matches the current size
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                _push_candidate(_normalize(prev_arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4. Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] retained {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 5. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Local refinements – continuation L‑BFGS on the top few seeds
    # ------------------------------------------------------------------
    max_refinements = min(8, len(seeds))  # we will try a few restarts per seed
    for idx, seed_vec in enumerate(seeds[:max_refinements]):
        if time.time() > deadline - 5.0:
            break
        # number of random restarts per seed (1 – the original, plus a few noisy copies)
        restarts = 3
        for r in range(restarts):
            if time.time() > deadline - 5.0:
                break
            # perturb the seed a little before optimisation (except the first run)
            if r == 0:
                start_vec = seed_vec.copy()
            else:
                noise = rng.normal(scale=0.02, size=n)
                noisy = seed_vec + noise
                start_vec = _normalize(noisy)

            # give each refinement a share of the remaining time
            remaining = deadline - time.time()
            # reserve some time for the final hill‑climb
            refine_deadline = time.time() + max(5.0, remaining * 0.45 / (max_refinements * restarts))

            try:
                refined = _lbfgs_smooth_optimize(start_vec,
                                                 n,
                                                 refine_deadline)
            except RuntimeError:
                refined = start_vec  # fallback – SciPy missing

            refined_val = _objective_surrogate(refined)
            if refined_val < global_best_val:
                global_best_seq = refined
                global_best_val = refined_val
                print(f"[Refine seed {idx+1}·run {r+1}] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final greedy hill‑climb (use any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final hill‑climb ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Evaluate with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
