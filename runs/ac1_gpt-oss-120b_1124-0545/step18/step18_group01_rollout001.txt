def _next_pow2(x):

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

"""
Propose a candidate sequence of non‑negative heights that minimizes
``evaluate_sequence``.

Algorithm
---------
1. Initialise a length‑N distribution (N=4000).  If a previous
   ``height_sequence_1`` exists it is used, otherwise the uniform
   distribution is taken.

2. Run several *restarts* of Adam optimisation in the log‑softmax
   (mirror‑descent) space.  The objective is a smooth approximation of
   the max‑convolution:
        M_β(a) = (1/β)·log Σ_k exp(β·c_k)   with   c = a*a .
   The schedule β = [0.1, 0.5, 1, 5, 20, 80, …, 1e6] is used.
   For each β we perform a few thousand Adam steps, decreasing the
   learning‑rate as 1/√t.  The true surrogate value
        2·N·max(c)
   is monitored and the best vector is kept.

3. After the smooth phase we run a short *exact* exponentiated‑gradient
   (mirror‑descent) pass on the true max‑convolution to clean up any
   remaining smoothing error.

4. Finally the best sequence is evaluated with the provided
   ``evaluate_sequence`` function, stored globally as ``height_sequence_1``,
   and returned.

All steps respect the supplied time budget (default 1000 s) and only
use NumPy (SciPy is optional but not required).  The code is fully
self‑contained and can be run repeatedly – it will continue improving
the current best sequence.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
