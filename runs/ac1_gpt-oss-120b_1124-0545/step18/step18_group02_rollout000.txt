def _targeted_local_search(a, n, deadline, max_iter, step_frac):
    best = a.copy()
    best_val = _objective_exact(a)
    for it in range(max_iter):
        if time.time() > deadline: break
        g = _subgradient_maxconv(a) # gradient shape n
        i_max = np.argmax(g)
        i_min = np.argmin(g) # could be 0 but maybe many zero components; select among those with >0?
        if a[i_max] <= 0: continue
        delta = step_frac * a[i_max]
        # transfer delta from i_max to i_min (could be same as previously)
        # but ensure i_min != i_max. If i_min == i_max skip.
        if i_min == i_max:
            # choose next min
            idx_sorted = np.argsort(g)
            for idx in idx_sorted:
                if idx != i_max:
                    i_min = idx
                    break
        cand = a.copy()
        cand[i_max] -= delta
        cand[i_min] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()
            a = cand
        else:
            # optionally accept if val < current val
            cur_val = _objective_exact(a)
            if val < cur_val:
                a = cand
    return best, best_val

"""
Propose a step‑function minimizing the objective `evaluate_sequence`.

Improvements over the previous baseline
---------------------------------------

* Use a larger problem size (`n = 6000`) – more degrees of freedom allow a
  flatter auto‑convolution and therefore a smaller max‑value.
* Enrich the seed pool: besides the classic families we add
  – random “Sidon‑like” sparse supports (≈√n equally weighted points),
  – a finer power‑law sweep (exponents 0.2 … 0.8, many shifts),
  – mixtures of two power‑laws,
  – very sparse Dirichlet draws (conc < 0.2) that encourage non‑uniformity.
* Continuation L‑BFGS on the log‑sum‑exp surrogate is run with a **longer
  β‑schedule** (up to 1e9) and a larger per‑β iteration budget.
* After the smooth stage we perform a **targeted mirror‑descent** that
  explicitly moves mass from indices with the largest sub‑gradient
  (i.e. those that most increase the current maximum convolution) to
  indices with the smallest sub‑gradient.
* Finally a short stochastic hill‑climb (mass‑transfer) refines the result.

All stages respect the global 1000 s wall‑time limit.  The best sequence
encountered is stored in the global variable `height_sequence_1` and also
returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective (max‑convolution) – used for ranking and final evaluation
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (for a already scaled to target ℓ₁)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active lag)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (active lag only)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth approximation of the max‑convolution objective.

    Returns (value, gradient) w.r.t. logits `v`.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (via softmax of conv)
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """Continuation L‑BFGS on the log‑sum‑exp surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0, 50_000_000.0, 100_000_000.0,
            500_000_000.0, 1_000_000_000.0,
        ]

    # start from logits (avoid zero)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 2.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent with multiplicative updates (exact max‑conv objective)."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted mirror‑descent (mass transfer from worst to best gradient)
# ----------------------------------------------------------------------
def _targeted_local_search(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 30_000,
                           step_frac: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb: repeatedly move a small fraction of mass
    from the index with the largest sub‑gradient (most responsible for
    the current maximum convolution) to the index with the smallest
    sub‑gradient.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)
        i_max = int(np.argmax(g))
        # choose a minimum‑gradient index that is *different* from i_max
        # (if many minima exist, pick the first)
        i_min_candidates = np.where(g == g.min())[0]
        i_min = int(i_min_candidates[0])
        if i_min == i_max:
            # fallback: pick the second best (or any other)
            idx_sorted = np.argsort(g)
            for idx in idx_sorted:
                if idx != i_max:
                    i_min = int(idx)
                    break

        if a[i_max] <= 0.0:
            # nothing to move – break to avoid endless loop
            break

        delta = step_frac * a[i_max]
        if delta <= 0.0:
            break

        cand = a.copy()
        cand[i_max] -= delta
        cand[i_min] += delta
        val = _objective_exact(cand)

        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
        else:
            # accept only if we actually improve the current state
            cur = _objective_exact(a)
            if val < cur:
                a = cand

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic mass‑transfer hill‑climb (as in the original code)
# ----------------------------------------------------------------------
def _stochastic_local_search(start_a: np.ndarray,
                             n: int,
                             deadline: float,
                             iters: int = 30_000,
                             step_frac: float = 0.011) -> tuple[np.ndarray, float]:
    """Random hill‑climb: move a small mass fraction from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_power_law(n: int,
                     p1: float, sh1: float,
                     p2: float, sh2: float,
                     alpha: float) -> np.ndarray:
    """Mixture of two power‑law families, ℓ₁‑normalised."""
    a1 = _power_law(n, p1, sh1)
    a2 = _power_law(n, p2, sh2)
    mixed = (1.0 - alpha) * a1 + alpha * a2
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sidon_like_sparse(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a sparse vector with `m` equal‑weight entries at distinct, randomly
    chosen indices.  The set size `m` is typically around √n, mimicking a
    Sidon‑type support.
    """
    if m <= 0:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed collection (including many diverse families)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.20))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # Power‑law sweep (exponents and shifts)
    p_vals = np.linspace(0.2, 0.8, 13)          # finer range
    shifts = [0.0, 5.0, 20.0, 50.0, 100.0, 200.0, 400.0]
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # Double power‑law mixtures – a cheap way to get richer shapes
    for _ in range(15):
        p1 = rng.uniform(0.2, 0.6)
        sh1 = rng.uniform(0.0, 300.0)
        p2 = rng.uniform(0.4, 0.9)
        sh2 = rng.uniform(0.0, 300.0)
        alpha = rng.random()
        seeds.append(_double_power_law(n, p1, sh1, p2, sh2, alpha))

    # Sparse Sidon‑like supports (size ≈ √n)
    m = max(2, int(np.sqrt(n) * rng.uniform(0.8, 1.4)))
    for _ in range(12):
        seeds.append(_sidon_like_sparse(n, m, rng))

    # Random Dirichlet – favour sparsity with very low concentration
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse (few non‑zero entries)
    for frac in (0.02, 0.05, 0.10):
        for _ in range(4):
            m = max(2, int(frac * n))
            idx = rng.choice(n, size=m, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[idx] = 1.0 / m
            seeds.append(a)

    # Pairwise mixtures of existing seeds (explore convex hull)
    base = np.arange(len(seeds))
    for _ in range(120):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of promising candidates
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Warm‑start: previous best (if supplied)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Final clean‑up – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – targeted deterministic local search
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(80_000, remaining * 60))   # roughly 1‑2 µs per iter ~ ok
        if iters > 0:
            print("\n--- Targeted local search (gradient‑driven mass transfer) ---")
            seq_tgt, val_tgt = _targeted_local_search(
                global_best_seq,
                n,
                deadline,
                max_iters=iters,
                step_frac=0.015,
            )
            print(f"[Targeted] best = {val_tgt:.6f}")
            if val_tgt < global_best_val:
                global_best_seq = seq_tgt.copy()
                global_best_val = val_tgt
                print(f"[Update] New best after targeted search = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – stochastic local search (random hill‑climb) for fine tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(120_000, remaining * 40))
        if iters > 0:
            print("\n--- Stochastic local search (mass transfers) ---")
            seq_loc, val_loc = _stochastic_local_search(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.011,
            )
            print(f"[LocalSearch] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – final high‑β polishing L‑BFGS
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1200, int(min(4000, remaining * 0.05 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000.0, 10_000_000.0, 50_000_000.0,
                           100_000_000.0, 500_000_000.0, 1_000_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (using the supplied evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
