def _grid_search_best(n, deadline):
    best_val = np.inf
    best_shape = None
    best_params = None
    p_vals = np.linspace(0.30, 0.80, 26)
    delta_vals = np.linspace(0.0, 5.0, 11)
    for p in p_vals:
        for delta in delta_vals:
            if time.time() > deadline: break
            a = _power_law_offset(n, p, delta)
            val = _objective_exact(a)
            if val < best_val:
                best_val = val
                best_shape = a
                best_params = (p, delta)
    # refine around best
    p_center, delta_center = best_params
    p_fine = np.linspace(max(0.30, p_center-0.02), min(0.80, p_center+0.02), 21)
    delta_fine = np.linspace(max(0.0, delta_center-0.5), min(5.0, delta_center+0.5), 21)
    for p in p_fine:
        for delta in delta_fine:
            if time.time() > deadline: break
            a = _power_law_offset(n, p, delta)
            val = _objective_exact(a)
            if val < best_val:
                best_val = val
                best_shape = a
                best_params = (p, delta)
    return best_shape, best_val, best_params
```

"""
Optimiser for the “max‑convolution” problem.

The algorithm:

1.  Fixed problem size n (4000).  Build a few analytic families and
    perform a **dense grid search** over a two‑parameter family
    a_i ∝ (i+δ)^(-p) (offset power‑law).  The search is done in a
    coarse grid, then refined locally.  The best shape from this
    search becomes the starting point for the later stages.

2.  **Continuation L‑BFGS** on a smooth log‑sum‑exp surrogate of the
    max‑convolution objective.  The schedule of the smoothing
    parameter β runs from 1 to 5·10⁶, with a generous number of
    L‑BFGS iterations per β (≈2000).  The optimizer works in the
    unconstrained “logits” space and the simplex constraint is
    enforced by a soft‑max.

3.  If time permits, a **mass‑transfer stochastic local search**
    (hill‑climbing) moves a small fraction of mass from one index to
    another and accepts any improvement.  The number of iterations
    is scaled to the remaining time.

The best sequence found is stored globally as ``height_sequence_1``
and returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law_offset(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+delta+1)^{-p}, with delta≥0."""
    i = np.arange(1, n + 1, dtype=np.float64) + delta
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# exact objective (max‑convolution) – scaling‑invariant
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# smooth surrogate (log‑sum‑exp) for continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Surrogate of max‑convolution; returns (obj, grad) for L‑BFGS."""
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient of the smooth max w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
        ]

    # start from log‑space (logits) – avoids zero entries
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun,
                                x0=v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "disp": False,
                                         "ftol": 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:  # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# aggressive stochastic local search (mass transfers)
# ----------------------------------------------------------------------
def _local_search(start_a: np.ndarray,
                  n: int,
                  deadline: float,
                  iters: int = 150_000,
                  step_frac: float = 0.006) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val
