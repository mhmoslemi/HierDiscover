def propose_candidate(budget_s = 1000, ...):

    deadline = start_time + budget_s - 2.0

    n = 4000

        else: best_seq = best_seq / total
    else:

    # initial evaluation
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] initial surrogate {best_val}")

    # Phase A: L-BFGS smoothing if SciPy available
    if _opt is not None and time.time() < deadline:

        val_opt = _objective_surrogate(seq_opt)
        if val_opt < best_val:
            best_seq = seq_opt; best_val = val_opt
            print(f"[L-BFGS] improved surrogate {best_val}")

    # Phase B: Exponentiated gradient polish
    if time.time() < deadline:
        # allocate half remaining time for expgrad iterations
        remaining = deadline - time.time()
        # compute max iterations based on time roughly: each iteration ~ 0.0003 sec? Use estimation
        # but we can set to large number and break when time out.
        seq_exp, val_exp = _exp_gradient_opt(best_seq, n, deadline, max_iters=1_200_000)

            best_seq = seq_exp; best_val = val_exp
            print(f"[ExpGrad] improved to {best_val}")

    # Phase C: Swap hill climbing
    if time.time() < deadline:
        # allocate as many swaps as possible
        remaining = deadline - time.time()
        # estimate swaps per second: earlier 500k swaps took maybe ~150 sec => 3333 swaps per sec.
        # So time remaining seconds * 3000 approx.
        # We'll set max_swaps = int(min(5_000_000, remaining * 3000))

        max_swaps = max(500_000, max_swaps)
        seq_swap, val_swap = _perm_swap_opt(best_seq, n, deadline, max_swaps=max_swaps)
        if val_swap < best_val:
            best_seq = seq_swap; best_val = val_swap
            print(f"[Swap] improved to {best_val}")

    # Phase D: final small expgrad refinement
    if time.time() < deadline:

"""
Propose a candidate step‑function minimizing ``evaluate_sequence``.

Strategy
--------
1. **Parameter sweep** – Sample a few thousand shapes from several
   analytic families (Beta, sine‑window, Gaussian, power‑law,
   edge‑biased, Dirichlet, exponential).  Keep the best (lowest
   2·n·max(conv) surrogate).

2. **Smooth optimisation** – Run a continuation L‑BFGS optimiser on the
   log‑sum‑exp surrogate of the max‑convolution.  The schedule of the
   smoothing parameter β is long and each β gets many L‑BFGS iterations,
   allowing us to approach the nonsmooth optimum.

3. **Exact sub‑gradient polishing** – Mirror‑descent (exponentiated‑gradient)
   using the true max‑convolution sub‑gradient for many iterations.

4. **Heavy permutation hill‑climbing** – Randomly swap two entries and
   keep the swap only if the exact objective (max‑convolution) does not
   increase.  Up to several million swaps are performed, because the
   ordering of the non‑uniform heights strongly influences the max‑auto‑
   convolution.

5. **Final short exponentiated‑gradient pass** after the swaps.

All steps respect a global 1000 s wall‑clock budget and use only NumPy
(and SciPy if available).  The best sequence found is stored in the
global ``height_sequence_1`` variable for reuse in later rounds.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for ℓ₁‑normalised vectors:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_bias(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Optimise the smoothed max‑convolution objective using L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6
        ]

    # work in log‑probability space (stay on simplex via softmax)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_idx := rev_a, w)   # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # classic 1/√t schedule
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Swap‑based permutation hill‑climber
# ----------------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap only if it reduces the exact
    max‑convolution.  Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 100_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Parameter‑space seed search
# ----------------------------------------------------------------------
def _search_param_seeds(n: int,
                        rng: np.random.Generator,
                        deadline: float) -> np.ndarray:
    """
    Sample a few thousand analytic families and keep the best
    (lowest surrogate value).  Returns the best ℓ₁‑normalised vector.
    """
    best_seq = None
    best_val = np.inf

    # Helper to test a candidate and update the best
    def try_seq(seq):
        nonlocal best_seq, best_val
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq

    # 1. Beta‑shape random grid
    for _ in range(200):
        if time.time() > deadline:
            break
        a = 10 ** rng.uniform(-2, 1)   # 0.01 .. 10
        b = 10 ** rng.uniform(-2, 1)
        try_seq(_beta_shape(n, a, b))

    # 2. Sine‑window with random exponent
    for _ in range(120):
        if time.time() > deadline:
            break
        gamma = 10 ** rng.uniform(-1, 1)   # 0.1 .. 10
        try_seq(_sine_window(n, gamma=gamma))

    # 3. Gaussian width
    for _ in range(80):
        if time.time() > deadline:
            break
        sigma = rng.uniform(0.10, 0.45)
        try_seq(_gaussian(n, sigma))

    # 4. Power‑law
    for _ in range(80):
        if time.time() > deadline:
            break
        p = rng.uniform(0.0, 5.0)
        try_seq(_power(n, p))

    # 5. Edge‑biased
    for _ in range(40):
        if time.time() > deadline:
            break
        p = rng.uniform(0.25, 1.0)
        try_seq(_edge_bias(n, p))

    # 6. Random Dirichlet draws
    for _ in range(60):
        if time.time() > deadline:
            break
        conc = 10 ** rng.uniform(-2, 1)   # 0.01 .. 10
        try_seq(_random_dirichlet(n, conc, rng))

    # 7. Random exponential draws
    for _ in range(60):
        if time.time() > deadline:
            break
        try_seq(_random_exponential(n, rng))

    # 8. Uniform as a fallback
    try_seq(_uniform(n))

    # sanity
    if best_seq is None:
        best_seq = _uniform(n)

    # ensure ℓ₁‑normalisation (just in case)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = _uniform(n)

    print(f"[ParamSearch] best surrogate = {best_val:.6f}")
    return best_seq

    start_time = time.time()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 4000

    else:

        else:

        print("[Init] Loaded previous solution as starting point.")

    # randomise ordering a little – ordering matters for the subsequent swap search
    start_seq = start_seq[rng.permutation(n)]

    best_seq = start_seq.copy()

    # ------------------------------------------------------------------

        true_val = None

    return best_seq.tolist()
