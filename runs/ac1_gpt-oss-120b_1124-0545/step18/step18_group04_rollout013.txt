"""
Propose a step‑function (non‑negative heights) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The approach is a larger‑scale stochastic search:

1. **Broad seeding** – many analytical families (including a Sidon‑set
   construction) and a large batch of random Dirichlet vectors are generated,
   normalised to sum = 1 and the best few are kept in a heap.

2. **Continued refinement** – the top seeds are polished with a continuation
   L‑BFGS‑B optimiser that minimises a smooth log‑sum‑exp surrogate of the
   max‑convolution.  The β‑schedule is longer and the optimiser is allowed
   more iterations than in the reference code.

3. **Exploratory hill‑climb** – a simple simulated‑annealing style walk on the
   simplex moves tiny fractions of mass between two coordinates, accepting
   improvements or (with a decaying temperature) occasional worsening moves.
   This helps to escape shallow local minima.

The routine respects the supplied time budget (default 1000 s) and stores the
best vector it found in the global variable ``height_sequence_1`` for the next
call.  All helper functions are top‑level and avoid closures or lambda‐expresssions.
"""

import time
import math
import heapq
import numpy as np

# optional SciPy – required only for the L‑BFGS phase
try:
    import scipy.optimize as _opt
except Exception:   # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# FFT‑based linear convolution (zero‑padded)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (∑ = 1)."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Exact objective (surrogate) – assumes ∑a = 1
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """
    Exact evaluation for a probability vector:
        2·n·max(conv(vec, vec))
    """
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _double_power(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _arcsine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


def _sidon_candidate(n: int) -> np.ndarray:
    """
    Construct a Sidon‑like set using a prime–based quadratic construction.
    The produced vector has mass on a set of positions whose pairwise sums are
    (almost) unique – this yields a low max‑convolution.
    """
    # find a prime p with p^2 ≤ n and p as large as possible
    def _is_prime(m):
        if m < 2:
            return False
        if m % 2 == 0:
            return m == 2
        r = int(m ** 0.5) + 1
        for q in range(3, r, 2):
            if m % q == 0:
                return False
        return True

    p = int(math.isqrt(n))
    while p > 1 and not _is_prime(p):
        p -= 1
    if p < 2:  # fall back to a simple uniform block
        return _uniform(n)

    positions = []
    for i in range(p):
        pos = i * p + (i * i) % p
        if pos < n:
            positions.append(pos)
    m = len(positions)
    a = np.zeros(n, dtype=np.float64)
    a[positions] = 1.0 / m
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    nb = rng.integers(2, 6)
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for _ in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Optimisation runs in log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0,
                         1000000.0, 2000000.0, 5000000.0]

    # initialise logits (log‑softmax variables)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)          # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a  (2 * conv(reverse(a), w))
            rev_a = a[::-1]
            w = exp_beta / sum_exp
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple hill‑climb with occasional uphill moves (simulated annealing)
# ----------------------------------------------------------------------
def _hill_climb_sa(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style walk on the simplex.
    Moves a random fraction of mass from i → j.
    Accepts all improvements and, with a decaying temperature, occasional
    worsening moves.
    Returns the best vector found and its exact surrogate value.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    # temperature schedule (high → low)
    T0 = 0.1
    T_end = 1e-5

    while time.time() < deadline:
        # linear decay in log‑space
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T0 * ((T_end / T0) ** frac)

        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # move a random fraction of the mass at i (0.1 % … 5 %)
        eps = cur[i] * rng.uniform(0.001, 0.05)
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)

        delta = new_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-16)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, prints progress information,
    and stores the best vector in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – larger sizes give a finer search landscape.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Keep the top‑k seeds (min‑heap on surrogate value).
    # ------------------------------------------------------------------
    top_k = 30
    best_heap = []          # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # Deterministic analytical seeds (wide variety)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    _push_candidate(_sidon_candidate(n))

    for gamma in np.linspace(0.2, 3.0, 12):
        _push_candidate(_sine_window(n, gamma))
    for sigma in np.linspace(0.08, 0.30, 12):
        _push_candidate(_gaussian(n, sigma))
    for p in np.linspace(0.1, 3.0, 15):
        _push_candidate(_power(n, p))
        _push_candidate(_double_power(n, p))
    for _ in range(20):
        a = 10 ** rng.uniform(-2, 0)
        b = 10 ** rng.uniform(-2, 0)
        _push_candidate(_beta_shape(n, a, b))
    _push_candidate(_arcsine_window(n))
    for gamma in np.linspace(0.5, 4.0, 9):

    for _ in range(15):
        blk = rng.uniform(0.05, 0.35)
        lw = rng.random()
        _push_candidate(_two_block_uniform(n, block_ratio=blk, left_weight=lw))

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    for _ in range(random_seed_budget):
        if time.time() > deadline:

        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # occasional cyclic shift to change correlation pattern
        if rng.random() < 0.2:

            if shift:

        _push_candidate(vec)

    # exponential draws
    for _ in range(800):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        if rng.random() < 0.2:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # low‑frequency random curves
    for nf in (2, 3, 5, 8, 12):
        for _ in range(300):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            if rng.random() < 0.2:
                shift = rng.integers(n)
                if shift:
                    vec = np.roll(vec, shift)
            _push_candidate(vec)

    for _ in range(300):

        _push_candidate(_random_mixed_blocks(n, rng))

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:

            if prev_arr.size == n and prev_arr.sum() > 0:
                _push_candidate(_normalize(prev_arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] retained {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # Initialise the global best
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Local refinements – continuation L‑BFGS on the top seeds
    # ------------------------------------------------------------------
    max_refinements = min(7, len(seeds))   # a few top seeds get intensive polishing
    for idx in range(max_refinements):

        seed_vec = seeds[idx]

        remaining = deadline - time.time()

                                                            100000.0, 200000.0,
                                                            500000.0,
                                                            1000000.0],
                                             max_iter_per_beta=6000)
        except RuntimeError:

        refined_val = _objective_surrogate(refined)
        if refined_val < global_best_val:

    # ------------------------------------------------------------------
    true_val = None
    try:

              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
