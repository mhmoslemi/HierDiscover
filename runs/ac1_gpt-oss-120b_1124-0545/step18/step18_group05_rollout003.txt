"""
Mirror‑descent (Adam) on the exact max‑convolution objective, using an
active‑set subgradient, followed by a short L‑BFGS polish on a smooth
log‑sum‑exp surrogate.  The routine works on the probability simplex
(∑a=1, a_i≥0) and returns a non‑negative step‑function that improves
the previous best bound (≈1.5085 → target ≈1.503).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised) – used as seeds
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _reverse_power(n: int, p: float) -> np.ndarray:
    """Decreasing power law from the right end."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((n - idx + 1) ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exact surrogate (objective for a probability vector)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# L‑BFGS polish (smooth log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:          # pragma: no cover
    _opt = None


def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 2500) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:                              # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 1_00_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # log‑sum‑exp smoothing of the max
            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (from the smoothed max)
            w = exp_beta / sum_exp          # weights, sum = 1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)     # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                         # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑based mirror descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = None) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam momentum on the exact max‑convolution objective.
    Returns the best probability vector found and its surrogate value.
    """
    EPS = 1e-30
    logit = np.log(start_vec + EPS)               # initialise logits
    m = np.zeros_like(logit)
    v_vec = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    pow1 = 1.0
    pow2 = 1.0

    it = 0
    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)

    # fast periodic reporting interval
    report_every = 5000
    last_report = time.time()

    while time.time() < deadline:
        it += 1
        pow1 *= beta1
        pow2 *= beta2

        a = _softmax(logit)

        # ----- convolution and active set -----
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # treat values within a tiny relative tolerance as active
        tol = max(1e-12, 1e-8 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        # ----- sub‑gradient of max‑conv w.r.t. a -----
        grad_a = np.zeros_like(a)
        if active.size == 0:
            # Should not happen; fall back to gradient from the single max index
            active = [int(np.argmax(conv))]
        inv = 1.0 / active.size
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            rev_idx = k - idx
            grad_a[idx] += inv * 2.0 * a[rev_idx]

        # ----- gradient w.r.t. logits (mirror‑descent) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - pow1)
        v_hat = v_vec / (1.0 - pow2)

        # learning‑rate decays like 1/√t (helps with the non‑smooth landscape)
        base_lr = 0.5
        lr = base_lr / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        logit -= step

        # ----- occasional surrogate evaluation -----
        if it % report_every == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] it={it:,} new best surrogate {best_val:.6f}")

            now = time.time()
            if now - last_report > 30.0:
                elapsed = now - (deadline - (deadline - time.time()))
                print(f"[Adam] it={it:,} best={best_val:.6f} elapsed={now - (deadline - (deadline - time.time())):.1f}s")
                last_report = now

        if max_iters is not None and it >= max_iters:
            break

    # final evaluation on the last logits (could be a little better)
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_val = final_val
        best_seq = final_a.copy()
        print(f"[Adam] final iteration improves best to {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes the provided
    evaluation metric.  The method combines a rich seed pool, Adam‑based
    mirror‑descent on the exact max‑convolution objective (with an active
    set sub‑gradient), followed by a short L‑BFGS polish on a smooth
    log‑sum‑exp surrogate.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries provides a good trade‑off.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 1. Build a diverse seed pool
    # --------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _sin2(n),
        _hann_window(n),
        _arcsine(n),
    ]

    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))
        seeds.append(_reverse_power(n, p))

    # Random Dirichlet draws with a range of concentration parameters
    conc_params = np.logspace(-2, 1, 12)            # 0.01 … 10
    for conc in conc_params:
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # Add the previous best if it exists
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # Normalise everything onto the probability simplex
    def _norm(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / len(v))
        return v / s

    seeds = [_norm(np.asarray(v, dtype=np.float64)) for v in seeds]

    # --------------------------------------------------------------
    # 2. Quick surrogate ranking – pick the best starting seed
    # --------------------------------------------------------------
    best_seed = None
    best_seed_val = np.inf
    for seq in seeds:
        val = _objective_surrogate(seq)
        if val < best_seed_val:
            best_seed_val = val
            best_seed = seq.copy()
    print(f"\n[Init] best surrogate among seeds: {best_seed_val:.6f}")

    # --------------------------------------------------------------
    # 3. Mirror‑descent with Adam on the exact max‑convolution
    # --------------------------------------------------------------
    print("\n--- Adam mirror‑descent (active‑set) ---")
    adam_seq, adam_val = _adam_optimize(best_seed, n, deadline)
    print(f"[Adam] finished with best surrogate {adam_val:.6f}")

    best_seq = adam_seq
    best_val = adam_val

    # --------------------------------------------------------------
    # 4. L‑BFGS polish if time permits
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a modest slice for L‑BFGS (roughly 10‑15 % of remaining time)
        max_iter_per_beta = int(min(5000, remaining * 0.015 * 1e5))
        max_iter_per_beta = max(2000, max_iter_per_beta)
        print("\n--- L‑BFGS polishing ---")
        seq_polished = _lbfgs_optimize(best_seq, n, deadline,
                                       beta_schedule=[1, 5, 20, 80, 200,
                                                      500, 1_000, 2_000,
                                                      5_000, 20_000, 100_000],
                                       max_iter_per_beta=max_iter_per_beta)
        val_polished = _objective_surrogate(seq_polished)
        print(f"[L‑BFGS] polished surrogate = {val_polished:.6f}")
        if val_polished < best_val:
            best_val = val_polished
            best_seq = seq_polished.copy()
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
