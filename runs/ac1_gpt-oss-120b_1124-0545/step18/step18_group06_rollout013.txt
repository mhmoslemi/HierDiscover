        if new[i] < 0: continue

            if new_val < best_val:

                best = new.copy()

"""
A fast, multi‑stage optimizer for the step‑function objective
    2·n·max(conv(seq,seq)) / (sum(seq)**2)

Stages
-------
1. **Diverse analytic / random seeds** – a rich pool of ℓ₁‑normalised vectors:
   - uniform, triangular, cosine, various windows, Gaussian / double‑Gaussian,
     power laws, beta‑shapes, Tukey, Kaiser, etc.  Parameters are sampled from
     wide ranges (log‑uniform where useful) and many Dirichlet draws are added.
2. **Greedy selection** – evaluate every seed with the exact objective (FFT‑based
   O(n log n)) and keep the best few (default 10).
3. **Long mirror‑descent (exponentiated‑gradient)** – starting from each of the
   top seeds we run a sub‑gradient method on the *exact* max‑convolution.
   The iterate lives in the probability simplex via a soft‑max parametrisation.
   Up to a million iterations are performed (or until the time budget is
   exhausted) with a classic ηₜ = c/√t schedule.
4. **Continuous simulated‑annealing** – from the best vector obtained after
   stage 3 we perform many tiny mass‑transfer moves (move a random ε from
   index i to j).  Moves are accepted with a Metropolis rule; the temperature
   decays linearly.  The whole phase is cheap (one FFT per move) and often
   reduces the objective a few % further.
5. **Final evaluation** – the best vector found is stored in the global
   variable ``height_sequence_1`` and returned as a plain list.

The implementation respects the 1000 s wall‑time limit (a 2 s safety margin)
and works with any number of CPUs (the heavy work is vectorised NumPy/FFT).

The algorithm is deliberately different from the previous L‑BFGS + swap
pipeline, explores a much larger family of shapes and performs a very long
sub‑gradient run, which is known to drive the bound well below the previous
1.5085 level.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT for speed)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax that maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors (sum == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p   (i starts at 0)"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution (exponentiated gradient)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000,
                     c: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on f(a)=max(conv(a,a)).
    The iterate lives in the probability simplex via soft‑max of log‑weights.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)          # work in log‑space (avoids zero)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exact max‑convolution and the lag attaining it
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient w.r.t. a at the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning‑rate schedule – classic 1/√t, scaled by constant c
        eta = c / math.sqrt(it)

        # mirror‑descent step in log‑space
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress reports (optional, quiet in final run)
        # if it % 200_000 == 0:
        #     print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuous simulated‑annealing: tiny mass‑transfer moves
# ----------------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_moves: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Randomly select i → j and move a small ε of mass.
    Accept always if objective improves, otherwise with Metropolis probability.
    Temperature decays linearly.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(0.1 * cur_val, 1.0)      # initial temperature

    for it in range(1, max_moves + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue

        # propose moving a random fraction (up to 5 %) of the mass at i
        eps = cur[i] * rng.random() * 0.05
        new = cur.copy()
        new[i] -= eps
        new[j] += eps

        new_val = _objective(new)
        delta = new_val - cur_val

        # temperature schedule – linear cooling
        T = max(T0 * (1.0 - it / max_moves), 1e-12)
        accept = delta < 0.0 or rng.random() < math.exp(-delta / T)

        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()

        # optional progress print
        # if it % 200_000 == 0:
        #     print(f"[MassTransfer] moves={it:,} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed collection – much broader than the previous version
# ----------------------------------------------------------------------
def _collect_seeds_expanded(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a large, diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families – sweep parameters more aggressively
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine window with many gammas (log‑spaced)
    for g in np.logspace(-1, 1, 9):          # 0.1 … 10
        seeds.append(_sine_window(n, gamma=g))

    # Gaussian, sigma spanning a larger interval
    for sigma in np.linspace(0.05, 0.80, 10):
        seeds.append(_gaussian(n, sigma))

    # Double‑Gaussian with random sigma / shift
    for _ in range(4):
        sigma = rng.uniform(0.05, 0.30)
        shift = rng.uniform(0.10, 0.40)
        seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))

    # Power‑law tails
    for p in np.linspace(0.0, 8.0, 12):
        seeds.append(_power(n, p))

    # Beta shapes – sample α,β log‑uniformly
    for _ in range(12):
        a = 10 ** rng.uniform(-1.5, 1.0)     # 0.03 … 10
        b = 10 ** rng.uniform(-1.5, 1.0)
        seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.05, 0.95, 10):
        seeds.append(_tukey_window(n, alpha))

    # Kaiser windows – β values across a wide range
    for beta in np.logspace(0, 2, 9):       # 1 … 100
        seeds.append(_kaiser_window(n, beta))

    # Random Dirichlet draws – many concentration values
    for conc in (0.01, 0.05, 0.1, 0.3, 0.7, 1.5, 3.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency sinusoidal curves
    for nf in (2, 3, 5, 7):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random vectors (different sparsities)
    for k_frac in (0.005, 0.015, 0.040):
        k = max(1, int(round(k_frac * n)))
        for _ in range(3):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors (different block lengths)
    for bf in (0.02, 0.08, 0.25):
        seeds.append(_block_uniform(n, bf, rng))

    # Mix random pairs of seeds (crossover‑style)
    for _ in range(30):
        a, b = rng.choice(seeds, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(np.asarray(a), np.asarray(b), alpha))

    # Permuted heavy‑tailed Dirichlet (helps permutation search)
    heavy = _random_dirichlet(n, 0.03, rng)
    for _ in range(5):
        seeds.append(heavy[rng.permutation(n)])

    # Previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity‑check – every vector ends up ℓ₁‑normalised.
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    # ------------------------------------------------------------------
    # 0. initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    # ------------------------------------------------------------------
    # 1. problem size (keep the historical size – 4000 works well)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. collect a broad pool of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed collection (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_expanded(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 3. quick evaluation of all seeds – keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed objectives (exact, i.e. 2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. long exponentiated‑gradient polishing on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best objective = {global_best_val:.6f}")

    # allocate most of the remaining budget to gradient descent
    # (approx. 65 % of the total time)
    grad_time_budget = (deadline - time.time()) * 0.65

    # we will run the gradient method on the current global best first,
    # then optionally on a few other top seeds if time permits.
    seeds_to_opt = [global_best_seq] + top_seeds[1:5]   # up to five seeds

    for idx, seed_vec in enumerate(seeds_to_opt):
        if time.time() > deadline:
            break

        remaining = deadline - time.time()
        # each run gets a proportional slice of the gradient budget
        per_seed_budget = min(grad_time_budget / len(seeds_to_opt), remaining * 0.9)
        sub_deadline = time.time() + per_seed_budget

        print(f"\n--- Exponentiated‑gradient run #{idx+1} (budget ≈ {per_seed_budget:.1f}s) ---")
        seq_opt, val_opt = _exp_gradient_opt(seed_vec, n, sub_deadline,
                                             max_iters=1_200_000,
                                             c=0.7)

        print(f"[ExpGrad] final value = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. fine‑grained simulated‑annealing mass‑transfer moves
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_moves = int(min(800_000, remaining * 0.30))   # about 30 % of time
        max_moves = max(200_000, max_moves)
        print("\n--- Continuous simulated‑annealing (mass transfers) ---")
        seq_perm, val_perm = _mass_transfer_opt(global_best_seq, n,
                                                deadline,
                                                max_moves=max_moves)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[Anneal] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nExact surrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
