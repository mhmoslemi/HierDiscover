"""
minimax_lp_step_function.py

Search for a non‑negative step‑function (vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

The algorithm consists of three stages:

1. **Initialisation** – pick a good analytic seed (arcsine, Gaussian,
   Dirichlet draws, or the previous best if available) and normalise it to
   a probability simplex.

2. **Linear‑programming minimax iteration** – for the current distribution
   `f` we solve the LP

        minimise   λ
        subject to conv(f, b) ≤ λ            (for every shift)
                   ∑ b_i = 1 ,  b_i ≥ 0

   The LP gives a new distribution `b` whose self‑convolution is provably
   no larger than the previous maximum.  We replace `f ← b` and repeat.
   This is a small‐scale alternating‑minimax method and converges rapidly
   to a better bound (typically a few LP solves are enough).

3. **Polish** – a short mirror‑descent (exponentiated‑gradient) run on the
   active constraint of the final `f` is performed if time permits.

The whole routine respects the 1000 s time budget, prints progress and,
after the final optimisation, evaluates the true objective using
`evaluate_sequence`.  The best vector is stored globally as
``height_sequence_1`` for warm‑starting the next call.
"""

import time
import math
import numpy as np
import scipy.sparse as _sp
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Fast linear convolution via real FFT
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Analytic seeds (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Helper: build the (full) convolution matrix  conv(f,·)  as a sparse CSR.
# For a vector f of length n the matrix has shape (2n‑1) × n and
# entries  M[k,i] = f[i]  if 0 ≤ k‑i < n  else 0.
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> _sp.csr_matrix:
    n = len(f)
    # number of non‑zeros = n * n
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)

    pos = 0
    for i in range(n):
        length = n
        rows[pos:pos + length] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + length] = i
        data[pos:pos + length] = f[i]
        pos += length

    return _sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)


# ----------------------------------------------------------------------
# Linear‑programming minimax step
# ----------------------------------------------------------------------
def _lp_minimax_step(f: np.ndarray, deadline: float):
    """
    Given the current distribution f (∑f = 1, f ≥ 0) solve

        min λ
        s.t. conv(f, b) ≤ λ   (for every shift)
             ∑ b_i = 1 , b_i ≥ 0

    Returns (b, λ) on success, otherwise (None, None).
    """
    n = len(f)

    # Build full constraint matrix conv(f,·) ≤ λ
    A_conv = _conv_matrix(f)                     # shape (2n‑1, n)
    neg_one = -np.ones((A_conv.shape[0], 1), dtype=np.float64)
    lambda_col = _sp.csr_matrix(neg_one)       # column for –λ
    A_ub = _sp.hstack([A_conv, lambda_col], format='csr')
    b_ub = np.zeros(A_ub.shape[0], dtype=np.float64)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    # Objective: minimise λ  (the last variable)
    c = np.concatenate([np.zeros(n), np.ones(1)])

    # Variable bounds: b_i ≥ 0 , λ ≥ 0
    bounds = [(0, None)] * (n + 1)

    # Linprog options – allow a per‑call time limit
    remaining = max(0.0, deadline - time.time())
    opts = {"presolve": True, "time_limit": remaining}

    try:
        res = _opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                           A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds,
                           method="highs",
                           options=opts)
    except Exception as exc:                     # pragma: no cover
        print("[LP] exception while solving:", exc)
        return None, None

    if not res.success:
        print("[LP] optimisation failed:", res.message)
        return None, None

    lam = res.x[-1]
    b = res.x[:n]
    # Numerical clean‑up
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s <= 0.0:
        return None, None
    b /= s
    return b, lam

# Optional short mirror‑descent (exponentiated‑gradient) polish
# ----------------------------------------------------------------------
def _mirror_descent_polish(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iter: int = 250_000) -> np.ndarray:
    """A lightweight EG (soft‑max) descent, started from ``start_seq``."""
    cur = start_seq.copy()
    log_a = np.log(cur + 1e-30)

    it = 0
    best = cur.copy()
    best_val = 2.0 * n * np.max(_conv_fft(cur, cur))

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution entry
        g = np.zeros_like(cur)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * cur[rev_idx]

        eta = 0.4 / math.sqrt(it + 1.0)
        log_a -= eta * g
        cur = np.exp(log_a - np.max(log_a))
        cur /= cur.sum()

        if it % 20_000 == 0:
            cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
                print(f"[EG] it={it:,} new best {best_val:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a step‑function that minimises the evaluation metric.
    The routine:
        * builds a rich pool of analytic/random seeds,
        * picks the best seed (or the previously stored one),
        * runs a few LP‑based minimax iterations,
        * optionally finishes with a short exponentiated‑gradient polish,
        * returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – balance quality / speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build a diverse seed pool (same families as the reference code)
    # ------------------------------------------------------------------
    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sin2(n))
    seeds.append(_hann_window(n))
    seeds.append(_arcsine(n))

    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # random Dirichlet draws (wide range of concentration)
    conc_params = np.logspace(-2, 1, 12)            # 0.01 … 10
    for conc in conc_params:
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # include the previous best if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # Normalise everything onto the probability simplex
    def _norm(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / len(v))
        return v / s

    seeds = [_norm(np.asarray(v, dtype=np.float64)) for v in seeds]

    # ------------------------------------------------------------------
    # 2. Pick the best seed according to the exact surrogate (sum = 1)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for seq in seeds:
        conv = _conv_fft(seq, seq)
        val = 2.0 * n * np.max(conv)               # surrogate = true value when sum = 1
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"\n[Init] best surrogate among seeds: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Minimax linear‑programming iterations
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    lp_iter = 0
    while time.time() < deadline and lp_iter < 12:   # cap iterations to keep within budget
        lp_iter += 1
        print(f"\n[LP] iteration {lp_iter}")

        b_seq, lam = _lp_minimax_step(cur_seq, deadline)
        if b_seq is None:
            print("[LP] step failed – terminating LP loop")
            break

        # Evaluate the true objective for the new candidate
        conv_new = _conv_fft(b_seq, b_seq)
        new_val = 2.0 * n * np.max(conv_new)

        print(f"[LP] λ={lam:.6f}  →  surrogate={new_val:.6f}")

        if new_val + 1e-12 < best_val:               # genuine improvement
            best_val = new_val
            best_seq = b_seq.copy()
            cur_seq = b_seq.copy()
            print(f"[LP]   improvement! new best = {best_val:.6f}")
        else:
            print("[LP]   no improvement – stop LP phase")
            break

    # ------------------------------------------------------------------
    # 4. Optional short mirror‑descent polish (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 250))   # generous but cheap
        max_iters = max(50_000, max_iters)
        print("\n--- Mirror‑descent polish ---")
        polished = _mirror_descent_polish(best_seq, n, deadline, max_iter=max_iters)
        polished_val = 2.0 * n * np.max(_conv_fft(polished, polished))
        print(f"[Polish] polished surrogate = {polished_val:.6f}")
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished.copy()
            print(f"[Polish]   accepted new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for warm‑starting next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
