def _subgradient_descent(start_a, n, deadline, rng):

    a = start_a.copy()
    cur_val = _objective_surrogate(a)
    best_a = a.copy()
    best_val = cur_val

    step = 0.1
    min_step = 1e-5
    while time.time() < deadline and step > min_step:
        conv = _conv_fft(a, a)
        max_val = conv.max()
        max_idx = conv.argmax()

        # compute gradient for max index
        g = np.zeros_like(a)
        # compute range of indices j where k-j in [0, n-1]
        lo = max(0, max_idx - (n-1))
        hi = min(n-1, max_idx)
        # vectorized: slice = a[max_idx - np.arange(lo, hi+1)]
        idx = np.arange(lo, hi+1)
        g[idx] = 2.0 * a[max_idx - idx]
        # gradient direction is to decrease, so we subtract step*g
        # line search
        improved = False
        # try step factor decreasing
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step_factor = step * factor
            a_candidate = a - step_factor * g
            # project onto simplex
            a_candidate = _project_simplex(a_candidate)
            new_val = _objective_surrogate(a_candidate)
            if new_val < cur_val - 1e-12:
                a = a_candidate
                cur_val = new_val
                improved = True
                if new_val < best_val:
                    best_a = a_candidate.copy()
                    best_val = new_val
                break
        if not improved:
            step *= 0.5
        # optional: random perturbations occasional
        if rng.random() < 0.001:
            # random shuffle a small random step
            i, j = rng.integers(n, size=2)
            if i!=j:
                eps = rng.uniform(0, a[i]*0.01)
                a_pert = a.copy()
                a_pert[i] -= eps
                a_pert[j] += eps
                a_pert = _project_simplex(a_pert)
                val_pert = _objective_surrogate(a_pert)
                if val_pert < cur_val:
                    a = a_pert
                    cur_val = val_pert
                    if val_pert < best_val:
                        best_a = a_pert.copy()
                        best_val = val_pert

    return best_a, best_val

""" 
Efficient search for a non‑negative sequence minimising 2·n·max(conv(a,a))/∑a².
The algorithm:
1. Generate a diverse set of seeds from analytic windows, random Dirichlet,
   low‑frequency curves and random block patterns.
2. Keep the K best seeds (by the exact surrogate).
3. Polishing: for each top seed run a continuation L‑BFGS optimiser on the
   smooth log‑sum‑exp surrogate (large β schedule) – this exploits fast
   gradients in the log‑softmax space.
4. Sub‑gradient descent on the true max‑convolution objective – a simple
   projected‑gradient scheme that directly reduces the current maximal
   convolution entry.
5. A final tiny‑step hill‑climb.
The best sequence found is stored in ``height_sequence_1`` and returned.
"""

import time
import heapq
import math
import numpy as np

# optional SciPy import – required only for the L‑BFGS phase
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (zero‑padded)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a (possibly negative) vector onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact surrogate (used while searching – assumes ∑a = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑ vec = 1):
        2·n·max(conv(vec, vec))
    """
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return normalised vectors (∑ = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _double_power(n: int, p: float) -> np.ndarray:
    """Symmetric sum of two power laws – high at both ends, low in the centre."""
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return _normalize(a)
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return _normalize(a)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return _normalize(a)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return _normalize(a)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    return _normalize(a)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0
    return _normalize(a)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _arcsine_window(n: int) -> np.ndarray:
    """U‑shaped arcsine‑like density  (≈1/√(x·(1‑x)))"""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """|cos|‑shaped – large at both ends, small in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    """
    Two uniform blocks placed at the left and right ends.
    block_ratio – fraction of the total length occupied by each block.
    left_weight – total mass placed on the left block (right gets 1‑left).
    """
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with all concentration parameters = conc."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, cut to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose a few disjoint blocks and put uniform mass on each."""
    nb = rng.integers(2, 6)                 # number of blocks
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for b in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()      # share of the remaining mass
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates on log‑softmax variables to stay on the simplex.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0,
                         1000.0, 2000.0,
                         5000.0, 10000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0,
                         2000000.0]

    # initialise logits (log‑softmax variables)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c                     # stabilising shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            w = exp_beta / sum_exp            # soft‑max weights over conv entries
            full = _conv_fft(rev_a, w)        # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(start_vec: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient method that directly minimises
    2·n·max(conv(a,a)).  At each iteration the gradient of the presently
    maximal convolution entry is used.  A back‑tracking line‑search with
    geometric decay is employed; the step size is halved when no progress
    is obtained.
    """
    a = start_vec.copy()
    cur_val = _objective_surrogate(a)
    best_a = a.copy()
    best_val = cur_val

    step = 0.15          # initial step size
    min_step = 1e-6

    while time.time() < deadline and step > min_step:
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        max_idx = np.argmax(conv)

        # gradient for the chosen max index
        g = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        # indices j in [lo,hi]
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * a[max_idx - idx]

        improved = False
        # try a few decreasing step factors
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step_f = step * factor
            cand = a - step_f * g
            cand = _project_simplex(cand)
            cand_val = _objective_surrogate(cand)
            if cand_val < cur_val - 1e-14:
                a = cand
                cur_val = cand_val
                improved = True
                if cand_val < best_val:
                    best_a = cand.copy()
                    best_val = cand_val
                break

        if not improved:
            step *= 0.5
            continue

        # occasional random jitter to escape shallow plateaus
        if rng.random() < 0.002:
            i, j = rng.integers(n, size=2)
            if i != j and a[i] > 0:
                eps = a[i] * rng.uniform(1e-4, 1e-2)
                jitter = a.copy()
                jitter[i] -= eps
                jitter[j] += eps
                jitter = _project_simplex(jitter)
                jitter_val = _objective_surrogate(jitter)
                if jitter_val < cur_val:
                    a = jitter
                    cur_val = jitter_val
                    if jitter_val < best_val:
                        best_a = jitter.copy()
                        best_val = jitter_val

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves on the simplex.
    Each step moves a random fraction (0.1 %–1 %) of mass from i to j.
    Only improvements are kept.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        eps = cur[i] * rng.uniform(0.001, 0.01)
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val

"""
Efficient search for a non‑negative sequence (step‑function) that minimizes

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm follows four stages:

1. **Seeding** – a rich collection of analytic windows, random Dirichlet
   vectors, low‑frequency cosine curves, block structures and dense
   β‑shape draws is generated.  The best *K* candidates (by the exact
   surrogate 2·n·max(conv)) are kept.

2. **Local polishing** – the top seeds are refined with a continuation
   L‑BFGS optimizer that minimises a smooth log‑sum‑exp surrogate for the
   max‑convolution.  The β‑schedule is extended to very large values so
   the surrogate approaches the true objective.

3. **Sub‑gradient descent** – starting from the best polished seed a
   simple projected sub‑gradient method directly reduces the exact
   max‑convolution term.  At each iteration the gradient of the currently
   maximal convolution entry is used, a geometric back‑tracking line search
   selects a step size and the iterate is projected onto the probability
   simplex.

4. **Final hill‑climb** – tiny random mass‑transfer moves on the simplex
   accept only improvements, polishing any remaining slack.

The best vector found is stored in the global variable ``height_sequence_1``,
printed and returned as a plain list of floats.
"""

import time
import heapq
import numpy as np

# optional SciPy import – required only for the L‑BFGS phase
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# FFT‑based full linear convolution (zero‑padded)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a (maybe negative) vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact surrogate (assumes ∑a = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑ vec = 1):
        2·n·max(conv(vec, vec))
    """
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return normalised vectors (∑ = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _double_power(n: int, p: float) -> np.ndarray:
    """Symmetric sum of two power laws – high at both ends, low in the centre."""
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return _normalize(a)
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return _normalize(a)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return _normalize(a)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return _normalize(a)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    return _normalize(a)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # modified Bessel I0
    return _normalize(a)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _arcsine_window(n: int) -> np.ndarray:
    """U‑shaped arcsine‑like density  (≈1/√(x·(1‑x)))"""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """|cos|‑shaped – large at both ends, small in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    """
    Two uniform blocks placed at the left and right ends.
    block_ratio – fraction of the total length occupied by each block.
    left_weight – total mass placed on the left block (right gets 1‑left).
    """
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with all concentration parameters = conc."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, cut to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose a few disjoint blocks and put uniform mass on each."""
    nb = rng.integers(2, 6)                 # number of blocks
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for b in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()      # share of the remaining mass
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates on log‑softmax variables so the iterate always stays on the simplex.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0,
                         1000.0, 2000.0,
                         5000.0, 10000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0,
                         1000000.0,
                         2000000.0]

    # initialise logits (log‑softmax variables)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c                     # stabilising shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            w = exp_beta / sum_exp            # soft‑max weights over conv entries
            full = _conv_fft(rev_a, w)        # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(start_vec: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly minimising
    2·n·max(conv(a, a)).  Each iteration uses the gradient of the presently
    maximal convolution entry, a geometric back‑tracking line search and a
    projection onto the probability simplex.
    """
    a = start_vec.copy()
    cur_val = _objective_surrogate(a)
    best_a = a.copy()
    best_val = cur_val

    step = 0.15          # initial step size
    min_step = 1e-6

    while time.time() < deadline and step > min_step:
        conv = _conv_fft(a, a)
        max_idx = np.argmax(conv)

        # gradient of the selected max entry
        g = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * a[max_idx - idx]

        improved = False
        # try a few shrinking step factors
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step_f = step * factor
            cand = a - step_f * g
            cand = _project_simplex(cand)
            cand_val = _objective_surrogate(cand)
            if cand_val < cur_val - 1e-14:
                a = cand
                cur_val = cand_val
                improved = True
                if cand_val < best_val:
                    best_a = cand.copy()
                    best_val = cand_val
                break

        if not improved:
            step *= 0.5
            continue

        # occasional tiny random jitter to shake off plateaus
        if rng.random() < 0.002:
            i, j = rng.integers(n, size=2)
            if i != j and a[i] > 0:
                eps = a[i] * rng.uniform(1e-4, 1e-2)
                jitter = a.copy()
                jitter[i] -= eps
                jitter[j] += eps
                jitter = _project_simplex(jitter)
                jitter_val = _objective_surrogate(jitter)
                if jitter_val < cur_val:
                    a = jitter
                    cur_val = jitter_val
                    if jitter_val < best_val:
                        best_a = jitter.copy()
                        best_val = jitter_val

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves on the simplex.
    Each step moves a random fraction (0.1 %–1 %) of mass from i to j,
    accepting only improvements.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        eps = cur[i] * rng.uniform(0.001, 0.01)
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the time budget, prints progress,
    stores the best vector in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # problem size (same as the previous best)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # keep a heap of the best seeds (by exact surrogate)
    # --------------------------------------------------------------
    top_k = 25
    best_heap = []   # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # --------------------------------------------------------------
    # 1. Deterministic analytical families
    # --------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    _push_candidate(_hamming_window(n))
    _push_candidate(_blackman_window(n))
    _push_candidate(_nuttall_window(n))
    _push_candidate(_arcsine_window(n))

    for gamma in np.linspace(0.2, 4.0, 15):
        _push_candidate(_sine_window(n, gamma))
    for beta in np.linspace(1.0, 15.0, 15):
        _push_candidate(_kaiser_window(n, beta))
    for sigma in np.linspace(0.07, 0.35, 15):
        _push_candidate(_gaussian(n, sigma))
    for p in np.linspace(0.05, 4.0, 20):
        _push_candidate(_power(n, p))
        _push_candidate(_double_power(n, p))
    for alpha in np.linspace(0.2, 2.0, 9):
        for beta in np.linspace(0.2, 2.0, 9):
            _push_candidate(_beta_shape(n, alpha, beta))
    for a in np.linspace(0.0, 1.0, 9):
        _push_candidate(_tukey_window(n, a))
    for gamma in np.linspace(0.5, 4.5, 12):
        _push_candidate(_cosine_edge_window(n, gamma))
    for _ in range(20):
        blk = rng.uniform(0.05, 0.4)
        lw = rng.random()
        _push_candidate(_two_block_uniform(n, block_ratio=blk, left_weight=lw))

    # --------------------------------------------------------------
    # 2. Random seeding
    # --------------------------------------------------------------
    conc_choices = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for _ in range(3000):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        if rng.random() < 0.2:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    for _ in range(500):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        if rng.random() < 0.2:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    for nf in (2, 3, 5, 8, 12):
        for _ in range(200):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            if rng.random() < 0.2:
                shift = rng.integers(n)
                if shift:
                    vec = np.roll(vec, shift)
            _push_candidate(vec)

    for _ in range(400):
        if time.time() > deadline:
            break
        _push_candidate(_random_mixed_blocks(n, rng))

    # denser beta‑shape draws (log‑space)
    for _ in range(200):
        if time.time() > deadline:
            break
        a = 10 ** rng.uniform(-1.5, 0.0)   # roughly 0.03 … 1
        b = 10 ** rng.uniform(-1.5, 0.0)
        _push_candidate(_beta_shape(n, a, b))

    # --------------------------------------------------------------
    # 3. Re‑use previous solution if it matches the current size
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and np.sum(prev_arr) > 0:
                _push_candidate(_normalize(prev_arr))
        except Exception:
            pass

    # --------------------------------------------------------------
    # 4. Extract the best seeds (ordered from best to worst)
    # --------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] retained {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # --------------------------------------------------------------
    # 5. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Local refinements – continuation L‑BFGS on a few top seeds
    # --------------------------------------------------------------
    max_refinements = min(6, len(seeds))
    for idx in range(max_refinements):
        if time.time() > deadline - 10.0:
            break
        seed_vec = seeds[idx]
        remaining = deadline - time.time()
        refine_deadline = time.time() + max(5.0, remaining * 0.45 / max_refinements)

        if _opt is not None:
            try:
                refined = _lbfgs_smooth_optimize(seed_vec,
                                                 n,
                                                 refine_deadline,
                                                 beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                                200.0, 500.0,
                                                                1000.0, 2000.0,
                                                                5000.0, 10000.0,
                                                                20000.0, 50000.0,
                                                                100000.0, 200000.0,
                                                                500000.0,
                                                                1000000.0],
                                                 max_iter_per_beta=3000)
            except Exception as e:
                print(f"[L‑BFGS] exception {e}")
                refined = seed_vec
        else:
            refined = seed_vec

        refined_val = _objective_surrogate(refined)
        if refined_val < global_best_val:
            global_best_seq = refined
            global_best_val = refined_val
            print(f"[Refine {idx+1}] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Sub‑gradient descent on the exact max‑convolution objective
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\n--- Sub‑gradient descent ---")
        seq_sg, val_sg = _subgradient_descent(global_best_seq, n, deadline, rng)
        if val_sg < global_best_val:
            global_best_seq = seq_sg
            global_best_val = val_sg
            print(f"[Sub‑grad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 8. Final greedy hill‑climb (tiny mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final hill‑climb ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 9. Evaluate with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
