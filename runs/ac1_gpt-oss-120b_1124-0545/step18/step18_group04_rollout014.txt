"""
Propose a height sequence that minimises evaluate_sequence.
The algorithm builds a low‑dimensional basis of analytic window shapes,
optimises a convex combination of these shapes using a smoothed max‑convolution
objective (log‑sum‑exp surrogate) with L‑BFGS‑B, then performs a fine‑grained
projected gradient descent on the original vector.  The final result is
projected onto the simplex (non‑negative, unit sum) and returned.
"""

import time, math, heapq, numpy as np
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# FFT convolution functions
def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

# Simple utilities
def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _normalize(v):
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s

# Analytic families (all return normalised vectors)
def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)

def _sine_window(n, gamma=1.0):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)

def _gaussian(n, sigma_frac):
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)

def _power(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)

def _double_power(n, p):
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)

def _beta_shape(n, alpha, beta):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)

def _arcsine_window(n):
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)

def _cosine_edge_window(n, gamma=1.0):
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)

def _tukey_window(n, alpha):
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return _normalize(a)
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return _normalize(a)

def _two_block_uniform(n, block_ratio=0.2, left_weight=0.5):
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)

# Random generators (return normalised vectors)
def _random_dirichlet(n, conc, rng):
    return _normalize(rng.dirichlet(np.full(n, conc)))

def _random_exponential(n, rng):
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)

def _lowfreq_curve(n, num_freq, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)

def _random_mixed_blocks(n, rng):
    nb = rng.integers(2, 6)
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for _ in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)

# Optimisation of convex combination of basis shapes
# -------------------------------------------------
def _lbfgs_optimize_weights(start_w, basis, n, deadline,
                            beta_schedule=None,
                            max_iter_per_beta=5000):
    """
    Optimise a convex combination of the columns of `basis`.
    `start_w` must already lie on the simplex.
    Returns the Optimised weight vector (still on simplex).
    """
    if _opt is None:
        raise RuntimeError("SciPy not available")
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 10000.0,
                         20000.0, 50000.0, 100000.0, 200000.0,
                         500000.0, 1e6, 2e6, 5e6, 1e7]

    # initialise logits for softmax parametrisation
    v = np.log(start_w + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            w = _softmax(v_local)               # weight vector on simplex
            a = basis @ w                        # mixture, already sums to 1
            conv = _conv_fft(a, a)
            shift = np.max(conv)                 # used only for numerical stability
            exp_b = np.exp(beta_val * (conv - shift))
            sum_exp = exp_b.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (same as in original script)
            rev_a = a[::-1]
            soft_w = exp_b / sum_exp
            full = _conv_fft(rev_a, soft_w)      # length 3n-2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule: grad_w = B^T grad_a
            grad_w = basis.T @ grad_a

            # chain rule through softmax
            dot = np.dot(grad_w, w)
            grad_v = w * (grad_w - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[Weight L‑BFGS] exception {e} at β={beta_val}")
            break

    w_opt = _softmax(v)
    return w_opt

def _projected_gradient_descent(a, n, deadline, step=0.05, max_iters=2000):
    """
    Perform a few hundred iterations of projected gradient descent
    on the exact max‑convolution objective (using the sub‑gradient
    from the active index).  The step size is reduced over time.
    """
    cur = a.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(cur, cur)
        k = int(np.argmax(conv))
        # sub‑gradient of max conv (non‑smooth) is 2 * a flipped at offset k
        grad = np.zeros_like(cur)
        # grad[i] = 2 * cur[k - i] when inside bounds
        # vectorised version:
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        grad[mask] = 2.0 * cur[partner[mask]]
        # take a step opposite to gradient
        cur = cur - step * grad
        # project back onto simplex
        cur = _normalize(cur)
        # optional: adapt step size
        if (it + 1) % 200 == 0:
            step *= 0.7
    return cur

"""
Propose a height sequence that minimises ``evaluate_sequence``.

The algorithm works in two stages:

1️⃣  A low‑dimensional *basis* of analytic window shapes (all non‑negative,
    normalised to sum‑1) is built.  The search is performed on the simplex of
    convex‑combination weights of these shapes.  A log‑sum‑exp surrogate of the
    max‑convolution (smooth maximum) is minimised with L‑BFGS‑B.  The optimisation
    is carried out in log‑softmax space so that the weight constraints
    (non‑negative, sum‑1) are satisfied automatically.

2️⃣  The resulting full‑length vector is fine‑tuned by a few iterations of a
    projected sub‑gradient descent that directly targets the exact
    ``max(conv(a,a))`` (the active convolution index provides a sub‑gradient).

The best vector found is returned and stored in the global variable
``height_sequence_1`` for warm‑starts in subsequent calls.
"""

import time
import heapq
import numpy as np

# optional SciPy – required for the L‑BFGS phase
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# FFT‑based linear convolution (zero‑padded)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _double_power(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _arcsine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return _normalize(a)
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    """Two uniform blocks placed at the left and right ends."""
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float,
                     rng: np.random.Generator) -> np.ndarray:
    """Dirichlet sample with identical concentration parameters."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _lowfreq_curve(n: int, num_freq: int,
                   rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    """Put uniform mass on a random collection of disjoint blocks."""
    nb = rng.integers(2, 6)                 # number of blocks
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for _ in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)


# ----------------------------------------------------------------------
# Exact surrogate (used while searching – assumes sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑ vec = 1):
        2·n·max(conv(vec, vec))
    """
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Optimisation of a convex combination of basis shapes
# ----------------------------------------------------------------------
def _lbfgs_optimize_weights(start_w: np.ndarray,
                            basis: np.ndarray,
                            n: int,
                            deadline: float,
                            beta_schedule=None,
                            max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimise a weight vector w (simplex of size m) that mixes the columns of
    ``basis``.  The optimisation uses a smooth surrogate of the max‑convolution
    and L‑BFGS‑B in log‑softmax coordinates.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 10000.0,
                         20000.0, 50000.0, 100000.0, 200000.0,
                         500000.0, 1e6, 2e6, 5e6, 1e7]

    # initialise logits for the softmax parametrisation
    v = np.log(start_w + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            w = _softmax(v_local)                # weights on the simplex
            a = basis @ w                         # mixed shape (still ∑ a = 1)
            conv = _conv_fft(a, a)
            shift = np.max(conv)                  # numeric stabiliser
            exp_b = np.exp(beta_val * (conv - shift))
            sum_exp = exp_b.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (same as in the original script)
            rev_a = a[::-1]
            soft_w = exp_b / sum_exp
            full = _conv_fft(rev_a, soft_w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule: grad_w = Bᵀ grad_a
            grad_w = basis.T @ grad_a

            # chain rule through softmax
            dot = np.dot(grad_w, w)
            grad_v = w * (grad_w - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[Weight L‑BFGS] exception {e} at β={beta_val}")
            break

    w_opt = _softmax(v)
    return w_opt


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the full vector (exact max)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a: np.ndarray,
                               n: int,
                               deadline: float,
                               step: float = 0.05,
                               max_iters: int = 1200) -> np.ndarray:
    """
    A few hundred iterations of a projected sub‑gradient descent that uses the
    exact max‑convolution index as the active constraint.
    """
    cur = a.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(cur, cur)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) is 2 * a shifted by k
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[partner[mask]]

        # step opposite to the sub‑gradient
        cur -= step * grad
        cur = _normalize(cur)

        # slowly shrink the step size
        if (it + 1) % 200 == 0:
            step *= 0.7

    return cur


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a time budget, prints progress information, and stores
    the best vector in the global variable ``height_sequence_1`` for a warm start.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final ``evaluate_sequence`` call
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Problem size (kept identical to the previous best)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 1. Build a modest library of normalised analytic shapes
    # --------------------------------------------------------------
    basis = []

    # simple deterministic shapes
    basis.append(_uniform(n))
    basis.append(_triangular(n))

    for gamma in np.linspace(0.5, 4.0, 8):
        basis.append(_sine_window(n, gamma))

    # a couple of Gaussians
    for sigma in (0.12, 0.20):
        basis.append(_gaussian(n, sigma))

    for p in (0.5, 1.0, 2.0, 3.0):
        basis.append(_power(n, p))
        basis.append(_double_power(n, p))

    basis.append(_arcsine_window(n))

    for gamma in np.linspace(0.5, 4.0, 6):
        basis.append(_cosine_edge_window(n, gamma))

    # random beta‑shape draws (U‑shaped often)
    for _ in range(15):
        a = 10 ** rng.uniform(-2, 0)          # 0.01 … 1
        b = 10 ** rng.uniform(-2, 0)
        basis.append(_beta_shape(n, a, b))

    # two‑block uniform shapes with random geometry
    for _ in range(12):
        blk = rng.uniform(0.04, 0.35)
        lw = rng.random()
        basis.append(_two_block_uniform(n, block_ratio=blk, left_weight=lw))

    # a few low‑frequency random curves to enrich the space
    for _ in range(10):
        nf = rng.integers(2, 6)
        basis.append(_lowfreq_curve(n, nf, rng))

    # include the previously discovered solution if it matches the size
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and np.sum(arr) > 0:
                basis.append(_normalize(arr))
        except Exception:
            pass

    basis = np.stack(basis, axis=1)                # shape (n, m)
    m = basis.shape[1]
    print(f"[Setup] built basis of {m} shapes for n={n}")

    # --------------------------------------------------------------
    # 2. Seed the weight space (simplex of size m)
    # --------------------------------------------------------------
    top_k = 25
    heap = []                                     # stores (-value, w)

    def _push(w):
        val = _objective_surrogate(basis @ w)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, w))
        else:
            if val < -heap[0][0]:
                heapq.heappushpop(heap, (-val, w))

    # deterministic uniform mixture
    _push(np.full(m, 1.0 / m))

    # Dirichlet draws – a few concentrations encourage sparsity
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(200):
            w = _random_dirichlet(m, conc, rng)
            _push(w)

    # exponential draws (another simple Dirichlet‑like distribution)
    for _ in range(200):
        w = _random_exponential(m, rng)
        w = _normalize(w)
        _push(w)

    # --------------------------------------------------------------
    # 3. Pick the best few weight vectors and run the heavy optimisation
    # --------------------------------------------------------------
    seeds = [vec for _, vec in sorted(heap, key=lambda x: -x[0])]
    print(f"[Seeding] best surrogate among mixtures = "
          f"{_objective_surrogate(basis @ seeds[0]):.6f}")

    best_vec = basis @ seeds[0]
    best_val = _objective_surrogate(best_vec)
    print(f"[Init] global best surrogate = {best_val:.6f}")

    max_refine = min(5, len(seeds))
    for idx in range(max_refine):
        if time.time() > deadline - 3.0:
            break
        w0 = seeds[idx]
        remaining = deadline - time.time()
        refine_deadline = time.time() + max(5.0, remaining * 0.5 / max_refine)

        try:
            w_opt = _lbfgs_optimize_weights(w0, basis, n, refine_deadline)
        except RuntimeError:
            w_opt = w0

        vec_opt = basis @ w_opt
        val_opt = _objective_surrogate(vec_opt)
        if val_opt < best_val:
            best_val = val_opt
            best_vec = vec_opt.copy()
            print(f"[Refine {idx+1}] improved surrogate to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Fine‑grained projected sub‑gradient descent on the full vector
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Projected gradient fine‑tune ---")
        cur = _projected_gradient_descent(best_vec, n, deadline,
                                          step=0.05, max_iters=1200)
        cur_val = _objective_surrogate(cur)
        if cur_val < best_val:
            best_val = cur_val
            best_vec = cur
            print(f"[PGD] improved surrogate to {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Evaluate with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
