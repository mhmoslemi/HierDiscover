"""
Improved step‑function optimizer.

The design centres on a fast *gradient‑move* local search that repeatedly
shifts a little probability mass from the indices that most increase the
current maximal convolution (the active constraint) to the indices that
least increase it.  The move is performed directly on the probability
vector (the ℓ₁‑normalised step heights) and the objective
    2·n·max(conv(a,a)) / (sum(a)²)
is equal to the surrogate ``2·n·max(conv)`` because we keep ``sum(a)=1``.
The algorithm works as follows:

1. Build a modest pool of analytical seeds (uniform, triangular, Gaussian,
   power‑law, arcsine) and a few random Dirichlet draws.
2. Keep the best few seeds according to the exact surrogate.
3. For each kept seed run a *local gradient move* optimiser:
   * compute the active convolution index,
   * build the sub‑gradient ``g_i = 2·a_{k‑i}``,
   * move a small fraction of probability mass from the index with the
     largest ``g`` to the one with the smallest ``g``,
   * accept the move only if the surrogate value improves,
   * adapt the step size automatically.
   This simple “mass‑transfer” descent rapidly drives the surrogate
   down and is able to find improvements beyond the long exponentiated‑
   gradient run used previously.
4. The best result is polished with a short L‑BFGS optimisation on a
   smooth log‑sum‑exp surrogate (as in the reference implementation).
5. Finally a brief simulated‑annealing walk refines the solution.
6. The best vector is returned (and stored globally as ``height_sequence_1``).

The whole routine respects the 1000 s budget and prints occasional progress
messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for L‑BFGS‑B (used only for polishing)

try:

except Exception:  # pragma: no cover

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exact surrogate (objective when sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _max_conv_and_gradient(seq: np.ndarray):
    """Return (max_conv, sub‑gradient of the max‑convolution)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    k = int(np.argmax(conv))
    grad = np.zeros_like(seq)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    rev = k - idx
    grad[idx] = 2.0 * seq[rev]
    return float(conv[k]), grad


# ----------------------------------------------------------------------
# Local “mass‑transfer” optimiser
# ----------------------------------------------------------------------
def _local_optimize(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple gradient‑move optimiser.
    Repeatedly moves a small fraction of probability mass from the
    index with the largest sub‑gradient to the one with the smallest,
    accepting only improving moves.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # adaptive step size (as a fraction of the source mass)
    step_frac = 0.05
    it = 0

    while time.time() < deadline:
        it += 1
        max_val, g = _max_conv_and_gradient(a)

        src = int(np.argmax(g))
        tgt = int(np.argmin(g))

        if a[src] <= 1e-12:
            # nothing to move – stop this run
            break

        delta = min(step_frac * a[src], a[src])

        a[src] -= delta
        a[tgt] += delta

        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # success – we may increase the step a bit
            step_frac = min(step_frac * 1.02, 0.5)
        else:
            # revert and shrink step
            a[src] += delta
            a[tgt] -= delta
            step_frac *= 0.5
            if step_frac < 1e-8:
                break

        # occasional renormalisation to fight numerical drift
        if it % 500 == 0:
            total = a.sum()
            if abs(total - 1.0) > 1e-12:
                a /= total

        # progress report
        if it % 2000 == 0:
            print(f"[LocalOpt] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth surrogate)
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 1500) -> np.ndarray:
    """Polish ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a log‑sum‑exp surrogate."""
    if _opt is None:  # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1, 5, 20, 80, 200,
            500, 1_000, 2_000, 5_000,
            20_000, 50_000, 100_000,
            200_000, 500_000, 1_000_000,
        ]

    v = np.log(start_vec + 1e-30)  # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)  # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)  # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Simulated‑annealing refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

        remaining = deadline - time.time()
        max_sa = int(min(400_000, remaining * 2000))
        max_sa = max(100_000, max_sa)
        print("\n--- Simulated‑Annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_opt(best_seq, n, deadline,
                                                  max_iters=max_sa)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] final surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
