def generate_seeds_for_n(n, rng):
    seeds = []
    # Uniform
    seeds.append(_uniform(n))

    # Triangular
    seeds.append(_triangular(n))

    # Exponential decay rates logspace
    for rate in np.logspace(-4, -0.5, 8):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Gaussian sigma_frac
    for sigma_frac in np.linspace(0.05, 0.35, 10):
        seeds.append(_gaussian(n, sigma_frac))

    # Sine window gamma
    for gamma in np.linspace(0.2, 3.0, 10):
        seeds.append(_sine_window(n, gamma))

    # Power law
    for p in np.linspace(0.0, 4.0, 10):
        seeds.append(_power(n, p))

    # Beta shape random
    for _ in range(12):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws
    for conc in [0.02, 0.05, 0.1, 0.25, 0.5, 1.0, 2.0]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Exponential random draws
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # Low frequency curves
    for nf in [2,3,4,5]:
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mix uniform + lowfreq
    for mix in [0.25, 0.5, 0.75]:
        for nf in [2,4]:
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix))

    # Permutations of some base shapes
    base = [_exp_decay(n, 0.01), _sine_window(n, 1.5), _beta_shape(n,2.0,2.0)]
    for b in base:
        for _ in range(2):
            seeds.append(np.random.permutation(b))

"""
Propose a step‑function (list of heights) that minimises
    evaluate_sequence(seq)
using a multi‑scale random‑seed generation + mirror‑descent + L‑BFGS
refinement strategy.

The routine works inside a fixed time budget (default 1000 s).  It
explores several problem sizes (n = 512, 1024, 2048, 4096), creates a
large and diverse pool of ℓ₁‑normalised seed vectors from analytic families,
keeps the best few seeds (across all n) and refines each of them with

* exponentiated‑gradient (mirror‑descent) on the exact max‑convolution,
* continuation L‑BFGS optimisation on a smooth log‑sum‑exp surrogate,
* a short stochastic hill‑climbing phase.

The best refined vector is polished with a long EG run and finally
evaluated with the provided `evaluate_sequence`.  The best sequence is
stored in the global variable ``height_sequence_1`` for possible reuse.

All helper functions are top‑level (no closures, no lambdas) and respect
the time budget.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (O(N log N)).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a vector with ℓ₁‑norm = 1:
        2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Linear combination of a few low‑frequency cosine waves, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    if s > 0.0:
        return mixed / s
    return a.copy()


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

# Optimisation primitives

def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Works in log‑space to preserve positivity and ℓ₁‑norm.
    Returns best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑space representation
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Continuation L‑BFGS on a smoothed max‑convolution objective.
    Optimisation is performed in log‑softmax space to guarantee feasibility.
    """
    if _opt is None:
        # SciPy missing – fall back to the input vector
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c  # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smoothed max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smoothed max w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # first n entries of the central part

            # convert to gradient w.r.t. logits v
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:  # pragma: no cover
            break

    return _softmax(v)


def _hill_climbing(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   max_attempts: int = 200_000,
                   sigma_start: float = 0.005,
                   sigma_decay: float = 0.9995):
    """
    Simple random‑perturbation hill‑climbing.
    Returns best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    sigma = sigma_start
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s
        val = _objective_surrogate(ns)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns
        sigma *= sigma_decay
        attempts += 1

    return best_seq, best_val

def _generate_seeds_for_n(n: int, rng: np.random.Generator):
    """
    Produce a diverse list of ℓ₁‑normalised candidates for a given length n.
    """
    seeds = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Sine windows (γ sweep)
    for gamma in np.linspace(0.1, 3.5, 12):
        seeds.append(_sine_window(n, gamma))

    # Gaussian windows (σ fraction sweep)
    for sigma_frac in np.linspace(0.05, 0.35, 12):
        seeds.append(_gaussian(n, sigma_frac))

    # Power‑law families
    for p in np.linspace(0.0, 4.0, 12):
        seeds.append(_power(n, p))

    # Exponential decay (rate sweep)
    for rate in np.logspace(-4, -0.5, 10):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Beta‑shape random draws
    for _ in range(20):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Pure exponential draws (no analytical rate)
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform and low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Random permutations of a few base shapes
    base_shapes = [_exp_decay(n, 0.01), _sine_window(n, 1.5), _beta_shape(n, 2.0, 2.0)]
    for base in base_shapes:
        for _ in range(2):
            seeds.append(np.random.permutation(base))

    return seeds

        except Exception:
            prev_arrs = {}

    # 1. Generate a large pool of seeds (quick surrogate evaluation)

        # Allocate a slice of remaining time for this seed
        remaining = deadline - time.time()
        # keep 15 % for each upcoming seed + final polish

        if local_val < global_best_val:
            global_best_val = local_val
            global_best_seq = local_seq.copy()
            print(f"    *** New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing of the overall best candidate
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 len(global_best_seq),
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"    *** Improved after final EG: {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Compute the true objective with the provided function
    # ------------------------------------------------------------------
    true_val = None
    if global_best_seq is not None:
        try:
            true_val = evaluate_sequence(global_best_seq.tolist())
            print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
                  f"true evaluate_sequence = {true_val:.6f}")
        except Exception as exc:  # pragma: no cover
            print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the winner for possible reuse
    globals()["height_sequence_1"] = (global_best_seq.tolist()
                                      if global_best_seq is not None else [])

    return global_best_seq.tolist() if global_best_seq is not None else []
