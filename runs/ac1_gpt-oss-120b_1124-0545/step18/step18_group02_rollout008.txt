"""
Propose a step‑function that minimises ``evaluate_sequence``.

Strategy
--------
* Use a richer pool of seed shapes (including heavy‑tailed Beta(α,β) with
  α,β∈(0,1) and a finer power‑law sweep).
* Run a continuation L‑BFGS on a log‑sum‑exp smooth approximation of the
  max‑convolution objective – the β‑schedule is longer and the per‑β
  iteration budget is larger.
* Refine the best candidates with an Adam‑style mirror‑descent optimiser
  that works on the exact non‑smooth objective (sub‑gradients of the
  active lag are used).  The Adam step‑size is annealed as 1/√t.
* Perform a very long stochastic mass‑transfer hill‑climb.
* End with a short high‑β L‑BFGS polishing pass.

The whole search respects the 1000 s wall‑time limit and stores the best
vector in the global ``height_sequence_1`` variable.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of 2·n·max(conv) / (∑a)².
    Returns (obj, grad) – gradient w.r.t. logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)              # full convolution
    shift = np.max(conv)                # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see derivation in the original code)
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0, 50_000_000.0, 100_000_000.0,
            5_000_000_000.0, 1e10,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)

        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 150_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam‑style mirror descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _adam_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on logits (mirror descent) using sub‑gradients."""
    a = start_a.copy()
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (1/√t) multiplied by Adam’s adaptive factor
        step = eta0 / (np.sqrt(it) + 1.0)
        log_a -= step * (m_hat / (np.sqrt(v_hat) + eps))

        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic mass‑transfer local search
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 120_000,
    step_frac: float = 0.011,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on {0,…,n‑1} (ℓ₁‑normalised)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Seed‑pool construction (augmented with extra beta/power‑law families)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # extra heavy‑tailed beta families (α,β < 1)
    for a, b in [(0.5, 0.5), (0.3, 0.3), (0.7, 0.7), (0.4, 0.6), (0.6, 0.4)]:
        seeds.append(_beta_shape(n, a, b))
        seeds.append(_beta_shape(n, b, a))

    # power‑law sweep – extended range
    p_vals = np.linspace(0.2, 1.0, 9)            # more exponents
    shifts = [0.0, 2.0, 5.0, 15.0, 40.0, 80.0, 120.0]
    for p in p_vals:
        for sh in shifts:
            vec = _power_law(n, float(p), float(sh))
            seeds.append(vec)
            seeds.append(vec[::-1])

    # random Dirichlet families (incl. very sparse)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.7, 1.5):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency curves
    for nf in (2, 3, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures of existing seeds
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – we pick a power‑of‑two for faster FFTs
    # --------------------------------------------------------------
    n = 4096                                 # a bit larger than before
    target_sum = np.sqrt(2.0 * n)

    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # --------------------------------------------------------------
    # Seed pool (including previous best if available)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # --------------------------------------------------------------
    # Evaluate seeds, keep the elite pool
    # --------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        scaled = v * target_sum
        val = _objective_exact(scaled)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    # keep the top‑k seeds for heavy L‑BFGS stage
    seeds.sort(key=lambda vec: _objective_exact(vec * target_sum))
    elite = seeds[:min(30, len(seeds))]      # a larger elite set

    global_best_seq = best_seq * target_sum
    global_best_val = best_val

    # --------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on smooth surrogate (aggressive)
    # --------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 50_000_000.0, 100_000_000.0,
        5_000_000_000.0, 1e10,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break

        seed_vec = seed_vec * target_sum
        # adapt per‑β budget according to remaining time
        remaining = deadline - time.time()
        per_beta = max(1200, int(min(4000, remaining * 0.05 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β≈{per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – Adam‑style mirror descent (multiple restarts)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 4
        iters_per_restart = int(min(1_500_000, remaining * 0.13 / restarts))
        print("\n--- Adam‑mirror descent restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            # start from a fresh random Dirichlet (conc < 1 favours sparsity)
            conc = rng.uniform(0.01, 0.9)
            start_vec = _random_dirichlet(n, conc, rng) * target_sum
            seq_adam, val_adam = _adam_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=1.5,
            )
            print(f"[Adam {r + 1}/{restarts}] best = {val_adam:.6f}")
            if val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Update] New best after Adam = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – long exponentiated‑gradient run from current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(5_000_000, remaining * 0.30 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=2.5,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – aggressive stochastic local search
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(250_000, remaining * 60))
        if iters > 0:
            print("\n--- Stochastic local search (mass transfers) ---")
            seq_loc, val_loc = _local_search(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.009,
            )
            print(f"[LocalSearch] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – final high‑β polishing L‑BFGS
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(1200, int(min(4000, remaining * 0.04 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000.0, 10_000_000.0, 50_000_000.0, 100_000_000.0, 5_000_000_000.0, 1e10],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (safety wrapper)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
