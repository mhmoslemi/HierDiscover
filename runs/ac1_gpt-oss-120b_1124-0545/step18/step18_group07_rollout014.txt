def _exp_gradient_adagrad_opt(start_seq, n, deadline, max_iters=500_000, base_eta=0.5, eps=1e-8):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    g_sq = np.zeros_like(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))
        g = np.zeros_like(a)
        i_start = max(0,kmax-(n-1))
        i_end = min(n-1,kmax)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # AdaGrad update
        g_sq += g**2
        eta_t = base_eta / (np.sqrt(g_sq) + eps)
        log_a = log_a - eta_t * g
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _perm_swap_anneal(start_seq, n, deadline, max_swaps=2_000_000, init_temp=0.1, final_temp=1e-4):
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    swaps = 0
    while swaps < max_swaps and time.time() < deadline:
        # Temperature schedule linear or exponential based on progress
        progress = swaps / max_swaps
        T = init_temp * (1 - progress) + final_temp * progress
        i, j = rng.integers(0,n, size=2)
        if i == j:
            continue
        # Swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        swaps += 1
        if swaps % 200_000 == 0:
            print(f"[PermSwap SA] swaps={swaps} best={best_val:.6f}")
    return best_seq, best_val

"""
Propose a high‑quality step‑function sequence.

The optimisation proceeds in several phases:

1. **Broad seed generation** – dozens of analytic families (beta‑shape,
   power‑law, windows, Gaussians, random Dirichlet draws, etc.) together
   with many randomised parameterisations.  All seeds are ℓ₁‑normalised

   we run mirror‑descent with per‑coordinate AdaGrad step‑sizes.  This
   converges faster than the simple 1/√t schedule used previously.

5. **Permutation simulated‑annealing** – a large number of random swaps
   are performed.  Swaps that improve the objective are always accepted,
   otherwise they are accepted with a temperature‑dependent probability.
   This explores the ordering of the entries more thoroughly than a pure
   hill‑climber.

6. **Final evaluation** – the best sequence found is stored in the global
   ``height_sequence_1`` variable and returned as a plain list.  The
   official ``evaluate_sequence`` function is called once at the end to
   report the true objective value.

All heavy work respects a global 1000 s wall‑clock budget; frequent
time‑checks guarantee termination before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for ℓ₁‑normalised vectors:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _edge_bias(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution concentrating mass near the ends."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection – massive, diverse and randomly parametrised
# ----------------------------------------------------------------------
def _collect_seeds_extensive(n: int,
                             rng: np.random.Generator,
                             previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a large, diverse pool of ℓ₁‑normalised seeds."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    # parametrised families – wide grid / random sampling
    # sine window
    for gamma in np.linspace(0.2, 8.0, 12):
        seeds.append(_sine_window(n, gamma))

    # Gaussian
    for sigma in np.linspace(0.07, 0.45, 10):
        seeds.append(_gaussian(n, sigma))

    # double Gaussian (few combos)
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.20, shift_frac=0.30))

    # Kaiser window
    for beta in np.linspace(3, 30, 8):
        seeds.append(_kaiser_window(n, beta))

    # Power law
    for p in np.linspace(0.0, 10.0, 13):
        seeds.append(_power(n, p))

    # Beta‑shape – random log‑uniform sampling
    for _ in range(80):
        a = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        b = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, a, b))

    # Tukey window
    for alpha in np.linspace(0.05, 0.95, 10):
        seeds.append(_tukey_window(n, alpha))

    # Edge‑bias
    for p in np.linspace(0.2, 2.0, 10):
        seeds.append(_edge_bias(n, p))

    # Low‑frequency random curves
    for nf in (1, 2, 3):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.6):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Sparse random vectors (very few spikes)
    for k_frac in (0.001, 0.002, 0.005):
        k = max(1, int(k_frac * n))
        for _ in range(3):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors
    for bf in (0.02, 0.05, 0.10, 0.20):
        for _ in range(2):
            seeds.append(_block_uniform(n, bf, rng))

    # Alternating Dirichlet (helps later permutation)
    seeds.append(_alternating_dirichlet(n, rng))

    # Heavy‑tailed Dirichlet and its permutations
    heavy = _random_dirichlet(n, 0.05, rng)
    for _ in range(5):
        perm = heavy[rng.permutation(n)]
        seeds.append(perm)
        seeds.append(perm[::-1])      # reversed version

    # Mixtures of families – random convex combinations
    for _ in range(30):
        a = seeds[rng.integers(len(seeds))]
        b = seeds[rng.integers(len(seeds))]
        alpha = rng.random()
        seeds.append(_mix_shapes(a, b, alpha))

    # Previous solution (if any) – both orientation
    if previous is not None and previous.size == n and previous.sum() > 0:
        prev_norm = previous / previous.sum()
        seeds.append(prev_norm)
        seeds.append(prev_norm[::-1])

    # Normalise and sanity‑check
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    # Limit the pool size (avoid pathological blow‑up)
    if len(cleaned) > 20000:
        rng.shuffle(cleaned)
        cleaned = cleaned[:20000]
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised probability vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6, 1e7
        ]

    # work in log‑probability space (stay on simplex via softmax)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] Exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# AdaGrad exponentiated‑gradient optimiser (exact max)
# ----------------------------------------------------------------------
def _exp_gradient_adagrad_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 500_000,
                             base_eta: float = 0.5,
                             eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution with per‑coordinate AdaGrad.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    g_sq = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # AdaGrad update on log‑prob space
        g_sq += g * g
        eta_vec = base_eta / (np.sqrt(g_sq) + eps)
        log_a = log_a - eta_vec * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress print
        if it % 100_000 == 0:
            print(f"[AdaGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation simulated‑annealing hill‑climber
# ----------------------------------------------------------------------
def _perm_swap_anneal(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int = 1_500_000,
                     init_temp: float = 0.05,
                     final_temp: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; accept improvements always and
    occasional worsening moves according to a temperature schedule.
    Returns the best sequence discovered and its surrogate value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        # linear temperature schedule
        T = init_temp * (1 - it / max_swaps) + final_temp * (it / max_swaps)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective_surrogate(cur_seq)

        # acceptance test
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            if rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
                cur_val = new_val
                # keep the worsening move (but still update best if by chance)
                if new_val < best_val:
                    best_val = new_val
                    best_seq = cur_seq.copy()
            else:
                # revert swap
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 250_000 == 0:
            print(f"[PermSA] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the allotted time.
    """
    # ------------------------------------------------------------------
    # 0. RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Load previous solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list):
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 3. Seed collection (very broad)
    # ------------------------------------------------------------------
    print(f"\n--- Seed collection (n={n}) ---")
    seeds = _collect_seeds_extensive(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seeds.")

    # ------------------------------------------------------------------
    # 4. Surrogate evaluation – keep the top few
    # ------------------------------------------------------------------
    print("\n--- Evaluating surrogate for all seeds ---")
    sur_vals = [_objective_surrogate(s) for s in seeds]
    sorted_idx = np.argsort(sur_vals)
    top_k = 15
    top_seeds = [seeds[i] for i in sorted_idx[:top_k]]
    print(f"Best surrogate values among seeds:")
    for rank, idx in enumerate(sorted_idx[:top_k]):
        print(f"  #{rank+1:2d}: {sur_vals[idx]:.6f}")

    # ------------------------------------------------------------------
    # 5. L‑BFGS refinement on the elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # beta schedule (same as in the helper)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1e3, 2e3, 5e3,
        1e4, 2e4, 5e4, 1e5,
        2e5, 5e5, 1e6, 2e6, 5e6, 1e7
    ]

    for seed_idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing all L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        # give each seed a generous budget but keep overall runtime bounded
        max_iter = int(min(20000, remaining * 0.04))
        max_iter = max(8000, max_iter)

        print(f"\n--- L‑BFGS optimisation on seed #{seed_idx+1} (max_iter/beta={max_iter}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. AdaGrad exponentiated‑gradient polishing (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 0.80))
        max_iters = max(200_000, max_iters)
        print(f"\n--- AdaGrad polishing (max_iters={max_iters:,}) ---")
        seq_adagrad, val_adagrad = _exp_gradient_adagrad_opt(global_best_seq,
                                                             n,
                                                             deadline,
                                                             max_iters=max_iters)
        if val_adagrad < global_best_val:
            global_best_seq = seq_adagrad.copy()
            global_best_val = val_adagrad
            print(f"[AdaGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Permutation simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(1_500_000, remaining * 0.30))
        max_swaps = max(500_000, max_swaps)
        print(f"\n--- Permutation SA (max_swaps={max_swaps:,}) ---")
        seq_perm, val_perm = _perm_swap_anneal(global_best_seq,
                                               n,
                                               deadline,
                                               max_swaps=max_swaps,
                                               init_temp=0.05,
                                               final_temp=1e-5)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[PermSA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
