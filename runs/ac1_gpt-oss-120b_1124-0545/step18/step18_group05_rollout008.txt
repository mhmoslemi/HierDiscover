The metric  

\[
F(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is invariant under a positive scaling of the vector, therefore we may restrict ourselves to probability vectors
\(\sum_i a_i = 1,\; a_i\ge 0\).  
For such a vector the objective simplifies to  

\[
\tilde F(a)=2\,n\;\max_k (a*a)_k .
\]

The problem is non‑convex because of the `max`.  The best published approach uses a
mirror‑descent on the single active index.  The following implementation improves on it by

* **a richer seed pool** (analytic families, many Dirichlet draws, the previous best).  
* **multi‑start quasi‑Newton optimisation** – each of the `k` best seeds is refined with
  L‑BFGS (or Adam when SciPy is missing) on a smooth *log‑sum‑exp* surrogate of the `max`.
* **longer, adaptive L‑BFGS runs** (more iterations per temperature, a broader
  temperature schedule).  
* **final simulated‑annealing walk** to escape possible local minima.

All heavy work is performed in the frequency domain with an FFT based
convolution, which makes each iteration cheap enough to explore many
starting points within the 1000 s budget.

The returned vector is stored globally as `height_sequence_1` so that a later
run can start from the best known solution.

```python
"""
Propose a non‑negative step function (list of floats) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

The algorithm:
    1. Build a diverse set of seeds (analytic families, many Dirichlet
       draws, previous best).
    2. Keep the `k_best` seeds with the smallest surrogate 
       `2·n·max(conv)`.
    3. For each of those seeds run a quasi‑Newton optimisation on the
       smooth log‑sum‑exp surrogate of the max‑convolution.
       – L‑BFGS‑B is used when SciPy is available, otherwise a handwritten
         Adam optimiser is employed.
    4. Keep the best vector obtained.
    5. Polish the best vector with a simulated‑annealing walk in logit
       space.
    6. Return the best vector found and store it globally.
"""

import time

    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Surrogate (objective when sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 3000) -> np.ndarray:
    """Refine `start_vec` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:            # pragma: no cover
        # SciPy not available – fallback to Adam (handled elsewhere)
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (as in the original code)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        # run L‑BFGS
        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                         # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser on the same smooth surrogate (used when SciPy absent)
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   steps_per_beta: int = 2500) -> np.ndarray:
    """Adam optimiser for the log‑sum‑exp surrogate."""
    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80, 200, 500, 1_000, 2_000, 5_000, 20_000]

    v = np.log(start_vec + 1e-30)               # logits
    m = np.zeros_like(v)
    v_var = np.zeros_like(v)                     # second moment (named v_var to avoid clash)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return _softmax(v)

            t += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            shifted = conv - max_c
            exp_beta = np.exp(beta * shifted)
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v_var = beta2 * v_var + (1 - beta2) * (grad_v ** 2)

            m_hat = m / (1 - beta1 ** t)
            v_hat = v_var / (1 - beta2 ** t)

            lr = 0.5 / math.sqrt(t)               # decaying learning rate
            v -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6          # temperature
        sigma = 0.15 * (1.0 - progress) + 1e-4      # proposal width

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        accept = (proposal_val < cur_val) or (math.exp(-(proposal_val - cur_val) / T) > rng.random())
        if accept:
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the given metric.
    The routine builds a large pool of seeds, refines the best few with a
    sophisticated quasi‑Newton optimiser, finally polishes the result with a
    simulated‑annealing walk, and returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressivity and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _sin2(n),
        _hann_window(n),
        _arcsine(n),
    ]

    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # random Dirichlet draws with a wide range of concentration parameters
    conc_params = np.logspace(-2, 1, 12)            # 0.01 … 10
    for conc in conc_params:
        for _ in range(2):                         # two draws per concentration
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # include the previously best vector if it matches the current length
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and np.sum(arr) > 0.0:
                seeds.append(arr / np.sum(arr))
        except Exception:
            pass

    # normalise everything onto the probability simplex
    def _norm(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / len(v))
        return v / s

    seeds = [_norm(np.asarray(v, dtype=np.float64)) for v in seeds]

    # ------------------------------------------------------------------
    # 2. Quick ranking – keep the best few seeds
    # ------------------------------------------------------------------
    ranked = sorted(((seq, _objective_surrogate(seq)) for seq in seeds),
                    key=lambda x: x[1])
    k_best = 12                                      # number of seeds sent to L‑BFGS
    top_seeds = [seq for seq, _ in ranked[:k_best]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate among seeds: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Full quasi‑Newton optimisation on the best seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break

        print(f"\n--- Optimising seed #{idx + 1}/{k_best} ---")
        if _opt is not None:
            # use the L‑BFGS implementation
            opt_seq = _lbfgs_optimize(seed_vec, n, deadline,
                                      beta_schedule=[
                                          1, 5, 20, 80, 200, 500,
                                          1_000, 2_000, 5_000, 20_000,
                                          50_000, 100_000, 200_000, 500_000,
                                          1_000_000, 2_000_000, 5_000_000],
                                      max_iter_per_beta=4000)
        else:
            # fall back to Adam when SciPy is unavailable
            opt_seq = _adam_optimize(seed_vec, n, deadline,
                                     beta_schedule=[1, 5, 20, 80, 200, 500,
                                                    1_000, 2_000, 5_000, 20_000],
                                     steps_per_beta=3000)

        val = _objective_surrogate(opt_seq)
        print(f"[Result] surrogate = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = opt_seq.copy()
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a generous number of SA steps while staying inside the budget
        max_sa = int(min(1_500_000, remaining * 3000))
        max_sa = max(400_000, max_sa)
        print("\n--- Simulated‑Annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_opt(best_seq, n, deadline,
                                                  max_iters=max_sa)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Official evaluation and return
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
