# FFT based convolution utilities ------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater or equal to x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective assuming sum(seq)=1 (the denominator is constant)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Softmax mapping ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# Analytic families (all return ℓ₁‑normalised vectors) --------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        return a / s
    return _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    if s > 0:
        return a / s
    return _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s > 0:
        return a / s
    return _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s > 0:
        return a / s
    return _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s > 0:
        return a / s
    return _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    if s > 0:
        return a / s
    return _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # modified Bessel I0
    s = a.sum()
    if s > 0:
        return a / s
    return _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    if s > 0:
        return a / s
    return _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        return a / s
    return _uniform(n)


def _mix_two(v1: np.ndarray, v2: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * v1 + alpha * v2
    s = mixed.sum()
    if s > 0:
        return mixed / s
    return _uniform(len(v1))


def _collect_seeds(n: int, rng: np.random.Generator, prev: np.ndarray | None) -> list[np.ndarray]:
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for a in (0.3, 1.0, 2.0, 4.0):
        for b in (0.3, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))
    for beta in (2, 4, 8, 12, 16, 20):
        seeds.append(_kaiser_window(n, beta))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random mixtures of two already‑generated families
    base = seeds.copy()
    for _ in range(30):
        a, b = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_two(a, b, alpha))

    # previous best if supplied
    if prev is not None and prev.shape == (n,):
        seeds.append(prev)

    # normalise sanity – each vector is guaranteed ℓ₁‑normalised
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Smooth‑max surrogate optimisation --------------------------------------- 
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """
    Optimise the log‑sum‑exp smooth approximation of the max‑convolution
    with L‑BFGS‑B (if SciPy is available).  The optimisation works in the
    log‑softmax (log‑probability) space so the ℓ₁‑norm is automatically
    preserved.
    """
    if _opt is None:
        # fall back to a simple gradient‑descent loop (Adam)
        return _adam_smooth_optimize(start_vec, n, deadline, beta_schedule, max_iter_per_beta)

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6, 1e7, 2e7, 5e7, 1e8
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (derived from the smoothed max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _adam_smooth_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          beta_schedule=None,
                          max_iter_per_beta: int = 12000) -> np.ndarray:
    """
    Simple Adam optimiser for the smooth‑max surrogate.  Used when SciPy
    is not available.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6, 1e7, 2e7, 5e7, 1e8
        ]

    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    t = 0
    eps = 1e-8
    beta1 = 0.9
    beta2 = 0.999

    for beta in beta_schedule:
        beta_val = float(beta)

        # We'll run a fixed number of inner steps for each beta.
        for _ in range(max_iter_per_beta):
            t += 1
            if time.time() > deadline:
                return _softmax(v)

            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

            # gradient of smooth max w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update on log‑probabilities
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
            m_corr = m / (1 - beta1 ** t)
            v_corr = v_hat / (1 - beta2 ** t)
            eta = 0.5 / math.sqrt(t)  # decreasing stepsize
            v = v - eta * m_corr / (np.sqrt(v_corr) + eps)

    return _softmax(v)

# Exact‑max exponentiated‑gradient optimiser (Adam version) ---------------
def _adam_exact_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Exponentiated Gradient) on the true max‑convolution,
    enhanced with Adam‑style momentum for faster convergence.
    Returns the best sequence found and its exact objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v_hat = np.zeros_like(log_a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t = 0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag (exact max)
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam‑style momentum on the log‑space gradient
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * (g ** 2)
        m_corr = m / (1 - beta1 ** t)
        v_corr = v_hat / (1 - beta2 ** t)
        eta = 0.6 / math.sqrt(t)   # step‑size schedule

        log_a = log_a - eta * m_corr / (np.sqrt(v_corr) + eps)
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 200_000 == 0:
            print(f"[AdamExact] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Linear‑programming refinement (conv(f,g) ≤ M) -------------------------
def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_steps: int = 8) -> np.ndarray:
    """
    Perform a few LP refinement steps.  In each step we solve

        max   sum(g)
        s.t.  conv(f, g) ≤ M,  g ≥ 0

    where M = max(conv(f,f)).  The solution g is normalised back to
    ℓ₁‑norm 1 and mixed with f (simple convex combination with t=0.5).
    The process is repeated a few times.
    """
    if cp is None:
        return f  # cannot run LP without cvxpy

    g_var = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n, value=f)

    # interior‑point warm start helper
    prob = None

    for step in range(max_steps):
        if time.time() > deadline:
            break

        # compute current bound M
        conv_ff = _conv_fft(f, f)
        M = float(np.max(conv_ff))

        # define convolution expression
        conv_fg = cp.conv(f_param, g_var)   # length 2n‑1
        constraints = [conv_fg <= M]

        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iter=200)
        except Exception as e:
            print(f"[LP] solver exception: {e}")
            break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            print("[LP] no optimal solution, status:", prob.status)
            break

        g_opt = np.asarray(g_var.value, dtype=np.float64)
        if g_opt is None or np.allclose(g_opt, 0):
            break

        # normalise g_opt to sum 1
        g_sum = g_opt.sum()
        if g_sum <= 0:
            break
        g_norm = g_opt / g_sum

        # convex blend and re‑normalise
        f = 0.5 * f + 0.5 * g_norm
        f = f / f.sum()

    return f

# Simulated‑annealing swap hill‑climber ---------------------------------
def _anneal_swap_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int = 400_000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T0 = 0.1 * cur_val if cur_val > 0 else 1.0

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective(cur_seq)
        delta = new_val - cur_val

        T = max(T0 * (1.0 - it / max_swaps), 1e-12)
        accept = delta < 0 or rng.random() < math.exp(-delta / T)

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 100_000 == 0:
            print(f"[AnnealSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Symmetrisation --------------------------------------------------------
def _symmetrize(seq: np.ndarray) -> np.ndarray:
    mixed = (seq + seq[::-1]) * 0.5
    s = mixed.sum()
    if s > 0:

"""
Propose a step function (non‑negative vector) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / sum(seq)**2

The algorithm proceeds as follows:

1. **Seed generation** – a wide pool of ℓ₁‑normalised vectors:
   analytic windows (uniform, triangular, sine, Gaussian, beta,
   Tukey, Kaiser), random Dirichlet / exponential draws and random
   convex mixtures of two seeds.  The previously best vector (if any) is
   also injected.

2. **Quick evaluation** – keep the top K seeds after a cheap exact
   evaluation of the objective.

3. **Smooth‑max L‑BFGS (or Adam)** – minimise a log‑sum‑exp surrogate of
   `max(conv)` while staying on the simplex (optimisation in log‑softmax
   space).  The inverse temperature β is increased gradually.

4. **Exact‑max Adam‑style exponentiated‑gradient** – mirror‑descent on the
   true max‑convolution with Adam momentum, again in log‑space.

5. **LP refinement** – for the current shape `f` compute `M = max(conv(f,f))`
   and solve the linear programme  
   `max sum(g)  s.t.  conv(f,g) ≤ M,  g ≥ 0`.  
   The solution `g` (normalised) is blended with `f`.  Several such
   refinement steps are performed (cvxpy/ECOS).

6. **Symmetrisation** – average the vector with its reversal; for this
   symmetric problem it often reduces the max‑convolution.

7. **Simulated‑annealing swap hill‑climber** – random swaps of positions
   with a Metropolis acceptance rule to polish the final vector.

All vectors are kept ℓ₁‑normalised, so the denominator of the objective
is constant and the whole search concentrates on reducing `max(conv)`.
The routine respects a total time budget (default 1000 s) and returns
the best sequence found.  The global variable `height_sequence_1` is
updated with the final candidate.
"""

import time
import math
import numpy as np

# Optional imports – the code works without them (fallbacks are used)
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective assuming sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_two(v1: np.ndarray, v2: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * v1 + alpha * v2
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(v1))


def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   prev: np.ndarray | None) -> list[np.ndarray]:
    """Build a diversified pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for a in (0.3, 1.0, 2.0, 4.0):
        for b in (0.3, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))
    for beta in (2, 4, 8, 12, 16, 20):
        seeds.append(_kaiser_window(n, beta))

    # random Dirichlet & exponential draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random convex mixtures of two already‑generated vectors
    base = seeds.copy()
    for _ in range(30):
        a, b = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_two(a, b, alpha))

    # previous best, if compatible
    if prev is not None and prev.shape == (n,):
        seeds.append(prev)

    # final sanity normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Smooth‑max optimisation (L‑BFGS or Adam fallback)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """Optimise the log‑sum‑exp surrogate of max‑convolution via L‑BFGS‑B."""
    if _opt is None:
        # No SciPy – fall back to the Adam implementation below
        return _adam_smooth_optimize(start_vec, n, deadline,
                                     beta_schedule, max_iter_per_beta)

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6,
            1e7, 2e7, 5e7, 1e8
        ]

    # work in log‑probability (softmax) space so simplex is automatic
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a for the smooth max
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _adam_smooth_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          beta_schedule=None,
                          max_iter_per_beta: int = 12000) -> np.ndarray:
    """Adam optimiser for the log‑sum‑exp surrogate (fallback when SciPy missing)."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6,
            1e7, 2e7, 5e7, 1e8
        ]

    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    for beta in beta_schedule:
        beta_val = float(beta)
        for _ in range(max_iter_per_beta):
            t += 1
            if time.time() > deadline:
                return _softmax(v)

            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

            # grad for smooth max w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam step on logits
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
            m_corr = m / (1 - beta1 ** t)
            v_corr = v_hat / (1 - beta2 ** t)
            eta = 0.5 / math.sqrt(t)
            v = v - eta * m_corr / (np.sqrt(v_corr) + eps)

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact‑max Adam‑style exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _adam_exact_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution with Adam momentum.
    Returns the best sequence seen and its exact objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v_hat = np.zeros_like(log_a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam update on log‑space
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * (g ** 2)
        m_corr = m / (1 - beta1 ** t)
        v_corr = v_hat / (1 - beta2 ** t)
        eta = 0.6 / math.sqrt(t)

        log_a = log_a - eta * m_corr / (np.sqrt(v_corr) + eps)
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 200_000 == 0:
            print(f"[AdamExact] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (requires cvxpy)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_steps: int = 8) -> np.ndarray:
    """Iteratively tighten the max‑convolution bound via a simple LP."""
    if cp is None or not hasattr(cp, "conv"):
        return f  # LP not available – return unchanged

    g_var = cp.Variable(n, nonneg=True)
    # We'll reuse the Parameter object by updating its value each iteration
    f_param = cp.Parameter(n, value=f)

    for step in range(max_steps):
        if time.time() > deadline:
            break

        conv_ff = _conv_fft(f, f)
        M = float(np.max(conv_ff))

        conv_fg = cp.conv(f_param, g_var)          # length 2n‑1
        constraints = [conv_fg <= M]

        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=200)
        except Exception as e:
            print(f"[LP] solver exception: {e}")
            break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            print("[LP] no optimal solution, status:", prob.status)
            break

        g_opt = np.asarray(g_var.value, dtype=np.float64)
        if g_opt is None or g_opt.sum() == 0:
            break

        # normalise and blend
        g_norm = g_opt / g_opt.sum()
        f = 0.5 * f + 0.5 * g_norm
        f = f / f.sum()
        f_param.value = f  # update parameter for the next step

    return f


# ----------------------------------------------------------------------
# Simulated‑annealing swap hill‑climber
# ----------------------------------------------------------------------
def _anneal_swap_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int = 400_000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T0 = 0.1 * cur_val if cur_val > 0 else 1.0

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective(cur_seq)
        delta = new_val - cur_val

        T = max(T0 * (1.0 - it / max_swaps), 1e-12)
        accept = delta < 0 or rng.random() < math.exp(-delta / T)

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 100_000 == 0:
            print(f"[AnnealSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------
def _symmetrize(seq: np.ndarray) -> np.ndarray:
    mixed = (seq + seq[::-1]) * 0.5
    s = mixed.sum()
    return mixed / s if s > 0 else seq.copy()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate size keeps the search fast while still
    # providing enough resolution to attain a good bound.
    # ------------------------------------------------------------------
    n = 2000
    print(f"\n[Info] Problem size n={n}")

    # ------------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr / arr.sum() if arr.sum() > 0 else None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"[Info] Collected {len(seeds)} seeds.")

    # ------------------------------------------------------------------
    # 2. Quick evaluation – keep the top K seeds
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Info] Top seed objectives (exact):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Optimisation over the selected seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 1e3, 2e3, 5e3,
                     1e4, 2e4, 5e4, 1e5,
                     2e5, 5e5, 1e6, 2e6, 5e6,
                     1e7, 2e7, 5e7, 1e8]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all seeds.")
            break

        remaining = deadline - time.time()
        max_iter_per_beta = int(min(20000, remaining * 0.07))
        max_iter_per_beta = max(12000, max_iter_per_beta)

        print(f"\n--- L‑BFGS (smooth‑max) on seed #{idx+1} "
              f"(max_iter_per_beta={max_iter_per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_per_beta)
        val_opt = _objective(seq_opt)
        print(f"[L‑BFGS] exact objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best (L‑BFGS) = {global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3a. Exact‑max Adam‑style polishing
        # ------------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp_iters = int(min(1_200_000, remaining * 0.12))
            max_exp_iters = max(800_000, max_exp_iters)
            print(f"[AdamExact] polishing with up to {max_exp_iters:,} iterations")
            seq_exp, val_exp = _adam_exact_opt(global_best_seq,
                                               n,
                                               deadline,
                                               max_iters=max_exp_iters)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[AdamExact] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. LP refinement (if cvxpy is available)
    # ------------------------------------------------------------------
    if time.time() < deadline and cp is not None:
        print("\n--- LP refinement steps ---")
        seq_lp = _lp_refine(global_best_seq, n, deadline, max_steps=6)
        val_lp = _objective(seq_lp)
        print(f"[LP] objective after refinement = {val_lp:.6f}")
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[LP] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sym = _symmetrize(global_best_seq)
        val_sym = _objective(seq_sym)
        if val_sym < global_best_val:
            global_best_seq = seq_sym
            global_best_val = val_sym
            print(f"[Symmetrize] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final permutation annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(400_000, remaining * 0.15))
        max_swaps = max(250_000, max_swaps)
        print("\n--- Simulated‑annealing swap local search ---")
        seq_perm, val_perm = _anneal_swap_opt(global_best_seq,
                                              n,
                                              deadline,
                                              max_swaps=max_swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[AnnealSwap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
