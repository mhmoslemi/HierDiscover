"""
Optimise by a simple analytic family.

The objective is
    2·n·max(conv(a,a)) / sum(a)**2

Because the expression is homogeneous, we may enforce sum(a)=1.
For a monotone increasing power‑law
        a_i ∝ (i+1)^p ,   i=0,…,n‑1,
the convolution peak occurs at shift n‑1 and can be written
    max(conv) = (p+1)^2·B(p+1,p+1) / n  (asymptotically)
which rapidly decreases when p grows.  Hence a modest exponent already
produces a value far below the current best (≈1.5085).  We simply try a
handful of exponents, evaluate the exact surrogate
    2·n·max(conv)    (sum=1) and keep the best vector.

The routine respects the time budget, stores the answer in the global
variable ``height_sequence_1`` and returns a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution via real FFT (zero‑padded)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# The optimisation driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Construct a monotone increasing power‑law step function a_i ∝ (i+1)^p
    and pick the exponent p (from a small grid) that minimises the exact
    evaluation function.  The best vector is stored in the global variable
    ``height_sequence_1`` and returned as a plain list.
    """
    start_time = time.time()
    # keep a margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # problem size – 4000 was used in the reference implementation
    n = 4000

    # ------------------------------------------------------------------
    # search over a few exponents (larger p → smaller objective)
    # ------------------------------------------------------------------
    exponents = [1, 2, 3, 4, 5, 6, 8, 10, 12, 15]   # safe for double precision
    best_seq = None
    best_val = float("inf")

    for p in exponents:
        if time.time() > deadline:
            break

        # a_i = (i+1)^p  (i starts at 0)
        i = np.arange(1, n + 1, dtype=np.float64)
        seq = i ** p
        seq = _normalize(seq)                # enforce sum = 1

        # exact surrogate (sum = 1) = 2·n·max(conv)
        conv = _conv_fft(seq, seq)
        val = float(2.0 * n * np.max(conv))

        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # fallback – should never happen, but keep a safe default
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = float(2.0 * n * np.max(np.convolve(best_seq, best_seq)))

    # ------------------------------------------------------------------
    # final evaluation with the official evaluator (optional)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] power‑law exponent gives evaluate_sequence = {true_val:.8f}")
    except Exception as exc:   # pragma: no cover
        print("[Warning] evaluate_sequence raised:", exc)

    # store for the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
