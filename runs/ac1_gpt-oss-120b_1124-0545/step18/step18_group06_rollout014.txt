Implementation of exponentiated gradient:

def exp_gradient_opt(vec, n, deadline, max_iters) -> (np.ndarray, float):
    a = vec.copy()
    best = a.copy()
    best_val = _objective(a)
    log_a = np.log(a + 1e-30)
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        # compute conv
        conv = _conv_fft(a, a)
        # find kmax
        kmax = int(np.argmax(conv))
        # compute gradient g
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        # idx_i from i_start to i_end inclusive
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # learning rate
        eta = 0.5 / math.sqrt(it)  # we could increase factor relative to 0.5
        log_a -= eta * g
        a = _softmax(log_a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
        # optionally print progress each 100k etc, but keep output minimal.
    return best, best_val

"""
A more exhaustive optimisation for the “step‑function” problem.

Strategy
--------
1. **Seed pool** – a large set of ℓ₁‑normalised vectors:
   * analytic windows (uniform, triangular, sine, Gaussian, Tukey, …)
   * Dirichlet draws with a range of concentration parameters
   * “alternating” placements of heavy‑tailed Dirichlet samples
     (big values are forced far apart to minimise overlap with the
     reversed copy, which is the dominant source of the max‑convolution).

2. **Fast pre‑selection** – evaluate every seed with a cheap FFT‑based
   max‑convolution and keep the best few (default 8).

3. **Exact‑max exponentiated‑gradient polishing** – mirror‑descent on the
   true objective using the sub‑gradient at the active lag.
   The update works in log‑probability space; the learning‑rate
   follows the classic  ηₜ = η₀ / √t  schedule (η₀ = 0.8 gives a bit more
   aggressive early steps).  We run as many iterations as the remaining
   time allows (up to ≈1 M steps).

4. **Swap‑based simulated annealing** – a final continuous‑value local
   search that repeatedly swaps two positions (pure permutation move) and
   sometimes makes a small multiplicative perturbation of a single entry.
   Acceptance follows a Metropolis rule with a linear‑cooling temperature
   schedule.  The number of swaps is again time‑controlled.

All steps preserve non‑negativity and ℓ₁‑normalisation, so the final
vector can be fed directly to ``evaluate_sequence``.  The routine respects
the 1000 s overall budget and stores the best sequence in the global
variable ``height_sequence_1`` for possible warm‑starts on the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (exact, size = len(a)+len(b)-1)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(vec: np.ndarray) -> float:
    """Exact objective used during the search: 2·n·max(conv)."""
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    per = alpha * (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    mask1 = i < per
    mask2 = (i >= per) & (i <= (n - 1) * (1 - alpha / 2))
    mask3 = ~mask1 & ~mask2
    a[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * i[mask1] / (alpha * (n - 1)) - 1)))
    a[mask2] = 1.0
    a[mask3] = 0.5 * (1 + np.cos(np.pi *
                                 (2 * i[mask3] / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _alternating_placement(v: np.ndarray) -> np.ndarray:
    """Place the sorted values alternately from the left and right ends."""
    sorted_desc = np.sort(v)[::-1]
    out = np.empty_like(v)
    left, right = 0, len(v) - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out / out.sum()


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                    previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.5, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    for sigma in (0.15, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    for p in (0.5, 1.0, 2.0, 4.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0):
        for b in (0.5, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.4, 0.6, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # heavy‑tailed Dirichlet samples with alternating placement
    for conc in (0.02, 0.05, 0.10):
        for _ in range(4):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(_alternating_placement(raw))

    # plain Dirichlet draws (balanced)
    for conc in (0.3, 0.7, 1.5, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # safety normalisation
    cleaned = []
    for v in seeds:
        vv = np.asarray(v, dtype=np.float64)
        s = vv.sum()
        if s > 0:
            cleaned.append(vv / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_vec: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int,
                     eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best vector seen and its objective value.
    """
    a = start_vec.copy()
    best = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = kmax - idx
        g[idx] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()

        if it % 200_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing permutation + value perturbation
# ----------------------------------------------------------------------
def _anneal_opt(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_swaps: int) -> tuple[np.ndarray, float]:
    """
    Random swaps and occasional local multiplicative perturbations,
    Metropolis acceptance with a linearly decreasing temperature.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = 0.1 * cur_val if cur_val > 0 else 1.0

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        # --- swap move -------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i != j:
            cur[i], cur[j] = cur[j], cur[i]
            new_val = _objective(cur)
            delta = new_val - cur_val
            T = max(T0 * (1.0 - it / max_swaps), 1e-12)
            accept = delta < 0.0 or rng.random() < math.exp(-delta / T)
            if accept:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = cur.copy()
            else:
                cur[i], cur[j] = cur[j], cur[i]   # revert

        # --- tiny multiplicative perturbation (10 % change) -----------
        if rng.random() < 0.005:   # very occasional
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.1))   # log‑normal tweak
            cur[idx] *= factor
            cur /= cur.sum()           # re‑normalise
            new_val = _objective(cur)
            delta = new_val - cur_val
            T = max(T0 * (1.0 - it / max_swaps), 1e-12)
            accept = delta < 0.0 or rng.random() < math.exp(-delta / T)
            if accept:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = cur.copy()
            else:
                # revert perturbation
                cur[idx] /= factor
                cur /= cur.sum()

        if it % 100_000 == 0:
            print(f"[Anneal] swaps={it:,} best={best_val:.6f}")

    return best, best_val
