"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s wall‑time)
--------------------------------
1. **Seed generation** – deterministic families (uniform, triangular,
   Gaussian, power‑law, truncated exponential/power‑law, sin²‑shape) and a
   large random Dirichlet/sparse pool.  All seeds are ℓ₁‑normalised.
2. **Elite selection** – keep the best few seeds (by the exact objective).
3. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp surrogate of
   ``max(conv)`` with an aggressive β‑schedule on every elite seed.
4. **Mirrored exponential‑gradient** – several restarts of the exact
   sub‑gradient descent (mirror‑descent on the simplex).
5. **Targeted local search** – deterministic hill‑climb that repeatedly
   moves a small fraction of mass from the index contributing most to the
   current maximal convolution entry to the index with the smallest weight.
   The convolution is updated incrementally (O(n) per move) and only
   accepted if it strictly reduces the objective.
6. **Final high‑β polishing** – one more L‑BFGS run with very large β.
7. Return the best sequence found (scaled to the required ℓ₁‑mass
   ``sqrt(2·n)``) and store it in the global ``height_sequence_1`` variable.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of 2·n·max(conv) / (∑a)².
    Returns (obj, grad) – gradient w.r.t. logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)              # full convolution, length 2n‑1
    shift = np.max(conv)                # shift for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # soft‑max over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # back‑prop through softmax (mirror‑descent style)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0, 50_000_000.0, 100_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 150_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted deterministic local search (incremental convolution updates)
# ----------------------------------------------------------------------
def _targeted_local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    step_frac_start: float = 0.018,
    step_frac_end: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing: move a small fraction of mass from the index that
    contributes most to the current maximal convolution entry to the
    index with the smallest weight.  Convolution is updated incrementally.
    """
    a = start_a.copy()
    conv = _conv_fft(a, a)          # length 2n‑1
    sum_a = a.sum()                # constant during the search
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current step fraction (linear schedule)
        step_frac = step_frac_start * (1 - it / max_iters) + step_frac_end * (it / max_iters)

        # find lag attaining the maximum
        idx = int(np.argmax(conv))

        # indices that participate in this lag
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range

        # contribution of each participating index to the max lag
        contrib = a[i_range] * a[rev]
        src = i_range[np.argmax(contrib)]

        # destination – the globally smallest entry (break ties deterministically)
        tgt = int(np.argmin(a))
        if src == tgt:
            # if the smallest entry also happens to be the source, pick the second smallest
            sorted_idxs = np.argsort(a)
            tgt = int(sorted_idxs[1])

        delta = step_frac * a[src]
        if delta <= 0.0:
            continue

        # keep a copy of convolution to revert if needed
        conv_old = conv.copy()
        a_src_old = a[src]
        a_tgt_old = a[tgt]

        # ----- removal from src (using old a) -----
        conv[src:src + n] -= 2.0 * delta * a

        # ----- apply the mass transfer -----
        a[src] = a_src_old - delta
        a[tgt] = a_tgt_old + delta

        # ----- addition at tgt (using the updated a) -----
        conv[tgt:tgt + n] += 2.0 * delta * a

        # evaluate new objective (sum_a unchanged)
        cur_val = 2.0 * n * np.max(conv) / (sum_a * sum_a)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        else:
            # revert – the move was not improving
            conv = conv_old
            a[src] = a_src_old
            a[tgt] = a_tgt_old

    return best_a, best_val


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _truncated_exponential(n: int, lam: float, L: int) -> np.ndarray:
    """Exponential decay on a prefix of length L, zeros elsewhere."""
    if L <= 0:
        return _uniform(n)
    i = np.arange(L, dtype=np.float64)
    a = np.exp(-lam * i)
    full = np.zeros(n, dtype=np.float64)
    full[:L] = a
    s = full.sum()
    return full / s if s > 0 else _uniform(n)


def _truncated_power_law(n: int, p: float, L: int) -> np.ndarray:
    """Power‑law on a prefix of length L, zeros elsewhere."""
    if L <= 0:
        return _uniform(n)
    i = np.arange(L, dtype=np.float64)
    a = (i + 1.0) ** (-p)
    full = np.zeros(n, dtype=np.float64)
    full[:L] = a
    s = full.sum()
    return full / s if s > 0 else _uniform(n)


def _sin2_shape(n: int) -> np.ndarray:
    """a_i ∝ sin²(π (i+0.5) / (2n)).  Positive, bell‑shaped."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / (2.0 * n)) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Diversified ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.20))
    seeds.append(_exponential_decay(n, 0.005))
    seeds.append(_sin2_shape(n))

    # power‑law sweep – wider range than before
    p_vals = np.linspace(0.20, 1.20, 11)   # exponents
    shifts = [0.0, 5.0, 20.0, 60.0, 150.0, 350.0]  # broader shifts
    for p in p_vals:
        for sh in shifts:
            seeds.append(_power_law(n, float(p), float(sh)))
            seeds.append(_power_law(n, float(p), float(sh))[::-1])

    # truncated families
    for lam in (0.001, 0.003, 0.007):
        for L_frac in (0.25, 0.40, 0.60):
            L = int(L_frac * n)
            seeds.append(_truncated_exponential(n, lam, L))
            seeds.append(_truncated_exponential(n, lam, L)[::-1])

    for p in (0.6, 1.0, 1.5):
        for L_frac in (0.30, 0.50):
            L = int(L_frac * n)
            seeds.append(_truncated_power_law(n, p, L))
            seeds.append(_truncated_power_law(n, p, L)[::-1])

    # random Dirichlet families (concentration favouring sparsity)
    for conc in (0.02, 0.07, 0.15, 0.5, 1.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse families
    for frac in (0.02, 0.05, 0.12, 0.25):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency curves
    for nf in (2, 3, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures (explore convex hull)
    base = np.arange(len(seeds))
    for _ in range(120):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising candidates
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best if supplied
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce non‑negativity & ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # Problem size – larger n gives more degrees of freedom but costs more.
    # 4000 works well for the time budget; feel free to raise it slightly.
    n = 4200
    target_sum = np.sqrt(2.0 * n)

    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed generation (including warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Evaluate seeds, keep elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v * target_sum)   # scale to ℓ₁‑mass required by the objective
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    # Keep the top‑k seeds for heavy L‑BFGS stage
    seeds.sort(key=_objective_exact)
    elite = seeds[:min(25, len(seeds))]

    global_best_seq = best_seq.copy() * target_sum
    global_best_val = best_val
    print(f"[Init] global best objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on smooth surrogate
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 50_000_000.0, 100_000_000.0,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break

        seed_vec = seed_vec * target_sum
        # allocate a modest per‑β iteration budget respecting remaining time
        time_left = deadline - time.time()
        per_beta = max(900, int(min(3000, time_left * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β≈{per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient restarts (exact objective)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 8
        iters_per_restart = int(min(500_000, remaining * 0.11 / restarts))
        print("\n--- EG restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.02, 1.2)
            start_vec = _random_dirichlet(n, conc, rng) * target_sum
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.0,
            )
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long EG run from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(8_000_000, remaining * 0.26 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=2.5,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – targeted deterministic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(500_000, remaining * 50))
        if iters > 0:
            print("\n--- Targeted local search (mass transfers) ---")
            seq_loc, val_loc = _targeted_local_search(
                global_best_seq,
                n,
                deadline,
                max_iters=iters,
                step_frac_start=0.018,
                step_frac_end=0.001,
            )
            print(f"[TargetedLocal] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after targeted local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final polishing L‑BFGS with very high β
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(900, int(min(3000, remaining * 0.05 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[5_000_000.0, 10_000_000.0, 50_000_000.0, 100_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
