def _adam_opt(start_vec, n, deadline, max_iters=None, beta=...):
   v = np.log(start_vec + eps)
   m = np.zeros_like(v)
   v2 = np.zeros_like(v)
   b1, b2 = 0.9, 0.999
   eps = 1e-8
   t=0
   while time.time()<deadline and (max_iters is None or t<max_iters):
        t+=1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        # choose beta schedule: start moderate, increase
        current_beta = beta if not isinstance(beta, (list, tuple)) else beta[t % len(beta)]
        # compute w = exp(beta*(conv - max_c)) etc.
        exp_beta = np.exp(current_beta * (conv - max_c))
        sum_exp = exp_beta.sum()
        if sum_exp == 0.0:
            continue  # fallback maybe uniform
        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]   # shape n
        # convert to gradient w.r.t v
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = b1 * m + (1-b1) * grad_v
        v2 = b2 * v2 + (1-b2) * (grad_v**2)
        m_hat = m / (1 - b1**t)
        v2_hat = v2 / (1 - b2**t)
        step = 0.1 * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        if t % 5000 == 0:
            cur_val = _objective(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()
                print(f"[Adam] iteration {t} best {best_val:.6f}")

   return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimises

    evaluate_sequence(sequence) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm proceeds as follows

1. **Seed generation** – a large, diverse pool of ℓ₁‑normalised vectors,
   built from analytic windows, Dirichlet draws and a few random patterns.
   The best few seeds are retained.

2. **Adam optimisation on a smooth‑max surrogate** – we work in the log‑softmax
   (log‑probability) domain, so the simplex constraint is satisfied automatically.
   For a given β the surrogate of the “max” is a soft‑max (log‑sum‑exp).  The
   gradient with respect to the underlying vector a is obtained analytically
   via a single convolution.  An Adam‑style adaptive step‑size is used,
   starting with a modest β and increasing it gradually, thereby transitioning
   from a smooth to the exact max problem.

3. **Fine‑tune with exact‑max Adam** – after the smooth stage we continue
   optimisation using the true sub‑gradient (one‑hot at the active lag) and Adam.

4. **Permutation‑swap annealing** – random swaps of positions are performed
   with a Metropolis acceptance rule to further reduce the peak of the
   self‑convolution.

5. **Optional symmetrisation** – averaging with the reversed copy often
   lowers the maximum coefficient.

All stages respect a global 1000 s budget and stop early if the deadline is
reached.  The best sequence found is stored in the global variable
`height_sequence_1` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:

    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0

def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:

    mu1 = shift_frac * (n - 1)

    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()

def _sinc_window(n: int, width_frac: float = 0.2) -> np.ndarray:

    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a large, diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 8, 12, 16, 20):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for a in (0.3, 1.0, 2.0, 4.0):
        for b in (0.3, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))
    seeds.append(_sinc_window(n, width_frac=0.2))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random vectors
    for k in (max(1, int(0.01 * n)), max(1, int(0.03 * n))):
        for _ in range(2):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors
    for bf in (0.05, 0.15, 0.30):
        for _ in range(2):
            seeds.append(_block_uniform(n, bf, rng))

    # Heavy‑tailed alternating Dirichlet
    seeds.append(_alternating_dirichlet(n, rng))

    # Permuted heavy‑tailed seed (helps permutation search)
    heavy = _random_dirichlet(n, 0.05, rng)
    for _ in range(3):
        seeds.append(heavy[rng.permutation(n)])

    # Previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity check
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

def _adam_opt(start_vec: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns (best_sequence, best_objective).
    """
    # initialise in log‑softmax space
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = _softmax(v)
    best_val = _objective(best_seq)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(v)                 # ℓ₁‑normalised a
        conv = _conv_fft(a, a)          # length 2n‑1
        shift = np.max(conv)            # location of the max

        # gradually increase β to tighten the surrogate
        if t < 50000:
            beta = 1e3
        elif t < 150000:
            beta = 1e4
        elif t < 250000:
            beta = 1e5
        else:
            beta = 1e6

        # soft‑max weights over the convolution coefficients
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        if sum_exp == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            sum_exp = 1.0
        else:
            w = exp_beta / sum_exp

        # gradient wrt a (derived from the smoothed max)
        grad_a = 2.0 * _conv_fft(a[::-1], w)[n - 1:n - 1 + n]

        # back‑propagate through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        step = 0.05 * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        # occasional logging & best‑track
        if t % 10000 == 0:
            cur_seq = _softmax(v)
            cur_val = _objective(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[AdamSmooth] it={t:,} best={best_val:.6f}")

    return best_seq, best_val


def _adam_exact_max(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the true max (hard sub‑gradient).  Similar to an
    exponentiated‑gradient step but with adaptive learning rates.
    Returns (best_sequence, best_objective).
    """
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = _softmax(v)
    best_val = _objective(best_seq)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # back‑prop through softmax
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        step = 0.05 * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        if t % 10000 == 0:
            cur_seq = _softmax(v)
            cur_val = _objective(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[AdamExact] it={t:,} best={best_val:.6f}")

    return best_seq, best_val


def _anneal_swap_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Random swaps with Metropolis acceptance.
    Returns the best sequence encountered and its objective value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T0 = 0.1 * cur_val if cur_val > 0 else 1.0

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective(cur_seq)
        delta = new_val - cur_val
        T = max(T0 * (1.0 - it / max_swaps), 1e-12)
        accept = delta < 0 or rng.random() < math.exp(-delta / T)
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 50000 == 0:
            print(f"[AnnealSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Average a vector with its reversal and renormalise."""
    mixed = (seq + seq[::-1]) * 0.5
    s = mixed.sum()
    return mixed / s if s > 0 else seq.copy()

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    n = 4000                                 # length of the step‑function

    # ------------------------------------------------------------------
    # 1. Seed generation (reuse previous solution if available)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 2. Quick evaluation – keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed objective values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Optimisation on the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before completing seed optimisation.")
            break

        remaining = deadline - time.time()
        # allocate a modest budget for the Adam stages
        max_iters_smooth = int(min(300_000, remaining * 0.20))
        max_iters_exact = int(min(200_000, remaining * 0.10))

        print(f"\n--- Adam (smooth) on seed #{idx+1} (iters={max_iters_smooth}) ---")
        seq_smooth, val_smooth = _adam_opt(seed_vec, n, deadline, max_iters=max_iters_smooth)
        if val_smooth < global_best_val:
            global_best_seq = seq_smooth.copy()
            global_best_val = val_smooth
            print(f"[Update] new global best (smooth) = {global_best_val:.6f}")

        if time.time() > deadline:
            break

        print(f"--- Adam (exact max) on seed #{idx+1} (iters={max_iters_exact}) ---")
        seq_exact, val_exact = _adam_exact_max(global_best_seq, n, deadline,
                                                max_iters=max_iters_exact)
        if val_exact < global_best_val:
            global_best_seq = seq_exact.copy()
            global_best_val = val_exact
            print(f"[Update] new global best (exact) = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final fine‑tuning with exact‑max Adam (as much time as remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_final = int(min(500_000, remaining * 0.30))
        print("\n--- Final Adam exact‑max fine‑tune ---")
        seq_final, val_final = _adam_exact_max(global_best_seq, n, deadline,
                                                max_iters=max_iters_final)
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Permutation hill‑climber (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(800_000, remaining * 0.30))
        print("\n--- Simulated‑annealing swap local search ---")
        seq_perm, val_perm = _anneal_swap_opt(global_best_seq, n, deadline,
                                              max_swaps=max_swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[AnnealSwap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional symmetrisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sym = _symmetrize(global_best_seq)
        val_sym = _objective(seq_sym)
        if val_sym < global_best_val:
            global_best_seq = seq_sym
            global_best_val = val_sym
            print(f"[Symmetrize] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
