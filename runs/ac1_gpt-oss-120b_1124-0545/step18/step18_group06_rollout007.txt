def _adam_optimize(seq, n, deadline, max_iters=500_000, lr=0.5):
    v = np.log(seq+1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1=0.9; beta2=0.999; eps=1e-8
    t=0
    best_seq = seq.copy()
    best_val = _objective(seq)
    while time<deadline and t<max_iters:
        t+=1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))
        # compute subgradient g (size n)
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end+1)
        rev_idx = kmax - idx
        g[idx] = 2.0 * a[rev_idx]   # as before
        # gradient w.r.t v
        grad_v = a * (g - np.dot(g, a))

        # Adam update
        m = beta1*m + (1-beta1)*grad_v
        v_hat = beta2*v_hat + (1-beta2)*(grad_v**2)
        m_corr = m / (1 - beta1**t)
        v_corr = v_hat / (1 - beta2**t)
        # dynamic learning rate decays like 0.5 / sqrt(t)
        eta = lr / np.sqrt(t)
        v = v - eta * m_corr / (np.sqrt(v_corr) + eps)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geometric(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised to ℓ₁."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _shifted_geometric(n: int, r: float, shift: int) -> np.ndarray:
    """Zeros for first `shift` entries then geometric decay."""
    if shift >= n:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    length = n - shift
    i = np.arange(length, dtype=np.float64)
    a[shift:] = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """Optimise a log‑sum‑exp smoothing of max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS stage.")
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6, 1e7,
            2e7, 5e7, 1e8
        ]

    # Work in log‑probability space – softmax guarantees the simplex constraint.
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (derived from the smoothed max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            # convolution of rev_a (length n) with w (length 2n‑1) -> length 3n‑2
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 2_000_000,
                   base_lr: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent.
    Returns the best sequence (and its surrogate value) discovered.
    """
    a = start_seq.copy()
    v = np.log(a + 1e-30)                     # log‑probability representation
    m = np.zeros_like(v)                     # first moment
    v_hat = np.zeros_like(v)                 # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_seq = a.copy()
    best_val = _objective(a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        a = _softmax(v)                       # current simplex point

        # exact max‑convolution and active lag
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of max‑convolution (as in exponentiated‑gradient)
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient wrt log‑probabilities
        grad_v = a * (g - np.dot(g, a))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
        m_corr = m / (1.0 - beta1 ** t)
        v_corr = v_hat / (1.0 - beta2 ** t)

        # learning‑rate schedule – decays like 1/√t
        eta = base_lr / math.sqrt(t)
        v = v - eta * m_corr / (np.sqrt(v_corr) + eps)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # optional cheap progress print
        if t % 250_000 == 0:
            print(f"[Adam] iter={t:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Log‑space Gaussian perturbation with Metropolis acceptance
# ----------------------------------------------------------------------
def _perturb_logspace(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_ops: int = 500_000,
                      sigma0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Random Gaussian perturbation of the log‑vector.
    Accepts improvements (always) and occasional deteriorations
    according to a linear‑cooling temperature schedule.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_ops + 1):
        if time.time() > deadline:
            break
        # temperature linearly decreases from 0.1*cur_val to 1e-12
        T = max(0.1 * cur_val * (1.0 - it / max_ops), 1e-12)

        # perturb log‑probability vector
        v = np.log(cur_seq + 1e-30)
        sigma = sigma0 * (1.0 - it / max_ops)  # linear decay
        v_new = v + sigma * rng.normal(size=n)
        cand_seq = _softmax(v_new)

        cand_val = _objective(cand_seq)
        delta = cand_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur_seq = cand_seq
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand_seq.copy()

        if it % 100_000 == 0:
            print(f"[Perturb] ops={it:,} best={best_val:.6f}")

    return best_seq, best_val

def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Average a vector with its reversal and re‑normalise onto the simplex."""
    mixed = (seq + seq[::-1]) * 0.5
    s = mixed.sum()
    return mixed / s if s > 0 else seq.copy()

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within a time budget.
    Returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Problem size – a few thousand works well for the FFT‑based code.
    n = 4000

    # ------------------------------------------------------------------
    # 1. Seed generation (reuse previous solution if available)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 2. Quick evaluation of all seeds – keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS continuation on the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1e3, 2e3, 5e3,
        1e4, 2e4, 5e4, 1e5,
        2e5, 5e5, 1e6, 2e6, 5e6, 1e7, 2e7, 5e7, 1e8
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS seeds.")
            break

        remaining = deadline - time.time()
        # allocate a modest slice of remaining time to each seed
        max_iter_per_beta = int(min(24_000, remaining * 0.07))
        max_iter_per_beta = max(8_000, max_iter_per_beta)

        print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter_per_beta={max_iter_per_beta}) ---")
        try:
            seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             deadline,
                                             beta_schedule=beta_schedule,
                                             max_iter_per_beta=max_iter_per_beta)
        except RuntimeError:
            # SciPy not available – fall back to the seed itself
            seq_opt = seed_vec.copy()

        val_opt = _objective(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best (L‑BFGS) = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam mirror‑descent polishing (runs as long as time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam mirror‑descent polishing ---")
        remaining = deadline - time.time()
        max_adam_iters = int(min(2_000_000, remaining * 1.2))
        max_adam_iters = max(300_000, max_adam_iters)
        seq_adam, val_adam = _adam_optimize(global_best_seq,
                                            n,
                                            deadline,
                                            max_iters=max_adam_iters,
                                            base_lr=0.8)
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Log‑space perturbation annealing (fine‑grained exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Log‑space perturbation annealing ---")
        remaining = deadline - time.time()
        max_ops = int(min(600_000, remaining * 1.0))
        max_ops = max(200_000, max_ops)
        seq_pert, val_pert = _perturb_logspace(global_best_seq,
                                               n,
                                               deadline,
                                               max_ops=max_ops,
                                               sigma0=0.5)
        if val_pert < global_best_val:
            global_best_seq = seq_pert.copy()
            global_best_val = val_pert
            print(f"[Perturb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional symmetrisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sym = _symmetrize(global_best_seq)
        val_sym = _objective(seq_sym)
        if val_sym < global_best_val:
            global_best_seq = seq_sym
            global_best_val = val_sym
            print(f"[Symmetrize] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
