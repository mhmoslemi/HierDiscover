"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

The algorithm combines several ideas:

* Build a diverse pool of analytic and random seeds (uniform, triangular,
  Gaussian, power‑law, Beta‑distribution, Dirichlet, etc.).
* Pick the best few seeds (by the exact surrogate 2·n·max(conv) with sum=1).
* Refine the best seed with a robust L‑BFGS‑B optimisation on a *smooth*
  max‑convolution surrogate (log‑sum‑exp).  The gradient is derived
  analytically (the previous implementation missed a factor 2 n and the
  extra weight coming from the active maximum).  A schedule of increasing
  β values drives the surrogate towards the true max.
* A short simulated‑annealing walk in logit‑space polishes the result.
* The final vector is returned and stored as ``height_sequence_1`` for the
  next run.

All heavy work is limited by the ``budget_s`` argument (default 1000 s).
The implementation uses only NumPy (and SciPy if available) and respects
the security checks of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for L‑BFGS‑B (used only for polishing)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_dist(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) on the grid i/(n‑1), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    eps = 1e-12
    a = np.power(np.maximum(x, eps), alpha - 1.0) * np.power(
        np.maximum(1.0 - x, eps), beta - 1.0
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exact surrogate (the objective when sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# L‑BFGS polishing with a *correct* gradient for the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize_correct(start_vec: np.ndarray,
                            n: int,
                            deadline: float,
                            beta_schedule=None,
                            max_iter_per_beta: int = 2000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) using L‑BFGS‑B on a smooth max‑conv.
    The gradient now includes the missing factor ``2 n`` and the extra weight
    for the active maximum.
    """
    if _opt is None:  # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1, 5, 20, 80, 200, 500, 1_000, 2_000, 5_000,
            20_000, 50_000, 200_000, 500_000, 1_000_000,
            2_000_000, 5_000_000, 10_000_000,
        ]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            max_idx = int(np.argmax(conv))

            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            # weights for the smooth max (include the extra weight for the active index)
            w = exp_beta / sum_exp
            w[max_idx] += 1.0                 # +1 accounts for the sub‑gradient of the max

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)        # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n] * (2.0 * n)  # full factor 2·n

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)       # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] unexpected exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

    selects the most promising one, refines it with a corrected L‑BFGS
    optimiser on a smooth surrogate, and finishes with a short simulated‑
    annealing walk.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a size of a few thousand balances quality / speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build a diverse seed pool
    # ------------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sin2(n))
    seeds.append(_hann_window(n))
    seeds.append(_arcsine(n))

    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # Beta‑distribution shapes (including very heavy edge concentrations)
    for a, b in ((0.2, 0.2), (0.1, 0.1), (0.05, 0.05),
                 (0.1, 0.3), (0.3, 0.1), (0.2, 0.5), (0.5, 0.2)):
        seeds.append(_beta_dist(n, a, b))

    # Random Dirichlet draws with a wide range of concentration parameters
    conc_params = np.logspace(-4, 2, 12)      # 1e‑4 … 100
    for conc in conc_params:
        for _ in range(3):
            try:
                seeds.append(rng.dirichlet(np.full(n, conc)))
            except Exception:  # pragma: no cover
                pass

    # Random permutations of a few analytic families (break symmetry)
    for base in (_gaussian(n, 0.15), _power(n, 0.5), _beta_dist(n, 0.2, 0.2)):
        for _ in range(4):
            perm = np.arange(n)
            rng.shuffle(perm)
            seeds.append(base[perm])

    # include the previous best if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # normalise everything onto the probability simplex
    def _norm(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / len(v))
        return v / s

    seeds = [_norm(np.asarray(v, dtype=np.float64)) for v in seeds]

    # ------------------------------------------------------------------
    # 2. Quick surrogate ranking – keep the best few seeds
    # ------------------------------------------------------------------
    surrogate_vals = [_objective_surrogate(s) for s in seeds]
    # indices of the 5 best seeds
    best_idxs = np.argsort(surrogate_vals)[:5]
    best_seeds = [seeds[i] for i in best_idxs]

    print("\n[Init] surrogate values of top seeds:")
    for i, idx in enumerate(best_idxs):
        print(f"  seed {i+1}: value = {surrogate_vals[idx]:.6f}")

    # ------------------------------------------------------------------
    # 3. Heavy L‑BFGS polishing (correct gradient) on each top seed
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    for i, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS polish on top seed #{i} ---")
        seq_polished = _lbfgs_optimize_correct(
            seed_vec, n, deadline,
            beta_schedule=[
                1, 5, 20, 80, 200, 500, 1_000,
                2_000, 5_000, 20_000, 50_000,
                200_000, 500_000, 1_000_000, 2_000_000,
                5_000_000, 10_000_000
            ],
            max_iter_per_beta=2500
        )
        val_polished = _objective_surrogate(seq_polished)
        print(f"[L‑BFGS] seed #{i} polished surrogate = {val_polished:.6f}")

        if val_polished < best_val:
            best_val = val_polished
            best_seq = seq_polished.copy()
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(500_000, remaining * 2000))
        max_sa = max(200_000, max_sa)
        print("\n--- Simulated‑Annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_opt(best_seq, n, deadline,
                                                  max_iters=max_sa)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Official evaluation and return
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
