"""
improved_candidate.py

A self‑contained optimiser for the “step‑function” challenge.

The code builds many analytic and random seeds, then repeatedly
refines the promising ones by two complementary strategies:

* **Fast convex mixture** – convex combinations of the K best seeds are
  optimised with a cheap ternary search (O(K²) work).

* **Local stochastic search** – a simple simulated‑annealing / coordinate‑
  descent loop that moves a tiny amount of mass between two entries while
  staying on the simplex.  The surrogate objective (exact max‑convolution)
  is used for the inner loop because it is cheap (`FFT` based, O(n log n)).
  When the time budget is almost exhausted we polish the current best with
  the exponentiated‑gradient (mirror‑descent) pass and a continuation L‑BFGS
  optimiser (the same smooth‑max surrogate used in the reference solution).

The optimiser respects the global 1000 s budget.  The best sequence found
is stored in the global variable ``height_sequence_1`` for reuse in later
runs.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – produces a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective under the ℓ₁‑normalisation Σseq = 1:
        2·n·max(conv(seq, seq))

    ``seq`` is assumed to be a 1‑D NumPy array of non‑negative numbers.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _dirichlet_draw(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _exponential_draw(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (both already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s

# Local optimisation primitives
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 150_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)               # stay in interior

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (standard formula for auto‑conv)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000):
    """
    Smooth‑max L‑BFGS optimisation (variables live in log‑softmax space).
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0, 200000.0,
                         500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            mx = np.max(conv)
            shift = mx
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:      # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Pairwise mixing – ternary search on α ∈ [0,1]
# ----------------------------------------------------------------------


def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None

    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0

        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)

        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)

        if val1 < val2:
            hi = m2
            if best_val is None or val1 < best_val:
                best_val = val1
                best_seq = seq1
        else:
            lo = m1
            if best_val is None or val2 < best_val:
                best_val = val2
                best_seq = seq2

    # final refinement at the middle point
    if best_seq is None:
        return a.copy(), _objective_surrogate(a)
    mid = (lo + hi) / 2.0
    cand = _mix_shapes(a, b, mid)
    cand_val = _objective_surrogate(cand)
    if cand_val < best_val:
        best_seq, best_val = cand, cand_val
    return best_seq, best_val

# Simple stochastic local search (coordinate‑mass transfer)
# ----------------------------------------------------------------------


def _local_mass_transfer(start_seq: np.ndarray,
                        deadline: float,
                        max_iters: int = 200_000,
                        init_step: float = 0.02):
    """
    Very cheap hill‑climbing / simulated‑annealing on the simplex.
    A tiny mass 'δ' is moved from a random source index to a random
    target index; the move is kept if it improves the surrogate.
    """
    n = len(start_seq)
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    step = init_step

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simulated‑annealing temperature schedule (optional)
        T = max(0.0, 1.0 - it / max_iters)

        # pick source with enough mass
        src = rng.integers(0, n)
        while cur[src] < 1e-12:
            src = rng.integers(0, n)
        dst = rng.integers(0, n)
        while dst == src:
            dst = rng.integers(0, n)

        # propose a transfer
        delta = step * cur[src] * rng.random()
        if delta <= 0.0:
            continue
        prop = cur.copy()
        prop[src] -= delta
        prop[dst] += delta

        # evaluate surrogate (fast)
        val = _objective_surrogate(prop)

        # accept if improvement, otherwise maybe accept downhill with prob.
        accept = False
        if val < cur_val:
            accept = True
        elif T > 0.0:
            prob = math.exp(-(val - cur_val) / (0.001 * T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = prop
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = prop.copy()
            # occasionally shrink step size to fine‑tune
            if it % 5000 == 0:
                step = max(step * 0.95, 1e-4)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best list of heights found and stores
    it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # tiny safety margin

    # ------------------------------------------------------------------
    # Fixed problem size – can be tweaked if desired (larger → more work)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            a = np.asarray(prev, dtype=np.float64)
            if a.shape == (n,):
                prev_arr = a / a.sum() if a.sum() > 0.0 else None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (analytic + random)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # analytic families (moderate grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 8):
        seeds.append(_sine_window(n, gamma))
    for sigma in np.linspace(0.08, 0.30, 9):
        seeds.append(_gaussian(n, sigma))
    for a_ in np.linspace(0.2, 5.0, 6):
        for b_ in np.linspace(0.2, 5.0, 6):
            seeds.append(_beta_shape(n, a_, b_))

    # random Dirichlet / exponential draws
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_dirichlet_draw(n, conc, rng))
    for _ in range(6):
        seeds.append(_exponential_draw(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # add a few perturbed copies of the previous best (if present)
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(4):
            noisy = prev_arr + 0.005 * rng.normal(size=n)
            noisy = np.maximum(noisy, 0.0)
            seeds.append(noisy / noisy.sum())

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of raw seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop raw seeds (surrogate values):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Pairwise mixture optimisation (ternary search)
    # ------------------------------------------------------------------
    print("\n--- Pairwise mixing of top seeds ---")
    mix_deadline = deadline - 2.0
    mixtures = []
    for i in range(len(top_seeds)):
        for j in range(i + 1, len(top_seeds)):
            if time.time() > mix_deadline:
                break
            mixed_seq, mixed_val = _best_mixed_pair(
                top_seeds[i], top_seeds[j], mix_deadline, steps=20)
            mixtures.append((mixed_seq, mixed_val))

    candidates = evaluated[:top_k] + mixtures
    candidates.sort(key=lambda x: x[1])
    best_seq = candidates[0][0].copy()
    best_val = candidates[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic local search on top candidates
    # ------------------------------------------------------------------
    print("\n--- Stochastic local search ---")
    local_deadline = deadline - 2.0
    for idx, (seed_vec, _) in enumerate(candidates[:4]):   # a handful only
        if time.time() >= deadline:
            break
        allotted = (deadline - time.time()) * 0.35
        sub_deadline = time.time() + allotted
        print(f"\n[Local #{idx + 1}] starting search (seed surrogate "

        seq_local, val_local = _local_mass_transfer(seed_vec.copy(),
                                                    sub_deadline,
                                                    max_iters=300_000,
                                                    init_step=0.04)
        # short EG refinement
        seq_eg, val_eg = _exp_gradient_opt(seq_local,
                                           n,

                                               n,

                                               max_iter_per_beta=3000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs = seq_eg

        local_best_seq, local_best_val = min(options, key=lambda x: x[1])

            best_val = val_final

        seq_final_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                                n,
                                                deadline,
                                                max_iter_per_beta=3000)
        val_final_lbfgs = _objective_surrogate(seq_final_lbfgs)
        if val_final_lbfgs < best_val:
            best_seq = seq_final_lbfgs
            best_val = val_final_lbfgs
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())

    except Exception as exc:            # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
