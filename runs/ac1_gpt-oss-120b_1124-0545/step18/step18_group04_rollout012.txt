def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities (probability simplex handling)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (∑=1)."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its sum equals *target*."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s == 0.0:
        return np.full_like(v, target / len(v))
    return v * (target / s)


# ----------------------------------------------------------------------
# Exact surrogate (used while searching – assumes ∑vec = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑vec = 1):
        2·n·max(conv(vec, vec))
    """
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return *probability* vectors – sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _double_power(n: int, p: float) -> np.ndarray:
    """Symmetric sum of two power laws – high at both ends, low in centre."""
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return _normalize(a)
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return _normalize(a)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return _normalize(a)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return _normalize(a)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    return _normalize(a)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0
    return _normalize(a)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _arcsine_window(n: int) -> np.ndarray:
    """U‑shaped arcsine‑like density (≈1/√(x·(1‑x)))"""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """|cos|‑shaped – large at both ends, small in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    """
    Two uniform blocks placed at the left and right ends.
    block_ratio – fraction of the total length occupied by each block.
    left_weight – total mass placed on the left block (right gets 1‑left).
    """
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return *probability* vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with all concentration parameters = conc."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, cut to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose a few disjoint blocks and put uniform mass on each."""
    nb = rng.integers(2, 6)                 # number of blocks
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for _ in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0]

    # initialise logits (log‑softmax variables)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c                     # stabilising shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            w = exp_beta / sum_exp            # soft‑max weights over conv entries
            full = _conv_fft(rev_a, w)        # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Greedy hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Random tiny transfers of mass between two coordinates.
    Accepts only moves that improve the exact surrogate.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # transfer a modest fraction of mass from i to j
        eps = cur[i] * rng.uniform(0.001, 0.01)
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Sub‑gradient optimiser on the *exact* L∞‑objective
# ----------------------------------------------------------------------
def _subgradient_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          target_sum: float,
                          max_iters: int = 5000) -> np.ndarray:
    """
    Directly minimise max(conv(a,a)) using a simple sub‑gradient method.
    a is kept on the affine set {a ≥ 0,  Σa = target_sum}.
    Returns the best vector seen before the deadline.
    """
    # start_vec is a probability vector (∑ = 1); scale to target_sum
    a = start_vec * (target_sum / np.sum(start_vec))
    best_a = a.copy()
    conv = _conv_fft(a, a)
    best_M = conv.max()

    # initial step size – a few percent of total mass
    eta = target_sum * 0.05
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a, a)
        M = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑convolution at index k
        g = np.zeros_like(a)
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        if start_i <= end_i:
            i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
            j = k - i_range
            g[i_range] = 2.0 * a[j]

        # trial update
        a_trial = a - eta * g
        a_trial[a_trial < 0.0] = 0.0
        # re‑normalise to the required total mass
        s = np.sum(a_trial)
        if s == 0.0:
            # fallback: random Dirichlet restart (rare)
            rng = np.random.default_rng()
            a_trial = _random_dirichlet(n, 0.1, rng) * target_sum
        else:
            a_trial = a_trial * (target_sum / s)

        conv_trial = _conv_fft(a_trial, a_trial)
        M_trial = conv_trial.max()

        # accept if we improve
        if M_trial < M:
            a = a_trial
            M = M_trial
            if M_trial < best_M:
                best_M = M_trial
                best_a = a_trial.copy()
            # try a slightly larger step next time
            eta = eta * 1.05
        else:
            # otherwise shrink the step
            eta = eta * 0.5

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, prints progress, and stores the
    best vector in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final call to ``evaluate_sequence``
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size – a modestly large n gives a richer search space.
    # ------------------------------------------------------------------
    n = 6000                     # feel free to increase – 6000 works comfortably

    target_sum = math.sqrt(2.0 * n)   # scaling for which the true objective equals max(conv)

    # ------------------------------------------------------------------
    # 2. Heap for the best ``top_k`` seeds (by the exact surrogate)
    # ------------------------------------------------------------------
    top_k = 30
    best_heap = []          # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert a probability vector into the heap, keeping only the best ``top_k``."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # 3. Deterministic analytical seeds (wide variety)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    _push_candidate(_hamming_window(n))
    _push_candidate(_blackman_window(n))
    _push_candidate(_nuttall_window(n))

    for gamma in np.linspace(0.2, 3.0, 14):
        _push_candidate(_sine_window(n, gamma))

    for beta in np.linspace(1.0, 15.0, 12):
        _push_candidate(_kaiser_window(n, beta))

    for sigma in np.linspace(0.06, 0.30, 20):
        _push_candidate(_gaussian(n, sigma))

    for p in np.linspace(0.05, 5.0, 25):
        _push_candidate(_power(n, p))
        _push_candidate(_double_power(n, p))

    # beta‑shape – heavy U‑shape (α,β < 1) often performs well
    for _ in range(60):
        a = 10 ** rng.uniform(-2.0, 0.0)   # log‑uniform between 0.01 and 1
        b = 10 ** rng.uniform(-2.0, 0.0)
        _push_candidate(_beta_shape(n, a, b))

    _push_candidate(_arcsine_window(n))
    for gamma in np.linspace(0.5, 5.0, 12):
        _push_candidate(_cosine_edge_window(n, gamma))

    for _ in range(30):
        blk = rng.uniform(0.05, 0.4)
        lw = rng.random()
        _push_candidate(_two_block_uniform(n, block_ratio=blk, left_weight=lw))

    # ------------------------------------------------------------------
    # 4. Massive random seeding
    # ------------------------------------------------------------------
    conc_choices = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_seed_budget = 8000
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # occasional cyclic shift adds diversity
        if rng.random() < 0.3:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # exponential draws – heavier tails
    for _ in range(500):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        if rng.random() < 0.3:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # low‑frequency random curves
    for nf in (2, 3, 5, 8, 12):
        for _ in range(250):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            if rng.random() < 0.3:
                shift = rng.integers(n)
                if shift:
                    vec = np.roll(vec, shift)
            _push_candidate(vec)

    # mixed block structures
    for _ in range(300):
        if time.time() > deadline:
            break
        _push_candidate(_random_mixed_blocks(n, rng))

    # ------------------------------------------------------------------
    # 5. Re‑use the previous solution if size matches
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and np.sum(prev_arr) > 0.0:
                _push_candidate(_normalize(prev_arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Extract top seeds (ordered best → worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] kept {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 7. Initialise the global best
    # ------------------------------------------------------------------
    global_best = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Local refinements – continuation L‑BFGS on a handful of seeds
    # ------------------------------------------------------------------
    max_refinements = min(8, len(seeds))
    for idx in range(max_refinements):
        if time.time() > deadline - 10.0:
            break
        seed_vec = seeds[idx]
        # share remaining time fairly among remaining refinements
        remaining = deadline - time.time()
        portion = max(5.0, remaining * 0.55 / max_refinements)

        refined = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         time.time() + portion,
                                         beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                        200.0, 500.0, 1000.0,
                                                        2000.0, 5000.0, 10000.0,
                                                        20000.0, 50000.0,
                                                        100000.0, 200000.0,
                                                        500000.0],
                                         max_iter_per_beta=6000)
        refined_val = _objective_surrogate(refined)
        if refined_val < global_best_val:
            global_best = refined
            global_best_val = refined_val
            print(f"[Refine {idx+1}] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Sub‑gradient descent on the exact L∞ objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Sub‑gradient descent phase ---")
        sub_deadline = deadline - 3.0
        sg_best = _subgradient_optimize(global_best,
                                        n,
                                        sub_deadline,
                                        target_sum=target_sum,
                                        max_iters=8000)
        sg_val = _objective_surrogate(sg_best)
        if sg_val < global_best_val:
            global_best = sg_best
            global_best_val = sg_val
            print(f"[Sub‑grad] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Final greedy hill‑climb (tiny transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final hill‑climb ---")
        hc_seq, hc_val = _hill_climb(global_best, n, deadline, rng)
        if hc_val < global_best_val:
            global_best = hc_seq
            global_best_val = hc_val
            print(f"[Hill‑climb] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 11. Evaluate with the official evaluator and report
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for next invocation
    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
