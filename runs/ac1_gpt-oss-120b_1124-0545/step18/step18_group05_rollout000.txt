"""
fast_optimize.py

Search for a non‑negative step function (a probability vector) that minimizes

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm works in three stages:

1. **Seed generation** – a diverse collection of analytic shapes (uniform,
   triangular, Gaussian, sinusoidal windows, power‑law, arcsine) together with
   random Dirichlet draws.  All seeds are normalised to sum to one.

2. **Adam optimisation** – each of the best seeds is refined with an
   Adam optimiser on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The surrogate parameter β is increased from a low value (≈5) to a large
   value (≈5000) during the run, turning the smooth objective gradually
   into the true max‑convolution.  Adam’s adaptive learning‑rate helps to
   escape shallow local minima quickly.

3. **Polishing** – if time permits a short L‑BFGS‑B optimisation on the same
   surrogate is performed, followed by a simulated‑annealing walk in logit
   space.  The best vector found overall is returned and stored globally as
   ``height_sequence_1`` for use as a warm‑start in later calls.

The whole routine respects the 1000 s time budget and never exceeds the
allowed CPU usage.  All helper functions are defined at module level and
avoid closures or lambda expressions.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for L‑BFGS‑B (used only for polishing)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:          # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exact surrogate used for quick ranking (sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 150_000,
                   beta_start: float = 5.0,
                   beta_end: float = 5000.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed surrogate

        Fβ(a) = 2·n·[ max(conv) + log Σ exp(β·(conv‑max)) / β ].

    Returns the best vector found (normalised) and its surrogate value.
    """
    v = np.log(start_vec + 1e-30)      # logits
    m = np.zeros_like(v)               # first moment
    v2 = np.zeros_like(v)              # second moment

    # Adam hyper‑parameters
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_seq = None
    best_val = np.inf

    for it in range(1, max_iters + 1):
        # time check – abort early if we are out of budget
        if time.time() > deadline:
            break

        # linear schedule for β
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)

        a = _softmax(v)                     # current probability vector
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shifted = conv - max_c
        exp_beta = np.exp(beta * shifted)   # all entries ≤ 0 → safe
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp               # soft weights, sum = 1

        # gradient wrt a of the smoothed objective
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)         # chain rule through softmax

        # Adam update in logit space
        m = b1 * m + (1 - b1) * grad_v
        v2 = b2 * v2 + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v2_hat = v2 / (1 - b2 ** it)

        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

        # periodic evaluation of the *exact* surrogate (no smoothing)
        if it % 3000 == 0:
            cur_val = _objective_surrogate(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()
                print(f"[Adam] it={it:,} best surrogate {best_val:.6f}")

    # final evaluation if we never updated best_seq inside the loop
    final_a = _softmax(v)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_val = final_val
        best_seq = final_a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 1500) -> np.ndarray:
    """Polish a probability vector with L‑BFGS‑B on the smoothed surrogate."""
    if _opt is None:                         # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80, 200, 500, 1_000, 2_000, 5_000, 20_000]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)   # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                 # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6      # temperature schedule
        sigma = 0.15 * (1.0 - progress) + 1e-4   # proposal scale

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

    The search consists of:
      1. building a pool of diverse seeds,
      2. refining the best seeds with Adam on a smooth surrogate,
      3. optional L‑BFGS polishing,
      4. optional simulated‑annealing walk.
    The best vector found is stored globally as ``height_sequence_1``
    and returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 3000 offers a good speed / quality trade‑off
    # ------------------------------------------------------------------
    n = 3000

    # ------------------------------------------------------------------
    # 1. Build a diverse seed pool
    # ------------------------------------------------------------------
    def _norm(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / len(v))
        return v / s

    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sin2(n))
    seeds.append(_hann_window(n))
    seeds.append(_arcsine(n))

    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # random Dirichlet draws with a variety of concentration parameters
    conc_params = np.logspace(-2, 1, 15)   # 0.01 … 10
    for conc in conc_params:
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # include previous best if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,) and arr.sum() > 0.0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # normalise everything onto the probability simplex
    seeds = [_norm(np.asarray(v, dtype=np.float64)) for v in seeds]

    # ------------------------------------------------------------------
    # 2. Rank seeds quickly and keep a few elite candidates
    # ------------------------------------------------------------------
    seed_vals = [_objective_surrogate(s) for s in seeds]
    elite_idx = np.argsort(seed_vals)[:5]   # keep the 5 best seeds
    elite_seeds = [seeds[i].copy() for i in elite_idx]

    print(f"[Init] best surrogate among raw seeds: {seed_vals[elite_idx[0]]:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam optimisation on each elite seed
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = np.inf

    for idx, seed_vec in enumerate(elite_seeds, 1):
        remaining = deadline - time.time()
        if remaining < 30.0:
            print("[Adam] insufficient time left, aborting further restarts")
            break

        # allocate a modest number of iterations; _adam_optimize will
        # stop early if the deadline is hit.
        max_iters = int(min(120_000, remaining * 1800))  # rough iteration budget
        print(f"[Adam] start seed {idx}/{len(elite_seeds)} – max_iters={max_iters}")

        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)

        print(f"[Adam] seed {idx} finished – best surrogate {val:.6f}")

        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Update] new global best surrogate after Adam: {global_best_val:.6f}")

    if global_best_seq is None:
        # fall back to the best raw seed (should never happen)
        best_raw_idx = elite_idx[0]
        global_best_seq = seeds[best_raw_idx].copy()
        global_best_val = seed_vals[best_raw_idx]

    # ------------------------------------------------------------------
    # 4. Optional L‑BFGS polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a fraction of the remaining budget to L‑BFGS
        max_iter_per_beta = max(1200, int(remaining * 0.02 * 1e5))
        print("\n--- L‑BFGS polishing ---")
        seq_polished = _lbfgs_optimize(global_best_seq, n, deadline,
                                       beta_schedule=[1, 5, 20, 80, 200,
                                                      500, 1_000, 2_000,
                                                      5_000, 20_000],
                                       max_iter_per_beta=max_iter_per_beta)
        val_polished = _objective_surrogate(seq_polished)
        print(f"[L‑BFGS] polished surrogate = {val_polished:.6f}")
        if val_polished < global_best_val:
            global_best_val = val_polished
            global_best_seq = seq_polished.copy()
            print(f"[Update] new global best after L‑BFGS = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa_iters = int(min(500_000, remaining * 1500))
        max_sa_iters = max(200_000, max_sa_iters)
        print("\n--- Simulated‑Annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_opt(global_best_seq, n, deadline,
                                                  max_iters=max_sa_iters)
        print(f"[SA] final surrogate = {val_sa:.6f}")
        if val_sa < global_best_val:
            global_best_val = val_sa
            global_best_seq = seq_sa.copy()
            print(f"[Update] new global best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
