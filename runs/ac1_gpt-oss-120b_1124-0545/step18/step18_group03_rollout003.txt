def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0

    n = 4000

    def objective(seq):
        # compute objective quickly
        conv = _conv_fft(seq, seq)
        max_conv = np.max(conv)
        sum_a = seq.sum()
        if sum_a < 1e-12:
            return np.inf
        return 2.0 * n * max_conv / (sum_a * sum_a)

   # seed generation:
   seeds = []
   seeds.append(_uniform(n))
   seeds.append(_triangular(n))
   for p in np.linspace(0.2, 1.2, 10):
        seeds.append(_power_law(n, p))
   for sigma in [n/5, n/4, n/3, n/2, n/1.5, n]:
        seeds.append(_gaussian(n, sigma, mu=n/2))
   # random Dirichlet
   for conc in [0.5, 1.0, 2.0, 5.0, 10.0]:
        seeds.append(_random_dirichlet(n, conc, rng))
   # random sparse
   for frac in [0.02, 0.05, 0.1]:
        seeds.append(_random_sparse(n, frac, rng))

   Evaluate each seed, keep best.

   best_seq = best; best_val = best_val.

   # Additional: start from previous best if available.

   if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.shape[0]==n:
                seeds.append(prev/prev.sum())

   # main loop
   iteration = 0
   accept = 0
   while time.time() < deadline:
        iteration += 1
        # Choose operation based on iteration number:
        if iteration % 10 ==0:
            # gradient step with exponentiated gradient maybe
            g = _subgradient_maxconv(best_seq)
            eta = 0.5 / np.sqrt(iteration)
            log_best = np.log(best_seq + 1e-30) - eta * g
            cand = _softmax(log_best)
        else:
            # random mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i==j or best_seq[i]<=0:
                continue
            # step fraction decays over time
            step_frac = 0.01 * max(0.001, 1.0 - iteration/500000.0)
            delta = step_frac * best_seq[i]
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
        val = objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
            accept += 1
        # occasionally restart from random Dirichlet as new candidate (exploratory)
        if iteration % 20000 == 0:
            start_vec = _random_dirichlet(n, rng.uniform(0.5,5.0), rng)
            val_rand = objective(start_vec)
            if val_rand < best_val:
                best_val = val_rand
                best_seq = start_vec
                print(f"[Exploration] new best from random Dirichlet at iter {iteration}: {best_val:.6f}")

        # log progress occasionally
        if iteration % 5000 == 0:
            print(f"[Iter {iteration}] best_val={best_val:.6f}, accept={accept}")

   # final evaluation with evaluate_sequence
   try:
        true_val = evaluate_sequence(best_seq.tolist())
   except Exception as e:
        true_val = None
   print(f"Finished after {iteration} iterations. Best objective approx {best_val:.6f}, evaluate_sequence = {true_val}")

   globals()["height_sequence_1"] = best_seq.tolist()
   return best_seq.tolist()

    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0:
        return np.full_like(v, 1.0/len(v))
    return e / s

    """Full linear convolution via real FFT (O(n log n))."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv

    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

        if cand_val < cur_val or rng.random() < np.exp(-(cand_val - cur_val) * 10.0):
            cur_seq = cand
            cur_val = cand_val
            accept_cnt += 1

        # --- update global best ------------------------------------
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand.copy()
            print(f"[Iter {iteration}] new global best = {best_val:.6f}")

        # --- occasional progress report -----------------------------
        if iteration % report_every == 0:
            elapsed = time.time() - start_time
            print(f"[Iter {iteration}] time={elapsed:.1f}s best={best_val:.6f} accept={accept_cnt}")

        # --- random restarts for exploration ------------------------
        if iteration % 50000 == 0:
            conc = rng.uniform(0.1, 5.0)
            restart = _random_dirichlet(n, conc, rng)
            restart_val = _objective(restart)
            if restart_val < best_val:
                best_val = restart_val
                best_seq = restart.copy()
                cur_seq = restart.copy()
                cur_val = restart_val
                print(f"[Restart] improved best to {best_val:.6f} at iter {iteration}")

    # --------------------------------------------------------------
    # optional final polishing using a high‑β L‑BFGS run (if time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        polished = _lbfgs_polish(best_seq, n, deadline)
        polished_val = _objective(polished)
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished
            print(f"[Polish] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # final evaluation with the official evaluator (just for logging)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished === best≈{best_val:.6f}, evaluate_sequence={true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in future calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
