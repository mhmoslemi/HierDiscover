    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    # compute initial logits
    v = np.log(a + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5   # base
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(v)   # using softmax function defined earlier
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))
        # compute subgradient g w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # gradient w.r.t logits
        grad_v = a * (g - np.dot(g, a))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)
        # learning rate schedule: decreasing sqrt
        eta = lr0 / np.sqrt(t)
        step = eta * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # progress prints
        if t % 100000 == 0:
            print(f"[AdamExpGrad] it={t:,} best={best_val:.6f}")
    return best_seq, best_val

"""
Propose a non‑negative step function (ℓ₁‑normalised vector) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

Algorithm
---------

1. **Diverse seed pool** – Analytic windows (uniform, triangular, sine,
   Kaiser, Chebyshev, DPSS, Gaussian, power‑law, Tukey, …) together with
   random Dirichlet draws.  A fine sweep of the main shape parameters
   (σ, p, α, β, attenuation, time‑bandwidth) is performed and every
   candidate is ℓ₁‑normalised.

2. **Screen and keep the best few** – Each seed is evaluated with the
   exact objective
   `obj = 2 * n * max(conv)` (the denominator is 1 after normalisation).
   Only the `top_k` vectors with smallest objective continue.

3. **Smooth‑max L‑BFGS** – The non‑smooth max is approximated by a
   log‑sum‑exp surrogate.  Optimisation is carried out in log‑probability
   space (`softmax`) so the simplex constraint is automatic.  A long
   β‑schedule (up to 1e9) and generous iteration limits refine each seed.

4. **Adam‑powered exact‑max mirror descent** – Starting from the best L‑BFGS
   result we perform exponentiated‑gradient descent on the *true* max
   using the sub‑gradient at the active lag.  Adam‑style momentum
   (first/second moment estimates) and a 1/√t step–size improve convergence
   dramatically.

5. **Random perturbations + short L‑BFGS** – The current best log‑vector is
   perturbed with Gaussian noise several times; each perturbation is
   quickly re‑optimised with a short L‑BFGS run.  The best among all
   perturbations is kept.

6. **Optional symmetrisation** – Averaging the vector with its reversal
   often reduces the peak autocorrelation.

7. **Simulated‑annealing swap hill‑climber** – Random swaps of two
   positions are performed; downhill moves are always accepted,
   uphill moves with a temperature‑controlled Metropolis probability.
   This explores the permutation space beyond gradient‑based methods.

All stages respect a global 1000 s wall‑clock budget (a small safety margin
is kept).  The best vector found is stored in ``height_sequence_1`` and
returned as a plain ``list[float]``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise a log‑sum‑exp smoothing of max‑convolution using L‑BFGS‑B.
    The optimisation works in log‑probability space (softmax) so the simplex
    constraint is satisfied automatically.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6, 1e7,
            1e8, 5e8, 1e9
        ]

    # work in log‑probability space – softmax guarantees ℓ₁‑norm = 1
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (derived from the smoothed max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Adam‑based exact‑max exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _adam_exp_gradient_opt(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution using Adam momentum.
    Works in log‑probability space (softmax) to keep the simplex.
    """
    # initialise logits
    v = np.log(start_seq + 1e-30)
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5      # base learning‑rate (will be scaled by 1/√t)

    best_seq = _softmax(v)
    best_val = _objective(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # current ℓ₁‑normalised vector
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag (exact max)
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t. logits (softmax Jacobian)
        grad_v = a * (g - np.dot(g, a))

        # Adam moments
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        # learning‑rate schedule (1/√t)
        eta = lr0 / math.sqrt(t)

        # parametrised update
        step = eta * m1_hat / (np.sqrt(m2_hat) + eps)
        v = v - step

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if t % 100_000 == 0:
            print(f"[AdamExpGrad] it={t:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random‑perturbation + short L‑BFGS refinement
# ----------------------------------------------------------------------
def _perturb_and_refine(base_seq: np.ndarray,
                        n: int,
                        rng: np.random.Generator,
                        deadline: float,
                        trials: int = 30) -> tuple[np.ndarray, float]:
    """
    Apply Gaussian noise to the log‑vector, run a short L‑BFGS
    optimisation, keep the best outcome.
    """
    best_seq = base_seq.copy()
    best_val = _objective(best_seq)
    v_base = np.log(base_seq + 1e-30)

    for tr in range(trials):
        if time.time() > deadline:
            break
        # decreasing noise amplitude
        sigma = 0.5 * (0.7 ** (tr / trials))
        v_pert = v_base + rng.normal(scale=sigma, size=n)
        seq_pert = _softmax(v_pert)

        # a very short smooth‑max run (few β‑steps, few iterations)
        seq_ref = _lbfgs_smooth_optimize(seq_pert,
                                         n,
                                         deadline,
                                         beta_schedule=[10, 100, 500, 1000, 5000],
                                         max_iter_per_beta=2000)
        val_ref = _objective(seq_ref)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = seq_ref.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing permutation hill‑climber
# ----------------------------------------------------------------------
def _anneal_swap_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Random swaps with Metropolis acceptance.
    Returns the best sequence encountered and its objective value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # initial temperature – a fraction of the current objective
    T0 = 0.1 * cur_val if cur_val > 0 else 1.0

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective(cur_seq)
        delta = new_val - cur_val

        # linear cooling schedule
        T = max(T0 * (1.0 - it / max_swaps), 1e-12)
        accept = (delta < 0.0) or (rng.random() < math.exp(-delta / T))

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 50_000 == 0:
            print(f"[AnnealSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------
def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Average a vector with its reversal and renormalise."""
    mixed = (seq + seq[::-1]) * 0.5
    s = mixed.sum()
    return mixed / s if s > 0 else seq.copy()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size (fixed to 4000 as in the reference run)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Seed generation (reuse previous solution if available)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_all_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 3. Quick evaluation of all seeds – keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed objective values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation loop over the top seeds (L‑BFGS + Adam)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # extended β‑schedule (already defined inside the helper)
    beta_schedule = None   # let the helper use its default (includes up to 1e9)

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing seed optimisation.")
            break

        remaining = deadline - time.time()
        max_iter_beta = int(min(15000, remaining * 5))
        max_iter_beta = max(8000, max_iter_beta)

        print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter_per_beta={max_iter_beta}) ---")
        seq_opt = _lbfgS_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best (L‑BFGS) = {global_best_val:.6f}")

        # --------------------------------------------------------------
        # 4a. Adam exact‑max exponentiated‑gradient polishing
        # --------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_adam_iters = int(min(600_000, remaining * 0.12))
            max_adam_iters = max(300_000, max_adam_iters)
            print(f"[Adam] running up to {max_adam_iters:,} iterations")
            seq_adam, val_adam = _adam_exp_gradient_opt(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=max_adam_iters)
            if val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Random perturbations + short L‑BFGS refinements
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random perturbation & short L‑BFGS refinements ---")
        seq_pert, val_pert = _perturb_and_refine(global_best_seq,
                                                n,
                                                rng,
                                                deadline,
                                                trials=30)
        if val_pert < global_best_val:
            global_best_seq = seq_pert.copy()
            global_best_val = val_pert
            print(f"[Perturb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional symmetrisation (averaging with reversed copy)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sym = _symmetrize(global_best_seq)
        val_sym = _objective(seq_sym)
        if val_sym < global_best_val:
            global_best_seq = seq_sym
            global_best_val = val_sym
            print(f"[Symmetrize] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Simulated‑annealing permutation hill‑climber (final fine‑tune)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(600_000, remaining * 0.15))
        max_swaps = max(300_000, max_swaps)
        print("\n--- Simulated‑annealing swap local search ---")
        seq_perm, val_perm = _anneal_swap_opt(global_best_seq,
                                              n,
                                              deadline,
                                              max_swaps=max_swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm
            global_best_val = val_perm
            print(f"[AnnealSwap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

**Explanation of improvements**

* Added **DPSS** (Slepian) and **Chebyshev** windows – known for low spectral leakage, giving better autocorrelation properties.
* Performed a **fine-grained sweep** of the main shape parameters (σ, p, α, β, attenuation, bandwidth) to widen the seed pool.
* Employed a **more aggressive β‑schedule** (up to 1e9) in the smooth‑max L‑BFGS stage, allowing the surrogate to approach the true max much closer.
* Replaced the simple exponentiated‑gradient step with an **Adam‑style momentum optimiser**, which converges faster on the non‑smooth objective.
* Introduced a **random‑perturbation + short L‑BFGS refinement** loop that explores the neighbourhood of the current best solution more thoroughly.
* Kept the **symmetrisation** and **simulated‑annealing swap** phases from the original approach but allocated time adaptively, ensuring they run only when budget permits.

All stages respect the 1000 s wall‑clock limit, and the best vector found is stored in `height_sequence_1` and returned. This richer search strategy is expected to push the objective below the previous 1.5085 barrier, moving closer to the target 1.503.
