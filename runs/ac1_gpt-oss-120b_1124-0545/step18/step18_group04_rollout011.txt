"""
search for a step‑function (non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The approach used here is a very aggressive simulated‑annealing / hill‑climb
that works directly on the soft‑max (log‑probability) parametrisation.
All candidates are normalised to sum‑one so the objective reduces to
`2 * n * max(conv)`.

The algorithm:
  1. Create a small pool of analytical “seed” vectors (uniform, triangular,
     a few windows, beta‑shapes, random Dirichlet draws …).
  2. Keep the best few seeds (by the exact objective).
  3. Starting from the current best, run a long simulated‑annealing loop.
     Each iteration perturbs the log‑softmax vector, evaluates the exact
     objective (using a fast FFT‑based convolution) and accepts the move
     if it improves or with a temperature‑dependent probability.
  4. Return the best vector found and store it in the global variable
     ``height_sequence_1`` for the next call.
"""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(N log N))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for long vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _conv_max(vec: np.ndarray) -> float:
    """Return max entry of the linear self‑convolution."""
    # Using the FFT routine is faster for n≈4000 than np.convolve in pure Python.
    conv = _conv_fft(vec, vec)
    return float(np.max(conv))


# ----------------------------------------------------------------------
# Simple mathematics on the simplex
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Exact objective (assumes the vector already sums to 1)
# ----------------------------------------------------------------------
def _objective(vec: np.ndarray) -> float:
    """Exact evaluate_sequence value for a unit‑sum vector."""
    n = len(vec)
    max_b = _conv_max(vec)
    # sum(vec) == 1, therefore denominator = 1
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _arcsine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    """Two flat blocks placed at the ends."""
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet sample with all concentration parameters equal to ``conc``."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _random_blocks(n: int, rng: np.random.Generator,
                   max_blocks: int = 5) -> np.ndarray:
    """Create a few disjoint uniform blocks with random sizes and masses."""
    nb = rng.integers(2, max_blocks + 1)
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for b in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget (≈1000 s) and stores the best
    vector in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 1. Problem size – a compromise between quality and speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Gather a modest pool of analytical seeds
    # --------------------------------------------------------------
    seed_vectors = []
    def _push(v: np.ndarray):
        seed_vectors.append(_normalize(v))

    _push(_uniform(n))
    _push(_triangular(n))
    for g in np.linspace(0.2, 3.0, 10):
        _push(_sine_window(n, g))
    for a in np.logspace(-2, 0, 12):
        for b in np.logspace(-2, 0, 12):
            _push(_beta_shape(n, a, b))
    _push(_arcsine_window(n))
    for blk in np.linspace(0.05, 0.35, 12):
        for lw in np.linspace(0.1, 0.9, 9):
            _push(_two_block_uniform(n, block_ratio=blk, left_weight=lw))

    # --------------------------------------------------------------
    # 3. Massive random seeding (Dirichlet + exponential + blocks)
    # --------------------------------------------------------------
    conc_choices = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_budget = 5000
    for _ in range(random_budget):
        if time.time() > deadline:
            break
        if rng.random() < 0.6:          # Dirichlet family
            conc = rng.choice(conc_choices)
            vec = _random_dirichlet(n, conc, rng)
        elif rng.random() < 0.8:        # Exponential family
            vec = _random_exponential(n, rng)
        else:                           # Block‑type structures
            vec = _random_blocks(n, rng, max_blocks=6)
        # occasional cyclic shift (changes overlap pattern)
        if rng.random() < 0.3:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        seed_vectors.append(vec)

    # --------------------------------------------------------------
    # 4. Re‑use the previous best if available
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0.0:
                seed_vectors.append(_normalize(prev_arr))
        except Exception:
            pass

    # --------------------------------------------------------------
    # 5. Evaluate all seeds and keep the top‑k
    # --------------------------------------------------------------
    top_k = 20
    heap = []                         # min‑heap on the *negative* objective
    for vec in seed_vectors:
        val = _objective(vec)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, vec))
        else:
            if val < -heap[0][0]:
                heapq.heappushpop(heap, (-val, vec))

    best_vec = heap[0][1].copy()
    best_val = -heap[0][0]
    print(f"[Init] best surrogate from seed pool = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Simulated annealing / hill‑climb on the best seed
    # --------------------------------------------------------------
    # Working in log‑softmax space guarantees feasibility.
    logits = np.log(best_vec + 1e-30)
    temperature = 1.0
    cooling = 0.9985

    while time.time() < deadline:
        # perturb logits – Gaussian noise scaled with the current temperature
        perturb = rng.normal(scale=temperature, size=n)
        new_log = logits + perturb
        new_vec = _softmax(new_log)          # stays on the simplex
        new_val = _objective(new_vec)

        # classic SA acceptance rule
        if new_val < best_val or rng.random() < math.exp(-(new_val - best_val) / max(temperature, 1e-12)):
            logits = new_log
            best_vec = new_vec
            best_val = new_val

        temperature *= cooling

    # --------------------------------------------------------------
    # 7. Final exact evaluation and bookkeeping
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
