Hybrid random‑seed + L‑BFGS + sub‑gradient + simulated‑annealing optimizer
for the step‑function evaluation problem.

The algorithm works in three stages:

1. **Global seeding** – a large, diverse set of analytically‑parameterised
   and fully random vectors (Dirichlet, mixtures of Gaussians, sparse spikes,

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and ‖v‖₁ = 1.
    """
    # First clip to non‑negative.
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        return np.full_like(v, 1.0 / len(v))

    # Classic O(n log n) projection.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – fall back to uniform.
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact surrogate (sum = 1 assumed)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """Exact objective for a probability vector: 2·n·max(conv(vec, vec))."""
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _proj_simplex(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _proj_simplex(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _proj_simplex(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _proj_simplex(a)


def _double_power(n: int, p: float) -> np.ndarray:
    """Symmetric sum of two power laws – high at both ends, low in the centre."""
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _proj_simplex(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _proj_simplex(a)


def _arcsine_window(n: int) -> np.ndarray:
    """U‑shaped arcsine‑like density  (≈1/√(x·(1‑x)))"""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _proj_simplex(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """|cos|‑shaped – large at both ends, small in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _proj_simplex(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    """
    Two uniform blocks placed at the left and right ends.
    block_ratio – fraction of the total length occupied by each block.
    left_weight – total mass placed on the left block (right gets 1‑left).
    """
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _proj_simplex(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with all concentration parameters = conc."""
    return _proj_simplex(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _proj_simplex(a)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, cut to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _proj_simplex(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose a few disjoint blocks and put uniform mass on each."""
    nb = rng.integers(2, 6)                 # number of blocks
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for _ in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()      # share of the remaining mass
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _proj_simplex(a)


def _random_gaussian_mixture(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of *k* Gaussians with random centres, widths and amplitudes."""
    a = np.zeros(n, dtype=np.float64)
    xs = np.arange(n, dtype=np.float64)
    for _ in range(k):
        centre_frac = rng.random()
        mu = centre_frac * (n - 1)
        sigma_frac = rng.uniform(0.01, 0.2)
        sigma = sigma_frac * n
        amp = rng.random()
        a += amp * np.exp(-((xs - mu) ** 2) / (2.0 * sigma ** 2))
    return _proj_simplex(a)


def _random_spike_vector(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k random spikes of random weight (then normalised)."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.random(k)
    a[idx] = weights
    return _proj_simplex(a)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """LBFGS on the smoothed max‑convolution.  Operates in log‑softmax space."""
    if _opt is None:
        raise RuntimeError("SciPy optimizer required for L‑BFGS phase.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1000.0, 2000.0, 5000.0, 10000.0,
            20000.0, 50000.0, 100000.0, 200000.0,
            500000.0, 1e6, 2e6, 5e6, 1e7,
        ]

    # initialise logits (log‑softmax variables)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)  # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smoothed max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            w = exp_beta / sum_exp
            full = _conv_fft(rev_a, w)  # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(start_vec: np.ndarray,
                         n: int,
                         deadline: float,
                         init_step: float = 0.1) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient method for the true max‑convolution objective."""
    cur = start_vec.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val
    step = init_step

    while time.time() < deadline:
        conv = _conv_fft(cur, cur)
        max_val = conv.max()
        # mask of the indices attaining the max (could be multiple)
        mask = (conv == max_val).astype(np.float64)

        # sub‑gradient = 2 * conv(reverse(cur), mask) on the relevant slice
        rev_cur = cur[::-1]
        full = _conv_fft(rev_cur, mask)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # descent step with projection onto simplex
        trial = cur - step * grad
        trial = _proj_simplex(trial)
        trial_val = _objective_surrogate(trial)

        if trial_val < cur_val:
            cur, cur_val = trial, trial_val
            if trial_val < best_val:
                best, best_val = trial, trial_val
            # after a successful step we keep the step size unchanged
        else:
            # decrease step size when no improvement
            step *= 0.5
            if step < 1e-12:
                break

    return best, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def _annealing_hill_climb(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves on the simplex.
    Occasionally accepts worsening moves according to a linear temperature schedule.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    # Initial temperature – 1 % of the current objective as a heuristic.
    T0 = cur_val * 0.01
    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best, best_val

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        T = T0 * max(0.0, 1.0 - elapsed / total_time)

        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        # Transfer a random fraction (0 % … 10 %) of mass from i to j.
        eps = cur[i] * rng.uniform(0.0, 0.10)
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = _proj_simplex(cand)          # safeguard the simplex (tiny rounding errors)

        cand_val = _objective_surrogate(cand)
        delta = cand_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best, best_val = cand, cand_val

    return best, best_val


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, prints progress information,
    and stores the best vector in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size – a compromise between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Keep the best ``top_k`` seeds (heap of size top_k).
    # ------------------------------------------------------------------
    top_k = 30
    best_heap = []          # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert a candidate into the heap while keeping only the best ``top_k``."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # 3. Deterministic analytical seeds (wide variety)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        _push_candidate(_sine_window(n, gamma))
    _push_candidate(_arcsine_window(n))
    for gamma in np.linspace(0.5, 4.0, 9):
        _push_candidate(_cosine_edge_window(n, gamma))
    _push_candidate(_hamming_window if 'hamming_window' in globals() else _sine_window(n, 1.0))
    # two‑block shapes
    for _ in range(20):
        blk = rng.uniform(0.05, 0.3)
        lw = rng.random()
        _push_candidate(_two_block_uniform(n, block_ratio=blk, left_weight=lw))
    # beta‑shape family (U‑shaped)
    for _ in range(25):
        a = 10 ** rng.uniform(-2, 0)   # log‑uniform 0.01 … 1
        b = 10 ** rng.uniform(-2, 0)
        _push_candidate(_beta_shape(n, a, b))

    # ------------------------------------------------------------------
    # 4. Massive random seeding
    # ------------------------------------------------------------------
    # Dirichlet draws with a wide range of concentration parameters
    conc_choices = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_seed_budget = 6000
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # occasional cyclic shift
        if rng.random() < 0.2:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # Exponential draws – heavier tails
    for _ in range(800):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        if rng.random() < 0.2:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # Low‑frequency cosine curves
    for nf in (2, 3, 5, 8):
        for _ in range(300):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            if rng.random() < 0.2:
                shift = rng.integers(n)
                if shift:
                    vec = np.roll(vec, shift)
            _push_candidate(vec)

    # Random mixed‑block structures
    for _ in range(500):
        if time.time() > deadline:
            break
        _push_candidate(_random_mixed_blocks(n, rng))

    # Gaussian mixture families
    for k in (2, 3, 4):
        for _ in range(400):
            if time.time() > deadline:
                break
            vec = _random_gaussian_mixture(n, k, rng)
            _push_candidate(vec)

    # Sparse spike vectors (very few non‑zero entries)
    for k in (3, 5, 10):
        for _ in range(300):
            if time.time() > deadline:
                break
            vec = _random_spike_vector(n, k, rng)
            _push_candidate(vec)

    # ------------------------------------------------------------------
    # 5. Re‑use the previous solution if it exists and matches the size.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                _push_candidate(_proj_simplex(prev_arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Extract the best seeds (ordered from best to worst).
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] retained {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 7. Initialise the global best.
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Local refinement – continuation L‑BFGS on a few top seeds.
    # ------------------------------------------------------------------
    max_refinements = min(7, len(seeds))
    for idx in range(max_refinements):
        if time.time() > deadline - 10.0:
            break
        seed_vec = seeds[idx]
        remaining = deadline - time.time()
        # reserve ~40 % of remaining time for this seed (but never less than 5 s)
        refine_deadline = time.time() + max(5.0, remaining * 0.4 / max_refinements)

        try:
            refined = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             refine_deadline,
                                             beta_schedule=None,
                                             max_iter_per_beta=6000)
        except RuntimeError:
            refined = seed_vec  # fallback – SciPy missing

        refined_val = _objective_surrogate(refined)
        if refined_val < global_best_val:
            global_best_seq, global_best_val = refined, refined_val
            print(f"[L‑BFGS Refine {idx+1}] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Sub‑gradient polishing (few seconds per candidate).
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.2)
    if time.time() < subgrad_deadline:
        polished, polished_val = _subgradient_descent(global_best_seq,
                                                      n,
                                                      subgrad_deadline,
                                                      init_step=0.2)
        if polished_val < global_best_val:
            global_best_seq, global_best_val = polished, polished_val
            print(f"[Sub‑gradient] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Final simulated‑annealing hill‑climb (use any leftover time).
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final annealing hill‑climb ---")
        hc_seq, hc_val = _annealing_hill_climb(global_best_seq,
                                               n,
                                               deadline,
                                               rng)
        if hc_val < global_best_val:
            global_best_seq, global_best_val = hc_seq, hc_val
            print(f"[Annealing] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 11. Evaluate with the official evaluator.
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
