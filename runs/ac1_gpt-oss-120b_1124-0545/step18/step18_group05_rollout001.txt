"""
Optimisation of a non‑negative step function (length 4000) to minimise

        2·n·max(conv(a, a)) / (sum(a)²)

The trick is to enforce symmetry (a[i]=a[n‑1‑i]), which halves the
number of free variables.  The variables are represented by logits of a
probability vector on the half‑interval; the full vector is obtained by
mirroring and scaling by 0.5 so that the sum of the full vector is 1.
A smooth surrogate of the max‑convolution (log‑sum‑exp) is minimised
with Adam in logit space, using a schedule of increasing β (the
smoothness parameter).  After the main optimisation a short simulated‑
annealing walk in the full logit space is performed for polishing.
The best vector found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# Optional L‑BFGS polishing (only if SciPy is available)
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Symmetric construction helpers
# ----------------------------------------------------------------------
def _sym_full(p_half: np.ndarray) -> np.ndarray:
    """
    Build the full symmetric vector from a half‑probability vector.
    The returned vector sums to 1.
    """
    return np.concatenate([p_half, p_half[::-1]]) * 0.5


# ----------------------------------------------------------------------
# Exact surrogate (objective when sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(full_seq: np.ndarray) -> float:
    """Exact value 2·n·max(conv(full_seq, full_seq)) for a probability vector."""
    n = len(full_seq)
    conv = _conv_fft(full_seq, full_seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (half‑length, already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform_half(m: int) -> np.ndarray:
    return np.full(m, 1.0 / m, dtype=np.float64)


def _triangular_half(m: int) -> np.ndarray:
    i = np.arange(m, dtype=np.float64)
    c = (m - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform_half(m)


def _gaussian_half(m: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * m
    mu = (m - 1) / 2.0
    i = np.arange(m, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform_half(m)


def _power_half(m: int, p: float) -> np.ndarray:
    idx = np.arange(1, m + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform_half(m)


def _sin2_half(m: int) -> np.ndarray:
    i = np.arange(m, dtype=np.float64)
    x = i / (m - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform_half(m)


def _hann_half(m: int) -> np.ndarray:
    i = np.arange(m, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (m - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform_half(m)


def _arcsine_half(m: int) -> np.ndarray:
    i = np.arange(m, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (m - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform_half(m)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate (symmetric variables)
# ----------------------------------------------------------------------
def _optimize_symmetric(p0: np.ndarray,
                        n: int,
                        deadline: float,
                        beta_schedule=None,
                        steps_per_beta: int = 2000) -> tuple[np.ndarray, float]:
    """
    Optimise a half‑probability vector `p0` (size n/2) using Adam on the
    smooth surrogate  f_β(a) = M + (1/β) log Σ exp(β(conv-M)).
    Returns the best full vector and its exact surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 2_000.0, 8_000.0, 20_000.0,
            50_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    m = len(p0)                 # = n/2
    v = np.log(p0 + 1e-30)      # logits for the half‑vector
    m1 = np.zeros_like(v)       # Adam first moment
    m2 = np.zeros_like(v)       # Adam second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0

    best_full = None
    best_val = np.inf

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # adapt a modest learning‑rate that shrinks with β
        base_lr = 0.08 / (beta ** 0.25 + 1.0)
        lr = base_lr

        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            t += 1

            # probability on half‑interval
            p_half = _softmax(v)                # sum = 1
            # full symmetric vector (sums to 1)
            a_full = _sym_full(p_half)          # length n

            # full convolution and its maximum
            conv_full = _conv_fft(a_full, a_full)
            M = np.max(conv_full)

            # smooth weights for the max‑convolution
            w = np.exp(beta * (conv_full - M))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of Σ w_i * conv_i  w.r.t. a_full
            rev_a = a_full[::-1]
            grad_conv = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * grad_conv[n - 1:n - 1 + n]  # central n entries

            # map gradient back to half‑variables (mirrored)
            grad_half = 0.5 * (grad_a[:m] + grad_a[-1:-m - 1:-1])

            # chain rule through the soft‑max
            dot = np.dot(p_half, grad_half)
            grad_v = p_half * (grad_half - dot)

            # Adam update in logit space
            m1 = beta1 * m1 + (1.0 - beta1) * grad_v
            m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

            m1_hat = m1 / (1.0 - beta1 ** t)
            m2_hat = m2 / (1.0 - beta2 ** t)

            v -= lr * m1_hat / (np.sqrt(m2_hat) + eps)

            # keep logits in a reasonable range to avoid overflow
            if np.max(np.abs(v)) > 50.0:
                v = np.clip(v, -50.0, 50.0)

            # occasional full‑objective check
            if t % 250 == 0:
                cur_val = _objective_surrogate(a_full)
                if cur_val < best_val:
                    best_val = cur_val
                    best_full = a_full.copy()

    # final check if never updated inside loop
    if best_full is None:
        best_full = _sym_full(_softmax(v))
        best_val = _objective_surrogate(best_full)

    return best_full, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (full logit space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

    # ------------------------------------------------------------------
    seeds = []

    seeds.append(_uniform_half(half))
    seeds.append(_triangular_half(half))
    seeds.append(_sin2_half(half))
    seeds.append(_hann_half(half))
    seeds.append(_arcsine_half(half))

    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian_half(half, sigma))

    for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power_half(half, p))

    # random Dirichlet draws with a range of concentration parameters
    conc_params = np.logspace(-2, 1, 12)  # 0.01 … 10
    for conc in conc_params:
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(half, conc)))

    # include the previous best if available (symmetrised)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                # symmetrise the previous vector (simple averaging)
                sym = 0.5 * (arr + arr[::-1])
                sym = np.maximum(sym, 0.0)
                sym /= sym.sum()
                # extract half‑vector
                half_vec = sym[:half] * 2.0   # because full = 0.5 * [p,p_rev]
                half_vec = np.maximum(half_vec, 0.0)
                half_vec /= half_vec.sum()
                seeds.append(half_vec)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Optimise each seed (symmetric Adam optimiser)
    # ------------------------------------------------------------------
    best_full = None
    best_val = np.inf
    seed_counter = 0

    for p_seed in seeds:
        if time.time() > deadline:
            break
        seed_counter += 1
        print(f"\n[Seed {seed_counter}] starting optimisation")
        full_seq, val = _optimize_symmetric(p_seed, n, deadline,
                                            beta_schedule=[1, 5, 20, 80, 200,
                                                           500, 2000, 8000,
                                                           20000, 50000, 200000,
                                                           500000, 1_000_000],
                                            steps_per_beta=2500)
        print(f"[Seed {seed_counter}] best surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_full = full_seq.copy()
            print(f"[Update] new global best = {best_val:.6f}")

    if best_full is None:
        # fallback: uniform full vector
        best_full = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3. Optional L‑BFGS polishing on the full vector (if time permits)
    # ------------------------------------------------------------------
    if _opt is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # give L‑BFGS at most 15 % of the remaining time
        max_seconds = max(0.0, remaining * 0.15)

        print("\n--- L‑BFGS polishing (full vector) ---")
        v_full = np.log(best_full + 1e-30)

        def smooth_obj(v_local):
            a = _softmax(v_local)      # stays on simplex
            conv = _conv_fft(a, a)
            M = np.max(conv)
            shifted = conv - M
            # use a relatively large β for sharp approximation
            beta = 5e5
            exp_beta = np.exp(beta * shifted)
            Z = exp_beta.sum()
            obj = 2.0 * n * (M + np.log(Z) / beta)

            w = exp_beta / Z
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=smooth_obj,
                                x0=v_full,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxfun': int(1e5),
                                         'ftol': 1e-12,
                                         'disp': False})
            polished = _softmax(res.x)
            polished_val = _objective_surrogate(polished)
            print(f"[L‑BFGS] polished surrogate = {polished_val:.6f}")
            if polished_val < best_val:
                best_val = polished_val
                best_full = polished.copy()
                print(f"[Update] new global best after L‑BFGS = {best_val:.6f}")
        except KeyboardInterrupt:
            print("[L‑BFGS] time limit reached, exiting polish")
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] exception:", e)

    # ------------------------------------------------------------------
    # 4. Final Simulated‑Annealing refinement (full logit space)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa_iters = int(min(600_000, remaining * 2000))
        max_sa_iters = max(200_000, max_sa_iters)
        print("\n--- Simulated‑Annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_opt(best_full, n, deadline,
                                                  max_iters=max_sa_iters)
        if val_sa < best_val:
            best_val = val_sa
            best_full = seq_sa.copy()
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_full.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_full.tolist()
    return best_full.tolist()
