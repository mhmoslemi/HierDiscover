"""
Propose a step‑function (non‑negative heights) that minimises

    evaluate_sequence(seq)  =  2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm is a multi‑stage heuristic:

1. Generate a large and diverse set of *seeds* (analytic families,
   Dirichlet samples, low‑frequency curves, random spikes, block mixtures).
   All seeds are normalised to sum = 1 and kept in a small heap of the
   best candidates according to the exact surrogate 2·n·max(conv).

2. For the top few seeds run a continuation L‑BFGS‑B optimiser that
   minimises a smoothed version of the max‑convolution (log‑sum‑exp).
   The optimiser works in log‑softmax space, guaranteeing feasibility.

3. Refine each L‑BFGS result with a simple *projected gradient descent*
   on the true (non‑smooth) max‑convolution.  The gradient of the max is
   easy to compute: for the index `k*` where the convolution attains its
   maximum, ∇ max(conv) = 2·rev(a)[k*] .  After each gradient step we project
   back onto the probability simplex.

4. Combine good candidates by random convex mixtures (α·x+(1‑α)·y) and keep
   any improvement.

5. A final *directional hill‑climb* moves a small fraction of mass from the
   coordinates that contribute most to the current peak towards those that
   contribute least, always accepting only improvements.

6. The best sequence found is evaluated with the official `evaluate_sequence`,
   stored globally as ``height_sequence_1`` and returned.

The whole search respects a 1000 s wall‑clock budget.
"""

import time
import heapq
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (zero‑padded)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stably convert logits to a probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Exact surrogate (used while searching – assumes ∑a = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """Exact objective for a probability vector (∑vec = 1):
       2·n·max(conv(vec, vec)).
    """
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _double_power(n: int, p: float) -> np.ndarray:
    """Symmetric sum of two power laws – high at both ends, low in the centre."""
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / ((idx + 1) ** p)) + (1.0 / ((n - idx) ** p))
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return _normalize(a)
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return _normalize(a)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return _normalize(a)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return _normalize(a)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    return _normalize(a)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0
    return _normalize(a)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _arcsine_window(n: int) -> np.ndarray:
    """U‑shaped arcsine‑like density  (≈1/√(x·(1‑x)))"""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return _normalize(a)


def _cosine_edge_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """|cos|‑shaped – large at both ends, small in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** gamma
    return _normalize(a)


def _two_block_uniform(n: int, block_ratio: float = 0.2,
                       left_weight: float = 0.5) -> np.ndarray:
    """
    Two uniform blocks placed at the left and right ends.
    block_ratio – fraction of the total length occupied by each block.
    left_weight – total mass placed on the left block (right gets 1‑left).
    """
    block_len = max(1, int(block_ratio * n))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = left_weight / block_len
    a[-block_len:] = (1.0 - left_weight) / block_len
    return _normalize(a)


# ----------------------------------------------------------------------
# Random generators (return normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with all concentration parameters = conc."""
    return _normalize(rng.dirichlet(np.full(n, conc)))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _normalize(a)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, cut to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _random_mixed_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose a few disjoint blocks and put uniform mass on each."""
    nb = rng.integers(2, 6)                 # number of blocks
    a = np.zeros(n, dtype=np.float64)
    remaining = 1.0
    for b in range(nb):
        block_len = rng.integers(1, max(2, n // nb))
        start = rng.integers(0, n - block_len + 1)
        mass = remaining * rng.random()
        a[start:start + block_len] += mass / block_len
        remaining -= mass
    return _normalize(a)


def _random_spikes(n: int, num_spikes: int, rng: np.random.Generator) -> np.ndarray:
    """Place a few spikes (large masses) at random positions, the rest is tiny."""
    a = np.full(n, 1e-12, dtype=np.float64)   # tiny background to keep positivity
    positions = rng.choice(n, size=num_spikes, replace=False)
    # distribute the bulk mass (≈1) among the spikes via a Dirichlet draw
    spike_weights = rng.dirichlet(np.ones(num_spikes))
    a[positions] = spike_weights
    return _normalize(a)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """Optimise a smoothed max‑convolution objective using L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 10000.0,
                         20000.0, 50000.0, 100000.0, 200000.0,
                         500000.0, 1_000_000.0, 5_000_000.0]

    v = np.log(start_vec + 1e-30)   # logits (log‑softmax variables)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth approximation of max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a : 2 * conv(rev(a), w)
            rev_a = a[::-1]
            w = exp_beta / sum_exp                # soft‑max weights over conv entries
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected gradient descent on the true max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _projected_gradient_descent(start_vec: np.ndarray,
                                n: int,
                                deadline: float,
                                rng: np.random.Generator) -> np.ndarray:
    """Simple sub‑gradient descent on max(conv) with simplex projection."""
    a = start_vec.copy()
    cur_val = _objective_surrogate(a)
    step = 0.1
    min_step = 1e-7

    while time.time() < deadline and step > min_step:
        conv = _conv_fft(a, a)
        k_max = int(np.argmax(conv))

        # compute sub‑gradient of max(conv) at k_max
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1, dtype=np.int64)
        opp = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[opp]          # 2*a[opp] (handles the i=j case automatically)

        # take a step and project back onto simplex
        a_new = a - step * grad
        a_new = _normalize(a_new)

        new_val = _objective_surrogate(a_new)

        if new_val < cur_val:
            a = a_new
            cur_val = new_val
            step = min(step * 1.2, 1.0)   # gently accelerate (but keep ≤1)
        else:
            step *= 0.5

    return a


# ----------------------------------------------------------------------
# Random convex mixing of two vectors
# ----------------------------------------------------------------------
def _mix_vectors(v1: np.ndarray, v2: np.ndarray, alpha: float) -> np.ndarray:
    """Return α·v1 + (1‑α)·v2 normalised to sum = 1."""
    mixed = alpha * v1 + (1.0 - alpha) * v2
    return _normalize(mixed)


# ----------------------------------------------------------------------
# Directional hill‑climb moving mass from high‑impact to low‑impact indices
# ----------------------------------------------------------------------
def _directional_hill_climb(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves guided by the gradient of the current peak."""
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        conv = _conv_fft(cur, cur)
        k_max = int(np.argmax(conv))

        # gradient weights for the current maximum
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1, dtype=np.int64)
        opp = k_max - idx
        weights = cur[opp]   # proportional contribution of each idx to the peak

        # source: index with largest weight, target: smallest weight (but different)
        src = idx[np.argmax(weights)]
        tgt = idx[np.argmin(weights)]

        if src == tgt:
            # fallback: random pair
            src, tgt = rng.integers(n, size=2)
            if src == tgt:
                continue

        # move a modest fraction of mass from src to tgt
        eps = cur[src] * rng.uniform(0.001, 0.01)
        if eps <= 0.0:
            continue

        new = cur.copy()
        new[src] -= eps
        new[tgt] += eps
        new = _normalize(new)        # re‑normalise to keep sum = 1
        new_val = _objective_surrogate(new)

        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns a plain ``list`` of heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a compromise between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Heap of the top‑k seeds (kept by surrogate value)
    # ------------------------------------------------------------------
    top_k = 30
    best_heap = []                 # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert a candidate into the heap while keeping only the best ``top_k``."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # 1. Deterministic analytic seeds (wide variety)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    _push_candidate(_arcsine_window(n))
    _push_candidate(_hamming_window(n))
    _push_candidate(_blackman_window(n))
    _push_candidate(_nuttall_window(n))

    for gamma in np.linspace(0.2, 3.0, 10):
        _push_candidate(_sine_window(n, gamma))
    for beta in np.linspace(1.0, 15.0, 8):
        _push_candidate(_kaiser_window(n, beta))
    for sigma in np.linspace(0.08, 0.30, 12):
        _push_candidate(_gaussian(n, sigma))
    for p in np.linspace(0.05, 2.5, 12):
        _push_candidate(_power(n, p))
        _push_candidate(_double_power(n, p))
    for gamma in np.linspace(0.5, 4.0, 8):
        _push_candidate(_cosine_edge_window(n, gamma))
    for alpha in np.linspace(0.0, 1.0, 6):
        _push_candidate(_tukey_window(n, alpha))

    # beta‑shape (U‑shaped) – explore several log‑uniform parameters
    for _ in range(30):
        a = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
        b = 10 ** rng.uniform(-2, 0)
        _push_candidate(_beta_shape(n, a, b))

    # two‑block uniform families
    for _ in range(20):
        blk = rng.uniform(0.05, 0.35)
        lw = rng.random()
        _push_candidate(_two_block_uniform(n, block_ratio=blk, left_weight=lw))

    # ------------------------------------------------------------------
    # 2. Massive random seeding
    # ------------------------------------------------------------------
    conc_choices = [0.001, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    random_seed_budget = 8000
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # occasional cyclic shift to change correlation pattern
        if rng.random() < 0.25:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # Exponential draws (heavier tails)
    for _ in range(800):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        if rng.random() < 0.25:
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
        _push_candidate(vec)

    # Low‑frequency random curves
    for nf in (2, 3, 5, 8, 12):
        for _ in range(250):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            if rng.random() < 0.25:
                shift = rng.integers(n)
                if shift:
                    vec = np.roll(vec, shift)
            _push_candidate(vec)

    # Mixed block structures
    for _ in range(300):
        if time.time() > deadline:
            break
        _push_candidate(_random_mixed_blocks(n, rng))

    # Random spikes – a few isolated large masses
    for spikes in (2, 3, 5, 8):
        for _ in range(200):
            if time.time() > deadline:
                break
            vec = _random_spikes(n, spikes, rng)
            _push_candidate(vec)

    # ------------------------------------------------------------------
    # 3. Re‑use previous solution if it matches current size
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                _push_candidate(_normalize(prev_arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4. Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seeding] kept {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 5. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Local refinements – L‑BFGS on the top few seeds
    # ------------------------------------------------------------------
    max_refinements = min(8, len(seeds))
    for idx in range(max_refinements):
        if time.time() > deadline - 10.0:
            break
        seed_vec = seeds[idx]
        remaining = deadline - time.time()
        # reserve a bit for later stages
        refine_deadline = time.time() + max(5.0, remaining * 0.55 / max_refinements)

        try:
            refined = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             refine_deadline,
                                             beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                            200.0, 500.0,
                                                            1000.0, 2000.0,
                                                            5000.0, 10000.0,
                                                            20000.0, 50000.0,
                                                            100000.0, 200000.0,
                                                            500000.0,
                                                            1_000_000.0],
                                             max_iter_per_beta=4000)
        except RuntimeError:
            refined = seed_vec

        # a quick projected‑gradient fine‑tune
        refined = _projected_gradient_descent(refined, n,
                                              deadline - 0.5,
                                              rng)

        refined_val = _objective_surrogate(refined)
        if refined_val < global_best_val:
            global_best_seq = refined
            global_best_val = refined_val
            print(f"[Refine {idx+1}] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Random convex mixes of the best few candidates
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mix_deadline = deadline - 2.0
        mix_start = time.time()
        mix_trials = 1500
        while time.time() < mix_deadline and mix_trials > 0:
            i, j = rng.choice(len(seeds), size=2, replace=False)
            alpha = rng.random()
            cand = _mix_vectors(seeds[i], seeds[j], alpha)
            cand_val = _objective_surrogate(cand)
            if cand_val < global_best_val:
                global_best_seq = cand
                global_best_val = cand_val
                print(f"[Mix] improved to {global_best_val:.6f}")
            mix_trials -= 1

    # ------------------------------------------------------------------
    # 8. Final directional hill‑climb (gradient‑guided mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final directional hill‑climb ---")
        seq_hc, val_hc = _directional_hill_climb(global_best_seq,
                                                 n,
                                                 deadline,
                                                 rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Evaluate with the official evaluator (exact objective)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
