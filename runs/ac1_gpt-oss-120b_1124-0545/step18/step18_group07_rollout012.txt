"""
Propose a step‑function (non‑negative vector) that minimizes the evaluation
function

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm is a three‑stage pipeline:

1. **Seed generation** – a large, diverse collection of ℓ¹‑normalised vectors
   (analytic windows, Dirichlet draws, edge‑biased U‑shapes, previous solution).

2. **Coarse optimisation** – a short L‑BFGS optimisation on a smooth
   approximation of the max‑convolution, followed by an exact exponentiated‑
   gradient (mirror‑descent) pass.  If the previous best solution is present we
   start from it directly.

3. **Fine‑grained local search** – a massive “mass‑transfer” hill‑climber that
   updates the convolution incrementally, moving a tiny amount of probability
   mass from one coordinate to another only if the exact max‑convolution
   decreases.  The update is O(n) and uses a copy of the convolution vector,
   allowing millions of cheap moves within the 1000 s budget.

The final sequence is stored in the global ``height_sequence_1`` for reuse in
subsequent runs.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate (ℓ¹‑normalised vectors): 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:

    s = a.sum()

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _edge_bias(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution concentrating mass near the ends (p ≥ 0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / ((i) * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse pool of ℓ¹‑normalised vectors.
    """
    seeds = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for g in (0.3, 0.5, 1.0, 2.0, 4.0, 8.0):
        seeds.append(_sine_window(n, gamma=g))

    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    for beta in (2, 4, 8, 12, 16, 20, 30):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))

    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))

    # beta‑shape grid – extend into the very‑U region
    for a in (0.05, 0.1, 0.3, 1.0, 2.0, 4.0):
        for b in (0.05, 0.1, 0.3, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Edge‑biased families – widen the range of p
    for p in (0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 2.5):
        seeds.append(_edge_bias(n, p))

    # Random Dirichlet draws ------------------------------------------------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random vectors (very few spikes)
    for k in (max(1, int(0.005 * n)), max(1, int(0.01 * n))):
        for _ in range(2):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors
    for bf in (0.05, 0.15, 0.30):
        for _ in range(2):
            seeds.append(_block_uniform(n, bf, rng))

    # Alternating Dirichlet (helps later permutation search)
    seeds.append(_alternating_dirichlet(n, rng))

    # Permuted heavy‑tailed seed (helps later permutation search)
    heavy = _random_dirichlet(n, 0.05, rng)
    for _ in range(3):
        perm = heavy[rng.permutation(n)]
        seeds.append(perm)

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity‑check (very rarely needed)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ¹‑normalised probability vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 1e7
        ]

    # work in log‑probability space (stay on simplex via softmax)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ¹‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # classic 1/√t schedule (slightly aggressive constant)
        eta = 0.7 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mass‑transfer local search (fast incremental updates)
# ----------------------------------------------------------------------
def _mass_transfer_local_search(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               max_moves: int = 2_000_000) -> np.ndarray:
    """
    Hill‑climbing that moves a tiny amount of mass from one index to another.
    The full linear convolution is kept incrementally, so each trial costs O(n).
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    # initial full convolution via FFT (fast)
    conv = _conv_fft(a, a)
    cur_max = float(conv.max())
    cur_obj = 2.0 * n * cur_max
    best_obj = cur_obj
    best_a = a.copy()

    L = 2 * n - 1
    move = 0
    no_improve = 0

    while time.time() < deadline and move < max_moves:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # amount of mass to move: up to 30 % of source entry
        delta = rng.random() * a[i] * 0.30
        if delta <= 0.0:
            continue

        # build candidate convolution (copy + slice updates)
        new_conv = conv.copy()
        # add 2·δ·a to the slice j:j+n and subtract from i:i+n
        slice_a = a * (2.0 * delta)          # vector of length n
        new_conv[j:j + n] += slice_a
        new_conv[i:i + n] -= slice_a

        # delta² corrections (scalar updates)
        d2 = delta * delta
        new_conv[2 * j] += d2
        new_conv[2 * i] += d2
        new_conv[i + j] -= 2.0 * d2

        new_max = float(new_conv.max())
        new_obj = 2.0 * n * new_max

        if new_obj < cur_obj - 1e-12:   # accept only strict improvements
            # apply the move
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        move += 1
        if move % 200_000 == 0:
            print(f"[MassTransfer] moves={move:,} best_obj={best_obj:.6f}")

        # occasional random restart if stuck for a long time
        if no_improve >= 500_000:
            # small perturbation to escape possible flat region
            idx1 = rng.integers(0, n)
            idx2 = rng.integers(0, n)
            if idx1 != idx2 and a[idx1] > 1e-12:
                eps = min(a[idx1] * 0.1, 1e-4)
                a[idx1] -= eps
                a[idx2] += eps
                conv = _conv_fft(a, a)       # recompute (rare)
                cur_max = float(conv.max())
                cur_obj = 2.0 * n * cur_max
                no_improve = 0

    # Return the best version found
    print(f"[MassTransfer] finished with {move:,} moves, best surrogate={best_obj:.6f}")
    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # Starting point: use previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    start_seq = None
    if isinstance(previous, list) and len(previous) == n:
        arr = np.asarray(previous, dtype=np.float64)
        s = arr.sum()
        if s > 0.0:
            start_seq = arr / s
            print("[Init] Using previous best sequence as start.")
    if start_seq is None:
        print("[Init] No previous solution – generating seeds.")
        seeds = _collect_seeds_extended(n, rng, None)
        # evaluate cheap surrogate on all seeds and keep the best few
        evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
        evaluated.sort(key=lambda x: x[1])
        top_k = 6
        top_seeds = [vec for vec, _ in evaluated[:top_k]]
        print(f"Collected {len(seeds)} seeds, best surrogate among seeds = {evaluated[0][1]:.6f}")

        # simple coarse optimisation: pick the best seed, run L‑BFGS + EG
        best_seed = top_seeds[0]
        if _opt is not None:
            try:
                seq = _lbfgs_smooth_optimize(best_seed,
                                             n,
                                             deadline,
                                             beta_schedule=None,
                                             max_iter_per_beta=12000)
                print("[L‑BFGS] finished.")
            except Exception as e:  # pragma: no cover
                print("[L‑BFGS] fell back to seed directly:", e)
                seq = best_seed
        else:
            seq = best_seed

        # brief exponentiated‑gradient polishing
        seq, _ = _exp_gradient_opt(seq, n, deadline, max_iters=200_000)
        start_seq = seq
        print("[ExpGrad‑init] coarse polishing done.")

    # ------------------------------------------------------------------
    # Stage 1: mass‑transfer local search (largest allocation of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # we estimate that roughly 200 000 moves cost ~1 s, so allocate the bulk
        max_moves = int(min(5_000_000, remaining * 3000))
        max_moves = max(500_000, max_moves)
        print(f"\n--- Mass‑Transfer Local Search (max_moves={max_moves:,}) ---")
        seq_mt = _mass_transfer_local_search(start_seq, n, deadline,
                                             max_moves=max_moves)
        seq = seq_mt
        cur_obj = _objective_surrogate(seq)
        print(f"[After MassTransfer] surrogate = {cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Stage 2: final short exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iter = int(min(300_000, remaining * 600))
        max_iter = max(100_000, max_iter)
        print(f"\n[ExpGrad‑final] polishing with {max_iter:,} iterations")
        seq, final_val = _exp_gradient_opt(seq, n, deadline, max_iters=max_iter)
        print(f"[ExpGrad‑final] final surrogate = {final_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq.tolist())
        print(f"\n=== Finished ===\nOfficial evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = seq.tolist()
    return seq.tolist()
