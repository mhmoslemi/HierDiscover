"""
propose_candidate
-----------------

A hybrid optimisation that combines:
  * a large diverse seed set (analytic windows, beta‑shapes, random Dirichlet/exp,
    low‑frequency cosine curves, and the previous best solution);
  * a lightweight Differential‑Evolution (DE) loop that works on the probability
    simplex (projection after every mutation/crossover);
  * a mirror‑descent (exponentiated‑gradient) polishing that uses the exact
    sub‑gradient of the max‑autoconvolution;
  * an optional L‑BFGS fine‑tuning on a smooth log‑sum‑exp surrogate
    (only if SciPy is available).

The objective for ℓ₁‑normalised vectors a∈ℝⁿ₊, Σa=1, is
    F(a) = 2·n·max( a * a ),
where “*” denotes the full linear convolution.
The algorithm stays entirely inside the simplex, thus guarantees a
valid sequence for ``evaluate_sequence``.  Time is split roughly as
   55 % DE  – global exploration,
   35 % EG – local descent,
    8 % L‑BFGS – final smoothing.
The best sequence found is stored in the global variable
``height_sequence_1`` for the next invocation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (real input -> real output)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns probabilities that sum to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate / exact objective (for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact objective for a simplex‑vector: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                    # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b, re‑normalised to stay on the simplex."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution (sub‑gradient on active entry)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (see reasoning in the prompt)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500):
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)    # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # pick the middle n entries

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)            # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Differential‑Evolution (simple, simplex‑constrained)
# ----------------------------------------------------------------------
def _de_optimize(initial_pop: list[np.ndarray],
                 n: int,
                 deadline: float,
                 F: float = 0.7,
                 CR: float = 0.9,
                 max_gens: int = 10_000):
    """
    Simple DE on the probability simplex.
    Returns the best individual and its surrogate value.
    """
    pop = [vec.copy() for vec in initial_pop]
    fitness = [_objective_surrogate(v) for v in pop]
    best_idx = int(np.argmin(fitness))
    best_val = fitness[best_idx]
    best_vec = pop[best_idx].copy()

    popsize = len(pop)
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # select three distinct individuals different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            x1, x2, x3 = pop[r1], pop[r2], pop[r3]

            # mutation
            mutant = x1 + F * (x2 - x3)
            mutant = np.maximum(mutant, 0.0)          # keep non‑negative
            mutant = _proj_simplex(mutant)            # project onto simplex

            # crossover (binomial)
            cross_mask = np.random.rand(n) < CR
            if not np.any(cross_mask):
                cross_mask[np.random.randint(0, n)] = True
            trial = pop[i].copy()
            trial[cross_mask] = mutant[cross_mask]
            trial = _proj_simplex(trial)

            val = _objective_surrogate(trial)
            if val < fitness[i]:
                pop[i] = trial
                fitness[i] = val
                if val < best_val:
                    best_val = val
                    best_vec = trial.copy()
                    best_idx = i
                    # occasional progress report
                    if gen % 50 == 0:
                        print(f"[DE] gen {gen}, new best surrogate {best_val:.6f}")

        # safety check to avoid endless loops when time is low
        if time.time() > deadline - 0.5:
            break

    return best_vec, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found and stores it in the global variable
    ``height_sequence_1`` for later runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size (fixed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (rich family)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic families (coarse grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 10):
        seeds.append(_sine_window(n, gamma=gamma))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 10):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 12):
        seeds.append(_beta_shape(n, sigma, sigma))   # symmetric beta
    # Small‑α/β beta shapes (including arcsine‐type)
    for a_ in (0.05, 0.1, 0.2, 0.5):
        for b_ in (0.05, 0.1, 0.2, 0.5):
            seeds.append(_beta_shape(n, a_, b_))
    # Power law tails
    for p in np.linspace(0.0, 2.5, 13):
        seeds.append(_power(n, p))
    # Double Gaussian blobs
    for sigma in (0.10, 0.12):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))
    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))
    # Random Dirichlet draws (wide concentration range)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))
    # Alternating Dirichlet (highly non‑smooth)
    def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
        raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
        sorted_desc = np.sort(raw)[::-1]
        a = np.empty_like(sorted_desc)
        left, right = 0, n - 1
        for idx, val in enumerate(sorted_desc):
            if idx % 2 == 0:
                a[left] = val
                left += 1
            else:
                a[right] = val
                right -= 1
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    seeds.append(_alternating_dirichlet(n, rng))

    # Add perturbed copies of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(5):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                seeds.append(noisy / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Differential‑Evolution on the simplex
    # ------------------------------------------------------------------
    print("\n--- Differential Evolution (global) ---")
    # Build initial population (mix of top seeds + fresh random Dirichlet)
    pop = []
    pop.extend(top_seeds)
    while len(pop) < 30:
        pop.append(_random_dirichlet(n, conc=0.2, rng=rng))
    # Ensure all vectors are on the simplex
    pop = [_proj_simplex(v) for v in pop]

    de_deadline = start_time + budget_s * 0.55
    best_vec, best_val = _de_optimize(pop, n, de_deadline,
                                      F=0.7, CR=0.9, max_gens=5000)
    print(f"[DE] best surrogate after global search = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Local refinement – exponentiated gradient
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient refinement (local) ---")
        eg_deadline = deadline - 0.5   # keep a tiny margin for final steps
        seq_eg, val_eg = _exp_gradient_opt(best_vec.copy(),
                                          n,
                                          eg_deadline,
                                          max_iters=800_000,
                                          eta0=6.0)
        if val_eg < best_val:
            best_vec, best_val = seq_eg, val_eg
            print(f"[EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final smoothing using L‑BFGS (if SciPy is present)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS smoothing (final) ---")
        lbfgs_deadline = deadline - 0.2
        seq_lbfgs = _lbfgs_smooth_optimize(best_vec,
                                           n,
                                           lbfgs_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1000.0,
                                                          2000.0, 5000.0,
                                                          20000.0, 50000.0,
                                                          100000.0, 200000.0,
                                                          500000.0, 1000000.0],
                                           max_iter_per_beta=2000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_vec, best_val = seq_lbfgs, val_lbfgs
            print(f"[L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation (exact)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
