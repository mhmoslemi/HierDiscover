def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Choose length n
    n = 4000  # maybe adapt if time remaining.
    # Get initial sequence
    previous = globals().get("height_sequence_1")
    if previous is not None and len(previous) == n:
        a = np.asarray(previous, dtype=np.float64)
        a = a / a.sum()
    else:
        # start with uniform or mixture of shapes
        a = np.full(n, 1.0/n, dtype=np.float64)

    best_a = a.copy()
    best_val = 2*n*np.max(_conv_fft(a,a))

    # Parameters
    eta0 = 0.5
    tau0 = 0.5
    tau_max = 30.0
    # number of iterations to reach tau_max
    # We'll gradually increase tau over e.g., 200k iterations
    tau_inc_iters = 200000

    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Iter {it}] new best surrogate {best_val:.6f}")

        # schedule tau
        tau = tau0 + (tau_max - tau0) * min(1.0, it / tau_inc_iters)

        # compute softmax weights
        # shift by max to improve stability
        shift = conv.max()
        w = np.exp(tau * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w / w_sum

        # compute gradient via convolution with reversed w
        w_rev = w[::-1]
        grad_full = _conv_fft(a, w_rev)  # length n + len(w) - 1 = n + 2n-2 = 3n-2
        offset = len(w) - 1  # = 2n-2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # learning rate schedule
        eta = eta0 / math.sqrt(it)

        # additive update and projection
        a_tmp = a - eta * grad
        a_tmp = np.maximum(a_tmp, 0.0)
        sum_tmp = a_tmp.sum()
        if sum_tmp <= 0:
            a = _uniform(n)  # fallback
        else:
            a = a_tmp / sum_tmp

        # optional: occasional random perturbation to escape plateau
        if it % 10000 == 0:
            # small random jitter
            jitter = rng.normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            sum_j = a.sum()
            if sum_j > 0:
                a = a / sum_j
            else:
                a = _uniform(n)

    # final evaluation
    try:
        true_val = evaluate_sequence(best_a.tolist())
        print(f"\n=== Finished ===\nSurrogate {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")

        true_val = None

    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()

def _smooth_grad_descent(start, n, deadline,
                         eta0=0.5, tau0=0.5, tau_max=30.0,
                         max_iters=None):
    a = start.copy()
    best_a = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * conv.max()
    it = 0
    tau_inc_iters = 200000  # number of iters to ramp tau
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SmoothGD] it={it} new best {best_val:.6f}")

        # schedule tau
        tau = tau0 + (tau_max - tau0) * min(1.0, it / tau_inc_iters)

        # compute weights (softmax)
        shift = max_conv  # subtract max for stability
        exp_arg = tau * (conv - shift)
        # Clip large exponent for stability
        exp_arg = np.clip(exp_arg, -700, 700)  # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # gradient
        w_rev = w[::-1]
        grad_full = _conv_fft(a, w_rev)
        offset = len(w) - 1
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # learning rate schedule
        eta = eta0 / math.sqrt(it)

        # additive update + projection
        a_tmp = a - eta * grad
        a_tmp = np.maximum(a_tmp, 0.0)
        sum_tmp = a_tmp.sum()
        if sum_tmp <= 0.0:
            a = _uniform(n)
        else:
            a = a_tmp / sum_tmp

    return best_a, best_val
```

"""
Smooth‐gradient descent + exponential‑gradient + simulated‑annealing
search for a non‑negative step function that minimises the
``evaluate_sequence`` objective.

Stage A – a differentiable surrogate of the max–convolution is built
using a soft‑max (log‑sum‑exp).  Projected additive updates on the
probability simplex are performed with a decreasing stepsize.
Stage B – a short exponential‑gradient (mirror‑descent) polishing on
the original non‑smooth objective.
Stage C – a final simulated‑annealing tweak to escape possible
local minima.

All stages respect the overall time budget (default 1000 s).  The best
sequence found is stored globally as ``height_sequence_1`` and returned.
"""

from __future__ import annotations
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n·log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _uniform(n: int) -> np.ndarray:
    """Uniform ℓ₁‑normalised vector of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) (Σseq is assumed = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_grad_descent(start: np.ndarray, n: int,
                         deadline: float,
                         eta0: float = 0.5,
                         tau0: float = 0.5,
                         tau_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Projected additive gradient descent on the smooth surrogate
    f_τ(a) = (1/τ)·log Σ_k exp(τ·(a*a)[k]).

    Returns the best vector seen and its surrogate value.
    """
    a = start.copy()
    best_a = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)
    it = 0
    # number of iterations over which τ is linearly increased
    tau_inc_iters = 200_000

    while time.time() < deadline:
        it += 1

        # current convolution and surrogate value
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SmoothGD] it={it:,} new best surrogate {best_val:.6f}")

        # -------------------- temperature schedule --------------------
        tau = tau0 + (tau_max - tau0) * min(1.0, it / tau_inc_iters)

        # -------------------- soft‑max weights ------------------------
        # shift by the maximum for numerical stability
        shift = max_conv
        exp_arg = tau * (conv - shift)          # ≤ 0
        # avoid overflow/underflow in np.exp
        exp_arg = np.clip(exp_arg, -700.0, 0.0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # -------------------- gradient of the smooth surrogate --------
        # g_i = 2· Σ_j a_j·w_{i+j}
        w_rev = w[::-1]
        grad_full = _conv_fft(a, w_rev)               # length = 3·n‑2
        offset = len(w) - 1                           # = 2·n‑2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # -------------------- stepsize -------------------------------
        eta = eta0 / math.sqrt(it)

        # -------------------- additive update + renormalisation -----
        a_tmp = a - eta * grad
        a_tmp = np.maximum(a_tmp, 0.0)
        s = a_tmp.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a_tmp / s

        # occasional tiny random jitter to jump out of plateaus
        if it % 5_000 == 0:
            jitter = np.random.default_rng().normal(scale=1e-5, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponential‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative weights)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift (as in the reference code)
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size schedule
        eta = eta0 / math.sqrt(it)

        # multiplicative update; clip exponent for safety
        delta = -eta * g
        delta = np.clip(delta, -50.0, 50.0)
        x = x * np.exp(delta)

        # renormalise onto the simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50_000 == 0:
                print(f"[ExpGrad] it={it:,} best surrogate {best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # geometric cooling
        T = max(T * 0.99995, min_temp)

        # two move types
        if rng.random() < 0.6:
            # reshuffle a random subset (size 3‑12)
            k = rng.integers(3, 13)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # small transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:,} best surrogate {best_val:.6f}")

    return best, best_val
