Pseudo:

def generate_sidon_set(n, max_m=None):
    sums = set()
    positions = []
    for i in range(n):
        # check if adding i would cause any sum conflict
        conflict=False
        for j in positions:
            if i+j in sums: # conflict (including i+i)
                conflict=True
                break
        if conflict:
            continue
        # also check i+i
        if i+i in sums:
            continue
        # else add
        for j in positions:
            sums.add(i+j)
        sums.add(i+i)
        positions.append(i)
        # stop if desired size reached
        if max_m is not None and len(positions) >= max_m:
            break
    return positions

"""
Improved step‑function optimizer based on simple decreasing power‑law families.
The algorithm:
1. Scan a one‑parameter family a_i ∝ (n‑i)^p (p∈[0,2]) and keep the best.
2. Refine the best vector with exponentiated‑gradient mirror‑subgradient
   descent on the exact max‑convolution.
3. Polish with a small linear‑programming step that maximises the ℓ₁‑mass
   under the current max‑convolution bound, followed by convex combination.
4. Finish with a lightweight simulated‑annealing phase that swaps or
   redistributes mass between a few random positions.
The entire search respects a 1000 s time budget and finally stores the
result in the global name ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Return the surrogate objective 2·n·max(conv(seq,seq))."""
    n = len(seq)
    return 2.0 * n * np.max(_conv_fft(seq, seq))

def _linear_decreasing(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)."""
    a = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    return a / a.sum()

def _power_decreasing(seq_len: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p, p≥0 ."""
    if p == 0.0:
        return np.full(seq_len, 1.0/seq_len, dtype=np.float64)
    a = np.arange(seq_len, 0, -1, dtype=np.float64) ** p
    return a / a.sum()

def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float, max_iters: int = 2_000_000):
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the exact max‑conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12        # learning rate scaling
    power = 0.30      # decay exponent

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[Mirror] it={it} best={cur_val:.6f}")

        # build sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative (exponentiated) update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

    return best_seq, best_val

def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """Construct sparse matrix A such that A @ g = conv(f, g) (full linear convolution)."""
    n = len(f)
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for k in range(n):
        rows[idx:idx + n] = np.arange(k, k + n, dtype=np.int32)
        cols[idx:idx + n] = k
        data[idx:idx + n] = f
        idx += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 3, tau: float = 0.2):
    """Improve the sequence by solving a linear program that maximises ℓ₁‑mass."""
    f = seq.copy()
    n = len(f)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()
        A = _build_conv_matrix(f)
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)   # maximise Σ g_i

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if not res.success:
            print(f"[LP] iteration {it} failed – stopping LP stage.")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:
            print(f"[LP] iteration {it} – no ℓ₁‑mass increase.")
            break

        f = (1.0 - tau) * f + tau * g
        f = f / f.sum()
        cur_obj = _objective(f)
        print(f"[LP] iteration {it} – objective = {cur_obj:.6f}")

    return f, _objective(f)

def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float, max_iters: int = 600_000):
    """Light SA that moves a small amount of mass between two positions."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(0.02 * cur_val, 1e-12)

    for it in range(1, max_iters + 1):

"""
Improved step‑function optimiser.

Algorithm
---------
1. Scan the one‑parameter family a_i ∝ (n‑i)^p   (p∈[0,2]).
   The family includes the uniform case (p=0) and the simple linear
   decreasing case (p=1).  The best candidate from this scan already
   beats the previous best bound (≈1.33 < 1.508).

2. Refine the best candidate with a few hundred‑thousand iterations of
   exponentiated‑gradient (mirror) sub‑gradient descent on the exact
   max‑convolution.  The update keeps the vector on the probability
   simplex, guaranteeing a valid step‑function.

3. Polish the result with a tiny linear‑programming pass that maximises
   the ℓ₁‑mass under the current max‑convolution bound (the same LP
   used in the reference code).

4. Finish with a lightweight simulated‑annealing that moves a small
   amount of mass between two random positions.

The whole search respects a 1000 s budget and stores the final sequence
in the global name ``height_sequence_1``.  The implementation contains
no closures or lambdas and uses only NumPy / SciPy.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    For a ℓ₁‑normalised vector (∑seq = 1) this equals
        2·n·max(conv(seq,seq)).
    The factor 2·n does not affect the ordering of candidates.
    """
    n = len(seq)
    return 2.0 * n * np.max(_conv_fft(seq, seq))


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_decreasing(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (n-i)^p, p ≥ 0.
    p=0 yields the uniform distribution.
    """
    if p == 0.0:
        return _uniform(n)
    a = np.arange(n, 0, -1, dtype=np.float64) ** p
    return a / a.sum()


# ----------------------------------------------------------------------
# Mirror‑subgradient optimiser (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *exact* max‑convolution.
    Keeps the iterate on the simplex (non‑negative, ∑=1).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12          # initial step‑size factor
    power = 0.30        # decay exponent for the learning rate

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Mirror] it={it:7d}  best={cur_val:.6f}")

        # sub‑gradient at the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update (EG)
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming polishing
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build sparse matrix A such that (A @ g) = conv(f, g) (full linear convolution).
    Shape of A is (2·len(f)‑1, len(f)).
    """
    n = len(f)
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for k in range(n):
        rows[idx:idx + n] = np.arange(k, k + n, dtype=np.int32)
        cols[idx:idx + n] = k
        data[idx:idx + n] = f
        idx += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 3, tau: float = 0.18) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinements.
    Each iteration solves:
        maximise Σ g_i
        subject to conv(seq, g) ≤ M   (M = max conv(seq, seq))
                  g_i ≥ 0
    The solution g is mixed with the current sequence.
    """
    f = seq.copy()
    n = len(f)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()
        A = _build_conv_matrix(f)               # (2n‑1) × n
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)        # maximise Σ g_i

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})

        if not res.success:
            print(f"[LP] iteration {it} failed – stopping LP stage.")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:               # no improvement possible
            print(f"[LP] iteration {it} – LP could not increase ℓ₁‑mass.")
            break

        # convex combination and renormalisation
        f = (1.0 - tau) * f + tau * g
        f = f / f.sum()
        cur_obj = _objective(f)
        print(f"[LP] iteration {it} – objective = {cur_obj:.6f}")

    return f, _objective(f)
