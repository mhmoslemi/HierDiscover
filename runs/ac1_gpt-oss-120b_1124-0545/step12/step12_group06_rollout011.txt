def _lp_refine(start_seq, n, deadline):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    iter = 0
    # Keep track of previous M to detect convergence
    while time.time() < deadline and iter < 20:
        iter += 1
        conv_a = _conv_fft(a, a)
        M = np.max(conv_a)
        # Determine active constraints
        eps = 0.01  # 1% tolerance
        active_idx = np.where(conv_a >= M * (1 - eps))[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(conv_a)], dtype=int)
        # Build sparse matrix
        rows = []
        cols = []
        data = []
        for r, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # j range
            for j in range(i_start, i_end + 1):
                rows.append(r)
                cols.append(j)
                data.append(a[k - j])
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        b_ub = np.full(len(active_idx), M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
        # Solve LP
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,
                                    method='highs', options={'time_limit': max(time_left, 0.001)})
        if not res.success:
            print(f"[LP] iteration {iter}: linprog failed ({res.message})")
            break
        g = res.x
        sum_g = np.sum(g)
        if sum_g <= 0:
            print("[LP] zero sum g")
            break
        g_norm = g / sum_g
        # Determine step size t (maybe adapt)
        # simple backtracking to ensure improvement
        improved = False
        for t in [0.5, 0.3, 0.1, 0.05]:
            new_a = (1 - t) * a + t * g_norm

"""
Lightweight optimizer for the step‑function objective.

We generate a diverse set of analytic / random seeds, run a short
Adam refinement on a smooth log‑sum‑exp surrogate, improve the
candidate with exponentiated‑gradient mirror descent, and finally
apply a cutting‑plane linear‑programming iteration (the classic
“LP‑descent” from the reference) that drives the max‑convolution
downwards.  All phases respect the global 1000 s budget and the best
vector is stored in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ---- fast convolution utilities ----
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used in the competition (ℓ₁‑normalised vectors).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ---- helpers for optimisation ----
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ---- analytic families (all returned ℓ₁‑normalised) ----
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ---- ordering helpers ----
def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ---- random low‑frequency curves ----
def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ---- seed pool construction ----
def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # sine windows
    for gamma in np.linspace(0.5, 6.0, 12):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussians
    for sigma in np.linspace(0.10, 0.55, 10):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # power laws
    for p in np.linspace(0.0, 8.0, 17):
        seeds.append(_power(n, p))
        seeds.append(_alternate_order(_power(n, p)))

    # beta family
    for _ in range(15):
        alpha = 10 ** rng.uniform(-2, 0.6)
        beta = 10 ** rng.uniform(-2, 0.6)
        seeds.append(_beta_shape(n, alpha, beta))
        seeds.append(_alternate_order(_beta_shape(n, alpha, beta)))

    # exponentials
    for rate in np.logspace(-4, -1, 12):
        seeds.append(_exponential(n, rate))
        seeds.append(_alternate_order(_exponential(n, rate)))

    # logarithmic
    for eps in np.logspace(-4, -1, 8):
        seeds.append(_logarithmic(n, eps))
        seeds.append(_alternate_order(_logarithmic(n, eps)))

    # Dirichlet draws – many concentrations & orderings
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(8):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(curve)
            seeds.append(_alternate_order(curve))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = _mix_shapes(_uniform(n), curve, mix_alpha)
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Random permutations of a few good deterministic shapes
    for base in [_gaussian(n, 0.2), _power(n, 3.0), _beta_shape(n, 0.5, 0.5)]:
        for _ in range(5):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ---- optimisation primitives ----
def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray | None, float]:
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 0.5
    beta_inc = 0.5
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20  # base learning rate

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional alternating re‑ordering – often helpful
        if it % 100_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {val_alt:.6f}")

    return best_seq, best_val


# ---- new LP‑based refinement ----
def _lp_refine(start_seq: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Cutting‑plane LP refinement.
    Starting from ``start_seq`` (ℓ₁‑normalised) we repeatedly solve

        maximise   Σ_j g_j
        subject to  g ≥ 0,
                    (f * g)[k] ≤ M    for a subset of k

    where M = max(f*f).  The LP direction g is blended into f
    (convex combination) with a small step size; the procedure is
    repeated until the max‑convolution value stops improving or the
    time budget expires.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    iter_cnt = 0
    # tolerance for selecting “active’’ convolution entries
    active_rel_tol = 0.02   # 2 % of the current peak
    # step‑size candidates for the convex combination
    step_candidates = [0.5, 0.3, 0.15, 0.07, 0.03]

    while time.time() < deadline and iter_cnt < 20:
        iter_cnt += 1
        conv_a = _conv_fft(a, a)
        M = np.max(conv_a)

        # pick constraints that are near the current peak
        active_idx = np.where(conv_a >= M * (1.0 - active_rel_tol))[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv_a)], dtype=int)

        # build the sparse matrix A_ub (rows = active constraints)
        rows = []
        cols = []
        data = []
        for r, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # column j runs from i_start … i_end
            for j in range(i_start, i_end + 1):
                rows.append(r)
                cols.append(j)
                data.append(a[k - j])
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        b_ub = np.full(len(active_idx), M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)  # maximise Σ g -> minimise –Σ g

        time_left = deadline - time.time()
        if time_left <= 0:
            break

        # hiGHS respects a ``time_limit`` in seconds
        res = scipy.optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"time_limit": max(time_left, 0.001)},
        )

        if not res.success:
            print(f"[LP] iteration {iter_cnt}: linprog failed ({res.message})")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print("[LP] zero‑sum solution, aborting")
            break
        g_norm = g / sum_g

        # try a few step sizes, keep the first that improves the objective
        improved = False
        for t in step_candidates:
            new_a = (1.0 - t) * a + t * g_norm
            new_val = _objective(new_a)
            if new_val < best_val - 1e-9:
                a = new_a
                best_seq = new_a.copy()
                best_val = new_val
                improved = True
                print(f"[LP] iter {iter_cnt:2d} step {t:.3f} → {new_val:.6f}")
                break
        if not improved:
            # no step produced improvement – likely we are at a local optimum
            print(f"[LP] iter {iter_cnt}: no improving step, terminating")
            break

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    rng = np.random.default_rng(seed)

    # length – a compromise between expressive power and speed

    if previous is not None:

    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives (exact):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – short Adam runs on the elite seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_budget = remaining * 0.12  # reserve most of the time for later phases
    adam_deadline = time.time() + adam_budget

    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > adam_deadline:
            break
        per_deadline = time.time() + (adam_deadline - time.time()) / max(1, len(best_seeds) - idx + 1)
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} ---")

        if seq_adam is not None and val_adam < global_best_val:

        # also try the alternating order of the Adam result
        if seq_adam is not None:
            alt_seq = _alternate_order(seq_adam)
            alt_val = _objective(alt_seq)
            if alt_val < global_best_val:
                global_best_seq = alt_seq
                global_best_val = alt_val

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent sub‑gradient optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:

        seq_mirror, val_mirror = _mirror_subgradient_optimize(

            global_best_val = val_mirror

    # ------------------------------------------------------------------
    # Phase 3 – LP‑based cutting‑plane refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP‑refinement ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, n, deadline)

            global_best_val = val_lp

    )

    return global_best_seq.tolist()

"""
Lightweight optimizer for the step‑function evaluation problem.

The algorithm works in four stages:

1. **Seed generation** – a large pool of analytically‑defined and
   random ℓ₁‑normalised vectors (uniform, triangular, Gaussian,
   beta, low‑frequency cosine sums, Dirichlet draws, permutations…).

2. **Adam on a smooth surrogate** – each of the best seeds is refined
   by Adam optimisation of a log‑sum‑exp approximation of the
   max‑convolution objective.

3. **Mirror‑descent (exponentiated‑gradient) sub‑gradient** – exact
   maximisation of the true objective (2·n·max(conv)).
   This quickly drives the peak down while staying on the probability
   simplex.

4. **LP‑based cutting‑plane refinement** – the classic iteration from the
   reference paper:

   * given the current vector `f`, let `M = max(f*f)`;
   * solve a linear programme that maximises `Σ g_j` under the linear
     constraints `f * g ≤ M` (only the most‑binding convolution
     indices are kept as constraints);
   * blend the solution `g` back into `f` with a small step size.
   Repeating this step monotonically reduces the maximal convolution
   value and typically improves the objective beyond the previous
   best (≈ 1.5087).  All phases respect the global 1000 s budget; the
   final vector is stored in the global name ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for an ℓ₁‑normalised sequence:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------
def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Misc random smooth curves
# ----------------------------------------------------------------------
def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # sine windows
    for gamma in np.linspace(0.5, 6.0, 12):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussians
    for sigma in np.linspace(0.10, 0.55, 10):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # power laws
    for p in np.linspace(0.0, 8.0, 17):
        seeds.append(_power(n, p))
        seeds.append(_alternate_order(_power(n, p)))

    # beta family
    for _ in range(15):
        alpha = 10 ** rng.uniform(-2, 0.6)
        beta = 10 ** rng.uniform(-2, 0.6)
        seeds.append(_beta_shape(n, alpha, beta))
        seeds.append(_alternate_order(_beta_shape(n, alpha, beta)))

    # exponentials
    for rate in np.logspace(-4, -1, 12):
        seeds.append(_exponential(n, rate))
        seeds.append(_alternate_order(_exponential(n, rate)))

    # logarithmic
    for eps in np.logspace(-4, -1, 8):
        seeds.append(_logarithmic(n, eps))
        seeds.append(_alternate_order(_logarithmic(n, eps)))

    # Dirichlet draws – many concentrations & orderings
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(8):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(curve)
            seeds.append(_alternate_order(curve))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = _mix_shapes(_uniform(n), curve, mix_alpha)
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Random permutations of a few good deterministic shapes
    for base in [_gaussian(n, 0.2), _power(n, 3.0), _beta_shape(n, 0.5, 0.5)]:
        for _ in range(5):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    # Previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 0.5
    beta_inc = 0.5
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the exact max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20  # base learning rate

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional alternating re‑ordering – often helpful
        if it % 100_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {val_alt:.6f}")

    return best_seq, best_val


def _lp_refine(start_seq: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Cutting‑plane LP refinement.

    Repeatedly:
        * M = max(f*f)
        * solve   maximise Σ g_j
                  s.t. g ≥ 0 and (f*g)[k] ≤ M  for a small set of “active’’ k
        * blend g (normalised to sum = 1) back into f with a modest step size.
    The process stops when no improving step is found or the time budget
    expires.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    active_rel_tol = 0.02          # rows with conv ≥ (1‑tol)·M are kept
    step_candidates = [0.5, 0.3, 0.15, 0.07, 0.03]

    for it in range(1, 21):
        if time.time() > deadline:
            break

        conv_a = _conv_fft(a, a)
        M = np.max(conv_a)

        active_idx = np.where(conv_a >= M * (1.0 - active_rel_tol))[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv_a)], dtype=int)

        # Build sparse matrix A_ub where row r corresponds to constraint for k = active_idx[r]
        rows, cols, data = [], [], []
        for r, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            for j in range(i_start, i_end + 1):
                rows.append(r)
                cols.append(j)
                data.append(a[k - j])
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        b_ub = np.full(len(active_idx), M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)  # maximise Σ g → minimise –Σ g

        time_left = deadline - time.time()
        if time_left <= 0:
            break

        res = scipy.optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"time_limit": max(time_left, 0.001)},
        )

        if not res.success:
            print(f"[LP] iteration {it}: linprog failed ({res.message})")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print("[LP] zero‑sum solution, aborting")
            break
        g_norm = g / sum_g

        # Try a few step sizes – keep the first that improves the objective
        improved = False
        for t in step_candidates:
            new_a = (1.0 - t) * a + t * g_norm
            new_val = _objective(new_a)
            if new_val < best_val - 1e-9:
                a = new_a
                best_seq = new_a.copy()
                best_val = new_val
                improved = True
                print(f"[LP] iter {it:2d} step {t:.3f} → {new_val:.6f}")
                break
        if not improved:
            print(f"[LP] iter {it}: no improving step, terminating")
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence (list of floats) and stores it in the
    global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length – 4000 gives a good balance between expressiveness and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 0 – generate a rich seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n = {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives (exact):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – brief Adam optimisation on the elite seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_budget = remaining * 0.12
    adam_deadline = time.time() + adam_budget

    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > adam_deadline:
            break
        per_deadline = time.time() + (adam_deadline - time.time()) / max(
            1, len(best_seeds) - idx + 1
        )
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, per_deadline)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq = alt
                global_best_val = alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent sub‑gradient optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP‑based cutting‑plane refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP‑refinement ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, n, deadline)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
