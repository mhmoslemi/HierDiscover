"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three phases:

1. **Broad seed generation** – a large collection of analytically
   defined shapes (Beta, power‑law, Gaussian, mixtures, random Dirichlet,
   low‑frequency curves).  The family parameters are explored much more
   widely than in the reference code (e.g. Beta parameters down to 1e‑2).

2. **Selection & local refinement** – the best ≈10 seeds (by the exact
   surrogate objective) are each refined by a cascade of gradient‑based
   methods:
        • log‑space Adam on a smoothed max (log‑sum‑exp temperature)
        • exact sub‑gradient descent with Euclidean projection onto the
          probability simplex
        • exponentiated‑gradient (mirror‑descent) on the true max‑convolution
   The number of iterations for each stage is adapted to the remaining
   time budget, guaranteeing that the whole routine finishes before the
   outer deadline.

3. **Final evaluation** – the best sequence found is evaluated with the
   official ``evaluate_sequence`` function and stored in the global variable
   ``height_sequence_1`` for possible reuse.

All helper functions are defined at top‑level, no closures are used.
The implementation respects the 1000 s budget (a couple of seconds are
reserved for the final evaluation call).  It returns a plain list of
floats ready for the scorer.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_conv = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the discrete set {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^{-p} (i starts at 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred on the middle of the window."""
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(
    n: int, sigma_frac: float = 0.12, shift_frac: float = 0.25
) -> np.ndarray:
    """Mixture of two symmetric Gaussians placed at shift and 1‑shift."""
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet( conc,…,conc ) and normalise."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(
    n: int, num_freq: int, rng: np.random.Generator
) -> np.ndarray:
    """Sum a few low‑frequency cosine components, clip, and normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _logspace_adam_optimize_long(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑softmax parametrisation with a
    log‑sum‑exp smoothing of the max.  Returns the best vector found
    and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature
    beta_inc = 0.5
    beta_max = 2000.0

    lr0 = 1.0                # initial learning rate (log‑space)

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # ℓ₁‑norm = 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smoothed max (log‑sum‑exp)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # Gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Jacobian of softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a decreasing step‑size and Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)          # polyak‑style step‑size
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Works directly on the simplex via the log‑softmax representation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)          # decreasing step size
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)                # back to simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple annealer that reshuffles mass on small random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:          # reshuffle a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # small pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)  # move ≤5 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _run_local_optim(start_seq: np.ndarray, n: int, deadline: float):
    """
    Run the three main local optimisation stages on a given start vector.
    Returns the best sequence and its surrogate value encountered.
    """
    # ensure proper normalisation
    start_seq = np.maximum(start_seq, 0.0)
    s = start_seq.sum()
    if s <= 0.0:
        start_seq = _uniform(n)
    else:
        start_seq = start_seq / s

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # ---------- Adam (smooth max) ----------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_it = int(min(1_200_000, remaining * 4000))
        seq_adam, val_adam = _logspace_adam_optimize_long(
            best_seq, n, deadline, max_iters=max_it
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"[Stage‑Adam] new best = {val_adam:.6f}")

    # ---------- Sub‑gradient ----------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_it = int(min(2_000_000, remaining * 2000))
        seq_sub, val_sub = _subgradient_proj(
            best_seq, n, deadline, max_iters=max_it
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"[Stage‑Sub] new best = {val_sub:.6f}")

    # ---------- Exponentiated gradient ----------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_it = int(min(1_200_000, remaining * 1500))
        seq_exp, val_exp = _exp_gradient_optimize(
            best_seq, n, deadline, max_iters=max_it
        )
        if val_exp < best_val:
            best_seq = seq_exp.copy()
            best_val = val_exp
            print(f"[Stage‑Exp] new best = {val_exp:.6f}")

    # ---------- Light simulated annealing ----------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_it = int(min(300_000, remaining * 1200))
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_it
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"[Stage‑SA] new best = {val_sa:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # reserve a few seconds for the final evaluation call
    overall_deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine problem size (use previous solution length if available)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and prev:
        n = len(prev)
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.sum() <= 0.0:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        n = 8000
        prev_arr = None

    print(f"\n[Info] Working with n = {n}")

    # ------------------------------------------------------------------
    # Phase 0 – generate a rich pool of analytic seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))

    # Beta family – explore a much wider range than before
    beta_params = np.concatenate(
        [
            np.logspace(-2, -0.3, 6),   # 0.01 … 0.5
            np.linspace(0.6, 5.0, 9),   # 0.6 … 5
            np.logspace(0.7, 1.5, 5),   # 5 … 30
        ]
    )
    for a in beta_params:
        for b in beta_params:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # power‑law family – extend exponent up to 15
    for p in np.concatenate(
        (np.linspace(0.0, 5.0, 11), np.linspace(6.0, 15.0, 5))
    ):
        seeds.append(_power_law(n, float(p)))

    # Gaussian family – several sigmas
    for sigma in np.linspace(0.05, 0.45, 9):
        seeds.append(_gaussian(n, float(sigma)))

    # Double‑Gaussian mixtures
    for sigma in (0.08, 0.12, 0.15):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=float(sigma),
                                          shift_frac=float(shift)))

    # Random Dirichlet draws with a wide range of concentration
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, float(conc), rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixture of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        curve = _lowfreq_curve(n, 3, rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Include previous solution, if any
    if prev_arr is not None and prev_arr.size == n and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())

    # Normalise everything robustly and eliminate any that are all‑zero
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean_seeds.append(v / s)
        else:
            clean_seeds.append(_uniform(n))

    # ------------------------------------------------------------------
    # Phase 1 – cheap surrogate evaluation and selection
    # ------------------------------------------------------------------
    evals = [(vec, _objective_surrogate(vec)) for vec in clean_seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 12
    top_candidates = [vec for vec, _ in evals[:top_k]]

    print("\n[Seed ranking] top surrogate values:")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # Initialise global best with the very best seed
    global_best_seq = top_candidates[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – local refinement of the selected candidates
    # ------------------------------------------------------------------
    for idx, candidate in enumerate(top_candidates):
        remaining = overall_deadline - time.time()
        if remaining < 30.0:  # not enough time for another full refinement
            break
        print(f"\n--- Refinement of candidate #{idx+1} (remaining {remaining:.1f}s) ---")
        cand_deadline = time.time() + remaining - 5.0  # keep a small buffer
        refined_seq, refined_val = _run_local_optim(candidate, n, cand_deadline)
        if refined_val < global_best_val:
            global_best_seq = refined_seq.copy()
            global_best_val = refined_val
            print(f"[Update] new global best surrogate = {refined_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – optional random restarts (if time permits)
    # ------------------------------------------------------------------
    while True:
        remaining = overall_deadline - time.time()
        if remaining < 40.0:  # need a minimum chunk to be worthwhile
            break
        # generate a fresh random start
        conc = rng.choice([0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
        rand_start = _random_dirichlet(n, float(conc), rng)
        print(
            f"\n--- Random restart (conc={conc}) – remaining {remaining:.1f}s ---"
        )
        rand_deadline = time.time() + remaining - 5.0
        refined_seq, refined_val = _run_local_optim(rand_start, n, rand_deadline)
        if refined_val < global_best_val:
            global_best_seq = refined_seq.copy()
            global_best_val = refined_val
            print(f"[Update] new global best surrogate = {refined_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
