def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0
    # Choose length n optimum (maybe 5000)
    n = 5000
    # load previous solution if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev = np.asarray(prev, dtype=np.float64)
        if prev.sum() > 0:
            prev = prev / prev.sum()
    else:
        prev = None

    # Generate seeds
    seeds = []
    seeds.append(np.full(n, 1.0/n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, sigma_frac=0.1))
    seeds.append(_gaussian(n, sigma_frac=0.08))
    seeds.append(_beta_shape(n, alpha=0.5, beta=0.5))
    # add previous
    if prev is not None:
        if prev.shape[0] != n:
            # if different length, resample via interpolation? Could just tile or truncate
            # We'll adjust length via simple interpolation
            prev_resized = np.interp(np.linspace(0, len(prev)-1, n), np.arange(len(prev)), prev)
            seeds.append(prev_resized / prev_resized.sum())
        else:
            seeds.append(prev)

    # Random Dirichlet seeds
    concs = [0.01, 0.05, 0.2, 0.5, 1.0]
    for conc in concs:
        for _ in range(5):
            s = rng.dirichlet(np.full(n, conc))
            seeds.append(s)

    # Evaluate seeds, keep top K
    evaluated = [(s, _objective(s)) for s in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    best_seq, best_val = evaluated[0][0], evaluated[0][1]

    # Mirror descent optimization on top seeds
    for seq, _ in evaluated[:top_k]:
        if time.time() > deadline: break
        seq_opt, val_opt = _mirror_subgradient_optimize(seq, n, deadline, max_iters=200_000)
        if val_opt < best_val:
            best_seq, best_val = seq_opt, val_opt

    # Mass transfer hill climbing

def _mass_transfer_optimize(start_seq, n, deadline, max_iters=500_000):
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    init_eps = 0.02
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        eps = init_eps / np.sqrt(it)  # decays
        # choose source index with cur[i] > eps
        src_candidates = np.where(cur > eps)[0]
        if src_candidates.size == 0:
            continue
        i = rng.choice(src_candidates)
        j = rng.integers(0, n)  # can equal i; if equal skip
        if j == i:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        # ensure nonnegative (already) and sum unchanged
        new_val = _objective(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[Mass] it={it} best={best_val:.6f}")
        else:
            # optionally accept uphill with some probability (sim anneal)
            T = cur_val * 0.001
            if rng.random() < np.exp(-(new_val - cur_val) / T):
                cur = new
                cur_val = new_val
    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

Algorithm
---------
1. Choose a moderate length (n = 5000) – large enough to give
   flexibility, still fast for FFT‑based convolutions.
2. Build a diversified pool of seed vectors:
   - deterministic analytic shapes (uniform, triangular, Gaussian,
     beta, Laplace, etc.);
   - Dirichlet draws with several concentration parameters;
   - the previous best solution (if any), resized to length n.
3. Keep the top few seeds according to the surrogate objective
   `2·n·max(conv)`.
4. Refine each top seed with a short mirror‑subgradient (exponentiated
   gradient) descent on the *exact* max‑convolution objective.
5. Starting from the current global best, run a large‑scale *mass‑transfer*
   hill‑climb: repeatedly move a small amount of mass from a larger entry
   to a smaller one if the move reduces the surrogate objective.
6. Apply a simple swap‑based simulated annealing on the permutation of
   the entries – this often lowers the peak further.
7. Re‑order the vector by the “alternating order” (large, small,
   2nd‑large, 2nd‑small …) and keep the improvement if any.
8. Finish with a longer Adam optimisation (log‑softmax parameters,
   smoothed max via log‑sum‑exp) for fine polishing.
9. Return the best vector found (as a list) and store it globally as
   ``height_sequence_1`` for possible reuse.

All steps respect a global time budget (default 1000 s).  The code
uses only NumPy and the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (numerator of evaluate_sequence with sum = 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)==1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helper functions
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Re‑order as large, small, 2nd‑large, 2nd‑small, … and renormalise."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


# ----------------------------------------------------------------------
# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.4) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the true max‑conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.30  # step‑size prefactor

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 400_000) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax parameters using a smoothed max (log‑sum‑exp)."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 0.5          # temperature for smooth max
    beta_inc = 0.5
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mass_transfer_optimize(start_seq: np.ndarray, n: int, deadline: float,
                           max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that moves a tiny amount of mass from a larger entry
    to a smaller one if the move reduces the surrogate objective.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    it = 0
    init_frac = 0.02  # initial fraction of the source entry to move

    while it < max_iters and time.time() < deadline:
        it += 1
        # Decaying step size (fraction of source mass)
        frac = init_frac / np.sqrt(it)
        # pick a source with enough mass
        src_candidates = np.where(cur > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        if cur[i] * frac < 1e-12:
            continue
        eps = cur[i] * frac
        # pick a distinct target
        j = rng.integers(0, n)
        if j == i:
            continue

        new = cur.copy()
        new[i] -= eps
        new[j] += eps

        new_val = _objective(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[Mass] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _perm_hill_climb(start_seq: np.ndarray, n: int,
                     deadline: float,
                     max_swaps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Swap‑based simulated‑annealing on the permutation of entries.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    it = 0
    T0 = cur_val * 0.005
    T = T0

    while it < max_swaps and time.time() < deadline:
        it += 1
        T = max(T * 0.9999, 1e-12)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 20_000 == 0:
                    print(f"[Perm] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    evaluate_sequence() within the given time budget.
    The result is stored as the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Length – 5000 gives a good compromise between expressiveness
    # and speed for FFT‑based convolution.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Load previous best (if any) and normalise it.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0.0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Seed generation – a rich mix of analytic families and random draws.
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    for sigma in np.linspace(0.07, 0.25, 6):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))
    for alpha in [0.4, 0.8, 2.0]:
        for beta in [0.4, 0.8, 2.0]:
            seeds.append(_beta_shape(n, alpha, beta))
            seeds.append(_alternate_order(_beta_shape(n, alpha, beta)))

    # Dirichlet draws with various concentration parameters
    concs = [0.01, 0.05, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_alternate_order(raw))

    # Previous solution (if any) – resize if length differs
    if previous is not None:
        if previous.shape[0] != n:
            # linear interpolation to the new length
            x_old = np.linspace(0, 1, previous.shape[0])
            x_new = np.linspace(0, 1, n)
            resized = np.interp(x_new, x_old, previous)
            resized = resized / resized.sum()
            seeds.append(resized)
            seeds.append(_alternate_order(resized))
        else:
            seeds.append(previous)
            seeds.append(_alternate_order(previous))

    print(f"\n--- Seed generation (length {n}) ---")
    print(f"Number of raw seeds: {len(seeds)}")

    # ------------------------------------------------------------------
    # Evaluate seeds, keep the top few.
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise global best with the very best seed.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short mirror‑descent refinements on top seeds.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        rem = deadline - time.time()
        # allocate at most 0.2 s per seed (approx 200 iters per sec)
        max_iters = int(min(200_000, rem * 400))
        max_iters = max(10_000, max_iters)
        print(f"\n--- Mirror descent on seed {idx}/{top_k} (max_iters={max_iters}) ---")
        seq_m, val_m = _mirror_subgradient_optimize(seed_vec, n, deadline,
                                                    max_iters=max_iters)
        if val_m < global_best_val:
            global_best_seq, global_best_val = seq_m.copy(), val_m
            print(f"[global‑Mirror] new best = {val_m:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – mass‑transfer hill‑climb (continuous values).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mass‑transfer hill‑climbing ---")
        seq_mt, val_mt = _mass_transfer_optimize(global_best_seq, n, deadline,
                                                max_iters=800_000)
        if val_mt < global_best_val:
            global_best_seq, global_best_val = seq_mt.copy(), val_mt
            print(f"[global‑Mass] new best = {val_mt:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – permutation annealing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation hill‑climbing (annealed swaps) ---")
        seq_perm, val_perm = _perm_hill_climb(global_best_seq, n, deadline,
                                              max_swaps=200_000)
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[global‑Perm] new best = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – alternating order tweak.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        alt_seq = _alternate_order(global_best_seq)
        alt_val = _objective(alt_seq)
        if alt_val < global_best_val:
            global_best_seq, global_best_val = alt_seq.copy(), alt_val
            print(f"[global‑Alt] new best after alternating order = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – long Adam polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam polishing (log‑softmax) ---")
        seq_adam, val_adam = _adam_optimize(global_best_seq, n, deadline,
                                            max_iters=400_000)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam.copy(), val_adam
            print(f"[global‑Adam] final best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nSurrogate best ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
