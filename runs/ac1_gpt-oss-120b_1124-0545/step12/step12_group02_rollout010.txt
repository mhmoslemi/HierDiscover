def _lp_step(f, M, idxs):
    # f: np.ndarray (n,)
    # idxs: list of lag indices (int) to enforce constraints
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    constraints = []
    for k in idxs:
        # compute convolution row coefficients
        # for j from max(0, k-(n-1)) to min(n-1, k):
        # coefficient for b[j] is f[k-j]
        coeff = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for each j = i_range? Actually j = k - i
        # i runs i_start..i_end; then j = k - i in range.
        # So we can fill coefficients
        idx_j = k - np.arange(i_start, i_end+1)
        coeff[idx_j] = f[i_start:i_end+1] # assign f[i] to b[j]
        constraints.append(coeff @ b <= M)
    # optionally also include constraint sum(b) >= epsilon ?

    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if b.value is None:
        # fall back to uniform
        return np.full(n, 1.0 / n)
    return np.array(b.value, dtype=np.float64)

"""
Propose a step‑function sequence that minimises the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The strategy combines a very diverse seed pool, several Adam‑type
optimisers working on the log‑softmax (probability‑simplex) parametrisation,
a greedy multi‑restart scheme, and an aggressive simulated‑annealing
polish that runs until the time budget is exhausted.  All operations are
performed with FFT‑based convolutions (O(n log n)) and respect the
non‑negativity and normalisation constraints required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import cvxpy as cp  # used only for very cheap LP sub‑steps (fallback)

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector
    (∑seq = 1).  It preserves the ordering of candidates for the true
    evaluator.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows as _scipy_windows
        a = _scipy_windows.kaiser(n, beta)
    except Exception:                               # pragma: no cover
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        t = beta * np.sqrt(1 - y * y)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    try:
        from scipy.signal import windows as _scipy_windows
        a = _scipy_windows.dpss(n, nw, Kmax=1, return_ratios=False)[0]
    except Exception:                               # pragma: no cover
        return _uniform(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _chebwin_window(n: int, att: float) -> np.ndarray:
    try:
        from scipy.signal import windows as _scipy_windows
        a = _scipy_windows.chebwin(n, at=att)
    except Exception:                               # pragma: no cover
        return _uniform(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15, shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few random cosine components, clip to non‑negative and normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed collection

def _collect_seeds_extended(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a large and diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for nw in (2.0, 3.0, 4.0, 5.0):
        seeds.append(_dpss_window(n, nw))
    for att in (30, 40, 60, 80):
        seeds.append(_chebwin_window(n, att))

    # Gaussian families -------------------------------------------------------
    for sigma in np.linspace(0.10, 0.45, 9):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    # Power‑law ---------------------------------------------------------------
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family grid --------------------------------------------------------
    alphas = np.linspace(0.2, 5.0, 9)
    betas = np.linspace(0.2, 5.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows -----------------------------------------------------------
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet / exponential -----------------------------------------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves ---------------------------------------------
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with a low‑frequency curve -------------------------
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Mixtures of two random Dirichlet draws ---------------------------------
    for _ in range(5):
        a = _random_dirichlet(n, rng.uniform(0.05, 1.5), rng)
        b = _random_dirichlet(n, rng.uniform(0.05, 1.5), rng)
        alpha = rng.uniform(0.1, 0.9)
        seeds.append(_mix_shapes(a, b, alpha))

    # previous solution, if any -----------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final normalisation ------------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned

def _logspace_adam_optimize_long(start_vec: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 500_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑softmax parametrisation.
    Uses a log‑sum‑exp smoothing of the maximum with a temperature β
    that is increased slowly so that the landscape stays smooth early
    on.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 1.0               # start temperature – quite smooth
    beta_inc = 0.5
    beta_max = 2500.0

    lr0 = 1.0                # initial learning rate (log‑space)

    best_seq = None
    best_val = float("inf")
    cur_val = _objective_surrogate(_softmax(v))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # ----- smoothed max (log‑sum‑exp) ------------------------------------
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Jacobian of softmax (logits → probabilities)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a diminishing step size and Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / np.sqrt(it)          # Polyak‑style step‑size
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) directly on the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (same as in _subgradient_proj)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)          # decreasing step size
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)              # back to simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing_improved(start_seq: np.ndarray,
                                  n: int,
                                  deadline: float,
                                  max_iters: int = 1_500_000,
                                  init_temp_factor: float = 0.08,
                                  min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Aggressive SA that reshuffles mass on random subsets with a slow cooling."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # very slow cooling – keeps exploration alive longer
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.65:          # reshuffle a random subset
            k = rng.integers(2, 20)          # subset size 2 … 19
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # small pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.03)   # move ≤3 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA‑imp] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _greedy_local_improvement(start_seq: np.ndarray,
                              n: int,
                              deadline: float,
                              max_trials: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly move a small amount of mass from a random
    source index to a random destination index, accepting only improvements.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    rng = np.random.default_rng()

    for it in range(max_trials):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.random() * cur[i] * 0.07   # up to 7 % of source mass
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_val = _objective_surrogate(new)
        if new_val + 1e-12 < cur_val:          # strict improvement
            cur = new
            cur_val = new_val
            if it % 5000 == 0:
                print(f"[Greedy] it={it:7d} cur={cur_val:.6f}")

    return cur, cur_val


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalisation."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects a wall‑clock budget (default 1000 s) and returns the
    best sequence found as a plain ``list`` of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # small safety margin for the final evaluation call
    overall_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Stage 0 – generate a massive pool of analytic and random seeds
    # ------------------------------------------------------------------
    n0 = 4000
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print(f"\n--- Seed generation (n={n0}) ---")
    seeds = _collect_seeds_extended(n0, rng, prev_arr)

    # cheap surrogate ranking ------------------------------------------------
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 15
    best_seeds = [vec for vec, _ in evals[:top_k]]
    print(f"Selected top {top_k} seeds (surrogate values):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample to a finer grid (if time allows)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    n_up = n0 * 2                     # 8000 points
    if remaining > 30.0:
        print(f"\n--- Upsampling to n={n_up} ---")
        global_best_seq = _upsample_sequence(global_best_seq, n_up)
        global_best_seq = global_best_seq / global_best_seq.sum()
        global_best_val = _objective_surrogate(global_best_seq)
        print(f"[Upsample] surrogate after interpolation = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – multiple short Adam restarts (diversify the search)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 80.0:
        restarts = 12
        iters_per = int(min(80_000, remaining * 500))  # keep each run modest
        print(f"\n--- Multiple short Adam restarts (iters={iters_per}) ---")
        for r in range(restarts):
            if time.time() > overall_deadline:
                break
            # random perturbation of the current best
            pert = _random_dirichlet(n_up, rng.uniform(0.05, 1.5), rng)
            mixed_start = _mix_shapes(global_best_seq, pert, alpha=0.35)
            seq_adam, val_adam = _logspace_adam_optimize_long(
                mixed_start, n_up, overall_deadline, max_iters=iters_per)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Restart {r}] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long log‑space Adam optimisation (fine‑tune)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 60.0:
        max_iters_adam = int(min(500_000, remaining * 4000))
        print(f"\n--- Long Adam optimisation (max_iters={max_iters_adam}) ---")
        seq_adam, val_adam = _logspace_adam_optimize_long(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters_adam)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam‑long] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – exact sub‑gradient projection
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 25.0:
        max_iters_sub = int(min(1_200_000, remaining * 2000))
        print(f"\n--- Sub‑gradient projection (max_iters={max_iters_sub}) ---")
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters_sub)
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[Sub‑proj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 15.0:
        max_iters_exp = int(min(600_000, remaining * 1500))
        print(f"\n--- Exponentiated‑gradient refinement (max_iters={max_iters_exp}) ---")
        seq_exp, val_exp = _exp_gradient_optimize(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[Exp‑grad] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – aggressive simulated‑annealing polishing
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 8.0:
        max_iters_sa = int(min(1_500_000, remaining * 2500))
        print(f"\n--- Simulated‑annealing polishing (max_iters={max_iters_sa}) ---")
        seq_sa, val_sa = _simulated_annealing_improved(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters_sa)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA‑polish] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 7 – final greedy local improvement
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        max_trials = int(min(300_000, remaining * 5000))
        print(f"\n--- Greedy local improvement (trials={max_trials}) ---")
        seq_greedy, val_greedy = _greedy_local_improvement(
            global_best_seq, n_up, overall_deadline, max_trials=max_trials)
        if val_greedy < global_best_val:
            global_best_seq = seq_greedy.copy()
            global_best_val = val_greedy
            print(f"[Greedy] new global best = {val_greedy:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
