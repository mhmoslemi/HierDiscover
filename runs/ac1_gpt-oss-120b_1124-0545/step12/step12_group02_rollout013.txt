def lp_optimize(f, max_iters=200, t=0.5, epsilon=1e-6):
    for it in range(max_iters):
        conv = _conv_fft(f, f)
        M = conv.max()
        # Build constraints A * g <= M
        # A is shape (2n-1, n)
        # data: for each row k, column i with coefficient f_i if i in window
        # We'll build using sparse csr
        A = build_conv_matrix(f)   # shape (2n-1, n)
        b = np.full(2*n - 1, M, dtype=np.float64)
        # Solve LP maximize sum(g) => minimize -sum(g)
        res = linprog(-np.ones_like(f), A_ub=A, b_ub=b,
                      bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x  # shape (n,)
        # combine
        f = (1 - t) * f + t * g
        # Project onto simplex
        f = _proj_simplex(f)
        # compute new max conv; break if not improving
        new_conv = _conv_fft(f, f)
        new_M = new_conv.max()
        if M - new_M < epsilon:
            break
    return f

   with a long log‑space Adam optimisation on a smooth
   soft‑max approximation of the max‑convolution.  The final
   vector is projected onto the probability simplex and evaluated
   with the official ``evaluate_sequence`` routine.

The whole routine respects the 1000 s budget and prints intermediate
progress.  It falls back to a pure Adam optimiser if SciPy is unavailable.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy).real[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector (∑seq=1).
    It is proportional to the true evaluate_sequence value.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple analytic seeds (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a symmetric Dirichlet and normalise to sum 1."""
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few low‑frequency cosines, clip to ≥0 and renormalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, then renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalisation."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Adam optimiser (log‑softmax, smoothed max)
# ----------------------------------------------------------------------
def _logspace_adam_optimize_long(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑softmax parametrisation with an aggressive
    temperature schedule.  Returns the best vector found and its surrogate
    value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature
    beta_inc = 0.5
    beta_max = 2000.0

    lr0 = 1.0                # initial learning rate (log‑space)

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probability simplex, ℓ₁‑norm = 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # --- smoothed max using log‑sum‑exp ---------------------------------
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # Gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Gradient on logits (log‑softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # Temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (requires SciPy)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, iterations: int, t: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a few convex‑combination steps using the LP described in the
    statement.  Returns the refined sequence and its maximal convolution value.
    """
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog

    n = len(f)

    # pre‑compute index arrays that do not depend on f
    row_idx = np.add.outer(np.arange(n), np.arange(n)).ravel()
    col_idx = np.tile(np.arange(n), n)

    for it in range(iterations):
        if time.time() > deadline:
            break

        conv = _conv_fft(f, f)
        M = conv.max()

        # Build the (2n‑1) × n constraint matrix A where A·g = f*g
        data = np.repeat(f, n)                 # f repeated for each column
        A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        # Linear programme: maximise 1ᵀg  (i.e. minimise –1ᵀg)
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # Give the solver a tiny time budget per call to stay safe
        remaining = deadline - time.time()
        linopt_time = max(0.5, min(30.0, remaining - 5.0))
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=(0, None),
                      method='highs',
                      options={'presolve': True, 'time_limit': linopt_time})

        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g = res.x
        # convex combination and simplex projection
        f = (1.0 - t) * f + t * g
        f = _proj_simplex(f)

        # optional early stop if improvement stalls
        new_conv = _conv_fft(f, f)
        new_M = new_conv.max()
        if abs(M - new_M) < 1e-12:
            break

        if it % 2 == 0:
            print(f"[LP] iter {it + 1:3d}  max_conv={new_M:.6e}")

    # final max‑convolution value
    final_M = _conv_fft(f, f).max()
    return f, final_M


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted time.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a safety margin for the final scorer call
    overall_deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Choose a modest resolution for the LP stage
    # --------------------------------------------------------------
    n_small = 2000
    print(f"\n--- LP‑phase on n={n_small} ---")

    # a few diverse deterministic/random seeds (ℓ₁‑normalised)
    seeds = [
        _uniform(n_small),
        _triangular(n_small),
        _random_dirichlet(n_small, conc=0.2, rng=rng),
        _random_dirichlet(n_small, conc=0.5, rng=rng),
        _random_exponential(n_small, rng),
        _lowfreq_curve(n_small, num_freq=3, rng=rng),
        _lowfreq_curve(n_small, num_freq=4, rng=rng),
    ]

    # evaluate surrogate and keep the best few
    evals = [(s, _objective_surrogate(s)) for s in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [s for s, _ in evals[:top_k]]
    print("Best surrogate values for the raw seeds:")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i + 1}: {val:.6f}")

    # --------------------------------------------------------------
    # 1. LP refinement on each promising seed
    # --------------------------------------------------------------
    best_f = best_seeds[0].copy()
    best_M = _conv_fft(best_f, best_f).max()

    remaining = overall_deadline - time.time()
    lp_iters_per_seed = 5               # keep it cheap
    t_lp = 0.5                           # convex‑combination weight

    for idx, seed_vec in enumerate(best_seeds):
        if remaining < 30.0:
            break
        print(f"\n[LP] refining seed #{idx + 1}")
        f_ref, M_ref = _lp_refine(seed_vec, iterations=lp_iters_per_seed,
                                  t=t_lp, deadline=overall_deadline)
        if M_ref < best_M:
            best_f = f_ref
            best_M = M_ref
            print(f"   --> improvement: new max_conv = {M_ref:.6e}")
        remaining = overall_deadline - time.time()

    # --------------------------------------------------------------
    # 2. Upsample to a finer grid for final polishing
    # --------------------------------------------------------------
    if remaining > 60.0:
        n_big = 8000
        print(f"\n--- Upsampling to n={n_big} ---")
        best_f = _upsample_sequence(best_f, n_big)
        best_f = best_f / best_f.sum()
        remaining = overall_deadline - time.time()
    else:
        n_big = len(best_f)

    # --------------------------------------------------------------
    # 3. Log‑space Adam optimisation (smooth max) – the final polish
    # --------------------------------------------------------------
    if remaining > 30.0:
        max_iters_adam = int(min(300_000, remaining * 4000))
        print(f"\n--- Long Adam optimisation (max_iters={max_iters_adam}) ---")
        seq_adam, val_adam = _logspace_adam_optimize_long(
            best_f, n_big, overall_deadline, max_iters=max_iters_adam
        )
        if seq_adam is not None:
            best_f = seq_adam
            best_M = _conv_fft(best_f, best_f).max()
            print(f"[Adam] final surrogate = {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4. Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_f.tolist()
    try:
        true_val = evaluate_sequence(final_seq)
        print("\n=== Finished ===")
        print(f"Surrogate ≈ {_objective_surrogate(best_f):.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = final_seq
    return final_seq
