"""
Efficient search for a non‑negative step‑function that minimizes

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The optimizer works with a probability vector (ℓ₁‑normalised, sum = 1) –
the scaling cancels out in the final score.  The algorithm consists of three
stages and runs entirely inside the supplied 1000 s budget.

Stage 1 – Broad seed pool
    * analytic families (uniform, geometric, power‑law, beta, Tukey,
      tri‑/trapezoidal, Gaussian, sine‑window)
    * very‑sparse “spike” vectors (≈ 50‑150 non‑zero entries) – these are
      good candidates for a low ℓ∞‑convolution because only few ordered
      pairs contribute to any sum.
    * random Dirichlet draws with a wide range of concentration parameters.
    * the previous best solution (if any).

Stage 2 – Deterministic refinement
    1. **Entropic mirror descent** – multiplicative updates on the simplex,
       using the sub‑gradient of the currently active (maximal) convolution
       entry.  A decreasing learning rate ηₜ = η₀/√t stabilises the descent.
    2. **Pure max‑sub‑gradient descent** – we keep only the single index
       k = argmax(conv) and move mass away from the coordinates that
       participate in this entry (g = 2·A·eₖ).  After each step we project
       back onto the simplex.  This aggressively eliminates the current
       peak.

Stage 3 – Stochastic polishing
    * “block‑Dirichlet” moves: pick a random subset of coordinates,
      redistribute the total mass of that block with a Dirichlet draw.
    * “pair‑swap” moves: exchange a tiny ε‑portion of mass between two
      indices (preserves the sum exactly).
    * classic simulated‑annealing acceptance with an exponential temperature
      schedule.

All convolutions are performed with a real FFT (complexity O(n log n)).
The code respects the time budget, prints progress, stores the best
sequence in the global variable ``height_sequence_1`` and finally returns
the best list of heights.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán (2013) – “Projection onto the probability simplex”.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_spike(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """m non‑zero equal spikes (a_i = 1/m on a random support of size m)."""
    if m <= 0 or m > n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # 1️⃣ deterministic / analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):

    for gamma in np.linspace(0.0, 4.0, 9):

    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    # geometric – we sweep much closer to 1 than the original code
    r_vals = np.concatenate([
        np.linspace(0.2, 0.98, 12),
        np.linspace(0.98, 0.9999, 12)   # far‑right tail

        seeds.append(_geom(n, r))

    # power‑law
    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    # beta shapes
    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # 2️⃣ sparse spike candidates (different support sizes)

    # 4️⃣ previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:

    return seeds

def _exp_mirror_descent(start_vec: np.ndarray, n: int, deadline: float,
                        max_iters: int = 1_000_000,
                        eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Entropic mirror descent on the max‑convolution objective.
    The update a ← a·exp(-η·g) stays in the simplex after normalisation.
    g is the sub‑gradient of the active convolution entry.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[Mirror] it={it:6d} best={best_val:.6f}")

        # sub‑gradient for the active shift k:
        # (a * shift_of_a)[i] contributes a[i]·a[k−i] for all i with 0≤k−i<n
        rev = k - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        # Numerical safety – stay non‑negative
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s
    return best_seq, best_val


def _max_subgradient(start_vec: np.ndarray, n: int, deadline: float,
                    init_lr: float = 0.35) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that always follows the *single* active
    constraint (the maximal convolution entry).
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[MaxSub] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        rev = k - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.06,
                         min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Light‑weight simulated annealing with two move types:
    – block Dirichlet reshuffle,
    – pair‑swap of a tiny ε‑mass.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        # ---- move selection -------------------------------------------------
        if rng.random() < 0.6:
            # block Dirichlet move
            k = rng.integers(3, 15)               # size of the block
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total > 0.0:
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
                new = cur.copy()
                new[idx] = new_alloc
            else:
                continue
        else:
            # pair‑swap move
            i, j = rng.choice(n, size=2, replace=False)
            eps = 1e-4   # tiny fraction of the total mass
            delta = min(eps, cur[i])          # amount we can take from i
            new = cur.copy()
            new[i] -= delta
            new[j] += delta

        new_val = _objective(new)
        delta_val = new_val - cur_val

        if delta_val < 0.0 or rng.random() < np.exp(-delta_val / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start with the previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the candidate vector – we take 6000 (still fast with FFT)
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation + quick evaluation
    # ------------------------------------------------------------------
    print(f"\n--- Gathering seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(7, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deterministic optimisation
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        # Mirror descent – 20 % of remaining time
        mirror_deadline = time.time() + 0.20 * time_left
        print("\n--- Exponential‑mirror descent ---")
        seq_m, val_m = _exp_mirror_descent(global_best_seq, n, mirror_deadline,
                                           max_iters=2_000_000, eta0=0.7)
        if val_m < global_best_val:
            global_best_seq, global_best_val = seq_m, val_m
            print(f"[Mirror] new global best = {val_m:.6f}")

    time_left = deadline - time.time()
    if time_left > 0:
        # Max‑subgradient – majority of the remaining budget
        sub_deadline = time.time() + 0.60 * time_left
        print("\n--- Max‑subgradient refinement ---")
        seq_sg, val_sg = _max_subgradient(global_best_seq, n, sub_deadline,
                                          init_lr=0.35)
        if val_sg < global_best_val:
            global_best_seq, global_best_val = seq_sg, val_sg
            print(f"[MaxSub] new global best = {val_sg:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline,
                                              max_iters=1_200_000,
                                              init_temp_factor=0.06)
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose the result for the next run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
