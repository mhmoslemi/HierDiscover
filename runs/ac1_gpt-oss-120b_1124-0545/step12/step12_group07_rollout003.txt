"""
A faster‑converging three‑stage optimiser for the autocorrelation
max‑norm objective.

The idea is simple but effective:

1️⃣ **Rich seed pool** – in addition to the analytic families already
   used in the reference code we generate a large number of
   randomised parameter draws (beta, Gaussian, Tukey, power,
   geometric, Dirichlet).  All vectors are ℓ₁‑normalised.

2️⃣ **Iterated first‑order descent** – we run several short mirror–descent
   runs from the best few seeds (different learning‑rate schedules) and
   a long projected sub‑gradient descent where the active set contains
   the `top_k` largest convolution entries (here `top_k` is enlarged
   to 30).  The sub‑gradient method uses an Adam‑style adaptive step
   size without any external libraries.

3️⃣ **Heavy stochastic polishing** – a simulated‑annealing phase that
   repeatedly redistributes mass on a random block of coordinates.
   The temperature decays slowly and the block size varies up to 30,
   allowing the algorithm to escape plateaux.

All work respects a user‑supplied time budget (default 1000 s) and the
final sequence is stored in the global variable ``height_sequence_1``.
The routine returns the best list of heights it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate objective for ℓ₁‑normalised sequences (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Seed generation (augmented)
# ----------------------------------------------------------------------


def _collect_rich_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a large and diverse pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families (same as before but a bit denser)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for wf in np.linspace(0.05, 0.40, 9):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    for gamma in np.linspace(0.0, 5.0, 13):
        seeds.append(_sine_window(n, gamma=gamma))

    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    # geometric families – finer grid
    r_vals = np.concatenate([np.linspace(0.2, 0.98, 15), np.linspace(0.98, 0.9999, 10)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # power law
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # beta family – random draws (log‑uniform for parameters)
    for _ in range(30):
        a = np.exp(rng.uniform(np.log(0.1), np.log(10.0)))
        b = np.exp(rng.uniform(np.log(0.1), np.log(10.0)))
        seeds.append(_beta_shape(n, a, b))

    # Gaussian – many σ values
    for sigma in np.linspace(0.08, 0.40, 13):
        seeds.append(_gaussian(n, sigma))

    # Random Dirichlet with a wide concentration spectrum
    for conc in (0.01, 0.03, 0.07, 0.15, 0.30, 0.60, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(15):
        seeds.append(_random_exponential(n, rng))

    # Perturb previous solution (if any) – add a few jittered copies
    if previous is not None and previous.shape == (n,):
        base = previous.astype(np.float64)
        for eps in (0.002, 0.005, 0.010):
            noisy = base + rng.normal(scale=eps, size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                seeds.append(noisy / s)

    return seeds

def _exp_mirror_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Classic multiplicative (entropic) mirror descent.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[Mirror] it={it:7d} best={best_val:.6f}")

        # sub‑gradient for active shift
        rev = max_idx - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _subgradient_adam(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    top_k: int = 30,
    lr: float = 0.20,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam‑style sub‑gradient descent.
    The gradient is the average of the sub‑gradients of the ``top_k``
    largest convolution entries.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)

        # indices of the largest entries
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        cur_max = conv[idxs].max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 15000 == 0:
                print(f"[AdamSub] it={t:7d} best={best_val:.6f}")

        # aggregate gradient of the selected entries
        g = np.zeros_like(a)
        for k in idxs:
            rev = k - np.arange(n)
            mask = (rev >= 0) & (rev < n)
            g[mask] += 2.0 * a[rev[mask]]
        g /= float(len(idxs))

        # Adam update (on the simplex we use a projected additive step)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """
    Polishing phase – a more aggressive simulated‑annealing that reshuffles
    up to 30 coordinates at a time and uses a slower temperature decay.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9998, min_temp)

        k = rng.integers(2, 31)          # block size
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 20000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _short_restart_descent(
    n: int,
    rng: np.random.Generator,
    global_best: np.ndarray,
    remaining_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a quick mirror‑descent run from a random Dirichlet seed.
    This function is called repeatedly while time permits to explore many
    basins.  It returns the best sequence it discovered (or the input best
    if no improvement occured).
    """
    # allocate a tiny slice of the remaining budget
    slice_deadline = time.time() + min(6.0, max(0.5, remaining_deadline * 0.02))

    # random Dirichlet with a randomly chosen concentration
    conc = rng.choice([0.02, 0.05, 0.12, 0.3, 0.7, 1.5, 3.0])
    seed = _random_dirichlet(n, conc, rng)

    # mirror descent with a relatively aggressive schedule
    seq, val = _exp_mirror_descent(
        seed,
        n,
        slice_deadline,
        max_iters=150_000,
        eta0=0.9,
    )
    # compare with current global best
    if val < _objective(global_best):
        return seq, val
    else:
        return global_best, _objective(global_best)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ----- previous solution (if any) ---------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ----- problem size ------------------------------------------------
    n = 4000          # length that previously yielded the best known value

    # ----- Stage 1 – rich seed pool -----------------------------------
    print(f"\n--- Collecting a rich seed pool (length {n}) ---")
    seeds = _collect_rich_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # quick surrogate evaluation of all seeds
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ----- Stage 2 – deterministic refinement --------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        # mirror descent from the best seed (40% of remaining time)
        mirror_deadline = time.time() + 0.40 * time_left
        print("\n--- Mirror descent (global) ---")
        seq_m, val_m = _exp_mirror_descent(
            global_best_seq, n, mirror_deadline,
            max_iters=2_500_000,
            eta0=0.6,
        )
        if val_m < global_best_val:
            global_best_seq, global_best_val = seq_m, val_m
            print(f"[Mirror] new global best = {val_m:.6f}")

    # sub‑gradient (Adam) – most of the remaining budget
    time_left = deadline - time.time()
    if time_left > 0:
        sub_deadline = time.time() + 0.55 * time_left
        print("\n--- Adam‑style sub‑gradient refinement (top_k=30) ---")
        seq_sub, val_sub = _subgradient_adam(
            global_best_seq, n, sub_deadline,
            top_k=30,
            lr=0.20,
        )
        if val_sub < global_best_val:
            global_best_seq, global_best_val = seq_sub, val_sub
            print(f"[AdamSub] new global best = {val_sub:.6f}")

    # ----- Stage 2b – random short restarts (optional) ---------------
    # Use any leftover time to run many cheap restarts.
    while time.time() < deadline - 30.0:   # keep ~30 s for polishing
        remaining = deadline - time.time()
        cand_seq, cand_val = _short_restart_descent(
            n, rng, global_best_seq, remaining,
        )
        if cand_val < global_best_val:
            global_best_seq, global_best_val = cand_seq, cand_val
            print(f"[Restart] improved global best = {cand_val:.6f}")

    # ----- Stage 3 – heavy stochastic polishing -----------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing (aggressive) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=1_200_000,
            init_temp_factor=0.06,
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ----- Final validation --------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for future warm‑starts
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
