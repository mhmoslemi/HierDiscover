"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

Main ideas
----------
* Use a *soft‑max* smoothing of the convolution to obtain a
  differentiable surrogate of the max‑operator.
* Compute the gradient of this smooth surrogate by a single
  FFT‑based correlation, which is O(n log n).
* Perform an entropic (multiplicative) mirror‑descent update with a
  decreasing steps‑size and a temperature schedule that slowly turns
  the soft‑max into the true max.
* Initialise the search from a small pool of analytically‑generated
  seeds (uniform, triangular, geometric, β‑shaped, Gaussian, …) and
  from the previous best sequence if it exists.
* After the deterministic phase a lightweight simulated‑annealing
  polishing is applied.
* The whole routine respects the supplied time budget (default 1000 s)
  and stores the best sequence in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for r in np.concatenate([np.linspace(0.2, 0.98, 12), np.linspace(0.98, 0.9995, 8)]):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))
    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))
    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))
    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Objective helpers (surrogate – the scaling by sum is irrelevant)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate objective for ℓ₁‑normalised sequences (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Gradient based optimiser – soft‑max mirror descent
# ----------------------------------------------------------------------


def _softmax_mirror_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
    beta_start: float = 0.01,
    beta_end: float = 15.0,
) -> tuple[np.ndarray, float]:
    """
    Entropic mirror descent using a soft‑max smoothing of the max‑conv
    term.  The “temperature” beta is increased from ``beta_start`` to
    ``beta_end`` during the run, gradually turning the smooth surrogate
    into the true max.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # 1) convolution a*a
        conv = _conv_fft(a, a)                # length 2n‑1

        # 2) soft‑max weights for the surrogate
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)
        # stabilise exponentials
        scores = beta * conv
        max_score = np.max(scores)
        w = np.exp(scores - max_score)
        w /= w.sum()                           # w has length 2n‑1, sums to 1

        # 3) gradient of the smooth surrogate:
        #    g_i = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]
        corr_full = _conv_fft(a, w_rev)        # length 3n‑2
        # slice indices n‑1 … 2n‑2 (inclusive) → n values
        slice_ = corr_full[n - 1 : 2 * n - 1]
        g = 2.0 * slice_[::-1]                # shape (n,)

        # 4) entropic update and projection onto simplex
        eta = eta0 / np.sqrt(it)              # decreasing step size
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # 5) keep the best so far
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SoftMD] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polishing (re‑used from baseline)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Polish a solution by random Dirichlet moves with a geometric cooling."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)                 # size of the reshuffled block
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start: previous best (if available)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Pick a moderate length – 2000 gives a good speed / quality trade‑off
    # ------------------------------------------------------------------
    n = 2000
    print(f"\n--- Preparing seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # quick evaluation of seeds, keep the top few
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(4, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values from seed pool:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Deterministic optimisation – soft‑max mirror descent
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        # allocate the majority of the remaining budget to the new optimiser
        md_deadline = time.time() + 0.80 * time_left
        print("\n--- Soft‑max mirror descent optimisation ---")
        seq_md, val_md = _softmax_mirror_descent(
            global_best_seq,
            n,
            md_deadline,
            max_iters=250_000,
            eta0=0.7,
            beta_start=0.01,
            beta_end=15.0,
        )
        if val_md < global_best_val:
            global_best_seq, global_best_val = seq_md, val_md
            print(f"[SoftMD] new global best = {val_md:.6f}")

    # ------------------------------------------------------------------
    # Optional second run from the second‑best seed (if time permits)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0 and top_k > 1:
        md_deadline = time.time() + 0.15 * time_left
        print("\n--- Second soft‑max run (alternative seed) ---")
        seq_md2, val_md2 = _softmax_mirror_descent(
            best_seeds[1],
            n,
            md_deadline,
            max_iters=150_000,
            eta0=0.7,
            beta_start=0.01,
            beta_end=15.0,
        )
        if val_md2 < global_best_val:
            global_best_seq, global_best_val = seq_md2, val_md2
            print(f"[SoftMD2] new global best = {val_md2:.6f}")

    # ------------------------------------------------------------------
    # Final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            max_iters=800_000,
            init_temp_factor=0.04,
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (the true target function)
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for warm‑starting future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
