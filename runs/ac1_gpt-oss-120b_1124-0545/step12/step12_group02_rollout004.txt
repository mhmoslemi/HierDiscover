def _hill_climb(seq, n, deadline, max_iters):
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    # compute conv and max index each iteration
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute current conv and max
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        # Identify max index
        k_max = int(np.argmax(conv))
        # Determine overlapping pairs indices
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        # range of indices contributing to max entry
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = k_max - idx_i
        # compute contributions = a[idx_i] * a[rev_idx]
        contrib = a[idx_i] * a[rev_idx]
        # sort indices by contribution descending
        sorted_idx = np.argsort(-contrib)  # descending
        # pick top pair
        top = sorted_idx[0]
        i = idx_i[top]
        j = rev_idx[top]
        # pick the larger of a[i], a[j] as source
        source = i if a[i] >= a[j] else j
        target = j if source == i else i

        # choose a small epsilon fraction of source
        eps = min(a[source] * 0.1, 0.001)  # shape
        if eps <= 0:
            # fallback random move
            source = np.random.randint(n)
            target = np.random.randint(n)
            eps = min(a[source] * 0.1, 0.001)
        # move epsilon from source to target
        new_a = a.copy()
        new_a[source] -= eps
        new_a[target] += eps
        # project onto simplex (though sum unchanged; still non-negative)
        if new_a[source] < -1e-12:
            # if negative due to epsilon > source, skip
            continue
        # could just keep as is because we adjust to keep sum 1.
        # but ensure nonneg:
        if new_a.min() < -1e-12:
            continue
        # No need to project; small move preserves sum and nonneg
        # Evaluate new objective
        new_conv = _conv_fft(new_a, new_a)
        new_val = 2.0 * n * np.max(new_conv)
        # Accept if better
        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_a.copy()
                print(f"[HillClimb] it={it} new best {new_val:.6f}")

        # Occasionally random restart: with prob small, random perturb
        if np.random.rand() < 0.001:

            perturb = _random_dirichlet(n, 0.5, np.random.default_rng())
            a = a * 0.9 + perturb * 0.1
            a = a / a.sum()
    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector that sums to 1) that minimises

    2·n·max(conv(a,a))

where conv is the full linear convolution.  The objective is exactly the
value returned by ``evaluate_sequence`` because the sum is fixed to 1.

The optimiser works in several stages:

1.  Build a diverse pool of analytic windows and random Dirichlet draws,
    keep the best few according to the cheap surrogate (ℓ1‑normalised).

2.  Upsample the current best to a finer grid (default 8000 points).

3.  Run several *log‑space Adam* optimisations with a slowly‑increasing
    temperature (log‑sum‑exp smooth max).  Each run starts from a different
    seed (the best seed, a random Dirichlet, and the previous solution).

4.  Refine with an exact sub‑gradient projector on the true max‑convolution
    (no smoothing).

5.  Refine again with exponentiated‑gradient (mirror‑descent) on the exact
    max‑convolution.

6.  A long simulated‑annealing / hill‑climbing polishing phase that
    repeatedly reshuffles a random subset of the mass, occasionally
    performs tiny pairwise transfers and accepts worsening moves with a
    slowly decreasing temperature.

All stages respect the global time budget (default 1000 s).  The best
vector found is stored in the global name ``height_sequence_1`` and
returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(seq,seq)) for a vector that is ℓ₁‑normalised
    (∑seq = 1).  The returned value is exactly the true ``evaluate_sequence``
    value when the sum equals 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0

         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows as _scipy_windows
        a = _scipy_windows.kaiser(n, beta)
    except Exception:
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        t = beta * np.sqrt(1 - y * y)
        # crude series for I0(t)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15, shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few low‑frequency cosine components, clip to non‑negative and normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian families
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    # Power‑law
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(3):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # normalise everything robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _logspace_adam_optimize_long(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.  Temperature ``beta``
    (used for the log‑sum‑exp smooth max) is increased *very* gradually so
    that the optimisation can wander before it focuses on the true max.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature (smooth max)
    beta_max = 3000.0
    beta_inc = (beta_max - beta) / max_iters   # linear schedule

    lr0 = 1.0                # initial learning rate (log‑space)

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # ℓ₁‑normalised vector
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smoothed max (log‑sum‑exp)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient on logits (log‑softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[Sub] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.30 / np.sqrt(it)          # Polyak‑style step‑size
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (same as in _subgradient_proj)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.45 / np.sqrt(it)          # decreasing step size
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)              # back to simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[Exp] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.08,
    min_temp: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """
    Aggressive annealing that repeatedly reshuffles a random subset of the
    mass, occasionally performs tiny pairwise transfers, and accepts
    worsening moves with a slowly decreasing temperature.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Exponential cooling (very slow)
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.65:          # reshuffle a random subset (size 3‑14)
            k = rng.integers(3, 15)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total > 0.0:
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
                new = cur.copy()
                new[idx] = new_alloc
            else:
                continue
        else:
            # tiny pairwise transfer (≤ 3 % of source mass)
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.03)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid (keeps shape) and ℓ₁‑renormalises."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin for the final call to evaluate_sequence
    overall_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Stage 0 – coarse seed generation (n = 4000)
    # ------------------------------------------------------------------
    n0 = 4000
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print(f"\n--- Seed generation (n={n0}) ---")
    seeds = _collect_seeds_extended(n0, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # cheap surrogate ranking (lower is better)
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample to finer grid (if time permits)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    n_up = n0 * 2                     # 8000 points
    if remaining > 25.0:
        print(f"\n--- Upsampling to n={n_up} ---")
        global_best_seq = _upsample_sequence(global_best_seq, n_up)
        global_best_seq = global_best_seq / global_best_seq.sum()
        global_best_val = _objective_surrogate(global_best_seq)
        print(f"[Upsample] surrogate after interpolation = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – multiple log‑space Adam runs (exploration)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 70.0:
        n_adam_runs = 3
        max_iters_per_run = int(min(200_000, remaining * 3000 / n_adam_runs))
        print(f"\n--- {n_adam_runs} Adam runs (max_iters≈{max_iters_per_run}) ---")
        for run in range(n_adam_runs):
            # pick a diverse start vector
            if run == 0:
                start_vec = global_best_seq.copy()
            elif run == 1:
                start_vec = _random_dirichlet(n_up, 0.5, rng)
            else:
                # perturb the current best a little
                perturb = _random_dirichlet(n_up, 0.8, rng)
                start_vec = 0.9 * global_best_seq + 0.1 * perturb
                start_vec = start_vec / start_vec.sum()

            seq_adam, val_adam = _logspace_adam_optimize_long(
                start_vec, n_up, overall_deadline, max_iters=max_iters_per_run
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Adam‑run{run+1}] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient projection (medium‑range refinement)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 20.0:
        max_iters_sub = int(min(400_000, remaining * 2000))
        print(f"\n--- Sub‑gradient projection (max_iters={max_iters_sub}) ---")
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters_sub
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[Sub] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – exponentiated‑gradient refinement (late‑stage fine‑tuning)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 15.0:
        max_iters_exp = int(min(300_000, remaining * 1500))
        print(f"\n--- Exponentiated‑gradient (max_iters={max_iters_exp}) ---")
        seq_exp, val_exp = _exp_gradient_optimize(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters_exp
        )
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[Exp] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – aggressive simulated‑annealing polishing (long run)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        max_iters_sa = int(min(1_200_000, remaining * 1800))
        print(f"\n--- Simulated annealing polishing (max_iters={max_iters_sa}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters_sa
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – final exact evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
