def _lp_maximize_sum(a, M):
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # for each i in [i_start,i_end], term a[i] * b[k-i] i.e., b[j] coefficient is a[i] where j = k-i
        # thus for each i in that range, we set A[k, j] = a[i] where j = k - i
        # We can equivalently iterate over i.
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints: A @ b <= M (vector)
    # plus b >= 0.
    c = -np.ones(n)  # maximize sum(b) => minimize -sum(b)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    if res.success:
        b = res.x
        sum_b = b.sum()
        return b, sum_b
    else:
        return None, None

"""
Improved step‑function optimizer.

The algorithm consists of four phases:

1. **Massive seed generation** – analytic families (uniform, triangular,
   cosine, β‑shapes, power‑law, exponential, low‑frequency sinusoids) and a
   large number of random Dirichlet draws are ℓ₁‑normalised and the best
   ~10 candidates are kept using the fast surrogate 2·n·max(conv).

2. **Smooth‑max L‑BFGS refinement** – each top seed is optimised with a
   sequence of increasing temperature (log‑sum‑exp) smoothings.  The L‑BFGS
   runs use up to 300 inner iterations, yielding a very accurate
   approximation of the true max‑convolution.

3. **Mirror‑descent (exponentiated‑gradient) sub‑gradient descent** – the
   exact non‑smooth objective is minimised with a multiplicative update on
   the simplex.  The step‑size decays as ηₜ = η₀ / t^0.30 and the loop runs
   for up to a few‑hundred‑thousand iterations (limited by the remaining
   time budget).

4. **Final simulated‑annealing polishing** – a lightweight SA pass (swap,
   Dirichlet‑redistribute, pair‑transfer moves) finishes the search.

All phases respect a hard deadline (default 1000 s).  The best sequence found
is stored globally as ``height_sequence_1`` and returned as a plain list of
floats.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (length = len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple softmax (for log‑sum‑exp parametrisation)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1.0)))
        elif i <= (n - 1) * (1.0 - alpha / 2.0):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2.0 / alpha + 1.0)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinc_squared(n: int) -> np.ndarray:
    """a_i = (sinc(pi*i/n))², ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    # avoid division by zero at i=0
    x = np.where(i == 0, 1.0, np.sin(np.pi * i / n) / (np.pi * i / n))
    a = x ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine_power(n: int, power: float) -> np.ndarray:
    """(0.5·(1‑cos(2πi/(n‑1))))^power, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Surrogate objective (identical ordering to official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max surrogate and gradient (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_max, grad_a) where smooth_max = log‑sum‑exp_beta(conv)/beta
    and grad_a = ∂/∂a smooth_max (ignores the external 2·n factor).
    """
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # gradient of weighted sum = 2 * (rev_a * weights) convolution
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 300,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """L‑BFGS on the smooth‑max surrogate (log‑sum‑exp of the convolution)."""
    eps = 1e-30
    v = np.log(start_seq + eps)          # logits, unconstrained

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        grad_v_scaled = 2.0 * n * grad_v
        return loss, grad_v_scaled.astype(np.float64)

    # SciPy may raise if the optimizer wants more iterations after deadline;
    # we guard against that with a try/except.
    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:  # pragma: no cover
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑conv objective
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 300_000,
                                 lr0: float = 0.12,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with multiplicative updates on the simplex."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple Simulated Annealing for polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                  n: int,
                                  deadline: float,
                                  max_iters: int = 200_000,
                                  init_temp_factor: float = 0.04,
                                  min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)   # 0=swap, 1=redistribute subset, 2=pair‑transfer
        if move_type == 0:               # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:             # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                            # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    `evaluate_sequence` objective within the allocated time budget.
    Returns the best sequence as a list of floats and stores it as the
    global ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final eval

    # ------------------------------------------------------------------
    # Length of the step‑function – a compromise between flexibility and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve any previously found solution for seeding
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 0 – generate a very rich set of candidate seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    top_k = 10                     # number of seeds kept for heavy optimisation
    best_seeds = []                # list of (val, vec) tuples, always sorted
    def _consider(vec):
        nonlocal best_seeds
        v = _objective(vec)
        if len(best_seeds) < top_k:
            best_seeds.append((v, vec))
            best_seeds.sort(key=lambda x: x[0])
        else:
            if v < best_seeds[-1][0]:
                best_seeds[-1] = (v, vec)
                best_seeds.sort(key=lambda x: x[0])

    # deterministic families
    _consider(_uniform(n))
    _consider(_triangular(n))
    _consider(_cosine_window(n))
    _consider(_sinc_squared(n))
    _consider(_tukey_window(n, 0.5))
    _consider(_raised_cosine_power(n, 2.0))

    # parameter sweeps over analytic families
    for gamma in np.linspace(0.3, 4.0, 12):
        _consider(_sine_window(n, gamma=gamma))
    for sigma in np.linspace(0.08, 0.45, 10):
        _consider(_gaussian(n, sigma_frac=sigma))
    for p in np.linspace(0.0, 4.0, 9):
        _consider(_power(n, p))
    for lam in np.linspace(0.0005, 0.015, 8):
        _consider(_exponential_decay(n, lam))
    for power in [0.5, 1.0, 1.5, 2.0, 3.0]:
        _consider(_raised_cosine_power(n, power))

    # random beta shapes
    for _ in range(1500):
        if time.time() > deadline:
            break
        alpha = 10 ** rng.uniform(-2.0, 0.7)   # roughly 0.01 – 5
        beta = 10 ** rng.uniform(-2.0, 0.7)
        _consider(_beta_shape(n, alpha, beta))

    # random power‑law
    for _ in range(1200):
        if time.time() > deadline:
            break
        p = rng.uniform(0.0, 5.0)
        _consider(_power(n, p))

    # random exponential
    for _ in range(800):
        if time.time() > deadline:
            break
        lam = 10 ** rng.uniform(-4.0, -1.0)   # 1e‑4 – 0.1
        _consider(_exponential_decay(n, lam))

    # random low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(200):
            if time.time() > deadline:
                break
            _consider(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws with a spread of concentration parameters
    conc_values = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for conc in conc_values:
        for _ in range(200):
            if time.time() > deadline:
                break
            _consider(_random_dirichlet(n, conc, rng))

    # previous solution, if any
    if previous is not None and previous.size == n:
        prev_norm = previous / previous.sum()
        _consider(prev_norm)

    print(f"Collected {len(best_seeds)} top seeds (surrogate values):")
    for i, (val, _) in enumerate(best_seeds, 1):
        print(f"  #{i}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0][1].copy()
    global_best_val = best_seeds[0][0]
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinement on the smooth‑max surrogate
    # ------------------------------------------------------------------
    temps = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0, 600.0]
    for idx, (seed_val, seed_vec) in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temps:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(seq, n, deadline,
                                                max_iter=300,
                                                temperature=temp)
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"[L‑BFGS] seed #{idx} temp={temp:.0f} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start next temperature

    # ------------------------------------------------------------------
    # Phase 2 – Mirror‑descent sub‑gradient optimisation (longer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline,
            max_iters=300_000,
            lr0=0.15,
            power=0.30,
        )
        if val_mirror < global_best_val:
            global_best_val = val_mirror
            global_best_seq = seq_mirror.copy()
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – A final L‑BFGS pass (high temperature) to polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS polishing ---")
        seq_opt, val_opt = _l_bfgs_optimize(global_best_seq, n, deadline,
                                            max_iter=400,
                                            temperature=800.0)
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Final L‑BFGS] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Light simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=200_000,
            init_temp_factor=0.03,
        )
        if val_sa < global_best_val:
            global_best_val = val_sa
            global_best_seq = seq_sa.copy()
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (informational)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by subsequent runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
