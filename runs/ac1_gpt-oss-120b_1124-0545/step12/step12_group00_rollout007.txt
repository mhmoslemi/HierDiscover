"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.
The search combines

1. a rich collection of analytic and random seeds;
2. a fast top‑k exponentiated‑gradient (mirror‑descent) optimiser on the
   simplex (the update is multiplicative, i.e. a KL‑mirror step);
3. a simulated‑annealing polishing phase;
4. a brief additive‑gradient refinement.

All stages respect a user‑provided time budget (default 1000 s) and
run on a single thread using only NumPy.  The final sequence is stored
in the global ``height_sequence_1`` variable.
"""

import time
import math
import numpy as np

# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself via a real FFT."""
    n = x.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    # element‑wise product (auto‑convolution)
    conv = np.fft.irfft(fx * fx, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective (assumes Σseq = 1):
        2 * n * max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few low‑frequency cosine components, then clip to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families (coarse)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    for rate in np.linspace(0.95, 0.9999, 12):
        seeds.append(_exponential_decay(n, rate))
    for p in np.linspace(0.2, 4.0, 12):
        seeds.append(_power(n, p))
    for a in np.linspace(0.3, 4.0, 6):
        for b in np.linspace(0.3, 4.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # random families
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.25, 0.5, 0.75):
        for nf in (2, 3):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if compatible length)
    if previous is not None:
        prev = np.asarray(previous, dtype=np.float64)
        if prev.size == n and prev.sum() > 0:
            seeds.append(prev / prev.sum())
        else:
            # Resample previous to required length (linear interpolation)
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            prev_res = np.interp(x_new, x_old, prev)
            if prev_res.sum() > 0:
                seeds.append(prev_res / prev_res.sum())

    # safety normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

# Exponentiated‑gradient optimiser (top‑k sub‑gradient)
# ----------------------------------------------------------------------
def _subgradient_topk(x: np.ndarray, n: int, top_shifts: np.ndarray) -> np.ndarray:
    """Aggregate sub‑gradient of the convolution entries listed in *top_shifts*."""
    g = np.zeros_like(x)
    for s in top_shifts:
        i_start = max(0, int(s) - (n - 1))
        i_end = min(n - 1, int(s))
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = int(s) - idx
        g[idx] += 2.0 * x[rev]
    return g


def _expgrad_optimize_topk(start: np.ndarray, n: int, deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 0.30,
                           top_k: int = 3) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the sub‑gradient of the *top_k*
    largest convolution entries.  Multiplicative update (exponential‑gradient)
    with a 1/√t step‑size schedule.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x)
        # indices of the top‑k largest entries (including ties)
        if top_k == 1:
            top_idx = np.array([int(np.argmax(conv))])
        else:
            # argpartition gives an unordered top‑k set; we keep them all
            top_idx = np.argpartition(-conv, top_k - 1)[:top_k]

        g = _subgradient_topk(x, n, top_idx)

        # step size
        eta = eta0 / math.sqrt(it)

        # multiplicative update, clamp exponent for stability
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # simplex projection (normalise)
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[ExpGrad‑topk] it={it:7d} best={best_val:.6f}")

    return best_x, best_val

def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 600_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature scaled to current surrogate value
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.99996, min_temp)

        # two families of moves
        if rng.random() < 0.6:
            # reshuffle a random subset (size 4‑14)
            k = rng.integers(4, 15)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total > 0:
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
                cand = cur.copy()
                cand[idx] = new_alloc
            else:
                continue
        else:
            # small transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.06)   # ≤ 6 % of source mass
            if eps <= 0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 150_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

def _tiny_additive_refine(x: np.ndarray, n: int, steps: int = 5_000,
                          step_size: float = 1e-3) -> np.ndarray:

                                                   eta0=0.35,
                                                   top_k=5)
        print(f"[Long EG] best surrogate after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – simulated‑annealing final tweak.
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1800))
        print("\n--- Simulated annealing polishing ---")
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_iters)
        print(f"[SA] surrogate after SA = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – tiny additive‑gradient clean‑up.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Tiny additive refinement ---")
        best_seq = _tiny_additive_refine(best_seq, n, steps=4_000, step_size=5e-4)
        best_val = _objective_surrogate(best_seq)
        print(f"[Refine] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
