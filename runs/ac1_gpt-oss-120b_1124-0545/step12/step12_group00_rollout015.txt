"""
Optimising a non‑negative step function for the ``evaluate_sequence`` objective.

The optimiser works on the probability simplex (∑a = 1, a ≥ 0) and uses
three complementary phases:

1. **Smoothed exponential‑gradient descent** – we minimise a
   temperature‑smoothed version of the max‑convolution
   ``max(conv(a,a))``.  The smooth surrogate is the log‑sum‑exp of the
   convolution entries, whose gradient can be computed with two FFT‑based
   convolutions.  A mirror‑descent (exponential‑gradient) update keeps the
   iterate inside the simplex.

2. **Sharp (sub‑gradient) exponential‑gradient refinement** – once the
   temperature is very low the smooth surrogate coincides with the true
   max, so we finish with the classic sub‑gradient EG used in the
   reference implementation.

3. **Simulated‑annealing polishing** – a cheap SA that reshuffles mass
   on random subsets to escape tiny local traps.

All stages respect a global time budget (default 1000 s).  The algorithm
starts from a diverse pool of analytic and random seeds (including the
previous best, if any) and keeps the best sequence seen.  The final
candidate is stored in the global ``height_sequence_1`` variable.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate: 2·n·max(conv(seq, seq)) (assumes Σseq = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    """Monotonically decreasing exponential, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    # Beta family (coarse grid)
    for a_ in (0.5, 1.0, 2.0, 3.0):
        for b_ in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a_, b_))
    # Tukey windows
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise all (safety)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _smooth_expgrad(start: np.ndarray, n: int, deadline: float,
                    max_iters: int, eta0: float, tau: float) -> tuple[np.ndarray, float]:
    """
    Minimise a temperature‑smoothed version of max(conv(a,a)).
    ``tau`` is the temperature (larger → smoother).
    Uses a mirror‑descent (exponential‑gradient) update.
    Returns the best vector seen and its surrogate value.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # 1. Full self‑convolution
        conv = _conv_fft(x, x)                     # length 2n‑1

        # 2. Soft‑max weights (temperature tau)
        maxc = conv.max()
        # stable softmax: shift by maxc
        exps = np.exp((conv - maxc) / tau)
        w = exps / exps.sum()                       # still length 2n‑1

        # 3. Gradient of the weighted sum of convolutions
        w_rev = w[::-1]
        conv_g = _conv_fft(x, w_rev)               # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)           # indices of interest
        g_weighted = 2.0 * conv_g[idx]             # ∇ (∑ w·conv)
        grad = 2.0 * n * g_weighted                # ∇ J where J = 2n·∑w·conv

        # 4. Exponential (mirror‑descent) update
        eta = eta0 / math.sqrt(it)
        delta = -eta * grad
        # clip to avoid overflow/underflow in exp
        delta = np.clip(delta, -50.0, 0.0)
        x = x * np.exp(delta)

        # 5. Projection onto the simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
        if it % 150000 == 0:
            print(f"[SmoothEG] it={it:7d}  tau={tau:.5f}  best={best_val:.6f}")

    return best_x, best_val

# Classic sub‑gradient exponential‑gradient optimiser (as in the reference)
# ----------------------------------------------------------------------
def _maxsubgrad_expgrad(start: np.ndarray, n: int, deadline: float,
                       max_iters: int = 400_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the original max‑convolution objective.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
        if it % 100000 == 0:
            print(f"[MaxEG] it={it:7d} best={best_val:.6f}")

    return best_x, best_val

# Simulated‑annealing polish (same as in the reference)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # geometric cooling
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            # reshuffle a random subset (size 3‑12)
            ksize = rng.integers(3, 13)
            idx = rng.choice(n, size=ksize, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(ksize, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length of the vector – 4000 is a good trade‑off between quality
    # and speed.  Feel free to change this value if you have extra time.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list):
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # ------------------------------------------------------------------
    # Keep a handful of the best seeds (according to the surrogate)
    # ------------------------------------------------------------------
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Optimisation loop over the selected seeds
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # Continuation schedule (temperature → 0)
    temp_schedule = [
        (0.020, 0.30, 200_000),   # relatively smooth
        (0.006, 0.25, 250_000),   # medium
        (0.002, 0.20, 250_000),   # near‑max
    ]

    for seed_idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        cur = seed_vec.copy()
        cur_val = _objective_surrogate(cur)

        # Run the three temperature phases
        for tau, eta0, max_it in temp_schedule:
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            # Rough estimate of how many iterations we can afford:
            it_per_sec = 1800   # empirically ~1800 it/s for n≈4000
            max_it_adj = int(min(max_it, remaining * it_per_sec))
            cur, cur_val = _smooth_expgrad(cur, n, deadline,
                                           max_iters=max_it_adj,
                                           eta0=eta0, tau=tau)
            if cur_val < best_val:
                best_val, best_seq = cur_val, cur.copy()
                print(f"[Seed {seed_idx+1}] improved to {best_val:.6f} (tau={tau})")

        # Final sharp refinement using the classic sub‑gradient EG
        if time.time() < deadline:
            remaining = deadline - time.time()
            it_per_sec = 1500
            max_it_adj = int(min(300_000, remaining * it_per_sec))
            cur, cur_val = _maxsubgrad_expgrad(cur, n, deadline,
                                               max_iters=max_it_adj,
                                               eta0=0.15)
            if cur_val < best_val:
                best_val, best_seq = cur_val, cur.copy()
                print(f"[Seed {seed_idx+1}] post‑sharp improvement to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polish (optional – cheap but can trim a few 1e‑4)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        it_per_sec = 1800
        max_it_adj = int(min(300_000, remaining * it_per_sec))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,

        print(f"[SA] final surrogate = {best_val:.6f}")

    try:

        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
