"""
Propose candidate step‑function.

This version builds a richer seed pool (more Dirichlet draws,
extra low‑frequency curves and several deterministic families),
optimises the best seeds with a long Adam run on a log‑sum‑exp
smoothed max‑convolution, refines the result with a multi‑index
mirror‑descent sub‑gradient method, and finishes with an aggressive
simulated‑annealing phase that works on both swaps and mass‑transfer
moves.  All phases respect a global 1000 s budget and the final
sequence is stored in the global name ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when the sequence
    is ℓ₁‑normalised (sum≈1):
        2·n·max(conv(seq,seq)) / (sum(seq)²)
    """
    n = len(seq)
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    return a / a.sum()


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    return a / a.sum()

# Ordering helpers
# ----------------------------------------------------------------------
def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    return a / a.sum()


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    return a / a.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    return out / out.sum()


# ----------------------------------------------------------------------
# Misc random smooth curves
# ----------------------------------------------------------------------
def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Return a list of ℓ₁‑normalised candidate sequences.
    A fairly large, diverse pool is assembled.
    """
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # sine window sweep
    for gamma in np.linspace(0.5, 8.0, 14):
        w = _sine_window(n, gamma)
        seeds.append(w)
        seeds.append(_alternate_order(w))
        seeds.append(_sorted_desc(w))
        seeds.append(_sorted_asc(w))

    # Gaussian sweep
    for sigma in np.linspace(0.08, 0.60, 12):
        g = _gaussian(n, sigma)
        seeds.append(g)
        seeds.append(_alternate_order(g))

    # Power‑law sweep
    for p in np.linspace(0.0, 10.0, 20):
        pw = _power(n, p)
        seeds.append(pw)
        seeds.append(_alternate_order(pw))

    # Beta family random draws
    for _ in range(30):
        alpha = 10 ** rng.uniform(-2, 0.7)
        beta = 10 ** rng.uniform(-2, 0.7)
        b = _beta_shape(n, alpha, beta)
        seeds.append(b)
        seeds.append(_alternate_order(b))

    # Exponential sweep
    for rate in np.logspace(-4, -0.7, 14):
        e = _exponential(n, rate)
        seeds.append(e)
        seeds.append(_alternate_order(e))

    # Logarithmic sweep
    for eps in np.logspace(-4, -0.5, 10):
        lg = _logarithmic(n, eps)
        seeds.append(lg)
        seeds.append(_alternate_order(lg))

    # Dirichlet draws – many concentrations & orderings
    concs = [0.005, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        for _ in range(10):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))
            # a couple of random permutations of the same multiset
            perm = rng.permutation(n)
            seeds.append(raw[perm])

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(6):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(curve)
            seeds.append(_alternate_order(curve))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.4, 0.6, 0.8):
        for nf in (2, 3, 4):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = _mix_shapes(_uniform(n), curve, mix_alpha)
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Random permutations of a few good deterministic shapes
    for base in [_gaussian(n, 0.2), _power(n, 3.0), _beta_shape(n, 0.5, 0.5)]:
        for _ in range(8):
            seeds.append(rng.permutation(base))

    # Previous solution, if any
    if previous is not None and previous.shape == (n,) and previous.sum() > 0.0:
        avg = np.asarray(previous, dtype=np.float64)
        seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 10_000_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on log‑softmax parameters using a log‑sum‑exp surrogate.
    Returns the best ℓ₁‑normalised vector observed before the deadline.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.9

    # Temperature schedule for the smoothed max
    beta = 0.5
    beta_inc = 0.2
    beta_max = 500.0

    best_seq = None
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max() / (a.sum() * a.sum())
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 4000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length ≈ 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # Log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature slowly
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent.
    Uses a multi‑index sub‑gradient (all points within a small
    relative tolerance of the current maximum) to obtain smoother
    updates.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.25  # base learning rate
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # pick all indices whose value is within 0.05% of the max
        tol = max_val * 0.0005
        active = np.where(conv >= max_val - tol)[0]

        # aggregate sub‑gradient over all active indices
        g = np.zeros_like(a)
        for idx in active:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = idx - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        if active.size > 1:
            g = g / active.size

        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # occasional re‑ordering attempt (alternating often helps)
        if it % 100_000 == 0:
            alt = _alternate_order(a)
            alt_val = _objective(alt)
            if alt_val < cur_val:
                a = alt
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {alt_val:.6f}")

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 10_000_000,
                                 init_temp_factor: float = 0.2,
                                 cooling: float = 0.9999,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with a richer move set:
        0 – swap two entries
        1 – redistribute a random subset (Dirichlet)
        2 – pair‑transfer
        3 – reverse a random sub‑array
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * cooling, min_temp)

        move_type = rng.integers(0, 4)
        if move_type == 0:                     # swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute subset
            k = rng.integers(3, 12)            # subset size 3‑11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        elif move_type == 2:                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:                                  # reverse sub‑array
            length = rng.integers(2, n // 4)
            start = rng.integers(0, n - length + 1)
            new = cur.copy()
            new[start:start + length] = new[start:start + length][::-1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 80_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _greedy_permutation_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_swaps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that swaps two entries only if the objective improves.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_val = _objective(a)

    for it in range(max_swaps):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        a[i], a[j] = a[j], a[i]                # try the swap
        val = _objective(a)
        if val < best_val:
            best_val = val
            # keep the swap
        else:
            a[i], a[j] = a[j], a[i]            # revert

    return a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence (list of floats) and stores it in the
    global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0               # safety margin

    # ------------------------------------------------------------------
    # Length – 8000 gives more degrees of freedom while keeping FFT cheap.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Retrieve previous solution if it exists.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 0 – seed generation & quick screening
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Evaluate each seed (exact objective) and keep the best ones
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:               # ensure we have some time left
        adam_budget = remaining * 0.30
        adam_deadline = time.time() + adam_budget
        print("\n--- Adam optimisation phase ---")
        # run Adam on the top three seeds (more budget per seed)
        seeds_to_opt = best_seeds[:3]
        for idx, seed_vec in enumerate(seeds_to_opt, 1):
            if time.time() > adam_deadline:
                break
            per_deadline = time.time() + (adam_deadline - time.time()) / max(1, len(seeds_to_opt) - idx + 1)
            print(f"\n[Adam] run {idx}/{len(seeds_to_opt)}")
            seq_adam, val_adam = _adam_optimize(seed_vec, n, per_deadline)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[overall‑Adam] new global best = {val_adam:.6f}")

            # try an alternating order of the Adam result as well
            if seq_adam is not None:
                alt = _alternate_order(seq_adam)
                alt_val = _objective(alt)
                if alt_val < global_best_val:
                    global_best_seq = alt
                    global_best_val = alt_val
                    print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Mirror‑descent sub‑gradient optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing polishing (high temperature)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            init_temp_factor=0.3, cooling=0.99995
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Greedy permutation hill‑climb (swap‑only)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Greedy permutation optimisation ---")
        seq_perm, val_perm = _greedy_permutation_optimize(
            global_best_seq, n, deadline
        )
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[overall‑Perm] new global best = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
