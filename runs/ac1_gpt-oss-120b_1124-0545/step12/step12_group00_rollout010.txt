def _adam_optimize(start, n, deadline, max_iters, gamma_initial, gamma_final, lr=0.01):
    a = start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # smoothing parameter
        gamma = gamma_initial + (gamma_final - gamma_initial) * (t-1) / (max_iters-1)
        conv = _conv_fft(a, a)
        # compute weights w
        # we can compute exp(gamma*conv - max) for stability
        mx = np.max(gamma * conv)
        exp_weights = np.exp(gamma * conv - mx)
        w = exp_weights / exp_weights.sum()
        # gradient = 2 * conv(w,a) slice(n-1:n-1+n)
        conv_wa = _conv_fft(w, a)
        grad = 2 * conv_wa[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex (non-negative sum=1)
        a = _proj_simplex(a)
        # track best surrogate (the real max)
        cur_max = np.max(_conv_fft(a, a))
        cur_val = 2 * n * cur_max  # since sum=1
        if cur_val < best_val:
            best_val = cur_val
            best_vec = a.copy()
            # print progress
    return best_vec, best_val
```

"""
Propose a non‑negative step function that minimises the provided
evaluate_sequence() objective.

Algorithm
---------
1. **Seed pool** – Generate a rich collection of analytic windows,
   random Dirichlet draws, low‑frequency random curves and (if present)
   the previous best solution.  All seeds are ℓ₁‑normalised (sum = 1)
   and the 8 best according to the surrogate 2·n·max(conv) are kept.

2. **Smooth Adam optimisation** – For each good seed we run a limited
   projected Adam optimisation on a smooth surrogate of the max
   convolution (log‑sum‑exp).  The smoothing parameter `γ` grows from a
   modest value (≈5) to a larger one (≈80) while Adam adapts the step
   size.  The iterates are projected onto the probability simplex.

3. **Long‑run Adam polishing** – The current best vector is refined for
   the remaining time with a longer Adam run (more iterations,
   stronger smoothing).

4. **Exponential‑gradient polishing** – A short mirror‑descent
   (exponential‑gradient) pass is applied as a final “exact” optimiser
   that directly reduces the true max‑convolution value.

5. **Final evaluation** – The best vector found is passed to
   evaluate_sequence() and stored globally as ``height_sequence_1``.
   The plain Python list is returned.

All heavy work is performed with FFT‑based convolutions (O(n log n)),
which makes it feasible to run several hundred thousand iterations on
n≈4000 within the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (efficient for large n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) (requires Σseq = 1)."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    # beta family (coarse grid)
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    # tukey windows
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution, if available
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final safety normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned

# Exponential‑gradient optimiser (existing implementation)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (exponential‑gradient)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow
        x = x * np.exp(delta)

        # simplex projection (Euclidean)
        s = x.sum()
        x = _proj_simplex(x) if s > 0 else _uniform(n)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Smooth Adam optimiser (log‑sum‑exp smoothing of max)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   gamma0: float = 5.0,
                   gamma1: float = 30.0,
                   lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate
        f_γ(a) = (1/γ)·log Σ exp(γ·conv(a,a))
    The iterates are projected onto the probability simplex.
    Returns the best vector seen and its *true* surrogate value.
    """
    a = start.copy()
    a = a / np.sum(a)                     # ensure ℓ₁ = 1
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear schedule for the smoothing parameter
        gamma = gamma0 + (gamma1 - gamma0) * (t - 1) / (max_iters - 1)

        # 1) compute convolution a*a  (needed for the softmax weights)
        conv_aa = _conv_fft(a, a)

        # 2) soft‑max weights over the convolution entries
        scaled = gamma * conv_aa
        mx = np.max(scaled)
        exp_w = np.exp(scaled - mx)
        w = exp_w / exp_w.sum()               # shape (2n‑1,)

        # 3) gradient = 2·(w * a) (restricted to the n‑dimensional slice)
        conv_wa = _conv_fft(w, a)
        grad = 2.0 * conv_wa[n - 1:n - 1 + n]

        # 4) Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 5) projection onto the simplex
        a = _proj_simplex(a)

        # occasional evaluation of the true (non‑smoothed) objective
        if t % 5000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Adam] iter={t:7d} best={best_val:.6f}")

    # final check
    final_val = _objective_surrogate(best_a)
    return best_a, final_val


# ----------------------------------------------------------------------
# Simulated‑annealing (lightweight version for final polish)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 250_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            # reshape a random small subset
            k = rng.integers(3, 13)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two coordinates
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # --------------------------------------------------------------
    # Length selection – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Stage 0 – seed collection (including the previous best, if any)
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # evaluate surrogate and keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = min(8, len(scored))
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – quick Adam refinements on the top seeds
    # --------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest number of iterations proportionally to remaining time
        max_iters = int(min(40_000, remaining * 1300))
        cand_seq, cand_val = _adam_optimize(seed_vec, n, deadline,
                                            max_iters=max_iters,
                                            gamma0=5.0, gamma1=30.0,
                                            lr=0.05)
        if cand_val < best_val:
            best_val, best_seq = cand_val, cand_seq
            print(f"[Quick Adam] seed #{idx+1} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – long‑run Adam polishing on the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long‑run Adam polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1400))
        best_seq, best_val = _adam_optimize(best_seq, n, deadline,
                                            max_iters=max_iters,
                                            gamma0=10.0, gamma1=80.0,
                                            lr=0.05)
        print(f"[Long Adam] best surrogate after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – exponential‑gradient fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                               max_iters=max_iters,
                                               eta0=0.25)
        print(f"[ExpGrad] best surrogate after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – optional simulated‑annealing tweak (light)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_iters)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
