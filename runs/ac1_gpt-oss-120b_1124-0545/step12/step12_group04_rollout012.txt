"""
search_step_function.py

A self‑contained optimiser that looks for a non‑negative step‑function
(sequence of heights) minimising the official `evaluate_sequence`
objective.  The method expands the original multi‑stage pipeline:

1. **Very rich seed pool** – analytic families (including new
   exponential·power hybrids, extended β‑shapes) and many random
   Dirichlet draws are generated and immediately evaluated with a fast
   FFT‑based surrogate (`2·n·max(conv)`).

2. **Aggressive L‑BFGS refinement** – the best ~30 seeds are polished
   using L‑BFGS on a smooth‑max (log‑sum‑exp) surrogate, exploring a
   longer temperature schedule and more iterations than the reference
   implementation.

3. **Long mirror‑descent (exponentiated‑gradient)** – exact sub‑gradient
   descent on the true max‑convolution with a higher iteration budget
   and a slightly larger learning‑rate schedule.

4. **Heavy‑weight simulated‑annealing polishing** – a geometric‑cooling
   SA pass with three elementary move types is run for a large number
   of steps.

All stages respect the global deadline (default 1000 s).  The best
sequence found is stored in the global name ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import heapq
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Softmax utilities (stable)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_power_combo(n: int, lam: float, p: float) -> np.ndarray:
    """exp(-lam*i) * (i+1)^(-p)  (ℓ₁‑normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) * np.maximum(i + 1.0, 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1.0)))
        elif i <= (n - 1) * (1.0 - alpha / 2.0):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2.0 / alpha + 1.0)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinc_squared(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = np.where(i == 0, 1.0, np.sin(np.pi * i / n) / (np.pi * i / n))
    a = x ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine_power(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Surrogate objective (identical ordering to official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max helper (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_max, grad_a) where smooth_max = log‑sum‑exp_beta(conv)/beta
    and grad_a = ∂/∂a smooth_max (ignoring the external 2·n factor).
    """
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # gradient of weighted sum: 2 * (rev_a *weights) convolution
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 300,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """
    L‑BFGS on the smooth‑max surrogate.  Returns the refined vector
    and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_seq + eps)   # logits – unconstrained

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        grad_v_scaled = 2.0 * n * grad_v
        return loss, grad_v_scaled.astype(np.float64)

    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:   # pragma: no cover
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑conv objective
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000,
                                 lr0: float = 0.15,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with multiplicative updates on the simplex."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:8d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # project back onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated Annealing polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 init_temp_factor: float = 0.06,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        move_type = rng.integers(0, 3)   # 0=swap, 1=redistribute subset, 2=pair‑transfer
        if move_type == 0:                  # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                               # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:8d} best={new_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    `evaluate_sequence` objective within the allocated time budget.
    Returns the best sequence as a list of floats and stores it as the
    global ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # -------------------------------
    # Length of the step‑function
    # -------------------------------
    n = 4000

    # -------------------------------
    # Bring forward any previously found solution
    # -------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # -------------------------------
    # Stage 0 – generate a huge, diverse seed pool
    # -------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    top_k = 30                      # keep the best `top_k` seeds
    heap = []                       # max‑heap (store -value to keep smallest)

    def _push_candidate(vec):
        """Evaluate `vec` (already ℓ₁‑normalised) and keep it if among top_k."""
        val = _objective(vec)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, vec))
        else:
            if -val > heap[0][0]:      # larger (worse) values are stored as negative
                heapq.heapreplace(heap, (-val, vec))

    # deterministic analytic families
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    _push_candidate(_cosine_window(n))
    _push_candidate(_sinc_squared(n))
    _push_candidate(_gaussian(n, sigma_frac=0.2))
    _push_candidate(_raised_cosine_power(n, power=1.0))
    _push_candidate(_tukey_window(n, alpha=0.5))

    # sine windows with a range of gammas
    for gamma in np.linspace(0.3, 4.0, 10):
        _push_candidate(_sine_window(n, gamma=gamma))

    # exponential family (fine sweep around the promising region)
    for lam in np.linspace(0.00025, 0.0012, 12):
        _push_candidate(_exponential_decay(n, lam))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 17):
        _push_candidate(_power(n, p))

    # beta‑shape grid – include very asymmetric cases
    alphas = np.concatenate((
        np.linspace(0.05, 0.4, 8),
        np.linspace(0.5, 5.0, 10)
    ))
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            _push_candidate(_beta_shape(n, a, b))

    # exponential × power hybrids (new family)
    for lam in np.linspace(0.00025, 0.0012, 8):
        for p in np.linspace(0.0, 3.0, 7):
            _push_candidate(_exp_power_combo(n, lam, p))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            _push_candidate(_lowfreq_curve(n, nf, rng))

    # random Dirichlet draws (various concentration)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            raw = _random_dirichlet(n, conc, rng)
            _push_candidate(raw)                     # raw ordering
            _push_candidate(np.sort(raw)[::-1])      # descending
            _push_candidate(np.sort(raw))            # ascending

    # include the previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            _push_candidate(avg / avg.sum())

    # extract the best seeds (sorted by true surrogate value, low → high)
    heap.sort(key=lambda x: -x[0])          # now smallest (most negative) first
    best_seeds = [vec for _, vec in heap[:top_k]]
    print(f"Generated >{len(heap)} raw candidates, kept top {top_k}.")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on smooth‑max surrogate
    # --------------------------------------------------------------
    temps = [10.0, 30.0, 100.0, 300.0, 600.0]   # extended temperature ladder
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temps:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline, max_iter=250, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp:.0f} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start the next temperature

    # --------------------------------------------------------------
    # Stage 2 – Mirror‑descent (exact max) fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=5_000_000,
            lr0=0.15,
            power=0.30,
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            init_temp_factor=0.06,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (informational)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by subsequent runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
