def _adam_optimize(start, n, deadline, max_iters, lr, beta1, beta2, eps, lambda_start, lambda_factor, lambda_max, verbose=False):
    x = start.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    lam = lambda_start
    best_x = x.copy()
    # compute initial objective:
    conv = _conv_fft(x, x)
    best_val = 2 * n * np.max(conv)
    # Also best surrogate via actual value.

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)  # full conv
        # compute softmax weights
        w, smooth_max = _softmax_weights(conv, lam)
        # compute gradient of smooth_max
        grad = _grad_smooth_max(x, w)  # shape n

        # update moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        # bias correction
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        # parameter update
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # project onto simplex
        # may have negative values due to step; projection ensures nonneg.
        x = _proj_simplex(x)

        # update lambda
        if lam < lambda_max:
            lam = min(lambda_max, lam * lambda_factor)

        # periodic evaluation
        if it % 1000 == 0 or it == max_iters:
            max_conv = np.max(conv)
            cur_val = 2.0 * n * max_conv
            if cur_val < best_val:
                best_val = cur_val
                best_x = x.copy()
                if verbose:
                    print(f"[Adam] it={it:7d} best={best_val:.6f}")
        # optional: random perturbation
        # For now skip random.

    return best_x, best_val

"""
Propose a non‑negative step function that minimises the “evaluate_sequence”
objective.

Algorithm
---------
1. **Initial pool** – use the previous solution (if any), a uniform
   distribution and a few random Dirichlet draws (different concentration).
   All vectors are ℓ₁‑normalised.

2. **Smooth‑max Adam optimisation** – for each seed we run a projected Adam
   optimiser on the *smooth* approximation of the maximal convolution entry:
        L(x) = (1/λ)·log Σₖ exp(λ·(x⋆x)[k])
   The gradient of L w.r.t. the vector x is
        ∇L = 2·conv( w , rev(x) )
   where wₖ = softmaxₖ(λ·(x⋆x)[k]) and rev(x) is x reversed.
   After every Adam step we project back onto the probability simplex.
   The smoothing parameter λ is increased slowly (λ←min(λ·1.001, λ_max))
   so that the optimisation first sees a smooth landscape and later focuses
   on the true maximum.

3. **Best‑of‑restarts** – each restart keeps the best vector seen (according to
   the true surrogate 2·n·max(x⋆x)).  The overall best across all restarts is
   returned.

4. **Final evaluation** – the best vector is passed to the official
   ``evaluate_sequence`` function for the true score and stored globally as
   ``height_sequence_1``.

The whole procedure respects the 1000 s deadline, uses only NumPy and runs on a
single process (compatible with the 2‑CPU limit).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = a.shape[0] + b.shape[0] - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx_i = 1, x_i ≥ 0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple analytic seeds
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max utilities for the smooth maximum
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, lam: float):
    """
    Return the soft‑max weights wₖ = exp(λ·cₖ)/ Σ exp(λ·c) and the smooth max
    value (optional, not used in the optimiser).
    """
    m = np.max(conv)
    # shift to avoid overflow
    exps = np.exp(lam * (conv - m))
    sum_exps = np.sum(exps)
    w = exps / sum_exps
    smooth_max = m + np.log(sum_exps) / lam
    return w, smooth_max


def _grad_smooth_max(x: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max of the convolution w.r.t. x.
    Implements  ∇L = 2·conv(w, rev(x)) restricted to the slice that matches x.
    """
    rev_x = x[::-1]
    conv = _conv_fft(w, rev_x)          # length = len(w)+len(x)-1 = 3n-2
    start = len(x) - 1                  # first index that corresponds to shift 0
    grad = 2.0 * conv[start:start + len(x)]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smooth maximum
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr: float = 0.1,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   lambda_start: float = 10.0,
                   lambda_factor: float = 1.001,
                   lambda_max: float = 500.0,
                   verbose: bool = False):
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector (ℓ₁‑normalised) seen and its true surrogate value
    2·n·max(conv).
    """
    x = start.copy()
    # initialise Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    lam = lambda_start
    best_x = x.copy()
    # initial surrogate (true objective)
    conv = _conv_fft(x, x)
    best_val = 2.0 * n * np.max(conv)

    for it in range(1, max_iters + 1):
        # time guard – cheap check every 100 iterations
        if (it & 0x3F) == 0 and time.time() > deadline:
            break

        # full self‑convolution
        conv = _conv_fft(x, x)

        # soft‑max weights for the current smoothing
        w, _ = _softmax_weights(conv, lam)

        # gradient of the smooth max
        g = _grad_smooth_max(x, w)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # onto simplex
        x = _proj_simplex(x)

        # slowly increase the smoothing parameter
        if lam < lambda_max:
            lam = min(lambda_max, lam * lambda_factor)

        # periodically evaluate the true surrogate and keep the best
        if it % 1000 == 0 or it == max_iters:
            cur_val = 2.0 * n * np.max(conv)
            if cur_val < best_val:
                best_val = cur_val
                best_x = x.copy()
                if verbose:
                    print(f"[Adam] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low ``evaluate_sequence``
    score.  The routine respects the ``budget_s`` time limit (default 1000 s).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between expressiveness
    # and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    seeds = []

    # previous solution, if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # deterministic simple families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # a few random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # make sure all are normalised
    seeds = [s / np.sum(s) for s in seeds if s.sum() > 0]

    # ------------------------------------------------------------------
    # Optimisation – run a short Adam run for each seed, keep the best.
    # ------------------------------------------------------------------
    best_seq = None
    best_obj = float("inf")
    restart_idx = 0

    for seed_vec in seeds:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # Rough estimate: ~6000 iterations per second on a modest machine.
        est_iters = int(min(200_000, remaining * 6000))
        if est_iters < 1_000:
            break
        restart_idx += 1
        print(f"\n[Restart {restart_idx}/{len(seeds)}] allocated iterations: {est_iters}")

        candidate, cand_val = _adam_optimize(seed_vec,
                                             n,
                                             deadline,
                                             max_iters=est_iters,
                                             lr=0.10,
                                             beta1=0.9,
                                             beta2=0.999,
                                             eps=1e-8,
                                             lambda_start=10.0,
                                             lambda_factor=1.001,
                                             lambda_max=500.0,
                                             verbose=True)

        if cand_val < best_obj:
            best_obj = cand_val
            best_seq = candidate.copy()
            print(f"[Restart {restart_idx}] NEW BEST surrogate = {best_obj:.6f}")

    # fallback – uniform if nothing was produced
    if best_seq is None:
        best_seq = _uniform(n)

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate (≈) = {best_obj:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in future calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
