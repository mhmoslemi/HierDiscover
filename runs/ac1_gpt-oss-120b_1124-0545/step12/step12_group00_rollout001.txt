        max_val = conv.max()

"""
Improved search for a non‑negative step function that minimises
the ``evaluate_sequence`` objective.

Key ideas
---------
1. **Richer seed pool** – besides the deterministic families used
   before we draw many random instances of the same families with
   continuously sampled parameters (beta, gaussian, sine‑window,
   t‑ukey, power‑law, exponential‑decay) and a few extra Dirichlet
   draws.  The top‑k seeds (by the surrogate 2·n·max(conv)) are kept.

2. **Multi‑active‑index exponentiated‑gradient** – the original EG
   used only the sub‑gradient at the single maximal convolution entry.
   Here we collect all indices whose convolution value is within a
   small relative tolerance of the maximum (default 0.1 %) and
   average their sub‑gradients.  This drives down the whole peak
   region much faster.

3. **Two‑phase EG** – a short “quick” EG run on each top seed with a
   larger step‑size, followed by a long polishing run on the best
   candidate using the multi‑active gradient and a decreasing step‑size.

4. **Aggressive simulated‑annealing polish** – after EG we reshuffle
   small subsets and perform tiny mass transfers, accepting any move
   that improves the surrogate (or with a small probability otherwise).

5. **Length fallback** – the algorithm first works with n=4000 (as
   before).  If enough time remains it repeats the whole pipeline for
   n=3000 and keeps the better of the two results.

All stages respect the global deadline (default 1000 s).  Only NumPy
and the provided ``evaluate_sequence`` are used; the code runs on a
single process and is safe for the competition sandbox.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective: 2·n·max(conv(seq, seq))  (assumes Σseq = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    """Monotonically decreasing exponential, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (deterministic + random parametric)

def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None,
                            extra_parametric: int = 120) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families (same as before)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet / exponential draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Extra parametric random draws
    for _ in range(extra_parametric):
        choice = rng.integers(0, 6)
        if choice == 0:  # sine window with random gamma
            gamma = rng.uniform(0.5, 5.0)
            seeds.append(_sine_window(n, gamma))
        elif choice == 1:  # Gaussian with random sigma_frac
            sigma_frac = rng.uniform(0.05, 0.5)
            seeds.append(_gaussian(n, sigma_frac))
        elif choice == 2:  # Exponential decay with random rate
            rate = rng.uniform(0.9, 0.99999)
            seeds.append(_exponential_decay(n, rate))
        elif choice == 3:  # Power‑law with random exponent
            p = rng.uniform(0.1, 5.0)
            seeds.append(_power(n, p))
        elif choice == 4:  # Beta shape with random alphas
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            seeds.append(_beta_shape(n, a, b))
        else:  # Tukey window with random alpha
            a = rng.uniform(0.0, 1.0)
            seeds.append(_tukey_window(n, a))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise all (safety)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (single‑active‑index)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with KL divergence (exponential‑gradient) using a
    single active index (the current maximal convolution entry).
    Returns the best vector seen and its surrogate value.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient at the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # project onto simplex
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val

# Exponentiated‑gradient optimiser (multi‑active‑index)
# ----------------------------------------------------------------------
def _expgrad_optimize_multi(start: np.ndarray, n: int, deadline: float,
                            max_iters: int = 600_000,
                            eta0: float = 0.25,
                            active_frac: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Same as _expgrad_optimize but averages sub‑gradients over all
    convolution entries whose value is within `active_frac` of the
    current maximum.  This smoothes the descent and reduces the peak
    more aggressively.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        max_val = conv.max()
        # indices close to the maximum (relative tolerance)
        thresh = max_val * (1.0 - active_frac)
        active_idxs = np.where(conv >= thresh)[0]

        # accumulate gradient contributions
        g = np.zeros_like(x)
        for k in active_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] += 2.0 * x[rev]

        if active_idxs.size > 1:
            g /= float(active_idxs.size)

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # simplex projection
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 100000 == 0:
                print(f"[ExpGrad‑Multi] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (unchanged, but a bit more aggressive)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 600_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # geometric cooling
        T = max(T * 0.99995, min_temp)

        # move type
        if rng.random() < 0.6:
            # reshuffle a random subset (size 3‑12)
            k = rng.integers(3, 13)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # small transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)  # at most 7 % of source mass
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence (list of floats) found.
    """
    # Seed the RNG
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # We'll try two lengths (4000 and a fallback 3000) and keep the best.
    candidate_lengths = [4000, 3000]

    best_overall_seq = None
    best_overall_val = float('inf')
    best_overall_n = None

    for n in candidate_lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # ------------------------------------------------------------------
        # 1) Seed collection (rich)
        # ------------------------------------------------------------------
        previous = globals().get("height_sequence_1")
        prev_arr = None
        if previous is not None:
            try:
                prev_arr = np.asarray(previous, dtype=np.float64)
            except Exception:
                prev_arr = None

        seeds = _collect_seeds_extended(n, rng, prev_arr, extra_parametric=120)
        print(f"Generated {len(seeds)} seeds.")

        # Score seeds with surrogate (fast) and keep the top few.
        scored = [(s, _objective_surrogate(s)) for s in seeds]
        scored.sort(key=lambda x: x[1])
        top_k = 12
        best_seeds = [vec for vec, _ in scored[:top_k]]

        print(f"Top {top_k} surrogate values after seeding:")
        for i, (_, val) in enumerate(scored[:top_k]):
            print(f"  #{i+1:2d}: {val:.6f}")

        # ------------------------------------------------------------------
        # 2) Quick EG on each top seed (few thousand iterations)
        # ------------------------------------------------------------------
        best_seq = best_seeds[0].copy()
        best_val = _objective_surrogate(best_seq)
        print(f"[Init] best surrogate = {best_val:.6f}")

        quick_iters = 70_000
        for idx, seed_vec in enumerate(best_seeds[:5]):
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            iters = int(min(quick_iters, remaining * 2000))
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=iters, eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) Long‑run multi‑active‑index EG polishing
        # ------------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            long_iters = int(min(800_000, remaining * 1500))
            print(f"[Long EG] polishing with up to {long_iters} iterations")
            best_seq, best_val = _expgrad_optimize_multi(
                best_seq, n, deadline, max_iters=long_iters,
                eta0=0.25, active_frac=0.001)
            print(f"[Long EG] best surrogate after polishing = {best_val:.6f}")

        # ------------------------------------------------------------------
        # 4) Simulated annealing final tweak
        # ------------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            sa_iters = int(min(600_000, remaining * 1800))
            print(f"[SA] polishing with up to {sa_iters} iterations")
            best_seq, best_val = _simulated_annealing(
                best_seq, n, deadline, max_iters=sa_iters)
            print(f"[SA] final surrogate = {best_val:.6f}")

        # ------------------------------------------------------------------
        # 5) True evaluation
        # ------------------------------------------------------------------
        try:
            true_val = evaluate_sequence(best_seq.tolist())
            print(f"True evaluate_sequence value (n={n}) = {true_val:.6f}")
        except Exception as exc:  # pragma: no cover
            print("Warning: evaluate_sequence raised an exception:", exc)
            true_val = None

        if true_val is not None and true_val < best_overall_val:
            best_overall_val = true_val
            best_overall_seq = best_seq
            best_overall_n = n
            print(f"*** New overall best (n={n}) = {true_val:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (fallback to whatever is available)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – use the last best_seq we have (should not happen)
        best_overall_seq = best_seq
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    print(f"\n=== Finished overall ===\nBest length = {best_overall_n}, "
          f"objective = {best_overall_val:.6f}")
    return best_overall_seq.tolist()
