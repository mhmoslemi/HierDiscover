   deadline = start_time + budget_s - 2.0

   n = 2000

   # generate seeds (rich)
   seeds = _collect_seeds_extended(n, rng, previous)

   seeds_evaluated.sort()

   global_best_seq = best_seeds[0].copy()

   # Stage 1: L-BFGS on smooth surrogate
   for idx, seed_vec in enumerate(best_seeds, start=1):

"""
Optimiser for the step‑function target.
The approach:
1. Generate a diverse seed pool (analytic families, random Dirichlet,
   low‑frequency curves, and mixtures, plus the previous best if any).
2. Keep the ~10 best seeds (by fast surrogate 2·n·max(conv)).
3. For each seed run a warm‑started L‑BFGS optimisation on the
   smooth‑max surrogate (log‑sum‑exp) with an increasing temperature
   schedule.  This gives a high‑quality local optimum.
4. Refine the best candidate with a long mirror‑descent (exponentiated
   gradient) using the exact sub‑gradient of the max‑convolution.
5. A simulated‑annealing polishing phase performs swap / redistribution /
   pair‑transfer moves with a geometric cooling schedule.
6. Finally a cheap hill‑climbing sweep moves a tiny amount of mass
   between random coordinates; any improvement is kept.
The routine respects a hard deadline (default 1000 s) and stores the
result in the global name ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (length = len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shifted(n: int, sigma_frac: float, shift_frac: float) -> np.ndarray:
    """Gaussian centred at shift_frac * (n‑1)."""
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = shift_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1.0)))
        elif i <= (n - 1) * (1.0 - alpha / 2.0):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2.0 / alpha + 1.0)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_growth(n: int, lam: float) -> np.ndarray:
    a = np.exp(lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinc_squared(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = np.where(i == 0, 1.0, np.sin(np.pi * i / n) / (np.pi * i / n))
    a = x ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine_power(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_vectors(v1: np.ndarray, v2: np.ndarray, alpha: float) -> np.ndarray:
    """Return convex combination alpha*v1 + (1-alpha)*v2, ℓ₁‑normalised."""
    a = alpha * v1 + (1.0 - alpha) * v2
    s = a.sum()
    return a / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (fast)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – operates on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max utilities for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """Return smooth max (log‑sum‑exp) of convolution and its gradient."""
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 150,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """L‑BFGS on the smooth‑max surrogate (log‑sum‑exp)."""
    eps = 1e-30
    v = np.log(start_seq + eps)  # logits, unconstrained

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        grad_v_scaled = 2.0 * n * grad_v
        return loss, grad_v_scaled.astype(np.float64)

    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:  # pragma: no cover
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    conv = _conv_fft(best_a, best_a)
    best_val = 2.0 * n * np.max(conv)
    return best_a, best_val


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax producing a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Mirror‑descent sub‑gradient on the exact max‑conv objective
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 3_000_000,
                                 lr0: float = 0.12,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true max‑conv objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:,} best={cur_val:.6f}")

        # Sub‑gradient for the active index (handle ties by averaging)
        # Identify all indices within tolerance of the max
        eps = 1e-12
        tie_mask = np.abs(conv - conv[max_idx]) <= eps
        idxs = np.where(tie_mask)[0]
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx] / len(idxs)  # average over ties

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 1_500_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)   # 0=swap, 1=redistribute subset, 2=pair‑transfer
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:,} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (rich pool)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))
    seeds.append(_sinc_squared(n))

    # sine windows
    for gamma in np.linspace(0.3, 4.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian families (centered and shifted)
    for sigma in np.linspace(0.08, 0.45, 10):
        seeds.append(_gaussian(n, sigma_frac=sigma))
    for sigma in np.linspace(0.08, 0.45, 6):
        for shift in np.linspace(0.2, 0.8, 5):
            seeds.append(_gaussian_shifted(n, sigma_frac=sigma, shift_frac=shift))

    # Exponential decay / growth
    for lam in np.linspace(0.0005, 0.015, 8):
        seeds.append(_exponential_decay(n, lam))
    for lam in np.linspace(0.0005, 0.015, 6):
        seeds.append(_exponential_growth(n, lam))

    # Power‑law
    for p in np.linspace(0.0, 4.0, 9):
        seeds.append(_power(n, p))

    # Beta shapes, extended range
    alphas = np.concatenate((np.linspace(0.02, 0.3, 5),
                             np.linspace(0.3, 5.0, 9)))
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.05, 0.95, 9):
        seeds.append(_tukey_window(n, alpha))

    # Raised‑cosine‑power
    for power in [0.5, 1.0, 1.5, 2.0, 3.0]:
        seeds.append(_raised_cosine_power(n, power))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of two families
    base_families = [_uniform, _gaussian, _beta_shape, _sine_window, _exponential_decay]
    for _ in range(20):
        f1, f2 = rng.choice(base_families, size=2, replace=False)
        # generate parameters for each selector
        if f1 is _uniform:
            v1 = f1(n)
        elif f1 is _gaussian:
            v1 = f1(n, sigma_frac=rng.uniform(0.05, 0.4))
        elif f1 is _beta_shape:
            v1 = f1(n, alpha=rng.uniform(0.05, 4.0), beta=rng.uniform(0.05, 4.0))
        elif f1 is _sine_window:
            v1 = f1(n, gamma=rng.uniform(0.5, 3.0))
        else:  # exponential_decay
            v1 = f1(n, lam=rng.uniform(0.001, 0.02))

        if f2 is _uniform:
            v2 = f2(n)
        elif f2 is _gaussian:
            v2 = f2(n, sigma_frac=rng.uniform(0.05, 0.4))
        elif f2 is _beta_shape:
            v2 = f2(n, alpha=rng.uniform(0.05, 4.0), beta=rng.uniform(0.05, 4.0))
        elif f2 is _sine_window:
            v2 = f2(n, gamma=rng.uniform(0.5, 3.0))
        else:
            v2 = f2(n, lam=rng.uniform(0.001, 0.02))

        alpha = rng.uniform(0.0, 1.0)
        seeds.append(_mix_vectors(v1, v2, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

            # alternating high‑low pattern
            a = np.sort(raw)
            lo, hi = 0, len(a) - 1
            alt = np.empty_like(a)
            pos = 0
            while lo <= hi:
                if pos % 2 == 0:
                    alt[pos] = a[hi]
                    hi -= 1
                else:
                    alt[pos] = a[lo]
                    lo += 1
                pos += 1
            seeds.append(alt)

    # Include the previous solution if supplied
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a list of floats and stores it globally
    as ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    # Choose a moderate length – large enough to give a good constant,
    # small enough to allow many optimisation steps.
    n = 2000

    # Retrieve possible previous solution
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Fast surrogate ranking – keep the best few
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for _, vec in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on the smooth‑max surrogate
    # ------------------------------------------------------------------
    temperature_schedule = [5.0, 20.0, 80.0, 300.0, 800.0]
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temperature_schedule:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline, max_iter=180, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp:.0f} → new best {val_opt:.6f}")
            seq = seq_opt  # warm‑start for next temperature

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent (exact max) fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=4_000_000,
            lr0=0.22,
            power=0.28,
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_500_000,
            init_temp_factor=0.2,
            min_temp=1e-14,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Tiny hill‑climbing tweaks (mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Hill‑climbing fine tuning ---")
        seq = global_best_seq.copy()
        best_val = _objective(seq)
        it = 0
        while time.time() < deadline and it < 200_000:
            i, j = rng.integers(0, n, size=2, endpoint=False)
            if i == j:
                continue
            eps = 1e-5 * seq[i]  # very small relative move
            if eps <= 0.0:
                continue
            new_seq = seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            if new_seq[i] < 0.0:
                continue
            new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < best_val:
                seq = new_seq
                best_val = new_val
                if best_val < global_best_val:
                    global_best_seq = new_seq.copy()
                    global_best_val = best_val
                    print(f"[Hill] it={it:,} new best {best_val:.6f}")
            it += 1

    # ------------------------------------------------------------------
    # Final official evaluation (informational)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()

"""
Optimiser for the step‑function challenge.

The algorithm proceeds in several stages:

1. **Seed generation** – analytic families (uniform, triangular, cosine,
   sine‑window, Gaussian, shifted Gaussian, beta, Tukey, exponential,
   sinc‑squared, raised‑cosine‑power) together with many random Dirichlet
   draws, low‑frequency random curves and convex mixtures are created and
   ℓ₁‑normalised.

2. **Fast surrogate ranking** – the surrogate objective
   `2·n·max(conv(a,a))` (identical to the official objective for a
   normalised vector) is evaluated with an FFT‑based convolution and only
   the best ~12 seeds are retained.

3. **L‑BFGS refinement** – each top seed is optimised with SciPy’s
   L‑BFGS‑B on a smooth‑max (log‑sum‑exp) approximation of the max‑convolution.
   The temperature is increased in a schedule
   `[5, 20, 80, 300, 800]` to approach the true non‑smooth objective while
   keeping the problem tractable.

4. **Mirror‑descent sub‑gradient** – the best candidate is further polished
   with a multiplicative (exponentiated‑gradient) update on the simplex,
   using the exact sub‑gradient of the true maximal convolution.
   The learning‑rate schedule is `ηₜ = 0.22 / t^0.28`.

5. **Simulated‑annealing polishing** – a lightweight SA pass (swap,
   Dirichlet‑redistribute, pair‑transfer moves) with a geometric cooling
   schedule (`init_temp_factor=0.2`) helps escape shallow minima.

6. **Hill‑climbing fine‑tuning** – tiny mass transfers between random
   coordinates are attempted; any improvement is kept.

All stages respect a hard deadline (default 1000 s).  The best sequence is
stored in the global name ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
import scipy.optimize


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (length = len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shifted(n: int, sigma_frac: float, shift_frac: float) -> np.ndarray:
    """Gaussian centred at shift_frac·(n‑1)."""
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = shift_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1.0)))
        elif i <= (n - 1) * (1.0 - alpha / 2.0):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2.0 / alpha + 1.0)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_growth(n: int, lam: float) -> np.ndarray:
    a = np.exp(lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinc_squared(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = np.where(i == 0, 1.0, np.sin(np.pi * i / n) / (np.pi * i / n))
    a = x ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine_power(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, projected to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_vectors(v1: np.ndarray, v2: np.ndarray, alpha: float) -> np.ndarray:
    """Convex combination α·v1 + (1‑α)·v2, ℓ₁‑normalised."""
    a = alpha * v1 + (1.0 - alpha) * v2
    s = a.sum()
    return a / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (fast)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Softmax utilities (used by L‑BFGS)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """Return smooth max of the convolution and its gradient w.r.t. a."""
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    weights = exp_term / w_sum if w_sum != 0.0 else np.zeros_like(conv)
    smooth = shift + np.log(w_sum) / beta
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 150,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """L‑BFGS on the log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-30
    v = np.log(start_seq + eps)  # logits – unconstrained

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)               # Jacobian of softmax
        grad_v_scaled = 2.0 * n * grad_v
        return loss, grad_v_scaled.astype(np.float64)

    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:  # pragma: no cover
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    conv = _conv_fft(best_a, best_a)
    best_val = 2.0 * n * np.max(conv)
    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent sub‑gradient on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 3_000_000,
                                 lr0: float = 0.12,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent using the exact sub‑gradient."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:,} best={cur_val:.6f}")

        # Sub‑gradient (average over ties to improve stability)
        eps = 1e-12
        tie_mask = np.abs(conv - conv[max_idx]) <= eps
        active = np.where(tie_mask)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx] / len(active)

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 1_500_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)   # 0=swap, 1=redistribute, 2=pair‑transfer
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:,} best={new_val:.6f}")

    return best_seq, best_val

            v1 = f1(n, gamma=rng.uniform(0.5, 3.0))
        else:
            v1 = f1(n, lam=rng.uniform(0.001, 0.02))

        if f2 is _uniform:
            v2 = f2(n)
        elif f2 is _gaussian:
            v2 = f2(n, sigma_frac=rng.uniform(0.05, 0.4))
        elif f2 is _beta_shape:
            v2 = f2(n, alpha=rng.uniform(0.05, 5.0), beta=rng.uniform(0.05, 5.0))
        elif f2 is _sine_window:
            v2 = f2(n, gamma=rng.uniform(0.5, 3.0))
        else:
            v2 = f2(n, lam=rng.uniform(0.001, 0.02))

        alpha = rng.uniform(0.0, 1.0)
        seeds.append(_mix_vectors(v1, v2, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

            # alternating high‑low pattern
            a = np.sort(raw)
            lo, hi = 0, len(a) - 1
            alt = np.empty_like(a)
            pos = 0
            while lo <= hi:
                if pos % 2 == 0:
                    alt[pos] = a[hi]
                    hi -= 1
                else:
                    alt[pos] = a[lo]
                    lo += 1
                pos += 1
            seeds.append(alt)

    # Include previous solution if it matches the current length
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a list of floats and stores it globally
    as ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    # Moderate problem size – large enough for a good constant, small enough
    # to allow many optimisation iterations.
    n = 2000

    # Retrieve any previously discovered solution for seeding.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Fast surrogate evaluation, keep the most promising ones.
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for _, vec in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on the smooth‑max surrogate
    # ------------------------------------------------------------------
    temperature_schedule = [5.0, 20.0, 80.0, 300.0, 800.0]
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temperature_schedule:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline, max_iter=180, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp:.0f} → new best {val_opt:.6f}")
            seq = seq_opt  # warm‑start next temperature

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent (exact max) fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=4_000_000,
            lr0=0.22,
            power=0.28,
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_500_000,
            init_temp_factor=0.2,
            min_temp=1e-14,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Tiny hill‑climbing tweaks (mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Hill‑climbing fine tuning ---")
        seq = global_best_seq.copy()
        best_val = _objective(seq)
        it = 0
        while time.time() < deadline and it < 200_000:
            i, j = rng.integers(0, n, size=2, endpoint=False)
            if i == j:
                continue
            eps = 1e-5 * seq[i]  # very small relative move
            if eps <= 0.0:
                continue
            new_seq = seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            if new_seq[i] < 0.0:
                continue
            new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < best_val:
                seq = new_seq
                best_val = new_val
                if best_val < global_best_val:
                    global_best_seq = new_seq.copy()
                    global_best_val = best_val
                    print(f"[Hill] it={it:,} new best {best_val:.6f}")
            it += 1

    # ------------------------------------------------------------------
    # Final official evaluation (informational)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
