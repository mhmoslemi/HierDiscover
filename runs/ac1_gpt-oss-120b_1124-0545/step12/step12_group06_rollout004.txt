"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(seq,seq)) / (sum(seq)²)

The search proceeds in five stages:

1. **Rich seed pool** – analytic families, Dirichlet draws,
   low‑frequency random curves and permutations.  All seeds are ℓ₁‑normalised.
2. **Exact‑gradient Adam** – the vector is represented by soft‑max logits.
   The true max‑convolution sub‑gradient (not a smooth surrogate) is used.
   Adam’s adaptivity quickly drives the objective down.
3. **Mirror (exponentiated‑gradient) sub‑gradient descent** – multiplicative
   updates on the simplex using a *multi‑active* sub‑gradient (the sum of the
   gradients of the top few convolution entries).  This refines the shape
   far beyond the single‑active version.
4. **Random perturbation + short Adam runs** – to escape possible local minima
   we repeatedly jitter the current best vector and run a brief Adam optimisation.
5. **Final sanity check** – evaluate the best vector with the official
   ``evaluate_sequence`` function and store it in the global variable
   ``height_sequence_1`` for reuse.

All phases respect the overall time budget (default 1000 s) and keep the
best feasible sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective (without the denominator – denominator = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector (sum=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, … (keeps sum = 1)."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of diverse ℓ₁‑normalised start vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # sine windows + alternates
    for gamma in np.linspace(0.5, 6.0, 12):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussians
    for sigma in np.linspace(0.10, 0.55, 10):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # power laws
    for p in np.linspace(0.0, 8.0, 17):
        seeds.append(_power(n, p))
        seeds.append(_alternate_order(_power(n, p)))

    # beta family
    for _ in range(15):
        alpha = 10 ** rng.uniform(-2, 0.6)
        beta = 10 ** rng.uniform(-2, 0.6)
        seeds.append(_beta_shape(n, alpha, beta))
        seeds.append(_alternate_order(_beta_shape(n, alpha, beta)))

    # exponentials
    for rate in np.logspace(-4, -1, 12):
        seeds.append(_exponential(n, rate))
        seeds.append(_alternate_order(_exponential(n, rate)))

    # logarithmic
    for eps in np.logspace(-4, -1, 8):
        seeds.append(_logarithmic(n, eps))
        seeds.append(_alternate_order(_logarithmic(n, eps)))

    # Dirichlet draws – many concentrations & orderings
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(8):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw
            seeds.append(_alternate_order(raw))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(curve)
            seeds.append(_alternate_order(curve))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = _mix_shapes(_uniform(n), curve, mix_alpha)
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Random permutations of a few good deterministic shapes
    for base in [_gaussian(n, 0.2), _power(n, 3.0), _beta_shape(n, 0.5, 0.5)]:
        for _ in range(5):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    # Previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_exact_optimize(start_vec: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 5_000_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the *exact* max‑convolution objective.
    The vector is stored as logits ``v`` such that ``softmax(v)`` lives on the simplex.
    Returns the best feasible vector seen during the run.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # initialise logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # probability vector (ℓ₁‑normalised)
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_val = 2.0 * n * max_val

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[AdamExact] it={it:7d} best={cur_val:.6f}")

        # ---- sub‑gradient of true max (multi‑active) ----
        # select the top few active shifts (at most 5) to stabilise the sub‑gradient
        top_k = 5
        # indices of the largest convolution entries
        active_idxs = np.argpartition(-conv, range(min(top_k, conv.size)))[:top_k]
        # ensure we include the global maximum
        max_idx = int(np.argmax(conv))
        if max_idx not in active_idxs:
            active_idxs[0] = max_idx

        grad_a = np.zeros_like(a)
        for k in active_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            grad_a[idx_i] += 2.0 * a[rev_idx]

        # chain‑rule for soft‑max (log‑space) parameters
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent.
    Uses a multi‑active sub‑gradient (sum of gradients of the top few
    convolution entries) which is more informative than the single‑active
    version.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20           # base learning rate
    max_active = 5       # how many top shifts to include in the sub‑gradient

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_val = 2.0 * n * max_val

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # ---- multi‑active sub‑gradient ----
        # pick the indices of the largest convolution values (up to max_active)
        active_idxs = np.argpartition(-conv, range(min(max_active, conv.size)))[:max_active]
        # guarantee that the absolute max is included
        max_idx = int(np.argmax(conv))
        if max_idx not in active_idxs:
            active_idxs[0] = max_idx

        g = np.zeros_like(a)
        for k in active_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional alternating re‑ordering – often helpful
        if it % 100_000 == 0:
            a_alt = _alternate_order(a)
            alt_val = _objective(a_alt)
            if alt_val < cur_val:
                a = a_alt
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {alt_val:.6f}")

    return best_seq, best_val


def _perturb(seq: np.ndarray, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Mix the sequence with a fresh Dirichlet draw (α controls the blend)."""
    n = len(seq)
    draw = rng.dirichlet(np.full(n, 1.0))
    pert = (1.0 - alpha) * seq + alpha * draw
    s = pert.sum()
    return pert / s if s > 0 else _uniform(n)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence (list of floats) and stores it in the global
    name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0               # safety margin

    # ------------------------------------------------------------------
    # Length – 4000 offers a good trade‑off between expressiveness and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution if it exists.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – rich seed pool & quick screening
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} exact objective values:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam with exact max‑gradient (run on the current best)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_budget = remaining * 0.30                # give roughly 30 % to AdamExact
        adam_deadline = time.time() + adam_budget
        print("\n--- Exact‑gradient Adam optimisation ---")
        seq_adam, val_adam = _adam_exact_optimize(global_best_seq, n, adam_deadline)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # try alternating order as a cheap extra move
        alt = _alternate_order(global_best_seq)
        alt_val = _objective(alt)
        if alt_val < global_best_val:
            global_best_seq = alt
            global_best_val = alt_val
            print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Mirror‑descent sub‑gradient optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Random perturbations + short AdamExact runs (exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exploration via perturbations + AdamExact ---")
        while time.time() < deadline:
            # allocate a tiny budget for each exploration run
            chunk = min(30.0, deadline - time.time())
            if chunk <= 0.0:
                break
            pert = _perturb(global_best_seq, alpha=0.25, rng=rng)
            sub_deadline = time.time() + chunk
            seq_p, val_p = _adam_exact_optimize(pert, n, sub_deadline)
            if seq_p is not None and val_p < global_best_val:
                global_best_seq = seq_p.copy()
                global_best_val = val_p
                print(f"[explore‑Adam] improved to {val_p:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
