"""
Creative evolutionary optimizer for the step‑function bound.

The algorithm proceeds in four phases:

1. **Rich seed pool** – deterministic shapes (uniform, double‑Gaussian,
   two‑sided exponential, power‑law, Beta, Dirichlet draws, mixtures)
   are ℓ₁‑normalised.  The very best seeds become a small population.

2. **Evolutionary search** – a lightweight (μ+λ) evolutionary loop
   (crossover + Gaussian mutation + Dirichlet‑redistribution moves) keeps
   the population on the simplex and continuously tries to lower the
   surrogate objective `2·n·max(conv(seq,seq))`.  The loop runs until the
   time budget expires.

3. **Permutation hill‑climbing** – using only swaps of entries (the
   multiset of values is fixed) we try to improve the best individual
   found by the evolution.

4. **Final evaluation** – the best vector is returned (as a plain list)
   and stored in the global name ``height_sequence_1`` for reuse.

All heavy work uses a fast FFT based convolution that computes
`conv(seq,seq)` in O(n log n).  The code respects the 1000 s budget,
prints progress, and works for any length `n` (default 4000).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(seq: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a vector with itself using a real FFT.
    Returns the full length (2·len(seq)-1) array.
    """
    n = len(seq)
    N = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(f * f, n=N)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective: 2·n·max(conv(seq,seq)).
    The input is assumed to be ℓ₁‑normalised (sum = 1) and non‑negative.
    """
    n = len(seq)
    conv = _conv_self(seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """
    Re‑order entries as large, small, 2nd‑large, 2nd‑small, … and renorm.
    This tends to spread large values over the whole index range.
    """
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.07,
                    offset_frac: float = 0.35) -> np.ndarray:
    """Two narrow Gaussians placed symmetrically around the centre."""
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _two_sided_exponential(n: int, lam: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float = 0.5) -> np.ndarray:
    """U‑shaped power‑law: a_i ∝ (i+1)^(-p) + (n-i)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p) + (n - i + 1) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta density discretised on {1,…,n} and ℓ₁‑normalised."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    """Simple decreasing exponential (only left side)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_draw(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc) and normalise."""
    raw = rng.dirichlet(np.full(n, conc))
    s = raw.sum()
    return raw / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed generation – much richer than the reference
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))
    seeds.append(_two_sided_exponential(n, lam=0.15))
    seeds.append(_power_law(n, p=0.4))

    # Double‑Gaussian grid (more aggressive parameters) --------------------
    for sigma in np.linspace(0.02, 0.12, 7):
        for offset in np.linspace(0.20, 0.45, 6):
            dg = _double_gaussian(n, sigma_frac=sigma, offset_frac=offset)
            seeds.append(dg)
            seeds.append(_alternate_order(dg))

    # Two‑sided exponential grid -------------------------------------------
    for lam in np.logspace(-2, 0, 8):
        te = _two_sided_exponential(n, lam=lam)
        seeds.append(te)
        seeds.append(_alternate_order(te))

    # Power‑law grid --------------------------------------------------------
    for p in [0.3, 0.5, 0.7, 1.0]:
        pl = _power_law(n, p=p)
        seeds.append(pl)
        seeds.append(_alternate_order(pl))

    # Beta family – fine‑grained exploration ---------------------------------
    for _ in range(30):
        alpha = 10 ** rng.uniform(-4, 0.9)   # 0.0001 … ≈ 8
        beta = 10 ** rng.uniform(-4, 0.9)
        b = _beta_shape(n, alpha, beta)
        seeds.append(b)
        seeds.append(_alternate_order(b))

    # Exponential shapes – wide rate range ----------------------------------
    for rate in np.logspace(-5, -0.5, 12):
        expv = _exponential(n, rate)
        seeds.append(expv)
        seeds.append(_alternate_order(expv))

    # Dirichlet draws – many concentration values ---------------------------
    concs = [0.001, 0.005, 0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(3):
            raw = _dirichlet_draw(n, conc, rng)
            seeds.append(raw)
            seeds.append(_alternate_order(raw))

    # Random mixtures of two independent Beta shapes ------------------------
    for _ in range(20):
        a1 = _beta_shape(n, 10 ** rng.uniform(-3, 0.5), 10 ** rng.uniform(-3, 0.5))
        a2 = _beta_shape(n, 10 ** rng.uniform(-3, 0.5), 10 ** rng.uniform(-3, 0.5))
        w = rng.random()
        mix = w * a1 + (1 - w) * a2
        mix /= mix.sum()
        seeds.append(mix)
        seeds.append(_alternate_order(mix))

    # Previous solution (if any) --------------------------------------------
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            norm = avg / avg.sum()
            seeds.append(norm)
            seeds.append(_alternate_order(norm))

    return seeds


# ----------------------------------------------------------------------
# Evolutionary optimisation on the simplex
# ----------------------------------------------------------------------
def _evolutionary_search(pop: list[tuple[np.ndarray, float]],
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_swaps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    (μ+λ) style search: crossover + Gaussian mutation + Dirichlet-
    redistribution moves.  The population list contains (seq, value) tuples.
    """
    pop_size = len(pop)
    best_seq, best_val = min(pop, key=lambda x: x[1])

    # pre‑compute decreasing mutation strength
    it = 0
    while time.time() < deadline and it < max_swaps:
        it += 1
        # decay schedule for mutation std (from 0.1 to 0.001)
        sigma = 0.10 * (1.0 - it / max_swaps) + 0.001

        # 50 % crossover, 50 % mutation
        if rng.random() < 0.5:
            # --- Crossover (convex combination) ---
            i, j = rng.choice(pop_size, size=2, replace=False)
            p1 = pop[i][0]
            p2 = pop[j][0]
            w = rng.random()
            child = w * p1 + (1.0 - w) * p2
        else:
            # --- Mutation (Gaussian) ---
            i = rng.integers(pop_size)
            parent = pop[i][0]
            child = parent + rng.normal(scale=sigma, size=n)
            child = np.clip(child, 0.0, None)

        # occasional Dirichlet redistribution on a random small subset
        if rng.random() < 0.15:
            k = rng.integers(3, 9)      # size of the subset
            idx = rng.choice(n, size=k, replace=False)
            total = child[idx].sum()
            if total > 0.0:
                child[idx] = rng.dirichlet(np.full(k, 1.0)) * total

        # renormalise onto the simplex
        s = child.sum()
        if s <= 0.0:
            child = _uniform(n)
        else:
            child = child / s

        # evaluate
        val = _objective(child)

        # replace the worst individual if we improved
        worst_idx = max(range(pop_size), key=lambda idx: pop[idx][1])
        if val < pop[worst_idx][1]:
            pop[worst_idx] = (child, val)

        # keep track of the overall best
        if val < best_val:
            best_val = val
            best_seq = child.copy()

        # very cheap occasional alternate‑order perturbation of the best
        if it % 5000 == 0:
            alt = _alternate_order(best_seq)
            alt_val = _objective(alt)
            if alt_val < best_val:
                best_seq, best_val = alt, alt_val
                # optionally insert into population
                worst_idx = max(range(pop_size), key=lambda idx: pop[idx][1])
                pop[worst_idx] = (best_seq, best_val)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple swap‑based hill‑climbing on the permutation of entries
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int = 80_000) -> tuple[np.ndarray, float]:
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    T0 = cur_val * 0.005
    T = T0
    it = 0
    while it < max_swaps and time.time() < deadline:
        it += 1
        T = max(T * 0.9999, 1e-12)
        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur, cur_val = new, new_val
            if new_val < best_val:
                best_seq, best_val = new.copy(), new_val
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises

        evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

    The function runs under the given time budget, returns the best
    sequence found and stores it in the global variable ``height_sequence_1``.
    """
    # --------------------------------------------------------------
    # Basic setup
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – 4000 offers a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best (if any) and normalise it.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Phase 1 – construct a diverse pool of analytic seeds.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Evaluate all seeds (surrogate objective) and keep the best few.
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – initialise a small population for evolutionary search.
    # --------------------------------------------------------------
    population = [(seq, _objective(seq)) for seq in best_seeds]
    # If we have fewer than 30 individuals, pad with random Dirichlet draws.
    while len(population) < 30 and time.time() < deadline:
        conc = rng.choice([0.001, 0.005, 0.01, 0.05, 0.1])
        cand = _dirichlet_draw(n, conc, rng)
        population.append((cand, _objective(cand)))

    print(f"\n--- Evolutionary optimisation (population size = {len(population)}) ---")
    best_seq, best_val = min(population, key=lambda x: x[1])
    print(f"[Init] Best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – evolutionary loop (crossover, mutation, Dirichlet moves).
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_val = _evolutionary_search(population, n, deadline, rng)
        print(f"[Evolution] New best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – permutation hill‑climbing (swap‑only local search).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation hill‑climbing ---")
        seq_perm, val_perm = _perm_hill_climb(best_seq, n, deadline)
        if val_perm < best_val:
            best_seq, best_val = seq_perm, val_perm
            print(f"[Perm] Improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final sanity‑check with the official evaluator.
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} (length {len(best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
