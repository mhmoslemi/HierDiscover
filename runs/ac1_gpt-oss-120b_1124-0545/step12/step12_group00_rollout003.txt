assistantfinal**Solution Overview**

The goal is to minimise

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where \((a*a)_k\) is the linear convolution of the vector with itself.
Because the objective is homogeneous, we may enforce the simplex constraint
\(\sum_i a_i = 1\) and minimise  

\[
\hat F(a)=2\,n\;\max_k (a*a)_k .
\]

The optimisation problem is non‑convex and the “max’’ makes the gradient
discontinuous.  
We use a **smooth surrogate**:

\[
\phi_\beta(a)=\frac{1}{\beta}\log\!\Bigl(\sum_k e^{\beta\,(a*a)_k}\Bigr)
\]

(soft‑max of the autocorrelation).  
For a fixed smoothing parameter \(\beta\) the gradient is

\[
\nabla_a\phi_\beta(a)=2\; (w * a_{\text{rev}})_{[n-1:n-1+n]},
\quad
w_k=\frac{e^{\beta (a*a)_k}}{\sum_j e^{\beta (a*a)_j}} .
\]

All convolutions are performed by FFT ( \(O(n\log n)\) ).  
The algorithm proceeds in three stages:

1. **Seed generation** – a large pool of analytic and random windows, all
   normalised to the simplex. The 5 best seeds (according to the surrogate)
   are kept.
2. **Smooth exponential‑gradient descent** – for each seed we run a short
   run of a *mirror‑descent* update  
   \[
   a \leftarrow a\;\exp(-\eta_t \,\nabla\phi_\beta(a)),
   \qquad a\gets a/\!\sum a ,
   \]
   with a decreasing stepsize \(\eta_t=\eta_0/\sqrt{t}\) and a continuation
   schedule for \(\beta\) ( \(\beta=1,5,15,30\) ).  The best vector over all
   seeds is retained.
3. **Long‑run polishing** – the current best vector is refined for the
   remaining time with the same update but a larger number of iterations and
   the largest \(\beta\) (30).  The surrogate value is monitored and the best
   vector ever seen is saved.

Finally the vector is handed to the official `evaluate_sequence`
function and stored globally as `height_sequence_1`.

The whole search respects the 1000 s deadline, uses only NumPy and plain
Python, and runs on a single process (≤ 2 CPU cores).

```python
"""
search for a non‑negative step function with minimal evaluate_sequence
value.

Algorithm
---------
1. generate a rich set of ℓ₁‑normalised seeds
2. for each seed run a short “smooth exponential‑gradient” descent
   (soft‑max surrogate, exponential update, simplex renormalisation)
3. polish the current best with a long run of the same optimiser
4. return the best sequence (list of floats)

All convolutions are performed with a real FFT, the algorithm adapts its
stepsize (η_t = η0/√t) and uses a continuation on the smoothness parameter
β ∈ {1,5,15,30}.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (O(n log n)).
    Returns a 1‑D array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _uniform(n: int) -> np.ndarray:
    """Uniform simplex vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, keep only the positive part, normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified list of ℓ₁‑normalised seeds."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0):
        for b in (0.5, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution, if any
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # safety normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        clean.append(v / s if s > 0 else _uniform(n))
    return clean


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Surrogate: 2·n·max(conv(a,a)) – assumes Σa = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _calc_grad(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smooth surrogate φ_β at point a.

    Returns
    -------
    grad : np.ndarray
        ∇_a φ_β (same shape as a)
    max_conv : float
        max_k (a*a)_k (needed for the true surrogate value)
    """
    n = a.shape[0]

    # full linear autocorrelation
    conv = _conv_fft(a, a)                     # length 2n-1
    max_conv = np.max(conv)

    # soft‑max weights (numerically stable)
    shifted = conv - max_conv                  # max becomes 0
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()             # length 2n-1, sums to 1

    # gradient = 2 * (w * a_rev) central part
    a_rev = a[::-1]
    conv2 = _conv_fft(w, a_rev)               # length 3n-2
    start = n - 1
    grad = 2.0 * conv2[start:start + n]       # size n
    return grad, max_conv


# ----------------------------------------------------------------------
# Optimiser: smooth exponential‑gradient descent
# ----------------------------------------------------------------------
def _smooth_expgrad(start: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule: list[float],
                    steps_per_beta: int,
                    eta0: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the smooth surrogate.
    Multiplicative update: a ← a·exp(-η·∇φ_β), followed by renormalisation.
    Returns best vector seen and its surrogate value.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_a, best_val

            grad, max_conv = _calc_grad(a, beta)

            it += 1
            eta = eta0 / math.sqrt(it)

            # multiplicative update, clip exponent to avoid overflow
            delta = -eta * grad
            delta = np.clip(delta, -50.0, 50.0)
            a = a * np.exp(delta)

            # renormalise onto the simplex (non‑negative, sum = 1)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = _uniform(n)

            cur_val = 2.0 * n * max_conv
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                if it % 20000 == 0:
                    print(f"[SmoothEG] it={it:6d} β={beta:.1f} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    evaluate_sequence objective within the allotted time budget.
    Returns the best sequence as a plain Python list.
    """
    # deterministic RNG for reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # --------------------------------------------------------------
    # Length of the step function – 4000 works well for the time budget
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Stage 0 – collect a diverse set of seeds
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Produced {len(seeds)} seeds.")

    # keep the best few according to the surrogate
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – quick smooth‑EG runs on each top seed
    # --------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    beta_sched_short = [1.0, 5.0, 15.0]          # smoother at the start
    steps_per_beta_short = 15000                # ≈ 45 k iterations per seed

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        cand, val = _smooth_expgrad(seed_vec, n, deadline,
                                    beta_schedule=beta_sched_short,
                                    steps_per_beta=steps_per_beta_short,
                                    eta0=0.35)
        if val < best_val:
            best_val, best_seq = val, cand
            print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – long‑run polishing on the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long‑run smooth‑EG polishing ---")
        remaining = deadline - time.time()
        # allocate the majority of the remaining budget here
        # estimate a step takes ≈ 0.0007 s → ≈ 1 M steps for 700 s
        est_iters = int(min(1_200_000, remaining * 1500))
        # map the estimated iterations onto beta schedule (use only the largest β)
        steps_per_beta_long = max(1, est_iters // 30)   # 30 blocks of equal size
        beta_sched_long = [30.0] * 30                 # keep β high for final refinement
        best_seq, best_val = _smooth_expgrad(best_seq, n, deadline,
                                            beta_schedule=beta_sched_long,
                                            steps_per_beta=steps_per_beta_long,
                                            eta0=0.20)
        print(f"[Polish] best surrogate after long run = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in subsequent calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
