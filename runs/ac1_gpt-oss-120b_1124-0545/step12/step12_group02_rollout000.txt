"""
Improved search for a step‑function minimizing

    2·n·max(conv(a,a)) / (sum(a)²)

The idea is very simple:

*   For a fixed ℓ₁‑norm the quantity to minimise equals 2·n·max(conv).
    If the mass is spread over a random *sparse* support the self‑convolution
    behaves like a random sum‑set: each ordered pair (i,j) contributes to the
    entry k=i+j, and for a random subset the number of pairs per k is
    ≈ S²/(2n).  Since a_i=1/S on the support, the resulting peak is ≈1/(2n),
    i.e. the surrogate value ≈ 1, well below the target 1.5.

*   We therefore generate many candidates of the form

        a_i = {1/S  if i∈support, 0 otherwise}

    where the support is a random subset of size `S ≈ n/2`.  Each candidate
    is evaluated with a fast FFT‑based convolution (O(n log n)).  The best
    candidate is kept.

*   A short projected sub‑gradient descent on the exact max‑convolution is
    applied to the current best vector – it removes the remaining dominant
    peak while keeping the vector on the probability simplex.

*   Finally the official `evaluate_sequence` is called once (the only
    expensive operation) and the resulting sequence is stored globally as
    ``height_sequence_1`` for possible reuse.

The whole routine respects a user supplied time budget (default 1000 s)
and never exceeds it.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate(a: np.ndarray, n: int) -> float:
    """Fast surrogate 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

        frac = rng.uniform(0.30, 0.90)
        S = max(2, int(frac * n))
        indices = rng.choice(n, size=S, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[indices] = 1.0 / S                     # ℓ₁‑normalised by construction
        val = _surrogate(a, n)
        sample_iter += 1
        if val < best_surrogate:
            best_surrogate = val
            best_seq = a.copy()
            print(f"[Sample] iter={sample_iter:5d} new best surrogate = {val:.6f}")

        # occasional progress report (does not affect time budget)
        if sample_iter % 500 == 0:
            elapsed = time.time() - start_time
            print(f"[Sample] {sample_iter} candidates, best={best_surrogate:.6f}, elapsed={elapsed:.1f}s")

    # --------------------------------------------------------
    # Stage 2 – short projected sub‑gradient descent
    # --------------------------------------------------------
    # The sub‑gradient of max(conv) is easy to compute (see comments below).
    # We run only a few thousand iterations – enough to shave a few 1e‑4 off
    # the surrogate while staying well within the time budget.
    remaining = deadline - time.time()
    if remaining > 1.0 and best_seq is not None:
        max_iters = int(min(10_000, remaining * 1500))   # crude scaling
        a = best_seq.copy()
        best_local_seq = a.copy()
        best_local_val = best_surrogate
        print(f"\n--- Sub‑gradient refinement (max_iters={max_iters}) ---")
        for it in range(1, max_iters + 1):
            # compute convolution and locate the active index
            conv = _conv_fft(a, a)
            max_idx = int(np.argmax(conv))

            # surrogate value for this iterate
            cur_val = 2.0 * n * conv[max_idx]
            if cur_val < best_local_val:
                best_local_val = cur_val
                best_local_seq = a.copy()
                if it % 1000 == 0:
                    print(f"[Sub] iter={it:5d} best surrogate={cur_val:.6f}")

            # ----- sub‑gradient for the active entry -----------------
            # For a_k appearing in the max, the derivative of conv[max_idx]
            # w.r.t. a_i equals 2·a_{max_idx‑i} (when the index is valid).
            g = np.zeros_like(a)
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            g[idx_i] = 2.0 * a[rev_idx]

            # simple diminishing step size and Euclidean projection onto the simplex
            lr = 0.30 / math.sqrt(it)          # step‑size schedule
            a = a - lr * g
            a = _proj_simplex(a)

            # early exit if time is exhausted
            if time.time() > deadline - 0.5:
                break

        # update global best if improvement happened
        if best_local_val < best_surrogate:
            best_surrogate = best_local_val
            best_seq = best_local_seq.copy()
            print(f"[Sub] final best surrogate after refinement = {best_local_val:.6f}")

    # --------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                                            # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
