    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0

    # set n
    n = 8000

    # seed generation
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in [0.5, 1.0, 2.0]:
        seeds.append(_sine_window(n, g))
    for sigma in np.linspace(0.12,0.30,5):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0,4.0,9):
        seeds.append(_power(n,p))
    for conc in [0.05, 0.2, 0.5, 1.0]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(10):
        seeds.append(_random_exponential(n, rng))
    # previous best
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if prev.shape[0]==n:
            seeds.append(prev/prev.sum())
        else:
            # upsample/dnsize
            pass
    # evaluate surrogates
    best_seq = None
    best_val = float('inf')
    for v in seeds:
        val = _objective_surrogate(v)   # surrogate already assumes sum=1
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"Initial best surrogate {best_val}")

    # Adam optimization
    def run_adam(init_seq, max_iters, beta_max=2000.0):
        v = np.log(init_seq + 1e-30)
        m = np.zeros_like(v)
        v_sq = np.zeros_like(v)
        b1=0.9; b2=0.999; eps=1e-8
        beta=0.5; beta_inc=0.5
        best_seq_local = init_seq.copy()
        best_val_local = _objective_surrogate(init_seq)
        for it in range(1, max_iters+1):
            if time.time() > deadline: break
            a = _softmax(v)
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv) / (np.sum(a)**2) # sum=1 each
            if cur_val < best_val_local:
                best_val_local = cur_val
                best_seq_local = a.copy()
            # compute weights
            shift = np.max(conv)
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0:
                continue
            w = w / w_sum
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w) # length 3n-2
            grad_a = 2.0 * full[n-1:n-1+n]

            grad_v = a * (grad_a - dot)

            lr = 1.0 / np.sqrt(it)
            m = b1*m + (1-b1)*grad_v
            v_sq = b2*v_sq + (1-b2)*(grad_v*grad_v)

            v_hat = v_sq/(1 - b2**it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            v = v - step
            if beta < beta_max:
                beta = min(beta + beta_inc, beta_max)
        return best_seq_local, best_val_local

            a = a - lr * g
            a = _proj_simplex(a)
        return best_local, best_val_local

    remaining = deadline - time.time()
    max_sub_iters = int(min(800_000, remaining * 1500))
    if max_sub_iters > 5000:
        seq, val = run_subgrad(best_seq, max_sub_iters)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"After subgrad best surrogate {best_val}")

    # Simulated annealing
    def run_sa(seq, max_iters):
        rng_local = np.random.default_rng()
        cur = seq.copy()
        cur_val = _objective_surrogate(cur)
        best_local = cur.copy()
        best_val_local = cur_val
        T0 = cur_val * 0.1 + 1e-12
        T = T0
        decay = 0.99995
        for it in range(1, max_iters+1):
            if time.time() > deadline: break
            # temperature decay
            T = max(T * decay, 1e-12)
            if rng_local.random() < 0.6:
                k = rng_local.integers(2, 20)
                idx = rng_local.choice(n, size=k, replace=False)
                total = cur[idx].sum()
                if total <= 0:
                    continue
                new_alloc = rng_local.dirichlet(np.full(k, 1.0)) * total
                new_seq = cur.copy()
                new_seq[idx] = new_alloc
            else:
                i, j = rng_local.choice(n, size=2, replace=False)
                eps = rng_local.random() * min(cur[i], 0.03)  # up to 3%
                if eps <= 0:
                    continue
                new_seq = cur.copy()
                new_seq[i] -= eps
                new_seq[j] += eps
            new_val = _objective_surrogate(new_seq)
            delta = new_val - cur_val
            if delta < 0 or rng_local.random() < math.exp(-delta / T):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val_local:
                    best_val_local = new_val
                    best_local = new_seq.copy()
                    if it % 50000 == 0:
                        print(f"[SA] iter {it} best {best_val_local}")
        return best_local, best_val_local

    remaining = deadline - time.time()
    max_sa_iters = int(min(2_000_000, remaining * 2000))
    if max_sa_iters > 1000:
        seq, val = run_sa(best_seq, max_sa_iters)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"After SA best surrogate {best_val}")

    # final evaluation
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as e:
        true_val = None
        print("Error evaluating true objective:", e)

    print(f"Final surrogate {best_val}, true {true_val}")
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Propose a non‑negative step‑function (a list of floats) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm works in three phases:

1. **Seed generation** – a diverse set of ℓ₁‑normalised analytic windows
   (uniform, triangular, sine, Gaussian, beta, random Dirichlet/exponential,
   low‑frequency curves, …).  The best few seeds (according to the cheap
   surrogate 2·n·max(conv) which is exact when the sum is 1) are retained.

2. **Gradient‑based refinement** – a long Adam optimisation on the
   log‑softmax parametrisation of the sequence.  The “soft‑max’’ of the
   convolution (with a temperature β that grows up to 2000) smoothes the
   max‑operator, giving a well‑behaved gradient.  The optimiser works
   directly on the probability simplex (the ℓ₁‑sum is always 1).

3. **Intensive simulated‑annealing polishing** – starting from the best
   Adam solution a temperature‑controlled Metropolis search reshuffles the
   mass on random subsets (Dirichlet redistribution) and performs small
   pairwise transfers.  The temperature decays very slowly (factor 0.99995)
   so that many millions of proposals can be examined within the time budget.

All stages respect the global 1000 s deadline.  The final sequence is stored
in the global name ``height_sequence_1`` for possible reuse and is returned
as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numerics
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) for a vector with ℓ₁‑sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few random cosine components, clip to non‑negative and normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diversified pool of ℓ₁‑normalised vectors."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # beta family grid (coarse)
    for a in (0.5, 1.0, 2.0, 4.0):
        for b in (0.5, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency curves
    for nf in (2, 4, 6):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_a in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_a))

    # previous solution, if compatible
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final normalisation guard
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray, n: int,
                   deadline: float, max_iters: int) -> tuple[np.ndarray, float]:
    """Adam optimisation on the log‑softmax parametrisation."""
    eps = 1e-30
    v = np.log(start_seq + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature
    beta_inc = 0.5
    beta_max = 2000.0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # smoothed max via softmax weights
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of the weighted sum of convolutions
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Jacobian of the softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update (learning‑rate decays as 1/√it)
        lr = 1.0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_optimize(start_seq: np.ndarray, n: int,
                          deadline: float, max_iters: int) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.5 / np.sqrt(it)          # a bit larger than the original 0.2
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int,
                         deadline: float, max_iters: int) -> tuple[np.ndarray, float]:
    """Metropolis search that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # start temperature roughly 10 % of the current value
    T = max(0.1 * cur_val, 1e-12)
    decay = 0.99995                     # very slow cooling

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, 1e-12)

        if rng.random() < 0.65:                 # reshuffle a random subset
            k = rng.integers(2, 15)            # subset size 2 … 14
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # small pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.03)   # move up to 3 % of source
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    overall_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 8000                               # length of the vector
    print(f"\n--- Starting optimisation (n={n}) ---")

    # ------------------------------------------------------------------
    # Stage 0 – seed generation
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap surrogate ranking (vectors are ℓ₁‑normalised → surrogate == objective)
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 15
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation (medium length)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()

    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()

        seq_sub, val_sub = _subgradient_optimize(global_best_seq, n,

        if val_sub < global_best_val:
            global_best_seq = seq_sub
            global_best_val = val_sub
            print(f"[SubGrad] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Aggressive simulated‑annealing polishing
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()

        seq_sa, val_sa = _simulated_annealing(global_best_seq, n,

        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "

        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
