"""
Propose a non‑negative step‑function (list of heights) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The search consists of:
1. Analytic seeds (uniform, triangular, Gaussian, Beta, Laplace, …)
   plus a large random Dirichlet pool (wide range of concentration).
2. Keep the top few seeds by the surrogate 2·n·max(conv) (the sum is
   already normalised to 1, so this equals the numerator of the
   official evaluator).
3. For each of those seeds run a mirror‑subgradient (exponentiated‑gradient)
   optimiser on the exact max‑convolution objective.  The optimiser
   updates on the probability simplex – i.e. it works directly with
   the ℓ₁‑normalised vectors.
4. Apply a focused swap‑based hill‑climbing on the current best vector
   (up to 2·10⁵ swaps).  Swaps that increase the objective are accepted
   with a tiny temperature so that occasional uphill moves are possible.
5. Finish with a short simulated‑annealing polish (swap / pair‑transfer /
   small‑subset redistribution moves) that runs for the remaining time.
6. Return the best sequence found and store it as the global variable
   ``height_sequence_1`` for future runs.
"""

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    Sequence is assumed to be ℓ₁‑normalised (sum = 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """
    Re‑order as large, small, 2nd‑large, 2nd‑small, …
    and re‑normalise to ℓ₁‑norm = 1.
    """
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.07,
                    offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, exponent: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-exponent}  (descending)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exponent)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))

    # sine windows on a fine grid
    for gamma in np.linspace(0.5, 6.0, 12):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussian on a fine sigma grid
    for sigma in np.linspace(0.07, 0.30, 12):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # beta family – explore a broader domain
    for _ in range(25):
        alpha = 10 ** rng.uniform(-3, 0.8)   # 0.001 … ~6.3
        beta = 10 ** rng.uniform(-3, 0.8)
        beta_vec = _beta_shape(n, alpha, beta)
        seeds.append(beta_vec)
        seeds.append(_alternate_order(beta_vec))

    # exponential shapes
    for rate in np.logspace(-5, -1, 10):
        exp_vec = _exponential(n, rate)
        seeds.append(exp_vec)
        seeds.append(_alternate_order(exp_vec))

    # logarithmic shapes
    for eps in np.logspace(-5, -1, 8):
        log_vec = _logarithmic(n, eps)
        seeds.append(log_vec)
        seeds.append(_alternate_order(log_vec))

    # power‑law shapes
    for exp in np.linspace(0.5, 2.5, 8):
        pl = _power_law(n, exp)
        seeds.append(pl)
        seeds.append(_alternate_order(pl))

    # Dirichlet draws with various concentrations (raw & alt)
    concs = [0.001, 0.003, 0.01, 0.03, 0.07, 0.12, 0.2, 0.5]
    for conc in concs:
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)
            seeds.append(_alternate_order(raw))

    # Mixtures of uniform with low‑frequency sinusoidal curves
    for nf in (2, 3, 4):
        for _ in range(2):
            t = np.arange(n, dtype=np.float64)
            curve = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                curve += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            curve = np.maximum(curve, 0.0)
            curve /= curve.sum()
            mixed = 0.7 * _uniform(n) + 0.3 * curve
            mixed /= mixed.sum()
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            normalised = avg / avg.sum()
            seeds.append(normalised)
            seeds.append(_alternate_order(normalised))

    return seeds

def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the *exact* max‑convolution.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.30          # initial step size

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index (only entries that appear in the max‑antidiagonal)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional re‑ordering to escape plateaus
        if it % 60_000 == 0:
            a_alt = _alternate_order(a)
            alt_val = _objective(a_alt)
            if alt_val < cur_val:
                a = a_alt
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved to {alt_val:.6f}")

    return best_seq, best_val


def _perm_hill_climb(start_seq: np.ndarray, n: int,
                     deadline: float,
                     max_swaps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple swap‑based hill‑climbing on the permutation of entries.
    Uses a tiny temperature so that a few uphill moves are possible.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    it = 0
    T0 = cur_val * 0.005   # initial temperature
    T = T0

    while it < max_swaps and time.time() < deadline:
        it += 1
        T = max(T * 0.9999, 1e-12)   # geometric cooling

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 20_000 == 0:
                    print(f"[Perm] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 500_000,
                                 init_temp_factor: float = 0.02,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polish with three move types:
    – swap two entries,
    – redistribute a small random subset,
    – pair‑transfer.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a small random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

    # Length – 4000 offers a good trade‑off between granularity and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous best, if any.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – rich seed pool (analytic + large random Dirichlet pool)
    # ------------------------------------------------------------------
    print("\n--- Seed generation (analytic) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Analytic seeds generated: {len(seeds)}")

    # Random Dirichlet draws – wide concentration sweep
    print("\n--- Random Dirichlet sampling ---")
    rand_draws = 3000
    for i in range(rand_draws):
        if time.time() >= deadline:
            break
        # concentration drawn uniformly in log‑space between 1e‑3 and ~3.2
        conc = 10 ** rng.uniform(-3, 0.5)
        vec = rng.dirichlet(np.full(n, conc))
        seeds.append(vec)
        # also keep the alternating permutation of the same draw
        seeds.append(_alternate_order(vec))
    print(f"Total seeds after random sampling: {len(seeds)}")

    # ------------------------------------------------------------------
    # Keep the top K seeds by the surrogate objective.
    # ------------------------------------------------------------------
    top_k = 30
    best_heap = []  # max‑heap of size top_k storing (-value, vector)
    for vec in seeds:
        val = _objective(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if -val > best_heap[0][0]:  # val better than current worst
                heapq.heapreplace(best_heap, (-val, vec))

    best_seeds = [item[1] for item in sorted(best_heap, key=lambda x: -x[0])]
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print("\n--- After seed selection ---")
    print(f"Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent sub‑gradient optimisation on the top seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    mirror_deadline = time.time() + remaining * 0.5   # allocate ~50 % time
    print("\n--- Mirror‑descent optimisation phase ---")
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() >= mirror_deadline:
            break
        time_left = mirror_deadline - time.time()
        # estimate feasible iterations (≈800 iters/sec on typical hardware)
        max_iters = int(min(200_000, max(5_000, time_left * 800)))
        print(f"[Mirror] seed #{idx}/{len(best_seeds)} – max_iters={max_iters}")
        seq_m, val_m = _mirror_subgradient_optimize(seed_vec, n,
                                                    time.time() + time_left,
                                                    max_iters=max_iters)
        if val_m < global_best_val:
            global_best_seq, global_best_val = seq_m.copy(), val_m
            print(f"[Overall‑Mirror] new global best = {val_m:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – focused permutation hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation hill‑climbing (focused) ---")
        swaps = 200_000
        seq_perm, val_perm = _perm_hill_climb(global_best_seq, n,
                                              deadline, max_swaps=swaps)
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[Overall‑Perm] new global best = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing polish (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(500_000, max(50_000, remaining * 800)))
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(global_best_seq, n,
                                                       deadline,
                                                       max_iters=max_iters,
                                                       init_temp_factor=0.02)
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[Overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
