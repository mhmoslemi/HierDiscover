"""
Propose a non‑negative step function that (approximately) minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Seed pool** – generate a modest set of ℓ¹‑normalised candidates from
   analytical families (uniform, triangular, geometric, beta‑shaped,
   Gaussian, Tukey, …) together with a few Dirichlet draws and the
   previous best solution (if any).  Keep the best few seeds.

2. **Smoothed gradient descent** – minimise a smooth approximation of the
   max‑convolution objective using an Adam optimiser.
   For a temperature β we use the soft‑max
       Sβ(a) = (1/β)·log Σ exp(β·(a∗a)_k)
   whose gradient is 2·corr(a, p) where
       p_k ∝ exp(β·(a∗a)_k) .
   β is increased slowly (continuation method) so the problem evolves
   from a smooth average to the true max.  After every step the vector is
   projected onto the probability simplex.

3. **Fine‑grained mirror descent** – a short entropic mirror‑descent run
   (multiplicative updates) on the exact max‑convolution objective to
   polish the solution obtained from step 2.

4. **Simulated‑annealing polishing** – a lightweight SA that repeatedly
   redistributes mass among a random subset of coordinates (Dirichlet
   move) and accepts moves according to a geometric temperature schedule.

The routine respects the supplied time budget (default 1000 s) and stores
the best sequence found in the global variable ``height_sequence_1`` for
future runs.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate objective for ℓ¹‑normalised sequences (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ¹‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12), np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _smooth_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_beta: float = 5.0,
    max_beta: float = 5000.0,
    beta_factor: float = 1.3,
    init_lr: float = 0.07,
    min_lr: float = 0.005,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑based gradient descent on a smoothed max‑convolution objective.
    ``beta`` controls the smoothness (larger ⇒ closer to true max).
    The schedule raises beta geometrically while the learning rate decays.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    # compute initial surrogate value (max conv)
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step = 0
    beta = init_beta
    lr = init_lr

    while time.time() < deadline:
        step += 1

        # convolution and soft‑max weights
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Smooth] step={step:6d} best={best_val:.6f} beta={beta:.1f}")

        # soft‑max (numerically stable)
        scaled = beta * conv
        max_scaled = np.max(scaled)
        exp_terms = np.exp(scaled - max_scaled)
        p = exp_terms / np.sum(exp_terms)          # shape (2n‑1,)

        # gradient = 2 * correlation(a, p)
        p_rev = p[::-1]
        corr_full = _conv_fft(a, p_rev)            # length 3n‑2
        start = len(p) - 1                         # =2n‑2
        g = 2.0 * corr_full[start:start + n]       # shape (n,)

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * g
        v = adam_beta2 * v + (1.0 - adam_beta2) * (g * g)
        m_hat = m / (1.0 - adam_beta1 ** step)
        v_hat = v / (1.0 - adam_beta2 ** step)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _proj_simplex(a)

        # schedule updates
        if step % 2000 == 0:
            # raise temperature, shrink learning rate
            beta = min(beta * beta_factor, max_beta)
            lr = max(lr * 0.9, min_lr)

    return best_seq, best_val


def _exp_mirror_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropic) mirror descent on the exact max‑convolution
    objective (the same routine as in the previous baseline).
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Mirror] it={it:6d} best={best_val:.6f}")

        # sub‑gradient for the active shift
        rev = max_idx - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        eta = eta0 / np.sqrt(it)          # decreasing learning rate
        a = a * np.exp(-eta * g)          # entropic step
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Light‑weight simulated‑annealing polishing that repeatedly reshuffles a
    random subset of coordinates (Dirichlet move) and accepts moves according
    to a geometric temperature schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)                # size of reshuffled block
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a plain Python list) and stores it in
    the global variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) for warm‑starting
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the candidate vector – 4000 offers a good compromise.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and quick evaluation
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smoothed gradient descent (continuation)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        smooth_deadline = time.time() + 0.55 * time_left
        print("\n--- Smoothed gradient descent (Adam + continuation) ---")
        seq_smooth, val_smooth = _smooth_descent(
            global_best_seq,
            n,
            smooth_deadline,
            init_beta=5.0,
            max_beta=5000.0,
            beta_factor=1.3,
            init_lr=0.07,
            min_lr=0.005,
        )
        if val_smooth < global_best_val:
            global_best_seq, global_best_val = seq_smooth, val_smooth
            print(f"[Smooth] new global best = {val_smooth:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – deterministic mirror descent (fine‑grained)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        mirror_deadline = time.time() + 0.30 * time_left
        print("\n--- Exponential‑mirror descent (refinement) ---")
        seq_mirror, val_mirror = _exp_mirror_descent(
            global_best_seq, n, mirror_deadline, max_iters=2_000_000, eta0=0.6
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror, val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=800_000, init_temp_factor=0.04
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (may be slightly different due to rounding)
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
