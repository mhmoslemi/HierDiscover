    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    lr = init_lr
    beta = beta0

    while time.time() < deadline and it < max_iters:
        it += 1

        # compute conv for objective
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:

            if it % 1000 == 0:

            w = np.full_like(w, 1.0 / len(w))

            w = w / w_sum

"""
propose_candidate: search for a non‑negative step function that minimises
``evaluate_sequence``.

The optimiser is built around three ideas:

1. **Rich seed pool** – analytic families (including very U‑shaped
   β‑distributions), random Dirichlet draws, sorted Dirichlet draws,
   mixtures and the previous solution.

2. **Smoothed gradient descent** – a soft‑max approximation of the
   ``max`` in the objective provides a dense sub‑gradient.
   The update is a projected gradient step on the simplex
   (Euclidean projection).  A slowly increasing temperature
   (``beta``) drives the surrogate from a smooth average to the true
   maximum.

3. **Polishing phases** – the classic exponential‑mirror descent,
   a top‑k sub‑gradient method and a simulated‑annealing move that
   reshuffles larger blocks.  All phases respect the global time
   budget (default 1000 s) and the best sequence seen so far is stored
   in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _corr_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (x ⋆ y)[k] = Σ_j x_j y_{j+k}.
    Returned length = len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    corr = np.fft.irfft(fx * np.conj(fy), n=N)[:L]
    return corr


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate objective for ℓ₁‑normalised sequences (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped density on {1,…,n}. Allows α,β < 1 (U‑shaped)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector and sort it in decreasing order (monotone)."""
    v = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    v.sort()
    v = v[::-1]  # decreasing
    return v


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12), np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    # fine beta‑shape grid, including very U‑shaped parameters
    for a in np.linspace(0.01, 0.3, 5):
        for b in np.linspace(0.01, 0.3, 5):
            seeds.append(_beta_shape(n, a, b))
    for a in np.linspace(0.4, 1.5, 4):
        for b in np.linspace(0.4, 1.5, 4):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sorted Dirichlet draws (monotone decreasing)
    for conc in (0.1, 0.5, 2.0):
        for _ in range(2):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _exp_mirror_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """Multiplicative (entropic) mirror descent on the max‑convolution objective."""
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Mirror] it={it:6d} best={best_val:.6f}")

        rev = max_idx - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _subgradient_multi(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    top_k: int = 10,
    init_lr: float = 0.30,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent averaging the gradients of the top‑k convolution entries."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)

        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        cur_max = conv[idxs].max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[MultiSub] it={it:7d} best={best_val:.6f}")

        g = np.zeros_like(a)
        for k in idxs:
            rev = k - np.arange(n)
            mask = (rev >= 0) & (rev < n)
            g[mask] += 2.0 * a[rev[mask]]
        g /= float(len(idxs))

        lr = init_lr / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _softmax_gradient_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    init_lr: float = 0.45,
    beta0: float = 1.0,
    beta_factor: float = 1.015,
    max_beta: float = 25.0,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of the max,
    using a soft‑max with temperature 1/beta.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    beta = beta0

    while time.time() < deadline and it < max_iters:
        it += 1

        # exact objective for bookkeeping
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[Soft] it={it:6d} best={best_val:.6f} beta={beta:.2f}")

        # soft‑max weights over all convolution entries
        exponents = beta * (conv - cur_max)          # max shifted to zero
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * (w ⋆ a) evaluated at the required offsets
        corr = _corr_fft(w, a)                      # length = (2n‑1) + n ‑ 1
        L = len(conv)                               # = 2n‑1
        g = 2.0 * corr[(L - 1) - np.arange(n)]

        # step size schedule
        lr = init_lr / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        # gradually sharpen the soft‑max
        beta = min(beta * beta_factor, max_beta)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Light‑weight simulated‑annealing polishing."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        # larger reshuffle blocks (2‑100 elements)
        k = rng.integers(2, 101)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start from previous run (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Problem size – 4000 offers a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and quick evaluation
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – intensive smoothed gradient descent (soft‑max)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        soft_deadline = time.time() + 0.55 * time_left
        print("\n--- Softmax‑gradient descent (smooth max) ---")
        # run on the currently best seed; if time permits also on the second one
        seq_soft, val_soft = _softmax_gradient_descent(
            global_best_seq, n, soft_deadline,
            init_lr=0.45, beta0=1.0, beta_factor=1.015, max_beta=25.0,
            max_iters=400_000
        )
        if val_soft < global_best_val:
            global_best_seq, global_best_val = seq_soft, val_soft
            print(f"[Soft] new global best = {val_soft:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – deterministic refinement (mirror + multi‑subgradient)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        mirror_deadline = time.time() + 0.15 * time_left
        print("\n--- Exponential‑mirror descent ---")
        seq_mirror, val_mirror = _exp_mirror_descent(
            global_best_seq, n, mirror_deadline,
            max_iters=2_000_000, eta0=0.6
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror, val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    time_left = deadline - time.time()
    if time_left > 0:
        sub_deadline = time.time() + 0.20 * time_left
        print("\n--- Multi‑sub‑gradient refinement (top_k=10) ---")
        seq_sub, val_sub = _subgradient_multi(
            global_best_seq, n, sub_deadline,
            top_k=10, init_lr=0.30
        )
        if val_sub < global_best_val:
            global_best_seq, global_best_val = seq_sub, val_sub
            print(f"[MultiSub] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final stochastic polishing (larger blocks)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing (large blocks) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=800_000,
            init_temp_factor=0.04
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check – evaluate with the official function.
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
