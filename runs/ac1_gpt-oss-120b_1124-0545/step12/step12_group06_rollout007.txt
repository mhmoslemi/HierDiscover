    # two-step shapes

    # Ensure positive

        val = _objective(seq)

"""
Propose a step‑function with a lower upper‑bound for the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Shape library** – generate a modest collection (≈30) of analytic
   non‑negative ℓ₁‑normalised vectors (uniform, triangular, sine‑window,
   Gaussian, power‑law, Beta, exponential and a few hand‑crafted two‑step
   shapes).

2. **Mixture optimisation** – treat a candidate as a convex combination
   of the library vectors.  Sample many Dirichlet weight vectors,
   evaluate the exact objective (via an FFT based convolution) and keep
   the best mixture.

3. **Continuous refinement** – run an Adam optimiser on the log‑softmax
   parameters of the best mixture, using a log‑sum‑exp surrogate for
   the max‑convolution.  The optimiser is stopped when the global time
   budget expires.

4. **Mirror (exponentiated‑gradient) sub‑gradient descent** – a second
   fine‑tuning step on the true max‑convolution objective.

5. **Permutation polishing** – apply a few cheap permutation heuristics
   (alternating order, sorted descending / ascending) and finish with a
   greedy hill‑climb that swaps two entries only if the objective
   improves.

All phases respect the 1000 s deadline and the best sequence found is
stored in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when ``sum(seq) == 1``.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _two_step(n: int, prop: float, factor: float) -> np.ndarray:
    """
    Simple two‑step shape: first ``prop``‑fraction of entries get
    ``factor`` times the weight of the remaining entries.
    Returned vector is ℓ₁‑normalised.
    """
    k = max(1, int(round(prop * n)))
    w = 1.0 / ((k * factor) + (n - k))
    v = factor * w
    a = np.full(n, w, dtype=np.float64)
    a[:k] = v
    return a / a.sum()


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------
def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on log‑softmax parameters using a log‑sum‑exp surrogate.
    Returns the best ℓ₁‑normalised vector seen before the deadline.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    # temperature schedule for the smoothed max
    beta = 0.5
    beta_inc = 0.5
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress output (silently suppressed in final runs)
            # print(f"[Adam] it={it} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
    max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20  # base learning rate

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # print(f"[Mirror] it={it} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional re‑ordering attempt
        if it % 100_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    # print(f"[Mirror‑alt] it={it} improved to {val_alt:.6f}")

    return best_seq, best_val


def _greedy_permutation_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_swaps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing that only swaps two entries if the objective improves.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_val = _objective(a)

    for it in range(max_swaps):
        if time.time() > deadline:
            break

        i, j = rng.choice(n, size=2, replace=False)
        a[i], a[j] = a[j], a[i]                # try the swap
        val = _objective(a)
        if val < best_val:
            best_val = val
            # keep the swap
        else:
            a[i], a[j] = a[j], a[i]            # revert

    return a, best_val


# ----------------------------------------------------------------------
# Shape library construction
# ----------------------------------------------------------------------
def _build_shape_library(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return an array of shape (m, n) containing a diverse set of ℓ₁‑normalised shapes."""
    shapes = []

    # Deterministic families
    shapes.append(_uniform(n))
    shapes.append(_triangular(n))

    # Sine windows
    for gamma in np.linspace(0.5, 5.0, 6):
        shapes.append(_sine_window(n, gamma))

    # Gaussians
    for sigma in np.linspace(0.10, 0.45, 5):
        shapes.append(_gaussian(n, sigma))

    # Power laws
    for p in np.linspace(0.0, 5.0, 6):
        shapes.append(_power(n, p))

    # Beta families (randomised)
    for _ in range(6):
        alpha = 10 ** rng.uniform(-2, 0.6)
        beta = 10 ** rng.uniform(-2, 0.6)
        shapes.append(_beta_shape(n, alpha, beta))

    # Exponentials
    for rate in np.logspace(-4, -1, 4):
        shapes.append(_exponential(n, rate))

    # Two‑step handcrafted shapes
    for prop in (0.10, 0.20, 0.30):
        for factor in (2.0, 3.0, 5.0, 10.0):
            shapes.append(_two_step(n, prop, factor))

    return np.vstack(shapes)   # shape (m, n)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The function respects the global
    time budget and stores the best found sequence in the global name
    ``height_sequence_1``.
    """
    # ------------------------------------------------------------------
    # Basic set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Length of the candidate – 4000 is a good compromise for FFT speed.
    n = 4000

    # ------------------------------------------------------------------
    # Build a library of analytic shapes
    # ------------------------------------------------------------------
    shape_lib = _build_shape_library(n, rng)   # (m, n)
    m = shape_lib.shape[0]

    # ------------------------------------------------------------------
    # Phase 1 – coarse random mixtures (Dirichlet draws)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    num_dirichlet = 12_000  # can be tuned; early break on timeout

    print("\n--- Phase 1: random mixtures ---")
    for it in range(num_dirichlet):
        if time.time() > deadline:
            break
        w = rng.dirichlet(np.ones(m))          # convex weights
        seq = np.dot(w, shape_lib)             # linear mixture
        # mixture is already ℓ₁‑normalised because both weights and shapes sum to 1
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq
            # print(f" Dirichlet #{it+1:05d} new best = {val:.6f}")

    if best_seq is None:
        # fallback – uniform shape
        best_seq = _uniform(n)
        best_val = _objective(best_seq)

    print(f"[Phase 1] best objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (smooth surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: Adam refinement ---")
        seq_adam, val_adam = _adam_optimize(best_seq, n, deadline)
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam
            best_val = val_adam
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Mirror sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: Mirror sub‑gradient descent ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(best_seq, n, deadline)
        if val_mirror < best_val:
            best_seq = seq_mirror
            best_val = val_mirror
            print(f"[Mirror] improved to {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Permutation polishing (order tricks + greedy swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: permutation polishing ---")
        # try a handful of deterministic reorderings
        candidates = [
            best_seq,
            _alternate_order(best_seq),
            _sorted_desc(best_seq),
            _sorted_asc(best_seq),
        ]
        for cand in candidates:
            val = _objective(cand)
            if val < best_val:
                best_seq, best_val = cand, val
                print(f"[Perm‑fixed] improved to {val:.6f}")

        # final greedy swap hill‑climb
        seq_perm, val_perm = _greedy_permutation_optimize(best_seq, n, deadline)
        if val_perm < best_val:
            best_seq, best_val = seq_perm, val_perm
            print(f"[Perm‑greedy] improved to {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    final_val = None
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective (internal) ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
