"""
Improved step‑function optimisation.

The search builds a diverse seed pool (analytic families,
Dirichlet draws, random spike patterns) and keeps the most promising
candidates.  It then spends the allotted time in three complementary
optimisation phases:

1. **Adam on a log‑sum‑exp surrogate** – many short runs from the top
   seeds with an aggressive temperature schedule (β up to 2000) to
   quickly locate good basins.

2. **Projected‑gradient sub‑gradient descent** – optimisation on the
   exact max‑convolution objective using Euclidean projection onto the
   simplex.  This often reduces the objective further than mirror‑descent.

3. **Simulated‑annealing + greedy hill‑climb** – a final polishing
   phase that swaps and redistributes mass while accepting occasional
   uphill moves.

All phases respect the global time budget (default 1000 s) and keep the
best feasible sequence in the global variable ``height_sequence_1``.
The returned list can be passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) for a sequence with sum=1."""
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


def _spike_pattern(n: int, num_spikes: int, rng: np.random.Generator) -> np.ndarray:
    """A few random spikes (non‑zero entries) distributed uniformly."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=num_spikes, replace=False)
    a[idx] = rng.random(num_spikes) + 1e-6
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # sine windows (various gammas) + alternated order
    for gamma in np.linspace(0.5, 6.0, 12):
        s = _sine_window(n, gamma)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Gaussians
    for sigma in np.linspace(0.10, 0.55, 10):
        g = _gaussian(n, sigma)
        seeds.append(g)
        seeds.append(_alternate_order(g))

    # power laws
    for p in np.linspace(0.0, 8.0, 17):
        pw = _power(n, p)
        seeds.append(pw)
        seeds.append(_alternate_order(pw))

    # beta family
    for _ in range(15):
        alpha = 10 ** rng.uniform(-2, 0.6)
        beta = 10 ** rng.uniform(-2, 0.6)
        b = _beta_shape(n, alpha, beta)
        seeds.append(b)
        seeds.append(_alternate_order(b))

    # exponentials
    for rate in np.logspace(-4, -1, 12):
        e = _exponential(n, rate)
        seeds.append(e)
        seeds.append(_alternate_order(e))

    # logarithmic
    for eps in np.logspace(-4, -1, 8):
        lg = _logarithmic(n, eps)
        seeds.append(lg)
        seeds.append(_alternate_order(lg))

    # Dirichlet draws – many concentrations & orderings
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(8):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw
            seeds.append(_alternate_order(raw))
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(curve)
            seeds.append(_alternate_order(curve))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = _mix_shapes(_uniform(n), curve, mix_alpha)
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Spike patterns – very sparse constructions
    for spikes in (5, 10, 20, 50, 100):
        for _ in range(3):
            sp = _spike_pattern(n, spikes, rng)
            seeds.append(sp)
            seeds.append(_alternate_order(sp))

    # Random permutations of a few good deterministic shapes
    for base in [_gaussian(n, 0.2), _power(n, 3.0), _beta_shape(n, 0.5, 0.5)]:
        for _ in range(5):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    # Previous solution, if any, normalised
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(
    start_vec: np.ndarray, n: int, deadline: float, max_iters: int = 4_000_000
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best ℓ₁‑normalised vector seen and its exact objective.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    # temperature schedule (β in the log‑sum‑exp)
    beta = 0.5
    beta_inc = 0.6
    beta_max = 2000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Adam] it={it:,d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # chain‑rule for softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _pgd_subgradient_optimize(
    start_seq: np.ndarray, n: int, deadline: float, max_iters: int = 5_000_000
) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent on the exact max‑convolution objective.
    Uses Euclidean projection onto the simplex after each sub‑gradient step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    lr0 = 0.5
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[PGD] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient w.r.t. a at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        lr = lr0 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with swaps, subset redistribution and pair‑transfer moves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new, n)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


def _greedy_permutation_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_swaps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that accepts only improving swaps.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_val = _objective(a, n)

    for it in range(max_swaps):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        a[i], a[j] = a[j], a[i]            # tentative swap
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            # keep the swap
        else:
            a[i], a[j] = a[j], a[i]        # revert

    return a, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    The best sequence is stored in the global name ``height_sequence_1``
    and also returned as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # small safety margin

    # ------------------------------------------------------------------
    # Length – 4000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve a previous solution if present.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Seed generation & quick screening
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec, n)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seeds (exact objective):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq, n)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (≈30 % of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_budget = remaining * 0.30
    adam_deadline = time.time() + adam_budget

    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > adam_deadline:
            break
        per_seed_deadline = time.time() + (adam_deadline - time.time()) / max(1, len(best_seeds) - idx + 1)
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, per_seed_deadline)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # try alternating order of the Adam result
        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt, n)
            if alt_val < global_best_val:
                global_best_seq = alt
                global_best_val = alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected‑gradient sub‑gradient descent (≈40 % of remaining)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Projected‑gradient sub‑gradient optimisation ---")
        pgd_deadline = time.time() + (deadline - time.time()) * 0.40
        seq_pgd, val_pgd = _pgd_subgradient_optimize(global_best_seq, n, pgd_deadline)
        if val_pgd < global_best_val:
            global_best_seq = seq_pgd.copy()
            global_best_val = val_pgd
            print(f"[overall‑PGD] new global best = {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing polishing (≈20 % of remaining)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        sa_deadline = time.time() + (deadline - time.time()) * 0.20
        seq_sa, val_sa = _simulated_annealing_optimize(global_best_seq, n, sa_deadline)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Greedy permutation hill‑climb (spare time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Greedy permutation optimisation ---")
        perm_deadline = time.time() + (deadline - time.time())
        seq_perm, val_perm = _greedy_permutation_optimize(global_best_seq, n, perm_deadline)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[overall‑Perm] new global best = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
