"""
Propose Candidate – “Projected sub‑gradient + soft‑max smoothing”

Algorithm Overview
------------------
1. **Seed pool** – generate a rich set of ℓ₁‑normalised candidates:
   uniform, triangular, sine‑windows, Gaussian, beta‑shapes,
   Tukey windows, random Dirichlet/exponential draws and low‑frequency
   cosine mixtures.  The already‑found solution (global
   ``height_sequence_1``) is also inserted.

2. **Fast PGD stage** – for the top K seeds run a *projected
   sub‑gradient descent* on the true surrogate 2·n·max(conv).  The
   sub‑gradient corresponds to the active shift (the index where the
   convolution attains its maximum).  Additive updates are followed
   by Euclidean projection onto the probability simplex.

3. **Long‑run PGD polishing** – continue the above descent on the
   current best vector for as many iterations as time permits.  A
   simple 1/√t step‑size schedule is used.

4. **Soft‑max refinement** – once the PGD phase has converged,
   switch to a smooth approximation of the max using a soft‑max
   (log‑sum‑exp) with an annealing parameter α that grows from 1 to
   200.  The gradient of the smoothed objective is computed via two
   FFT‑based convolutions, then a projected gradient step is taken.

5. **Optional simulated‑annealing tweak** – a light SA phase
   reshuffles mass on random subsets and performs tiny transfers;
   any improvement in the surrogate is accepted.

6. **Final evaluation** – the best vector is handed to the official
   ``evaluate_sequence`` function, stored globally as
   ``height_sequence_1`` and returned as a plain ``list``.

All heavy work is done with NumPy/FFT; the routine respects the
global deadline (default 1000 s) and never performs file or network I/O.
"""

import time
import math

# ----------------------------------------------------------------------
# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector w ≥ 0 with sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) – assumes Σseq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (always ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    """Monotonically decreasing exponential, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, then clip to non‑negative and normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse list of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0, 4.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.25, 0.3):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        seeds.append(_power(n, p))
    # beta grid (coarser)
    for a in (0.5, 1.0, 2.0, 3.0, 5.0):
        for b in (0.5, 1.0, 2.0, 3.0, 5.0):
            seeds.append(_beta_shape(n, a, b))
    # Tukey windows
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet (varied concentration)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous best, if available
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final safety normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        clean.append(v / s if s > 0 else _uniform(n))
    return clean


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (PGD) on the max‑convolution surrogate
# ----------------------------------------------------------------------
def _pgd_optimize(start: np.ndarray, n: int, deadline: float,
                  max_iters: int = 250_000, eta0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Additive projected sub‑gradient descent.
    Returns the best sequence seen and its surrogate value.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute full auto‑convolution
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))          # active shift (max index)

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # additive update + simplex projection
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[PGD] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent – annealing version
# ----------------------------------------------------------------------
def _softmax_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 100_000,
                      eta0: float = 2.0,
                      alpha_start: float = 1.0,
                      alpha_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp surrogate:
        f_α(x) = (1/α)·log Σ_k exp(α·conv_k(x,x))

    The gradient is 2·(w * flip(x)) where w are the soft‑max weights.
    The algorithm gradually raises α (smooth→hard max) while doing
    projected gradient steps.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current α (linear schedule)
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max_iters

        # convolution of current iterate with itself
        conv = _conv_fft(x, x)

        # soft‑max weights (stable)
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        expw = np.exp(scaled - max_scaled)
        w = expw / np.sum(expw)                       # shape: (2·n‑1)

        # gradient = 2·conv(w, flip(x)) restricted to original length
        grad_full = _conv_fft(w, np.flip(x))
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # step size schedule (1/√t)
        eta = eta0 / math.sqrt(it)

        # additive update + simplex projection
        x = x - eta * grad
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 20000 == 0:
                print(f"[SoftMax] it={it:7d} α={alpha:.1f} best={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (light version)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # geometric cooling
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            # reshuffle a random subset (size 3‑12)
            k = rng.integers(3, 13)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    `evaluate_sequence` objective within the time budget.
    Returns the best sequence as a plain list of Python floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – 4000 works well and stays inside the time budget.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # keep the most promising ones (according to the surrogate)
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – quick PGD on a few top seeds.
    # --------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick PGD refinement on top seeds ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # about 0.0004 s per iteration → allocate a modest budget
            max_it = int(min(40_000, remaining * 2000))
            cand, val = _pgd_optimize(seed_vec, n, deadline,
                                      max_iters=max_it, eta0=0.25)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick PGD] seed #{idx+1} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – long‑run PGD polishing on the current best.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long‑run PGD polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1500))
        best_seq, best_val = _pgd_optimize(best_seq, n, deadline,
                                           max_iters=max_iters, eta0=0.20)
        print(f"[Long PGD] best surrogate after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – soft‑max gradient refinement (annealing).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Soft‑max refinement (gradual α‑annealing) ---")
        remaining = deadline - time.time()
        max_iters = int(min(120_000, remaining * 250))
        best_seq, best_val = _softmax_optimize(best_seq, n, deadline,
                                               max_iters=max_iters,
                                               eta0=2.0,
                                               alpha_start=1.0,
                                               alpha_end=200.0)
        print(f"[Soft‑max] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – optional simulated‑annealing final tweak.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_iters)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store globally for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
