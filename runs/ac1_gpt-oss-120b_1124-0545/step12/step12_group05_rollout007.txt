def simplex_proj(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose – a fast multi‑stage optimizer for the step‑function problem.

Algorithm
---------
1️⃣  Build a diverse pool of analytic “seed” shapes (uniform, triangular,
    sine‑window, Gaussian, Beta, Laplace, Dirichlet draws, random Fourier
    mixtures).  All seeds are L₁‑normalised (sum = 1).

2️⃣  Keep the best few seeds (by the surrogate 2·n·max(conv)).

3️⃣  For every top seed run a long Adam optimiser on the *smooth* max‑convolution
    (log‑sum‑exp temperature).  The parameters are kept on the log‑softmax
    simplex, guaranteeing non‑negativity.

4️⃣  Starting from the best Adam result run a **projected sub‑gradient descent**
    on the *exact* max‑convolution.  The sub‑gradient is non‑zero only for the
    active convolution index.  After each Euclidean step we project back onto
    the simplex (the usual ‖·‖₂‑projection).

5️⃣  Polish the obtained vector with a **simulated‑annealing** that repeatedly
    performs three move types:
        – swap two entries,
        – redistribute the total mass of a random subset (Dirichlet draw),
        – transfer mass between a random pair.
    After each move we accept it if it improves the objective or with a
    Metropolis probability that slowly cools.

6️⃣  Return the best sequence (as a list of floats) and store it as the global
    variable ``height_sequence_1`` for possible reuse.

All stages respect the global 1000 s time budget.  The implementation works
with only NumPy (and the supplied ``evaluate_sequence``) and is deliberately
written with no closures or lambdas so that it can be safely executed in the
challenge environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact numerator of evaluate_sequence when sum(seq)=1:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all numbers are zero‑ish – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return w / w_sum


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Re‑order as large, small, 2nd‑large, 2nd‑small, … and renormalise."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    return _simplex_projection(out)


# ----------------------------------------------------------------------
# Analytic shape families – all return L₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return _simplex_projection(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _simplex_projection(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _simplex_projection(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    return _simplex_projection(a)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    return _simplex_projection(a)


def _double_gaussian(n: int, sigma_frac: float = 0.07,
                     offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    return _simplex_projection(a)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    return _simplex_projection(a)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    return _simplex_projection(a)


def _random_fourier(n: int, rng: np.random.Generator,
                    nfreq: int = 3) -> np.ndarray:
    """Sum of a few cosines with random phase – then rectify and renormalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(nfreq):
        freq = rng.integers(1, n // 2)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.0)
    return _simplex_projection(a)


# ----------------------------------------------------------------------
# Seed generation – richer than the reference
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))

    # Sine windows (varying shape)
    for gamma in np.linspace(0.5, 6.0, 12):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussians
    for sigma in np.linspace(0.07, 0.30, 12):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # Beta family – broad parameter sweep
    for _ in range(30):
        alpha = 10 ** rng.uniform(-3, 0.9)   # 0.001 … ~7.9
        beta = 10 ** rng.uniform(-3, 0.9)
        vec = _beta_shape(n, alpha, beta)
        seeds.append(vec)
        seeds.append(_alternate_order(vec))

    # Exponential shapes
    for rate in np.logspace(-5, -1, 12):
        vec = _exponential(n, rate)
        seeds.append(vec)
        seeds.append(_alternate_order(vec))

    # Logarithmic shapes
    for eps in np.logspace(-5, -1, 10):
        vec = _logarithmic(n, eps)
        seeds.append(vec)
        seeds.append(_alternate_order(vec))

    # Dirichlet draws (wide range of concentrations)
    concs = [0.001, 0.005, 0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_alternate_order(raw))

    # Random Fourier mixtures – produce surprising patterns
    for _ in range(12):
        seeds.append(_random_fourier(n, rng, nfreq=rng.integers(2, 6)))
        seeds.append(_alternate_order(_random_fourier(n, rng,
                                                       nfreq=rng.integers(2, 6))))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            normalised = avg / avg.sum()
            seeds.append(normalised)
            seeds.append(_alternate_order(normalised))

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int,
                   deadline: float, max_iters: int) -> tuple[np.ndarray, float]:
    """Adam on log‑softmax parameters using a log‑sum‑exp surrogate."""
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits – stay on simplex via softmax
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    # Temperature schedule for smooth max
    beta = 0.5            # start temperature
    beta_inc = 0.5
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum conv w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _projected_subgradient(start_seq: np.ndarray, n: int,
                           deadline: float,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.50          # initial step size (larger than reference)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[PGD] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index (mirror of Mirror‑subgradient)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Euclidean step and projection onto simplex
        lr = lr0 / np.sqrt(it)
        a = a - lr * g
        a = _simplex_projection(a)

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 3_000_000,
                                 init_temp_factor: float = 0.02,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polish with three move types (swap, subset‑redistribute,
    pair‑transfer).  No explicit temperature schedule – exponential cooling.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a small random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence (list of floats) and stores it in the global
    variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Length – 2000 offers a very good speed / quality trade‑off.
    # --------------------------------------------------------------
    n = 2000

    # --------------------------------------------------------------
    # Load previous best, if any, and normalise it.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0.0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Phase 1 – generate a diverse seed pool.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise the global best with the absolute best seed.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smoothed objective.
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest number of Adam iterations based on remaining time
        max_iters = int(min(150_000, remaining * 35))   # ~35 iters / sec
        max_iters = max(20_000, max_iters)
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam.copy(), val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # Try alternating order of the Adam result – often reduces the peak
        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq, global_best_val = alt, alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent (Euclidean) for deep polishing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Projected sub‑gradient descent (PGD) ---")
        seq_pgd, val_pgd = _projected_subgradient(
            global_best_seq, n, deadline, max_iters=4_000_000
        )
        if val_pgd < global_best_val:
            global_best_seq, global_best_val = seq_pgd.copy(), val_pgd
            print(f"[overall‑PGD] new global best = {val_pgd:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated‑annealing polishing (heavy final sweep).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=4_000_000,
            init_temp_factor=0.025,
            min_temp=1e-13
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final sanity‑check with the official evaluator.
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
