def _smooth_max_adam(start_seq, n, deadline, beta0=1.0, beta_max=200.0, beta_growth=1.001, lr=0.01, eps=1e-12):
    a = start_seq.copy()
    # ensure sum to 1
    a = a / a.sum()
    # initialize z = np.log(a + eps)
    z = np.log(a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta = beta0
    t = 0
    best_seq = a.copy()
    best_obj = _objective(a)  # surrogate
    while time.time() < deadline:
        t += 1
        # compute a from z
        expz = np.exp(z - np.max(z))  # stability
        a = expz / expz.sum()
        # compute conv
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        # compute smooth max and gradient
        # if beta too large, may cause overflow, limit.
        max_val = max_conv
        shifted = conv - max_val
        # compute exponentials safely
        exp_beta = np.exp(beta * shifted)
        sum_exp = exp_beta.sum()
        weights = exp_beta / sum_exp  # size 2n-1, sum=1
        # compute gradient of smoothed max w.r.t a
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, weights)
        start = n - 1
        grad_sm = 2.0 * conv_rev[start:start + n]
        # full objective gradient
        grad_obj = 2.0 * n * grad_sm
        # gradient w.r.t z
        grad_z = a * (grad_obj - np.dot(grad_obj, a))
        # Adam updates
        m = 0.9 * m + 0.1 * grad_z
        v = 0.999 * v + 0.001 * (grad_z ** 2)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # update best based on true max conv (surrogate)
        cur_obj = 2.0 * n * max_conv
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
        # increase beta gradually
        beta = min(beta_max, beta * beta_growth)
        # optionally break after some max_iters
    return best_seq, best_obj

"""
search function that uses a smooth-max Adam optimizer to flatten the auto‑correlation
of a non‑negative step function, combined with a short simulated‑annealing polish.
The optimizer works on the probability simplex via a soft‑max parametrisation
(a_i = exp(z_i)/∑exp(z_i)).  A temperature β controls a smooth approximation of
the max‑convolution; β is increased during the run so that the optimisation
first sees a smooth landscape and gradually focuses on the true
max‑convolution.  Gradient computation is performed with two FFT‑based
convolutions per iteration, giving O(n log n) per step.  The routine respects
the supplied time budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (same as before)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate objective for ℓ₁‑normalised sequences (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12), np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _adam_smooth_max(start_seq: np.ndarray, n: int, deadline: float,
                     beta0: float = 0.5,
                     beta_max: float = 200.0,
                     beta_growth: float = 1.001,
                     lr: float = 0.05,
                     eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of the max‑convolution.
    The variables are kept on the simplex via a soft‑max (z‑parametrisation).
    ``beta`` controls the smoothness; it starts low and is increased
    geometrically up to ``beta_max``.
    Returns the best sequence seen and its surrogate objective.
    """
    # initialise
    a = start_seq.copy()
    a = a / a.sum()
    # log‑space parameters (add eps to avoid log(0))
    z = np.log(a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta = beta0
    best_seq = a.copy()
    best_val = _objective(a)

    t = 0
    while time.time() < deadline:
        t += 1

        # recompute a from current z (soft‑max)
        max_z = np.max(z)
        expz = np.exp(z - max_z)          # stability
        a = expz / expz.sum()

        # full convolution (FFT)
        conv = _conv_fft(a, a)            # length 2n-1
        max_conv = conv.max()

        # smooth max (log‑sum‑exp trick)
        shift = conv - max_conv           # <=0
        exp_beta = np.exp(beta * shift)   # safe, all <=1
        sum_exp = exp_beta.sum()
        weights = exp_beta / sum_exp       # softmax over conv indices, sum = 1

        # gradient of the smooth max w.r.t a
        a_rev = a[::-1]

        conv_rev = _conv_fft(a_rev, weights)   # length n + (2n-1) -1 = 3n-2
        start = n - 1
        grad_sm = 2.0 * conv_rev[start:start + n]   # ∂/∂a of smoothed max

        # full objective gradient (objective = 2 n * smoothed_max)
        grad_obj = 2.0 * n * grad_sm

        # chain rule to obtain gradient w.r.t z (soft‑max Jacobian)
        grad_z = a * (grad_obj - np.dot(grad_obj, a))

        # Adam update
        b1, b2 = 0.9, 0.999
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep track of best (using the true max‑convolution)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # increase smoothing temperature
        beta = min(beta_max, beta * beta_growth)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 600_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Light‑weight simulated‑annealing polishing that reshuffles a random
    block of coordinates via a Dirichlet move.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)          # block size
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _next_pow2(x: int) -> int:

    a = start_seq.copy()
    a = a / a.sum()
    z = np.log(a + eps)                     # initialise log‑space variables

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta = beta0
    best_seq = a.copy()
    best_val = _objective(a)

    t = 0
    while time.time() < deadline:
        t += 1

        # soft‑max to obtain a on the simplex
        max_z = np.max(z)
        expz = np.exp(z - max_z)            # stability shift
        a = expz / expz.sum()

        # full convolution (FFT)
        conv = _conv_fft(a, a)               # length 2n‑1
        max_conv = conv.max()

        # smooth max via log‑sum‑exp
        shift = conv - max_conv              # ≤0
        exp_beta = np.exp(beta * shift)      # safely ≤1
        sum_exp = exp_beta.sum()
        weights = exp_beta / sum_exp          # softmax over conv indices

        # gradient of smoothed max w.r.t a
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, weights)  # length 3n‑2
        start = n - 1
        grad_sm = 2.0 * conv_rev[start:start + n]   # ∂/∂a of smoothed max

        # full objective gradient (objective = 2·n·smoothed_max)
        grad_obj = 2.0 * n * grad_sm

        # chain rule through the soft‑max parametrisation
        grad_z = a * (grad_obj - np.dot(grad_obj, a))

        # Adam update
        b1, b2 = 0.9, 0.999
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # track best (true max‑convolution objective)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # increase temperature (more focus on the true max)
        beta = min(beta_max, beta * beta_growth)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 600_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Light‑weight simulated‑annealing polishing that reshuffles a random
    block of coordinates via a Dirichlet move.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)                     # block size
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, starts from a diverse pool of analytic
    and random seeds, refines each promising seed with a smooth‑max Adam
    optimiser, and finishes with a short simulated‑annealing polish.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from previous run, if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the candidate vector – 4000 offers a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & quick surrogate evaluation
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    seed_vals = [(vec, _objective(vec)) for vec in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(5, len(seed_vals))
    best_seeds = [vec for vec, _ in seed_vals[:top_k]]

    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(seed_vals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max Adam optimisation on the most promising seeds
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        per_seed_budget = 0.70 * time_left / len(best_seeds)
        for idx, seed_vec in enumerate(best_seeds):
            seed_deadline = time.time() + per_seed_budget
            print(
                f"\n--- Adam optimisation on seed #{idx+1} "
                f"(budget {per_seed_budget:.1f}s) ---"
            )
            seq_adam, val_adam = _adam_smooth_max(
                seed_vec, n, seed_deadline,
                beta0=0.5, beta_max=200.0, beta_growth=1.001,
                lr=0.05
            )
            if val_adam < global_best_val:
                global_best_seq, global_best_val = seq_adam, val_adam
                print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing (short SA)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing (final) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=600_000,
            init_temp_factor=0.04
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible warm‑start in a future run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
