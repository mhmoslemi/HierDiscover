      m = beta1 * m + (1 - beta1) * g

      v = beta2 * v + (1 - beta2) * (g * g)

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------

A three‑stage global optimiser works within the 1000 s budget.

1. **Seed generation** – a diverse pool of analytically‑generated and random
   ℓ₁‑normalised vectors (uniform, triangular, trapezoidal, β‑shapes,
   geometric, power‑law, Gaussian, Dirichlet draws, …).  The previous best
   solution (if any) is also added.

2. **Differential‑Evolution (DE) global search** – a modest population (≈20)
   evolves by mutation ``a + F·(b‑c)`` and binomial crossover.  After each
   trial the vector is projected onto the probability simplex (negative
   entries are set to zero and the result is renormalised).  The DE run takes
   most of the time budget and discovers a strong basin of attraction.

3. **Local refinement** – the best DE individual is polished with two
   complementary methods:

   * Adam‑style projected gradient descent on the *max‑convolution* (the
     active shift provides a sub‑gradient).  The step‑size decays as
     ``lr/√t`` and the iterate is projected onto the simplex each step.
   * Light‑weight simulated‑annealing that reshuffles a random block of
     coordinates via a Dirichlet move.

The routine respects the user‐supplied time limit, prints progress, stores
the final vector in ``height_sequence_1`` and returns it as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(N log N))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective for a ℓ₁‑normalised sequence (sum == 1):
    2·n·max(conv(seq,seq)).
    This coincides with ``evaluate_sequence`` because the latter is
    homogeneous of degree 0.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple projection onto the simplex (positivity + unit ℓ₁‑norm)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp negatives to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fall back to the uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Right‑hand geometric series (r∈(0,1))."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return _project_to_simplex(a)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Create a modest collection of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12), np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – a handful of concentrations
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        avg = _project_to_simplex(avg)
        seeds.append(avg)

    return seeds


# ----------------------------------------------------------------------
# Differential Evolution (global search)
# ----------------------------------------------------------------------


def _differential_evolution(
    init_pop: list[np.ndarray],
    n: int,
    deadline: float,
    popsize: int = 20,
    max_gens: int = 10_000,
    f_low: float = 0.5,
    f_high: float = 0.9,
    cr: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Simple DE on the simplex (positivity + unit sum).  The mutation factor
    ``F`` is drawn uniformly from [f_low, f_high] for each trial.
    """
    rng = np.random.default_rng()
    # initialise population
    pop = [vec.copy() for vec in init_pop[:popsize]]
    while len(pop) < popsize:
        pop.append(_project_to_simplex(rng.random(n)))
    fitness = [_objective(v) for v in pop]

    best_idx = int(np.argmin(fitness))
    best_seq = pop[best_idx].copy()
    best_val = fitness[best_idx]

    gen = 0
    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # pick three distinct individuals different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a, b, c = pop[a_idx], pop[b_idx], pop[c_idx]

            F = rng.uniform(f_low, f_high)
            mutant = a + F * (b - c)
            mutant = _project_to_simplex(mutant)

            # binomial crossover
            cross_mask = rng.random(n) < cr
            # guarantee at least one element from mutant
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = _project_to_simplex(trial)

            trial_val = _objective(trial)

            if trial_val < fitness[i]:
                pop[i] = trial
                fitness[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_seq = trial.copy()
                    # occasional progress report
                    if gen % 200 == 0:
                        print(f"[DE] gen={gen:5d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (local refinement)
# ----------------------------------------------------------------------


def _adam_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_lr: float = 0.10,
    beta1: float = 0.9,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv) using Adam for the active (max) shift.
    The iterate is kept on the simplex after each update.
    """
    a = start_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active shift (see analysis)
        rev = max_idx - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = init_lr / np.sqrt(t)  # mild decay

        a = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = _project_to_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 10_000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated Annealing polishing (lightweight)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Dirichlet‑move block annealing."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)

    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9993, min_temp)

        # pick a random block (size 2‑12)
        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        # redistribute mass inside the block
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # ==================================================================
    # Retrieve previous solution (if any)
    # ==================================================================
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ==================================================================
    # Choose length – 4000 has proven a good trade‑off
    # ==================================================================
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed creation
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)

    # evaluate all seeds (very cheap)
    seed_vals = [_objective(v) for v in seeds]
    # keep the top few for the initial population
    top_k = 10
    top_indices = np.argsort(seed_vals)[:top_k]
    init_pop = [seeds[i] for i in top_indices]

    # ensure we have enough individuals
    while len(init_pop) < 20:
        init_pop.append(_project_to_simplex(rng.random(n)))

    global_best_seq = init_pop[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Differential Evolution (global search)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        de_deadline = time.time() + 0.60 * time_left
        print("\n--- Differential Evolution (global) ---")
        seq_de, val_de = _differential_evolution(
            init_pop, n, de_deadline, popsize=20, max_gens=20_000
        )
        if val_de < global_best_val:
            global_best_seq, global_best_val = seq_de, val_de
            print(f"[DE] new global best = {val_de:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Adam‑style local refinement
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        adam_deadline = time.time() + 0.25 * time_left
        print("\n--- Adam‑style projected gradient descent ---")
        seq_adam, val_adam = _adam_refine(
            global_best_seq, n, adam_deadline, init_lr=0.12
        )
        if val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam, val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=800_000
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
