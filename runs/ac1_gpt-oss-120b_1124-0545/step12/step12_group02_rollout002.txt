"""
Improved multi‑stage optimizer for the step‑function objective

Goal: minimise 2·n·max(conv(a,a))  (a_i ≥ 0, Σa_i = 1).

The optimiser builds a large, diverse pool of analytic and random “seed” windows,
keeps the best few (using a cheap ℓ₁‑normalised surrogate) and then refines the
candidate with several complementary local‑search phases:

1️⃣  **Adam on a smoothed max** – log‑sum‑exp smoothing with a temperature
    that grows up to 1e⁵.  This provides a fast descent while still
    approximating the true L∞‑objective.

2️⃣  **Projected sub‑gradient descent** – works directly on the exact max‑entry
    (no smoothing), with a decreasing step‑size and Euclidean projection onto
    the probability simplex.

3️⃣  **Exponentiated‑gradient (mirror‑descent)** – similar to (2) but on the
    log‑softmax parametrisation; helps escape shallow plateaux.

4️⃣  **Simulated‑annealing polishing** – random mass reshuffles on
    small subsets, with an aggressively cooling schedule.

5️⃣  **Restart loop** – while time permits we repeatedly draw a fresh random
    seed (Dirichlet, sparse spikes, low‑frequency curves) and run a short
    “micro‑optimisation” (≈ 5 k Adam + 5 k sub‑gradient).  The best result of all
    restarts becomes the starting point for a final long‑run optimisation.

All stages respect the global deadline (default 1000 s).  The best sequence
found is stored in the global name ``height_sequence_1`` and returned as a list
of floats.
"""

import time
import math
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog  # optional, used only for tiny sanity checks

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(seq,seq)) for a vector that is ℓ₁‑normalised
    (Σseq = 1).  The returned value equals the true evaluate_sequence value
    (up to the clipping performed by that function) and preserves ordering.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid (keeps the shape) and renormalises."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k random positions receive exponential mass, the rest stay zero."""
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = vals
    s = seq.sum()
    if s == 0.0:
        return _uniform(n)
    return seq / s


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few random cosine components, clip to non‑negative and renorm."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, then renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.20))

    # beta family grid (coarse)
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws with many concentration parameters
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse spikes
    for k in (5, 10, 20, 40):
        for _ in range(3):
            seeds.append(_random_sparse(n, k, rng))

    # low‑frequency random curves
    for nf in (2, 3, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        curve = _lowfreq_curve(n, 3, rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution, if any
    if previous is not None and previous.size == n:
        seeds.append(previous / previous.sum())

    # ensure every seed is ℓ₁‑normalised and non‑negative
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 1_000_000,
                   beta_max: float = 1e5) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑softmax parametrisation with an aggressive
    temperature schedule (β → beta_max). Returns the best vector found
    (ℓ₁‑normalised) and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5
    beta_inc = 0.5
    lr0 = 1.0

    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        a = _softmax(v)                     # ℓ₁‑normalised
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smoothed max (log‑sum‑exp)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient on logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Works in the primal domain with Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)          # diminishing step size
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Operates on the log‑softmax representation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.08,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.65:          # reshuffle a random subset
            k = rng.integers(3, 15)          # size 3 … 14
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # tiny pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.03)  # ≤3 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    overall_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Stage 0 – seed generation (coarse resolution)
    # ------------------------------------------------------------------
    n0 = 4000
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print(f"\n--- Seed generation (n={n0}) ---")
    seeds = _collect_seeds(n0, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # cheap surrogate ranking
    evaluations = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluations.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in evaluations[:top_k]]

    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluations[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample if time permits
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    up_factor = 2            # we aim for ~8000 points
    n_up = n0 * up_factor
    if remaining > 25.0:
        print(f"\n--- Upsampling to n={n_up} ---")
        global_best_seq = _upsample_sequence(global_best_seq, n_up)
        global_best_seq = global_best_seq / global_best_seq.sum()
        global_best_val = _objective_surrogate(global_best_seq)
        print(f"[Upsample] surrogate after interpolation = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long Adam optimisation (first pass)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 60.0:
        max_iters = int(min(800_000, remaining * 4000))
        print(f"\n--- Adam optimisation (first pass, max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters, beta_max=1e5
        )
        if val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam, val_adam
            print(f"[Adam‑1] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – sub‑gradient projection
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        max_iters = int(min(600_000, remaining * 2000))
        print(f"\n--- Sub‑gradient projection (max_iters={max_iters}) ---")
        seq_sub, val_sub = _subgradient_optimize(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters
        )
        if val_sub < global_best_val:
            global_best_seq, global_best_val = seq_sub, val_sub
            print(f"[SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 20.0:
        max_iters = int(min(500_000, remaining * 1500))
        print(f"\n--- Exponentiated‑gradient refinement (max_iters={max_iters}) ---")
        seq_exp, val_exp = _exp_gradient_optimize(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters
        )
        if val_exp < global_best_val:
            global_best_seq, global_best_val = seq_exp, val_exp
            print(f"[ExpGrad] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 15.0:
        max_iters = int(min(1_000_000, remaining * 1200))
        print(f"\n--- Simulated annealing polishing (max_iters={max_iters}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA‑polish] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – micro‑restarts (quick bursts) while time permits
    # ------------------------------------------------------------------
    while True:
        remaining = overall_deadline - time.time()
        if remaining < 12.0:
            break

        # generate a fresh random seed (Dirichlet or sparse)
        if rng.random() < 0.6:
            conc = rng.choice([0.03, 0.07, 0.15, 0.3])
            seed_vec = _random_dirichlet(n_up, conc, rng)
        else:
            k = rng.integers(5, 30)
            seed_vec = _random_sparse(n_up, k, rng)

        # a very short Adam + sub‑gradient combo (≈ 5 k iters each)
        mini_deadline = time.time() + min(8.0, remaining * 0.5)

        seq_a, val_a = _adam_optimize(
            seed_vec, n_up, mini_deadline, max_iters=8_000, beta_max=1e4
        )
        seq_s, val_s = _subgradient_optimize(
            seq_a, n_up, mini_deadline, max_iters=8_000
        )
        if val_s < global_best_val:
            global_best_seq, global_best_val = seq_s, val_s
            print(f"[Restart] improved global best = {val_s:.6f}")

    # ------------------------------------------------------------------
    # Optional final long Adam run (use any remaining time)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        max_iters = int(min(300_000, remaining * 4000))
        print(f"\n--- Final Adam run (max_iters={max_iters}) ---")
        seq_final, val_final = _adam_optimize(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters, beta_max=2e5
        )
        if val_final < global_best_val:
            global_best_seq, global_best_val = seq_final, val_final
            print(f"[FinalAdam] new global best = {val_final:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
