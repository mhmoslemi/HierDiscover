"""
Optimiser for the “step‑function” challenge.

The algorithm builds a large and diverse pool of analytically‑oriented seeds,
selects the most promising ones (according to the fast surrogate
2·n·max(conv)), and then refines them with a series of optimisation stages:

1. **Heavy L‑BFGS on a smooth‑max surrogate** – a log‑sum‑exp
   approximation of the max‑convolution is minimised with L‑BFGS.
   Temperatures are increased (10 → 30 → 100 → 300 → 800 → 1500) and
   the optimiser is allowed up to 300 iterations per temperature.

2. **Extended exponentiated‑gradient (mirror) descent** – an exact
   sub‑gradient of the true max‑convolution is used together with a
   multiplicative update on the simplex.  The step‑size is larger than in
   the reference implementation (η₀=0.20) and the decay exponent is
   slightly slower (0.30) to enable deeper polishing.

3. **Long simulated‑annealing** – a SA pass with three move types
   (swap, Dirichlet‑redistribute, pair‑transfer) runs for up to 2 M
   iterations with a geometric cooling schedule.

4. **Final L‑BFGS fine‑tuning** – a short high‑temperature L‑BFGS run
   (temperature = 1500, 500 iterations) polishes the result.

All stages stop when the global deadline (budget – a safety margin) is
reached.  The best sequence discovered is stored in the global name
``height_sequence_1`` and also returned as a plain ``list`` of floats.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Soft‑max utilities (stable)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decreasing(n: int) -> np.ndarray:
    """Monotone decreasing line (sum = 1)."""
    a = (n - np.arange(n, dtype=np.float64)) / (n * (n - 1) / 2.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1.0)))
        elif i <= (n - 1) * (1.0 - alpha / 2.0):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2.0 / alpha + 1.0)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_mixture(n: int, lam1: float, lam2: float, w: float) -> np.ndarray:
    """Mixture of two exponentials (w·exp(-lam1·i)+(1‑w)·exp(-lam2·i))."""
    i = np.arange(n, dtype=np.float64)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hyperbolic(n: int, beta: float) -> np.ndarray:
    """a_i ∝ 1/(i+beta)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinc_squared(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = np.where(i == 0, 1.0, np.sin(np.pi * i / n) / (np.pi * i / n))
    a = x ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine_power(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Random helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Surrogate objective (fast, same ordering as official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max + gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 300,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """L‑BFGS on the smooth‑max surrogate."""
    eps = 1e-30
    v = np.log(start_seq + eps)  # logits – unconstrained

    def fun(vv):
        a = _softmax(vv)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        grad_v_scaled = 2.0 * n * grad_v
        return loss, grad_v_scaled.astype(np.float64)

    # honour deadline
    if time.time() > deadline:
        return start_seq, _objective(start_seq)

    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:  # pragma: no cover
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑conv objective
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback
        return np.maximum(v, 0.0) / max(np.sum(v), 1e-12)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 3_500_000,
                                 lr0: float = 0.20,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with multiplicative updates."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:8d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # projection onto simplex
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated Annealing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_500_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)
        move_type = rng.integers(0, 3)

        if move_type == 0:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:  # Dirichlet redistribute a random subset
            k = rng.integers(3, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:  # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:8d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))
    seeds.append(_linear_decreasing(n))
    seeds.append(_sinc_squared(n))

    # sine‑window sweep
    for gamma in np.linspace(0.3, 4.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian sweep
    for sigma in np.linspace(0.08, 0.45, 10):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Power‑law sweep
    for p in np.linspace(0.0, 4.0, 9):
        seeds.append(_power(n, p))

    # Beta‑shape (wide grid – provides monotone decreasing shapes)
    alphas = np.concatenate((np.linspace(0.05, 0.3, 4), np.linspace(0.3, 3.0, 7)))
    betas = np.linspace(1.5, 4.0, 5)  # β > α to bias decreasing
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.05, 0.95, 9):
        seeds.append(_tukey_window(n, alpha))

    # Exponential families
    for lam in np.linspace(0.0005, 0.015, 10):
        seeds.append(_exponential_decay(n, lam))

    # Mixture of two exponentials (randomised)
    for _ in range(12):
        lam1 = rng.uniform(0.0003, 0.005)
        lam2 = rng.uniform(0.005, 0.02)
        w = rng.uniform(0.2, 0.8)
        seeds.append(_exp_mixture(n, lam1, lam2, w))

    # Hyperbolic family
    for beta in np.linspace(0.5, 5.0, 6):
        seeds.append(_hyperbolic(n, beta))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform and low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_uniform(n) * (1 - mix_alpha) + curve * mix_alpha)

    # Random Dirichlet draws (various concentrations)
    for conc in (0.001, 0.005, 0.01, 0.05, 0.1, 0.2):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw ordering
            seeds.append(np.sort(raw)[::-1])      # descending (monotone)
            seeds.append(np.sort(raw))            # ascending

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function under the official evaluation.
    Returns the best sequence as a list of floats and stores it in the
    global ``height_sequence_1`` for possible reuse.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Length of the vector – 4000 gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Retrieve previous solution (if any) for seeding
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Stage 0 – generate a rich seed pool
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # --------------------------------------------------------------
    # Fast surrogate evaluation, keep the top candidates
    # --------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – L‑BFGS on smooth‑max surrogate (more iterations)
    # --------------------------------------------------------------
    temps = [10.0, 30.0, 100.0, 300.0, 800.0, 1500.0]
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temps:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline, max_iter=300, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp:.0f} → new best {val_opt:.6f}")
            seq = seq_opt  # warm‑start next temperature

    # --------------------------------------------------------------
    # Stage 2 – Extended mirror‑descent (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_m, val_m = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=3_500_000,
            lr0=0.20,
            power=0.30,
        )
        if val_m < global_best_val:
            global_best_seq = seq_m.copy()
            global_best_val = val_m
            print(f"[Mirror] new global best = {val_m:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing (longer)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_500_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – Final high‑temperature L‑BFGS fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS fine‑tuning ---")
        seq_final, val_final = _l_bfgs_optimize(
            global_best_seq,
            n,
            deadline,
            max_iter=500,
            temperature=1500.0,
        )
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Final L‑BFGS] improved best = {val_final:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (information only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
