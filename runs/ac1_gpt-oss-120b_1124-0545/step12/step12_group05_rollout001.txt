"""
Step‑function optimizer for ``evaluate_sequence``.

The algorithm combines a rich pool of analytically‑generated seeds,
large‑scale first‑order optimisation (Adam on a log‑softmax
parameterisation with a temperature‑controlled log‑sum‑exp surrogate
for the max‑convolution), aggressive permutation‑focused simulated
annealing (swap + segment‑reverse moves), a cheap greedy mass‑transfer
descent that moves probability mass from the most “responsible”
indices to the least, and a final mirror‑subgradient polishing phase.

All phases respect a global time budget (default 1000 s).  The best
vector found (as a plain ``list[float]``) is stored in the global name
``height_sequence_1`` for possible reuse and is also returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective (the numerator in ``evaluate_sequence`` when the
    ℓ₁‑norm of the sequence equals 1).

    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(x: np.ndarray) -> np.ndarray:
    """Interleave largest / smallest values, then renormalise."""
    a = np.sort(x)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


# ----------------------------------------------------------------------
# Analytic seed families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.07, offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (p>0)."""
    if p <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))

    # sine windows (various gammas) -----------------------------------------
    for gamma in np.linspace(0.5, 6.0, 12):
        s = _sine_window(n, gamma)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Gaussian (wider sigma range) -----------------------------------------
    for sigma in np.linspace(0.07, 0.55, 13):
        s = _gaussian(n, sigma)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Beta shapes ------------------------------------------------------------
    for _ in range(30):
        alpha = 10 ** rng.uniform(-3, 0.8)   # 0.001 … ~6.3
        beta = 10 ** rng.uniform(-3, 0.8)
        s = _beta_shape(n, alpha, beta)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Exponential (broad rate range) ----------------------------------------
    for rate in np.logspace(-6, -0.5, 12):
        s = _exponential(n, rate)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Logarithmic ------------------------------------------------------------
    for eps in np.logspace(-6, -0.5, 9):
        s = _logarithmic(n, eps)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Power‑law ---------------------------------------------------------------
    for p in np.linspace(0.3, 2.5, 12):
        s = _powerlaw(n, p)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Dirichlet draws (various concentration) -------------------------------
    for conc in [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_alternate_order(raw))

    # Mixtures of low‑frequency cosine waves ---------------------------------
    for nf in (2, 3, 4):
        for _ in range(2):
            t = np.arange(n, dtype=np.float64)
            curve = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                curve += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            curve = np.maximum(curve, 0.0)
            curve_sum = curve.sum()
            if curve_sum > 0:
                curve /= curve_sum
                mixed = 0.7 * _uniform(n) + 0.3 * curve
                mixed /= mixed.sum()
                seeds.append(mixed)
                seeds.append(_alternate_order(mixed))

    # Previous solution (if any) --------------------------------------------
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            avg = avg / avg.sum()
            seeds.append(avg)
            seeds.append(_alternate_order(avg))

    return seeds

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on log‑softmax parameters with a temperature‑controlled
    log‑sum‑exp surrogate for the max‑convolution.
    Returns the best feasible vector (ℓ₁‑normalised) seen.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    # temperature schedule (smooth max)
    beta = 0.5
    beta_inc = 0.6
    beta_max = 500.0

    best_seq = None
    best_val = float("inf")

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                         # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smooth max via log‑sum‑exp ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a  (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # back‑prop through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature slowly
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
    max‑convolution objective.  Returns the best vector seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.30

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # Sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional permutation diversification (alternating order)
        if it % 120_000 == 0:
            alt = _alternate_order(a)
            alt_val = _objective(alt)
            if alt_val < cur_val:
                a = alt
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {alt_val:.6f}")

    return best_seq, best_val


def _perm_sa_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing focused on permutation moves (swap & reverse‑segment).
    The multiset of values is kept unchanged.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    T = 0.02 * cur_val + 1e-12

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9999, 1e-12)

        move_type = rng.integers(0, 2)   # 0 = swap, 1 = reverse segment
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            i, j = sorted(rng.choice(n, size=2, replace=False))
            new = cur.copy()
            new[i:j + 1] = new[i:j + 1][::-1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[Perm‑SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _grad_mass_transfer_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    eps_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Very cheap greedy mass‑transfer descent.
    At each step we move a small amount of probability from the index
    with the largest sub‑gradient (most “responsible” for the peak) to the
    index with the smallest sub‑gradient.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rng = np.random.default_rng()
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution entry
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if g[i_max] <= g[i_min] + 1e-12:
            # No beneficial transfer possible – break early
            break

        # transfer a small fraction of the source mass
        delta = min(a[i_max], eps_frac * a[i_max])
        if delta <= 0.0:
            break
        a[i_max] -= delta
        a[i_min] += delta

        # renormalise (numerical drift)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = _uniform(n)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mass‑Transfer] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under a time budget.
    The resulting list is stored in the global variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # Length of the candidate. 4000 offers a good speed / quality trade‑off.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Bring forward any previously discovered solution.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Phase 1 – generate a broad pool of seeds.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise global best -------------------------------------------------
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam refinements (one per top seed, with restarts).
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 45))
        max_iters = max(30_000, max_iters)
        print(f"\n--- Adam run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam.copy(), val_adam
            print(f"[Overall‑Adam] new global best = {val_adam:.6f}")

        # a quick perturb‑and‑re‑Adam restart can escape shallow basins
        if seq_adam is not None:
            pert = seq_adam + rng.normal(scale=0.001, size=n)
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            seq_adam2, val_adam2 = _adam_optimize(pert, n, deadline, max_iters=int(max_iters * 0.3))
            if seq_adam2 is not None and val_adam2 < global_best_val:
                global_best_seq, global_best_val = seq_adam2.copy(), val_adam2
                print(f"[Overall‑Adam‑restart] new global best = {val_adam2:.6f}")

        # try alternating order of the Adam result
        alt = _alternate_order(seq_adam) if seq_adam is not None else None
        if alt is not None:
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq, global_best_val = alt, alt_val
                print(f"[Overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – permutation‑focused simulated annealing on the best vector.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation‑SA polishing ---")
        seq_perm, val_perm = _perm_sa_optimize(
            global_best_seq, n, deadline, max_iters=500_000
        )
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[Overall‑Perm‑SA] new global best = {val_perm:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – cheap greedy mass‑transfer descent.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Greedy mass‑transfer descent ---")
        seq_mt, val_mt = _grad_mass_transfer_optimize(
            global_best_seq, n, deadline, max_iters=300_000
        )
        if val_mt < global_best_val:
            global_best_seq, global_best_val = seq_mt.copy(), val_mt
            print(f"[Overall‑Mass‑Transfer] new global best = {val_mt:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – long mirror‑subgradient polishing (full gradient updates).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑subgradient polishing ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=3_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror.copy(), val_mirror
            print(f"[Overall‑Mirror] new global best = {val_mirror:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – final simulated‑annealing with full move set.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final SA (full move set) ---")
        seq_final, val_final = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            init_temp_factor=0.015,
            min_temp=1e-14,
        )
        if val_final < global_best_val:
            global_best_seq, global_best_val = seq_final.copy(), val_final
            print(f"[Overall‑Final‑SA] new global best = {val_final:.6f}")

    # --------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()


# ----------------------------------------------------------------------
# Additional routine used in Phase 6 (full‑move SA)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polish that combines three move types:
      0 – swap two entries,
      1 – redistribute mass among a random small subset (Dirichlet),
      2 – pair‑wise mass transfer.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # subset redistribution
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 75_000 == 0:
                    print(f"[Final‑SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val
