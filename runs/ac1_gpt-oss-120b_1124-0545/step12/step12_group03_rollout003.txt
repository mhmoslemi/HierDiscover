def _project_onto_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _proj_subgradient_optimize(start_seq, n, deadline, max_iters=2_000_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    lr0 = 0.5
    power = 0.5
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute conv
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ProjGrad] it={it} best={cur_val:.6f}")
        # subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - idx
        g[idx] = 2.0 * a[rev]
        # step
        lr = lr0 / (it ** power)
        a = a - lr * g
        # project onto simplex
        a = _project_onto_simplex(a)
    return best_seq, best_val

   the current convolution bound.

5. **Final evaluation** – the best vector found is returned, stored in
   the global name ``height_sequence_1`` and its official value printed.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used for ranking candidates.
    seq must be ℓ₁‑normalised (Σseq = 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                        (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve built from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """Adam on logits with a log‑sum‑exp surrogate of the max."""
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta = 0.5            # start temperature
    beta_inc = 0.5
    beta_max = 2000.0     # higher final temperature

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # simplex point
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # --- smoothed max via log‑sum‑exp ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a  (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _proj_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                              max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Additive projected sub‑gradient descent on the exact max‑convolution.
    Uses Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.5          # initial step size
    power = 0.5        # decay exponent

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ProjGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - idx
        g[idx] = 2.0 * a[rev]

        # additive step and projection onto simplex
        lr = lr0 / ((it) ** power)
        a = a - lr * g
        a = _project_onto_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming polishing (very few iterations)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the matrix A where (A @ g) = conv(f, g) (full linear convolution).
    Shape of A is (2·len(f)‑1, len(f)).
    """
    n = len(f)
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for k in range(n):
        rows[idx:idx + n] = np.arange(k, k + n, dtype=np.int32)
        cols[idx:idx + n] = k
        data[idx:idx + n] = f
        idx += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 3, tau: float = 0.18) -> tuple[np.ndarray, float]:
    """
    A compact LP polishing step.
    Solves:
        maximise   Σ g_i
        s.t.       conv(seq, g) ≤ M   (M = max conv(seq, seq))
                   g_i ≥ 0
    The solution g is mixed with the current seq (weight τ) and renormalised.
    """
    f = seq.copy()
    n = len(f)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()
        A = _build_conv_matrix(f)
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if not res.success:
            print(f"[LP] iteration {it} failed – stopping LP stage.")
            break

        g = res.x
        if g.sum() <= f.sum() + 1e-12:
            print(f"[LP] iteration {it} – no L1‑mass increase.")
            break

        f_new = (1.0 - tau) * f + tau * g
        f_new = f_new / f_new.sum()
        f = f_new

        cur_obj = _objective(f)
        print(f"[LP] iteration {it} – objective = {cur_obj:.6f}")

    return f, _objective(f)


# ----------------------------------------------------------------------
# Seed collection – richer parameter grids
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # Sine‑window family
    for gamma in np.linspace(0.2, 5.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian family
    for sigma in np.linspace(0.06, 0.50, 10):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Power‑law family
    for p in np.linspace(0.0, 6.0, 11):
        seeds.append(_power(n, p))

    # Beta‑shape (covers U‑shapes as well)
    for a in np.linspace(0.2, 3.5, 7):
        for b in np.linspace(0.2, 3.5, 7):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.05, 0.95, 10):
        seeds.append(_tukey_window(n, alpha))

    # Exponential decay
    for lam in np.linspace(0.0002, 0.02, 8):
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet draws – many concentrations
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform & low‑frequency curves
    for mix_alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution, if available
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a list of floats
    and stores it globally under the name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – compromise between expressiveness and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any) to enrich the seed pool.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size != n:
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.pad(previous, (0, n - previous.size), constant_values=0.0)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich pool of seeds and keep the best few.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n = {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} seed values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on the smoothed surrogate.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # Allocate a generous chunk of the remaining budget to each seed,
        # but never exceed a hard cap.
        max_iters = min(300_000, int(remaining * 60))
        print(f"\n--- Adam run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Projected sub‑gradient descent (exact max) on the remainder.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Projected sub‑gradient descent (exact max) ---")
        seq_proj, val_proj = _proj_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=5_000_000
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑ProjGrad] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Linear‑programming polishing (few iterations).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming polishing ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, deadline,
                                    max_iters=3, tau=0.18)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (informational)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
