"""
Propose a step‑function (non‑negative vector) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm builds a very diverse pool of analytic “seed” windows
(including Kaiser, Blackman, DPSS, etc.), keeps the best few, refines
them with a long Adam optimiser (log‑softmax parameters, smooth max),
applies two deterministic orderings (alternating and pendulum),
polishes the best seed with a mirror‑subgradient (exponentiated‑gradient)
descent, then performs an extensive permutation‑hill‑climb followed by
a simulated‑annealing (SA) polish.  All phases respect the global
time‑budget (default 1000 s).  The final vector is stored in the global
name ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# optional SciPy for extra windows (e.g. DPSS)
try:
    from scipy.signal import windows as scipy_windows
except Exception:                 # pragma: no cover
    scipy_windows = None


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective (numerator of evaluate_sequence with sum=1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Re‑order as large, small, 2nd‑large, 2nd‑small, … and renormalise."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


def _pendulum_order(vec: np.ndarray) -> np.ndarray:
    """Largest, then smallest, then second‑largest, then second‑smallest …"""
    desc = np.sort(vec)[::-1]          # descending
    out = np.empty_like(desc)
    left, right = 0, len(desc) - 1
    pos = 0
    while left <= right:
        out[pos] = desc[left]
        left += 1
        pos += 1
        if left <= right:
            out[pos] = desc[right]
            right -= 1
            pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


# ----------------------------------------------------------------------
# Analytic shapes – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-np.abs(i - c) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.07,
                     offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    c = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (c - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (c + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser (DPSS‑like) window; beta≥0."""
    if beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    arg = (i - alpha) / alpha
    a = np.i0(beta * np.sqrt(1.0 - arg * arg)) / np.i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42
         - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.355768
         - 0.487396 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.144232 * np.cos(4.0 * np.pi * i / (n - 1))
         - 0.012604 * np.cos(6.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Slepian (DPSS) window – requires SciPy."""
    if scipy_windows is None:          # pragma: no cover
        return _uniform(n)
    # the DPSS routine returns an array normalised in L2 norm
    a = scipy_windows.dpss(n, nw, Kmax=1, sym=True)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, steep: float = 6.0) -> np.ndarray:
    """A smoothed step – values increase from left to right."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(-steep * (i - c) / c))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    # Kaiser on a fine grid
    for beta in np.linspace(0.0, 12.0, 15):
        seeds.append(_kaiser_window(n, beta))
        seeds.append(_alternate_order(_kaiser_window(n, beta)))

    # Sine windows – wide gamma grid
    for gamma in np.linspace(0.2, 8.0, 20):
        s = _sine_window(n, gamma)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Gaussian – wide sigma grid
    for sigma in np.linspace(0.04, 0.55, 20):
        g = _gaussian(n, sigma)
        seeds.append(g)
        seeds.append(_alternate_order(g))

    # Beta family – broader exploration
    for _ in range(40):
        alpha = 10 ** rng.uniform(-3, 1.0)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1.0)
        bvec = _beta_shape(n, alpha, beta)
        seeds.append(bvec)
        seeds.append(_alternate_order(bvec))

    # Laplace with varied λ
    for lam in np.logspace(-2, 1, 12):
        seeds.append(_laplace_window(n, lam))
        seeds.append(_alternate_order(_laplace_window(n, lam)))

    # Exponential shapes
    for rate in np.logspace(-5, -0.5, 12):
        e = _exponential(n, rate)
        seeds.append(e)
        seeds.append(_alternate_order(e))

    # Logarithmic shapes
    for eps in np.logspace(-5, -0.5, 8):
        logs = _logarithmic(n, eps)
        seeds.append(logs)
        seeds.append(_alternate_order(logs))

    # Logistic window – varied steepness
    for steep in np.linspace(2.0, 12.0, 10):
        l = _logistic_window(n, steep)
        seeds.append(l)
        seeds.append(_alternate_order(l))

    # DPSS windows if SciPy is present
    if scipy_windows is not None:
        for nw in np.linspace(2.0, 8.0, 9):
            d = _dpss_window(n, nw)
            seeds.append(d)
            seeds.append(_alternate_order(d))

    # Dirichlet draws – many concentration levels
    concs = [0.005, 0.02, 0.07, 0.15, 0.3, 0.6, 1.0, 2.0]
    for conc in concs:
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)
            seeds.append(_alternate_order(raw))
            seeds.append(_pendulum_order(raw))

    # Random permutations of already good deterministic seeds
    for base in [_triangular(n), _gaussian(n, 0.15), _kaiser_window(n, 6.0)]:
        for _ in range(4):
            perm = rng.permutation(base)
            seeds.append(perm / perm.sum())
            seeds.append(_alternate_order(perm))
            seeds.append(_pendulum_order(perm))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            norm = avg / avg.sum()
            seeds.append(norm)
            seeds.append(_alternate_order(norm))
            seeds.append(_pendulum_order(norm))

    return seeds

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 150_000) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax parameters using a temperature‑controlled smooth max."""
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 0.5          # temperature for smooth max
    beta_inc = 0.6
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
    max‑convolution objective. Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.35          # start step‑size (a bit larger helps early progress)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional aggressive re‑ordering (every 80k iters)
        if it % 80_000 == 0:
            alt = _alternate_order(a)
            alt_val = _objective(alt)
            if alt_val < cur_val:
                a = alt
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {alt_val:.6f}")

    return best_seq, best_val


def _perm_hill_climb(start_seq: np.ndarray, n: int,
                     deadline: float,
                     max_swaps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Swap‑based hill‑climbing on the permutation of the entries.
    Multiset of values stays unchanged.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    it = 0
    # initial temperature – a few percent of the current value
    T = cur_val * 0.004

    while it < max_swaps and time.time() < deadline:
        it += 1
        T = max(T * 0.99995, 1e-14)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 25_000 == 0:
                    print(f"[Perm] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 3_000_000,
                                 init_temp_factor: float = 0.025,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polish with three move types:
    swap, small Dirichlet mass‑redistribution, and pair‑transfer.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9998, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random small subset
            k = rng.integers(2, 7)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 75_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence (list of floats) and stores it in the global
    variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Length – a compromise between granularity and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous best, if any, and normalise it.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – generate a very rich seed pool.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # evaluate surrogate (2·n·max(conv)) – this ignores the denominator because
    # all seeds are ℓ₁‑normalised.
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinements on the top seeds.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest iteration budget proportionally to remaining time
        max_iters = int(min(200_000, remaining * 450))  # ~450 it/s estimate
        max_iters = max(30_000, max_iters)
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (iters ≤ {max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam.copy(), val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # try deterministic orderings of Adam result
        if seq_adam is not None:
            for ord_func, name in [( _alternate_order, "alt"),
                                   ( _pendulum_order,  "pend" )]:
                cand = ord_func(seq_adam)
                cand_val = _objective(cand)
                if cand_val < global_best_val:
                    global_best_seq, global_best_val = cand, cand_val
                    print(f"[overall‑Adam‑{name}] new global best = {cand_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long mirror‑descent polishing on the best seed.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent (first polishing) ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=2_500_000)
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror.copy(), val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – permutation hill‑climbing (re‑ordering only).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation hill‑climbing ---")
        seq_perm, val_perm = _perm_hill_climb(
            global_best_seq, n, deadline, max_swaps=180_000)
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[overall‑Perm] new global best = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – second mirror‑descent polishing (after re‑ordering).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent (second polishing) ---")
        seq_m2, val_m2 = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=1_500_000)
        if val_m2 < global_best_val:
            global_best_seq, global_best_val = seq_m2.copy(), val_m2
            print(f"[overall‑Mirror2] new global best = {val_m2:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – simulated‑annealing final polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing final polish ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=4_000_000,
            init_temp_factor=0.025,
            min_temp=1e-14)
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:                         # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
