def _conv_self_fft(a):
    n = len(a)
    L = 2*n-1
    N = _next_pow2(L)
    fx = np.fft.rfft(a, n=N)
    # product: abs(fx) ** 2 (real)
    conv = np.fft.irfft(fx * np.conj(fx), n=N)[:L]
    return conv

def _euclidean_projection_onto_simplex(v):
    # Project onto simplex {x >=0, sum x = 1}
    # using algorithm by Duchi
    v = np.asarray(v)
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm improves on the reference by

* using a longer vector (n≈5000) for a finer granularity,
* a faster self‑convolution based on a single real FFT,
* a more aggressive exponentiated‑gradient (mirror) descent with
  a larger initial step‑size and occasional alternating‑order swaps,
* a custom “balancing” phase that transfers mass from indices that
  contribute most to the current autocorrelation peak to indices that
  contribute least,
* a final simulated‑annealing polish.

All phases respect the global time budget (default 1000 s).  The best
vector found is stored in the global name ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast self‑convolution (linear, full length) via a single real FFT
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (full length)."""
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * np.conj(fa), n=N)[:L]   # real result
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective for a sequence normalised to ℓ₁‑norm 1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_self_fft(seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Re‑order as large, small, 2nd‑large, 2nd‑small, … and renormalise."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int,
                     sigma_frac: float = 0.07,
                     offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed generation – richer than the reference
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))

    # sine windows on a fine grid
    for gamma in np.linspace(0.5, 6.0, 12):
        s = _sine_window(n, gamma)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Gaussian on a fine sigma grid
    for sigma in np.linspace(0.07, 0.30, 12):
        s = _gaussian(n, sigma)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # beta family – explore a broader domain
    for _ in range(30):
        alpha = 10 ** rng.uniform(-3, 0.8)   # 0.001 … ≈6.3
        beta = 10 ** rng.uniform(-3, 0.8)
        b = _beta_shape(n, alpha, beta)
        seeds.append(b)
        seeds.append(_alternate_order(b))

    # exponential shapes
    for rate in np.logspace(-5, -1, 12):
        e = _exponential(n, rate)
        seeds.append(e)
        seeds.append(_alternate_order(e))

    # logarithmic shapes
    for eps in np.logspace(-5, -1, 10):
        l = _logarithmic(n, eps)
        seeds.append(l)
        seeds.append(_alternate_order(l))

    # Dirichlet draws with several concentration parameters
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_alternate_order(raw))

    # Mixtures of uniform with low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(3):
            t = np.arange(n, dtype=np.float64)
            curve = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                curve += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            curve = np.maximum(curve, 0.0)
            curve /= curve.sum()
            mixed = 0.7 * _uniform(n) + 0.3 * curve
            mixed /= mixed.sum()
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            avg = avg / avg.sum()
            seeds.append(avg)
            seeds.append(_alternate_order(avg))

    # a few completely random Dirichlet draws for extra diversity
    for _ in range(15):
        rand = rng.dirichlet(np.full(n, 0.5))
        seeds.append(rand)

    return seeds


# ----------------------------------------------------------------------
# Mirror‑subgradient optimiser (exponentiated‑gradient on the simplex)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.40          # a slightly larger initial step‑size than the reference

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - idx
        g[idx] = 2.0 * a[rev]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise to keep ℓ₁‑norm = 1
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional aggressive re‑ordering
        if it % 80_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {val_alt:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fine‑grained balancing – move mass from large contributors to small ones
# ----------------------------------------------------------------------
def _balancing_optimize(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_steps: int = 120_000,
                        transfer_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Greedy local search that repeatedly moves a fraction of the mass from the
    index contributing most to the current autocorrelation peak to the one
    contributing least.  The total ℓ₁‑norm stays equal to 1.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rng = np.random.default_rng()
    it = 0
    while it < max_steps and time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        max_idx = int(np.argmax(conv))
        # compute contributions a[i] * a[max_idx-i] for the active index
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - idx
        contrib = a[idx] * a[rev]

        # indices with maximal / minimal contribution
        i_max = idx[np.argmax(contrib)]
        i_min = idx[np.argmin(contrib)]

        if a[i_max] > 0:
            delta = transfer_frac * a[i_max]
            a[i_max] -= delta
            a[i_min] += delta

        # occasional random perturbations to escape plateaus
        if it % 5_000 == 0:
            if rng.random() < 0.5:
                i, j = rng.choice(n, size=2, replace=False)
                a[i], a[j] = a[j], a[i]
            else:
                k = rng.integers(2, 7)
                idx_sub = rng.choice(n, size=k, replace=False)
                tot = a[idx_sub].sum()
                if tot > 0:
                    a[idx_sub] = rng.dirichlet(np.ones(k)) * tot

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Bal] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (same as before, a few tweaks)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 init_temp_factor: float = 0.02,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a small random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allotted time budget.
    The result is stored in the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Choose a length that balances granularity and speed.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Load a previous solution, if any.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – generate a diverse pool of analytic seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"   #{i}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent refinement on the elite seeds.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest, time‑based budget per seed
        max_iters = int(min(300_000, remaining * 45))
        max_iters = max(40_000, max_iters)
        print(f"\n--- Mirror‑descent run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq, val = _mirror_subgradient_optimize(seed_vec, n, deadline,
                                                max_iters=max_iters)
        if val < global_best_val:
            global_best_seq, global_best_val = seq, val
            print(f"[overall‑Mirror] new global best = {val:.6f}")

        # try alternating order as well
        alt = _alternate_order(seq)
        alt_val = _objective(alt)
        if alt_val < global_best_val:
            global_best_seq, global_best_val = alt, alt_val
            print(f"[overall‑Mirror‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – dedicated balancing phase.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Balancing phase ---")
        seq_bal, val_bal = _balancing_optimize(global_best_seq, n, deadline,
                                               max_steps=120_000,
                                               transfer_frac=0.07)
        if val_bal < global_best_val:
            global_best_seq, global_best_val = seq_bal, val_bal
            print(f"[Balancing] new global best = {val_bal:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – a second round of mirror descent (polishing).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Second mirror‑descent polishing ---")
        seq_m2, val_m2 = _mirror_subgradient_optimize(global_best_seq, n,
                                                     deadline,
                                                     max_iters=300_000)
        if val_m2 < global_best_val:
            global_best_seq, global_best_val = seq_m2, val_m2
            print(f"[Mirror2] new global best = {val_m2:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final simulated‑annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(global_best_seq, n,
                                                       deadline,
                                                       max_iters=500_000,
                                                       init_temp_factor=0.02)
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
