Improved optimization for the convolution max objective.

Algorithm:
1. Build a pool of analytic and random seeds (uniform, triangular,
   sine windows, Dirichlet draws, previous solution, etc.) and ℓ1‑norm
   them.
2. Keep the best few seeds (by the surrogate 2·n·max(conv)).
3. From the best seed run a Frank‑Wolfe (conditional‑gradient) phase.
   In each FW step we solve a linear program

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   for all k,
                    b_j ≥ 0,

   where M = max(conv(f,f)).  The LP is solved with scipy.optimize.linprog
   and the feasible point b is renormalised and mixed with the current
   iterate.
4. After the FW phase perform a long mirror‑subgradient (exponentiated‑gradient)
   descent on the exact max‑convolution, using a decreasing step‑size.
5. Follow with a permutation hill‑climbing that swaps entries and
   occasionally performs a block‑reversal move.
6. Finish with a simulated‑annealing polishing that uses swaps,
   mass‑transfer, pair‑transfer and block‑reversal moves.
7. Return the best sequence found and store it in the global variable
   ``height_sequence_1``.
All steps respect a global time budget (default 1000 s).

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s > 0 else np.full_like(v, 1.0 / len(v))

def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))

def _pendulum_order(vec: np.ndarray) -> np.ndarray:
    """Place biggest, then smallest, then second biggest, then second smallest … but starting from the centre."""
    # This is a variant that sometimes works better for the convolution
    a = np.sort(vec)[::-1]               # descending
    out = np.empty_like(a)
    left, right = 0, len(a) - 1
    i = 0
    while left <= right:
        out[left] = a[i]; i += 1; left += 1
        if left <= right:
            out[right] = a[i]; i += 1; right -= 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))

# Analytic families – all return ℓ₁‑normalised vectors

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _double_gaussian(n: int, sigma_frac: float = 0.07, offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

                curve += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            curve = np.maximum(curve, 0.0)
            curve /= curve.sum()
            mixed = 0.7 * _uniform(n) + 0.3 * curve
            mixed /= mixed.sum()
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            normalised = avg / avg.sum()
            seeds.append(normalised)
            seeds.append(_alternate_order(normalised))

    return seeds

# ----------------------------------------------------------------------
# Optimisers – Adam (log‑softmax) --------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    eps = 1e-30
    v = np.log(start_vec + eps)           # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta = 0.5
    beta_inc = 0.5
    beta_max = 500.0
    best_seq = None
    best_val = float('inf')
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")
        # smoothed max
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)
        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)
    return best_seq, best_val

# ----------------------------------------------------------------------
# Mirror‑subgradient (exponentiated‑gradient) -------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    lr0 = 0.30
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")
        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # multiplicative update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s
        # occasional re‑ordering
        if it % 80_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {val_alt:.6f}")
    return best_seq, best_val

# Permutation hill‑climbing -------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray, n: int,
                     deadline: float,
                     max_swaps: int = 150_000) -> tuple[np.ndarray, float]:
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    it = 0
    T0 = cur_val * 0.005
    T = T0
    while it < max_swaps and time.time() < deadline:
        it += 1
        T = max(T * 0.9999, 1e-12)
        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 40_000 == 0:
                    print(f"[Perm] it={it:7d} best={best_val:.6f}")
    return best_seq, best_val

# Simulated‑annealing polish ------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000,
                                 init_temp_factor: float = 0.03,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)
        move_type = rng.integers(0, 4)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a small random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        elif move_type == 2:                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:                                   # block reversal
            i = rng.integers(0, n - 1)
            j = rng.integers(i + 1, n)          # exclusive upper bound
            new = cur.copy()
            new[i:j] = new[i:j][::-1]
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")
    return best_seq, best_val

# Frank‑Wolfe (conditional gradient) step using a linear programme ----
def _frank_wolfe_step(f: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Perform a single Frank‑Wolfe step: solve
       maximise Σ b_j   s.t.   (f∗b)_k ≤ M, b≥0,
       then mix the current iterate towards the normalised solution."""
    if not _SCIPY_AVAILABLE:
        # fall back to a cheap heuristic: move toward uniform
        g = _uniform(n)
        new_f = 0.5 * f + 0.5 * g
        new_f = new_f / new_f.sum()
        return new_f, _objective(new_f)

    if time.time() >= deadline:
        return f, _objective(f)

    conv_f = _conv_fft(f, f)
    M = float(np.max(conv_f))

    # Build a sparse Toeplitz matrix representing convolution with f
    # Each column j contributes f_i to row i+j
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')

    # Linear programme:
    # maximise   sum(b)   <=> minimise   -sum(b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Solve with HiGHS (fast interior‑point/simplex)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max(1.0, deadline - time.time())})
    if not res.success:
        # If the solver fails, fall back to a simple heuristic move toward the alternating order
        g = _alternate_order(f)
        new_f = 0.5 * f + 0.5 * g
        new_f /= new_f.sum()
        return new_f, _objective(new_f)

    b_opt = np.maximum(res.x, 0.0)
    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        g = _uniform(n)
    else:
        g = b_opt / sum_b          # ℓ₁‑normalised

    # Mix the current point and the LP solution.
    # The step size 0.5 works well in practice; we keep it fixed.
    t = 0.5
    new_f = (1.0 - t) * f + t * g
    new_f = new_f / new_f.sum()
    return new_f, _objective(new_f)

"""
Improved optimisation for the convolution‑max objective.

The algorithm proceeds in several stages:

1. **Seed generation** – a diverse pool of analytic shapes (uniform,
   triangular, sine, Gaussian, Beta, Laplace, mixtures, Dirichlet draws,
   previous best, ...) is built and ℓ₁‑normalised.

2. **Elite selection** – the best few seeds (by the surrogate 2·n·max(conv))
   are kept for further work.

3. **Adam on a smoothed‑max surrogate** – each elite seed is refined with
   exponentiated‑gradient (Adam) optimisation on a log‑soft‑max version of the
   max‑convolution.  After Adam we also test a couple of deterministic
   re‑orderings (alternating and pendulum) of the result.

4. **Frank‑Wolfe (conditional‑gradient) steps** – for the current iterate
   `f` we solve the linear programme

        maximise   Σ b_j
        subject to (f * b)_k ≤ M   ∀k,    b_j ≥ 0,

   where `M = max(conv(f,f))`.  The LP is solved with ``scipy.optimize.linprog``
   (HiGHS).  The normalised LP solution `g = b/Σb` is mixed with `f`
   (`f ← (1‑t)·f + t·g`, `t=0.5`).  A few such steps dramatically reduce the
   peak of the self‑convolution.

5. **Mirror‑subgradient descent** – a long exponentiated‑gradient (mirror)
   sub‑gradient descent on the exact max‑convolution further decreases the
   objective.

6. **Permutation hill‑climbing** – swap‑only hill climbing (with a tiny
   temperature for occasional uphill moves) re‑orders the multiset of values.

7. **Second mirror‑descent polish** – a short mirror‑descent pass after
   re‑ordering.

8. **Simulated‑annealing polish** – a final SA with four move types
   (swap, small‑subset Dirichlet redistribution, pair‑transfer,
   block‑reversal) refines the solution.

All phases respect a global time budget (default 1000 s).  The best
sequence found is stored globally as ``height_sequence_1`` and returned.

import time
import numpy as np

# optional SciPy – used for the exact linear programme
try:
    from scipy.optimize import linprog
    from scipy.sparse import diags, csr_matrix
    _SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s > 0 else np.full_like(v, 1.0 / len(v))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """
    Re‑order as large, small, 2nd‑large, 2nd‑small, … and renormalise.
    """
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


def _pendulum_order(vec: np.ndarray) -> np.ndarray:
    """
    Variant ordering: largest, smallest, 2nd‑largest, 2nd‑smallest, …
    (often gives a different distribution of large products).
    """
    a = np.sort(vec)[::-1]               # descending
    out = np.empty_like(a)
    left, right = 0, len(a) - 1
    i = 0
    while left <= right:
        out[left] = a[i]; i += 1; left += 1
        if left <= right:
            out[right] = a[i]; i += 1; right -= 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.07,
                    offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection (richer than the reference)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))

    # sine windows on a fine grid
    for gamma in np.linspace(0.5, 6.0, 12):
        s = _sine_window(n, gamma)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Gaussian on a fine sigma grid
    for sigma in np.linspace(0.07, 0.30, 12):
        g = _gaussian(n, sigma)
        seeds.append(g)
        seeds.append(_alternate_order(g))

    # beta family – explore a broader domain
    for _ in range(30):
        alpha = 10 ** rng.uniform(-3, 0.8)
        beta = 10 ** rng.uniform(-3, 0.8)
        b = _beta_shape(n, alpha, beta)
        seeds.append(b)
        seeds.append(_alternate_order(b))

    # exponential shapes, wide rate range
    for rate in np.logspace(-5, -1, 10):
        e = _exponential(n, rate)
        seeds.append(e)
        seeds.append(_alternate_order(e))

    # logarithmic shapes
    for eps in np.logspace(-5, -1, 8):
        l = _logarithmic(n, eps)
        seeds.append(l)
        seeds.append(_alternate_order(l))

    # Dirichlet draws
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(2):
            d = rng.dirichlet(np.full(n, conc))
            seeds.append(d)
            seeds.append(_alternate_order(d))

    # mixtures of uniform with low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(2):
            t = np.arange(n, dtype=np.float64)
            curve = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                curve += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            curve = np.maximum(curve, 0.0)
            curve /= curve.sum()
            mixed = 0.7 * _uniform(n) + 0.3 * curve
            mixed /= mixed.sum()
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # previous best (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            normalised = avg / avg.sum()
            seeds.append(normalised)
            seeds.append(_alternate_order(normalised))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a log‑softmax parametrisation
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 0.5          # temperature for smooth max
    beta_inc = 0.5
    beta_max = 500.0

    best_seq = None
    best_val = float("inf")
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d}  best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # chain rule through the softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror‑subgradient (exponentiated‑gradient) optimisation
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.30
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[Mirror] it={it:7d}  best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional aggressive re‑ordering
        if it % 80_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d}  improved alt to {val_alt:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation hill‑climbing (swap‑only)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray, n: int,
                     deadline: float,
                     max_swaps: int = 150_000) -> tuple[np.ndarray, float]:
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    it = 0
    T0 = cur_val * 0.005
    T = T0
    while it < max_swaps and time.time() < deadline:
        it += 1
        T = max(T * 0.9999, 1e-12)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        accept = delta < 0.0 or rng.random() < np.exp(-delta / T)
        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 40_000 == 0:
                    print(f"[Perm] it={it:7d}  best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish with extra block‑reversal moves
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000,
                                 init_temp_factor: float = 0.03,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 4)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a small random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        elif move_type == 2:                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:                                   # block reversal
            i = rng.integers(0, n - 1)
            j = rng.integers(i + 1, n)          # exclusive upper bound
            new = cur.copy()
            new[i:j] = new[i:j][::-1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d}  best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Frank‑Wolfe (conditional‑gradient) step via a linear programme
# ----------------------------------------------------------------------
def _frank_wolfe_step(f: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """One FW step: solve  maximise Σ b_j  s.t. (f∗b)_k ≤ M, b≥0."""
    if not _SCIPY_AVAILABLE:
        # fall back to a cheap heuristic move towards the alternating order
        g = _alternate_order(f)
        new_f = 0.5 * f + 0.5 * g
        new_f /= new_f.sum()
        return new_f, _objective(new_f)

    if time.time() >= deadline:
        return f, _objective(f)

    conv_f = _conv_fft(f, f)
    M = float(np.max(conv_f))

    # Build the Toeplitz convolution matrix A where A[k, j] = f_{k-j}
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')

    # Linear programme: maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Time limit for the LP – give it a few seconds at most
    lp_time = max(1.0, deadline - time.time())
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': lp_time})
    if not res.success:
        # fallback to alternating order if the LP failed
        g = _alternate_order(f)
        new_f = 0.5 * f + 0.5 * g
        new_f /= new_f.sum()
        return new_f, _objective(new_f)

    b_opt = np.maximum(res.x, 0.0)
    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        g = _uniform(n)
    else:
        g = b_opt / sum_b                     # ℓ₁‑normalised feasible direction

    # Mix current iterate with the LP solution (fixed step size 0.5)
    t = 0.5
    new_f = (1.0 - t) * f + t * g
    new_f /= new_f.sum()
    return new_f, _objective(new_f)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence (list of floats) and stores it in the global
    variable ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Length – 4000 is a good trade‑off between granularity and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best, if any, and normalise it.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Phase 0 – generate a diverse pool of seeds.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise the global best with the very best seed.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on the elite seeds.
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        max_iters = int(min(180_000, remaining * 40))
        max_iters = max(20_000, max_iters)
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam.copy(), val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # Try a couple of deterministic reorderings of the Adam result
        if seq_adam is not None:
            for reorder in (_alternate_order, _pendulum_order):
                alt = reorder(seq_adam)
                alt_val = _objective(alt)
                if alt_val < global_best_val:
                    global_best_seq, global_best_val = alt, alt_val
                    print(f"[overall‑Adam‑{reorder.__name__}] new global best = {alt_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Frank‑Wolfe (LP) steps.
    # --------------------------------------------------------------
    fw_steps = 3
    for step in range(fw_steps):
        if time.time() > deadline:
            break
        print(f"\n--- Frank‑Wolfe step {step + 1}/{fw_steps} ---")
        global_best_seq, fw_val = _frank_wolfe_step(global_best_seq, n, deadline)
        if fw_val < global_best_val:
            global_best_val = fw_val
            print(f"[FW] improved surrogate to {fw_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – long mirror‑descent sub‑gradient optimisation.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=4_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror.copy(), val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – permutation hill‑climbing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation hill‑climbing ---")
        seq_perm, val_perm = _perm_hill_climb(
            global_best_seq, n, deadline, max_swaps=120_000
        )
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[overall‑Perm] new global best = {val_perm:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – second mirror‑descent polishing after re‑ordering.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Second mirror‑descent polishing ---")
        seq_m2, val_m2 = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=2_000_000
        )
        if val_m2 < global_best_val:
            global_best_seq, global_best_val = seq_m2.copy(), val_m2
            print(f"[overall‑Mirror2] new global best = {val_m2:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – simulated‑annealing polishing (final refinement).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=5_000_000,
            init_temp_factor=0.03,
            min_temp=1e-12
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final sanity‑check with the official evaluator.
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:            # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
