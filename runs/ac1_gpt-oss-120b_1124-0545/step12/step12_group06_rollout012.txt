    start = time.time()
    deadline = start + budget_s - 2.0   # margin

    def conv(seq):
        L = len(seq) + len(seq) - 1
        N = 1 << ((L - 1).bit_length())
        fx = np.fft.rfft(seq, n=N)
        conv = np.fft.irfft(fx * fx, n=N)[:L]

"""
Project: minimise 2·n·max(conv(seq,seq)) / (sum(seq)²)
Author : an expert optimisation script (2024‑06)

Approach
--------
1.  Work on ℓ₁‑normalised vectors – the denominator becomes 1,
    so the objective is simply 2·n·max(conv(seq,seq)).
2.  Generate a large but cheap pool of *analytic* families:
        • exponential   a_i ∝ exp(-λ·i)
        • power‑law    a_i ∝ (i+1)^{-p}
        • beta‑shape   a_i ∝ i^{α‑1} (n‑i)^{β‑1}
   All families are normalised to sum‑to‑one.
3.  Scan each family on a coarse grid, keep the best candidate
    and then perform a small *local* random search around its
    parameters (log‑normal perturbations).  This exhausts the
    most promising region in a few seconds.
4.  For the current best vector also try the “alternating order’’
    (large, small, 2nd‑large, …) – this often decorrelates large
    entries and reduces the convolution peak.
5.  Finally a cheap hill‑climbing / simulated‑annealing phase
    (swap, Dirichlet re‑allocation, tiny Gaussian noise) is run
    until the time budget expires.  Every improvement is kept.
6.  The sequence is returned as a plain list of floats and is also
    stored in the global variable ``height_sequence_1`` for possible
    reuse by later runs.

The whole routine respects the 1000 s budget, prints a few progress
messages and never uses file or network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself via real FFT."""
    L = len(x) + len(x) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    conv = np.fft.irfft(fx * fx, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a ℓ₁‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Flat vector (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exponential(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) , normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} , normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ i^{α‑1}·(n‑i‑1)^{β‑1} , normalised."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------
def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    # re‑normalise (tiny rounding errors may appear)
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence (list of floats) and stores it in the
    global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # Length – a power of two is convenient for the FFT.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Helper to evaluate a candidate (exact objective, no extra checks)
    # --------------------------------------------------------------
    def eval_seq(v: np.ndarray) -> float:
        """Fast exact evaluation (assumes ℓ₁‑normalised vector)."""
        return _objective(v)

    # --------------------------------------------------------------
    # Keep the best candidate seen so far
    # --------------------------------------------------------------
    best_seq = _uniform(n)          # a safe fallback
    best_val = eval_seq(best_seq)

    # --------------------------------------------------------------
    # 1) EXPONENTIAL family – coarse grid + random refinement
    # --------------------------------------------------------------
    print("\n[Search] Exponential family (coarse grid)")
    rates = np.logspace(-5, -0.5, 200)          # 1e‑5 … ≈0.3
    best_rate = None
    for rate in rates:
        if time.time() > deadline:
            break
        cand = _exponential(n, rate)
        val = eval_seq(cand)
        if val < best_val:
            best_val, best_seq, best_rate = val, cand, rate
    # random local search around the best rate (log‑space perturbations)
    if best_rate is not None:
        print("[Refine] Exponential (log‑normal perturbations)")
        for _ in range(300):
            if time.time() > deadline:
                break
            rate = best_rate * np.exp(rng.normal(scale=0.2))
            if rate <= 0:
                continue
            cand = _exponential(n, rate)
            val = eval_seq(cand)
            if val < best_val:
                best_val, best_seq, best_rate = val, cand, rate

    # --------------------------------------------------------------
    # 2) POWER‑LAW family
    # --------------------------------------------------------------
    print("\n[Search] Power‑law family")
    p_vals = np.linspace(0.0, 8.0, 120)   # dense sweep
    for p in p_vals:
        if time.time() > deadline:
            break
        cand = _power(n, p)
        val = eval_seq(cand)
        if val < best_val:
            best_val, best_seq = val, cand

    # --------------------------------------------------------------
    # 3) BETA‑SHAPE family (grid over α,β)
    # --------------------------------------------------------------
    print("\n[Search] Beta‑shape family")
    alphas = np.logspace(-2, 1, 30)      # 0.01 … 10
    betas = np.logspace(-2, 1, 30)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            cand = _beta_shape(n, a, b)
            val = eval_seq(cand)
            if val < best_val:
                best_val, best_seq = val, cand

    # --------------------------------------------------------------
    # 4) MIXED exponential + uniform (adds a flat background)
    # --------------------------------------------------------------
    print("\n[Search] Mixed exponential‑+‑uniform")
    for _ in range(200):
        if time.time() > deadline:
            break
        rate = 10 ** rng.uniform(-5, -0.5)          # same range as before
        mix = rng.random()                         # weight of the exponential part
        exp_vec = _exponential(n, rate)
        mixed = mix * exp_vec + (1.0 - mix) * _uniform(n)
        mixed /= mixed.sum()
        val = eval_seq(mixed)
        if val < best_val:
            best_val, best_seq = val, mixed

    # --------------------------------------------------------------
    # 5) Alternating order of the current best (often helpful)
    # --------------------------------------------------------------
    print("\n[Post‑process] Alternating‑order test")
    alt = _alternate_order(best_seq)
    alt_val = eval_seq(alt)
    if alt_val < best_val:
        best_val, best_seq = alt_val, alt
        print(f"  → improvement after alternation: {best_val:.6f}")

    # --------------------------------------------------------------
    # 6) Local polishing – simple hill‑climbing / simulated‑annealing
    # --------------------------------------------------------------
    print("\n[Polish] Random local moves")
    move_counter = 0
    while time.time() < deadline:
        move_counter += 1
        # choose a move type
        mtype = rng.integers(0, 3)
        if mtype == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            cand = best_seq.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif mtype == 1:                   # redistribute a small random subset
            k = rng.integers(2, 7)         # size of the subset
            idx = rng.choice(n, size=k, replace=False)
            total = best_seq[idx].sum()
            if total == 0.0:
                continue
            cand = best_seq.copy()
            cand[idx] = rng.dirichlet(np.ones(k)) * total
        else:                               # tiny Gaussian jitter then re‑project
            cand = best_seq + rng.normal(scale=1e-4, size=n)
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s

        val = eval_seq(cand)
        if val < best_val:
            best_val, best_seq = val, cand
            # occasional progress report
            if move_counter % 2000 == 0:
                print(f"[Polish] Improvement #{move_counter}: {best_val:.6f}")

    # --------------------------------------------------------------
    # Final sanity check with the official evaluator
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest   objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
