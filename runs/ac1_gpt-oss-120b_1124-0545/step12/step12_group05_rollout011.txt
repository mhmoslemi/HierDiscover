"""
Improved search for minimal step‑function value.

Algorithm:

1. Generate a large and diverse pool of parametric seed vectors (uniform,
   triangular, sine, Gaussian, Beta, Dirichlet, random mixtures).  All are
   ℓ₁‑normalised.
2. Evaluate the surrogate 2·n·max(conv) for every seed and keep the best
   few.
3. For each promising seed run a mirror‑descent on a smooth ‑p‑norm
   approximation of the max‑convolution (p is increased from 4 to 64).  The
   iterate lives on the simplex via an exponentiated‑gradient update.
4. Take the best shape and try a handful of deterministic re‑orderings
   (alternating, “balanced‑alternating”, reversed, sorted).  The best
   permutation is kept.
5. Polish the permutation with a short simulated‑annealing that mixes
   swaps, pair‑mass transfers and small Dirichlet‑redistributions.
6. Return the best vector found and store it as ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Re‑order as large, small, 2nd‑large, 2nd‑small, … and renormalise."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


def _balanced_alternate(vec: np.ndarray) -> np.ndarray:
    """Place the largest value at the left end, the second at the right end,
    the third next to the left, the fourth next to the right, …"""
    desc = np.sort(vec)[::-1]            # descending
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    toggle = True
    for v in desc:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    s = out.sum()
    return out / s if s > 0 else out


# ----------------------------------------------------------------------
# Analytic shape families – return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.07, offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))

    # sine windows on a fine grid
    for gamma in np.linspace(0.5, 6.0, 12):

    # Dirichlet draws
    concs = [0.001, 0.005, 0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)
            seeds.append(_alternate_order(raw))

    # Random mixtures of low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(3):
            t = np.arange(n, dtype=np.float64)
            curve = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                curve += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            curve = np.maximum(curve, 0.0)
            curve /= curve.sum()
            mixed = 0.7 * _uniform(n) + 0.3 * curve
            mixed /= mixed.sum()
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Additional random parameter draws (wide search)
    for _ in range(80):
        typ = rng.choice(['beta', 'gauss', 'sine', 'dirichlet'])
        if typ == 'beta':
            alpha = 10 ** rng.uniform(-4, 2)
            beta = 10 ** rng.uniform(-4, 2)
            sv = _beta_shape(n, alpha, beta)
        elif typ == 'gauss':
            sigma_frac = rng.uniform(0.02, 0.5)
            sv = _gaussian(n, sigma_frac)
        elif typ == 'sine':
            gamma = rng.uniform(0.3, 12.0)
            sv = _sine_window(n, gamma)
        else:  # dirichlet
            conc = rng.uniform(0.001, 3.0)
            sv = rng.dirichlet(np.full(n, conc))
        seeds.append(sv)
        seeds.append(_alternate_order(sv))

    # Previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            norm = avg / avg.sum()
            seeds.append(norm)
            seeds.append(_alternate_order(norm))

    return seeds


# ----------------------------------------------------------------------
# Mirror‑descent optimizer based on a smooth p‑norm approximation of max
# ----------------------------------------------------------------------
def _p_norm_mirror_optimize(start_vec: np.ndarray, n: int,
                            deadline: float,
                            max_iters: int = 200_000,
                            p_start: float = 4.0,
                            p_end: float = 64.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the objective
        J_p(a) = 2·n·||conv(a,a)||_p ,
    where the p‑norm approaches the L∞ norm as p→∞.
    The iterate stays on the probability simplex.
    """
    a = np.copy(start_vec)
    # guarantee ℓ₁‑norm = 1
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = _uniform(n)

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # schedule p (increase linearly)
        p = p_start + (p_end - p_start) * it / max_iters

        conv = _conv_fft(a, a)
        # p‑norm of conv (avoid overflow)
        Lp = np.linalg.norm(conv, ord=p)
        if Lp == 0.0:
            continue

        # weight vector for gradient of the p‑norm
        # w_i = p * conv_i^{p-1} / Lp^{p-1}
        # conv >= 0, safe to raise
        w = np.power(conv, p - 1)
        denom = (Lp ** (p - 1)) if p > 1 else 1.0
        w = (p / (denom + 1e-30)) * w

        # gradient of 2·n·Lp w.r.t. a: 2·n·∇Lp,
        # but we can omit the constant factor because the step size
        # will be tuned anyway.
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]   # central n entries

        # exponentiated‑gradient update (mirror descent on simplex)
        lr = 0.30 / np.sqrt(it)                  # simple decreasing schedule
        a = a * np.exp(-lr * grad)

        # renormalise to stay on the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # track best surrogate (exact max‑conv)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[pNorm] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation hill‑climbing (swap‑only) – same as reference but a little
# more aggressive.
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray, n: int,
                     deadline: float,
                     max_swaps: int = 150_000) -> tuple[np.ndarray, float]:
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    it = 0
    # mild temperature for occasional uphill moves
    T0 = cur_val * 0.005
    T = T0

    while it < max_swaps and time.time() < deadline:
        it += 1
        T = max(T * 0.99995, 1e-14)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30_000 == 0:
                    print(f"[Perm] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (swap & mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 800_000,
                                 init_temp_factor: float = 0.015,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-14)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random small subset
            k = rng.integers(2, 10)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Deterministic permutation helpers
# ----------------------------------------------------------------------
def _evaluate_permutations(base_seq: np.ndarray, n: int) -> np.ndarray:
    """Return the best permutation among a handful of deterministic schemes."""
    candidates = [
        base_seq,
        _alternate_order(base_seq),
        _balanced_alternate(base_seq),
        base_seq[::-1],                    # pure reversal
        np.sort(base_seq),                 # ascending
        np.sort(base_seq)[::-1],           # descending (no special ordering)
    ]
    # add a few random shuffles for extra chance
    rng = np.random.default_rng()
    for _ in range(5):
        perm = rng.permutation(n)
        candidates.append(base_seq[perm])

    best_seq = candidates[0]
    best_val = _objective(best_seq)
    for cand in candidates[1:]:
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand
    return best_seq

"""
Improved search for a step‑function that minimises evaluate_sequence().

The algorithm proceeds in several stages (all respecting a 1000 s budget):

1. Build a very diverse pool of analytic “seed” shapes (uniform,
   triangular, sine, Gaussian, Beta, Laplace, mixtures, Dirichlet draws,
   previous best, …) and ℓ₁‑normalise them.
2. Keep the best few seeds according to the surrogate 2·n·max(conv).
3. Refine each promising seed with an exponentiated‑gradient (mirror‑descent)
   optimiser that minimises a smooth p‑norm approximation of the max‑convolution.
   The p‑norm is ramped from 4 to 64, so the objective increasingly resembles
   the true L∞ norm while remaining differentiable.
4. Apply a collection of deterministic re‑orderings (alternating, balanced‑alternating,
   reversal, sorted) to the best shape and keep the ordering with the smallest
   exact max‑convolution.
5. Run an extended swap‑only hill‑climbing on the current best permutation.
6. Finish with a simulated‑annealing polish that combines swaps,
   small Dirichlet‑redistributions and pair‑mass‑transfers.
7. Return the best sequence (list of floats) and store it globally as
   ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, … and re‑normalise."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


def _balanced_alternate(vec: np.ndarray) -> np.ndarray:
    """Largest left, 2nd‑largest right, 3rd left+1, 4th right‑1, …."""
    desc = np.sort(vec)[::-1]  # descending
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    toggle = True
    for v in desc:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    s = out.sum()
    return out / s if s > 0 else out


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(
    n: int, sigma_frac: float = 0.07, offset_frac: float = 0.35
) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed generation – richer than the reference
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))

    # sine windows on a grid
    for gamma in np.linspace(0.5, 6.0, 12):
        sv = _sine_window(n, gamma)
        seeds.append(sv)
        seeds.append(_alternate_order(sv))

    # Gaussians on a grid
    for sigma in np.linspace(0.07, 0.30, 12):
        gv = _gaussian(n, sigma)
        seeds.append(gv)
        seeds.append(_alternate_order(gv))

    # Beta shapes – random draws
    for _ in range(40):
        alpha = 10 ** rng.uniform(-3, 1.0)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1.0)
        bv = _beta_shape(n, alpha, beta)
        seeds.append(bv)
        seeds.append(_alternate_order(bv))

    # Exponential
    for rate in np.logspace(-5, -0.5, 12):
        ev = _exponential(n, rate)
        seeds.append(ev)
        seeds.append(_alternate_order(ev))

    # Logarithmic
    for eps in np.logspace(-5, -0.5, 10):
        lv = _logarithmic(n, eps)
        seeds.append(lv)
        seeds.append(_alternate_order(lv))

    # Dirichlet draws (various concentrations)
    concs = [0.001, 0.005, 0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)
            seeds.append(_alternate_order(raw))

    # Random cosine mixtures
    for nf in (2, 3, 4):
        for _ in range(3):
            t = np.arange(n, dtype=np.float64)
            curve = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                curve += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            curve = np.maximum(curve, 0.0)
            curve /= curve.sum()
            mixed = 0.7 * _uniform(n) + 0.3 * curve
            mixed /= mixed.sum()
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Wide random sweep of parameters
    for _ in range(80):
        typ = rng.choice(["beta", "gauss", "sine", "dirichlet"])
        if typ == "beta":
            alpha = 10 ** rng.uniform(-4, 2)
            beta = 10 ** rng.uniform(-4, 2)
            sv = _beta_shape(n, alpha, beta)
        elif typ == "gauss":
            sigma_frac = rng.uniform(0.02, 0.5)
            sv = _gaussian(n, sigma_frac)
        elif typ == "sine":
            gamma = rng.uniform(0.3, 12.0)
            sv = _sine_window(n, gamma)
        else:
            conc = rng.uniform(0.001, 3.0)
            sv = rng.dirichlet(np.full(n, conc))
        seeds.append(sv)
        seeds.append(_alternate_order(sv))

    # Include previous solution if available
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            norm = avg / avg.sum()
            seeds.append(norm)
            seeds.append(_alternate_order(norm))

    return seeds


# ----------------------------------------------------------------------
# Mirror‑descent optimiser based on a smooth p‑norm approximation of max
# ----------------------------------------------------------------------
def _p_norm_mirror_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    p_start: float = 4.0,
    p_end: float = 64.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on
        J_p(a) = 2·n·||conv(a,a)||_p,
    with p linearly increased from p_start to p_end.
    The iterate lives on the simplex (ℓ₁‑norm = 1).
    """
    a = np.copy(start_vec)
    s = a.sum()
    a = a / s if s > 0 else _uniform(n)

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        p = p_start + (p_end - p_start) * it / max_iters

        conv = _conv_fft(a, a)
        Lp = np.linalg.norm(conv, ord=p)
        if Lp == 0.0:
            continue

        # weight for the gradient of the p‑norm
        w = np.power(conv, p - 1)
        denom = Lp ** (p - 1) if p > 1 else 1.0
        w = (p / (denom + 1e-30)) * w

        # gradient of 2·n·Lp w.r.t. a (constant factor omitted)
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # exponentiated‑gradient (mirror) update
        lr = 0.30 / np.sqrt(it)
        a = a * np.exp(-lr * grad)

        # project back to the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[pNorm] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation hill‑climbing (swap‑only) – a bit more aggressive than the reference
# ----------------------------------------------------------------------
def _perm_hill_climb(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_swaps: int = 150_000,
) -> tuple[np.ndarray, float]:
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    it = 0
    T0 = cur_val * 0.005
    T = T0

    while it < max_swaps and time.time() < deadline:
        it += 1
        T = max(T * 0.99995, 1e-14)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30_000 == 0:
                    print(f"[Perm] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (swap & mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.015,
    min_temp: float = 1e-14,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-14)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:  # redistribute a small random subset
            k = rng.integers(2, 10)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:  # pair transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Deterministic permutation utilities
# ----------------------------------------------------------------------
def _evaluate_permutations(base_seq: np.ndarray, n: int) -> np.ndarray:
    """Return the best permutation among a handful of deterministic schemes."""
    candidates = [
        base_seq,
        _alternate_order(base_seq),
        _balanced_alternate(base_seq),
        base_seq[::-1],
        np.sort(base_seq),
        np.sort(base_seq)[::-1],
    ]
    rng = np.random.default_rng()
    for _ in range(5):
        candidates.append(base_seq[rng.permutation(n)])

    best_seq = candidates[0]
    best_val = _objective(best_seq)
    for cand in candidates[1:]:
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand
    return best_seq

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence() within the supplied time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # Length – 4000 offers a good trade‑off.
    n = 4000

    # Load previous best, if any.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Phase 0 – generate a rich pool of seed vectors.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise global best.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – p‑norm mirror descent refinement of the top seeds.
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        max_iters = int(min(150_000, max(5_000, remaining * 30)))
        print(f"\n--- p‑norm mirror descent run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_opt, val_opt = _p_norm_mirror_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[overall‑pNorm] new global best = {val_opt:.6f}")

        # Try a few deterministic reorderings of the refined vector.
        perm_seq = _evaluate_permutations(seq_opt, n)
        perm_val = _objective(perm_seq)
        if perm_val < global_best_val:
            global_best_seq, global_best_val = perm_seq.copy(), perm_val
            print(f"[overall‑perm] after reordering new best = {perm_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – extended permutation hill‑climbing on the current best.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation hill‑climbing (extended) ---")
        seq_perm, val_perm = _perm_hill_climb(
            global_best_seq, n, deadline, max_swaps=200_000
        )
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[overall‑hill] new global best = {val_perm:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final simulated‑annealing polish.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final simulated‑annealing polish ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=1_200_000,
            init_temp_factor=0.015,
            min_temp=1e-14,
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible reuse by later runs.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
