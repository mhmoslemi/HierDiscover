"""
Propose a high‑quality step‑function sequence for the ``evaluate_sequence`` challenge.

Algorithm
---------
1. **Extensive seeding** – deterministic analytic shapes (uniform,
   triangular, sine‑window, Gaussian, beta, Tukey, exponential,
   cosine) together with a very large collection of Dirichlet draws and
   low‑frequency random curves.  Every seed is ℓ₁‑normalised (∑ a = 1).

2. **Fast surrogate ranking** – the exact objective
   ``2·n·max(conv(a,a))`` (with sum = 1) is cheap to compute via FFT.
   The best *top_k* seeds are kept.

3. **Adam on a smooth log‑sum‑exp surrogate** – for each of the best
   seeds we run Adam on the log‑softmax parametrisation of the vector.
   The temperature of the log‑sum‑exp is ramped up to 1000, making the
   surrogate arbitrarily close to the true max.  This yields a high‑
   quality starting point.

4. **Multiplicative mirror‑subgradient descent** – an exact
   sub‑gradient of ``max(conv)`` is applied with an exponentiated‑gradient
   (mirror) update on the probability simplex.  The step‑size decays
   slowly (power ≈ 0.25) and the loop runs as long as time permits.

5. **Iterated LP polishing** – for the current vector *f* we solve
   ``max ∑ g_i`` subject to the *linear* constraints ``conv(f,g) ≤ M``,
   where ``M = max(conv(f,f))``.  The solution ``g`` is mixed into *f*
   with a relatively large weight (τ ≈ 0.35) and renormalised.
   Several LP rounds are performed until no further L₁‑mass gain.

6. **Second mirror phase** – after LP polishing we run a short mirror
   refinement to exploit the now flatter convolution landscape.

7. **Light simulated‑annealing** – final few seconds are spent on a
   lightweight SA that swaps, redistributes or transfers mass between
   random positions.

All stages respect the global time budget (default 1000 s).  The best
sequence is stored globally as ``height_sequence_1`` and returned as a

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (scale independent) – 2·n·max(conv(seq,seq))
    with the sequence already ℓ₁‑normalised (∑seq = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve built from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a large, diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # Fine grids for analytic families
    for gamma in np.linspace(0.1, 6.0, 25):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma in np.linspace(0.04, 0.60, 20):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    for p in np.linspace(0.0, 8.0, 21):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.15, 4.0, 15)
    betas = np.linspace(0.15, 4.0, 15)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for alpha in np.linspace(0.02, 0.98, 20):
        seeds.append(_tukey_window(n, alpha))

    for lam in np.linspace(0.0001, 0.03, 20):
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet draws – many concentrations and repetitions
    conc_vals = [0.001, 0.004, 0.008, 0.015, 0.03, 0.06,
                 0.12, 0.25, 0.5, 1.0, 2.0]
    for conc in conc_vals:
        for _ in range(80):  # 80 draws per concentration → 880 seeds
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(30):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.05, 0.15, 0.30, 0.45, 0.60, 0.80):
        for nf in (2, 3, 5):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    """Adam on logits (softmax) with a temperature‑controlled log‑sum‑exp."""
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.7
    beta = 0.5            # start temperature
    beta_inc = 0.5
    beta_max = 1000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probability simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # --- smooth max via log‑sum‑exp ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_optimize(start_seq: np.ndarray, n: int, deadline: float,
                    max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max‑conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.18          # start step size (larger than the reference)
    power = 0.25        # slower decay → more aggressive later steps

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # Sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update (mirror descent)
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 500_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Lightweight SA polishing (swap / redistribution / pair‑transfer)."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:            # swap two positions
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:          # redistribute mass among k random indices
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                         # split / merge between a pair
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """Return A such that (A @ g) = conv(f, g) (full linear convolution)."""
    n = len(f)
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for k in range(n):
        start = k
        end = k + n
        rows[idx:idx + n] = np.arange(start, end, dtype=np.int32)
        cols[idx:idx + n] = k
        data[idx:idx + n] = f
        idx += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 15, tau: float = 0.35) -> tuple[np.ndarray, float]:
    """
    Perform several LP polishing steps.
    Each step solves: maximise ∑g_i  s.t. conv(seq,g) ≤ M   (with M = max(conv(seq,seq))).
    The solution g is mixed back with weight ``tau``.
    """
    f = seq.copy()
    n = len(f)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()
        A = _build_conv_matrix(f)            # (2n‑1) × n, sparse
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)    # maximise Σ g_i

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if not res.success:
            print(f"[LP] iteration {it} failed – aborting LP phase.")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:          # no L1‑mass increase → stop
            print(f"[LP] iteration {it} – LP could not increase L1‑mass.")
            break

        # convex combination and renormalisation
        f = (1.0 - tau) * f + tau * g
        f = f / f.sum()

        cur_obj = _objective(f)
        print(f"[LP] iter {it} – objective = {cur_obj:.6f}")

    return f, _objective(f)

        print(f"\n--- Adam run {idx}/{max_adam} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent sub‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 45))
        print("\n--- Mirror‑descent refinement ---")
        seq_mirror, val_mirror = _mirror_optimize(global_best_seq, n, deadline,
                                                  max_iters=max_iters)
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Linear‑programming polishing (multiple iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming polishing (up to 15 rounds) ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, deadline,
                                    max_iters=15, tau=0.35)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Second mirror phase (fine‑tuning after LP)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(200_000, remaining * 30))
        print("\n--- Second mirror‑descent phase ---")
        seq_mirror2, val_mirror2 = _mirror_optimize(global_best_seq, n, deadline,
                                                    max_iters=max_iters)
        if val_mirror2 < global_best_val:
            global_best_seq = seq_mirror2.copy()
            global_best_val = val_mirror2
            print(f"[overall‑Mirror2] new global best = {val_mirror2:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Simulated‑annealing polishing (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(500_000, remaining * 2e3))  # attempt many cheap moves
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=max_iters
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (purely informational)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
