"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a))²

The ratio is scale‑invariant, therefore the search works on ℓ₁‑normalised
vectors (sum(a)=1) and optimises the surrogate objective

    2·n·max(conv(a,a))

The pipeline is:

1. **Rich seed pool** – analytic families (uniform, triangular, cosine,
   sine‑window, Gaussian, β‑shape, Tukey, exponential, raised‑cosine,
   logistic) together with many random Dirichlet draws and low‑frequency
   cosine curves.

2. **Fast surrogate ranking** – the surrogate is evaluated via an FFT‑based
   convolution and the best few seeds are kept.

3. **Smooth‑max L‑BFGS refinement** – each top seed is optimised with
   SciPy’s L‑BFGS‑B on a log‑sum‑exp approximation of the max.  The
   temperature is increased gradually (5 → 10 → 30 → 75 → 150 → 300 → 600)
   while allowing many iterations per temperature.

4. **Exact sub‑gradient polishing** – two complementary schemes are applied:
   *exponentiated‑gradient* (mirror descent) on the simplex with a
   carefully‑scaled exact sub‑gradient, and an *Adam* optimiser on the
   softmax logits using the true sub‑gradient of the max‑convolution.
   Both run until the time budget is exhausted.

5. **Simulated‑annealing polishing** – lightweight SA with swap,
   Dirichlet‑redistribute and pair‑transfer moves escapes any remaining
   shallow minima.

All stages respect a hard 1000 s deadline; the best sequence found is
stored in the global name ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# FFT helpers (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Analytic families – each returns an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _cosine_window(n)
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1.0)))
        elif i <= (n - 1) * (1.0 - alpha / 2.0):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2.0 / alpha + 1.0)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine_power(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, steep: float = 10.0) -> np.ndarray:
    """Logistic CDF shaped hill, centred, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(-steep * (i - centre) / n))
    a = a - a.min()
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Surrogate objective (fast, identical ordering to official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient w.r.t. the vector a
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """Return (smooth_max, grad_a) where smooth_max approximates max(conv)."""
    conv = _conv_fft(a, a)
    shift = np.max(conv)               # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    w = exp_term / exp_term.sum()      # softmax weights, sum to 1
    smooth = shift + np.log(exp_term.sum()) / beta

    # Gradient of weighted sum: d/d a_i  Σ_j w_j (a * a)[j]
    # (a*a)[j] = Σ_t a_t a_{j-t}
    # ∂/∂a_i = Σ_t a_t (δ_{i, j-t} + δ_{i,t}) = 2 Σ_t a_t δ_{i, j-t}
    # which is exactly 2 * (rev_a convolved with w) as used before.
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad


# ----------------------------------------------------------------------
# L‑BFGS on smooth‑max surrogate
# ----------------------------------------------------------------------
def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 300,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the log‑sum‑exp surrogate."""
    eps = 1e-30
    v = np.log(start_seq + eps)                     # logits – unconstrained

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        # back‑prop through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return loss, (2.0 * n * grad_v).astype(np.float64)

    # SciPy optimizer – we guard against time overruns outside
    res = scipy.optimize.minimize(
        fun,
        v,
        method="L-BFGS-B",
        jac=True,
        options={"maxiter": max_iter, "disp": False},
    )
    best_a = _softmax(res.x)
    return best_a, _objective(best_a)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) w.r.t. a
# ----------------------------------------------------------------------
def _max_conv_subgrad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (max_val, grad_a) where grad_a is a sub‑gradient of max(conv)."""
    conv = _conv_fft(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    n = len(a)
    grad = np.zeros_like(a)

    # For each i such that 0 <= i <= n-1 and 0 <= j = max_idx-i <= n-1
    # contribution is a_j (or 2*a_i when i == j)
    for i in range(n):
        j = max_idx - i
        if 0 <= j < n:
            if i == j:
                grad[i] += 2.0 * a[i]
            else:
                grad[i] += a[j]
    return max_val, grad


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑conv objective
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 4_000_000,
                                 lr0: float = 0.15,
                                 power: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative updates on the simplex using the exact sub‑gradient."""
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        cur_val, g = _max_conv_subgrad(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:,d}  best={best_val:.6f}")

        # learning rate schedule
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)          # multiplicative update

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on logits using the *exact* max‑convolution sub‑gradient
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 2_000_000,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam on logits v (a = softmax(v)) using the exact sub‑gradient of max(conv)."""
    v = np.log(start_seq + 1e-30)          # initialise logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    best_a = start_seq.copy()
    best_val = _objective(best_a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        cur_val, g_a = _max_conv_subgrad(a)            # g_a = ∂ max(conv) / ∂ a
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if t % 200_000 == 0:
                print(f"[Adam] it={t:,d}  best={best_val:.6f}")

        # back‑prop through softmax
        dot = np.dot(g_a, a)
        g_v = a * (g_a - dot)                         # ∂O/∂v
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (lightweight) for final polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9993, min_temp)

        move_type = rng.integers(0, 3)   # 0=swap, 1=redistribute, 2=pair‑transfer
        if move_type == 0:               # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:             # redistribute a random subset
            k = rng.integers(2, 15)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                            # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 250_000 == 0:
                    print(f"[SA] it={it:,d}  best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_gaussian(n, sigma_frac=0.2))
    seeds.append(_raised_cosine_power(n, power=1.0))
    seeds.append(_logistic_shape(n, steep=8.0))
    seeds.append(_tukey_window(n, alpha=0.5))
    seeds.append(_exponential_decay(n, lam=0.005))

    # parameter sweeps for families
    for gamma in np.linspace(0.4, 3.5, 10):
        seeds.append(_sine_window(n, gamma=gamma))
    for sigma in np.linspace(0.1, 0.35, 8):
        seeds.append(_gaussian(n, sigma_frac=sigma))
    for p in np.linspace(0.0, 5.0, 9):
        seeds.append(_power(n, p))
    for alpha in np.linspace(0.05, 0.95, 9):
        seeds.append(_tukey_window(n, alpha))

    # beta‑shape grid (covers “U‑shaped” and bell‑shaped)
    alphas = np.concatenate((np.linspace(0.05, 0.3, 4), np.linspace(0.3, 3.0, 7)))
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform & low‑freq curves
    for mix in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_uniform(n) * (1 - mix) + curve * mix)

    # perturbations of the previous best (if any)
    if previous is not None and previous.shape == (n,):
        base = previous / previous.sum()
        for _ in range(5):
            pert = base + rng.normal(scale=0.01, size=n)
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            seeds.append(pert)

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    `evaluate_sequence` objective within the allocated time budget.
    Returns the best sequence as a list of floats and stores it as the
    global ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose a length – 4500 works well: FFT still cheap, extra DOF helps.
    # ------------------------------------------------------------------
    n = 4500
    print(f"\n=== Starting optimisation (n={n}) ===")

    # ------------------------------------------------------------------
    # Load any previous solution for seeding
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.shape != (n,):
                # resize if length mismatches (simple trunc/zero‑pad)
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.concatenate([previous,
                                               np.zeros(n - previous.size)])
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich pool of seeds
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 25
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i:2d}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – smooth‑max L‑BFGS refinement (gradual temperature schedule)
    # ------------------------------------------------------------------
    temps = [5.0, 10.0, 30.0, 75.0, 150.0, 300.0, 600.0]
    for seed_idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        a = seed_vec.copy()
        for temp in temps:
            if time.time() > deadline:
                break
            a_opt, val_opt = _l_bfgs_optimize(
                a, n, deadline, max_iter=300, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = a_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{seed_idx}, temp={temp:.0f} → new best {val_opt:.6f}")
            a = a_opt   # warm‑start next temperature

    # ------------------------------------------------------------------
    # Stage 2 – exact sub‑gradient polishing (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent polishing (exponentiated‑gradient) ---")
        seq_md, val_md = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=4_000_000,
            lr0=0.15,
            power=0.25,
        )
        if val_md < global_best_val:
            global_best_seq = seq_md.copy()
            global_best_val = val_md
            print(f"[Mirror] new global best = {val_md:.6f}")

    # ------------------------------------------------------------------
    # Stage 2b – Adam on logits (exact sub‑gradient of max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam polishing on logits ---")
        seq_adam, val_adam = _adam_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            lr=0.05,
        )
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for diagnostics only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
