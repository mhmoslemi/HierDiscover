def _smooth_max_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    # conv: length L=2*n-1
    L = conv.shape[0]
    # compute weights (softmax)
    # subtract max to avoid overflow
    scaled = beta * conv
    max_s = np.max(scaled)
    exp_scaled = np.exp(scaled - max_s)  # shape L
    w = exp_scaled / np.sum(exp_scaled)  # softmax weights
    # compute gradient using convolution of a with w_rev
    w_rev = w[::-1]
    # conv_w_rev_a = full conv of w_rev and a (len = L + n -1 = 3n -2)

    # g_i = 2 * conv_wa[(L-1) - np.arange(n)] (since L = 2n-1)
    n = a.shape[0]
    g = 2.0 * conv_wa[(L - 1) - np.arange(n)]
    return g
```

"""
Hybrid variance‑flattening + smoothed‑max mirror descent.

1. Generate a modest pool of analytically‑named seeds (uniform,
   triangular, trapezoidal, geometric, beta, Gaussian …) and random
   Dirichlet draws.  The best seed (minimum surrogate 2·n·max(conv))
   becomes the start point.

2. **Variance flattening** – projected gradient descent on the variance
   of the full autocorrelation `a*a`.  Since the mean of the convolution
   is fixed (sum(a)^2/(2n‑1)), minimising the variance pushes all
   entries of the convolution towards the flat value, i.e. towards the
   theoretical lower bound 2n/(2n‑1)≈1.

3. **Smoothed‑max mirror descent** – a multiplicative (entropic) mirror
   step using a soft‑max approximation of the `max` (temperature `β`).
   The schedule `β = 2, 4, 8, …, 256` gradually turns the smooth
   objective into the true `max`.  After each step the iterate is
   projected back onto the probability simplex.

4. Optional tiny stochastic polishing (Dirichlet reshuffle) for a few
   seconds.

The routine respects the user supplied time budget (default 1000 s) and
stores the best sequence in the global variable ``height_sequence_1``.
All numbers are non‑negative and the final sequence is returned as a
plain Python list.
"""

import time
import numpy as np
from typing import List, Tuple

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray, n: int) -> float:
    """Exact surrogate objective for ℓ₁‑normalised sequences (sum = 1):
    2·n·max(conv).
    """
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> List[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidates."""
    seeds: List[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12), np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _variance_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    init_lr: float = 0.15,
    max_no_improve: int = 2000,
) -> Tuple[np.ndarray, float]:
    """
    Projected gradient descent on the variance of the autocorrelation.
    The mean of the convolution is fixed (1/(2n‑1)), so minimising the
    variance pushes the whole convolution towards the flat target.
    """
    a = start_vec.copy()
    L = 2 * n - 1
    avg = 1.0 / L                      # mean of convolution for sum(a)=1
    best_seq = a.copy()
    best_val = _objective(a, n)

    cur_val = best_val
    it = 0
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        it += 1
        conv = _conv_fft(a, a)                     # length L
        w = conv - avg                              # deviation from flatness
        # gradient = (4/L) * convolution( w_rev , a )
        w_rev = w[::-1]
        conv_w = _conv_fft(w_rev, a)                # length 3n‑2
        g = (4.0 / L) * conv_w[(L - 1) - np.arange(n)]

        # Scale the gradient to unit norm to obtain a robust step size.
        g_norm = g / (np.linalg.norm(g) + 1e-12)

        # diminishing learning rate
        lr = init_lr / np.sqrt(it)

        # try a simple back‑track line search (max 3 attempts)
        accepted = False
        trial_lr = lr
        for _ in range(3):
            a_new = a - trial_lr * g_norm
            a_new = _proj_simplex(a_new)
            val_new = _objective(a_new, n)
            if val_new < cur_val:
                a = a_new
                cur_val = val_new
                accepted = True
                no_improve = 0
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                break
            else:
                trial_lr *= 0.5

        if not accepted:
            no_improve += 1

        if it % 5000 == 0:
            print(f"[VarDescent] it={it:6d} cur={cur_val:.6f} best={best_val:.6f}")

    return best_seq, best_val


def _smooth_max_mirror_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: List[float] = None,
    init_eta: float = 0.6,
) -> Tuple[np.ndarray, float]:
    """
    Entropic mirror descent using a soft‑max approximation of the
    `max` (temperature β).  β is increased geometrically, turning the
    smoothed objective into the true ℓ∞‑objective.
    """
    if beta_schedule is None:
        beta_schedule = [2 ** k for k in range(1, 9)]  # 2,4,…,256

    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    it_total = 0
    for beta in beta_schedule:
        # Give each β a fair share of the remaining time
        phase_deadline = time.time() + (deadline - time.time()) * 0.5
        while time.time() < phase_deadline and time.time() < deadline:
            it_total += 1
            conv = _conv_fft(a, a)                     # length L = 2n‑1
            # soft‑max weights
            scaled = beta * conv
            max_s = np.max(scaled)
            exp_scaled = np.exp(scaled - max_s)        # avoid overflow
            w = exp_scaled / np.sum(exp_scaled)       # Σ w = 1

            # gradient of smooth‑max w.r.t a: g_i = 2 * Σ_j w_{i+j} a_j
            w_rev = w[::-1]
            conv_w = _conv_fft(w_rev, a)                # length 3n‑2
            g = 2.0 * conv_w[(2 * n - 2) - np.arange(n)]

            # entropic (multiplicative) update
            eta = init_eta / np.sqrt(it_total)
            a = a * np.exp(-eta * g)
            a = _proj_simplex(a)

            cur_val = _objective(a, n)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it_total % 10000 == 0:
                    print(f"[SmoothMirror] β={beta} it={it_total} best={best_val:.6f}")

    return best_seq, best_val


def _local_dirichlet_polish(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Very light stochastic polishing – repeatedly select a small random
    block and reshuffle its mass via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        k = rng.integers(2, 13)            # block size
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new, n)

        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[Polish] it={it} best={best_val:.6f}")

    return best_seq, best_val

        if val_var < global_best_val:
            global_best_seq, global_best_val = seq_var, val_var
            print(f"[Var] new global best = {val_var:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smoothed‑max mirror descent
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        mirror_deadline = time.time() + 0.55 * time_left
        print("\n--- Smoothed‑max mirror descent ---")
        seq_smooth, val_smooth = _smooth_max_mirror_descent(
            global_best_seq, n, mirror_deadline
        )
        if val_smooth < global_best_val:
            global_best_seq, global_best_val = seq_smooth, val_smooth
            print(f"[SmoothMirror] new global best = {val_smooth:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – tiny stochastic polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final stochastic polishing (Dirichlet moves) ---")
        seq_polish, val_polish = _local_dirichlet_polish(
            global_best_seq, n, deadline
        )
        if val_polish < global_best_val:
            global_best_seq, global_best_val = seq_polish, val_polish
            print(f"[Polish] final global best = {val_polish:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
