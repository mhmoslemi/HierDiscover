def _softmax_descent(start_vec, n, deadline, max_iters=20000, lr0=0.1, lambda0=5.0, lambda_factor=1.05):
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    lam = lambda0
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)  # length 2n-1
        # compute softmax weights
        # scaling conv to avoid overflow:
        scaled = lam * conv
        max_scaled = np.max(scaled)
        exps = np.exp(scaled - max_scaled)
        w = exps / np.sum(exps)
        # compute gradient via convolution of a_rev and w
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, w)
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]
        # optionally clip gradient magnitude to avoid huge steps
        lr = lr0 / np.sqrt(it)
        a = a - lr * grad
        # project onto simplex
        a = _proj_simplex(a)
        # compute objective:
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # schedule lambda (increase)
        lam = min(lam * lambda_factor, 200.0)  # cap
    return best_seq, best_val

"""
Optimiser for the “step‑function” challenge.

Approach
--------
1. **Seed pool** – a modest collection of analytically generated vectors
   (uniform, triangular, beta‑shaped, Gaussian…) together with a few random
   Dirichlet draws and the previous best solution (if any).  All vectors are
   ℓ₁‑normalised (sum = 1).

2. **Soft‑max gradient descent** – we minimise a smooth approximation of
   ``max(conv(a,a))``:
        φ(a) = (1/λ)·log∑ₖ exp(λ·(a∗a)ₖ) .
   Its gradient is
        ∇φ(a) = 2·(rev(a) ∗ w)   with   wₖ ∝ exp(λ·(a∗a)ₖ) .
   We update with a diminishing step‑size and project onto the probability
   simplex.  The soft‑max temperature λ is increased gradually, turning the
   smooth loss into the true max‑loss.  Random jitter (Dirichlet reshuffle)
   is injected occasionally to escape plateaus.

3. **Simulated annealing polishing** – a lightweight SA that repeatedly
   redistributes the mass of a random small block via a Dirichlet move and
   accepts worsening moves with a geometric temperature schedule.

The routine respects a user‑supplied time budget (default 1000 s) and
stores the best sequence found in the global ``height_sequence_1`` variable.

All heavy work (convolutions) is performed with a real FFT for speed.
The surrogate objective ``_objective`` equals the official one for
ℓ₁‑normalised vectors, so we can work with it safely.
"""

import time
import numpy as np
from scipy import sparse

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Reference: https://arxiv.org/abs/1101.6081 (Algorithm 1)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate objective for ℓ₁‑normalised sequences (sum = 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – a range of concentrations
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any) – re‑normalise to sum = 1
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)
    return seeds

def _softmax_gradient_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    lr0: float = 0.5,
    lambda0: float = 5.0,
    lambda_factor: float = 1.02,
    jitter_interval: int = 5_000,
    jitter_alpha: float = 0.05,
    rng: np.random.Generator | None = None,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation
        φ(a) = (1/λ)·log∑ₖ exp(λ·(a∗a)ₖ)

    Parameters
    ----------
    start_vec     – initial ℓ₁‑normalised vector
    n             – length of the vector (kept for readability)
    deadline      – Unix time when the routine must stop
    max_iters     – hard iteration cap (actual run may stop earlier)
    lr0           – initial learning‑rate (scaled by 1/√t)
    lambda0       – initial soft‑max temperature
    lambda_factor – multiplicative increase of λ each iteration
    jitter_interval – how many iterations between random Dirichlet jitter
    jitter_alpha  – mixing factor for the jitter (0 → no jitter)
    rng           – random generator (created if None)
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lam = lambda0
    last_improve_it = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full self‑convolution
        conv = _conv_fft(a, a)                     # length 2n‑1

        # soft‑max weights (stable version)
        scaled = lam * conv
        max_scaled = scaled.max()
        exps = np.exp(scaled - max_scaled)
        w = exps / exps.sum()                      # shape (2n‑1,)

        # gradient = 2·(rev(a) ∗ w)  (take the n‑length centre slice)
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, w)            # length 3n‑2
        start = n - 1
        grad = 2.0 * grad_full[start : start + n]

        # step‑size schedule
        lr = lr0 / np.sqrt(it)

        # gradient step & simplex projection
        a = a - lr * grad
        a = _proj_simplex(a)

        # occasional random perturbation to escape shallow plateaus
        if jitter_alpha > 0 and (it % jitter_interval == 0):
            jitter = rng.dirichlet(np.full(n, 1.0))
            a = (1.0 - jitter_alpha) * a + jitter_alpha * jitter
            a = _proj_simplex(a)

        # monitor best value
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve_it = it
            if it % 10_000 == 0:
                print(f"[Softmax] it={it:6d} best={best_val:.6f}")

        # increase λ slowly – it sharpens the approximation to the true max
        lam = min(lam * lambda_factor, 300.0)

        # early‑stop if no improvement for a long stretch
        if it - last_improve_it > 50_000:
            break

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Light‑weight simulated‑annealing polishing.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9993, min_temp)

        # pick a small random block and reshuffle its mass
        k = rng.integers(2, 13)               # block size 2‑12
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence found as a Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) for warm‑starting
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the candidate vector – a compromise between expressivity
    # and optimisation speed.
    # ------------------------------------------------------------------
    n = 2500

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and quick evaluation
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – soft‑max gradient descent (multiple restarts)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        descent_deadline = time.time() + 0.65 * time_left
        per_seed_deadline = descent_deadline - time.time()
        # Split remaining descent budget evenly among the top seeds
        per_seed_time = per_seed_deadline / top_k
        print("\n--- Soft‑max gradient descent ---")
        for idx, seed_vec in enumerate(best_seeds):
            print(f"  > Descent run #{idx+1}")
            it_dead = time.time() + per_seed_time
            seq, val = _softmax_gradient_descent(
                seed_vec,
                n,
                it_dead,
                max_iters=400_000,
                lr0=0.6,
                lambda0=5.0,
                lambda_factor=1.025,
                jitter_interval=4_000,
                jitter_alpha=0.07,
                rng=rng,
            )
            if val < global_best_val:
                global_best_seq, global_best_val = seq, val
                print(f"    [Update] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            init_temp_factor=0.04,
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        official_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible warm‑starts in later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
