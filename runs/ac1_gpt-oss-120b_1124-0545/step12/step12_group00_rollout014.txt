def _softmax_max_optimize(start, n, deadline, max_iters, eta0=0.1, alpha_start=10.0, alpha_end=50000.0, beta1=0.9, beta2=0.999):
    x = start.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_x = x.copy()
    best_val = _objective_surrogate(x)  # use surrogate 2*n*max(conv)
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute conv
        conv = _conv_fft(x, x)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        # maybe evaluate true objective periodic; not needed for surrogate
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
        # schedule alpha and eta
        alpha = alpha_start + (alpha_end - alpha_start) * (t - 1) / max_iters
        # compute softmax weights
        # For numerical stability, subtract max:
        exp_arg = alpha * (conv - conv.max())
        # Clip to avoid overflow: though values negative
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient via convolution
        # conv2 = a * w
        conv2 = _conv_fft(x, w)  # length 3n-2
        # conv_rev = a * rev(w)
        conv_rev = _conv_fft(x, w[::-1])
        L = len(conv)
        grad = conv2[:n] + conv_rev[(L-1) - np.arange(n)]
        grad = 2.0 * grad
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = eta0 / (np.sqrt(v_hat) + 1e-8)
        x = x - step * m_hat
        # projection onto simplex
        s = x.sum()
        if s <= 0 or np.any(np.isnan(x)):
            # reinitialize
            x = _proj_simplex(x)
        else:
            x = x / s
            # ensure non-negativity via projection
            x = _proj_simplex(x)  # this ensures sum=1
        # or maybe use _proj_simplex directly
        # we also ensure values remain low (clamp to 1000). Not needed.

    return best_x, best_val

-----------------

A self‑contained search for a non‑negative step function that minimises

The algorithm consists of three phases:

1. **Seeding** – a large library of analytic windows (uniform,

   FFTs (three convolutions per iteration).  An Adam optimiser on the
   Euclidean simplex (projection onto the probability simplex after each
   update) drives the convolution peak down.  The best seed is refined
   for a long run; shorter runs are performed on the other top seeds.

3. **Final simulated‑annealing polish** – a fast SA routine reshuffles
   mass on random subsets and performs tiny transfers between positions,
   accepting any surrogate improvement (occasionally a worsening move
   according to a slowly cooling temperature).  This cleans up any
   remaining local irregularities.

The whole search respects the global deadline (default 1000 s).  The
final sequence is stored globally as ``height_sequence_1`` and returned
as a plain ``list[float]``.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    For a sequence with Σseq = 1 it equals
        2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    """Monotonically decreasing exponential, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (richer than the previous version)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families ------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0, 4.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.25, 0.3):
        seeds.append(_gaussian(n, sigma))
    for rate in (0.999, 0.9995, 0.9998):
        seeds.append(_exponential_decay(n, rate))
    for p in (0.3, 0.5, 0.7, 1.0, 1.5):
        seeds.append(_power(n, p))
    # Beta shapes – finer grid including extreme values
    for a in (0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        for b in (0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    # Tukey windows
    for alpha in (0.1, 0.2, 0.4, 0.6, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mix low‑frequency curves with uniform (helps escape symmetry)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution, if supplied
    if previous is not None and previous.shape == (n,) and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise everything (safety) ----------------------------------------
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam on the simplex)
# ----------------------------------------------------------------------
def _smooth_max_optimize(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 0.05,
                         alpha_start: float = 10.0,
                         alpha_end: float = 50_000.0,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth surrogate
            F(a) = log‑sum‑exp(α·conv(a,a))

    Uses Adam‑style moments, Euclidean projection onto the simplex after
    each step.  ``alpha`` is linearly increased from ``alpha_start`` to
    ``alpha_end`` so that the optimisation first moves in a smooth
    landscape and later focuses on the true maximum.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    L = 2 * n - 1  # length of a full self‑convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # 1️⃣  current convolution and surrogate
        conv = _conv_fft(x, x)          # length L
        cur_val = 2.0 * n * np.max(conv)   # exact surrogate (Σx = 1)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

        # 2️⃣  smooth‑max weights
        #    w_i ∝ exp(α·conv_i)  (shifted by max for stability)
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max_iters
        # shift to avoid overflow
        shift = conv.max()
        exp_arg = alpha * (conv - shift)
        # clip extreme negatives to stay in float64 range
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / L)
        else:
            w /= w_sum

        # 3️⃣  gradient of the smooth surrogate
        #    grad_j = 2 * Σ_i a_i (w_{j+i} + w_{j-i})
        conv_fw = _conv_fft(x, w)                 # a * w
        conv_rev = _conv_fft(x, w[::-1])           # a * rev(w)

        #   term from w_{j-i}
        term2 = conv_fw[:n]                         # indices 0 … n‑1

        #   term from w_{j+i}
        term1 = conv_rev[(L - 1) - np.arange(n)]

        grad = 2.0 * (term1 + term2)

        # 4️⃣  Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = eta0 / (np.sqrt(v_hat) + eps)
        x = x - step * m_hat

        # 5️⃣  project back onto the simplex (keep non‑negativity and Σ=1)
        x = _proj_simplex(x)

        # occasional progress report (quietly throttled)
        if it % 50000 == 0:
            print(f"[SmoothMax] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (kept from the previous version)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 400_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        # two move types (re‑shuffle a small subset or tiny transfer)
        if rng.random() < 0.6:
            k = rng.integers(3, 13)          # size of the reshuffle set
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the supplied time budget.
    Returns the best sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a safety margin

    # --------------------------------------------------------------
    # Length selection – a compromise between quality and speed.
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed candidates.")

    # rank seeds by the cheap surrogate and keep the top few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – short smooth‑max refinements on each top seed.
    # --------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest budget per seed (≈0.5 % of remaining time)
        max_iters = int(min(50_000, remaining * 1200))
        if max_iters < 1_000:
            continue
        cand, val = _smooth_max_optimize(seed_vec, n, deadline,
                                         max_iters=max_iters,
                                         eta0=0.07,
                                         alpha_start=5.0,
                                         alpha_end=20_000.0)
        if val < best_val:
            best_val, best_seq = val, cand
            print(f"[QuickSmooth] seed #{idx+1} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – long‑run smooth‑max optimisation on the current best.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long‑run smooth‑max optimisation ---")
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 1200))
        best_seq, best_val = _smooth_max_optimize(best_seq, n, deadline,
                                                 max_iters=max_iters,
                                                 eta0=0.05,
                                                 alpha_start=20.0,
                                                 alpha_end=50_000.0)
        print(f"[LongSmooth] best surrogate after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – simulated‑annealing final tweak.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_iters)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
