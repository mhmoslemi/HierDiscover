"""
Improved optimiser for the “step‑function” challenge.

The algorithm combines three ideas that were not used together before:

1. **Sidon‑set seed** – a deterministic sparse seed built from triangular
   numbers (a classic Sidon set).  Its convolution has almost no
   collisions, giving a very low initial max‑convolution value.

2. **Linear‑programming refinement** – for a fixed non‑uniform shape `f`
   we solve a linear programme that maximises the total mass of a new
   vector `b` while keeping every entry of `f * b` below the current
   worst convolution value `M`.  This forces the mass of `b` into a
   pattern that reduces the active “peak’’ of the convolution.  The LP
   is solved with `scipy.optimize.linprog` on a sparse constraint matrix.

3. **Log‑space Adam + sub‑gradient polishing** – after the LP stage the
   vector is up‑sampled to the final length (8000) and a long Adam run
   on the log‑softmax parametrisation is performed, followed by an
   exact sub‑gradient projection.  Both respect the global time budget.

All stages abort gracefully when the deadline is reached.  The best
sequence found (as a plain list of floats) is stored in the global name
`height_sequence_1` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the LP stage)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

except Exception:                     # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector
    (∑seq = 1).  It is proportional to the true ``evaluate_sequence`` value.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw (conc < 1 gives sparse vectors)."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a /= a.sum()
    return a


def _sidon_seed(n: int) -> np.ndarray:
    """
    Deterministic Sidon‑set seed: use triangular numbers
    T_i = i(i+1)/2 < n.  Place equal mass on those indices.
    """
    idx = []
    i = 0
    while True:
        t = i * (i + 1) // 2
        if t >= n:
            break
        idx.append(t)
        i += 1
    if not idx:                     # fallback – uniform
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / len(idx)
    return a


def _mix_seeds(seeds):
    """Normalise every seed to ℓ₁ = 1."""
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(len(v)))
    return cleaned

def _prepare_lp_matrix(n: int):
    """
    Prepare the CSR matrix pattern for the constraints conv(f,b) ≤ M.
    The non‑zero pattern does not depend on the actual values of f,
    only on the size n.  Returns (row_idx, col_idx) arrays that can be
    reused – the data values are simply f[col_idx] for the current f.
    """
    rows = []
    cols = []
    for i in range(n):
        # column i contributes to rows i … i+n‑1
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def _lp_refine(
    f: np.ndarray,
    target_sum: float,
    t: float,
    max_iters: int,
    deadline: float,
    row_idx: np.ndarray,
    col_idx: np.ndarray,
) -> np.ndarray:
    """
    Perform a few Frank‑Wolfe style LP steps.
    f is assumed to have ℓ₁‑sum = target_sum.
    Each iteration:
        * compute M = max(conv(f,f))
        * solve   maximise Σ b_j
                  s.t.   conv(f,b) ≤ M ,  b ≥ 0
        * rescale b to the same ℓ₁‑norm (target_sum) and mix:
                f ← (1‑t)·f + t·b
    """
    if not _has_scipy:
        return f

    n = len(f)
    ub_template = None

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(f, f)
        M = np.max(conv)

        # build sparse matrix for the current f
        data = f[col_idx]                     # each entry = f_i for column i
        A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        # right‑hand side – same bound M for every row
        if ub_template is None or ub_template.shape[0] != 2 * n - 1:
            ub_template = np.full(2 * n - 1, M, dtype=np.float64)

        # objective: maximise sum(b)  ⇒ minimise –sum(b)
        c = -np.ones(n, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=ub_template,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # if LP fails (rare), abort the refinement
            break

        b = res.x
        sum_b = b.sum()
        if sum_b == 0:
            break

        # rescale to the target ℓ₁‑norm and convex‑combine
        g = b * (target_sum / sum_b)
        f = (1.0 - t) * f + t * g

    return f


# ----------------------------------------------------------------------
# Adam optimiser (log‑space, smooth max)
# ----------------------------------------------------------------------
def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑softmax parametrisation.  Uses a log‑sum‑exp
    approximation to the max with a temperature that grows up to 4000.
    Returns the best vector visited (ℓ₁‑norm = 1) and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature
    beta_inc = 0.5
    beta_max = 4000.0

    lr0 = 1.0                # initial learning rate (log‑space)

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probability simplex, ℓ₁ = 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Jacobian of softmax (log‑softmax → probabilities)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val

# Sub‑gradient projection (exact max)
# ----------------------------------------------------------------------
def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a decreasing Polyak‑style step‑size and Euclidean projection onto
    the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / np.sqrt(it)          # decreasing step
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalisation."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq

    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final evaluation call
    overall_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Stage 0 – tiny initial resolution (for the LP stage)
    # ------------------------------------------------------------------
    n0 = 1800                               # small enough for fast LPs
    target_sum = np.sqrt(2.0 * n0)          # ℓ₁‑norm required by the LP scheme
    print(f"\n--- LP‑stage initialisation (n={n0}) ---")

    # build a collection of diverse seeds (ℓ₁‑normalised)
    seeds = [
        _uniform(n0),
        _sidon_seed(n0),
        _random_dirichlet(n0, 0.2, rng),
        _random_dirichlet(n0, 0.5, rng),
        _random_dirichlet(n0, 1.0, rng),
        _random_dirichlet(n0, 2.0, rng),
    ]
    seeds = _mix_seeds(seeds)

    # prepare the sparse pattern for the LP constraints (re‑used each iteration)
    row_idx, col_idx = _prepare_lp_matrix(n0)

    best_seq = None
    best_val = float("inf")

    # try each seed, perform a short LP refinement and keep the best
    for i, seed_vec in enumerate(seeds, 1):
        if time.time() > overall_deadline:
            break

        # scale seed to the required ℓ₁‑norm (target_sum)
        f = seed_vec * (target_sum / seed_vec.sum())

        # run a few LP‑based refinement steps
        f_refined = _lp_refine(
            f,
            target_sum=target_sum,
            t=0.25,                     # moderate mixing factor
            max_iters=6,
            deadline=overall_deadline,
            row_idx=row_idx,
            col_idx=col_idx,
        )
        # bring back to ℓ₁ = 1 for surrogate evaluation
        a = f_refined / f_refined.sum()
        val = _objective_surrogate(a)
        print(f"[Seed {i:2d}] surrogate = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"   ** new global best (seed {i}) **")

    if best_seq is None:
        # fallback – uniform vector of length 2000
        best_seq = _uniform(n0)

    print(f"\n[LP‑stage] best surrogate after LP refinement = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample to the final resolution
    # ------------------------------------------------------------------
    n_final = 8000
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        print(f"\n--- Upsampling to n={n_final} ---")
        best_seq = _upsample_sequence(best_seq, n_final)
        best_seq = best_seq / best_seq.sum()
        best_val = _objective_surrogate(best_seq)
        print(f"[Upsample] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long log‑space Adam optimisation
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 60.0:
        max_iters_adam = int(min(400_000, remaining * 3500))
        print(f"\n--- Adam optimisation (max_iters={max_iters_adam}) ---")
        seq_adam, val_adam = _logspace_adam_optimize(
            best_seq, n_final, overall_deadline, max_iters=max_iters_adam
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient projection
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 20.0:
        max_iters_sub = int(min(1_200_000, remaining * 2200))
        print(f"\n--- Sub‑gradient projection (max_iters={max_iters_sub}) ---")
        seq_sub, val_sub = _subgradient_proj(
            best_seq, n_final, overall_deadline, max_iters=max_iters_sub
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"[SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:                        # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
