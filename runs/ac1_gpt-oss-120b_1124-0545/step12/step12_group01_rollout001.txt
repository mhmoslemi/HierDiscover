"""
Propose a step‑function (non‑negative vector) that minimizing

    2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the full linear convolution (numpy.convolve).

Algorithm
---------
1. Gather a modest set of analytically‑motivated seeds (uniform,
   triangular, a few window functions, power‑law / beta families and, if
   present, the previous best solution).

2. Pick the best few seeds w.r.t. the cheap surrogate
   `2·n·max(conv)` (i.e. the objective for ℓ₁‑normalised sequences).

3. For each selected seed run a continuation L‑BFGS optimiser.
   * Variables are logits; the actual sequence is the soft‑max of the
     logits (guarantees non‑negativity and ℓ₁‑normalisation).
   * The smooth objective is a shifted log‑sum‑exp
     `max(conv) + log∑exp(β·(conv-max)) / β`.  The schedule
     `β = [1, 5, 20, 80, 200, 500, 1000]` gradually tightens the
     approximation to the true max.
   * The optimiser runs until the global deadline.

4. Starting from the best L‑BFGS result, perform an extensive
   sub‑gradient descent on the exact non‑smooth objective
   `2·n·max(conv)`.  The sub‑gradient is computed analytically
   (only the active convolution entry contributes).  After each step
   we project back onto the probability simplex.

5. Finally a cheap random‑local‑search (Gaussian perturbation + simplex
   projection) is performed for the remaining time – it is able to
   escape tiny local minima and often shaves a few‑hundredths off the
   surrogate bound.

6. The best sequence found is evaluated with the official
   `evaluate_sequence` function, printed and stored in the global
   variable `height_sequence_1`.

The whole pipeline respects the 1000 s budget (a 5 s safety margin)
and works with a single thread (numpy + SciPy use only one CPU by
default).  All helper functions are top‑level and contain no closures.
"""

import time
import math
import numpy as np

# SciPy is part of the execution environment (used for L‑BFGS)
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ onto the simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    (Implementation of the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán, 2013.)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised seq."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α (re‑normalised)."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection – modest but diverse
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.7, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    for sigma in (0.15, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    # power‑law family
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    # beta family (coarse grid)
    for a in (0.5, 1.0, 2.0):
        for b in (0.5, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet & exponential draws
    for conc in (0.1, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(3):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency curves
    for nf in (2, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixes of uniform and a low‑freq curve
    for mix_alpha in (0.2, 0.5, 0.8):
        curve = _lowfreq_curve(n, 3, rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous best (if present)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # safety normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimises a smoothed version of the max‑convolution objective using
    L‑BFGS‑B on the logits (log‑softmax parameters). Returns an ℓ₁‑
    normalised vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimisation module not available.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

    # initialise logits – small positive value added for stability
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            """Smooth max objective and its gradient w.r.t. logits."""
            a = _softmax(v_local)                     # probability simplex
            conv = _conv_fft(a, a)                     # full autocorrelation
            max_c = np.max(conv)
            # shifted log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - max_c))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            # weights proportional to exp(beta*conv)
            w = exp_beta / sum_exp

            # gradient wrt a: 2 * conv(rev_a, w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Jacobian of softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # deadline‑aware callback
        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            # Time limit – abort remaining β values
            break
        except Exception as exc:  # pragma: no cover
            print(f"[L‑BFGS] exception {exc} (beta={beta_val})")
            break

    # final simplex vector
    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000,
                         lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """Performs projected sub‑gradient descent on 2·n·max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        step = lr0 / math.sqrt(it)
        a = a - step * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:,d}  best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random local search (Gaussian perturbation + projection)
# ----------------------------------------------------------------------
def _random_local_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000,
                         epsilon: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: add a small Gaussian perturbation,
    re‑project onto the simplex and keep the move if it improves the
    surrogate bound.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    rng = np.random.default_rng(0xC0FFEE)
    while it < max_iters and time.time() < deadline:
        it += 1
        perturb = rng.normal(scale=epsilon, size=n)
        cand = a + perturb
        cand = np.maximum(cand, 0.0)          # keep non‑negative
        s = cand.sum()
        if s <= 0.0:
            continue
        cand = cand / s

        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand.copy()
            a = cand               # move to the better point
            # slowly shrink epsilon to allow finer adjustments
            epsilon *= 0.9995

        # occasional reset of epsilon to avoid stagnation
        if it % 50000 == 0:
            epsilon = max(epsilon, 1e-5)

    return best_seq, best_val
