def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT length)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:

    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:

        return _uniform(n)

    mu = (n - 1) / 2.0

    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:

    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few low‑frequency cosine terms, clip to ≥0, normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_constant(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly split the index range into a few constant blocks."""
    # choose 3‑10 blocks
    k = rng.integers(3, 11)
    cut_points = np.sort(rng.choice(n - 1, size=k - 1, replace=False) + 1)
    cuts = np.concatenate(([0], cut_points, [n]))
    vals = rng.exponential(scale=1.0, size=k)
    a = np.empty(n, dtype=np.float64)
    for i in range(k):
        a[cuts[i]:cuts[i + 1]] = vals[i]
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, then re‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# -------------------------------------------------------------
# Seed generation
# -------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_gaussian(n, 0.15))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_power(n, 0.5))
    seeds.append(_power(n, 2.0))
    seeds.append(_beta_shape(n, 0.5, 0.5))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_tukey_window(n, 0.5))

    # random Dirichlet – a broad range of concentrations
    for conc in (0.02, 0.1, 0.5, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency cosine curves
    for nf in (2, 4, 6):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # piecewise‑constant random shapes
    for _ in range(5):
        seeds.append(_piecewise_constant(n, rng))

    # mixtures with uniform (helps when a good shape is already known)
    if previous is not None:
        seeds.append(previous)                       # reuse the exact previous result
        for a in (0.2, 0.5, 0.8):
            seeds.append(_mix(_uniform(n), previous, a))

    # final normalisation guard (should already be normalised)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        ss = v.sum()
        if ss > 0:
            cleaned.append(v / ss)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Optimisation primitives
# -------------------------------------------------------------
def _smooth_adam(start_vec: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 300_000,
                 beta_start: float = 1.0,
                 beta_target: float = 4000.0) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑softmax parametrisation.
    The loss is a smoothed max of the convolution using a temperature β.
    """
    eps = 1e-30
    logit = np.log(start_vec + eps)          # logits → ℓ₁ simplex after softmax
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = beta_start
    beta_inc = (beta_target - beta_start) / max_iters

    best = start_vec.copy()
    best_val = _objective(start_vec)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(logit)                     # current candidate (ℓ₁‑norm = 1)

        # both surrogate and true objective are proportional → we can use true
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if it % 5000 == 0:
                print(f"[smooth‑Adam] it={it:7d} best={cur_val:.6f}")

        # ----- smooth max ----------
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w /= w.sum()                     # normalized weights → gradient of smooth max

        # gradient of Σ w_k * conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)      # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Jacobian of softmax
        dot = np.dot(grad_a, a)
        grad_logit = a * (grad_a - dot)

        # Adam step
        lr = 1.0 / np.sqrt(it)          # simple decay
        m = b1 * m + (1 - b1) * grad_logit
        v = b2 * v + (1 - b2) * (grad_logit ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logit -= step

        # temperature schedule
        if beta < beta_target:
            beta = min(beta + beta_inc, beta_target)

    return best, best_val


def _projected_subgrad(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a decreasing Polyak step‑size and Euclidean projection onto the simplex.
    """
    a = start_vec.copy()
    best = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        lr = 0.25 / np.sqrt(it)          # more aggressive than the earlier version
        a = a - lr * g
        a = _proj_simplex(a)

    return best, best_val


def _exp_gradient(start_vec: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Operates directly on the simplex via the log‑softmax representation.
    """
    a = start_vec.copy()
    best = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        eta = 0.6 / np.sqrt(it)          # a bit larger step‑size
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best, best_val


def _simulated_annealing(start_vec: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.08,
                         min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Aggressive SA: random subset reshuffling + pairwise transfers.
    The temperature decays geometrically.
    """
    rng = np.random.default_rng()
    cur = start_vec.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.99995, min_temp)

        # ----- propose move -----
        if rng.random() < 0.65:                 # reshuffle a random subset
            k = rng.integers(2, 14)            # size 2 … 13
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.03)   # move ≤3 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best, best_val


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalisation."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new /= new.sum()
    return new

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.

    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    overall_deadline = start_time + budget_s - 2.0   # tiny safety margin

    # -----------------------------------------------------------------
    # Stage 0 – generate a rich pool of seeds (coarse resolution)
    # -----------------------------------------------------------------
    n0 = 4000
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n0:
                prev_vec = prev_arr
        except Exception:
            prev_vec = None

    print(f"\n--- Seed generation (n={n0}) ---")
    seed_pool = _collect_seeds(n0, rng, prev_vec)
    print(f"Generated {len(seed_pool)} seeds.")

    # cheap surrogate ranking, keep the top few
    scored = [(v, _surrogate(v)) for v in seed_pool]
    scored.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [v for v, _ in scored[:top_k]]

    # initialise global best
    global_best = best_seeds[0].copy()
    global_best_val = _objective(global_best)
    print(f"[Init] Objective = {global_best_val:.6f}")

    # -----------------------------------------------------------------
    # Stage 1 – up‑sample to a finer grid (if we have time)
    # -----------------------------------------------------------------
    remaining = overall_deadline - time.time()
    n_fine = n0 * 2                 # default = 8000
    if remaining > 20.0:
        print(f"\n--- Upsampling to n={n_fine} ---")
        global_best = _upsample(global_best, n_fine)
        global_best_val = _objective(global_best)
        print(f"[Upsample] Objective = {global_best_val:.6f}")

    # -----------------------------------------------------------------
    # Stage 2 – multi‑restart smooth‑max Adam (fast exploration)
    # -----------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 120.0:
        extra_seeds = []
        # add a few more random Dirichlet draws at the fine resolution
        for conc in (0.05, 0.2, 0.7, 1.0):
            for _ in range(3):
                extra = _random_dirichlet(n_fine, conc, rng)
                extra_seeds.append(extra)

        # also upsample a couple of the coarse top‑seeds for extra diversity
        for v in best_seeds[:5]:
            extra_seeds.append(_upsample(v, n_fine))

        # run a short Adam from each of them
        max_iters_per = 80_000
        for idx, seed_vec in enumerate(extra_seeds, 1):
            if time.time() > overall_deadline:
                break
            seq, val = _smooth_adam(seed_vec, n_fine, overall_deadline,
                                    max_iters=max_iters_per,
                                    beta_start=1.0,
                                    beta_target=2000.0)
            if val < global_best_val:
                global_best = seq
                global_best_val = val
                print(f"[Multi‑restart] new best from seed #{idx} = {val:.6f}")

    # -----------------------------------------------------------------
    # Stage 3 – long smooth‑max Adam (main optimiser)
    # -----------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 300.0:
        max_iters = int(min(500_000, remaining * 1500))
        print(f"\n--- Long smooth‑max Adam (max_iters={max_iters}) ---")
        seq, val = _smooth_adam(global_best, n_fine, overall_deadline,
                                max_iters=max_iters,
                                beta_start=10.0,
                                beta_target=4000.0)
        if val < global_best_val:
            global_best = seq
            global_best_val = val
            print(f"[Long‑Adam] new global best = {val:.6f}")

    # -----------------------------------------------------------------
    # Stage 4 – exact sub‑gradient projection
    # -----------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 100.0:
        max_iters = int(min(1_200_000, remaining * 2500))
        print(f"\n--- Sub‑gradient projection (max_iters={max_iters}) ---")
        seq, val = _projected_subgrad(global_best, n_fine, overall_deadline,
                                      max_iters=max_iters)
        if val < global_best_val:
            global_best = seq
            global_best_val = val
            print(f"[SubProj] new global best = {val:.6f}")

    # -----------------------------------------------------------------
    # Stage 5 – exponentiated‑gradient fine‑tuning
    # -----------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 60.0:
        max_iters = int(min(600_000, remaining * 2000))
        print(f"\n--- Exponentiated‑gradient (max_iters={max_iters}) ---")
        seq, val = _exp_gradient(global_best, n_fine, overall_deadline,
                                 max_iters=max_iters)
        if val < global_best_val:
            global_best = seq
            global_best_val = val
            print(f"[ExpGrad] new global best = {val:.6f}")

    # -----------------------------------------------------------------
    # Stage 6 – simulated‑annealing polishing
    # -----------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 20.0:
        max_iters = int(min(800_000, remaining * 1800))
        print(f"\n--- Simulated annealing polishing (max_iters={max_iters}) ---")
        seq, val = _simulated_annealing(global_best, n_fine, overall_deadline,
                                        max_iters=max_iters)
        if val < global_best_val:
            global_best = seq
            global_best_val = val
            print(f"[SA] new global best = {val:.6f}")

    # -----------------------------------------------------------------
    # Final evaluation (just for logging – the caller will evaluate again)
    # -----------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
