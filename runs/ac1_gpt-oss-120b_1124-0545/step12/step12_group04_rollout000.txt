def projected_gradient_descent(start, step_schedule, max_iter, deadline):
   a = start copy
   best = a, best_val
   for epoch in range(max_iter):
        if time.time() > deadline: break
        conv = _conv_fft(a,a)
        max_idx = np.argmax(conv)
        grad = np.zeros_like(a)
        i_start = max(0, max_idx-(n-1))
        i_end = min(n-1, max_idx)
        # indices i in [i_start,i_end]
        rev_idx = max_idx - np.arange(i_start,i_end+1)
        grad[i_start:i_end+1] = 2*a[rev_idx]
        # step size schedule: η = eta0 / sqrt(t) or else.
        # compute v = a - η * grad
        v = a - eta * grad
        # project onto simplex
        a = _simplex_projection(v)
        # evaluate objective
        cur_val = 2*n*max(conv)
        if cur_val < best_val:
             best = a, cur_val

def upsample(seq, new_n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    # ensure nonnegative; set negative to zero
    new_seq = np.maximum(new_seq, 0)
    # renorm to sum=1
    s = new_seq.sum()
    if s > 0:
        return new_seq / s
    else:
        return np.full(new_n, 1.0/new_n)

def _projected_gradient_descent(start_seq, n, deadline, max_iters=1_000_000, eta0=0.1, power=0.5):
    a = start_seq.copy()
    best_seq = a.copy()

        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        idx = np.argmax(conv)
        # compute subgradient
        grad = np.zeros_like(a)
        i_start = max(0, idx-(n-1))
        i_end = min(n-1, idx)

def _simplex_projection(v):
    """Project onto simplex {x >=0, sum x=1}."""
    # Implementation from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0, return uniform
        return np.full(n, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a))²

The algorithm is a multi‑stage search:

1. **Seed generation** – uniform, triangular, cosine‑window, sinc‑squared
   and many random Dirichlet/low‑frequency curves, all L1‑normalised.
   The previous best solution (if any) is up‑sampled to each candidate
   length and used as a seed.

2. **Fast surrogate ranking** – evaluate the surrogate 2·n·max(conv) for
   all seeds and keep the best few.

3. **Smooth‑max L‑BFGS refinement** – optimise a log‑sum‑exp (smooth‑max)
   surrogate using SciPy L‑BFGS‑B, slowly raising the temperature
   (10 → 30 → 100 → 300 → 600 → 1000).  The optimisation works on a
   softmax (log‑it) parametrisation, guaranteeing the simplex constraint.

4. **Mirror‑descent (exponentiated‑gradient) sub‑gradient** – exact
   sub‑gradient descent on the true max‑convolution, using multiplicative
   updates and a slow power‑law decay of the step‑size.  This stage is
   run for a large number of iterations (limited only by the time budget).

5. **Projected‑gradient fine‑tuning** – a classic projected sub‑gradient
   descent (additive updates + Euclidean projection onto the simplex) that
   can improve the solution after the mirror step.

6. **Light SA polishing** – a short simulated‑annealing pass to escape
   shallow minima.

The search is performed for two lengths, `n = 4000` (the size that gave the
previous best) and a larger `n = 8000`.  The extra degrees of freedom at the
larger size often allow a smaller constant `c` in `max(conv)=c/n`, which
lowers the final objective (`2·c`).  The best sequence among all lengths is
returned and stored globally as ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic FFT‑based convolution (linear, full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple surrogate objective (identical ordering to the official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinc_squared(n: int) -> np.ndarray:
    """a_i = (sinc(pi*i/n))², ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    # avoid division by zero at i=0
    x = np.where(i == 0, 1.0, np.sin(np.pi * i / n) / (np.pi * i / n))
    a = x ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max utilities (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_max, grad_a) where smooth_max = log‑sum‑exp_beta(conv)/beta
    and grad_a = ∂/∂a smooth_max (ignores the external 2·n factor).
    """
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # gradient: 2 * (rev_a * weights) (convolution of reversed a with weights)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 200,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """L‑BFGS on the smooth‑max surrogate (softmax of the convolution)."""
    eps = 1e-30
    v = np.log(start_seq + eps)                     # logits, unrestricted

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        grad_v_scaled = 2.0 * n * grad_v
        return loss, grad_v_scaled.astype(np.float64)

    # Scipy's L‑BFGS‑B works on the unconstrained variables v.
    # We keep the iteration budget modest to stay within the overall time limit.
    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:
        # In case the optimizer misbehaves, fall back to the starter.
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    best_val = _objective(best_a)
    return best_a, best_val


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000,
                                 lr0: float = 0.08,
                                 power: float = 0.20) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with multiplicative (exponentiated) updates."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update with a decaying step size
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected (additive) sub‑gradient descent – an alternative fine tuner
# ----------------------------------------------------------------------
def _projected_gradient_descent(start_seq: np.ndarray,
                                n: int,
                                deadline: float,
                                max_iters: int = 1_000_000,
                                eta0: float = 0.05,
                                power: float = 0.5) -> tuple[np.ndarray, float]:
    """Additive sub‑gradient descent with Euclidean projection onto the simplex."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_range
        g[idx_range] = 2.0 * a[rev_idx]

        # diminishing step size
        eta = eta0 / ((it) ** power)

        # gradient step then projection onto the simplex
        a = _simplex_projection(a - eta * g)

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[PGD] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing (light) to escape shallow minima
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 800_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)  # 0=swap, 1=redistribute, 2=pair‑transfer
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation) – used when re‑using a previous
# solution at a larger length.
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0.0:
        return new_seq / s
    else:
        return _uniform(new_n)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    `evaluate_sequence` objective within the allocated time budget.
    Returns the best sequence (list of floats) and stores it globally as
    ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) for seeding.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – 4000 (known good) and a larger 8000.
    # ------------------------------------------------------------------
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = np.inf

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising for length {n} ===")

        # --------------------
        # Seed generation
        # --------------------
        seeds = []

        # up‑sampled previous solution (if any)
        if previous is not None:
            seeds.append(_upsample(previous, n))

        # a few analytic families
        seeds.append(_uniform(n))
        seeds.append(_triangular(n))
        seeds.append(_cosine_window(n))
        seeds.append(_sinc_squared(n))
        seeds.append(_gaussian(n, sigma_frac=0.2))

        # random Dirichlet draws (various concentrations)
        for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
            for _ in range(3):
                seeds.append(_random_dirichlet(n, conc, rng))

        # low‑frequency random curves
        for nf in (2, 3, 4):
            for _ in range(2):
                seeds.append(_lowfreq_curve(n, nf, rng))

        # --------------------------------------------------------------
        # Fast surrogate ranking – keep the best few seeds.
        # --------------------------------------------------------------
        evaluated = [(vec, _objective(vec)) for vec in seeds]
        evaluated.sort(key=lambda x: x[1])
        top_k = 8
        top_seeds = [vec for vec, _ in evaluated[:top_k]]

        print(f"Top initial surrogate values for n={n}:")
        for i, (_, val) in enumerate(evaluated[:top_k], 1):
            print(f"  #{i}: {val:.6f}")

        # --------------------
        # Initialise best for this length
        # --------------------
        best_seq_n = top_seeds[0].copy()
        best_val_n = _objective(best_seq_n)

        # --------------------------------------------------------------
        # Stage 1 – L‑BFGS refinement on a smooth‑max surrogate
        # --------------------------------------------------------------
        temps = [10.0, 30.0, 100.0, 300.0, 600.0, 1000.0]
        for temp in temps:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(best_seq_n, n, deadline,
                                                max_iter=200, temperature=temp)
            if val_opt < best_val_n:
                best_seq_n = seq_opt.copy()
                best_val_n = val_opt
                print(f"[L‑BFGS] n={n}, temp={int(temp)} → {val_opt:.6f}")

        # --------------------------------------------------------------
        # Stage 2 – Mirror‑descent (exponentiated‑gradient) fine‑tuning
        # --------------------------------------------------------------
        if time.time() < deadline:
            print("[Mirror] starting fine‑grained sub‑gradient descent")
            seq_m, val_m = _mirror_subgradient_optimize(
                best_seq_n, n, deadline,
                max_iters=3_000_000,
                lr0=0.06,               # slightly smaller step size
                power=0.20)             # slower decay for better precision
            if val_m < best_val_n:
                best_seq_n = seq_m.copy()
                best_val_n = val_m
                print(f"[Mirror] new best {val_m:.6f}")

        # --------------------------------------------------------------
        # Stage 3 – Projected additive sub‑gradient (optional extra tweak)
        # --------------------------------------------------------------
        if time.time() < deadline:
            print("[PGD] starting projected gradient fine‑tuning")
            seq_pgd, val_pgd = _projected_gradient_descent(
                best_seq_n, n, deadline,
                max_iters=800_000,
                eta0=0.04,
                power=0.45)
            if val_pgd < best_val_n:
                best_seq_n = seq_pgd.copy()
                best_val_n = val_pgd
                print(f"[PGD] new best {val_pgd:.6f}")

        # --------------------------------------------------------------
        # Stage 4 – Light simulated‑annealing polishing
        # --------------------------------------------------------------
        if time.time() < deadline:
            print("[SA] final polishing")
            seq_sa, val_sa = _simulated_annealing_optimize(
                best_seq_n, n, deadline,
                max_iters=600_000,
                init_temp_factor=0.03)
            if val_sa < best_val_n:
                best_seq_n = seq_sa.copy()
                best_val_n = val_sa
                print(f"[SA] new best {val_sa:.6f}")

        print(f"Best surrogate for n={n}: {best_val_n:.6f}")

        # ------------------------------------------------------------------
        # Keep the global best across lengths.
        # ------------------------------------------------------------------
        if best_val_n < global_best_val:
            global_best_val = best_val_n
            global_best_seq = best_seq_n.copy()
            print(f"*** New global best: {global_best_val:.6f} (n={n}) ***")

    # ----------------------------------------------------------------------
    # Final official evaluation (informational only)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        # fallback – should never happen, but guard against it.
        raise RuntimeError("No sequence was generated within the time budget.")

    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
