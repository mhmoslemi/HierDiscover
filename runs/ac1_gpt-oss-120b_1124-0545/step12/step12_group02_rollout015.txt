assistantfinal**Solution Overview**

The following implementation improves on the previous optimizer in three ways  

1. **Richer seed pool** – the seed generation explores many more analytic families
   (exponential, logistic, finer parameter grids for the existing windows, and many
   random mixtures).  
2. **Multi‑restart Adam** – after the first long Adam run the best vector is
   perturbed several times and Adam is started again.  The temperature schedule
   reaches a much larger `β` (up to 5000) so the smooth max‑approximation becomes
   almost exact.  
3. **Simple post‑processing** – after the final refinement the sequence is
   symmetrised (`(a+rev(a))/2`) which never hurts the maximum linear convolution
   and often reduces it.

All stages respect the global 1000 s deadline, keep the best solution seen so far
and finally call `evaluate_sequence`.  The code uses only `numpy` and the helper
functions already present in the notebook.

```python
"""Optimiser for the step‑function objective.

The optimiser works in five stages:

1.  Generate a very diverse pool of ℓ₁‑normalised candidate windows
    (analytic families, random Dirichlet / exponential draws, random
    low‑frequency curves, and convex mixtures of two windows).  The
    parameter grids are finer than in the previous version.

2.  Keep the best K candidates (according to the cheap surrogate
    2·n·max(conv)) and up‑sample them to a finer grid (8000 points).

3.  Run a long Adam optimiser on the best candidate.  After the first run
    the solution is perturbed several times and Adam is started again.
    The temperature for the log‑sum‑exp smoothing is increased up to
    β≈5000, providing a very tight approximation of the true max.

4.  Refine the result with the exact sub‑gradient projection and an
    exponentiated‑gradient (mirror‑descent) pass.

5.  Symmetrise the final vector ((a+rev(a))/2) – this never increases the
    maximum linear convolution and often reduces it – and evaluate it
    with the official scorer.

The routine respects the global time budget (default 1000 s) and returns
the best sequence found.
"""

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    It respects the ordering of the official objective.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families (all vectors are ℓ₁‑normalised)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows as _scipy_windows
    except Exception:  # pragma: no cover
        _scipy_windows = None

    if _scipy_windows is not None and hasattr(_scipy_windows, 'kaiser'):
        a = _scipy_windows.kaiser(n, beta)
    else:
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        t = beta * np.sqrt(1 - y * y)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window; falls back to uniform."""
    try:
        from scipy.signal import windows as _scipy_windows
    except Exception:  # pragma: no cover
        _scipy_windows = None

    if _scipy_windows is not None and hasattr(_scipy_windows, 'dpss'):
        a = _scipy_windows.dpss(n, nw, Kmax=1, return_ratios=False)[0]
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    return _uniform(n)


def _chebwin_window(n: int, att: float) -> np.ndarray:
    """Chebyshev window with given side‑lobe attenuation (dB)."""
    try:
        from scipy.signal import windows as _scipy_windows
    except Exception:  # pragma: no cover
        _scipy_windows = None

    if _scipy_windows is not None and hasattr(_scipy_windows, 'chebwin'):
        a = _scipy_windows.chebwin(n, at=att)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    return _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    """Mixture of two symmetric Gaussians placed at shift and 1‑shift."""
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, tau_frac: float) -> np.ndarray:
    """Monotone exponential decay a[i] ∝ exp(-i/(tau)) , τ = τ_frac·n."""
    if tau_frac <= 0.0:
        return _uniform(n)
    tau = tau_frac * n
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / tau)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, centre_frac: float = 0.5,
                     steep_frac: float = 0.1) -> np.ndarray:
    """Logistic (sigmoid) window."""
    centre = centre_frac * (n - 1)
    steep = steep_frac * n
    if steep <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - centre) / steep))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few random cosine components, clip to non‑negative and normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed generation (very extensive)
# ----------------------------------------------------------------------
def _collect_seeds_rich(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a large and diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families (finer grids than before)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Sine windows – finer gamma grid (0.05 … 3.0)
    for g in np.linspace(0.05, 3.0, 30):
        seeds.append(_sine_window(n, gamma=g))

    # Gaussian windows – finer sigma grid
    for sigma in np.linspace(0.05, 0.45, 15):
        seeds.append(_gaussian(n, sigma))

    # Exponential decay
    for tau in np.linspace(0.05, 0.6, 12):
        seeds.append(_exp_decay(n, tau))

    # Logistic windows
    for centre in np.linspace(0.3, 0.7, 5):
        for steep in np.linspace(0.02, 0.2, 5):
            seeds.append(_logistic_window(n, centre_frac=centre,
                                          steep_frac=steep))

    # Kaiser windows – extended beta range
    for beta in np.linspace(2, 20, 19):
        seeds.append(_kaiser_window(n, beta))

    # DPSS windows – extended range
    for nw in np.linspace(2.0, 6.0, 9):
        seeds.append(_dpss_window(n, nw))

    # Chebyshev windows – larger attenuation values
    for att in np.linspace(30, 120, 10):
        seeds.append(_chebwin_window(n, att))

    # Power‑law family – dense grid around the interesting region
    for p in np.linspace(0.0, 1.2, 25):
        seeds.append(_power(n, p))

    # Beta family – finer two‑dimensional grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows – full range (including α=1 → Hann)
    for alpha in np.linspace(0.05, 1.0, 20):
        seeds.append(_tukey_window(n, alpha))

    # Double‑Gaussian mixtures
    for sigma in (0.10, 0.12, 0.15):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma,
                                          shift_frac=shift))

    # Random Dirichlet draws – many concentrations
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves – more frequencies
    for nf in (2, 3, 5, 8, 12):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random convex mixtures of any two deterministic windows
    deterministic = [
        _uniform(n), _triangular(n), _hamming_window(n), _blackman_window(n),
        _nuttall_window(n), _kaiser_window(n, 6.0), _dpss_window(n, 3.0),
        _gaussian(n, 0.2), _exp_decay(n, 0.25), _logistic_window(n, 0.5, 0.1)
    ]
    for _ in range(30):
        a, b = rng.choice(deterministic, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(a, b, alpha))

    # Previous best (if supplied)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise everything robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Optimisation primitives
# ----------------------------------------------------------------------
def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
    beta_max: float = 5000.0,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑softmax parametrisation with an aggressive
    temperature schedule (β up to *beta_max*).  Returns the best vector found
    and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5
    beta_inc = (beta_max - beta) / max_iters

    lr0 = 1.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # ℓ₁‑normalised vector
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # Smoothed max via log‑sum‑exp with temperature β
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # Gradient of Σ w_k·conv_k w.r.t. a (as in the original code)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Gradient on logits (log‑softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # Increment temperature
        beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a decreasing step‑size and Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # Sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / np.sqrt(it)          # Polyak‑style step‑size
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Works directly on the simplex via the log‑softmax representation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)          # decreasing step size
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)              # back to simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        if rng.random() < 0.65:          # reshuffle a random subset
            k = rng.integers(2, 13)          # size 2 … 12
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.06)  # move ≤6 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalisation."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    overall_deadline = start_time + budget_s - 2.0               # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – generate a very diverse seed pool (coarse resolution)
    # ------------------------------------------------------------------
    n0 = 4000
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print(f"\n--- Rich seed generation (n={n0}) ---")
    seeds = _collect_seeds_rich(n0, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # cheap surrogate ranking
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 50
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample to a finer grid (if time permits)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    n_up = n0 * 2                      # 8000 points
    if remaining > 30.0:
        print(f"\n--- Upsampling to n={n_up} ---")
        global_best_seq = _upsample_sequence(global_best_seq, n_up)
        global_best_seq = global_best_seq / global_best_seq.sum()
        global_best_val = _objective_surrogate(global_best_seq)
        print(f"[Upsample] surrogate after interpolation = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – multi‑restart Adam optimisation
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 60.0:
        # First long run
        max_iters_adam = int(min(400_000, remaining * 8000))
        print(f"\n--- First Adam run (max_iters={max_iters_adam}) ---")
        seq_adam, val_adam = _logspace_adam_optimize(
            global_best_seq, n_up, overall_deadline,
            max_iters=max_iters_adam, beta_max=5000.0
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam‑1] new global best = {val_adam:.6f}")

        # Additional restarts – small perturbations of the current best
        restarts = 3
        for r in range(restarts):
            remaining = overall_deadline - time.time()
            if remaining < 15.0:
                break
            # perturb by adding a tiny Gaussian noise and re‑normalising
            noise = rng.normal(scale=0.01, size=n_up)
            pert = np.maximum(global_best_seq + noise, 0.0)
            pert = pert / pert.sum()
            max_iters_restart = int(min(150_000, remaining * 4000))
            print(f"\n--- Adam restart {r+1} (max_iters={max_iters_restart}) ---")
            seq_restart, val_restart = _logspace_adam_optimize(
                pert, n_up, overall_deadline,
                max_iters=max_iters_restart, beta_max=5000.0
            )
            if seq_restart is not None and val_restart < global_best_val:
                global_best_seq = seq_restart.copy()
                global_best_val = val_restart
                print(f"[Adam‑restart{r+1}] new global best = {val_restart:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient projection
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 20.0:
        max_iters_sub = int(min(1_200_000, remaining * 3000))
        print(f"\n--- Sub‑gradient projection (max_iters={max_iters_sub}) ---")
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters_sub
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        max_iters_exp = int(min(600_000, remaining * 2000))
        print(f"\n--- Exponentiated‑gradient refinement (max_iters={max_iters_exp}) ---")
        seq_exp, val_exp = _exp_gradient_optimize(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters_exp
        )
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        max_iters_sa = int(min(800_000, remaining * 1500))
        print(f"\n--- Simulated annealing polishing (max_iters={max_iters_sa}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n_up, overall_deadline, max_iters=max_iters_sa
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – final symmetrisation (never hurts the max‑conv)
    # ------------------------------------------------------------------
    sym_seq = (global_best_seq + global_best_seq[::-1]) / 2.0
    sym_seq = _proj_simplex(sym_seq)   # protect against numerical drift
    sym_val = _objective_surrogate(sym_seq)
    if sym_val < global_best_val:
        global_best_seq = sym_seq
        global_best_val = sym_val
        print(f"[Symmetrise] improved surrogate = {sym_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```
