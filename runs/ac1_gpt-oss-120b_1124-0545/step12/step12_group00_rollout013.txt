    # start: np.ndarray length n sum=1
    eps = 1e-12
    w = np.log(start + eps)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    lr0 = lr
    best_x = start.copy()
    best_val = _objective_surrogate(start)
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        # compute x via softmax
        max_w = np.max(w)
        exp_w = np.exp(w - max_w)  # for numerical stability
        x = exp_w / np.sum(exp_w)
        # conv and find active shift
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))
        # compute gradient g for x
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]
        # compute gradient w.r.t logits: d f / d w = x * (g - np.dot(g, x))
        dot = np.dot(g, x)
        grad_w = x * (g - dot)

        v = beta2 * v + (1 - beta2) * (grad_w ** 2)

        if val < best_val:

            best_x = x.copy()

"""
Propose a low‑autocorrelation (low max‑convolution) step function.

Algorithm
---------
1. **Rich seed library** – analytic windows (uniform, triangular,
   sine, Gaussian, exponential, power‑law, Beta, Tukey) together with
   random Dirichlet / exponential draws and low‑frequency Fourier
   series.  All seeds are ℓ₁‑normalised (∑=1).  The ten best seeds w.r.t.
   the surrogate ``2·n·max(conv)`` are kept.

2. **Adam optimiser on the simplex** – we maintain an unconstrained
   log‑weight vector ``w`` and obtain the probabilities via a soft‑max.
   The sub‑gradient of the max‑convolution entry is used; Adam (first‑
   and second‑moment adaptation) gives fast, low‑variance descent.
   A short Adam run is performed on each top seed, then a long run on
   the current global best.

3. **Simulated‑annealing polishing** – after Adam, tiny random mass
   transfers and subset‑reshuffles are applied while accepting all
   improvements (or occasional uphill moves) to escape remaining local
   minima.

4. **Final check** – the best vector is passed to the official
   ``evaluate_sequence`` function, stored globally as ``height_sequence_1``,
   and returned as a plain list of floats.

All stages respect a user‑specified time budget (default 1000 s).  The
implementation uses only NumPy (FFT based convolutions) and runs on a
single process (NumPy may use up to two threads internally).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq, seq)),
    assuming the sequence is ℓ₁‑normalised (∑seq = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p  (i starts at 0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    """Monotonically decreasing exponential, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amp/phase, ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mix (1‑α)·a + α·b  and renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray, n: int, deadline: float,
                   max_iters: int = 250_000, lr0: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the probability simplex.
    start : ℓ₁‑normalised (sum=1) starting point.
    Returns best vector seen and its surrogate value.
    """
    eps = 1e-12
    w = np.log(start + eps)               # unconstrained logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_x = None
    best_val = np.inf

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to obtain a point on the simplex
        max_w = np.max(w)
        exp_w = np.exp(w - max_w)          # stability
        x = exp_w / np.sum(exp_w)

        # find active shift (max convolution entry)
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(g, x)
        grad_w = x * (g - dot)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 * np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it)

        w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # evaluate surrogate
        val = _objective_surrogate(x)
        if val < best_val:
            best_val = val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

    # fallback in the unlikely case no improvement was recorded
    if best_x is None:
        best_x = start.copy()
        best_val = _objective_surrogate(best_x)

    return best_x, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (identical to the one from the reference)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # geometric cooling
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            # reshuffle a random subset (size 3‑12)
            k = rng.integers(3, 13)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)  # ≤ 7 % of source mass
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed collection (more exhaustive than the reference)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.20, 0.30):
        seeds.append(_gaussian(n, sigma))
    for rate in np.linspace(0.985, 0.9999, 8):
        seeds.append(_exponential_decay(n, rate))
    # Power‑law sweep (dense)
    for p in np.arange(0.1, 1.6, 0.1):
        seeds.append(_power(n, float(p)))
    # Beta family (moderately dense grid)
    for a in np.arange(0.2, 4.2, 0.4):
        for b in np.arange(0.2, 4.2, 0.4):
            seeds.append(_beta_shape(n, float(a), float(b)))
    # Tukey windows
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves (few harmonics)
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        curve = _lowfreq_curve(n, 4, rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution, if available
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise everything (safety)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        clean.append(v / s if s > 0 else _uniform(n))
    return clean

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # small safety margin for bookkeeping / prints
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length – a power of two gives the fastest FFT.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    # Score seeds using the surrogate and keep the most promising
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – quick Adam runs on a few best seeds.
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick Adam refinement on top seeds ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):   # a handful of quick runs
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # allocate iterations proportionally to remaining time
            max_it = int(min(80_000, remaining * 1600))
            cand, val = _adam_optimize(seed_vec, n, deadline,
                                       max_iters=max_it, lr0=0.25)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick Adam] seed #{idx+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long‑run Adam polishing on the current best.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long‑run Adam polishing ---")
        remaining = deadline - time.time()
        max_it = int(min(1_200_000, remaining * 1500))
        best_seq, best_val = _adam_optimize(best_seq, n, deadline,
                                            max_iters=max_it, lr0=0.18)
        print(f"[Long Adam] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing fine‑tuning.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_it = int(min(800_000, remaining * 2000))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_it)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible reuse in subsequent calls.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
