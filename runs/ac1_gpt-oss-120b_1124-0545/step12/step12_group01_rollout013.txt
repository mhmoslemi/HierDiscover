"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution (numpy.convolve).
The algorithm is a multi‑stage stochastic / gradient search:

1. **Seed generation** – a large, diverse set of ℓ₁‑normalised vectors
   (analytic windows, many random Dirichlet draws with a wide range of
   concentration parameters, low‑frequency cosine combinations, plus the
   previously found solution if any).

2. **Cheap surrogate ranking** – rank the seeds by the cheap surrogate
   `2·n·max(conv)`.  Keep the best few for the expensive optimisation.

3. **Continuation L‑BFGS** – optimise a smooth approximation of the max
   using a soft‑max (log‑sum‑exp) with a schedule of increasing β.
   We work in the log‑softmax (simplex) parametrisation and project
   back onto the probability simplex after each update.

4. **Local polishing** – a long sub‑gradient descent on the *exact* max‑
   convolution objective, followed by an exponentiated‑gradient (mirror‑
   descent) pass.

5. **Random restarts** – perturb the current best logits with small
   Gaussian noise and re‑run a short L‑BFGS from the perturbed point.

6. **Upsampling (optional)** – if enough time remains, linearly up‑sample
   the best vector to a finer grid (double the length) and repeat steps
   3‑4.

All stages respect an overall time‑budget (default 1000 s).  The final
vector is stored in the global name ``height_sequence_1`` and returned
as a plain list of floats.
"""

# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (more efficient than np.convolve)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns probabilities that sum to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Linear combination of a few low‑frequency cosine waves, made non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `conc` (small conc ⇒ spiky)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture of two ℓ₁‑normalised vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (used for cheap ranking)
# ----------------------------------------------------------------------
def _surrogate_objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    The variables are the logits of a soft‑max (hence always stay on the
    probability simplex).  Returns a probability vector (ℓ₁‑normalised).
    """
    if _opt is None:          # pragma: no cover
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0, 200.0, 500.0]

    # initialise logits from the starting probability vector
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        # ----- objective & gradient for a fixed beta -----------------
        def fun(v_local):
            a = _softmax(v_local)                     # simplex point
            conv = _conv_fft(a, a)                     # full autocorrelation
            shift = np.max(conv)                       # location of the true max
            # soft‑max over the convolution values
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # derivative of the smoothed max w.r.t. a
            w = exp_beta / sum_exp                     # weights (softmax)
            # grad_a = 2 * conv_rev(a, w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # ----- timeout handling ------------------------------------
        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            # out of time – just stop the whole schedule
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] unexpected exception {e} (beta={beta})")
            break

    best = _softmax(v)
    return best


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry (zero‑shift is always active)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        # step size ∝ 1/√t
        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    # return the best (surrogate) value we have seen
    return best_seq, _surrogate_objective(best_seq)


def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate_objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a finer grid, then re‑normalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq

    seeds.append(_triangular(n))

    seeds.append(_gaussian(n, 0.20))

    # low‑frequency random curves
    for nf in (2, 3, 5):
        for _ in range(3):

    # random Dirichlet with a wide range of concentration values
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures of uniform + low‑freq curves
    for a in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, a))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # safety normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap surrogate evaluation (fast: no denominator)
    evaluated = [(vec, _surrogate_objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("Top surrogate values after seeding:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate_objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all seeds.")
            break

        # allocate a fraction of the remaining budget to this seed
        time_left = deadline - time.time()
        max_iters_per_beta = max(200,
                                 int(min(3500,
                                         time_left * 0.07 * 1e5)))

        print(f"\n--- L‑BFGS run on seed #{idx+1}/{len(top_seeds)} "
              f"(max_iter/beta={max_iters_per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=max_iters_per_beta)

        val_opt = _surrogate_objective(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best = {global_best_val:.6f}")

        # ----------------------------------------------------------------
        # short sub‑gradient polishing of the current global best
        # ----------------------------------------------------------------
        if time.time() < deadline:
            sub_budget = int(min(250_000,
                                 (deadline - time.time()) * 0.06 * 1e5))
            seq_sub, val_sub = _subgradient_proj(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=sub_budget)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubProj] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient refinement (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_budget = int(min(300_000,
                             (deadline - time.time()) * 0.06 * 1e5))
        if exp_budget > 0:
            print("\n--- Exponentiated‑gradient refinement ---")
            seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=exp_budget)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – random restarts (perturb & re‑optimise)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random restarts ---")
        for r in range(3):
            if time.time() > deadline:
                break
            # perturb current logits
            logits = np.log(global_best_seq + 1e-30)
            noise = rng.normal(scale=0.05, size=logits.shape)
            perturbed_logits = logits + noise
            perturbed_seq = _softmax(perturbed_logits)

            # short L‑BFGS from perturbed start
            seq_restart = _lbfgs_smooth_optimize(perturbed_seq,
                                                 n,
                                                 deadline,
                                                 max_iter_per_beta=800)
            val_restart = _surrogate_objective(seq_restart)
            print(f"[Restart {r+1}] surrogate = {val_restart:.6f}")

            if val_restart < global_best_val:
                global_best_seq = seq_restart.copy()
                global_best_val = val_restart
                print(f"[Restart Update] new best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – up‑sampling to a finer grid (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 40.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample_sequence(global_best_seq, new_n)

            max_iters_up = int(min(3000, (deadline - time.time()) * 0.07 * 1e5))
            up_opt = _lbfgs_smooth_optimize(up_seq,
                                            new_n,
                                            deadline,
                                            max_iter_per_beta=max_iters_up)
            up_val = _surrogate_objective(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] after upsampling = {global_best_val:.6f}")

            # final polishing on up‑sampled grid
            if time.time() < deadline:
                sub2_budget = int(min(150_000,
                                      (deadline - time.time()) * 0.05 * 1e5))
                seq_sub2, val_sub2 = _subgradient_proj(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=sub2_budget)
                if val_sub2 < global_best_val:
                    global_best_seq = seq_sub2.copy()
                    global_best_val = val_sub2
                    print(f"[Final‑SubProj] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
