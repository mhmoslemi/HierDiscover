def _lp_iterative_refine(seq, deadline, max_iters=30, tau_candidates=None):
    if tau_candidates is None:
        tau_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]
    f = seq.copy()
    n = len(f)
    best_val = _objective(f)
    best_seq = f.copy()
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        M = _conv_fft(f, f).max()
        # build conv matrix
        A = _build_conv_matrix(f)
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
        if not res.success:
            print(f"[LP] iteration {it} failed, stopping.")
            break
        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:
            print(f"[LP] iteration {it} no mass increase, stopping.")
            break
        # try different taus
        best_local_seq = None
        best_local_val = float('inf')
        for tau in tau_candidates:
            cand = (1.0 - tau) * f + tau * g
            cand = cand / cand.sum()
            val = _objective(cand)

        if best_local_val < best_val:

            best_seq = f.copy()

"""
Advanced optimiser for the ``evaluate_sequence`` challenge.

Algorithm
---------
1. Initialise a high‑quality seed.  If a previous solution
   (global ``height_sequence_1``) exists it is used, otherwise a
   diverse collection of analytic families and random Dirichlet draws
   is generated and the top few (by a fast surrogate objective) are
   kept.

2. Adam on a log‑sum‑exp surrogate – a smooth approximation of the
   max‑convolution.  The temperature is ramped up to a very large
   value (β≈2000) so that the surrogate becomes virtually the true
   max.  This stage gives a strong local improvement.

3. Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
   A large number of cheap sub‑gradient steps rapidly flattens the
   autocorrelation.

4. Intensive linear‑programming polishing.  For the current shape
   ``f`` we solve

        maximise   Σ g_i
        subject to conv(f, g) ≤ M   (M = max(conv(f, f)))
                  g_i ≥ 0

   The LP is solved with HiGHS.  The solution ``g`` is mixed with ``f``
   using several candidate mixing factors τ; the mixture with the
   smallest surrogate value is kept.  The process is repeated until
   no further L1‑mass increase is possible (≈20‑30 iterations).

5. A short post‑LP mirror step refines the result.

6. Simulated‑annealing final polishing – cheap stochastic moves
   (swap, redistribution, pair‑transfer) with geometric cooling.

All stages respect the global 1000 s budget.  The best sequence is
stored in the global name ``height_sequence_1`` and returned as a
plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    The sequence is assumed to be L1‑normalised (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Softmax helper (used by Adam)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


# ----------------------------------------------------------------------
# Random / low‑frequency helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # sine window sweep
    for gamma in np.linspace(0.2, 5.0, 15):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian sweep
    for sigma in np.linspace(0.06, 0.50, 12):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Power‑law sweep
    for p in np.linspace(0.0, 6.0, 13):
        seeds.append(_power(n, p))

    # Beta family grid
    alphas = np.linspace(0.2, 3.5, 8)
    betas = np.linspace(0.2, 3.5, 8)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey sweep
    for alpha in np.linspace(0.05, 0.95, 12):
        seeds.append(_tukey_window(n, alpha))

    # Exponential decay sweep
    for lam in np.linspace(0.0002, 0.02, 12):
        seeds.append(_exponential_decay(n, lam))

    # Dirichlet draws – wide range of concentrations
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

            # alternating high‑low order
            a = np.sort(raw)
            lo, hi = 0, len(a) - 1
            alt = np.empty_like(a)
            pos = 0
            while lo <= hi:
                alt[pos] = a[hi] if pos % 2 == 0 else a[lo]
                hi -= pos % 2 == 0
                lo += pos % 2 == 1
                pos += 1
            seeds.append(alt)

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform and low‑frequency curves
    for mix_alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution as an extra seed
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Adam optimiser on a smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    eps = 1e-30
    v = np.log(start_vec + eps)                     # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta = 0.5            # start temperature
    beta_inc = 0.5
    beta_max = 2000.0     # higher temperature for a very tight max‑approximation

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # simplex point, sum == 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 50_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # --- smoothed max via log‑sum‑exp ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror‑descent subgradient optimiser (exact max)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.15
    power = 0.30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative (exponentiated‑gradient) update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming polishing – repeated
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """Build sparse matrix A such that (A @ g) = conv(f, g)."""
    n = len(f)
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for k in range(n):
        start = k
        end = k + n
        rows[idx:idx + n] = np.arange(start, end, dtype=np.int32)
        cols[idx:idx + n] = k
        data[idx:idx + n] = f
        idx += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_iterative_refine(seq: np.ndarray, deadline: float,
                         max_iters: int = 25,
                         tau_candidates: list[float] | None = None) -> tuple[np.ndarray, float]:
    """
    Repeated LP polishing.
    Each iteration solves:
        maximise   Σ g_i
        subject to conv(f, g) ≤ M   (M = max conv(f, f))
                  g_i ≥ 0
    The solution g is mixed with the current f using several τ values,
    the best mixture (by the surrogate objective) is kept.
    """
    if tau_candidates is None:
        tau_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]

    f = seq.copy()
    n = len(f)
    best_val = _objective(f)
    best_seq = f.copy()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()
        A = _build_conv_matrix(f)            # (2n‑1) × n
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)    # maximise Σ g_i

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if not res.success:
            print(f"[LP] iteration {it} failed – stopping LP stage.")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:
            print(f"[LP] iteration {it} – no L1‑mass increase, terminating.")
            break

        # try several mixing factors τ and keep the best
        best_local_seq = None
        best_local_val = float('inf')
        for tau in tau_candidates:
            cand = (1.0 - tau) * f + tau * g
            cand = cand / cand.sum()
            val = _objective(cand)
            if val < best_local_val:
                best_local_val = val
                best_local_seq = cand

        if best_local_seq is None:
            break

        f = best_local_seq
        if best_local_val < best_val - 1e-12:
            best_val = best_local_val
            best_seq = f.copy()
            print(f"[LP] iteration {it} – improved surrogate = {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing (lightweight)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 1_200_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:              # redistribute on a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.ones(k)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                             # pair‑wise mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective. Returns the best sequence as a list
    and registers it globally as ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin for the final official evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – a good trade‑off between expressiveness and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution, if any
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size != n:
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.pad(previous, (0, n - previous.size), constant_values=0.0)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Initialise the current best sequence
    # ------------------------------------------------------------------
    if previous is not None and previous.sum() > 0.0:
        best_seq = previous / previous.sum()
        print(f"Starting from previous solution (surrogate={_objective(best_seq):.6f})")
    else:
        print("\n--- Seed generation ---")
        seeds = _collect_seeds(n, rng, None)
        print(f"Generated {len(seeds)} raw seeds.")
        evaluated = [(vec, _objective(vec)) for vec in seeds]
        evaluated.sort(key=lambda x: x[1])
        best_seq = evaluated[0][0].copy()
        print(f"Initial surrogate = {_objective(best_seq):.6f}")

    best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # Stage 1 – Adam refinement (smooth surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam refinement ---")
        remaining = deadline - time.time()
        # allocate roughly 10 % of the remaining budget
        max_iters = int(min(400_000, remaining * 60))
        seq_adam, val_adam = _adam_optimize(best_seq, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam
            best_val = val_adam
            print(f"[Adam] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent sub‑gradient optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent optimisation ---")
        remaining = deadline - time.time()
        max_iters = int(min(1_000_000, remaining * 200))
        seq_mirror, val_mirror = _mirror_subgradient_optimize(best_seq, n, deadline, max_iters=max_iters)
        if val_mirror < best_val:
            best_seq = seq_mirror
            best_val = val_mirror
            print(f"[Mirror] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Intensive LP polishing (multiple iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Intensive LP polishing ---")
        seq_lp, val_lp = _lp_iterative_refine(best_seq, deadline, max_iters=25)
        if val_lp < best_val:
            best_seq = seq_lp
            best_val = val_lp
            print(f"[LP] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Post‑LP mirror refinement (short run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Post‑LP mirror refinement ---")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 150))
        seq_mirror2, val_mirror2 = _mirror_subgradient_optimize(best_seq, n, deadline, max_iters=max_iters)
        if val_mirror2 < best_val:
            best_seq = seq_mirror2
            best_val = val_mirror2
            print(f"[Mirror‑2] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Simulated‑annealing final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing final polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            best_seq, n, deadline,
            max_iters=800_000,
            init_temp_factor=0.05,
            min_temp=1e-12
        )
        if val_sa < best_val:
            best_seq = seq_sa
            best_val = val_sa
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (informational)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
