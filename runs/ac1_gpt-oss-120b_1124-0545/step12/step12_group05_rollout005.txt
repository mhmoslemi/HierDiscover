"""
Propose a step‑function (non‑negative vector) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

Our strategy differs from the reference in several ways:

1. **Richer seed pool** – many Dirichlet draws with a wide range of
   concentration parameters, plus the classic analytic families.

2. **Smooth‑max Adam optimisation** – we optimise the *log‑sum‑exp*
   approximation of the maximum convolution peak.  The temperature
   (β) is increased multiplicatively, giving a gentle transition from
   a smooth surrogate to the true max.

3. **Two deterministic re‑orderings** – “alternating” (large, small,
   …) and “interleaved‑ends” (largest at the left, second‑largest at the
   right, …).  Both are evaluated and the better one seeds the next
   phase.

4. **Heavy permutation annealing** – a simulated‑annealing on the
   ordering of the entries with three move types (swap, segment‑reverse,
   block‑move).  This explores the combinatorial space far more
   aggressively than a simple hill‑climber.

5. **Final polishing** – a short mirror‑subgradient (exponentiated‑gradient)
   run to clean up any remaining peaks.

All steps respect a global 1000 s time budget and make heavy use of
NumPy’s FFT‑based convolution for speed.  The best sequence found is
stored in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, … (then renormalise)."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


def _interleave_ends(vec: np.ndarray) -> np.ndarray:
    """
    Place the sorted values alternately at the leftmost and rightmost
    still‑free positions (largest→left, 2nd‑largest→right, …).
    """
    n = len(vec)
    vals = np.sort(vec)[::-1]            # descending
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for i, v in enumerate(vals):
        if (i & 1) == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / n)


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.07, offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed generation – deterministic + many random Dirichlet draws
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))

    # sine windows on a grid
    for gamma in np.linspace(0.5, 6.0, 10):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussian on a sigma grid
    for sigma in np.linspace(0.07, 0.30, 8):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # beta family – broader domain
    for _ in range(30):
        alpha = 10 ** rng.uniform(-3, 0.8)
        beta = 10 ** rng.uniform(-3, 0.8)
        b = _beta_shape(n, alpha, beta)
        seeds.append(b)
        seeds.append(_alternate_order(b))

    # exponential family
    for rate in np.logspace(-5, -1, 12):
        e = _exponential(n, rate)
        seeds.append(e)
        seeds.append(_alternate_order(e))

    # logarithmic family
    for eps in np.logspace(-5, -1, 10):
        l = _logarithmic(n, eps)
        seeds.append(l)
        seeds.append(_alternate_order(l))

    # Dirichlet draws – very diverse
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(20):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)
            seeds.append(_alternate_order(raw))

    # Mixtures with low‑frequency cosine components
    for nf in (2, 3):
        for _ in range(8):
            t = np.arange(n, dtype=np.float64)
            curve = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                curve += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            curve = np.maximum(curve, 0.0)
            curve /= curve.sum()
            mixed = 0.6 * _uniform(n) + 0.4 * curve
            mixed /= mixed.sum()
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            norm = avg / avg.sum()
            seeds.append(norm)
            seeds.append(_alternate_order(norm))

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_smooth_opt(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on the *log‑softmax* parameters, minimising a log‑sum‑exp
    surrogate for ``max(conv)``.  Returns the best feasible vector
    (already ℓ₁‑normalised) and its *exact* max‑convolution value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits → simplex via softmax
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    # temperature schedule – multiplicative growth
    beta = 1.0
    beta_mul = 1.025
    beta_max = 250.0

    best_seq = None
    best_val = float("inf")

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                     # stays on simplex, sum=1

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)

        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best_max={cur_max:.6e}")

        # ---- smoothed max -------------------------------------------------
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # degenerate β – increase it a little and continue
            beta = min(beta * beta_mul, beta_max)
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat_corr = v_hat / (1 - beta2 ** (it + 1))
        lr = lr0 / np.sqrt(it + 1)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

    return best_seq, best_val


def _mirror_subgradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent directly on the
    true ``max(conv)``.  Very cheap per iteration (just one FFT).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.30

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6e}")

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / np.sqrt(it + 1)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional aggressive re‑ordering
        if it % 60_000 == 0:
            alt = _alternate_order(a)
            val_alt = _objective(alt)
            if val_alt < cur_val:
                a = alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {val_alt:.6e}")

    return best_seq, best_val


def _perm_anneal(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_moves: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the *ordering* of the entries.
    Three move types:
        0 – swap two positions
        1 – reverse a short segment
        2 – cut a block and re‑insert elsewhere
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)

    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature ≈ 2 % of the current objective
    T = max(0.02 * cur_val, 1e-12)
    T_min = 1e-12
    cooling = 0.99993

    it = 0
    while it < max_moves and time.time() < deadline:
        it += 1
        T = max(T * cooling, T_min)

        move_type = rng.integers(0, 3)
        if move_type == 0:      # swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:    # reverse a short segment
            i, j = sorted(rng.choice(n, size=2, replace=False))
            if j - i < 2:
                continue
            new = cur.copy()
            new[i : j + 1] = new[i : j + 1][::-1]
        else:                   # block cut‑&‑paste
            seg_len = rng.integers(2, min(12, n // 4))
            i = rng.integers(0, n - seg_len + 1)
            block = cur[i : i + seg_len].copy()
            rest = np.concatenate([cur[:i], cur[i + seg_len :]])
            # new insertion point in the *rest*
            j = rng.integers(0, rest.size + 1)
            new = np.concatenate([rest[:j], block, rest[j:]])
            # ensure length stays n (it does)
        new_val = _objective(new)
        delta = new_val - cur_val
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 25_000 == 0:
                    print(f"[Perm‑SA] it={it:7d} best={best_val:.6e}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **_kwargs):
    """
    Search for a non‑negative step‑function minimising the official
    ``evaluate_sequence`` objective within ``budget_s`` seconds.
    The best sequence found is returned (list of floats) and stored in
    the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Length – a power of two for FFT speed.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Load previous best, if any.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Phase 1 – diverse seed pool.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6e}")

    # Initialise global best.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6e}")

    # --------------------------------------------------------------
    # Phase 2 – Adam (smooth max) on the elite seeds.
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest budget per seed (roughly 0.4 s per 10 k iters)
        max_iters = int(min(80_000, remaining * 30))
        max_iters = max(15_000, max_iters)
        print(
            f"\n--- Adam run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---"
        )
        seq_adam, val_adam = _adam_smooth_opt(
            seed_vec, n, deadline, max_iters=max_iters
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam.copy(), val_adam
            print(f"[overall‑Adam] new global best max={val_adam:.6e}")

        # try the deterministic re‑orderings of the Adam result
        if seq_adam is not None:
            for name, reorder in {
                "alt": _alternate_order,
                "inter": _interleave_ends,
                "rev": lambda x: x[::-1],
            }.items():
                alt = reorder(seq_adam)
                alt_val = _objective(alt)
                if alt_val < global_best_val:
                    global_best_seq, global_best_val = alt, alt_val
                    print(
                        f"[overall‑Adam‑{name}] new global best max={alt_val:.6e}"
                    )

    # --------------------------------------------------------------
    # Phase 3 – Mirror‑subgradient polishing (continuous values).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑subgradient polishing ---")
        seq_mirror, val_mirror = _mirror_subgradient_opt(
            global_best_seq, n, deadline, max_iters=250_000
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror.copy(), val_mirror
            print(f"[overall‑Mirror] new global best max={val_mirror:.6e}")

    # --------------------------------------------------------------
    # Phase 4 – Heavy permutation simulated annealing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation simulated annealing ---")
        seq_perm, val_perm = _perm_anneal(
            global_best_seq, n, deadline, max_moves=600_000
        )
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[overall‑Perm] new global best max={val_perm:.6e}")

    # --------------------------------------------------------------
    # Phase 5 – Final short mirror‑subgradient run (fine polishing).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final mirror polishing ---")
        seq_final, val_final = _mirror_subgradient_opt(
            global_best_seq, n, deadline, max_iters=100_000
        )
        if val_final < global_best_val:
            global_best_seq, global_best_val = seq_final.copy(), val_final
            print(f"[overall‑Final] new global best max={val_final:.6e}")

    # --------------------------------------------------------------
    # Final sanity‑check with the official evaluator.
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate max ≈ {global_best_val:.6e} "
        f"(length {len(global_best_seq)})"
    )

    # expose for reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
