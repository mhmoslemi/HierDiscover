"""
Search for a step‑function that minimizes ``evaluate_sequence``.

The algorithm builds a *low‑overlap* support set of indices:
for a chosen bound L (the maximum allowed number of unordered pairs
(i,j) with i≤j that share the same sum i+j) we repeatedly construct
sets S ⊂ {0,…,n‑1} by a greedy random process.  With a uniform weight
1/|S| on the chosen indices the convolution ``a*a`` counts, for each
shift, the ordered number of pairs (i,j) with i+j=shift.  By controlling
L we keep this count small, and by making the support as large as possible
we minimise the objective 2·n·max(conv)/(∑a)² = 2·n·max(conv)/(1)².

The search runs many independent greedy trials (different random orderings)
for several values of L (2‒4) and keeps the best vector found.
The construction is extremely cheap, so many trials are performed
well within the 1000 s budget.  The final sequence is stored in the
global variable ``height_sequence_1`` and also returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Small utilities
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate objective for ℓ₁‑normalised sequences (sum = 1)."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    return float(2.0 * n * np.max(conv))


def _greedy_low_overlap_set(n: int, L: int, rng: np.random.Generator) -> np.ndarray:
    """
    Construct a set S of indices such that each unordered pair (i,j) with i≤j
    satisfies ``i + j`` appears at most L times.
    Greedy random insertion, returns the set as a NumPy array of dtype int.
    """
    # count of unordered pairs for each possible sum (0 … 2n‑2)
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

    S = []  # list of chosen indices (Python ints)

    # a random permutation of all possible positions
    cand = np.arange(n, dtype=np.int32)
    rng.shuffle(cand)

    for x in cand:
        # self‑pair (x,x) would increase sum 2*x
        if sum_counts[2 * x] >= L:
            continue

        # check all existing members of S
        ok = True
        for y in S:
            if sum_counts[x + y] >= L:
                ok = False
                break
        if not ok:
            continue

        # ----- add x to the set -----
        S.append(int(x))
        sum_counts[2 * x] += 1                 # the self‑pair
        for y in S[:-1]:                       # all previous members
            sum_counts[x + y] += 1

    return np.array(S, dtype=np.int32)


def _evaluate_support(S: np.ndarray, n: int) -> tuple[float, int, int]:
    """
    Given a support set S, compute the surrogate objective for the uniform
    distribution on S.  Returns (objective, |S|, max ordered count).
    """
    k = S.size
    if k == 0:
        return np.inf, 0, 0

    # binary indicator of the support
    bin_vec = np.zeros(n, dtype=np.int8)
    bin_vec[S] = 1

    # full linear convolution (ordered pair counts)
    conv_counts = np.convolve(bin_vec, bin_vec)          # length 2n‑1, dtype int
    max_cnt = int(conv_counts.max())                     # ordered count for the worst shift

    # uniform weight = 1/k → conv_max = max_cnt / k²
    obj = 2.0 * n * max_cnt / (k * k)
    return obj, k, max_cnt


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Build a sparse step function with a very low max‑convolution value.
    The method repeatedly creates low‑overlap support sets (B₂[L] sets)
    for L = 2, 3, 4, keeps the one giving the smallest surrogate objective,
    and finally returns the corresponding normalized vector.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    # length of the candidate vector – 4000 has been a good trade‑off
    n = 4000

    best_obj = np.inf
    best_seq = None
    best_info = None

    # ------------------------------------------------------------------
    # Stage 1 – many greedy trials for various L
    # ------------------------------------------------------------------
    L_values = [2, 3, 4]          # L = allowed unordered collisions per sum
    trial = 0
    max_trials = 10_000           # generous upper bound; the loop will stop by deadline

    print(f"\n--- Greedy low‑overlap construction (n={n}) ---")
    while time.time() < deadline and trial < max_trials:
        for L in L_values:
            if time.time() >= deadline:
                break

            S = _greedy_low_overlap_set(n, L, rng)
            obj, k, max_cnt = _evaluate_support(S, n)

            if obj < best_obj:
                best_obj = obj
                best_info = (L, k, max_cnt)
                seq = np.zeros(n, dtype=np.float64)
                if k > 0:
                    seq[S] = 1.0 / k                # uniform weights on the support
                best_seq = seq

                print(
                    f"[Trial {trial:5d}] New best → L={L}, |S|={k}, "
                    f"max_cnt={max_cnt}, surrogate={obj:.6f}"
                )
        trial += 1

    # If for any reason we did not obtain a candidate, fall back to uniform
    if best_seq is None:
        print("No candidate produced – falling back to uniform vector.")
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_obj = _objective(best_seq)

    # ------------------------------------------------------------------
    # Stage 2 – optional short projected gradient polishing
    # (only a few thousand iterations; cheap but can shave a little)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:           # run polishing only if we have a few seconds left
        # Simple projected gradient descent on the exact objective
        a = best_seq.copy()
        best_val = best_obj
        lr = 0.8
        it = 0
        while time.time() < deadline and it < 30_000:
            it += 1
            conv = np.convolve(a, a)
            max_idx = int(np.argmax(conv))
            # sub‑gradient for the active shift (ordered pairs)
            rev = max_idx - np.arange(n)
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(a)
            g[mask] = 2.0 * a[rev[mask]]

            a = a - lr * g / np.sqrt(it)   # decreasing step size
            # Euclidean projection onto the probability simplex
            # (the same routine used in the baseline code)
            u = np.sort(a)[::-1]
            cssv = np.cumsum(u) - 1.0
            rho_arr = np.arange(n) + 1
            cond = u - cssv / rho_arr > 0
            if np.any(cond):
                rho = rho_arr[cond].max()
                theta = cssv[cond].max() / rho
                a = np.maximum(a - theta, 0.0)
            else:
                a = np.full_like(a, 1.0 / n)

            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Polish it={it}] improved → {best_val:.6f}")

        best_obj = best_val

    # ------------------------------------------------------------------
    # Final official evaluation & reporting
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning – evaluate_sequence raised:", exc)
        official_val = None

    L_best, k_best, cnt_best = best_info if best_info is not None else (None, None, None)
    print(
        f"\n=== Finished ===\n"
        f"Best surrogate ≈ {best_obj:.6f} (length {len(best_seq)}, support {k_best})\n"
        f"Parameters of best construction: L={L_best}, |S|={k_best}, max_cnt={cnt_best}"
    )

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
