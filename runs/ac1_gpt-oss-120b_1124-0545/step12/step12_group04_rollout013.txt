        indices = np.arange(n)

"""
Sidon‐set based optimizer.

The evaluation function is
    2·n·max(conv(a,a)) / (sum(a))².

If the indices where a_i>0 form a Sidon set (all pairwise sums i+j are
different), each entry of the convolution is either 0 or a_i·a_j.
Choosing all non‑zero entries equal to 1 gives max(conv)=1 and
sum(a)=m, where m is the number of chosen indices.  The objective then
becomes

    value = 2·n / m².

Thus a larger Sidon set gives a smaller value.  The theoretical limit
for length n is m ≈ √(2·n) (≈ 89 for n=4000) which yields a value close
to 1.0.  The simple random‑greedy construction below finds large
Sidon sets very quickly; a set of size ≥ 74 already beats the required
bound 1.503.  The search repeats this construction until the time budget
expires or a satisfactory value is found, then returns the corresponding
step‑function (a list of floats).

The algorithm runs only with the standard library, `numpy` and the
provided `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: random greedy construction of a Sidon (sum‑distinct) set
# ----------------------------------------------------------------------
def _random_sidon_set(length: int, rng: np.random.Generator) -> list[int]:
    """
    Return a list of indices 0 ≤ i < length such that all pairwise sums i+j
    are distinct (including i=j).  The set is built by scanning a random
    permutation of the indices and accepting an index iff it does not
    create a duplicate sum with the already chosen indices.
    """
    order = np.arange(length, dtype=np.int64)
    rng.shuffle(order)

    used_sums = set()          # sums i+j already occupied
    chosen = []                # indices kept so far

    for i in order:
        # quick rejection: double‑sum already taken?
        if (2 * i) in used_sums:
            continue

        # check all i + j for previously chosen j
        conflict = False
        for j in chosen:
            if (i + j) in used_sums:
                conflict = True
                break
        if conflict:
            continue

        # accept i
        for j in chosen:
            used_sums.add(i + j)   # i+j == j+i, store once
        used_sums.add(2 * i)
        chosen.append(int(i))

    return chosen


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    `evaluate_sequence`.  The search is based on constructing large Sidon
    sets; each successful set yields a vector of 0/1 with objective
    2·n / m² where m is the set size.  The routine repeats the greedy
    construction while time permits and keeps the best vector found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 1.5

    # length of the step function – 4000 is a good trade‑off
    n = 4000

    best_m = 0
    best_set = []
    best_val = float("inf")
    attempts = 0

    target_val = 1.503  # stop as soon as we beat the required bound

    print(f"\n--- Sidon‑set search (length {n}) ---")
    while time.time() < deadline:
        attempts += 1
        cand_set = _random_sidon_set(n, rng)
        m = len(cand_set)
        if m == 0:
            continue

        # For a perfect Sidon set max(conv) = 1, sum = m
        cand_val = 2.0 * n / (m * m)

        if cand_val < best_val:
            best_val = cand_val
            best_m = m
            best_set = cand_set.copy()
            print(f"[Attempt {attempts:4d}] m={m:3d}  value={cand_val:.6f}")

            if best_val <= target_val:
                print("Target achieved – exiting early.")
                break

    # ------------------------------------------------------------------
    # Build the final sequence (0/1 at the Sidon positions)
    # ------------------------------------------------------------------
    final_seq = [0.0] * n
    for idx in best_set:
        final_seq[idx] = 1.0

    # Verify with the official evaluator (just for information)
    try:
        official_val = evaluate_sequence(final_seq)
        print(f"\n[Eval] Official evaluate_sequence = {official_val:.6f}")
        # keep the officially computed value as the final metric
        best_val = official_val
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest Sidon set size = {best_m} "
        f" → objective ≈ {best_val:.6f}"
    )

    # expose for possible reuse by subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
