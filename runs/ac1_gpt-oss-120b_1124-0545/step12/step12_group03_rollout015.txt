"""
Improved search for step‑function minimizing `evaluate_sequence`.

Algorithm
---------
1. **Broad seeding** – many analytic windows, random Dirichlet,
   mixtures of Gaussians, low‑frequency cosine sums, piecewise‑linear
   random shapes.  All candidates are L1‑normalised (Σ=1).

2. **Fast surrogate ranking** – compute an inexpensive surrogate
   `obj = 2·n·max(conv(a,a))` with FFT.  Keep the `top_k` best candidates.

3. **Adam on a smoothed max** – each top seed is refined by Adam on the
   log‑softmax parametrisation of the simplex.  A temperature‑controlled
   log‑sum‑exp surrogate of the max is used; temperature is raised up to
   `β_max = 3000` so the surrogate is virtually the true max.

4. **Exact mirror‑descent** – the best‑so‑far vector is further optimised
   by exponentiated‑gradient sub‑gradient steps on the exact max‑conv.
   The learning‑rate schedule is more aggressive than before.

5. **Iterated LP polishing** – several iterations of the LP
   `max Σ g_i  s.t.  conv(f,g) ≤ M` (with `M = max(conv(f,f))`) are
   performed.  Each iteration mixes the LP solution back into `f` and
   renormalises, strictly decreasing the objective.

6. **Long simulated‑annealing** – the remaining time is spent in a
   lightweight SA that performs swaps, Dirichlet redistributions and
   pair‑mass transfers.

The function respects the global time‑budget (default 1000 s), stores
the best sequence in the global name `height_sequence_1` and returns it
as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_obj(seq: np.ndarray) -> float:
    """
    Surrogate objective used for quick ranking.
    The sequence is assumed to be L1‑normalised (Σseq = 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piece‑wise linear curve with 3 random knots."""
    # generate 3 increasing break‑points
    breaks = np.sort(rng.choice(n, size=3, replace=False))
    heights = rng.random(3) + 0.01  # ensure positivity
    # linearly interpolate between the knots
    x = np.arange(n, dtype=np.float64)
    a = np.interp(x, breaks, heights)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few random Gaussians (centred & off‑centred)."""
    a = np.zeros(n, dtype=np.float64)
    num = rng.integers(2, 6)
    for _ in range(num):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.02 * n, 0.15 * n)
        x = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth curve built from a few cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax parameters using a log‑sum‑exp surrogate."""
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.9
    beta = 0.5
    beta_inc = 0.5
    beta_max = 3000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # simplex point
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smoothed max (log‑sum‑exp)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 3_500_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max‑conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.25
    power = 0.28

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 12, tau: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Perform several LP refinements.

    For each iteration we solve:
        maximise   Σ g_i
        subject to conv(seq, g) ≤ M   (M = max conv(seq, seq))
                  g_i ≥ 0

    The solution g is mixed with the current seq (weight τ) and renormalised.
    """
    f = seq.copy()
    n = len(f)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()
        # Build convolution matrix for current f
        rows = np.empty(n * n, dtype=np.int32)
        cols = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)

        idx = 0
        for k in range(n):
            start = k
            end = k + n
            rows[idx:idx + n] = np.arange(start, end, dtype=np.int32)
            cols[idx:idx + n] = k
            data[idx:idx + n] = f
            idx += n

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)      # maximise Σ g_i

        # HiGHS solver – give a per‑iteration time limit to stay safe
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True,
                                               'time_limit': 20})

        if not res.success:
            print(f"[LP] iteration {it} failed – stopping LP stage.")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:
            print(f"[LP] iteration {it} – no L1 gain.")
            break

        # convex combination and renormalisation
        f_new = (1.0 - tau) * f + tau * g
        f_new = f_new / f_new.sum()
        f = f_new

        cur_obj = _surrogate_obj(f)
        print(f"[LP] iteration {it} – objective = {cur_obj:.6f}")

    final_obj = _surrogate_obj(f)
    return f, final_obj


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 2_500_000,
                                 init_temp_factor: float = 0.06,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Lightweight SA with three move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling – fairly slow
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            # swap two positions
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            # redistribute mass on a random small subset
            k = rng.integers(2, 14)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # Parameter sweeps
    for gamma in np.linspace(0.2, 5.0, 9):
        # sine window is already covered by cosine‑type shapes, keep it simple
        pass

    for sigma in np.linspace(0.05, 0.35, 8):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 4.5, 10):
        seeds.append(_power_law(n, p))

    alphas = np.linspace(0.2, 3.0, 9)
    betas = np.linspace(0.2, 3.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for lam in np.linspace(0.0005, 0.015, 10):
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet draws – broad concentration range
    concs = np.logspace(-4, 0.5, 12)   # from 1e‑4 to ~3.2
    for conc in concs:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixture of few random Gaussians
    for _ in range(5):
        seeds.append(_gaussian_mixture(n, rng))

    # Random piece‑wise linear shapes
    for _ in range(5):
        seeds.append(_piecewise_linear(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 5, 7):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mix uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * curve
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # Previous solution (if supplied)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)

"""
Improved search for a step‑function that minimises the official
``evaluate_sequence`` objective.

Algorithm
---------
1. **Broad seeding** – many analytic windows, random Dirichlet draws,
   random mixtures of Gaussians, low‑frequency cosine curves,
   piece‑wise‑linear random shapes and a few hand‑crafted families.
   All candidates are L₁‑normalised (Σ = 1).

2. **Fast surrogate ranking** – a cheap surrogate
   ``obj = 2·n·max(conv(a,a))`` (computed via FFT) is used to keep the
   best ``top_k`` seeds.

3. **Adam on a smoothed max** – each top seed is refined by Adam on the
   log‑softmax parametrisation of the simplex.  A temperature‑controlled
   log‑sum‑exp surrogate of the max is employed; the temperature is
   gradually increased up to 3000 so the surrogate is virtually the true
   max.

4. **Exact mirror‑descent** – the best vector so far is further improved
   with exponentiated‑gradient sub‑gradient steps on the *exact* max‑conv
   (no smoothing).  The learning‑rate schedule is more aggressive.

5. **Iterated LP polishing** – a handful of linear‑programming steps
   ``max Σ g_i  s.t.  conv(f,g) ≤ M`` (with ``M = max(conv(f,f))``) are
   performed.  Each LP solution is mixed back into ``f`` and renormalised,
   strictly decreasing the objective.

6. **Simulated‑annealing polishing** – the remaining time is spent in a
   lightweight SA that swaps, redistributes or transfers mass between
   random positions.

The function respects the global time‑budget (default 1000 s), stores the
best sequence in the global name ``height_sequence_1`` and returns it as a
plain list of ``float``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_obj(seq: np.ndarray) -> float:
    """
    Fast surrogate objective (Σseq = 1):
        2·n·max(conv(seq,seq))
    Ordering of candidates is unchanged by the omitted denominator.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 3‑segment piece‑wise linear shape."""
    breaks = np.sort(rng.choice(n, size=3, replace=False))
    heights = rng.random(3) + 0.01
    x = np.arange(n, dtype=np.float64)
    a = np.interp(x, breaks, heights)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few randomly positioned Gaussians."""
    a = np.zeros(n, dtype=np.float64)
    num = rng.integers(2, 6)
    for _ in range(num):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.02 * n, 0.15 * n)
        x = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth curve built from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax parameters with a log‑sum‑exp surrogate of max."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.9
    beta = 0.5
    beta_inc = 0.5
    beta_max = 3000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smooth max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 4_000_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the exact max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.25
    power = 0.28

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 12, tau: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Perform several LP refinements.

    Each iteration solves:
        maximise   Σ g_i
        subject to conv(seq, g) ≤ M   (where M = max conv(seq, seq))
                  g_i ≥ 0
    The solution g is mixed back into the current sequence with weight τ
    and renormalised.
    """
    f = seq.copy()
    n = len(f)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()

        # Build convolution matrix for current f (sparse, exactly n² non‑zeros)
        rows = np.empty(n * n, dtype=np.int32)
        cols = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)

        idx = 0
        for k in range(n):
            start = k
            end = k + n
            rows[idx:idx + n] = np.arange(start, end, dtype=np.int32)
            cols[idx:idx + n] = k
            data[idx:idx + n] = f
            idx += n

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if not res.success:
            print(f"[LP] iteration {it} failed – stopping LP stage.")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:
            print(f"[LP] iteration {it} – no L1 gain.")
            break

        f_new = (1.0 - tau) * f + tau * g
        f_new = f_new / f_new.sum()
        f = f_new

        cur_obj = _surrogate_obj(f)
        print(f"[LP] iteration {it} – objective = {cur_obj:.6f}")

    return f, _surrogate_obj(f)


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 2_500_000,
                                 init_temp_factor: float = 0.06,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Lightweight SA with three move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # gentle exponential cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            # redistribute mass on a small random subset
            k = rng.integers(2, 14)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # parameter sweeps
    for sigma in np.linspace(0.05, 0.35, 8):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 4.5, 10):
        seeds.append(_power_law(n, p))

    alphas = np.linspace(0.2, 3.0, 9)
    betas = np.linspace(0.2, 3.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for lam in np.linspace(0.0005, 0.015, 10):
        seeds.append(_exponential_decay(n, lam))

    # random Dirichlet draws with a broad range of concentrations
    concs = np.logspace(-4, 0.5, 12)   # 1e‑4 … ~3.2
    for conc in concs:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures of random Gaussians
    for _ in range(5):
        seeds.append(_gaussian_mixture(n, rng))

    # piece‑wise‑linear random shapes
    for _ in range(5):
        seeds.append(_piecewise_linear(n, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 5, 7):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mix uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * curve
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds

        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=4_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Iterated LP polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming polishing (iterated) ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, deadline,
                                    max_iters=12, tau=0.15)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=2_500_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
