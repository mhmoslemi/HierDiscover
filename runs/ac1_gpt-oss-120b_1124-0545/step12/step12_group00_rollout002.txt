def _adam_optimize(start_x, n, deadline, max_iters=200_000,
                   lam_start=5.0, lam_end=150.0, lr0=0.1,
                   beta1=0.9, beta2=0.999, eps=1e-8):
    w = np.log(start_x + 1e-12)  # shape (n,)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    best_x = start_x.copy()
    best_val = _objective_surrogate(start_x)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # convert to probability vector
        # careful: w may cause overflow exp(w); use stable softmax
        max_w = np.max(w)
        exp_w = np.exp(w - max_w)
        x = exp_w / exp_w.sum()
        
        # lambda schedule
        lam = lam_start + (lam_end - lam_start) * (it / max_iters)
        
        # compute conv and gradient
        conv = _conv_fft(x, x)
        max_conv = conv.max()
        
        # compute smooth gradient
        # compute wgt = exp(lam*(conv - max_conv))
        # subtract max to avoid overflow
        exp_c = np.exp(lam * (conv - max_conv))
        wgt = exp_c / exp_c.sum()
        
        # gradient wrt x
        wgt_rev = wgt[::-1]
        grad_x = 2.0 * _conv_fft(wgt_rev, x)[(2 * n - 2) - np.arange(n)]
        # compute gradient wrt w
        dot = np.dot(grad_x, x)
        grad_w = x * (grad_x - dot)  # shape n
        
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # learning rate schedule: decays about 1/sqrt(it) say
        lr = lr0 / math.sqrt(it)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # compute surrogate objective (real max)
        val = 2.0 * n * max_conv
        if val < best_val:
            best_val = val
            best_x = x.copy()
            # print progress
            if it % 50000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")
    return best_x, best_val

"""
Propose a low‑convolution step function.

Algorithm
---------
1. **Seed pool** – generate a diverse collection of analytic windows,
   random Dirichlet draws and low‑frequency cosine curves, all ℓ₁‑normalised.
   Keep the best few according to the surrogate objective
   `2·n·max(conv(x,x))` (identical to `evaluate_sequence` when Σx = 1).

2. **Smooth‑max Adam optimisation** – treat the vector as a soft‑max of
   unconstrained logits.  A temperature‑controlled soft‑max of the
   convolution values provides a *smooth* approximation of the max,
   giving a usable gradient for *all* shifts.  Using Adam on the logits
   (with a continuation schedule that raises the temperature λ from
   modest to large) quickly drives the true max down while avoiding the
   plateaus of pure sub‑gradient descent.

3. **Exponential‑gradient polishing** – the Adam phase works with
   strictly positive entries; a short mirror‑descent (the classic
   exponentiated‑gradient update) is applied afterwards to allow the
   mass to concentrate arbitrarily close to zero and to fine‑tune the
   ℓ₁‑simplex point.

4. **Final evaluation** – the best vector is passed to the official
   `evaluate_sequence` function, stored globally as ``height_sequence_1``,
   and returned as a plain Python list.

The whole search respects a user‑supplied time budget (default 1000 s)
and runs on a single process using only NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) (the quantity returned by
    `evaluate_sequence` when Σseq = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    """Monotonically decreasing exponential, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final safety normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        clean.append(v / s if s > 0 else _uniform(n))
    return clean


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Classic EG refinement – multiplicative update with simplex projection."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # Sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow
        x = x * np.exp(delta)

        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")
    return best_x, best_val


# ----------------------------------------------------------------------
# Adam optimiser using a smooth‑max approximation of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3_000_000,
    lam_start: float = 5.0,
    lam_end: float = 150.0,
    lr0: float = 0.12,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam on the soft‑max logits with a temperature λ that slowly grows.
    The loss is the *true* max‑convolution (not the smooth surrogate);
    a smooth soft‑max is only used to obtain a well‑behaved gradient.
    """
    # initialise logits (soft‑max parametrisation)
    w = np.log(start + 1e-12)
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    best_x = start.copy()
    best_val = _objective_surrogate(best_x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- soft‑max to obtain a probability vector -----
        max_w = np.max(w)
        exp_w = np.exp(w - max_w)
        x = exp_w / exp_w.sum()                     # Σx = 1, x ≥ 0

        # ----- annealing schedule for λ (temperature) -----
        lam = lam_start + (lam_end - lam_start) * (it / max_iters)

        # ----- convolution and smooth‑max weight -----
        conv = _conv_fft(x, x)                      # length 2n‑1
        max_conv = conv.max()
        # normalise to [0,1] before applying λ
        if max_conv == 0:
            weight = np.full_like(conv, 1.0 / conv.size)
        else:
            conv_norm = conv / max_conv
            # w_i ∝ exp( λ·(conv_i / max - 1) ), the max gets weight 1
            exp_wgt = np.exp(lam * (conv_norm - 1.0))
            weight = exp_wgt / exp_wgt.sum()

        # ----- gradient of the smooth max w.r.t x -----
        # g_x = 2 * Σ_i weight_i * shift_i(x)
        w_rev = weight[::-1]
        grad_full = _conv_fft(w_rev, x)            # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)           # extraction indices
        grad_x = 2.0 * grad_full[idx]

        # ----- chain rule through the soft‑max (Jacobian) -----
        dot = np.dot(grad_x, x)
        grad_w = x * (grad_x - dot)                # ∂L/∂w

        # ----- Adam update on logits -----
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / math.sqrt(it)                   # simple decay
        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- keep the best (real) objective seen so far -----
        cur_val = 2.0 * n * max_conv                # exact surrogate
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **_kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin for final eval

    # ------------------------------------------------------------------
    # Length of the step function – larger lengths give a slightly finer
    # search space without exploding the FFT size.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Gather a diversified seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # evaluate surrogates and keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Initialise from the best seed
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation with smooth‑max gradient
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # allocate almost the whole remaining budget here; the loop will
        # stop automatically once the deadline is reached.
        max_it = int(min(3_000_000, remaining * 1800))
        print(f"\n--- Adam optimisation (up to {max_it} iters) ---")
        seq, val = _adam_optimize(
            best_seq,
            n,
            deadline,
            max_iters=max_it,
            lam_start=5.0,
            lam_end=150.0,
            lr0=0.12,
        )
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Adam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – final exponential‑gradient polishing (allows true zeros)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_it = int(min(800_000, remaining * 1500))
        print("\n--- Final EG polishing ---")
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline, max_iters=max_it, eta0=0.25
        )
        print(f"[Final EG] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in subsequent runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
