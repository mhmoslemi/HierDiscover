def _lp_refine_iterative(seq, deadline, max_iters=30, tau_init=0.3, tau_decay=0.95):
    f = seq.copy()
    tau = tau_init
    best_val = _objective(f)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        M = _conv_fft(f, f).max()
        # build matrix
        A = _build_conv_matrix(f)
        b = np.full(A.shape[0], M)
        bounds = [(0, None)]*len(f)
        c = -np.ones(len(f))
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
        if not res.success:
            print(f"[LP] iteration {it} failed.")
            break
        g = res.x
        # combine
        f = (1 - tau) * f + tau * g
        f = f / f.sum()
        # compute new obj
        cur_val = _objective(f)
        print(f"[LP] iter {it}, tau={tau:.3f}, obj={cur_val:.6f}")
        if cur_val >= best_val - 1e-9:  # no improvement
            break
        best_val = cur_val
        tau *= tau_decay
    return f, best_val
```

"""
**Step‑function optimiser for the `evaluate_sequence` challenge**

The optimiser works in three phases:

1. **Initial seed** – we start from the best known sequence (`height_sequence_1`
   if it exists) or from a rich pool of analytically‑generated shapes.
2. **Mirror‑subgradient descent** – a multiplicative (exponentiated‑gradient)
   descent on the exact max‑convolution reduces the objective quickly.
3. **Iterated LP polishing** – repeatedly solve a linear programme that
   maximises the ℓ₁‑mass under the current max‑convolution bound and
   convex‑combine the solution.  With a large mixing weight (τ≈0.5) the bound
   drops substantially in a few iterations.  The LP matrix is pre‑computed
   once (𝑂(n²) memory, 𝑂(n) build per iteration).

After the LP stage we perform a short second mirror descent and a lightweight
simulated‑annealing sweep to fine‑tune the final answer.

All heavy work respects the global time budget (default 1000 s) and uses only
`numpy`, `scipy` and the already‑available `evaluate_sequence` function.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective (scale‑invariant).  seq is assumed ℓ₁‑normalised."""
    n = len(seq)
    return 2.0 * n * np.max(_conv_fft(seq, seq))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random / low‑frequency helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve built from a few cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed collection – richer parameter grids
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # fine grids
    for gamma in np.linspace(0.2, 5.0, 15):
        seeds.append(_sine_window(n, gamma=gamma))
    for sigma in np.linspace(0.06, 0.50, 12):
        seeds.append(_gaussian(n, sigma_frac=sigma))
    for p in np.linspace(0.0, 6.0, 13):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 3.5, 10)
    betas = np.linspace(0.2, 3.5, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.05, 0.95, 12):
        seeds.append(_tukey_window(n, alpha))
    for lam in np.linspace(0.0002, 0.02, 12):
        seeds.append(_exponential_decay(n, lam))

    # random Dirichlet draws – wide range of concentrations
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)
            seeds.append(np.sort(raw)[::-1])   # descending
            seeds.append(np.sort(raw))         # ascending
            # alternating order
            a = np.sort(raw)
            lo, hi = 0, len(a) - 1
            alt = np.empty_like(a)
            pos = 0
            while lo <= hi:
                if pos % 2 == 0:
                    alt[pos] = a[hi]
                    hi -= 1
                else:
                    alt[pos] = a[lo]
                    lo += 1
                pos += 1
            seeds.append(alt)

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _mirror_subgradient_optimize(

    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on max‑convolution, multiplicative update."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.15      # initial learning rate
    power = 0.30    # decay exponent

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative (exponentiated‑gradient) update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # project back onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (lightweight)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Three elementary moves – swap / reshuffle / pair‑transfer."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                               # swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                             # reshuffle a block
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                            # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming helper – pre‑compute index pattern once
# ----------------------------------------------------------------------
def _precompute_conv_indices(n: int):
    """Rows and columns for the (full) convolution matrix (constant pattern)."""
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    idx = 0
    for k in range(n):
        rows[idx : idx + n] = np.arange(k, k + n, dtype=np.int32)
        cols[idx : idx + n] = k
        idx += n
    return rows, cols


def _build_conv_matrix(f: np.ndarray, rows: np.ndarray, cols: np.ndarray) -> csr_matrix:
    """CSR matrix A such that (A @ g) = conv(f, g).  Shape (2n‑1, n)."""
    n = len(f)
    data = np.tile(f, n)                # O(n) fill
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Repeated LP refinement (aggressive τ schedule)
# ----------------------------------------------------------------------
def _lp_refine_iterative(
    seq: np.ndarray,
    deadline: float,
    max_iters: int = 30,
    tau_init: float = 0.5,
    tau_decay: float = 0.94,
) -> tuple[np.ndarray, float]:
    """
    Solve, mix and renormalise repeatedly:

        maximise Σ g_i   s.t. conv(seq, g) ≤ M   (M = max conv(seq, seq))

    The mix weight τ starts large (≈0.5) and decays geometrically.
    """
    f = seq.copy()
    n = len(f)
    rows, cols = _precompute_conv_indices(n)

    best_val = _objective(f)
    tau = tau_init

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()
        A = _build_conv_matrix(f, rows, cols)
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)   # maximise Σ g_i

        # Solve LP (HiGHS is the default efficient backend)
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            print(f"[LP] iteration {it} – solver failure, stopping LP stage.")
            break

        g = res.x
        # convex combination, then re‑normalise to ℓ₁ = 1
        f = (1.0 - tau) * f + tau * g
        f = f / f.sum()

        cur_val = _objective(f)
        print(f"[LP] iter {it:2d} τ={tau:.3f} objective={cur_val:.6f}")

        if cur_val + 1e-12 >= best_val:          # no improvement
            # stop early if we are no longer making progress
            break

        best_val = cur_val
        tau = max(tau * tau_decay, 0.02)         # keep τ positive but shrink

    return f, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence is stored globally as ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final official evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good balance of expressiveness / speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous solution if present (it is already a good candidate).
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev = np.asarray(prev, dtype=np.float64)
            if prev.size != n:
                if prev.size > n:
                    prev = prev[:n]
                else:
                    prev = np.pad(prev, (0, n - prev.size), constant_values=0.0)
        except Exception:
            prev = None

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, prev)
    print(f"Generated {len(seeds)} raw seeds.")

    # Quick surrogate evaluation – keep the most promising few
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # --------------------------------------------------------------
    # Initialise the current best – use the very best seed.
    # --------------------------------------------------------------
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Mirror‑subgradient descent (exact max).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑subgradient descent (phase 1) ---")
        max_iters = int(min(2_000_000, (deadline - time.time()) * 80))
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=max_iters
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror.copy(), val_mirror
            print(f"[Phase 1] new global best = {val_mirror:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Aggressive LP polishing (many iterations, large τ).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming polishing (phase 2) ---")
        seq_lp, val_lp = _lp_refine_iterative(
            global_best_seq,
            deadline,
            max_iters=40,
            tau_init=0.5,
            tau_decay=0.94,
        )
        if val_lp < global_best_val:
            global_best_seq, global_best_val = seq_lp.copy(), val_lp
            print(f"[Phase 2] new global best = {val_lp:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Short second mirror descent (fine‑tuning).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑subgradient descent (phase 3) ---")
        max_iters = int(min(500_000, (deadline - time.time()) * 80))
        seq_m2, val_m2 = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=max_iters
        )
        if val_m2 < global_best_val:
            global_best_seq, global_best_val = seq_m2.copy(), val_m2
            print(f"[Phase 3] new global best = {val_m2:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated‑annealing polishing (remaining time).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing (phase 4) ---")
        remaining = deadline - time.time()
        # roughly 2000 cheap SA flips per second on a typical VM
        max_iters = int(max(0, remaining * 2000))
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=max_iters
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[Phase 4] new global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation (purely informational)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
