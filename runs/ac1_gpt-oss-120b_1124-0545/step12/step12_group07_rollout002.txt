Improved search for the minimal‐value step function.

The algorithm consists of three phases:

1. **Seed generation** – a handful of analytically‑derived shapes (uniform,
   triangular, trapezoidal, β‑shapes, Gaussian, etc.) together with a few
   random Dirichlet draws and the previous best solution (if any).  All vectors
   are ℓ₁‑normalised (sum = 1).

2. **Smooth‑max Adam optimisation** – we minimise a smooth approximation of the
   true objective
       f(a)=2·n·max(conv(a,a))
   by replacing the max with a log‑sum‑exp (temperature β).  The gradient of the
   smoothed objective can be written in closed form using a correlation of the
   softmax weights with the current vector.  An Adam optimiser (with simplex
   projection after each update) drives the vector towards lower maximum
   autocorrelation.  The temperature β is annealed from a small value (≈0.5)
   to a large one (≈5000) so that the optimisation first moves in a smooth
   landscape and then focusses on the true max.

3. **Lightweight polishing** – a short simulated‑annealing stage that
   repeatedly reshuffles a random subset of coordinates via a Dirichlet move,
   accepting improvements (or uphill moves with a geometric temperature schedule).

All heavy work (convolution, correlation) is performed with real FFTs, and the
routine respects the user‑supplied time budget (default 1000 s).  The best
sequence found is stored in the global ``height_sequence_1`` variable.
"""

# Low‑level utilities (FFT, simplex projection, etc.)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerically unstable case – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors of length n)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Surrogate objective (for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)).  Assumes sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth‑max objective + gradient (used by Adam)
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray, beta: float, Nfft: int) -> tuple[float, np.ndarray]:
    """
    Returns a smoothed approximation of the true objective and its gradient.
    The smoothing uses the log‑sum‑exp with temperature beta.
    """
    n = len(a)
    L = 2 * n - 1

    # --- convolution a*a (full) ---------------------------------------
    A = np.fft.rfft(a, n=Nfft)
    conv_full = np.fft.irfft(A * A, n=Nfft)[:L]           # length 2n‑1

    # --- smooth max (log‑sum‑exp) ------------------------------------
    z = beta * conv_full
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)                             # stable
    sum_exp = np.sum(exp_z)
    # LSE = (z_max + log(sum_exp)) / beta
    lse = (z_max + np.log(sum_exp)) / beta
    obj = 2.0 * n * lse

    # softmax weights (sum to 1)
    w = exp_z / sum_exp                                   # length 2n‑1

    # --- gradient: 4·n·corr(w, a) ------------------------------------
    # correlation = sum_k w_{j+k} * a_k  ,  j=0..n‑1
    # compute via FFT: corr = ifft( fft(w) * conj( fft(a_rev) ) )
    fw = np.fft.rfft(w, n=Nfft)
    fa_rev = np.fft.rfft(a[::-1], n=Nfft)
    corr_full = np.fft.irfft(fw * fa_rev, n=Nfft)[:L + n - 1]   # length 3n‑2
    start = n - 1
    corr = corr_full[start:start + n]                          # length n
    grad = 4.0 * n * corr

    return obj, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   beta_start: float = 0.5,
                   beta_end: float = 5000.0,
                   max_iters: int = 30000,
                   lr: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector (in terms of the true surrogate) found during the run.
    """
    n = len(start)
    a = start.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    L = 2 * n - 1
    Nfft = _next_pow2(L + n)        # enough for convolution & correlation

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear annealing of the temperature β
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        obj_smooth, grad = _smooth_max_and_grad(a, beta, Nfft)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _proj_simplex(a)

        # occasional evaluation of the true surrogate
        if t % 200 == 0 or t == max_iters:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] iter={t:6d} surrogate={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny stochastic polishing (Dirichlet reshuffle of a random block)
# ----------------------------------------------------------------------


def _polish_simulated_annealing(start: np.ndarray,
                                deadline: float,
                                init_temp_factor: float = 0.03,
                                min_temp: float = 1e-12,
                                max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Very lightweight SA that moves mass among a random subset of entries."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    n = len(cur)
    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        T = max(T * 0.9995, min_temp)

        # pick a random block size between 2 and 15
        k = rng.integers(2, 16)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if best_val < cur_val:  # print only on improvement
                    print(f"[SA] new best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # ------------------------------------------------------------------
    # Length of the candidate vector – a balance between resolution and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) for warm‑starting
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.shape != (n,):
                previous = None
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Seed pool
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.08, 0.32, 5):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for a in np.linspace(0.3, 2.5, 5):
        for b in np.linspace(0.3, 2.5, 5):
            seeds.append(_beta_shape(n, a, b))
    for sigma in (0.12, 0.20, 0.28):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws with varied concentration
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # include previous solution if available
    if previous is not None:
        seeds.append(previous / previous.sum())

    # ------------------------------------------------------------------
    # Pick the best seed according to the exact surrogate
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"[Init] best surrogate from seeds = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on smooth max
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        adam_deadline = time.time() + 0.70 * time_left
        print("\n--- Adam optimisation (smooth‑max) ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq,
            adam_deadline,
            beta_start=0.5,
            beta_end=5000.0,
            max_iters=25000,
            lr=0.07,
        )
        if val_adam < best_val:
            best_seq, best_val = seq_adam, val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – final stochastic polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _polish_simulated_annealing(
            best_seq,
            deadline,
            init_temp_factor=0.03,
            max_iters=500_000,
        )
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official function
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
