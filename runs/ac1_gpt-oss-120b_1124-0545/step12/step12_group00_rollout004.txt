def _gradient_descent_smooth(start_seq, n, deadline):
    # use Adam
    x = start_seq.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr0 = 0.1
    it = 0
    best_x = x.copy(); best_val = float('inf')
    lam = 5.0
    lam_inc_iter = 50000
    max_iters = 5000000  # fallback
    while time.time() < deadline and it < max_iters:
        it += 1
        if it % lam_inc_iter == 0:
            lam = min(lam*2, 200.0)   # increase
        # compute conv, softmax, gradient
        conv = _conv_fft(x, x)
        max_conv = np.max(conv)
        # compute probabilities via softmax
        # for numeric stability subtract max
        expo = np.exp(lam * (conv - max_conv))
        sum_expo = expo.sum()
        p = expo / sum_expo
        # compute gradient of smooth max
        rev = x[::-1]
        conv_grad_full = _conv_fft(rev, p)  # size ~ len(rev)+len(p)-1
        # take slice
        start_idx = n - 1
        grad_smooth = 2.0 * conv_grad_full[start_idx:start_idx + n]
        # gradient of obj: simply grad_smooth as we keep sum=1 via projection.
        # Adam moments
        m = beta1 * m + (1 - beta1) * grad_smooth
        v = beta2 * v + (1 - beta2) * (grad_smooth ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size schedule
        lr = lr0 / (1 + it * 1e-5)   # decreasing small
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        x = _proj_simplex(x_new)
        # evaluate surrogate true max
        # compute true max conv each maybe every 2000 iterations
        if it % 2000 == 0:
            true_obj = 2 * n * max_conv   # since sum=1
            if true_obj < best_val:
                best_val = true_obj
                best_x = x.copy()
                print(f"[GD] it={it} max={max_conv:.6e} surrogate={true_obj:.6f}")
    return best_x, best_val

"""
Smooth‑max gradient descent on the simplex.

The evaluation function is
    obj(a) = 2·n·max_k (a*a)[k] / (sum a)^2 .

Since the objective is homogeneous we may fix Σa = 1.
We minimise  max_k (a*a)[k] by a smooth approximation

    S_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k])   (λ > 0),

whose gradient can be written in closed form using one extra FFT.
The algorithm is a projected Adam optimiser on the probability simplex.
It starts from the previous solution (if any) or from a uniform vector,
gradually increases λ (deterministic annealing) and injects a tiny
random jitter when progress stalls.  After the main phase a short
exponentiated‑gradient polishing step is performed.

The whole routine respects the time budget (default 1000 s).  The final
sequence is stored in the global variable ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0,
    Σx = 1}.  Implementation follows the algorithm of Wang & Carreira‑Perpinan.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerically all entries are the same (or negative)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max optimiser (projected Adam)
# ----------------------------------------------------------------------
def _smooth_max_optimize(start: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 10_000_000,
                         lr0: float = 0.05,
                         lam_start: float = 5.0,
                         lam_factor: float = 1.5,
                         lam_inc_iter: int = 40_000,
                         lam_max: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth approximation of the max
    convolution entry.  Returns the best vector (Σ = 1) seen and its
    true surrogate value 2·n·max(conv).
    """
    x = start.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    lam = lam_start
    best_seq = x.copy()
    best_val = float('inf')
    it = 0
    last_improve_it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- convolution and smooth max (soft‑max) -----
        conv = _conv_fft(x, x)               # length 2n‑1
        max_conv = conv.max()

        # softmax weights (numerically stable)
        shift = max_conv
        exp_weights = np.exp(lam * (conv - shift))
        sum_exp = exp_weights.sum()
        p = exp_weights / sum_exp             # again length 2n‑1, Σp = 1

        # ----- gradient of the smooth max -----
        rev = x[::-1]                         # reversed copy
        grad_full = _conv_fft(rev, p)         # length 3n‑2
        grad_smooth = 2.0 * grad_full[n - 1:n - 1 + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_smooth
        v = beta2 * v + (1.0 - beta2) * (grad_smooth ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + it * 1e-5)           # slowly decaying stepsize
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex (keeps Σx = 1) -----
        x = _proj_simplex(x_new)

        # ----- bookkeeping & progress output -----
        if it % 2000 == 0:
            surrogate = 2.0 * n * max_conv    # true surrogate (Σx = 1)
            if surrogate < best_val:
                best_val = surrogate
                best_seq = x.copy()
                last_improve_it = it
                print(f"[Smooth] it={it:7d}  surrogate={best_val:.6f}  max={max_conv:.6e}")

        # ----- anneal λ -----
        if it % lam_inc_iter == 0:
            lam = min(lam * lam_factor, lam_max)

        # ----- occasional jitter to escape shallow local minima -----
        if it - last_improve_it > 50_000:
            # add a tiny random perturbation and re‑project
            jitter = 0.02 * np.random.rand(n)
            x = _proj_simplex(x + jitter)
            last_improve_it = it

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (as in the previous code)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 150_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with KL divergence (exponential‑gradient) on the exact
    max‑convolution objective.  Returns the best vector seen.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = 2.0 * n * np.max(_conv_fft(x, x))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # stepsize
        eta = eta0 / math.sqrt(it)

        # multiplicative update (clip exponent to avoid overflow)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # projection onto the simplex
        s = x.sum()
        if s <= 0:
            x = np.full_like(x, 1.0 / n)
        else:
            x = x / s

        cur_val = 2.0 * n * np.max(_conv_fft(x, x))
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Helper – up‑sample an old solution to the new length
# ----------------------------------------------------------------------
def _upsample_to(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to change the length of v to new_n (keeps Σ = 1)."""
    old_n = len(v)
    if old_n == new_n:
        return v.copy()
    t_old = np.linspace(0.0, 1.0, old_n)
    t_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(t_new, t_old, v)
    s = up.sum()
    if s <= 0:
        return np.full(new_n, 1.0 / new_n)
    return up / s


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the official
    ``evaluate_sequence`` objective within the time budget.

    The method consists of:
    1. Initialise a probability vector (length ≈ 8000) – either the
       previous best solution (up‑sampled) or a uniform start.
    2. Smooth‑max Adam descent with deterministic annealing of λ.
    3. A short exponentiated‑gradient polishing phase.
    4. Final evaluation with the official evaluator.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # 2 s safety margin

    # ------------------------------------------------------------------
    # Choose a length – 8000 gives us more degrees of freedom without
    # exploding the FFT size (next power‑of‑two stays at 16384).
    # ------------------------------------------------------------------
    TARGET_N = 8000

    # ------------------------------------------------------------------
    # Seed creation – use previous solution if it exists, otherwise uniform.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) > 0:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.sum() > 0:
                init_seq = _upsample_to(prev_arr, TARGET_N)
                print(f"Using up‑sampled previous solution (length {len(prev_arr)} → {TARGET_N}).")
            else:
                raise ValueError
        except Exception:
            init_seq = np.full(TARGET_N, 1.0 / TARGET_N, dtype=np.float64)
            print("Previous solution invalid – falling back to uniform.")
    else:
        init_seq = np.full(TARGET_N, 1.0 / TARGET_N, dtype=np.float64)
        print("No previous solution – starting from uniform.")

    # --------------------------------------------------------------
    # Stage 1 – smooth‑max Adam optimisation.
    # --------------------------------------------------------------
    print("\n--- Smooth‑max Adam optimisation ---")
    best_seq, best_surrogate = _smooth_max_optimize(
        start=init_seq,
        n=TARGET_N,
        deadline=deadline,
        max_iters=5_000_000,       # generous upper bound; loop checks deadline
        lr0=0.05,
        lam_start=5.0,
        lam_factor=1.5,
        lam_inc_iter=40_000,
        lam_max=200.0
    )
    print(f"[Stage 1] best surrogate after smoothed optimisation = {best_surrogate:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – short EG polishing (helps with the true max objective).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        # allocate roughly half of remaining time to polishing
        max_iters = int(min(200_000, remaining * 1500))
        best_seq, best_surrogate = _expgrad_optimize(
            start=best_seq,
            n=TARGET_N,
            deadline=deadline,
            max_iters=max_iters,
            eta0=0.25
        )
        print(f"[Stage 2] final surrogate after EG polishing = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible reuse in a subsequent run.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
