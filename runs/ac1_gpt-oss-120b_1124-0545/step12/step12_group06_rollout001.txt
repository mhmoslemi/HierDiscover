"""
Propose a step‐function (non‑negative sequence) that minimises

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The search follows a simple yet effective multi‑phase pipeline:

1. **Seed generation** – a large and diverse pool of analytically‑defined
   families (uniform, triangular, Gaussian, power, beta, exponential,
   sine‑window, Tukey, low‑frequency random curves) together with many
   Dirichlet draws.  All seeds are ℓ₁‑normalised (sum = 1).

2. **Fast permutation sweep** – for the best few seeds we explore a
   handful of promising orderings (sorted, alternating, several
   coprime‑stride permutations).  Each ordering is evaluated exactly;
   the best ordering becomes the new global incumbent.

3. **Mirror‑descent sub‑gradient optimisation** – an exponentiated‑gradient
   method on the exact max‑convolution objective.  The gradient is
   computed from *all* near‑maximal convolution entries (within a small
   relative tolerance) via a single FFT‑based convolution, giving a
   smooth descent direction while staying on the probability simplex.

4. **Simulated‑annealing polishing** – swap / mass‑transfer moves with
   geometric cooling to escape shallow minima.

5. **Greedy permutation hill‑climb** – final local optimisation that
   accepts only improving swaps.

All phases respect the global time budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a sequence normalised to sum = 1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, ≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Arrange as large, small, 2nd‑large, 2nd‑small, …."""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


def _permute_by_stride(vec: np.ndarray, stride: int) -> np.ndarray:
    """
    Permute a vector by walking with a coprime stride:
    result[pos] = vec[i] where pos = (i*stride) % n.
    """
    n = len(vec)
    out = np.empty_like(vec)
    for i in range(n):
        pos = (i * stride) % n
        out[pos] = vec[i]
    # re‑normalise to guard against tiny numerical drift
    s = out.sum()
    return out / s if s > 0 else _uniform(n)

# ----------------------------------------------------------------------
# Mirror‑descent sub‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 rel_tol: float = 1e-3,
                                 max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on the true max‑convolution
    objective.  Gradient is taken from *all* convolution entries whose
    value lies within ``rel_tol`` of the current maximum (weighted
    uniformly).  The iterate stays on the simplex via multiplicative
    updates.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20          # base learning rate
    it = 0
    while it < max_iters:
        it += 1
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_obj = 2.0 * n * max_val
        if cur_obj < best_val:
            best_val = cur_obj
            best_seq = a.copy()

        # ---------- active set ----------
        threshold = max_val - rel_tol * max_val
        active_idxs = np.where(conv >= threshold)[0]
        if active_idxs.size == 0:
            # fallback: single max entry (should not really happen)
            active_idxs = np.array([np.argmax(conv)], dtype=int)

        # weight vector for active indices (uniform)
        w = np.zeros_like(conv)
        w[active_idxs] = 1.0 / active_idxs.size

        # gradient (central n entries of rev(a) * w)
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]         # centre n entries

        # multiplicative (exponentiated) update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * grad)

        # renormalise onto simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional re‑ordering (alternate order) – often helpful
        if it % 100_000 == 0:
            alt = _alternate_order(a)
            alt_val = _objective(alt)
            if alt_val < cur_obj:
                a = alt
                cur_obj = alt_val
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} new best {alt_val:.6f}")

        # periodic progress output (lightweight)
        if it % 250_000 == 0:
            print(f"[Mirror] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing (swap / mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 10_000_000,
                                 init_temp_factor: float = 0.03,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:  # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:  # redistribute a random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:  # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 40_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (swap‑only)
# ----------------------------------------------------------------------
def _greedy_permutation_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_swaps: int = 250_000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_val = _objective(a)

    for it in range(max_swaps):
        if time.time() > deadline:
            break

        i, j = rng.choice(n, size=2, replace=False)
        a[i], a[j] = a[j], a[i]                # try the swap
        val = _objective(a)
        if val < best_val:
            best_val = val
            # keep swap
        else:
            a[i], a[j] = a[j], a[i]            # revert

        if it % 50_000 == 0 and it > 0:
            print(f"[Greedy] swaps={it} best={best_val:.6f}")

    return a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that (approximately) minimises
    the evaluate_sequence objective within the supplied time budget.
    The best sequence is stored in ``height_sequence_1`` and also returned
    as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # Basic setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Length choice – 8000 gives a finer discretisation than 4000
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Retrieve previous solution, if any (to seed the pool)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – rich seed pool generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # Phase 2 – quick evaluation & keep the most promising seeds
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} seeds (exact objective values):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise global best
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – permutation sweep on the top seeds
    # ------------------------------------------------------------------
    print("\n--- Permutation sweep ---")
    # we will stop early if time becomes scarce
    perm_time_cutoff = deadline - 200.0   # reserve ~200 s for later phases

    for idx, base in enumerate(top_seeds, 1):
        if time.time() > perm_time_cutoff:
            print("Time budget low – stopping permutation sweep.")
            break

        # Work on the descending‑sorted version (most informative)
        base_desc = _sorted_desc(base)

        # Candidate permutations
        perms = []

        perms.append(base_desc)                       # plain sorted desc
        perms.append(_sorted_asc(base_desc))          # sorted asc
        perms.append(_alternate_order(base_desc))     # large‑small alternating

        # Several coprime‑stride permutations
        for _ in range(5):
            # pick a random stride coprime with n
            while True:
                stride = rng.integers(2, n - 2)
                if math.gcd(int(stride), n) == 1:
                    break
            perms.append(_permute_by_stride(base_desc, int(stride)))

        # reverse order (sometimes useful)
        perms.append(base_desc[::-1])

        # evaluate each permutation
        for cand in perms:
            val = _objective(cand)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = cand.copy()
                print(f"[Perm‑{idx}] new global best = {val:.6f}")

    print(f"\n[After perm] Best objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Mirror‑descent sub‑gradient optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        print("\n--- Mirror‑descent optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline - 100.0
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline - 30.0
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – Greedy permutation hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Greedy permutation optimisation ---")
        seq_greedy, val_greedy = _greedy_permutation_optimize(
            global_best_seq, n, deadline - 5.0
        )
        if val_greedy < global_best_val:
            global_best_seq = seq_greedy.copy()
            global_best_val = val_greedy
            print(f"[Greedy] new global best = {val_greedy:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check using the official evaluator (if available)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
