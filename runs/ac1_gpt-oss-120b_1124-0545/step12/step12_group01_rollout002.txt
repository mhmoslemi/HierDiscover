def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (maps Rⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(a: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)               # numpy’s modified Bessel function (approx.)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, projected onto the simplex."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.4, 0.7, 1.0, 1.5, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    for beta in (2, 5, 8, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian & double‑Gaussian families
    for sigma in np.linspace(0.12, 0.40, 5):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))

    # Power‑law grid
    for p in np.linspace(0.0, 4.0, 9):
        seeds.append(_power_law(n, p))

    # Beta‑family grid (coarse – enough diversity)
    for a in np.linspace(0.3, 4.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Random draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Convex mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.25, 0.5, 0.75):
        curve = _lowfreq_curve(n, 3, rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Continuation L‑BFGS on a smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Return a simplex‑projected vector after continuation L‑BFGS."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS phase.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 2000.0]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        # ---- smooth objective + gradient (log‑sum‑exp) -----------------
        def fun(v_local):
            a = _softmax(v_local)                     # probability simplex
            conv = _conv_fft(a, a)                     # full convolution
            max_c = np.max(conv)
            shift = max_c                               # for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smoothed objective (log‑sum‑exp)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # weights w_i = softmax(beta * conv_i)
            w = exp_beta / sum_exp

            # grad wrt a : 2 * conv(rev_a, w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # grad wrt logits via Jacobian of softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # ---- time‑aware callback ---------------------------------------
        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            # deadline hit inside optimiser – just stop the outer loop
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)

def _subgradient_opt(seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur = _objective_surrogate(a)
        if cur < best_val:
            best_val = cur
            best_seq = a.copy()
    return best_seq, best_val


def _expgradient_opt(seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur = _objective_surrogate(a)
        if cur < best_val:
            best_val = cur
            best_seq = a.copy()
    return best_seq, best_val

# Simple stochastic hill‑climbing in log‑softmax space
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 150_000,
                    sigma0: float = 0.35) -> tuple[np.ndarray, float]:
    """Randomised hill‑climbing on the smooth surrogate."""
    v = np.log(start_seq + 1e-30)               # start logits
    best_v = v.copy()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # linear cooling of the proposal width
        sigma = sigma0 * (1.0 - it / max_iters)
        if sigma <= 0.0:
            break
        v_new = v + sigma * np.random.randn(n)
        seq_new = _softmax(v_new)
        val_new = _objective_surrogate(seq_new)

        if val_new < best_val:
            best_val = val_new
            best_seq = seq_new.copy()
            best_v = v_new.copy()
            v = v_new          # move to the improvement
        else:
            # occasional acceptance of worse move (simulated‑annealing style)
            # temperature decays with it – we use a tiny probability to keep
            # the search alive.
            temp = sigma0 * 0.05 * (1.0 - it / max_iters)
            if temp > 0 and np.exp(-(val_new - best_val) / temp) > np.random.rand():
                v = v_new
    return best_seq, best_val

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a finer grid (returns ℓ₁‑normalised)."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq

    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(200_000, remaining * 0.05 * 1e5))
        if max_exp > 0:
            print("\n--- Exponentiated‑gradient fine‑tuning ---")
            seq_exp, val_exp = _expgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – optional up‑sampling to a finer grid
    # --------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 35.0:           # enough time for a meaningful up‑sample
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample(global_best_seq, new_n)

            max_iters_up = int(min(3000, time_left * 0.07 * 1e5))
            up_opt = _lbfgs_smooth_optimize(up_seq,
                                            new_n,
                                            deadline,
                                            max_iter_per_beta=max_iters_up)
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] New best after up‑sampling = {global_best_val:.6f}")

            # Final tiny sub‑gradient polish (if still time)
            if time.time() < deadline:
                rem = deadline - time.time()
                max_sub2 = int(min(120_000, rem * 0.04 * 1e5))
                if max_sub2 > 0:
                    seq_sub2, val_sub2 = _subgradient_opt(global_best_seq,
                                                          n,
                                                          deadline,
                                                          max_iters=max_sub2)
                    if val_sub2 < global_best_val:
                        global_best_seq = seq_sub2.copy()
                        global_best_val = val_sub2
                        print(f"[Final‑SubGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised:", exc)

    # store for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
