"""
Optimisation of a non‑negative step function for the provided
``evaluate_sequence``.  The approach combines three ideas:

1. **Diverse seeds** – analytical families (uniform, triangular, …) together
   with *very sparse* binary vectors whose pairwise sums appear only a few
   times (a relaxed “Sidon” construction).  The sparse vectors are built
   greedily while respecting a user‑defined bound on the maximum number of
   representations of any sum.

2. **Smooth‑max gradient descent** – minimise the surrogate
   ``2 * n * max(conv)`` using a differentiable approximation of the max
   (soft‑max with a temperature that grows during optimisation).  The
   gradient of the soft‑maxed convolution can be written as a convolution,
   so each iteration needs only a few FFTs and stays fast.

3. **Light‑weight simulated annealing** for final polishing.

The routine works entirely on the probability simplex (the sum of the
coefficients is always 1), which makes the scaling irrelevant for the
official ``evaluate_sequence``.  It respects the given time budget (default
1000 s) and stores the best sequence in the global variable
``height_sequence_1`` for warm‑starts in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective for ℓ₁‑normalised sequences (sum = 1).
    Exactly ``2 * n * max(conv)``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a modest pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12), np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous solution
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _binary_sparse_seed(
    n: int,
    max_unordered: int,
    rng: np.random.Generator,
    attempts: int = 30,
) -> np.ndarray:
    """
    Greedy construction of a binary vector of length ``n`` whose
    unordered pair‑sum multiplicities never exceed ``max_unordered``.
    Returns the best (largest) vector found after ``attempts`` trials.
    The vector is ℓ₁‑normalised (each non‑zero entry = 1/k).
    """
    best_vec = _uniform(n)  # fallback
    best_k = 0

    for _ in range(attempts):
        counts = np.zeros(2 * n - 1, dtype=np.int16)
        support = []
        # random permutation of indices
        order = rng.permutation(n)
        for idx in order:
            if not support:
                # first element always fits (self sum check)
                if counts[idx + idx] + 1 > max_unordered:
                    continue
                counts[idx + idx] += 1
                support.append(idx)
                continue

            # sums with already selected indices
            s = idx + np.array(support, dtype=np.int32)
            if np.any(counts[s] + 1 > max_unordered):
                continue
            if counts[idx + idx] + 1 > max_unordered:
                continue

            # accept the index
            counts[s] += 1
            counts[idx + idx] += 1
            support.append(idx)

        k = len(support)
        if k > best_k:
            best_k = k
            vec = np.zeros(n, dtype=np.float64)
            if k > 0:
                vec[support] = 1.0 / k
                best_vec = vec
            # early stop if we already have a huge support
            if best_k > int(0.6 * n):
                break

    return best_vec

def _smooth_max_opt(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    init_alpha: float = 0.5,
    max_alpha: float = 20.0,
    init_lr: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the *soft‑max* approximation of the max‑convolution.
    The temperature (alpha) is increased slowly so that the optimisation
    focuses on the true maximum towards the end.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    alpha = init_alpha

    while time.time() < deadline:
        it += 1
        # full convolution
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Smooth] iter={it:6d} best={best_val:.6f}")

        # soft‑max weights (numerically stable)
        c = conv - conv.max()
        exp_c = np.exp(alpha * c)
        p = exp_c / exp_c.sum()  # shape = len(conv) = 2n‑1

        # gradient = 2 * (p * a_rev) shifted
        a_rev = a[::-1]
        g_full = _conv_fft(p, a_rev)  # length = (2n-1)+n-1 = 3n-2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # step size schedule
        lr = init_lr / np.sqrt(it)

        a = a - lr * g
        a = _proj_simplex(a)

        # slowly raise the temperature
        if alpha < max_alpha:
            alpha *= 1.003  # gentle increase

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Light‑weight simulated‑annealing polishing that reshuffles a random
    subset of coordinates (Dirichlet move) and accepts moves according to a
    geometric temperature schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)  # size of reshuffled block
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (length ~4000) that minimises
    ``evaluate_sequence``.  The routine builds a rich seed pool (analytic
    families + low‑additive‑energy binary vectors), runs a smooth‑max
    gradient descent, and finishes with a short simulated‑annealing polish.
    The best sequence is stored in ``height_sequence_1`` for reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from previous run (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the candidate vector – 4000 gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation (analytic + sparse binary)
    # ------------------------------------------------------------------
    print(f"\n--- Collecting analytical seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)

    # sparse binary seeds with different max‑sum‑multiplicity caps
    print("\n--- Generating low‑additive‑energy binary seeds ---")
    for max_unordered, trials in [(8, 15), (10, 15), (12, 10)]:
        for _ in range(trials):
            vec = _binary_sparse_seed(n, max_unordered, rng, attempts=25)
            seeds.append(vec)

    print(f"Total seeds gathered: {len(seeds)}")

    # evaluate surrogate on all seeds
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {v:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max gradient descent (major part of the budget)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        opt_deadline = time.time() + 0.80 * time_left  # 80 % to optimiser
        print("\n--- Smooth‑max gradient descent ---")
        seq_opt, val_opt = _smooth_max_opt(
            global_best_seq, n, opt_deadline,
            init_alpha=0.5, max_alpha=20.0, init_lr=0.30
        )
        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt, val_opt
            print(f"[Smooth] new global best = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=800_000,
            init_temp_factor=0.04
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
