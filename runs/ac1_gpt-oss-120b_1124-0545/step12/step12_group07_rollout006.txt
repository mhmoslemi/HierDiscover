    max_c = conv.max()

def _adam_optimize(start_seq, n, deadline, total_iters, alpha_start, alpha_end):
    w = np.log(start_seq + 1e-12)  # initial logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_seq = None
    best_val = np.inf
    # compute expected number of iter from time left: maybe set max_iter huge, break by time.
    for t in range(1, total_iters+1):
        if time.time() > deadline:
            break
        a = np.exp(w - np.max(w))
        a /= a.sum()
        conv = _conv_fft(a, a)  # length 2n -1
        max_c = conv.max()
        # surrogate objective
        obj = 2 * n * max_c  # without denominator since sum=1
        if obj < best_val:
            best_val = obj
            best_seq = a.copy()
        
        # compute temperature for softmax weighting
        # naive schedule: alpha = alpha_start + (alpha_end - alpha_start) * min(1, t / total_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / total_iters)

        # compute weighted gradient w.r.t a
        g_a = _weighted_gradient(a, conv, alpha)  # shape n

        # gradient w.r.t w:
        avg = np.dot(g_a, a)  # scalar
        grad_w = a * (g_a - avg)  # elementwise

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.1  # base learning rate, can adapt
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional: continue
    return best_seq, best_val
```

"""
Optimisation of a non‑negative step function for the evaluation metric

    f(a) = 2·n·max(a*a) / (sum(a)**2)

The algorithm follows three stages:

1. **Broad seeding** – a rich library of analytic families (uniform,
   triangular, geometric, β‑shaped, Gaussian, …) together with a few random
   Dirichlet draws and the previous best solution (if any) are normalised to
   ℓ₁‑norm 1 and stored as seeds.

2. **Soft‑max‑Adam descent** – the sequence is re‑parameterised by a
   soft‑max (logits → probabilities) so the positivity and sum‑to‑one constraints
   are automatically satisfied.  The true objective (the max‑convolution)
   is smoothed by a temperature‑controlled soft‑max over all 2·n‑1 convolution
   entries.  The resulting smooth surrogate is differentiable; its gradient
   can be written as a *correlation* that is evaluated with a single FFT.
   Adam is applied to the logits, with the temperature rising from a gentle
   smoothing (α≈0.1) to a sharp focus on the true maximum (α≈10).  The best
   iterate seen during the whole run is kept.

3. **Light‑weight simulated‑annealing polish** – a final short SA phase
   perturbs a random block of coordinates via a Dirichlet move and accepts
   uphill moves with a geometric temperature schedule.

The routine respects the supplied time budget (default 1000 s) and stores the
final sequence in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N·log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_from_conv(n: int, conv: np.ndarray) -> float:
    """Surrogate objective when the ℓ₁‑norm of the vector equals 1."""
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Create a modest pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12), np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Optimiser: soft‑max Adam with temperature annealing
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max‑smoothed max‑convolution.
    The sequence is represented as a soft‑max of logits, guaranteeing
    non‑negativity and unit ℓ₁‑norm at every step.
    """
    # Initialise logits from the starting probability vector
    w = np.log(start_seq + 1e-12)  # shape (n,)

    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.1

    # Temperature schedule for the soft‑max over convolution entries
    alpha_start, alpha_end = 0.1, 12.0   # start very smooth, finish sharp
    max_iters = 200_000                # hard limit; the loop also respects the deadline

    best_seq = start_seq.copy()
    best_val = _objective_from_conv(n, _conv_fft(start_seq, start_seq))

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # ------- obtain current probability vector ----------
        a = np.exp(w - np.max(w))
        a /= a.sum()                     # guarantees a_i ≥ 0 and sum(a)=1

        # ------- compute convolution a * a -----------------
        conv = _conv_fft(a, a)            # length 2n‑1
        max_c = conv.max()
        obj = float(2.0 * n * max_c)

        # keep the best seen so far
        if obj < best_val:
            best_val = obj
            best_seq = a.copy()
            if t % 2000 == 0:
                print(f"[Adam] iter={t:6d}  best={best_val:.6f}")

        # ------- temperature (soft‑max smoothing) ----------
        # Use a smooth exponential ramp: α = α_start·(α_end/α_start)^{γ},
        # where γ = min(1, t / T_ramp) and T_ramp≈100 000.
        T_ramp = 100_000
        gamma = min(1.0, t / T_ramp)
        alpha = alpha_start * ((alpha_end / alpha_start) ** gamma)

        # ------- soft‑max weights over all convolution entries -----
        # weight_k ∝ exp(alpha·(conv_k - max_c))   (max_c subtraction avoids overflow)
        exp_term = np.exp(alpha * (conv - max_c))
        p = exp_term / exp_term.sum()               # shape (2n‑1,), ∑p = 1

        # ------- gradient of the weighted sum w.r.t. a ----------
        # g_a = 2 * correlation(p, a)   (see derivation in analysis)
        # correlation via convolution with the time‑reversed a
        corr = _conv_fft(p, a[::-1])                # length 3n‑2
        start = n - 1
        g_a = 2.0 * corr[start:start + n]           # shape (n,)

        # ------- gradient w.r.t. logits (chain rule) ----------
        avg = np.dot(g_a, a)                        # scalar ⟨g_a , a⟩
        grad_w = a * (g_a - avg)                    # shape (n,)

        # ------- Adam update ---------------------------------
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)

        # bias‑corrected estimates
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_seq, best_val

# Stochastic annealing polish (lightweight, same as baseline)

def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Polishing phase: randomly reshuffle a small block of coordinates
    via a Dirichlet move and accept according to a geometric temperature schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_from_conv(n, _conv_fft(cur, cur))

    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 13)          # size of the reshuffled block
        idx = rng.choice(n, size=k, replace=False)

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc

        new_val = _objective_from_conv(n, _conv_fft(new, new))

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA]  iter={it:7d}  best={best_val:.6f}")

    return best_seq, best_val

    # Stage 3 – final stochastic polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=800_000,
            init_temp_factor=0.04
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for hot‑restarts in the next run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
