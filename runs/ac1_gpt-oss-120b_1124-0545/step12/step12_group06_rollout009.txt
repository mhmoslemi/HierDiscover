from scipy.optimize import linprog
import numpy as np
import scipy.sparse as sp

def solve_lp_max_sum(f, M):
    n = len(f)
    m = 2*n-1
    # build A as CSR
    data = []
    rows = []
    cols = []
    for i in range(n):
        # rows i .. i+n-1 (inclusive)
        start = i          # row index
        end = i + n       # exclusive (i+n)
        rows.extend(range(start, end))
        cols.extend([i]*n)
        data.extend([f[i]]*n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n))
    # minimize -sum x => maximize sum
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(m, M), bounds=[(0, None)]*n,
                  method='highs', options={'presolve': True, 'maxiter': 1000})
    if not res.success:
        print('LP failed', res.message)
        return None
    return res.x
```

def mirror_descent(start_seq, deadline, max_iters=5_000_000):
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    lr0 = 0.2
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[Mirror] it={it} best={cur_val:.6f}")
        # compute gradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)
        # renorm
        s = a.sum()
        if s <= 0:
            a = _uniform(n)
        else:
            a = a / s
        # optional reordering
        if it % 100000 == 0:
            alt = _alternate_order(a)
            alt_val = _objective(alt)
            if alt_val < cur_val:
                a = alt
                cur_val = alt_val
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = alt.copy()
                    print(f"[Mirror-alt] it={it} improved to {alt_val:.6f}")
    return best_seq, best_val

def _gaussian(n, sigma_frac):

    old_n = len(old_seq)

    x_old = np.linspace(0, 1, old_n)

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(seq, seq)) / (sum(seq)²)

Algorithm summary
-----------------
1. Build a modest pool of analytic and random seed shapes (uniform,
   triangular, Gaussian, power‑law, Beta, Dirichlet draws, plus the
   previous solution if it exists) and keep the few best candidates.
2. Starting from the best seed run a small number of *LP‑updates*:
   given the current vector `f`, compute M = max(conv(f,f)).
   Solve the linear program

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k
                    g_i ≥ 0

   (the convolution constraints are linear in `g`).  Normalise the
   optimiser output to ℓ¹‑norm = 1, mix it with the current vector
   (`f ← (1‑t)·f + t·g`) and repeat a few times with a decreasing mixing
   weight `t`.  These steps aggressively push mass away from the
   convolution peak.
3. Polish the best vector with the exponentiated‑gradient (mirror)
   sub‑gradient method on the exact max‑convolution objective (the
   same routine used in the previous state‑of‑the‑art code).  This runs
   until the time budget is exhausted.
4. Finish with a cheap greedy permutation hill‑climb (swap‑only) that
   only accepts improving swaps.
5. Return the best sequence found and store it in the global variable
   ``height_sequence_1`` for possible reuse by a later run.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (ℓ¹‑normalised vectors):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple normalisation / ordering helpers
# ----------------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    s = v.sum()
    return v / s if s > 0 else v


def _sorted_desc(v: np.ndarray) -> np.ndarray:
    a = np.sort(v)[::-1]
    return _normalize(a)


def _sorted_asc(v: np.ndarray) -> np.ndarray:
    a = np.sort(v)
    return _normalize(a)


def _alternate_order(v: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …"""
    a = np.sort(v)  # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    return _normalize(out)


# ----------------------------------------------------------------------
# Analytic shape families (return ℓ¹‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    return _normalize(a)


# ----------------------------------------------------------------------
# Interpolation of a previous solution to a different length
# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(old_seq)
    if old_n == new_n:
        return _normalize(old_seq.astype(np.float64))
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, old_seq.astype(np.float64))
    return _normalize(new_seq)


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (maximise ℓ¹‑norm under convolution bound)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve:
        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k = 0 … 2n‑2
                    g_i ≥ 0.
    Returns the optimal g (may be all zeros) or None on failure.
    """
    n = len(f)
    m = 2 * n - 1

    # Build dense constraint matrix A where A[k,i] = f_i if i contributes to conv_k
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    # Linear programme (maximise → minimise –cᵀx)
    res = linprog(
        c=-np.ones(n),                # maximise Σ g_i
        A_ub=A,
        b_ub=np.full(m, M),
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 30.0}
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Mirror (exponentiated‑gradient) sub‑gradient optimiser
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                n: int,
                                deadline: float,
                                max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution objective.
    Returns the best vector seen and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20  # base learning rate

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative EG update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional re‑ordering – sometimes helpful
        if it % 100_000 == 0:
            alt = _alternate_order(a)
            alt_val = _objective(alt)
            if alt_val < cur_val:
                a = alt
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:7d} improved alt to {alt_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple greedy permutation optimiser (swap‑only hill‑climb)
# ----------------------------------------------------------------------
def _greedy_permutation_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_swaps: int = 200_000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_val = _objective(a)

    for it in range(max_swaps):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        a[i], a[j] = a[j], a[i]                # try the swap
        val = _objective(a)
        if val < best_val:
            best_val = val
            # keep the swap
        else:
            a[i], a[j] = a[j], a[i]            # revert

    return a, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _generate_initial_seeds(n: int,
                           rng: np.random.Generator,
                           previous_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Return a short list (≈5) of promising ℓ¹‑normalised vectors.
    """
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.20))
    seeds.append(_gaussian(n, 0.35))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 3.0))
    seeds.append(_beta_shape(n, 0.5, 0.5))

    # Dirichlet draws with a range of concentration parameters
    for conc in (0.01, 0.05, 0.2, 0.8, 2.0):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # include the previous solution if available (interpolate)
    if previous_seq is not None:
        seeds.append(_interpolate_to_length(previous_seq, n))

    # normalise (just in case) and evaluate
    seeds = [_normalize(s) for s in seeds]
    evaluated = [(s, _objective(s)) for s in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    return [vec for vec, _ in evaluated[:top_k]]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    The best sequence is stored in the global name ``height_sequence_1``.
    """
    import numpy as np
    import time

    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0               # safety margin

    # ------------------------------------------------------------------
    # Problem size – 2000 gives a good trade‑off between resolution
    # and speed for the LP/FFT steps.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Retrieve previous solution, if any.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 0 – obtain a few good seed vectors
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length={n}) ---")
    seed_pool = _generate_initial_seeds(n, rng, previous)
    best_seq = seed_pool[0].copy()
    best_val = _objective(best_seq)
    print(f"[Init] best seed objective ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP‑based peak‑reduction iterations
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    lp_budget = remaining * 0.35          # ~35 % of the remaining time
    lp_deadline = time.time() + lp_budget
    mix_weight = 0.20
    lp_iter = 0

    while time.time() < lp_deadline:
        lp_iter += 1
        # current max‑convolution value
        conv_ff = _conv_fft(best_seq, best_seq)
        M = conv_ff.max()

        # solve the LP (maximise ℓ¹‑norm under the same convolution bound)
        g0 = _solve_lp_max_sum(best_seq, M)
        if g0 is None:
            print("[LP] linear programme failed – exiting LP loop.")
            break
        if g0.sum() < 1e-12:
            print("[LP] LP returned near‑zero sum – exiting LP loop.")
            break

        # normalise the LP solution to ℓ¹‑norm = 1
        g = _normalize(g0)

        # convex combine and re‑normalise
        best_seq = (1.0 - mix_weight) * best_seq + mix_weight * g
        best_seq = _normalize(best_seq)

        # evaluate
        cur_val = _objective(best_seq)
        if cur_val < best_val:
            best_val = cur_val
            print(f"[LP] iter {lp_iter:3d} improved value = {cur_val:.6f}")

        # slowly decay the mixing weight
        mix_weight = max(0.02, mix_weight * 0.95)

        # occasionally test an alternating ordering (cheap cheap)
        if lp_iter % 5 == 0:
            alt = _alternate_order(best_seq)
            alt_val = _objective(alt)
            if alt_val < best_val:
                best_seq = alt
                best_val = alt_val
                print(f"[LP] alternating order improved to {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Mirror‑descent sub‑gradient polishing (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent polishing ---")
        best_seq, best_val = _mirror_subgradient_optimize(
            best_seq, n, deadline
        )
        print(f"[Mirror] polished best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final greedy swap hill‑climb (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Greedy permutation refinement ---")
        seq_perm, val_perm = _greedy_permutation_optimize(
            best_seq, n, deadline
        )
        if val_perm < best_val:
            best_seq = seq_perm
            best_val = val_perm
            print(f"[Greedy] improved to {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("[Eval] evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
