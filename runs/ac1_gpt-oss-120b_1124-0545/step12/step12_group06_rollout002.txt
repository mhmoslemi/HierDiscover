"""
Propose a step‑function (non‑negative sequence) that minimizes
the ``evaluate_sequence`` objective.

The search consists of four stages:

1. **Rich seed pool** – analytic windows, Dirichlet draws, low‑frequency
   random curves and the previous best solution (if any).  All seeds are
   ℓ₁‑normalised (sum = 1).

2. **Adam on a log‑sum‑exp surrogate** – for each of the top‑k seeds we run
   Adam in the log‑softmax parametrisation.  The surrogate smoothly
   approximates the max‑convolution; the temperature is increased
   gradually up to a large value (default 500) to focus on the true maximum.

3. **Multiplicative (mirror) sub‑gradient descent** on the exact max‑convolution.
   This stage receives the remaining time budget and performs many
   inexpensive iterations (FFT based).  Periodically we try a few
   re‑orderings (descending, alternating) which often improve the bound.

4. **Local greedy polishing** – a very cheap hill‑climb that attempts
   swaps, pair‑transfers and small redistributions while keeping the
   sequence on the simplex.

The best sequence found is stored globally as ``height_sequence_1`` and
returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for an ℓ₁‑normalised sequence:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Ordering helpers

def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Largest, smallest, 2nd‑largest, 2nd‑smallest, …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse list of ℓ₁‑normalised seed vectors."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_tukey_window(n, 0.5))

    # sine windows and their alternating orderings

        seeds.append(w)

        g = _gaussian(n, sigma)
        seeds.append(g)
        seeds.append(_alternate_order(g))

    # power laws
    for p in np.linspace(0.0, 8.0, 17):
        pw = _power(n, p)
        seeds.append(pw)
        seeds.append(_alternate_order(pw))

    # beta family
    for _ in range(15):
        alpha = 10 ** rng.uniform(-2, 0.6)
        beta = 10 ** rng.uniform(-2, 0.6)
        b = _beta_shape(n, alpha, beta)
        seeds.append(b)
        seeds.append(_alternate_order(b))

    # exponentials
    for rate in np.logspace(-4, -1, 12):
        e = _exponential(n, rate)
        seeds.append(e)
        seeds.append(_alternate_order(e))

        l = _logarithmic(n, eps)
        seeds.append(l)
        seeds.append(_alternate_order(l))

    # Dirichlet draws with many concentrations
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(8):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            cur = _lowfreq_curve(n, nf, rng)
            seeds.append(cur)
            seeds.append(_alternate_order(cur))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            cur = _lowfreq_curve(n, nf, rng)
            mixed = _mix_shapes(_uniform(n), cur, mix_alpha)
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Random permutations of a few deterministic shapes
    for base in [_gaussian(n, 0.2), _power(n, 3.0), _beta_shape(n, 0.5, 0.5)]:
        for _ in range(5):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    # If a previous solution exists, add a lightly jittered copy
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())
            jitter = avg + 0.001 * rng.standard_normal(size=n)
            jitter = np.maximum(jitter, 0.0)
            jitter /= jitter.sum()
            seeds.append(jitter)

    return seeds

# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best ℓ₁‑normalised vector observed before the deadline.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.9

    beta = 0.5
    beta_max = 500.0            # final temperature (very smooth)
    beta_mul = 1.001            # geometric increase each iteration

    best_seq = None
    best_val = float("inf")
    it_print = 100_000

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                 # stays on the simplex, sum = 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % it_print == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # --- smoothed max via log‑sum‑exp ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # chain rule for softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature (smoothness)
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 15_000_000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent on the exact
    max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.25  # base learning rate

    it_print = 50_000
    alt_print = 200_000

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % it_print == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # Occasionally try a different ordering – can open new descent directions
        if it % alt_print == 0:
            for reorder in (_sorted_desc, _alternate_order):
                a_alt = reorder(a)
                val_alt = _objective(a_alt)
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a_alt.copy()
                    a = a_alt
                    print(f"[Mirror‑alt] it={it:7d} improved to {val_alt:.6f}")

    return best_seq, best_val


def _local_search_optimize(start_seq: np.ndarray, n: int,
                           deadline: float,
                           max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Very cheap greedy hill‑climbing: swaps, pair‑transfers and small redistributions.
    Accepts a move only if it strictly improves the exact objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    it_print = 100_000

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        move_type = rng.integers(0, 3)
        if move_type == 0:                 # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:               # pair‑transfer (split mass of two entries)
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:                               # small redistribution among a random subset
            k = rng.integers(2, 8)          # subset size 2‑7
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc

        new_val = _objective(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % it_print == 0:
                    print(f"[Local] it={it:7d} new best={new_val:.6f}")

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the supplied time budget.

    Returns the best sequence (list of floats) and stores it globally as
    ``height_sequence_1`` for possible reuse in later runs.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0               # safety margin

    # ------------------------------------------------------------------
    # Length – 4000 is a good compromise (FFT size = 8192)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution, if any.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – seed generation and quick shortlisting
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives (exact):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the short list (≈10 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_budget = remaining * 0.12
    adam_deadline = time.time() + adam_budget

    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > adam_deadline:
            break
        per_deadline = time.time() + (adam_deadline - time.time()) / max(1, len(best_seeds) - idx + 1)
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, per_deadline)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # test an alternating re‑ordering of the Adam result
        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq = alt
                global_best_val = alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Mirror‑subgradient descent (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑subgradient descent (exact max) ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

        # Try a few deterministic reorderings after the mirror pass
        for reorder in (_sorted_desc, _sorted_asc, _alternate_order):
            cand = reorder(global_best_seq)
            cand_val = _objective(cand)
            if cand_val < global_best_val:
                global_best_seq = cand
                global_best_val = cand_val
                print(f"[Reorder] improved to {cand_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Local greedy polishing (fast hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Local greedy polishing ---")
        seq_local, val_local = _local_search_optimize(
            global_best_seq, n, deadline
        )
        if val_local < global_best_val:
            global_best_seq = seq_local.copy()
            global_best_val = val_local
            print(f"[overall‑Local] new global best = {val_local:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
