def project_simplex(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Implementation of mirror descent:

def mirror_descent(a0, n, deadline, lr0=0.3, power=0.35, max_iters=500000):
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    t=0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _conv_fft(a, a)
        k = np.argmax(conv)
        cur_val = 2.0 * n * conv[k]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # gradient
        g = np.zeros_like(a)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]
        # step-size
        eta = lr0 / ((t) ** power)
        a_new = a - eta * g
        # project
        a_new = project_simplex(a_new)
        # ensure nonneg ok
        a = a_new
    return best_seq, best_val

def _kaiser_window(n, beta):
    i = np.arange(n)
    alpha = (n - 1) / 2
    denominator = scipy.special.i0(beta)
    args = beta * np.sqrt(1 - ((i - alpha) / alpha)**2)
    a = scipy.special.i0(args) / denominator
    s = a.sum()

"""
propose_candidate – convex‐optimization / stochastic search for the
minimal‐max‑auto‑correlation step function.

The algorithm proceeds in four stages:

1. **Rich seed pool** – a broad collection of analytic windows
   (uniform, triangular, Hann, Kaiser, Blackman, Nuttall, …) together
   with many Dirichlet draws.  All vectors are ℓ₁‑normalised.

2. **Surrogate ranking** – fast evaluation of the scale‑invariant
   surrogate 2·n·max(conv) (identical ordering to the official target)
   keeps the best ~10 seeds.

3. **Adam optimisation on a smooth‑max surrogate** – each top seed is
   refined by minimising a log‑sum‑exp approximation to the maximum.
   The temperature β is increased (10 → 30 → 100 → 300 → 600 → 1000)
   while an Adam optimiser updates the soft‑max logits.  The best
   refined vector is remembered.

4. **Exact sub‑gradient polishing** – a projected‑gradient (mirror‑descent)
   loop on the true max‑convolution, with a decaying step‑size, followed
   by a light simulated‑annealing pass (swap, Dirichlet‑redistribute,
   pair‑transfer, Gaussian jitter).  The final vector is returned and
   stored globally as ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from scipy.special import i0  # Kaiser window
import scipy.optimize

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical edge case – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1.0)))
        elif i <= (n - 1) * (1.0 - alpha / 2.0):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2.0 / alpha + 1.0)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinc_squared(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = np.where(i == 0, 1.0, np.sin(np.pi * i / n) / (np.pi * i / n))
    a = x ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine_power(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – beta roughly controls side‑lobe level."""
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    arg = beta * np.sqrt(1 - ((i - alpha) / alpha) ** 2)
    a = i0(arg) / i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    a0, a1, a2, a3 = 0.355768, 0.487396, 0.144232, 0.012604
    i = np.arange(n, dtype=np.float64)
    a = (
        a0
        - a1 * np.cos(2.0 * np.pi * i / (n - 1))
        + a2 * np.cos(4.0 * np.pi * i / (n - 1))
        - a3 * np.cos(6.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)

# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) – assumes seq ℓ₁‑normalised."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_max, grad_a) where
    smooth_max = log‑sum‑exp_beta(conv)/beta
    and grad_a = ∂/∂a smooth_max.
    """
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    beta_sched: list[float],
    iters_per_beta: int,
) -> tuple[np.ndarray, float]:
    """Adam on log‑sum‑exp surrogate (softmax of the convolution)."""
    eps = 1e-12
    v = np.log(start_seq + eps)  # logits, unconstrained
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    t = 0
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_sched:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return best_seq, best_val
            t += 1
            a = _softmax(v)
            smooth, grad_a = _smooth_max_and_grad(a, beta)
            loss = 2.0 * n * smooth

            # gradient w.r.t logits (chain rule through softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            lr = 0.1  # fixed base learning rate
            v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

            # keep track of the best surrogate seen so far
            a_curr = _softmax(v)
            val_curr = _objective(a_curr)
            if val_curr < best_val:
                best_val = val_curr
                best_seq = a_curr.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exact sub‑gradient) with projection onto the simplex
# ----------------------------------------------------------------------
def _mirror_descent_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    lr0: float = 0.2,
    power: float = 0.35,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:,} best={cur_val:.6f}")

        # sub‑gradient for the active shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]

        # step‑size (multiplicative decay)
        eta = lr0 / (it ** power)
        a_new = a - eta * g
        a = _project_simplex(a_new)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑Annealing polishing (light but diverse moves)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    init_temp_factor: float = 0.05,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 4)  # added a Gaussian jitter move
        if move_type == 0:  # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:  # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            tot = cur[idx].sum()
            if tot <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
            new = cur.copy()
            new[idx] = new_alloc
        elif move_type == 2:  # pair transfer
            i, j = rng.choice(n, size=2, replace=False)
            tot = cur[i] + cur[j]
            if tot <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * tot
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:  # Gaussian jitter on a random subset
            k = rng.integers(3, 15)
            idx = rng.choice(n, size=k, replace=False)
            pert = rng.normal(scale=0.05, size=k) * cur[idx]
            new = cur.copy()
            new[idx] = np.maximum(new[idx] + pert, 0.0)
            # renormalise the perturbed part only, then globally project
            new = _project_simplex(new)

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 250_000 == 0:
                    print(f"[SA] it={it:,} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed aggregation – many analytic families + random draws
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))
    seeds.append(_sinc_squared(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    # Kaiser window – medium resolution
    for beta in np.linspace(2.0, 12.0, 9):
        seeds.append(_kaiser_window(n, beta))

    # sine windows (dense gamma grid)
    for gamma in np.linspace(0.3, 4.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian family
    for sigma in np.linspace(0.08, 0.45, 10):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Power‑law family
    for p in np.linspace(0.0, 4.0, 9):
        seeds.append(_power(n, p))

    # Beta shape family (wide parameter sweep)
    alphas = np.concatenate((np.linspace(0.05, 0.3, 4), np.linspace(0.3, 3.0, 7)))
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.05, 0.95, 9):
        seeds.append(_tukey_window(n, alpha))

    # Exponential decay family
    for lam in np.linspace(0.0005, 0.015, 8):
        seeds.append(_exponential_decay(n, lam))

    # Raised‑cosine‑power windows
    for power in [0.5, 1.0, 1.5, 2.0, 3.0]:
        seeds.append(_raised_cosine_power(n, power))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending
            # alternating high‑low pattern
            a = np.sort(raw)
            lo, hi = 0, len(a) - 1
            alt = np.empty_like(a)
            pos = 0
            while lo <= hi:
                if pos % 2 == 0:
                    alt[pos] = a[hi]
                    hi -= 1
                else:
                    alt[pos] = a[lo]
                    lo += 1
                pos += 1
            seeds.append(alt)

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_uniform(n) * (1 - mix_alpha) + curve * mix_alpha)

    # Include previous solution if available
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin for final eval

    # Problem size – 4000 offers a good trade‑off between quality and speed
    n = 4000

    # Retrieve a previous solution (if any) for seeding
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich pool of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Fast surrogate evaluation, keep the best few
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for _, vec in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, v) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {v:.6f}")

    best_seq = top_seeds[0].copy()
    best_val = _objective(best_seq)
    print(f"\n[Init] Best seed surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on smooth‑max surrogate
    # ------------------------------------------------------------------
    beta_schedule = [10.0, 30.0, 100.0, 300.0, 600.0, 1000.0]
    iters_per_beta = 250  # modest but enough for convergence

    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() > deadline:
            break
        print(f"\n[Adam] Optimising seed #{idx}")
        seq_opt, val_opt = _adam_optimize(
            seed_vec,
            n,
            deadline,
            beta_sched=beta_schedule,
            iters_per_beta=iters_per_beta,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"[Adam] New global best = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Exact sub‑gradient (mirror descent) polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent polishing ---")
        seq_mirror, val_mirror = _mirror_descent_optimize(
            best_seq,
            n,
            deadline,
            lr0=0.2,
            power=0.35,
            max_iters=2_000_000,
        )
        if val_mirror < best_val:
            best_val = val_mirror
            best_seq = seq_mirror.copy()
            print(f"[Mirror] New global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            init_temp_factor=0.05,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] New global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (informational)
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # expose for possible reuse by subsequent runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
