"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the ordinary linear convolution.  The objective is
scale‑invariant, therefore we work with ℓ₁‑normalised vectors and
optimise ``2·n·max(conv)``.

The search combines several complementary ideas:

1. **Rich seed pool** – analytic families (uniform, triangular, beta,
   Gaussian, exponential, arcsine, double‑exponential, two‑Gaussian
   mixture), Dirichlet draws with many concentrations, low‑frequency
   random curves and sparse random spikes.  All seeds are ℓ₁‑normalised
   and a few orderings (sorted, alternating) are added.

2. **Screening** – exact objective is computed for all seeds and the
   best few (``top_k``) are kept.

3. **Adam on a log‑softmax surrogate** – we minimise the
   log‑sum‑exp smoothed version of the max‑convolution.  A schedule for
   the smoothing temperature ``beta`` slowly raises the approximation
   to the true max.  Each kept seed receives a short Adam run; the best
   results are retained.

4. **Mirror (exponentiated‑gradient) sub‑gradient descent** – direct
   optimisation of the exact max‑convolution objective using the
   sub‑gradient at the active index.  Adaptive step‑size and occasional
   re‑ordering (alternating order) help escape plateaus.

5. **Simulated annealing polishing** – swap, mass‑redistribution,
   pair‑transfer and block‑shift moves with geometric cooling.

6. **Final Adam fine‑tuning** – a longer Adam run on the best vector
   found so far, using a high temperature to focus on the true
   max‑convolution.

7. **Greedy permutation hill‑climb** – simple swap‑only hill climbing
   for a last polish.

All phases respect the global time budget (default 1000 s).  The best
feasible sequence is stored in the global name ``height_sequence_1`` and
returned as a plain list of ``float``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for an ℓ₁‑normalised sequence:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    """Laplace‑type shape, peak at the centre and exponential tails."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _two_gaussian(n: int, sigma_frac: float, offset_frac: float) -> np.ndarray:
    """Mixture of two Gaussians placed symmetrically."""
    sigma = sigma_frac * n
    offset = offset_frac * n
    mu1 = (n - 1) / 2.0 - offset
    mu2 = (n - 1) / 2.0 + offset
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    """Beta(0.5, 0.5) – arcsine distribution."""
    return _beta_shape(n, 0.5, 0.5)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k random support points with a Dirichlet draw on the support."""
    idx = rng.choice(n, size=min(k, n), replace=False)
    raw = rng.dirichlet(np.ones(len(idx)))
    out = np.zeros(n, dtype=np.float64)
    out[idx] = raw
    return out / out.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Large, small, 2nd‑large, 2nd‑small, …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 3_000_000,
                   beta_max: float = 500.0) -> tuple[np.ndarray | None, float]:
    """
    Adam on log‑softmax parameters using a log‑sum‑exp surrogate.
    Returns the best ℓ₁‑normalised vector seen before the deadline.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits (unconstrained)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 0.5
    beta_inc = 0.5

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:8d} best={cur_val:.6f}")

        # ----- smoothed max (log‑sum‑exp) -----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # the central n entries

        # softmax chain‑rule
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 15_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
    max‑convolution objective.  Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.30  # a slightly larger base step

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:8d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise onto simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional re‑ordering (alternating) – often beneficial
        if it % 200_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
                    print(f"[Mirror‑alt] it={it:8d} improved alt to {val_alt:.6f}")

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 12_000_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with several move types.  Returns the best
    sequence seen.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 4)
        if move_type == 0:                 # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:               # redistribute a small random subset
            k = rng.integers(2, 8)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        elif move_type == 2:               # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:                               # block shift (move a contiguous chunk)
            block_len = rng.integers(2, 12)
            start = rng.integers(0, n - block_len + 1)
            dest = rng.integers(0, n - block_len + 1)
            if dest == start:
                continue
            new = cur.copy()
            chunk = new[start:start + block_len].copy()
            new[start:start + block_len] = 0.0
            # insert the mass at destination (simple additive placement)
            new[dest:dest + block_len] += chunk
            # renormalise (to avoid tiny rounding drift)
            new = new / new.sum()

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 75_000 == 0:
                    print(f"[SA] it={it:8d} best={new_val:.6f}")

    return best_seq, best_val


def _greedy_permutation_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_swaps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climbing that only accepts improving swaps.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_val = _objective(a)

    for it in range(max_swaps):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        a[i], a[j] = a[j], a[i]                # try swap
        val = _objective(a)
        if val < best_val:
            best_val = val
            # keep swap
        else:
            a[i], a[j] = a[j], a[i]            # revert
    return a, best_val


def _final_adam_finetune(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 5_000_000,
                         beta_target: float = 1200.0) -> tuple[np.ndarray, float]:
    """
    A longer Adam run on the smoothed max‑convolution, starting from the
    best vector obtained so far.  The temperature ``beta`` is taken to a
    high value (default 1200) so that the surrogate is very close to the
    true max.
    """
    eps = 1e-30
    v = np.log(start_seq + eps)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 5.0
    beta_inc = (beta_target - beta) / max_iters

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Finetune‑Adam] it={it:8d} best={cur_val:.6f}")

        # smoothed max gradient
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # temperature increase
        beta = min(beta + beta_inc, beta_target)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence (list of floats) and stores it globally as
    ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Choose a length that is a power of two (fast FFT) – 4096 is a good
    # trade‑off between resolution and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Retrieve a previously found solution, if present.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – create a diverse seed pool and keep the most promising.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 16
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} exact objective values after screening:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i:2d}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the top seeds.
    # Allocate roughly 25 % of the remaining time to Adam.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_budget = remaining * 0.25
    adam_deadline = time.time() + adam_budget

    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > adam_deadline:
            break
        per_seed_deadline = time.time() + (adam_deadline - time.time()) / max(1, len(best_seeds) - idx + 1)
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, per_seed_deadline)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # try alternating ordering of the Adam result
        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq = alt
                global_best_val = alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Mirror‑descent sub‑gradient optimisation (exact max).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final Adam fine‑tuning with a high temperature.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Adam fine‑tuning (high β) ---")
        seq_fine, val_fine = _final_adam_finetune(
            global_best_seq, n, deadline
        )
        if val_fine < global_best_val:
            global_best_seq = seq_fine.copy()
            global_best_val = val_fine
            print(f"[overall‑Fine] new global best = {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – Greedy permutation hill‑climb (swap‑only).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Greedy permutation optimisation ---")
        seq_perm, val_perm = _greedy_permutation_optimize(
            global_best_seq, n, deadline
        )
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[overall‑Perm] new global best = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
