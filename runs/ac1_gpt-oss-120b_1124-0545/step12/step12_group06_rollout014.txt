"""
Optimisation of a non‑negative step function for the objective
    2·n·max(conv(seq,seq)) / (sum(seq)²)

Approach
--------
1. Build a modest but diverse pool of ℓ₁‑normalised seed vectors
   (uniform, power‑law, beta, Dirichlet draws, low‑frequency random curves
   and a few deterministic shapes).  For each seed also store the
   descending, ascending and “alternating‑extremes’’ permutations,
   because ordering strongly influences the convolution peak.

2. Pick the best few seeds (by the exact objective) and run a long
   simulated‑annealing (SA) phase on each of them.  SA moves are:
   * swap two positions,
   * reverse a random sub‑array,
   * redistribute mass over a randomly chosen small subset
     (Dirichlet draw, total mass preserved).
   The temperature is geometric (≈0.9999 decay) and the acceptance
   criterion is Metropolis on the true objective.  The current best
   sequence is kept globally.

3. After the SA budget is exhausted, a few passes of a simple
   exponentiated‑gradient (mirror‑subgradient) optimiser are applied
   to the global best – this refines the solution in the neighbourhood
   of a good point.

4. Finally a greedy permutation hill‑climb (swap‑only) cleans up the
   remaining tiny improvements.

All steps respect the global time budget (default 1000 s) and the
resulting sequence is stored in the global name ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a sequence (sum(seq) assumed to be 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few cosine terms, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------
def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Largest, smallest, 2nd‑largest, 2nd‑smallest …"""
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # 1. Simple analytic families
    seeds.append(_uniform(n))
    for p in np.linspace(0.0, 8.0, 9):
        seeds.append(_power(n, p))

    # 2. Beta family (random parameters)
    for _ in range(12):
        alpha = 10 ** rng.uniform(-2, 0.6)
        beta = 10 ** rng.uniform(-2, 0.6)
        seeds.append(_beta_shape(n, alpha, beta))

    # 3. Dirichlet draws with a few concentrations
    for conc in (0.01, 0.05, 0.1, 0.5, 1.0):
        for _ in range(4):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))

    # 4. Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(curve)
            seeds.append(_alternate_order(curve))

    # 5. Random permutations of a few deterministic shapes
    for base in [_power(n, 2.0), _beta_shape(n, 0.5, 0.5), _beta_shape(n, 2.0, 2.0)]:
        for _ in range(4):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    # 6. Previous best (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    # 7. Apply simple orderings to all seeds (desc/asc/alternate)
    enriched = []
    for s in seeds:
        enriched.append(s)                # as‑is (already ℓ₁‑normalised)
        enriched.append(_sorted_desc(s))
        enriched.append(_sorted_asc(s))
        enriched.append(_alternate_order(s))
    # deduplicate a little – convert to list of unique tuples (cheap)
    uniq = []
    seen = set()
    for v in enriched:
        key = tuple(np.round(v, 12))
        if key not in seen:
            seen.add(key)
            uniq.append(v)
    return uniq


# ----------------------------------------------------------------------
# Simulated annealing optimiser
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray, n: int,
                         deadline: float,
                         init_temp_factor: float = 0.05,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    SA with three move types (swap, sub‑array reverse, small‑subset redistribution).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while time.time() < deadline:
        it += 1
        T = max(T * 0.9999, min_temp)      # slow geometric cooling

        move_type = rng.integers(0, 3)
        if move_type == 0:                 # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:               # reverse a random sub‑array
            lo = rng.integers(0, n - 2)
            hi = rng.integers(lo + 1, n)
            new = cur.copy()
            new[lo:hi] = new[lo:hi][::-1]
        else:                               # redistribute a small random subset
            k = rng.integers(2, 8)          # subset size
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] iter={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror (exponentiated‑gradient) sub‑gradient optimiser (exact max)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.20

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:,} best={cur_val:.6f}")

        # sub‑gradient w.r.t. the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Greedy permutation hill‑climb (swap‑only)
# ----------------------------------------------------------------------
def _greedy_permutation_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_swaps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly try a random swap; keep it only if it improves the objective.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_val = _objective(a)

    for it in range(max_swaps):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        a[i], a[j] = a[j], a[i]
        val = _objective(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]   # revert
    return a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine builds a diverse seed pool,
    runs a long simulated‑annealing phase on the best seeds,
    refines the result with mirror‑subgradient descent, finishes with a
    greedy swap hill‑climb and finally returns the best sequence found.
    The best sequence is also stored in the global variable
    ``height_sequence_1`` for reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Length of the step‑function – 3000 is a good trade‑off (FFT size 8192)
    # ------------------------------------------------------------------
    n = 3000

    # ------------------------------------------------------------------
    # Retrieve previous solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds (length = {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # quick evaluation of all seeds (exact objective)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("Top seed objectives:")
    for i, (_, v) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {v:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Current best objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Simulated‑annealing polishing – main time consumer
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate ~60 % of the remaining budget to SA, split equally among the seeds
    sa_budget = remaining * 0.60
    per_seed_budget = sa_budget / max(1, len(best_seeds))
    print(f"\n--- Simulated annealing phase (≈{int(sa_budget)} s) ---")
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seed_deadline = time.time() + per_seed_budget
        print(f"\n[SA] start {idx}/{len(best_seeds)} (budget ≈{int(per_seed_budget)} s)")
        seq_sa, val_sa = _simulated_annealing(seed_vec, n, seed_deadline)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA‑global] new best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Mirror‑subgradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑subgradient refinement ---")
        seq_mir, val_mir = _mirror_subgradient_optimize(
            global_best_seq, n, deadline
        )
        if val_mir < global_best_val:
            global_best_seq = seq_mir.copy()
            global_best_val = val_mir
            print(f"[Mirror‑global] new best = {val_mir:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final greedy permutation hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Greedy permutation hill‑climb ---")
        seq_greedy, val_greedy = _greedy_permutation_optimize(
            global_best_seq, n, deadline
        )
        if val_greedy < global_best_val:
            global_best_seq = seq_greedy.copy()
            global_best_val = val_greedy
            print(f"[Greedy‑global] new best = {val_greedy:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:
        print("[Eval] evaluate_sequence raised:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
