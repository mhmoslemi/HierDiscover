def _biased_move(a, rng, move_frac=0.1):

    # Choose number of coords to modify
    k = rng.integers(2, 15)
    # Choose indices randomly but encourage high and low selections
    # We'll select k indices with probability proportional to a (high values) for removal.
    probs = a / a.sum()

def _adam_subgradient(start_seq, n, deadline, top_k=20, init_lr=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    a = start_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_seq = a.copy()
    best_val = _objective(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        # get indices where conv is close to max
        max_val = conv.max()
        # select top_k
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        # compute gradient sum over selected indices
        g = np.zeros_like(a)
        for k in idxs:
            rev = k - np.arange(n)
            mask = (rev >= 0) & (rev < n)
            g[mask] += 2.0 * a[rev[mask]]
        g /= float(len(idxs))
        # Adam update
        lr = init_lr / np.sqrt(t)   # maybe not needed with Adam; keep constant but schedule optional
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _proj_simplex(a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # optional progress print
            if t % 10000 == 0:
                print(f"[Adam] it={t} best={best_val:.6f}")
    return best_seq, best_val

"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.  The algorithm expands the seed pool,
uses a fast Adam‑style sub‑gradient optimiser on the max‑convolution,
and finishes with a biased simulated‑annealing that moves mass away
from entries that contribute most to the current maximum.  All
operations respect the probability‑simplex (∑a_i = 1, a_i ≥ 0) so the
evaluation function’s scaling invariance is preserved.

Key ideas
---------

* **Richer seed set** – in addition to the analytic families we generate
  random convex combinations of a few seeds, extra Dirichlet draws with
  very small concentration and extreme β‑shapes (α,β≈0.05).  This gives
  a much more diverse starting population.

* **Adam sub‑gradient descent** – instead of a plain diminishing‑step‑size
  sub‑gradient we keep first‑ and second‑moment estimates (β₁=0.9,
  β₂=0.999) and adapt the learning rate per coordinate.  The gradient is
  the average of the sub‑gradients for the *top‑k* convolution entries
  (k≈20) which provides a smoother descent direction.

* **Biased simulated annealing** – a move consists of taking a small
  amount of probability mass from the currently “heavy’’ coordinates
  (probability proportional to a_i) and redistributing it to the
  “light’’ ones (probability proportional to 1‑a_i).  The temperature
  schedule is slower (multiplicative factor 0.9995) and the move set
  size varies between 2 and 15, allowing both fine and coarse
  perturbations.

The routine respects the supplied time budget (default 1000 s), stores
the best sequence in the global name ``height_sequence_1`` and returns
it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective for ℓ₁‑normalised sequences (∑a = 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a generous pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for wf in np.linspace(0.05, 0.35, 7):
        seeds.append(_trapezoidal(n, width_fraction=wf))
    for gamma in np.linspace(0.0, 4.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))
    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    # geometric (both sides)
    r_vals = np.concatenate([np.linspace(0.2, 0.98, 12), np.linspace(0.98, 0.9995, 8)])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # power law
    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    # beta shapes – exhaustive grid plus some extreme values
    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))
    for a in (0.05, 0.1, 0.15):
        for b in (0.05, 0.1, 0.15):
            seeds.append(_beta_shape(n, a, b))

    # gaussians
    for sigma in (0.12, 0.20, 0.28, 0.36):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # extra Dirichlet with very small concentration (highly spiky)
    for conc in (0.001, 0.002, 0.005):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # random convex combinations of a few existing seeds
    for _ in range(20):
        idxs = rng.choice(len(seeds), size=3, replace=False)
        weights = rng.dirichlet(np.full(3, 1.0))
        mixed = sum(weights[i] * seeds[idxs[i]] for i in range(3))
        mixed /= mixed.sum()
        seeds.append(mixed)

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _exp_mirror_descent(start_vec: np.ndarray, n: int, deadline: float,
                        max_iters: int = 2_000_000,
                        eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropic) mirror descent on the max‑convolution.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 15_000 == 0:
                print(f"[Mirror] it={it:7d} best={best_val:.6f}")

        # sub‑gradient for the active shift
        rev = max_idx - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        eta = eta0 / np.sqrt(it)          # decreasing learning rate
        a = a * np.exp(-eta * g)          # entropic step
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _adam_subgradient(start_seq: np.ndarray, n: int, deadline: float,
                     top_k: int = 20,
                     init_lr: float = 0.20,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on the max‑convolution.
    Averages the sub‑gradients of the ``top_k`` largest convolution entries.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)

        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        cur_max = conv[idxs].max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # aggregated sub‑gradient for the selected shifts
        g = np.zeros_like(a)
        for k in idxs:
            rev = k - np.arange(n)
            mask = (rev >= 0) & (rev < n)
            g[mask] += 2.0 * a[rev[mask]]
        g /= float(len(idxs))

        # Adam update (learning‑rate schedule kept simple)
        lr = init_lr / np.sqrt(it)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _proj_simplex(a)

    return best_seq, best_val


def _biased_move(cur: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a small amount of probability mass from heavy coordinates
    towards light ones.  Returns a new feasible vector (still on the simplex).
    """
    n = cur.shape[0]
    # number of coordinates to touch
    k = rng.integers(2, 15)

    # sample source indices with probability proportional to current mass
    src = rng.choice(n, size=k, replace=False, p=cur / cur.sum())
    total = cur[src].sum()
    if total == 0.0:
        return cur.copy()

    # make a copy and zero‑out the sources
    new = cur.copy()
    new[src] = 0.0

    # target indices – favour low‑mass coordinates
    inv = 1.0 - cur
    inv_sum = inv.sum()
    if inv_sum == 0.0:
        tgt = rng.choice(n, size=k, replace=False)
        probs = np.full(k, 1.0 / k)
    else:
        probs = inv / inv_sum
        tgt = rng.choice(n, size=k, replace=False, p=probs)

    # redistribute the removed mass uniformly via a Dirichlet draw
    alloc = rng.dirichlet(np.full(k, 1.0)) * total
    new[tgt] = alloc
    # projection is unnecessary – total mass is preserved and all entries ≥0
    return new


def _biased_simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 1_000_000,
                                init_temp_factor: float = 0.05,
                                min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with biased moves that preferentially
    shift probability away from heavy coordinates.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling – slightly slower than the baseline
        T = max(T * 0.9995, min_temp)

        new = _biased_move(cur, rng)
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[Biased SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previous run if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the candidate vector – 4000 gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and quick ranking
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Total seeds generated: {len(seeds)}")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deterministic descent (mirror + Adam sub‑gradient)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        # allocate ~15 % to mirror descent (fast, coarse exploration)
        mirror_deadline = time.time() + 0.15 * time_left
        print("\n--- Exponential mirror descent ---")
        seq_mirror, val_mirror = _exp_mirror_descent(
            global_best_seq, n, mirror_deadline,
            max_iters=2_000_000, eta0=0.9
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror, val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # Adam sub‑gradient refinement – biggest chunk of the budget
    time_left = deadline - time.time()
    if time_left > 0:
        adam_deadline = time.time() + 0.70 * time_left
        print("\n--- Adam‑style sub‑gradient refinement (top_k=20) ---")
        seq_adam, val_adam = _adam_subgradient(
            global_best_seq, n, adam_deadline,
            top_k=20,
            init_lr=0.20
        )
        if val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam, val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – stochastic polishing (biased simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Biased simulated‑annealing polishing ---")
        seq_sa, val_sa = _biased_simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=1_000_000,
            init_temp_factor=0.05
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[Biased SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check – official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
