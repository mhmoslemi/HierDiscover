def _mirror_subgradient_multi_optimize(start_seq, n, deadline, max_iters=2_000_000, active_tol=1e-6):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    lr0 = 0.4
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        # identify active indices within relative tolerance
        # treat absolute tolerance also
        thresh = max_val - active_tol
        active = conv >= thresh
        # compute weight vector w as active indicator (float)
        w = active.astype(np.float64)
        # gradient: 2 * (a *conv w)
        grad = 2.0 * _conv_fft(a, w)[n-1:n-1+n]
        # multiplicative update on simplex (exponentiated)
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * grad)
        s = a.sum()
        if s <= 0:
            a = _uniform(n)
        else:
            a = a / s
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[MirrorMulti] it={it} best={best_val:.6f}")
        # occasional reordering
        if it % 50000 == 0:
            a_alt = _alternate_order(a)
            alt_val = _objective(a_alt)
            if alt_val < cur_val:
                a = a_alt
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()
                    print(f"[MirrorMulti-alt] it={it} improved alt to {best_val:.6f}")
    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm is a heavyweight optimisation pipeline that
  * builds a large pool of analytic and stochastic “seed” shapes,
  * refines the best seeds with a long Adam optimiser on a smooth
    log‑sum‑exp surrogate,
  * runs a *multi‑active* mirror‑descent sub‑gradient optimiser
    (the sub‑gradient is taken on all convolution entries that are
    within a tiny tolerance of the current maximum – this gives a
    much richer descent direction than the single‑active version),
  * performs a permutation hill‑climbing phase,
  * refines further with a second mirror‑descent pass,
  * finally a simulated‑annealing polish.

All phases respect a global 1000 s time budget; the code constantly
checks the deadline and stops early if needed.  The best vector found
is stored globally as ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)).  Sequence is assumed L1‑normalised."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """
    Re‑order as large, small, 2nd‑large, 2nd‑small, … and renormalise to L1‑norm = 1.
    """
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


# ----------------------------------------------------------------------
# Analytic shape families – all return L1‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.07,
                     offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection – richer than the reference version
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))

    # sine windows on a fine grid
    for gamma in np.linspace(0.5, 6.0, 12):
        s = _sine_window(n, gamma)
        seeds.append(s)
        seeds.append(_alternate_order(s))

    # Gaussian on a sigma grid
    for sigma in np.linspace(0.07, 0.30, 12):
        g = _gaussian(n, sigma)
        seeds.append(g)
        seeds.append(_alternate_order(g))

    # beta family – fairly wide exploration
    for _ in range(30):
        alpha = 10 ** rng.uniform(-3, 0.8)
        beta = 10 ** rng.uniform(-3, 0.8)
        b = _beta_shape(n, alpha, beta)
        seeds.append(b)
        seeds.append(_alternate_order(b))

    # exponential shapes
    for rate in np.logspace(-5, -1, 12):
        e = _exponential(n, rate)
        seeds.append(e)
        seeds.append(_alternate_order(e))

    # logarithmic shapes
    for eps in np.logspace(-5, -1, 8):
        l = _logarithmic(n, eps)
        seeds.append(l)
        seeds.append(_alternate_order(l))

    # Dirichlet draws – very sparse to very uniform
    concs = [0.001, 0.005, 0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_alternate_order(raw))

    # Random mixtures of two disjoint uniform blocks
    for _ in range(5):
        block_len = rng.integers(5, n // 2)
        start = rng.integers(0, n - block_len)
        mix = np.zeros(n, dtype=np.float64)
        mix[start:start + block_len] = rng.uniform(0.5, 1.0)
        mix /= mix.sum()
        seeds.append(mix)
        seeds.append(_alternate_order(mix))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            normalised = avg / avg.sum()
            seeds.append(normalised)
            seeds.append(_alternate_order(normalised))

    return seeds

# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int,
                   deadline: float, max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on the log‑softmax (simplex) using a temperature‑controlled
    log‑sum‑exp surrogate for max(conv).  Returns best feasible vector
    seen and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)           # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 0.5          # start temperature for smooth max
    beta_inc = 0.7
    beta_max = 350.0

    best_seq = None
    best_val = float("inf")
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n

        # map to log‑space (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it + 1.0)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_multi_optimize(start_seq: np.ndarray, n: int,
                                      deadline: float,
                                      max_iters: int = 2_000_000,
                                      active_tol: float = 1e-9,
                                      lr0: float = 0.35) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) sub‑gradient optimiser.
    The sub‑gradient is taken on *all* convolution entries whose
    value lies within ``active_tol`` of the current maximum.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = np.max(conv)

        # active set – indices whose value is within the tolerance
        # of the maximum.  Using a tiny absolute tolerance avoids a
        # degenerate single‑active gradient when the peak is flat.
        active = conv >= (max_val - active_tol)

        # weight vector w (length 2n‑1) : 1 on active indices, 0 elsewhere
        w = active.astype(np.float64)

        # gradient of Σ w_k·conv_k  w.r.t. a : 2 * (a * w) convolution
        grad = 2.0 * _conv_fft(a, w)[n - 1:n - 1 + n]

        # multiplicative (exponentiated) update on the simplex
        lr = lr0 / np.sqrt(it + 1.0)
        a = a * np.exp(-lr * grad)

        # re‑normalise to the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror‑Multi] it={it:7d} best={best_val:.6f}")

        # every now and then try the alternating permutation
        if it % 70_000 == 0:
            a_alt = _alternate_order(a)
            alt_val = _objective(a_alt)
            if alt_val < cur_val:
                a = a_alt
                if alt_val < best_val:
                    best_val = alt_val
                    best_seq = a.copy()
                    print(f"[Mirror‑Multi‑alt] it={it:7d} improved alt to {best_val:.6f}")

    return best_seq, best_val


def _perm_hill_climb(start_seq: np.ndarray, n: int,
                     deadline: float,
                     max_swaps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple swap‑based hill‑climbing on the permutation of entries.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    it = 0
    # mild temperature for occasional uphill moves
    T0 = cur_val * 0.004
    T = T0

    while it < max_swaps and time.time() < deadline:
        it += 1
        # geometric cooling
        T = max(T * 0.9999, 1e-12)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30_000 == 0:
                    print(f"[Perm] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 4_500_000,
                                 init_temp_factor: float = 0.025,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing polish with three move types (swap, mass‑transfer,
    pair‑transfer).  The schedule is slightly slower than the reference
    to give the algorithm more time to settle.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a small random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 60_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the time budget.
    Returns the best sequence (list of floats) and stores it globally as
    ``height_sequence_1``.
    """
    # initialise RNG
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between granularity and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous best, if any, and normalise it.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – generate a diverse pool of seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # initialise the global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinements on the top seeds (longer runs).
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate up to ~150k iterations per seed, but guarantee a minimum
        max_iters = int(min(150_000, max(30_000, remaining * 80)))
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam.copy(), val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # try alternating permutation of the Adam result
        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq, global_best_val = alt, alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – multi‑active mirror‑descent sub‑gradient optimisation.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Multi‑active Mirror‑Descent optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_multi_optimize(
            global_best_seq, n, deadline, max_iters=2_500_000, active_tol=1e-8, lr0=0.35
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror.copy(), val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – permutation hill‑climbing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation hill‑climbing ---")
        seq_perm, val_perm = _perm_hill_climb(
            global_best_seq, n, deadline, max_swaps=200_000
        )
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[overall‑Perm] new global best = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – second mirror‑descent polishing (same multi‑active routine).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Second mirror‑descent polishing ---")
        seq_m2, val_m2 = _mirror_subgradient_multi_optimize(
            global_best_seq, n, deadline, max_iters=1_500_000, active_tol=1e-9, lr0=0.30
        )
        if val_m2 < global_best_val:
            global_best_seq, global_best_val = seq_m2.copy(), val_m2
            print(f"[overall‑Mirror2] new global best = {val_m2:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – simulated‑annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=4_500_000,
            init_temp_factor=0.025,
            min_temp=1e-12
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
