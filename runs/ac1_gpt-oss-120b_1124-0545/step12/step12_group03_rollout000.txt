"""
Propose a step‑function sequence that minimizes ``evaluate_sequence``.

The algorithm runs a multi‑stage search:

1. **Rich seed pool** – Analytic windows, beta‑shapes, Dirichlet draws,
   low‑frequency cosine curves and a few user‑defined exotic families.
   Every seed is ℓ₁‑normalised (sum = 1).

2. **Surrogate ranking** – Fast FFT‑based surrogate
   ``obj(a) = 2·n·max(conv(a,a))`` (the factor 2·n is the exact
   evaluation constant).  The best ``top_k`` seeds are kept.

3. **Adam on a smoothed max** – Parameters are the logits of a softmax.
   A temperature‑controlled log‑sum‑exp approximates the max; the
   temperature is increased *slowly* (by 0.05 per 1 k iterations) up to
   400, giving a high‑quality gradient in the early phase.

4. **Exact mirror‑subgradient descent** – Multiplicative updates on the
   probability simplex using the true max‑convolution gradient.
   Step‑size decays as η₀·t⁻ᵖ (η₀=0.12, p=0.28) and runs until the time
   budget is exhausted.

5. **Iterated LP polishing** – For the current vector ``f`` we solve

        maximise    Σ g_i
        subject to  conv(f,g) ≤ M,   g ≥ 0

   where ``M = max(conv(f,f))``.
   The solution ``g`` is mixed with ``f`` using a line‑search on τ
   (τ ∈ {0.05,0.1,0.2}) that yields the smallest new maximum.
   Up to 15 LP rounds are performed (or until no improvement).

6. **Simulated‑annealing fine‑tuning** – Three simple moves
   (swap, Dirichlet‑redistribute, pair‑transfer) are applied with an
   exponential cooling schedule.  This stage consumes the final
   seconds of the budget.

All stages respect a global deadline (budget‑5 s) to guarantee that a
result is returned.  The best vector is stored in the global variable
``height_sequence_1`` and also returned as a plain list of Python floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for large vectors)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact surrogate used during the search (2·n·max(conv))."""
    n = a.shape[0]
    return 2.0 * n * np.max(_conv_fft(a, a))


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_squared(n: int) -> np.ndarray:
    """Window a_i = sin^2(π·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, steep: float = 6.0) -> np.ndarray:
    """A smooth S‑shaped window based on the logistic function."""
    t = np.linspace(-1.0, 1.0, n)
    a = 1.0 / (1.0 + np.exp(-steep * t))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve produced by a few random cosine harmonics."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture and re‑normalisation."""
    m = (1.0 - alpha) * a + alpha * b
    s = m.sum()
    return m / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on logits of a softmax using a slowly‑raised temperature."""
    eps = 1e-30
    logits = np.log(start + eps)        # initialise logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta = 0.5                         # temperature
    beta_step = 0.05                    # slower increase
    beta_max = 400.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = np.exp(logits - np.max(logits))
        a = a / a.sum()                # simplex point

        # surrogate objective
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] iter {it:7d} best {cur_val:.6f}")

        # --- smoothed max via log‑sum‑exp ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted convolution w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_logits = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_logits
        v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_step, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start: np.ndarray, n: int, deadline: float,
                                max_iters: int = 6_000_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the exact max."""
    a = start.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12
    power = 0.28

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] iter {it:7d} best {cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g[i_range] = 2.0 * a[rev]

        # multiplicative EG update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """Construct sparse matrix A such that (A @ g) = conv(f, g)."""
    n = f.shape[0]
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for k in range(n):
        rows[idx:idx + n] = np.arange(k, k + n, dtype=np.int32)
        cols[idx:idx + n] = k
        data[idx:idx + n] = f
        idx += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 15, tau_candidates: tuple = (0.05, 0.1, 0.2)
               ) -> tuple[np.ndarray, float]:
    """Repeated LP polishing: maximise ℓ₁‑mass under the current max‑conv bound."""
    f = seq.copy()
    n = f.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()
        A = _build_conv_matrix(f)               # (2n‑1) × n
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)       # maximise Σ g_i

        # solve the LP (HiGHS is fast for this size)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})

        if not res.success:
            print(f"[LP] iteration {it} failed – skipping remaining LP steps.")
            break

        g = res.x
        # try a few τ values and keep the one giving smallest new max
        best_f = None
        best_obj = float("inf")
        for tau in tau_candidates:
            mixed = (1.0 - tau) * f + tau * g
            mixed /= mixed.sum()
            obj = _objective(mixed)
            if obj < best_obj:
                best_obj = obj
                best_f = mixed

        if best_f is None or best_obj >= _objective(f) - 1e-9:
            # no improvement → stop LP phase
            print(f"[LP] iteration {it} – no improvement, terminating LP stage.")
            break

        f = best_f
        print(f"[LP] iteration {it} – objective now {best_obj:.6f}")

    return f, _objective(f)


def _simulated_annealing_optimize(start: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 1_500_000,
                                 init_temp_factor: float = 0.045,
                                 min_temp: float = 1e-13) -> tuple[np.ndarray, float]:
    """Lightweight SA that mutates mass among a few positions."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9993, min_temp)

        move = rng.integers(0, 3)
        if move == 0:        # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move == 1:      # redistribute mass over a random small set
            k = rng.integers(3, 13)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new = cur.copy()
            new[idx] = rng.dirichlet(np.ones(k)) * total
        else:                # transfer mass between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet(np.ones(2)) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] iter {it:7d} best {new_val:.6f}")

    return best, best_val

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))
    seeds.append(_sine_squared(n))
    seeds.append(_logistic_window(n, steep=8.0))

    # Parameter sweeps
    for sigma in np.linspace(0.04, 0.55, 13):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    for p in np.linspace(0.0, 7.0, 15):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 3.5, 12)
    betas = np.linspace(0.2, 3.5, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for lam in np.concatenate([np.linspace(0.0001, 0.015, 10),
                               np.linspace(0.015, 0.05, 5)]):
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet draws with many concentration values
    for conc in (0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.3, 0.8, 1.5, 3.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                         # raw order
            seeds.append(np.sort(raw)[::-1])          # descending
            # interleaved order (high‑low‑high‑...)
            a = np.sort(raw)
            lo, hi = 0, n - 1
            alt = np.empty_like(a)
            pos = 0
            while lo <= hi:
                if pos % 2 == 0:
                    alt[pos] = a[hi]
                    hi -= 1
                else:
                    alt[pos] = a[lo]
                    lo += 1
                pos += 1
            seeds.append(alt)

    # Low‑frequency cosine curves
    for nf in (2, 3, 5, 7):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.15, 0.35, 0.55, 0.75):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix(_uniform(n), curve, mix_alpha))

    # Previous solution (if available)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

    n = 4000

    print(f"Top {top_k} surrogate values (lower is better):")

        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()

    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(best_seeds, 1):

        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent sub‑gradient (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:

        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=7_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – LP polishing (multiple iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming polishing ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, deadline,
                                    max_iters=15, tau_candidates=(0.05, 0.1, 0.2))
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=2_000_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Report official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
