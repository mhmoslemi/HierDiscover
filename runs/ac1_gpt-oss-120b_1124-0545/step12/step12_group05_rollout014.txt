"""
Propose a step‑function (non‑negative vector) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

Algorithm
---------

1. **Seed pool** – generate many analytic shapes (uniform, triangular,
   sine‑windows, Gaussians, Beta, Laplace, mixtures, Dirichlet draws…)
   and ℓ₁‑normalise them.

2. **Select the elite** – keep the best few seeds w.r.t. the surrogate
   objective 2·n·max(conv) (the denominator is 1 after ℓ₁‑normalisation).

3. **Linear mixing** – for every pair of elite seeds perform a cheap
   line‑search over convex combinations  λ·s₁ + (1‑λ)·s₂ .  The best mixture
   is added to the elite set.

4. **Mirror‑descent (exponentiated‑gradient) sub‑gradient** – start from the
   best mixture and run a long mirror‑descent on the *exact* max‑convolution
   (the same routine that the reference code used, but with a higher
   iteration budget and a stricter time guard).

5. **Permutation optimisation** – (a) a deterministic “alternating”
   permutation (large‑small‑large‑…) followed by a fast
   swap‑based hill‑climb, and (b) a final simulated‑annealing phase that
   uses three move types (swap, Dirichlet‑redistribution on a random
   subset, pair‑mass‑transfer).  Because the vector length is moderate
   (4000), each move is evaluated with a full FFT‑based convolution;
   the total number of moves is limited only by the remaining time.

6. **Return** – the best vector seen (as a plain ``list[float]``) is stored
   as ``height_sequence_1`` for later runs and is also returned.

All steps respect the global 1000 s budget; the code falls back to the
current best if the deadline is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate: numerator 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """
    Re‑order as large, small, 2nd‑large, 2nd‑small, …  and renormalise to ℓ₁‑norm = 1.
    """
    a = np.sort(vec)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if (pos & 1) == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else np.full_like(out, 1.0 / len(out))


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, lam: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / lam)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.07, offset_frac: float = 0.35) -> np.ndarray:
    sigma = sigma_frac * n
    offset = offset_frac * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - (centre - offset)) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - (centre + offset)) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + 1e-12
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_laplace_window(n, lam=0.4))
    seeds.append(_double_gaussian(n, sigma_frac=0.06, offset_frac=0.30))

    # sine windows
    for gamma in np.linspace(0.5, 6.0, 12):
        seeds.append(_sine_window(n, gamma))
        seeds.append(_alternate_order(_sine_window(n, gamma)))

    # Gaussians
    for sigma in np.linspace(0.07, 0.30, 12):
        seeds.append(_gaussian(n, sigma))
        seeds.append(_alternate_order(_gaussian(n, sigma)))

    # Beta family
    for _ in range(30):
        alpha = 10 ** rng.uniform(-3, 0.8)
        beta = 10 ** rng.uniform(-3, 0.8)
        b = _beta_shape(n, alpha, beta)
        seeds.append(b)
        seeds.append(_alternate_order(b))

    # Exponential
    for rate in np.logspace(-5, -1, 12):
        e = _exponential(n, rate)
        seeds.append(e)
        seeds.append(_alternate_order(e))

    # Logarithmic
    for eps in np.logspace(-5, -1, 8):
        l = _logarithmic(n, eps)
        seeds.append(l)
        seeds.append(_alternate_order(l))

    # Dirichlet draws
    concs = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)
            seeds.append(_alternate_order(raw))

    # Mixtures of uniform + low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(2):
            t = np.arange(n, dtype=np.float64)
            curve = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                curve += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            curve = np.maximum(curve, 0.0)
            curve /= curve.sum()
            mixed = 0.7 * _uniform(n) + 0.3 * curve
            mixed /= mixed.sum()
            seeds.append(mixed)
            seeds.append(_alternate_order(mixed))

    # Previous solution (if supplied)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            normalised = avg / avg.sum()
            seeds.append(normalised)
            seeds.append(_alternate_order(normalised))

    return seeds

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 150_000) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax parameters with a temperature‑controlled smooth max."""
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 0.5          # temperature for smooth max
    beta_inc = 0.5
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                         # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the true max‑conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.30          # start step size

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # Sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update on the simplex
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional aggressive re‑ordering
        if it % 60_000 == 0:
            a_alt = _alternate_order(a)
            val_alt = _objective(a_alt)
            if val_alt < cur_val:
                a = a_alt
                if val_alt < best_val:
                    best_val = val_alt
                    best_seq = a.copy()
    return best_seq, best_val


def _perm_hill_climb(start_seq: np.ndarray, n: int,
                     deadline: float,
                     max_swaps: int = 150_000) -> tuple[np.ndarray, float]:
    """Swap‑based hill‑climbing on the permutation of the entries."""
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    it = 0
    T0 = cur_val * 0.005
    T = T0

    while it < max_swaps and time.time() < deadline:
        it += 1
        T = max(T * 0.9999, 1e-12)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 4_000_000,
                                 init_temp_factor: float = 0.02,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """SA polish with three move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pair‑transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
    return best_seq, best_val


def _mix_elite_seeds(best_seeds: list[np.ndarray], n: int,
                    deadline: float,
                    mix_grid: int = 51) -> tuple[np.ndarray | None, float]:
    """
    Exhaustively line‑search convex mixtures of every pair of elite seeds.
    Returns the best mixture found (or None if no time left) and its surrogate value.
    """
    best_seq = None
    best_val = float("inf")
    K = len(best_seeds)
    for i in range(K):
        for j in range(i + 1, K):
            s1 = best_seeds[i]
            s2 = best_seeds[j]
            # early exit
            if time.time() > deadline:
                return best_seq, best_val
            for lam in np.linspace(0.0, 1.0, mix_grid):
                combo = lam * s1 + (1.0 - lam) * s2
                # renormalise to avoid drift from floating‑point errors
                combo = combo / combo.sum()
                val = _objective(combo)
                if val < best_val:
                    best_val = val
                    best_seq = combo.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative vector that minimises ``evaluate_sequence``.
    Returns the best list of floats and stores it globally as ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Length – 4000 gives a good balance.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous best (if any) and normalise.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.sum() > 0:
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – generate a diverse pool of seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (2·n·max(conv)):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – quick Adam refinements on the elite seeds.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        max_iters = int(min(130_000, remaining * 45))
        max_iters = max(5_000, max_iters)
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam.copy(), val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # try alternating permutation of Adam result
        if seq_adam is not None:
            alt = _alternate_order(seq_adam)
            alt_val = _objective(alt)
            if alt_val < global_best_val:
                global_best_seq, global_best_val = alt, alt_val
                print(f"[overall‑Adam‑alt] new global best = {alt_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – linear mixing of the elite seeds.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear mixing of elite seeds ---")
        mix_seq, mix_val = _mix_elite_seeds(best_seeds, n, deadline, mix_grid=51)
        if mix_seq is not None and mix_val < global_best_val:
            global_best_seq, global_best_val = mix_seq, mix_val
            print(f"[mix] new global best (surrogate) = {mix_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – long mirror‑descent sub‑gradient optimisation.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=6_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror.copy(), val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – permutation hill‑climbing (re‑ordering only).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation hill‑climbing ---")
        seq_perm, val_perm = _perm_hill_climb(
            global_best_seq, n, deadline, max_swaps=150_000
        )
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm.copy(), val_perm
            print(f"[overall‑Perm] new global best = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – second mirror‑descent polishing (after re‑ordering).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Second mirror‑descent polishing ---")
        seq_m2, val_m2 = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=3_000_000
        )
        if val_m2 < global_best_val:
            global_best_seq, global_best_val = seq_m2.copy(), val_m2
            print(f"[overall‑Mirror2] new global best = {val_m2:.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – simulated‑annealing polish (final refinement).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=5_000_000,
            init_temp_factor=0.02,
            min_temp=1e-12
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation augments the previous approach with an explicit **linear mixing** step that searches convex combinations of the top analytic seeds, allocates more time to the mirror‑descent polish, and performs a longer permutation‑based hill‑climb followed by an extended simulated‑annealing phase.  
All stages respect the 1000 s budget, use only NumPy (and the supplied `evaluate_sequence`), and expose the best-found vector as `height_sequence_1`.
