def project_simplex(v):
    # v may have negative values
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (length = len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Soft‑max utilities (log‑sum‑exp)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors (sum == 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverted_triangle(n: int) -> np.ndarray:
    """U‑shaped linear valley (zero at centre, maximum at ends)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.abs(i - centre) / centre      # 0 at centre, 1 at ends
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-9) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _cosine_window(n)
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1.0)))
        elif i <= (n - 1) * (1.0 - alpha / 2.0):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2.0 / alpha + 1.0)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinc_squared(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = np.where(i == 0, 1.0, np.sin(np.pi * i / n) / (np.pi * i / n))
    a = x ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine_power(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Surrogate objective (fast, identical ordering to official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient w.r.t. a
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_max, grad_a) where smooth_max = log‑sum‑exp_beta(conv)/beta
    (ignores the external 2·n factor) and grad_a = ∂smooth_max/∂a.
    """
    conv = _conv_fft(a, a)
    shift = np.max(conv)               # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum       # softmax weights over the full conv
    smooth = shift + np.log(w_sum) / beta

    # gradient: 2 * (rev(a) *weights)  (convolution)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 400,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """
    L‑BFGS on the smooth‑max surrogate (log‑sum‑exp).  The optimisation
    variable are logits (unconstrained); a softmax enforces the simplex.
    """
    eps = 1e-30
    v = np.log(start_seq + eps)    # logits

    def fun(v):
        a = _softmax(v)            # ℓ₁‑normalised probabilities
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth     # scale to match the true objective
        # back‑prop through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        grad_v_scaled = 2.0 * n * grad_v
        return loss, grad_v_scaled.astype(np.float64)

    # guard against overrunning the deadline
    if time.time() > deadline:
        return start_seq, _objective(start_seq)

    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:          # pragma: no cover
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000,
                                 lr0: float = 0.12,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent on the simplex."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:,d}  best={best_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight Simulated Annealing (final polishing)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)   # 0=swap, 1=redistribute subset, 2=pair‑transfer
        if move_type == 0:               # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:             # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                            # pair‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 250_000 == 0:
                    print(f"[SA] it={it:,d}  best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper: keep only the K lowest‑objective seeds (min‑heap)
# ----------------------------------------------------------------------
def _keep_top_k(seeds, values, k):
    """Return the k best (lowest) seeds given their objective values."""
    heap = []   # max‑heap on negative value
    for vec, val in zip(seeds, values):
        if len(heap) < k:
            heapq.heappush(heap, (-val, vec))
        else:
            if val < -heap[0][0]:
                heapq.heapreplace(heap, (-val, vec))
    best = [vec for (_, vec) in sorted(heap, key=lambda x: -x[0])]
    return best

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` objective within the given time budget.
    The best sequence is returned as a list of floats and stored in the
    global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Length of the step‑function (chosen to keep FFT cheap)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Retrieve any previously found solution for seeding
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.shape != (n,):
                previous = None
        except Exception:
            previous = None

    # --------------------------------------------------------------
    # Stage 0 – generate a huge pool of diverse seeds
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = []

    # Analytic families (all ℓ₁‑normalised)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_inverted_triangle(n))
    seeds.append(_cosine_window(n))
    seeds.append(_sinc_squared(n))

    # sine windows
    for gamma in np.linspace(0.3, 4.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian family (wide range)
    for sigma in np.linspace(0.05, 0.45, 12):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Power‑law family
    for p in np.linspace(0.0, 4.0, 9):
        seeds.append(_power_law(n, p))

    # Beta shape – random exploration plus a grid
    alphas_grid = np.concatenate((np.linspace(0.05, 0.30, 6),
                                  np.linspace(0.30, 3.0, 10)))
    betas_grid = alphas_grid.copy()
    for a in alphas_grid:
        for b in betas_grid:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # Random β‑shapes (continuous sampling)
    for _ in range(200):
        a = rng.uniform(0.02, 3.0)
        b = rng.uniform(0.02, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.05, 0.95, 9):
        seeds.append(_tukey_window(n, alpha))

    # Exponential decay
    for lam in np.linspace(0.0005, 0.015, 12):
        seeds.append(_exponential_decay(n, lam))

    # Raised‑cosine‑power
    for power in [0.5, 1.0, 1.5, 2.0, 3.0]:
        seeds.append(_raised_cosine_power(n, power))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform and low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_uniform(n) * (1 - mix_alpha) + curve * mix_alpha)

    # Large batch of Dirichlet draws with varied concentration
    N_dirichlet = 800
    for _ in range(N_dirichlet):
        # log‑uniform concentration in [0.01, 2.0]
        log_c = rng.uniform(np.log(0.01), np.log(2.0))
        conc = np.exp(log_c)
        raw = rng.dirichlet(np.full(n, conc))
        seeds.append(raw)                      # raw order
        seeds.append(np.sort(raw)[::-1])      # descending
        seeds.append(np.sort(raw))            # ascending

    # Previous solution (if any)
    if previous is not None:
        seeds.append(previous / previous.sum())

    # --------------------------------------------------------------
    # Fast surrogate evaluation & keep the best few seeds
    # --------------------------------------------------------------
    print(f"Generated {len(seeds)} raw seeds – evaluating surrogate...")
    values = [_objective(vec) for vec in seeds]
    top_k = 20
    best_seeds = _keep_top_k(seeds, values, top_k)
    best_vals = [_objective(vec) for vec in best_seeds]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, v in enumerate(sorted(best_vals)[:top_k], 1):
        print(f"  #{i}: {v:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on smooth‑max surrogate (continuation)
    # --------------------------------------------------------------
    temps = [10.0, 30.0, 100.0, 300.0, 500.0]  # increasing β → sharper approximation
    max_iter_per_temp = 400

    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temps:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline, max_iter=max_iter_per_temp, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, β={temp:.0f} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start next temperature

    # --------------------------------------------------------------
    # Stage 2 – Mirror‑descent (exact max) fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_m, val_m = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=5_000_000,
            lr0=0.12,
            power=0.30,
        )
        if val_m < global_best_val:
            global_best_seq = seq_m.copy()
            global_best_val = val_m
            print(f"[Mirror] new global best = {val_m:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing (optional)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation (informational)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for reuse by subsequent runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
