def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np, math
    try:
        import scipy.optimize as opt
    except Exception:
        opt = None
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    n = 4000
    # Gather previous best if any
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.shape[0] != n:
            # If length mismatch, truncate/upsample later
            prev_arr = None

    # Seed generation: include uniform, triangular, windows, random Dirichlet with varying concentration, random exponential draws, random low freq curves, previous solution
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, 0.5))
    # more analytic windows
    for beta in [2,4,6,8,10]:
        seeds.append(_kaiser_window(n, beta))
    # random Dirichlet with various conc
    for conc in [0.05,0.2,0.5,1.0,2.0,5.0]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    # random exponential
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))
    # low freq curves
    for nf in [2,3,4]:
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))
    if prev_arr is not None:
        seeds.append(prev_arr / prev_arr.sum())
    # Normalise each
    seeds = [s / s.sum() for s in seeds]

    # cheap surrogate
    sur_vals = [_objective_surrogate(s) for s in seeds]
    top_indices = np.argsort(sur_vals)[:6]
    top_seeds = [seeds[i] for i in top_indices]
    # track best
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)

    # L-BFGS for each seed
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         beta_schedule=[1.0,5.0,20.0,80.0,200.0],
                                         max_iter_per_beta=4000)
        val = _objective_surrogate(seq_opt)
        if val < best_val:
            best_seq, best_val = seq_opt, val

    # extended refinement on best
    if time.time() < deadline:
        seq_opt = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                         beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0],
                                         max_iter_per_beta=8000)
        val = _objective_surrogate(seq_opt)
        if val < best_val:
            best_seq, best_val = seq_opt, val

    # Random local search
    # Parameter for step fraction: start at 0.2 and decay
    a = best_seq.copy()
    # Keep a as simplest representation
    step_scale = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        # dynamic step decreasing with iterations
        lr = max(1e-4, step_scale / math.sqrt(it))
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] < 1e-12:
            continue
        delta = min(lr * a[i], a[i])  # move fraction of a[i]
        # propose move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # project onto simplex not needed: sum unchanged, nonneg guaranteed
        # compute objective
        val_new = _objective_surrogate(a_new)
        # accept if better
        if val_new < best_val:
            a = a_new
            best_seq, best_val = a_new, val_new
            # optionally reset iteration count for step scaling
            it = 0
        else:
            # maybe allow uphill with small probability
            pass
        # occasional random restart to avoid stagnation
        if it % 50000 == 0 and time.time() < deadline:
            # perturb current best slightly
            perturb = rng.normal(scale=0.01, size=n)
            a_perturbed = np.maximum(a + perturb, 0)
            a_perturbed /= a_perturbed.sum()
            a = a_perturbed

    # final evaluation
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluate_sequence = {true_val}")
    except Exception as e:
        print("Evaluation error:", e)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
propose_candidate – fast local‑search optimiser for the max‑convolution bound.

The function builds a modest pool of ℓ₁‑normalised “seed’’ vectors (simple windows,
Dirichlet draws and a few analytic families).  The best few seeds are refined
with a continuation L‑BFGS optimiser that minimises a smooth approximation of
the max‑convolution (log‑sum‑exp with an increasing β schedule).  After the
smooth optimisation a long sub‑gradient / exponentiated‑gradient polishing phase
runs until the time budget expires.  The final vector is stored in the global
name ``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional imports
# ----------------------------------------------------------------------
try:                     # SciPy is available in the execution environment
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    This equals the real objective because the sum of ``seq`` is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1.0 - y * y)
    a = np.i0(t)                     # numpy’s modified Bessel function
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    The variables are logits ``v``; ``softmax(v)`` always lies on the simplex.
    """
    if _opt is None:  # pragma: no cover
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        # -------------------- objective & gradient --------------------
        def fun(v_local):
            a = _softmax(v_local)                     # probability simplex
            conv = _conv_fft(a, a)                     # full autocorrelation
            # smooth approximation of the max (log‑sum‑exp)
            shift = np.max(conv)                       # for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth objective w.r.t. a
            w = exp_beta / sum_exp                      # softmax‑like weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through the softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # -------------------- timeout callback -----------------------
        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Sub‑gradient / exponentiated‑gradient refinements
# ----------------------------------------------------------------------
def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Stops automatically when the global deadline is hit.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / np.sqrt(it)          # diminishing step size
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.5, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    for beta in (2, 6, 8):
        seeds.append(_kaiser_window(n, beta))

    # random Dirichlet (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution, if compatible
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # robust normalisation
    final = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            final.append(v / s)
        else:
            final.append(_uniform(n))
    return final
