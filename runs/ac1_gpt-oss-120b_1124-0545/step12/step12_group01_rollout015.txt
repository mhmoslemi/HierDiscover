"""
Improved search for a step‑function minimising

    2·n·max(conv(a,a)) / (sum(a)²)

The core ideas
----------------
* A family of “inverse‑power’’ seeds  a_i ∝ (i+δ)^‑p  with p≈0.5 (the
  continuous kernel 1/√x has a flat self‑convolution).  Small shifts δ
  are swept to get a slightly better discrete approximation.
* All seeds are normalised to an ℓ¹‑sum of 1 and ranked by the *exact*
  objective (no surrogate).
* The best seeds are refined with a custom Adam optimiser that works on the
  smooth Log‑Sum‑Exp approximation of the max‑convolution.  The β‑parameter
  (sharpness) is increased progressively (continuation) and the Adam
  parameters are tuned for fast convergence.
* After Adam we run a long sub‑gradient / exponentiated‑gradient polishing
  phase (the same code as the reference implementation) to squeeze out the
  last few percent.
* The whole pipeline respects the supplied time budget (default 1000 s) and
  stores the best sequence in the global ``height_sequence_1`` variable.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no clipping)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv) / (np.sum(seq) ** 2)


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    a /= a.sum()
    return a


def _power_offset(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}  (offset>0 to avoid i=0 singularity)."""
    i = np.arange(n, dtype=np.float64) + offset
    a = i ** (-p)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Seed collection – focused around the 1/√ kernel
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> List[np.ndarray]:
    seeds: List[np.ndarray] = []

    # a few classic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.2):
        seeds.append(_sine_window(n, gamma=g))

    # power‑law around p≈0.5 with several offsets
    for offset in (0.0, 0.2, 0.5, 1.0, 2.0):
        for p in (0.45, 0.5, 0.55):
            seeds.append(_power_offset(n, p, offset))

    # Beta‑family – explore a region around the U‑shape
    for a in (0.4, 0.5, 0.6):
        for b in (0.4, 0.5, 0.6):
            seeds.append(_beta_shape(n, a, b))

    # Random draws – a few Dirichlets help avoid local minima
    for conc in (0.05, 0.2, 0.8, 1.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,) and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # clean up – make sure everything is a proper probability vector
    cleaned: List[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          beta_schedule=None,
                          max_iters_total: int = 25000) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp smoothed max.
    Parameters are the logits of a soft‑max (guarantees positivity and sum‑1).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.12          # a slightly aggressive learning rate

    # initialise logits so that softmax(start_vec)=start_vec
    v = np.log(start_vec + 1e-30)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)   # second moment (named v2 to avoid clash with input v)

    # allocate iterations per β (more for later, sharper β)
    wts = np.array([1, 2, 3, 4, 5, 6, 7], dtype=float)
    it_per_beta = np.maximum(1, (max_iters_total * wts / wts.sum()).astype(int))

    global_it = 0
    for beta, it_budget in zip(beta_schedule, it_per_beta):
        beta_val = float(beta)
        for _ in range(it_budget):
            if time.time() > deadline:
                break
            global_it += 1

            # ----- forward pass -------------------------------------------------
            a = _softmax(v)                                   # probability vector
            conv = _conv_fft(a, a)                             # full auto‑correlation
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))      # safe because shift= max(conv)
            sum_exp = exp_beta.sum()
            # smooth surrogate of the max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # ----- gradient w.r.t. a -------------------------------------------
            w = exp_beta / sum_exp                             # soft‑weights (≈soft‑max)
            # gradient of Σ w_k * Σ_i a_i a_{k-i}  w.r.t. a  equals 2 * conv_rev(a, w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                         # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # ----- gradient w.r.t. logits (softmax Jacobian) --------------------
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ----- Adam update -------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** global_it)
            v2_hat = v2 / (1.0 - beta2 ** global_it)

            step = base_lr * m_hat / (np.sqrt(v2_hat) + eps)
            v -= step

            # optional early stopping if objective stagnates (very cheap)
            if global_it % 4000 == 0:
                # re‑evaluate exact objective every few thousand steps – cheap check
                cur_obj = _objective(_softmax(v))
                if cur_obj < 1e-12:          # absurdly low → break
                    break

    return _softmax(v)


# ----------------------------------------------------------------------
# Sub‑gradient / exponentiated‑gradient refinements (same as before)
# ----------------------------------------------------------------------
def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # gradient of the active entry (ordered‑pair count)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubProj] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine builds a rich pool of analytic and
    random seeds, selects the most promising few, refines them with an
    Adam optimiser on a smooth surrogate of the max‑convolution and finally
    polishes the best candidate with long sub‑gradient / exponentiated‑gradient
    passes.  The best sequence is stored in ``height_sequence_1`` and returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for I/O

    # ------------------------------------------------------------------
    # Choose a resolution – 4000 gives a good trade‑off between speed
    # and the theoretical lower bound (≈1.0).
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation + cheap exact ranking
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # Evaluate the true objective (fast, no clipping)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"Top {top_k} seeds (exact objective):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the smooth surrogate
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] current best exact = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached during Adam passes.")
            break

        # allocate a slice of remaining time for this seed
        remaining = deadline - time.time()
        # target total iterations for Adam (more for the later seeds)
        max_iters = min(30_000, int(remaining * 0.06 * 1e5))

        print(f"\n--- Adam optimisation on seed #{idx+1} (max_iters≈{max_iters}) ---")
        seq_opt = _adam_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iters_total=max_iters)
        val_opt = _objective(seq_opt)
        print(f"[Adam] exact objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best = {global_best_val:.6f}")

        # quick sub‑gradient polishing (few thousand iters)
        if time.time() < deadline:
            sub_seq, sub_val = _subgradient_proj(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=120_000)
            if sub_val < global_best_val:
                global_best_seq = sub_seq.copy()
                global_best_val = sub_val
                print(f"[SubProj‑quick] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long polishing phase (sub‑gradient + exp‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate remaining time roughly equally to the two refinements
        max_sub = int(min(500_000, remaining * 0.45 * 1e5))
        max_exp = int(min(500_000, remaining * 0.45 * 1e5))

        if max_sub > 0:
            print("\n--- Extended sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_proj(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubProj‑ext] improved to {global_best_val:.6f}")

        if time.time() < deadline and max_exp > 0:
            print("\n--- Exponentiated‑gradient polishing ---")
            seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad‑ext] improved to {global_best_val:.6f}")

        print(f"\n=== Finished ===\nExact surrogate = {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")

        print("Warning: evaluate_sequence raised an exception:", exc)
