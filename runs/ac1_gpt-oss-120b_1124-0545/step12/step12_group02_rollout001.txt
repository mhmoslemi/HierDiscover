"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm works in three stages:

1.  Generate a diverse pool of ℓ₁‑normalised “seed” windows
    (uniform, triangular, sine, Gaussian, Dirichlet draws, low‑frequency
    cosine mixtures, and the previous best solution if available).  The
    best few seeds are kept as starting points.

2.  Upsample the best seed to a finer grid (default 12 000 points) and
    run a long *exponentiated‑gradient* optimisation on the exact
    max‑convolution objective.  The update is performed on the log‑softmax
    (probability simplex) representation with a slowly decaying step‑size.
    Iterations continue until only a few seconds of the global budget are
    left.

3.  A greedy *local‑reallocation* phase reshuffles the mass on random
    subsets of coordinates (Dirichlet redistributions).  Only moves that
    strictly improve the surrogate objective are accepted, guaranteeing a
    monotone descent.

All stages respect the 1000 s deadline.  The final best sequence is
returned as a plain list of floats and stored globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(seq, seq)) for an ℓ₁‑normalised vector
    (∑seq == 1).  This equals the true ``evaluate_sequence`` value.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all returned ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few random cosine components, clip to non‑negative and normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (diverse ℓ₁‑normalised candidates)
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    for sigma in np.linspace(0.10, 0.35, 5):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws (several concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(3):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency cosine mixtures
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        curve = _lowfreq_curve(n, 3, rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution, if provided
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # robust normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Upsampling utility (linear interpolation, ℓ₁‑renormalise)
# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (mirrored on the simplex)
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
    objective.  Runs until the deadline and keeps the best vector found.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active (max) convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        # decreasing step size – slower decay to keep mobility late in the run
        eta = 0.5 / (it ** 0.55)  # 0.55 gives a gentle slowdown
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Greedy local re‑allocation (Dirichlet reshuffling of a random subset)
# ----------------------------------------------------------------------
def _greedy_local_search(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: pick a random subset of coordinates,
    redistribute its total mass via a Dirichlet draw, keep the change
    iff the surrogate improves.  Iterates until the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0

    while time.time() < deadline:
        it += 1
        k = rng.integers(2, 9)                     # subset size 2 … 8
        idx = rng.choice(n, size=k, replace=False)
        total = a[idx].sum()
        if total <= 0.0:
            continue
        # moderate concentration encourages both exploration and exploitation
        conc = 0.5
        new_alloc = rng.dirichlet(np.full(k, conc)) * total
        new_a = a.copy()
        new_a[idx] = new_alloc

        cur_val = _objective_surrogate(new_a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = new_a.copy()
            a = new_a
            if it % 2000 == 0:
                print(f"[Greedy] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point (the function that will be called)
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best found sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final call and possible prints
    overall_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Stage 0 – generate a diverse seed pool (coarse resolution)
    # ------------------------------------------------------------------
    n0 = 4000
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print("\n=== Seed generation (n = {}) ===".format(n0))
    seeds = _collect_seeds_extended(n0, rng, prev_arr)

    # cheap surrogate ranking – keep a handful of the best
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print("Top %d surrogate values (lower is better):" % top_k)
    for i, (_, v) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # start from the absolute best seed
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample to a finer grid (more degrees of freedom)
    # ------------------------------------------------------------------
    n_up = 12000                     # larger grid helps push the max down
    print(f"\n=== Upsampling to n = {n_up} ===")
    best_seq = _upsample_sequence(best_seq, n_up)
    best_seq = best_seq / best_seq.sum()
    best_val = _objective_surrogate(best_seq)
    print(f"[Upsample] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long exponentiated‑gradient optimisation
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        print("\n=== Exponentiated‑gradient optimisation ===")
        # give it most of the remaining budget, but leave a small buffer
        exp_deadline = overall_deadline - 8.0
        seq_exp, val_exp = _exp_gradient_optimize(best_seq, n_up, exp_deadline)
        if val_exp < best_val:
            best_seq, best_val = seq_exp, val_exp
            print(f"[ExpGrad] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – greedy local re‑allocation (hill‑climbing)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        print("\n=== Greedy local search ===")
        local_deadline = overall_deadline - 0.5
        seq_local, val_local = _greedy_local_search(best_seq, n_up, local_deadline, rng)
        if val_local < best_val:
            best_seq, best_val = seq_local, val_local
            print(f"[Greedy] new global best = {val_local:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
