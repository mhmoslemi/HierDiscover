"""
Non‑negative step‑function optimiser (enhanced).

The routine searches for a probability vector a (a_i ≥ 0, Σ a_i = 1)
that minimises

        2 * n * max(conv(a, a))

where conv(a, a) is the linear convolution of a with itself.
The objective is the same value returned by ``evaluate_sequence``
(the denominator cancels because we keep Σ a_i = 1).

Main ideas
~~~~~~~~~~
* A fairly large pool of *seed* windows (geometric, power‑law,
  combined families, classic DSP windows, a few random soft‑max
  vectors and the previous best construction if available) is built.
* For each seed an Adam optimiser works on the *log‑softmax*
  parametrisation (v → a = softmax(v)).  The loss is a smooth
  approximation of the max, controlled by a temperature β that
  is increased geometrically.
* Hyper‑parameters (learning‑rate, β‑schedule, Adam moments) are
  randomly perturbed per‑seed – this diversifies the search.
* After the first pass the current best vector is refined many times
  with a smaller learning‑rate and a larger β (almost the true max).
* Finally, a cheap stochastic hill‑climber (Gaussian perturbation +
  projection onto the simplex) is executed while time permits.
* The whole procedure respects a user supplied time budget (default
  1000 s) and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, softmax)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using an FFT of size
    a power of two.  Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the algorithm from
    Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("only 1‑D vectors are supported")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of seed windows
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1 / (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric × power‑law a_i ∝ r**i / (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric shape a_i ∝ r**i + r**(n‑1‑i)."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _gaussian_window(n: int, sigma: float = 0.4) -> np.ndarray:
    """Discrete Gaussian (σ expressed as a fraction of the length)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / (sigma * centre)) ** 2)
    return a / a.sum()


# ----------------------------------------------------------------------
# Core optimiser for a single seed & length
# ----------------------------------------------------------------------


def _optimise_one_seed(
    a0: np.ndarray,
    n: int,
    deadline: float,
    *,
    lr0: float,
    beta1: float,
    beta2: float,
    eps_adam: float,
    beta_start: float,
    beta_mul: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the log‑softmax parameters of a single seed vector.
    Returns the best (a, value) pair seen during the run.
    """
    # initialise log‑softmax parameters
    v = np.log(a0 + 1e-30)          # avoid log(0)
    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    t = 0

    best_a = None
    best_val = float("inf")

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # always on the simplex

        # evaluate current objective (fast version)
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasional log output
            print(
                f"[adam] n={n:5d} it={it:5d} val={cur_val:.6f}"
                f" lr={lr0/np.sqrt(t):.4e}"
            )

        # ----------- smoothed‑max weights -------------
        beta = beta_start * (beta_mul ** it)
        # shift = cur_max makes the maximal entry exponent = 0 (no overflow)
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:          # numerics – skip this iteration
            continue
        w /= w_sum                # normalized softmax over lag indices

        # ----------- gradient w.r.t. a (smooth) -------
        # grad_a = 2 * (rev(a) * w) restricted to the central n entries
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]     # central slice

        # ----------- convert to gradient w.r.t. v (softmax) ----------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)                     # simple decay
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climber – cheap local search around a good point
# ----------------------------------------------------------------------


def _local_random_search(
    a_start: np.ndarray,
    n: int,
    deadline: float,
    *,
    steps: int,
    sigma_init: float = 0.02,
    decay: float = 0.95,
) -> tuple[np.ndarray, float]:
    """
    Simple random‑perturbation hill‑climber.
    Perturb the log‑softmax vector, project back onto the simplex,
    and keep improvements.  The step-size (Gaussian sigma) decays.
    """
    v_center = np.log(a_start + 1e-30)
    best_a = a_start.copy()
    conv = _conv_fft(best_a, best_a)
    best_val = 2.0 * n * np.max(conv)

    sigma = sigma_init

    while steps > 0 and time.time() < deadline:
        # propose a perturbed logit vector
        v_prop = v_center + sigma * np.random.randn(n)
        a_prop = _softmax(v_prop)

        conv = _conv_fft(a_prop, a_prop)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_prop.copy()
            v_center = np.log(best_a + 1e-30)  # recentre the walk
            print(f"[random] new local best={cur_val:.6f} sigma={sigma:.4f}")

        sigma *= decay
        steps -= 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_unused):
    """
    Search for a non‑negative height sequence that minimises the
    ``evaluate_sequence`` objective.  The routine runs for at most
    ``budget_s`` seconds (default 1000) and returns the best sequence
    encountered as a plain Python list of floats.
    """
    np.random.seed(seed)
    # leave a small safety margin for finalisation
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Lengths to explore – start with 4000, then 8000 if time permits,
    # finally a third larger length if we still have time.
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Iterate over the chosen lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        # --------------------------------------------------------------
        # Build a diversified pool of seed vectors for this length
        # --------------------------------------------------------------
        seeds = []

        # Geometric families (r = 1 - c/n)
        for c in np.linspace(0.5, 5.0, 12):
            r = 1.0 - c / n
            if r > 0.0:
                seeds.append(_geom(n, r))

        # Pure power‑law families
        for p in np.linspace(0.0, 2.2, 12):
            seeds.append(_power(n, p))

        # Combined geometric × power families
        for c in np.linspace(0.5, 4.0, 8):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 8):
                seeds.append(_geom_power(n, r, p))

        # Symmetric geometric
        for c in np.linspace(0.5, 4.5, 8):
            r = 1.0 - c / n
            if r > 0.0:
                seeds.append(_sym_geom(n, r))

        # Classical DSP windows
        seeds.append(_sine_window(n))
        seeds.append(_hann_window(n))
        seeds.append(_hamming_window(n))
        seeds.append(_blackman_window(n))
        seeds.append(_gaussian_window(n, sigma=0.35))
        seeds.append(_gaussian_window(n, sigma=0.45))

        # A few random soft‑max initialisations
        for _ in range(6):
            v_rand = np.random.randn(n)
            seeds.append(_softmax(v_rand))

        # Re‑use a previous construction if it exists
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                a_prev = np.asarray(prev, dtype=np.float64)
                if a_prev.ndim == 1:
                    if a_prev.shape[0] == n:
                        seeds.append(a_prev / a_prev.sum())
                    else:
                        # truncate / pad to desired size
                        a_adj = np.zeros(n, dtype=np.float64)
                        m = min(len(a_prev), n)
                        a_adj[:m] = a_prev[:m]
                        a_adj = a_adj / a_adj.sum()
                        seeds.append(a_adj)
            except Exception:
                pass

        # --------------------------------------------------------------
        # Quick scan of all seeds using the cheap internal evaluator
        # --------------------------------------------------------------
        for a in seeds:
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] len={n} new global best={cur_val:.6f}")

        # --------------------------------------------------------------
        # Adam optimisation on every seed (time‑aware)
        # --------------------------------------------------------------

        # Hyper‑parameter ranges (we randomise per seed to improve exploration)
        lr0_base = 0.8
        beta_start_base = 3.0
        beta_mul_base = 1.018

        for s_idx, a0 in enumerate(seeds):
            # sample slight variations to diversify the optimisation runs
            lr0 = lr0_base * np.exp(0.2 * (np.random.rand() - 0.5))
            beta_start = beta_start_base * np.exp(0.2 * (np.random.rand() - 0.5))
            beta_mul = beta_mul_base * np.exp(0.01 * (np.random.rand() - 0.5))

            # allow up to 15000 iterations per seed (bounded by remaining time)
            max_iters = min(15000, int(1e5 * (deadline - time.time())))  # generous upper bound

            seq, val = _optimise_one_seed(
                a0,
                n,
                deadline,
                lr0=lr0,
                beta1=0.9,
                beta2=0.999,
                eps_adam=1e-12,
                beta_start=beta_start,
                beta_mul=beta_mul,
                max_iters=max_iters,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall] len={n} seed={s_idx} NEW GLOBAL BEST={val:.6f}")

        # --------------------------------------------------------------
        # Local random‑search refinement (few thousand cheap proposals)
        # --------------------------------------------------------------
        if time.time() < deadline and global_best_seq is not None:
            # we operate on the best vector found for this length
            a_start = global_best_seq.copy()
            steps = 5000
            seq, val = _local_random_search(
                a_start,
                n,
                deadline,
                steps=steps,
                sigma_init=0.015,
                decay=0.97,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[refine] len={n} random search improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("[fallback] using uniform distribution")

    # Return as a plain Python list (compatible with the caller)
    return global_best_seq.tolist()
