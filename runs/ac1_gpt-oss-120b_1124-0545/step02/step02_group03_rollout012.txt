"""
Hybrid projected gradient descent (hard mask + softmax) with adaptive step size.
The algorithm starts with a sub‑gradient step using the active set (binary mask)
to quickly reduce the dominant autocorrelation peak, then gradually switches to
a smooth log‑sum‑exp approximation of the max using a temperature schedule.
Updates are projected onto the probability simplex and occasional random
mass‑transfer moves are performed to escape plateaus.  Restarts are triggered
if no improvement has been seen for a while.  The best sequence found within
the time budget is returned.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of real vectors via FFT (real‑to‑real)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]

def _max_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full autocorrelation a * a using a pre‑computed FFT length."""
    return _conv_fft(a, a, size)

def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Move a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise to keep ∑a = 1
    a_new /= a_new.sum()
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``2·n·max(conv(a,a))/(∑a)²`` within the given time budget.
    The routine returns the best list of floats found.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 8000))

    # --------------------------------------------------------------
    # initialise a starting distribution
    # --------------------------------------------------------------
    try:
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            seq0 = np.asarray(seq0, dtype=float)
            if seq0.size != n:
                # linear interpolation to the required length
                x_old = np.linspace(0, 1, seq0.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, seq0)
            else:
                a = seq0.copy()
        else:
            raise KeyError
    except Exception:
        a = np.random.rand(n)

    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.ones(n)
    a /= a.sum()   # keep Σa = 1 (simplex)

    # --------------------------------------------------------------
    # pre‑compute FFT lengths (powers of two)
    # --------------------------------------------------------------
    size_conv = _next_pow2(2 * n - 1)       # for a * a
    size_grad = _next_pow2(3 * n - 2)       # for a_rev * weights

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    init_step = 0.45          # base step for projected gradient
    rand_prob = 0.07          # probability of a random mass‑transfer move
    rand_scale = 0.03         # relative size of random transfer
    warmup_iters = 20000      # iterations before switching to softmax
    alpha_start = 0.2
    alpha_end = 250.0
    max_iters_est = 400000    # rough estimate for scaling schedules
    stagnation_secs = 30.0    # restart if no improvement for this long
    tolerance_factor = 1e-6   # for active‑set mask

    # bookkeeping
    conv_full = _max_convolution(a, size_conv)
    max_b = conv_full.max()
    best_score = 2.0 * n * max_b
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    last_improve = start
    it = 0

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # evaluate current objective
        conv_full = _max_convolution(a, size_conv)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        # keep best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # decide on move
        if np.random.rand() < rand_prob:
            # cheap exploration move
            a = _random_transfer(a, rand_scale)
        else:
            # gradient step
            if it < warmup_iters:
                # classic sub‑gradient using the active set (binary mask)
                tol = max(tolerance_factor * max_b, 1e-12)
                weights = (conv_full >= max_b - tol).astype(float)
            else:
                # smooth approximation of the max via softmax / log‑sum‑exp
                progress = min((it - warmup_iters) / max_iters_est, 1.0)
                alpha = alpha_start * (alpha_end / alpha_start) ** (progress ** 1.5)
                # shifted exponent to avoid overflow (max_b is subtracted)
                exp_vals = np.exp(alpha * (conv_full - max_b))
                sum_exp = exp_vals.sum()
                if sum_exp == 0.0 or not np.isfinite(sum_exp):
                    # fallback to hard mask if numerical trouble
                    weights = (conv_full >= max_b - 1e-12).astype(float)
                else:
                    weights = exp_vals / sum_exp

            # compute gradient = 2 * (a_rev * weights)_{central n}
            a_rev = a[::-1]
            full = _conv_fft(a_rev, weights, size_grad)
            grad = 2.0 * full[n - 1 : n - 1 + n]

            # projected gradient descent step with diminishing step size
            step = init_step / np.sqrt(it + 1.0)
            a_new = a - step * grad
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s == 0.0:
                a_new = np.ones_like(a) / n
            else:
                a_new /= s
            a = a_new

        # occasional restart if stagnating
        if time.time() - last_improve > stagnation_secs:
            # re‑initialise from the best known sequence with a small perturbation
            perturb = np.random.rand(n) * 0.01
            a = best_seq + perturb
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            print(f"[{it:7d}] restart due to stagnation")
            last_improve = time.time()   # reset timer

        it += 1

    # final report
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()

"""
Hybrid projected gradient descent for minimizing
    2·n·max(conv(a,a))/(∑a)²

The algorithm works on the probability simplex (∑a = 1) and proceeds in two
phases:

1. **Active‑set (hard‑mask) phase** – for the first few thousand iterations
   we use a classic sub‑gradient based on the indices where the autocorrelation
   attains its maximum.  This quickly reduces the dominant peak.

2. **Soft‑max (log‑sum‑exp) phase** – thereafter we replace the hard mask by a
   smooth approximation of the max using a temperature schedule (α grows from
   `alpha_start` to `alpha_end`).  The gradient of the smooth objective is
   computed via FFT‑based convolution and a simple projected gradient step with
   a diminishing step size is performed.

Occasional random “mass‑transfer’’ moves (probability `rand_prob`) help the
search escape flat plateaus, and a restart is triggered if no improvement has
been observed for `stagnation_secs` seconds.  The best sequence seen before
the time budget expires is returned.

All helper functions are top‑level; there are no closures or lambda
expressions.  The code respects the 1000 s budget and prints progress
messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (FFT‑based convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of real vectors via FFT (real‑to‑real)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _max_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full autocorrelation a * a."""
    return _conv_fft(a, a, size)


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns the best sequence (as a list of Python floats) found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # problem size (can be overridden via kwargs)
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 8000))

    # --------------------------------------------------------------
    # initialise a starting distribution
    # --------------------------------------------------------------
    try:
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            seq0 = np.asarray(seq0, dtype=float)
            if seq0.size != n:
                # linear interpolation to the required length
                x_old = np.linspace(0, 1, seq0.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, seq0)
            else:
                a = seq0.copy()
        else:
            raise KeyError
    except Exception:
        a = np.random.rand(n)

    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.ones_like(a)
    a /= a.sum()                     # keep Σa = 1 (simplex)

    # --------------------------------------------------------------
    # pre‑compute FFT lengths
    # --------------------------------------------------------------
    size_conv = _next_pow2(2 * n - 1)       # for a * a
    size_grad = _next_pow2(3 * n - 2)       # for a_rev * weight vector

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    init_step = 0.45          # base step size for projected gradient
    rand_prob = 0.07          # probability of a random transfer move
    rand_scale = 0.03         # relative magnitude of a transfer move
    warmup_iters = 20000      # iterations using the hard‑mask sub‑gradient
    alpha_start = 0.2
    alpha_end = 250.0
    max_iters_est = 400000    # for scaling the temperature schedule
    tolerance_factor = 1e-6   # active‑set tolerance (relative)
    stagnation_secs = 30.0    # restart if no improvement for this long

    # bookkeeping
    conv_full = _max_convolution(a, size_conv)
    max_b = conv_full.max()
    best_score = 2.0 * n * max_b        # because Σa = 1
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    last_improve = start
    it = 0

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- evaluate current objective -----
        conv_full = _max_convolution(a, size_conv)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ----- decide which move to apply -----
        if np.random.rand() < rand_prob:
            # cheap exploration
            a = _random_transfer(a, rand_scale)
        else:
            # ----- compute the weight vector (mask or soft‑max) -----
            if it < warmup_iters:
                # hard‑mask (active set) sub‑gradient
                tol = max(tolerance_factor * max_b, 1e-12)
                weights = (conv_full >= max_b - tol).astype(float)
            else:
                # smooth approximation of the max via log‑sum‑exp
                progress = min((it - warmup_iters) / max_iters_est, 1.0)
                # raise the temperature slowly (exponential schedule)
                alpha = alpha_start * (alpha_end / alpha_start) ** (progress ** 1.5)
                # shifted exponent to avoid overflow: max_b is subtracted
                exp_vals = np.exp(alpha * (conv_full - max_b))
                sum_exp = exp_vals.sum()
                if sum_exp == 0.0 or not np.isfinite(sum_exp):
                    # fall back to hard mask on numerical trouble
                    weights = (conv_full >= max_b - 1e-12).astype(float)
                else:
                    weights = exp_vals / sum_exp

            # ----- gradient of the (soft) max -----
            a_rev = a[::-1]
            full = _conv_fft(a_rev, weights, size_grad)
            grad = 2.0 * full[n - 1 : n - 1 + n]   # central n entries

            # ----- projected gradient step -----
            step = init_step / np.sqrt(it + 1.0)
            a_new = a - step * grad
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s == 0.0:
                a_new = np.ones_like(a) / n
            else:
                a_new /= s
            a = a_new

        # ----- restart if stagnating -----
        if time.time() - last_improve > stagnation_secs:
            # perturb the current best a little and continue
            a = best_seq + 0.01 * np.random.rand(n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            print(f"[{it:7d}] restart due to stagnation")
            last_improve = time.time()

        it += 1

    # --------------------------------------------------------------
    # final report
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
