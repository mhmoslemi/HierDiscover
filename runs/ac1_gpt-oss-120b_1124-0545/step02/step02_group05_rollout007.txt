Propose a step‑function sequence that minimizes the evaluation function.

The algorithm proceeds in three stages within the 1000 s budget:

1. **Flattening (variance minimisation).**  
   We minimise the squared deviation of the self‑convolution from its
   average value, i.e.  ∑ₖ ( (a*a)[k] – μ )²  where μ = (∑a)²/(2n−1).  
   This objective is smooth and its gradient can be computed by a single
   convolution.  An Adam optimiser with projection onto the ℓ₁‑simplex
   (∑a fixed, aⱼ ≥ 0) drives the sequence towards a flat autocorrelation,
   which strongly reduces the maximum entry.

2. **Fine‑tuning (log‑sum‑exp smoothing).**  
   When ~60 % of the time budget has elapsed we switch to a smoothed
   surrogate for the true max (log‑sum‑exp) and continue Adam updates.
   This sharpens the focus on the current peak while still keeping the
   optimisation stable.

3. **Local hill‑climbing.**  
   In the final ~30 % of the budget we perform cheap O(n) moves:
   transfer a small mass δ from a random index i to another j, update the
   convolution incrementally, and accept the move only if the maximal
   convolution entry decreases.  The step size shrinks with time, and we
   keep the best sequence seen throughout the run.

All updates respect the non‑negativity and ℓ₁‑norm constraints, and the
best sequence found is returned as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function height vector that minimises
    `evaluate_sequence`.  The routine combines a variance‑flattening phase,
    a log‑sum‑exp fine‑tuning phase and a final hill‑climbing refinement.
    """
    # --------------------------------------------------------------
    # reproducibility
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a0 = np.asarray(height_sequence_1, dtype=float)
    else:
        n = 2500               # safe default
        a0 = np.random.exponential(scale=1.0, size=n)

    target_sum = math.sqrt(2.0 * n)               # ∑a = √(2n)
    a0 = a0 / a0.sum() * target_sum               # normalise onto simplex

    # --------------------------------------------------------------
    # initialise state
    a = a0.copy()
    conv = np.convolve(a, a)                     # current self‑convolution
    cur_max = float(conv.max())
    best_seq = a.copy()
    best_score = cur_max
    best_it = 0

    # ADAM moments (used in both optimisation phases)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    adam_t = 0

    # constant average of the convolution (independent of a after normalisation)
    avg_conv = target_sum ** 2 / (2 * n - 1)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    it = 0
    last_improve = 0

    # --------------------------------------------------------------
    # main loop – adapt the optimisation strategy according to elapsed time
    while time.time() < deadline:
        elapsed = time.time() - start_time
        fractional = elapsed / (deadline - start_time)   # 0 → 1

        # ------------------------------------------------------------------
        # Phase 1 : variance flattening (≈ 0 – 0.60 of the budget)
        if fractional < 0.60:
            # ---- learning‑rate schedule (decrease exponentially) ----------
            lr_start = target_sum * 0.005   # about 0.35 for n≈2500
            lr_end   = lr_start * 0.05      # final ≈ 0.0175
            lr = lr_start * ((lr_end / lr_start) ** fractional)

            # ---- compute gradient of  Σ (conv‑avg)² -------------------------
            conv = np.convolve(a, a)                 # length 2n‑1
            D = conv - avg_conv                       # zero‑mean error
            a_rev = a[::-1]
            grad_full = np.convolve(D, a_rev)        # length 3n‑2
            grad = 4.0 * grad_full[n - 1:n - 1 + n]   # slice of length n

            # ---- Adam update ---------------------------------------------
            adam_t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** adam_t)
            v_hat = v / (1.0 - beta2 ** adam_t)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            a_new = _simplex_projection(a_new, target_sum)

            a = a_new
            conv = np.convolve(a, a)
            cur_max = float(conv.max())

            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_seq = a.copy()
                last_improve = it
                print(f"[{it:6d}] Var‑phase improve: max(conv) = {best_score:.6f} "
                      f"(t={elapsed:.1f}s)")

        # ------------------------------------------------------------------
        # Phase 2 : smoothed max (≈ 0.60 – 0.90 of the budget)
        elif fractional < 0.90:
            # smoothing parameter schedule (log‑sum‑exp)
            eps_start, eps_end = 0.5, 0.001
            eps = eps_start * ((eps_end / eps_start) ** fractional)

            lr_start = target_sum * 0.001
            lr_end   = target_sum * 0.00005
            lr = lr_start * ((lr_end / lr_start) ** fractional)

            # current convolution (already up‑to‑date from previous step)
            conv = np.convolve(a, a)

            # soft‑max weights (stability shift)
            w_shift = conv - conv.max()
            w_raw = np.exp(w_shift / eps)
            w_sum = w_raw.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = w_raw / w_sum

            # gradient of smooth maximum (same derivation as the reference code)
            # cumulative sums to obtain σ
            cumsum_w = np.cumsum(w)
            sigma = cumsum_w[n - 1:2 * n - 1] - np.concatenate(([0.0], cumsum_w[:n - 1]))

            a_rev = a[::-1]
            conv_wa = np.convolve(w, a_rev)                # length 3n‑2
            first = conv_wa[n - 1:n - 1 + n]

            grad = first + a * sigma                       # size n

            # Adam update
            adam_t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** adam_t)
            v_hat = v / (1.0 - beta2 ** adam_t)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            a_new = _simplex_projection(a_new, target_sum)

            a = a_new
            conv = np.convolve(a, a)
            cur_max = float(conv.max())

            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_seq = a.copy()
                last_improve = it
                print(f"[{it:6d}] Smooth‑max improve: max(conv) = {best_score:.6f} "
                      f"(t={elapsed:.1f}s)")

        # ------------------------------------------------------------------
        # Phase 3 : hill‑climbing (≈ 0.90 – 1.00 of the budget)
        else:
            # Ensure we have the current convolution cached
            if conv is None or conv.shape[0] != 2 * n - 1:
                conv = np.convolve(a, a)

            # Choose a source index with some mass
            i = random.randint(0, n - 1)
            if a[i] <= 1e-12:
                it += 1
                continue
            # Choose a destination index different from i
            j = random.randint(0, n - 1)
            if j == i:
                it += 1
                continue

            # Step size shrinks as we approach the deadline
            remaining = deadline - time.time()
            max_step_frac = 0.3 * (remaining / (deadline - start_time)) + 0.01
            delta = random.random() * a[i] * max_step_frac
            if delta <= 0:
                it += 1
                continue

            di = -delta
            dj = +delta

            # Incremental update of the convolution (O(n))
            conv_new = conv.copy()
            conv_new[i:i + n] += 2 * di * a
            conv_new[j:j + n] += 2 * dj * a
            conv_new[i + j] += 2 * di * dj
            conv_new[2 * i] += di * di
            conv_new[2 * j] += dj * dj

            new_max = float(conv_new.max())
            cur_max = float(conv.max())

            if new_max < cur_max - 1e-12:
                # Accept move
                a[i] += di
                a[j] += dj
                conv = conv_new
                cur_max = new_max
                if cur_max < best_score - 1e-12:
                    best_score = cur_max
                    best_seq = a.copy()
                    last_improve = it
                    print(f"[{it:6d}] Hill‑climb improve: max(conv) = {best_score:.6f} "
                          f"(t={elapsed:.1f}s)")

            # occasional large perturbation if stuck
            if it - last_improve > 3000:
                # reset a mixture of the current best and fresh noise
                noise = np.random.exponential(scale=1.0, size=n)
                noise = noise / noise.sum() * target_sum
                a = 0.5 * best_seq + 0.5 * noise
                a = _simplex_projection(a, target_sum)
                conv = np.convolve(a, a)
                last_improve = it

        # ------------------------------------------------------------------
        it += 1

    # ----------------------------------------------------------------------
    # Return the best sequence as a plain list
    return best_seq.tolist()
