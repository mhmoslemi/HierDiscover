def _simplex_projection(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    # based on algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.nonzero(u - css / np.arange(1, n+1) > 0)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Proposes a non‑negative step‑function sequence with a low evaluation score.

Main ideas
---------
* Initialise the sequence from a flexible parametric family (power‑law,
  piecewise‑power‑law).  The best of a few hundred random parameter choices
  is taken as a start.
* Refine the start by a projected gradient descent on a *smooth* approximation
  of the objective   max(conv(seq,seq)).   The smooth max is realised by a
  soft‑max with an annealing parameter β.  The gradient of the smoothed
  objective can be written as a convolution, so each iteration costs only
  O(n log n) time.
* After every (failed) step the step‑size is halved; after a successful
  step it is increased slightly.  β is increased slowly (annealed) to
  focus the optimisation on the true maximum.
* Simple random mass‑exchange moves are injected every few hundred
  iterations to escape shallow plateaus.
* The whole optimisation runs for at most ``budget_s`` seconds (default
  1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised power‑law sequence a_i ∝ (i+1+offset)^{-α}.
    The returned array has sum == 1 and all entries ≥ 0.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _piecewise_sequence(
    n: int,
    alpha1: float,
    alpha2: float,
    split: int,
    offset: float = 0.0,
) -> np.ndarray:
    """
    Piecewise power‑law:
        a_i ∝ (i+1+offset)^{-α1}   for i < split
        a_i ∝ (i+1+offset)^{-α2}   for i ≥ split
    Normalised to sum == 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = np.empty(n, dtype=float)
    a[:split] = idx[:split] ** (-alpha1)
    a[split:] = idx[split:] ** (-alpha2)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _mixture_sequence(
    n: int,
    alpha1: float,
    alpha2: float,
    weight: float,
    offset: float = 0.0,
) -> np.ndarray:
    """
    Mixture of two power‑laws:
        a = λ·(i+1+offset)^{-α1} + (1-λ)·(i+1+offset)^{-α2}
    Normalised to sum == 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a1 = idx ** (-alpha1)
    a2 = idx ** (-alpha2)
    a = weight * a1 + (1.0 - weight) * a2
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _score_from_sequence(seq: np.ndarray) -> float:
    """
    Exact score used by ``evaluate_sequence`` for a normalised sequence
    (i.e. sum(seq) == 1).  The function is cheap because it avoids the
    security checks performed by ``evaluate_sequence``.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ∈ ℝⁿ | x_i ≥ 0, Σ x_i = 1 }
    Implementation of the algorithm from Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u > css / np.arange(1, n + 1))[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_gradient(seq: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where ``grad`` is the gradient of the
    smoothed max‑convolution function
          f_β(a) = (1/β)·log Σ_k exp(β·conv_k(a))
    and ``max_conv`` is the actual maximum entry of the convolution
    (used to compute the true objective).
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)                 # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (stable computation)
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / exp_term.sum()
    # gradient: 2·(w * a)  (convolution of w with reversed a)
    grad_full = np.convolve(w, seq[::-1])
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # slice to length n
    return grad, max_c


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Searches for a non‑negative step‑function sequence that minimises the
    ``evaluate_sequence`` score.  The function respects the ``budget_s`` time
    limit (seconds) and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds margin for finalisation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 5000          # a good trade‑off between quality and speed

    # ------------------------------------------------------------------
    # 2. Randomised parametric initialisation
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # a) pure power‑law grid (coarse)
    for alpha in np.linspace(0.45, 0.60, 31):
        seq = _power_sequence(n, float(alpha))
        sc = _score_from_sequence(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[init] power α={alpha:.4f} → score={sc:.6f}")

    # b) random mixture / piecewise families
    trials = 120
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # randomly decide which family to use
        family = np.random.choice(["piecewise", "mixture", "power"])
        if family == "piecewise":
            a1 = np.random.uniform(0.35, 0.7)
            a2 = np.random.uniform(0.6, 1.4)
            split = np.random.randint(int(0.2 * n), int(0.8 * n))
            off = np.random.uniform(0.0, 5.0)
            seq = _piecewise_sequence(n, a1, a2, split, off)
        elif family == "mixture":
            a1 = np.random.uniform(0.35, 0.7)
            a2 = np.random.uniform(0.7, 1.4)
            w = np.random.uniform(0.0, 1.0)
            off = np.random.uniform(0.0, 5.0)
            seq = _mixture_sequence(n, a1, a2, w, off)
        else:  # pure power‑law with random offset
            a = np.random.uniform(0.45, 0.60)
            off = np.random.uniform(0.0, 5.0)
            seq = _power_sequence(n, a, off)

        sc = _score_from_sequence(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[init] {family} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected smooth‑gradient descent
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    cur_score = best_score
    step = 1.0
    min_step = 1e-9
    beta = 5.0                     # start with a gentle smoothing
    iteration = 0
    stagnation = 0                # counts iterations without improvement

    while time.time() < deadline:
        iteration += 1

        # gradient of the smooth approximation + true max value
        grad, max_conv = _smooth_gradient(seq, beta)
        cur_score = 2.0 * n * max_conv

        # ---- back‑tracking line search (projected) ----
        trial_step = step
        for _ in range(12):                     # at most 12 halvings
            cand = seq - trial_step * grad
            cand = _project_onto_simplex(cand)
            # exact score (max convolution) for the trial vector
            conv = np.convolve(cand, cand)
            cand_score = 2.0 * n * conv.max()
            if cand_score < cur_score - 1e-12:
                break
            trial_step *= 0.5

        # ---- accept / reject ----
        if cand_score < cur_score - 1e-12:
            seq = cand
            cur_score = cand_score
            step = trial_step * 1.12               # modest increase after success
            stagnation = 0
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(
                    f"[improve] iter={iteration} β={beta:.2f} step={step:.6f}"
                    f" → score={best_score:.6f}"
                )
        else:
            step *= 0.5
            stagnation += 1
            # after a while increase β to focus on the true maximum
            if stagnation >= 60:
                beta = min(beta * 1.5, 200.0)
                stagnation = 0

        # ---- periodic random mass exchange (escape plateaus) ----
        if iteration % 250 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(0.001, 0.1 * seq[i])
            cand = seq.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = _project_onto_simplex(cand)
            seq = cand  # never reject this cheap move

        # stop if step becomes tiny
        if step < min_step:
            break

        # safety check – avoid overrunning the allotted time
        if time.time() >= deadline:
            break

    print(f"[final] best_score={best_score:.6f}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
