    i = np.arange(n)
    a0 = 1.0/np.sqrt((i+0.5)*(n-i-0.5))

    v = np.log(a0 + 1e-12)  # log domain

    best_seq = None
    best_score = np.inf

    # parameters
    max_iters = ... but rely on time
    start = time.time()
    deadline = start + budget_s - 0.5

    # schedule for beta
    beta_start = 1.0
    beta_end = 200.0  # gradually increase

    # Adam parameters
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t_iter = 0
    while time.time() < deadline:
        # compute a via softmax
        a = np.exp(v)
        a /= a.sum()
        # compute conv
        conv = np.convolve(a, a)
        # compute beta

        beta = beta_start * (1 - progress) + beta_end * progress

        z = beta * conv

"""
A smooth‑gradient optimiser for the step‑function bound.

The objective to minimise is

    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2).

Because the expression is scale‑invariant we restrict ourselves to
probability vectors (sum = 1).  Direct sub‑gradient descent that only looks
at the current maximal convolution entry (as in the previous solution)
gets stuck in a narrow basin.  Here we replace the hard max by a smooth
log‑sum‑exp (soft‑max) with a temperature β that is increased during the
run.  The gradient of the smoothed objective is a weighted sum of all
convolution entries, which provides a much richer descent direction.
The variables are kept on the probability simplex by parametrising them
as a soft‑max of unconstrained real values v (a_i = softmax(v)_i).

The optimiser uses an Adam‑style update, a linear schedule for the
learning‑rate and for the soft‑max temperature, and occasional random
re‑initialisations when progress stalls.  The best exact score (computed
with the true max) seen so far is returned at the end of the allotted
budget (default 1000 s).  The code works for any length `n` (default
2000) and respects the required API `propose_candidate`.
"""

import time
import numpy as np

# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_shift = v - np.max(v)
    exp_v = np.exp(v_shift)
    return exp_v / exp_v.sum()


def exact_score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a probability vector `a`.
    The sum of `a` is (by construction) 1, so the score reduces to
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def initialise_a(n: int, seed: int = 42) -> np.ndarray:
    """
    Initialise a good starting point.
    If a global variable `height_sequence_1` exists we reuse it,
    otherwise we fall back to the classic arcsine shape.
    """
    rng = np.random.default_rng(seed)

    # Try to reuse a previously found sequence if it exists.
    global_vars = globals()
    if "height_sequence_1" in global_vars:
        a0 = np.array(global_vars["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)               # safety
        if a0.size == n:

    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()
    return a0

    1000 s) and returns the best sequence found.

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------- 1) initial point -------------------
    a = initialise_a(n, seed=seed)
    v = np.log(a + 1e-12)      # log‑parametrisation (soft‑max)

    # ------------------- optimisation hyper‑parameters -----
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    lr_start, lr_end = 0.05, 0.001       # learning‑rate schedule
    beta_start, beta_end = 5.0, 200.0    # soft‑max temperature schedule

            best_score = cur_score
            best_a = a.copy()
            last_improve_t = time.time()
            print(f"[{it:07d}] new best score = {best_score:.6f} (t={elapsed:.1f}s)")

        # ---- occasional stochastic restart to escape plateaus ----
        if it % 5000 == 0:
            if time.time() - last_improve_t > 30.0:
                jitter = np.random.randn(n) * 1e-3
                v = np.log(best_a + 1e-12) + jitter
                m.fill(0.0)
                v_hat.fill(0.0)
                print(f"[{it:07d}] restart after stagnation (t={elapsed:.1f}s)")

    total_time = time.time() - start_time
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total_time:.1f}s")

    # safety: guarantee non‑negativity and finite values
    best_a = np.maximum(best_a, 0.0)
    # The evaluation function caps values >1000, but we stay well below.
    return best_a.tolist()
