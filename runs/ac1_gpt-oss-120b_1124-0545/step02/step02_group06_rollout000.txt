"""
Propose a high‑quality non‑negative step‑function sequence.

Algorithm
---------

1. **Initialisation** – Use the global `height_sequence_1` if it exists,
   otherwise fall back to the classic 1/√(i+1) root law.

2. **Parametric shape search** –  Search a 3‑parameter family
   ``a_i ∝ (i+α+1)^‑p + λ·(n‑1‑i+α+1)^‑p`` (a decreasing part plus a
   reversed part) on the full target length.  Randomly sample
   `(p,α,λ)` in a generous range, keep the shape with the smallest
   evaluation score.  This explores many “U‑shaped’’ or “power‑law’’
   candidates far more efficiently than the pure random‐walk used
   before.

3. **Projected gradient refinement** – Starting from the best shape,
   repeatedly reduce the maximal entry of the self‑convolution.
   For the current active index `k_max = argmax (a*a)`,
   the sub‑gradient of the maximal entry is
   `g_i = 2·a_{k_max‑i}` on the feasible index range.
   We take a gradient step, project back onto the simplex
   (non‑negative, unit sum) and accept the step if it lowers the
   score; otherwise we shrink the step size.  The gradient is cheap
   (O(n)) and the convolution (required to locate `k_max`) is done
   with a real FFT (O(n log n)).  The loop runs until the time budget
   is exhausted.

The combination of a richer parametric family and a deterministic
gradient descent (instead of the mainly stochastic annealing in the
previous code) yields a tighter bound in practice while staying well
within the 1000 s time limit.
"""

def _initial_root_sequence(n: int) -> np.ndarray:
    """Root law a_i ∝ 1/√(i+1) normalised to unit sum."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _conv_max_and_arg(seq: np.ndarray) -> tuple[float, int]:
    """
    Return (max(conv(seq,seq)), argmax) using an FFT based convolution.
    """
    n = seq.size
    size = 2 * n - 1
    # next power of two for a fast real FFT
    m = 1 << ((size - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:size]
    arg = int(np.argmax(conv))
    return float(conv[arg]), arg


def _score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective used inside the search loop.
    The sequence must already be non‑negative and its sum ≈ 1
    (the exact scale does not matter for the objective).
    """
    n = seq.size
    max_b, _ = _conv_max_and_arg(seq)
    # sum(seq) is forced to 1, but we keep the exact formula for safety
    s = seq.sum()
    if s < 0.01:               # mirrors the safety check in evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _generate_power_mix_sequence(
    n: int,
    p: float,
    alpha: float,
    lam: float,
) -> np.ndarray:
    """
    Generate a normalised sequence of length ``n`` with shape

        a_i ∝ (i + α + 1)^‑p + λ·(n‑1‑i + α + 1)^‑p .

    The two terms give a decreasing tail (first) and a reversed tail
    (second); λ controls the amount of mass placed towards the right
    end of the interval.  ``lam = 0`` reduces to a pure power law.
    """
    idx = np.arange(n, dtype=float)

    # first (decreasing) component
    comp1 = (idx + alpha + 1.0) ** (-p)

    # reversed component – increasing when read left‑to‑right
    comp2 = ((n - 1 - idx) + alpha + 1.0) ** (-p)

    a = comp1 + lam * comp2
    total = a.sum()
    if total == 0.0 or np.isnan(total):
        # fall back to the root law – should never happen but protects against
        # extreme parameter choices.
        return _initial_root_sequence(n)
    a /= total
    return a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    This implementation follows the algorithm of Wang & Carreira‑Perpinan
    (2013) and runs in O(n log n).  For our use‑case we can also simply
    clip to non‑negative and renormalise (which is cheaper and sufficient
    because we never leave the simplex by a huge amount).  The function
    is kept for completeness and possible future tweaks.
    """
    # Simple clip‑and‑renorm – fast and works well for the small step sizes.
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        v /= s
    else:
        # If everything became zero (unlikely), fall back to a uniform vector.
        v[:] = 1.0 / v.size
    return v

            no_improve_iters = 0
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[improve] iter={iter_count:7d} score={best_score:.6f}")
        else:
            # step did not improve – shrink step size and optionally
            # perform a tiny random perturbation to escape plateaus.
            step *= 0.7
            no_improve_iters += 1
            if no_improve_iters % 2000 == 0:
                # random mass transfer (light perturbation)
                i, j = np.random.choice(n, 2, replace=False)
                eps = seq[i] * np.random.uniform(0.0005, 0.005)
                eps = min(eps, seq[i])
                cand = seq.copy()
                cand[i] -= eps
                cand[j] += eps
                cand = _project_onto_simplex(cand)
                cand_max, cand_k = _conv_max_and_arg(cand)
                cand_score = 2.0 * n * cand_max
                if cand_score < cur_score:
                    seq = cand
                    cur_score = cand_score
                    cur_max = cand_max
                    k_max = cand_k
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand.copy()
                        print(f"[perturb] iter={iter_count:7d} score={best_score:.6f}")

        # decay step slowly (even if the step was successful)
        step = max(step * step_decay, step_min)

        # occasional logging (every 5000 iterations)
        if iter_count % 5000 == 0:
            print(f"[log] iter={iter_count:7d} cur_score={cur_score:.6f} step={step:.2e}")

        # optional early exit if we reach a very good bound
        if best_score < 1.503:
            print("[early exit] target bound reached")
            break

    print(f"[final] total_iterations={iter_count}, best_score={best_score:.6f}")
    # Return the best candidate as a plain Python list.
    return best_seq.tolist()
