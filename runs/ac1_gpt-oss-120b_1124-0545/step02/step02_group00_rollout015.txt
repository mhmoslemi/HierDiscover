def gradient_move_multi(a, step_scale, K=5):
    b = np.convolve(a,a)
    # get indices of top K values
    top_idx = np.argpartition(-b, K-1)[:K]  # unsorted.
    # compute gradient
    g = np.zeros_like(a)
    n = len(a)
    for k in top_idx:
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        g[idx] += 2.0 * a[k - idx]
    # take step
    a_new = a - step_scale * g
    a_new = np.clip(a_new, 0, None)
    a_new = project_simplex(a_new)
    return a_new

def gradient_move_multi(a, step, K):
    n = len(a)
    b = np.convolve(a, a)
    # find top K indices (including ties)
    top_idx = np.argpartition(-b, K-1)[:K]

    g = np.zeros_like(a)
    for k in top_idx:
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        g[idx] += 2.0 * a[k - idx]
    a_new = a - step * g
    a_new = np.clip(a_new, 0, None)

def gradient_step(a, b, step_scale, K):
    n = len(a)
    # top K indices:
    if K >= len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argpartition(-b, K-1)[:K]
    g = np.zeros_like(a)
    for k in top_idx:
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        # counterpart indices: k-idx
        g[idx] += 2.0 * a[k - idx]

"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Analytic seeding** – a small pool of hand‑crafted families (uniform,
   beta‑distribution shapes, inverse‑sqrt families, the previously‑found
   construction if available) is generated and the best member is kept.

2. **Multi‑peak projected sub‑gradient descent** – at every iteration the
   convolution of the current sequence is computed, the K largest convolution
   entries are identified and a sub‑gradient that simultaneously pushes down
   all of them is applied.  The step size, temperature (Metropolis acceptance)
   and the number K are decreased linearly with the elapsed time.

3. **Exploration & restarts** – with a small probability a random mass‑transfer
   move is performed, symmetry is enforced occasionally, and after long
   periods without improvement a tiny jitter is added to the best known sequence.

All updates are projected onto the probability simplex (non‑negative,
sum = 1).  The routine respects the 1000 s budget, prints progress and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a (may be un‑normalised)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2.0 * len(a) * max_b / (s * s))


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Tiny stochastic transfer of mass from i to j."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def gradient_step_multi(a: np.ndarray, b: np.ndarray, step: float,
                       K: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Projected sub‑gradient step that reduces the K largest convolution entries.
    Returns the new sequence and its convolution.
    """
    n = a.shape[0]
    # indices of the K largest entries of the convolution
    if K >= b.size:
        top_idx = np.arange(b.size)
    else:
        top_idx = np.argpartition(-b, K - 1)[:K]

    g = np.zeros_like(a)
    for k in top_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        # counterpart indices are k - idx
        g[idx] += 2.0 * a[k - idx]   # standard sub‑gradient (double count)
    a_new = a - step * g
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_simplex(a_new)
    b_new = np.convolve(a_new, a_new)
    return a_new, b_new


def generate_analytic_candidates(n: int) -> list[np.ndarray]:
    """Create a small pool of analytically motivated sequences."""
    candidates = []

    # 0. Uniform
    candidates.append(np.ones(n) / n)

    # 1. Beta(½,½) – classic “arcsine” shape
    x = (np.arange(n, dtype=float) + 0.5) / n
    beta = (x ** (-0.5)) * ((1.0 - x) ** (-0.5))
    candidates.append(beta / beta.sum())

    # 2. Symmetric Beta(α,α) for a range of α
    for a in np.linspace(0.2, 1.2, 11):
        shape = (x ** (a - 1.0)) * ((1.0 - x) ** (a - 1.0))
        # safeguard against infinities (should not occur)
        shape[np.isinf(shape)] = 0.0
        candidates.append(shape / shape.sum())

    # 3. Inverse‑sqrt product with a small offset δ
    i = np.arange(n, dtype=float)
    for delta in (0.0, 0.5, 1.0):
        shape = 1.0 / np.sqrt((i + delta + 0.5) * (n - i - 0.5 + delta))
        candidates.append(shape / shape.sum())

    # 4. Mixed inverse‑sqrt powers p
    for p in (0.3, 0.4, 0.5, 0.6, 0.7, 0.8):
        shape = (i + 1.0) ** (-p) + (n - i) ** (-p)
        candidates.append(shape / shape.sum())

    # 5. If a previous best sequence is provided, add it
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=float)
        if prev.shape[0] == n:
            # normalise just in case
            prev = prev / prev.sum()
            candidates.append(prev)

    return candidates

                best_ever_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:06d}] new best = {best_ever_score:.6f} "
                      f"(t={elapsed:.1f}s)")

        # occasional symmetry enforcement (helps keep the two halves alike)
        # ------------------------------------------------------------------

            a = project_simplex(a)

            a = best_ever_seq + jitter
            a = project_simplex(a)
            print(f"[{iteration:06d}] restart (stagnation)")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_ever_score:.6f}, "
          f"time={total_time:.1f}s")
    return best_ever_seq.tolist()
