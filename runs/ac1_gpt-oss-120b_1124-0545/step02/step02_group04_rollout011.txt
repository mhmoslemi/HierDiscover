                a = a_new
                break

"""
Propose a non‑negative step‑function (vector) that minimizes

    E(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works on the simplex {x ≥ 0, sum(x)=√(2n)}.
It repeatedly tries to lower a target bound `t` on the maximal
convolution entry.  For a given bound it solves the feasibility
problem

    max(conv(a, a)) ≤ t

by minimizing the squared violation penalty

    L(a) = Σ_i max(conv_i - t, 0)²

with projected gradient descent (FFT based).  When a feasible
vector is found the bound is decreased (geometric schedule) and
the process repeats.  Several random restarts (uniform,
exponential, arcsine‑shaped and, if available, the previous best
construction) are performed while respecting the 1000 s budget.

The routine returns the best vector discovered as a plain list of
floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Projects a vector `v` onto the simplex {x ≥ 0, Σ x = s}
    using the O(n log n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # s = 0 case (should not happen here)
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def linear_full_convolution(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Returns the full linear convolution a * a (length 2·len(a)‑1) using FFT.
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def convolve_fft(x: np.ndarray, y: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution of `x` and `y` via FFT (full length).
    """
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    prod = np.fft.irfft(X * Y, n=nfft)
    return prod[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
def find_feasible(a_start: np.ndarray,
                  target_sum: float,
                  bound_curr: float,
                  bound_target: float,
                  n: int,
                  nfft_conv: int,
                  nfft_grad: int,
                  deadline: float,
                  max_iters: int = 1500,
                  step_scale: float = 0.20) -> np.ndarray | None:
    """
    Tries to produce a vector `a` with max(conv(a,a)) ≤ bound_target.
    Returns the feasible vector or ``None`` if the bound cannot be met
    within the allowed iterations.
    """
    a = a_start.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # ---- current violation -------------------------------------------------
        conv = linear_full_convolution(a, nfft_conv)                # length 2n‑1
        r = conv - bound_target
        r = np.where(r > 0, r, 0.0)                                 # only positive part
        loss = np.sum(r * r)

        if loss == 0.0:                                            # success
            return a

        # ---- gradient of the penalty -----------------------------------------
        rev_a = a[::-1]
        grad_full = convolve_fft(r, rev_a, nfft_grad)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]                  # slice to length n

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            return None

        # ---- simple back‑tracking line search ----------------------------------
        step = step_scale * target_sum / (norm + 1e-12)             # magnitude ≈ step_scale·√(2n)
        for _ in range(5):
            a_new = a - step * grad
            a_new = simplex_projection(a_new, target_sum)

            conv_new = linear_full_convolution(a_new, nfft_conv)
            r_new = conv_new - bound_target
            r_new = np.where(r_new > 0, r_new, 0.0)
            loss_new = np.sum(r_new * r_new)

            if loss_new <= loss:                                   # accept
                a = a_new
                break
            step *= 0.5                                            # shrink step

        else:
            # no improvement after back‑tracking → give up on this bound
            return None
    return None
