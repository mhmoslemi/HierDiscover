from scipy.optimize import linprog
...
def solve_lp(f, M):
    n = len(f)
    rows = 2*n-1
    # Build A
    diags = [f] * n
    offsets = np.arange(n)
    A = sp.diags(diags, offsets, shape=(rows, n), format='csc')
    b = np.full(rows, M)
    c = -np.ones(n)  # maximize sum => minimize -sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if not res.success:
        return None

"""
Hybrid optimizer for the step‑function self‑convolution problem.

* Start from a random (or a previously found) non‑negative vector
  with ℓ₁‑norm √(2 n) (the objective is invariant under scaling).
* Perform a fast simulated‑annealing / hill‑climbing loop that
  transfers mass, swaps entries or applies a “peak‑reduction’’ move.
* Every few dozen iterations solve a *linear program* that maximises the
  ℓ₁‑norm of a candidate g under the constraint   f∗g ≤ M,
  where M = max(f∗f).  The LP is solved with ``scipy.optimize.linprog``,
  using a sparse Toeplitz matrix that represents the convolution.
* The LP solution is scaled to the required ℓ₁‑norm and mixed
  with the current vector (convex combination).  This provably never
  increases the maximal convolution entry and often yields a sizable drop.
* All intermediate vectors are projected onto the probability simplex
  (exact ℓ₁‑norm) using an O(n log n) projection routine.
* The loop runs until the time budget (≈1000 s) is exhausted and returns
  the best sequence found.
"""

import time
import math
import random
import numpy as np

# Optional imports – fall back to a pure‑SA approach if unavailable.
try:
    import scipy.sparse as _sp
    from scipy.optimize import linprog
    _has_scipy = True
except Exception:  # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector v onto the simplex {x ≥ 0, sum(x)=s}.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray, nfft: int) -> float:
    """
    Return the maximal entry of the linear self‑convolution a∗a,
    computed with an FFT of length nfft (next power of two ≥ 2·len(a)‑1).
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Return the full linear convolution a∗a (length 2·len(a)‑1)
    using an FFT of length nfft.
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear program
        maximise    sum(g)
        subject to  f ∗ g ≤ M   (element‑wise)
                    g ≥ 0
    for a given non‑negative vector f and bound M.
    Returns the optimal g (as a dense np.ndarray) or None on failure.
    The convolution matrix is built as a sparse Toeplitz matrix.
    """
    if not _has_scipy:
        return None

    n = f.size
    rows = 2 * n - 1

    # Build convolution matrix A where (A @ g)[k] = (f ∗ g)[k].
    # Each column i contains the vector f shifted down by i positions.
    # This is exactly a Toeplitz matrix with diagonals equal to f.
    diags = [f] * n                 # each diagonal uses the current f
    offsets = np.arange(n)          # 0,1,…,n‑1
    A = _sp.diags(diags, offsets, shape=(rows, n), format='csc')

    # Right‑hand side: the bound M for every convolution entry.
    b = np.full(rows, M, dtype=np.float64)

    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # Solve with HiGHS (highly efficient for large sparse LPs).
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30.0})
    if res.success:
        return np.array(res.x, dtype=np.float64)
    return None

    best_val = cur_max

    # Timing
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin
    iteration = 0

    # Annealing / step‑size schedule
    T_start = 1.0
    T_end = 1e-4

    # How often we call the LP (every LP_EVERY iterations)
    LP_EVERY = 30

    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        # exponential temperature decay
        temperature = T_start * ((T_end / T_start) ** frac)

        # --------------------------------------------------------------
        # Choose a cheap move
        move = random.random()
        a_prop = a.copy()

        if move < 0.45:
            # 1) Random transfer of a small amount of mass between two slots
            i, j = np.random.randint(n, size=2)
            if i != j:
                max_delta = min(a_prop[i], target_sum * 0.04)
                delta = max_delta * temperature * random.random()
                if delta > 1e-12:
                    a_prop[i] -= delta
                    a_prop[j] += delta

        elif move < 0.70:
            # 2) Simple swap (permutes the multiset of values)
            i, j = np.random.randint(n, size=2)
            if i != j:
                a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        elif move < 0.90:
            # 3) Peak‑reduction move (target current maximal convolution entry)
            conv_full = _full_convolution_fft(a, nfft)
            max_idx = int(conv_full.argmax())
            idx = np.arange(n)
            partner = max_idx - idx
            mask = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[idx[mask]] * a[partner[mask]]

            src = int(np.argmax(contrib))                     # biggest contributor
            # destination: a coordinate with (almost) zero contribution
            zero_candidates = np.where(contrib == 0)[0]
            if zero_candidates.size:
                dst = int(np.random.choice(zero_candidates))
            else:
                dst = int(np.argmin(contrib))

            max_delta = min(a_prop[src], target_sum * 0.02)
            delta = max_delta * temperature * random.random()
            a_prop[src] -= delta
            a_prop[dst] += delta

        else:
            # 4) Sub‑gradient step for the current maximal entry
            conv_full = _full_convolution_fft(a, nfft)
            max_idx = int(conv_full.argmax())
            grad = np.zeros_like(a)
            partner = max_idx - np.arange(n)
            mask = (partner >= 0) & (partner < n)
            grad[mask] = a[partner[mask]]                    # gradient = 2·a_{max_idx‑i}
            # step size proportional to temperature
            step = target_sum * 0.15 * temperature
            a_prop -= step * grad

        # --------------------------------------------------------------
        # Projection back onto the simplex (ensures feasibility)
        a_prop = _simplex_projection(a_prop, target_sum)

        # --------------------------------------------------------------
        # Evaluate new candidate (fast max‑convolution)
        new_max = _max_convolution_fft(a_prop, nfft)

        # --------------------------------------------------------------
        # Acceptance rule (simulated annealing style)
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / (temperature + 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            a = a_prop
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_seq = a.copy()
                print(f"[{iteration:6d}] New best max‑conv = {best_val:.6f} (t={elapsed:.1f}s)")

        # --------------------------------------------------------------
        # Periodic LP‑based refinement
        if _has_scipy and (iteration % LP_EVERY == 0) and iteration > 0:
            # Current bound M
            conv_full = _full_convolution_fft(a, nfft)
            M = conv_full.max()

            g_raw = _solve_lp_max_sum(a, M)
            if g_raw is not None:
                sum_g = g_raw.sum()
                if sum_g > 0:
                    # Scale to the required ℓ₁‑norm and mix
                    scale = target_sum / sum_g
                    g_scaled = g_raw * scale
                    # Mixing factor – start relatively aggressive then shrink
                    mix = 0.3 * temperature
                    a = (1.0 - mix) * a + mix * g_scaled
                    a = _simplex_projection(a, target_sum)

                    # Re‑evaluate after the LP mix
                    cur_max = _max_convolution_fft(a, nfft)
                    if cur_max < best_val:
                        best_val = cur_max
                        best_seq = a.copy()
                        print(f"[{iteration:6d}] LP mix improved max‑conv = {best_val:.6f} (t={elapsed:.1f}s)")

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered as a plain Python list.
    return best_seq.tolist()
