    n = a.shape[0]
    grad = np.zeros_like(a)
    # compute a_{max_idx - i} part

def run_sa(a, target_sum, end_time, best_seq, best_val):
    n = a.shape[0]
    nfft = 1 << ((2*n - 1).bit_length())
    # compute initial conv, max
    cur_seq = a.copy()
    cur_max = _max_convolution(cur_seq, nfft)
    cur_val = 2 * n * cur_max / (target_sum**2)  # = cur_max
    best_seq = best_seq
    best_val = best_val
    start = time.time()
    iteration = 0
    # define parameters for SA
    T_start = 1.0
    T_end = 1e-6
    while time.time() < end_time:
        elapsed = time.time() - start
        fraction = elapsed / (end_time - start)
        temperature = T_start * ((T_end / T_start) ** fraction)
        # select move type
        move_type = random.random()
        a_prop = cur_seq.copy()
        if move_type < 0.25:
            # random transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                max_delta = min(a_prop[i], target_sum * 0.02)
                delta = max_delta * temperature * random.random()
                if delta > 1e-15:
                    a_prop[i] -= delta
                    a_prop[j] += delta
        elif move_type < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a_prop[i], a_prop[j] = a_prop[j], a_prop[i]
        elif move_type < 0.75:
            # peak reduction
            conv = np.convolve(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            a_prop = _peak_reduction_move(cur_seq, max_idx, target_sum, temperature)
        else:
            # subgradient
            conv = np.convolve(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            a_prop = _subgradient_move(cur_seq, max_idx, target_sum, temperature)
        # projection
        a_prop = _simplex_projection(a_prop, target_sum)
        # evaluate
        new_max = _max_convolution(a_prop, nfft)
        new_val = new_max  # since sum normalized
        # accept
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / (temperature + 1e-12))
            if random.random() < prob:
                accept = True
        if accept:
            cur_seq = a_prop
            cur_val = new_val
            # update best
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[SA] New best {best_val:.6f} at time {time.time() - start_time:.1f}s")

            cur_seq = np.maximum(cur_seq, 0)
            cur_seq = _simplex_projection(cur_seq, target_sum)
            cur_val = _max_convolution(cur_seq, nfft)
        iteration += 1
    return cur_seq, cur_val, best_seq, best_val

def _subgradient_move(a, max_idx, target_sum, temperature):
    n = a.size
    grad = np.zeros_like(a)

"""
Propose a non‑negative step function that minimizes
    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm (≈1000 s budget)

1. Initialise a sequence (use the provided best construction if present,
   otherwise a smooth half‑sine shape) and project it onto the ℓ₁‑simplex
   {x≥0, sum(x)=√(2n)}.

2. Phase A – Simulated annealing (≈20 % of the budget):
   * Four elementary moves (random transfer, swap, peak‑reduction,
     sub‑gradient) are mixed.
   * The temperature follows an exponential schedule; candidates are
     accepted with the Metropolis rule.
   * The best vector so far is kept.

3. Phase B – Linear‑programming improvement (≈50 % of the budget):
   * For the current vector f compute M = max(conv(f,f)).
   * Solve the LP
         maximise   Σ g_i
         subject to f * g ≤ M   (pointwise convolution)
                    g_i ≥ 0
     using a sparse Toeplitz matrix (SciPy ‘highs’ solver).
   * Scale the LP solution g to the required ℓ₁‑norm and update
         f ← (1‑t)·f + t·g   (t≈0.2).
   * Repeat until time runs out or the LP becomes infeasible.

4. Phase C – A final short annealing run (remaining time) to polish
   the result.

During all phases the current maximal convolution entry is obtained via
FFT for speed, and every candidate is projected back onto the simplex.
The best sequence found is returned as a plain list of floats.
"""

import time

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx = s} (Wang & Carreira‑Perpiñán)."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, nfft: int = None) -> float:
    """Return the maximal entry of the linear convolution a*a (FFT)."""
    n = a.size
    if nfft is None:
        nfft = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _peak_reduction_move(a: np.ndarray, max_idx: int,
                         target_sum: float, temperature: float) -> np.ndarray:
    """Move a small amount of mass from a high‑contributor to a low‑contributor."""
    n = a.size
    idx = np.arange(n)
    partner = max_idx - idx
    valid = (partner >= 0) & (partner < n)

    contrib = np.zeros_like(a)
    contrib[valid] = a[idx[valid]] * a[partner[valid]]

    top_k = 10
    src_candidates = np.argpartition(-contrib, top_k)[:top_k]
    src = src_candidates[np.argmax(contrib[src_candidates])]

    noncontrib = np.where(~valid)[0]
    if noncontrib.size > 0:
        dst = int(np.random.choice(noncontrib))
    else:
        low_k = 10
        dst_candidates = np.argpartition(contrib, low_k)[:low_k]
        dst = int(np.random.choice(dst_candidates))

    max_delta = min(a[src], target_sum * 0.02)
    delta = max_delta * temperature * random.random()
    if delta < 1e-12:
        return a.copy()

    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = _simplex_projection(a_new, target_sum)
    return a_new


# ----------------------------------------------------------------------
def _subgradient_move(a: np.ndarray, max_idx: int,
                      target_sum: float, temperature: float) -> np.ndarray:
    """Gradient descent step w.r.t. the current maximal convolution entry."""
    n = a.size
    grad = np.zeros_like(a)

    i = np.arange(n)
    j1 = max_idx - i                # partner on the left
    mask1 = (j1 >= 0) & (j1 < n)
    grad[mask1] += a[j1[mask1]]

    j2 = i - max_idx                # partner on the right
    mask2 = (j2 >= 0) & (j2 < n)
    grad[mask2] += a[j2[mask2]]

    eta = target_sum * 0.2 * temperature
    a_new = a - eta * grad
    a_new = _simplex_projection(a_new, target_sum)
    return a_new


# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   subject to   f * g ≤ M   (pointwise)
    and   g ≥ 0.  Returns the optimal vector g (unscaled) or None on failure.
    """
    n = f.size
    # Build sparse Toeplitz matrix A_{k,i} = f_{k-i} (zero outside)
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    # Fill column‑wise: column i uses the whole f vector shifted by i
    idx = 0
    base = np.arange(n, dtype=np.int32)  # 0..n-1
    for i in range(n):
        rows[idx: idx + n] = base + i          # k = i + j
        cols[idx: idx + n] = i                 # column i
        data[idx: idx + n] = f                 # f_j for all j
        idx += n

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M * (1.0 + 1e-9), dtype=np.float64)   # tiny epsilon

    # maximise Σ g_i  ⇔ minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    try:
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "disp": False},
        )
    except Exception:
        return None
    if res.success:
        return res.x
    return None

def _run_sa(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    best_seq: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float, np.ndarray, float]:
    """
    Perform simulated‑annealing on vector a until the given deadline.
    Returns (current_seq, current_val, best_seq, best_val).
    """
    n = a.size
    nfft = 1 << ((2 * n - 1).bit_length())
    cur = a.copy()
    cur_max = _max_convolution(cur, nfft)
    cur_val = cur_max  # because sum(cur)=target_sum
    start = time.time()
    iteration = 0
    T_start, T_end = 1.0, 1e-6

    while time.time() < deadline:
        # temperature
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temperature = T_start * ((T_end / T_start) ** frac)

        # pick a move
        move_type = random.random()
        cand = cur.copy()
        if move_type < 0.30:
            # random transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                max_delta = min(cand[i], target_sum * 0.02)
                delta = max_delta * temperature * random.random()
                if delta > 1e-12:
                    cand[i] -= delta
                    cand[j] += delta
        elif move_type < 0.55:
            # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.80:
            # peak‑reduction
            conv = np.convolve(cur, cur)
            max_idx = int(np.argmax(conv))
            cand = _peak_reduction_move(cur, max_idx, target_sum, temperature)
        else:
            # sub‑gradient
            conv = np.convolve(cur, cur)
            max_idx = int(np.argmax(conv))
            cand = _subgradient_move(cur, max_idx, target_sum, temperature)

        # projection onto simplex (keeps sum = target_sum)
        cand = _simplex_projection(cand, target_sum)

        # fast evaluation
        new_max = _max_convolution(cand, nfft)
        new_val = new_max

        # acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / (temperature + 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur.copy()
                print(
                    f"[SA] New best {best_val:.6f} at t={time.time() - start:.1f}s"
                )

        # occasional global jitter to escape stagnation
        if iteration % 30000 == 0 and iteration > 0:
            cur = best_seq + np.random.normal(
                scale=target_sum * 0.001, size=n
            )
            cur = np.maximum(cur, 0.0)
            cur = _simplex_projection(cur, target_sum)
            cur_val = _max_convolution(cur, nfft)

        iteration += 1

    return cur, cur_val, best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence minimizing the evaluation function.
    The routine combines simulated annealing with a linear‑programming
    refinement step and respects the time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # initialise sequence
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        x = np.arange(n)
        a = np.sin(np.pi * (x + 0.5) / n)  # smooth half‑sine start
        a = np.maximum(a, 0.0)

    target_sum = math.sqrt(2.0 * n)
    a = a / a.sum() * target_sum

    # --------------------------------------------------------------
    # timing control
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    # keep the global best
    best_seq = a.copy()
    best_val = _max_convolution(best_seq)   # already normalised

    # --------------------------------------------------------------
    # Phase A – short simulated annealing
    phase_a_end = start_time + (deadline - start_time) * 0.20
    cur_seq, cur_val, best_seq, best_val = _run_sa(
        a, target_sum, phase_a_end, best_seq, best_val
    )

    # --------------------------------------------------------------
    # Phase B – linear‑programming refinement
    # run until half of the remaining budget is spent or LP becomes infeasible
    phase_b_deadline = start_time + (deadline - start_time) * 0.70
    nfft = 1 << ((2 * n - 1).bit_length())
    while time.time() < phase_b_deadline:
        # current maximal convolution value
        M = _max_convolution(cur_seq, nfft)
        # solve LP
        g = _solve_lp(cur_seq, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # rescale to the required ℓ₁‑norm
        g = g * (target_sum / sum_g)

        # mix the new direction (mixing factor t can be tuned)
        t = 0.2
        cur_seq = (1.0 - t) * cur_seq + t * g
        cur_seq = _simplex_projection(cur_seq, target_sum)

        # evaluate new candidate (fast)
        cur_val = _max_convolution(cur_seq, nfft)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()
            print(
                f"[LP] New best {best_val:.6f} at t={time.time() - start_time:.1f}s"
            )

    # --------------------------------------------------------------
    # Phase C – final annealing on the remaining time
    remaining_deadline = deadline
    cur_seq, cur_val, best_seq, best_val = _run_sa(
        cur_seq, target_sum, remaining_deadline, best_seq, best_val
    )

    # --------------------------------------------------------------
    # return the best sequence found
    return best_seq.tolist()
