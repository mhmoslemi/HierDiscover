"""
Optimisation of a non‑negative step function.

The optimiser works in three stages:

1. **Seed generation** – a modest collection of analytically‑motivated
   probability vectors (geometric, power‑law, combined, symmetric,
   sine‑window) together with a few Dirichlet samples and a random
   soft‑max start.

2. **Fast scan** – evaluate the exact objective (2·n·max(conv)) for each
   seed using an FFT based convolution and keep the best one.

3. **Long‑run refinement** – starting from the best seed, run a single
   Adam optimiser on the log‑softmax parametrisation.  The loss is a
   smoothed (log‑sum‑exp) approximation of the maximum autocorrelation;
   the smoothing parameter β is increased geometrically each step so the
   optimiser eventually targets the true maximum.  The loop runs until
   the time budget expires, constantly updating the best sequence seen.

The final sequence is normalised to sum‑to‑one (the evaluation function
is scale‑invariant) and returned as a plain Python list of floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Output length is len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _refine_sequence(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 5.0,
    beta_mul: float = 1.02,
    lr0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best sequence (as np.ndarray) and its fast objective value.
    """
    v = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    best_seq = a0.copy()
    conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(conv)

    beta = start_beta
    it = 0

    while True:
        it += 1
        if time.time() > deadline:
            break

        # current probability vector (on the simplex)
        a = _softmax(v)

        # convolution and objective
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(
                f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}"
            )

        # ---- smoothed‑max weights ----
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerical issue – skip this iteration
            continue
        w /= w_sum

        # ---- gradient w.r.t. a (smooth loss) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n-2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]    # central n entries

        # ---- gradient w.r.t. logits v (softmax) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update ----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence minimising ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose a length – try the larger length first if time permits.
    # ------------------------------------------------------------------
    lengths = [8000, 4000]   # fall back to 4000 if 8000 would be too costly
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"[main] Optimising length = {n}")

        # --------------------------------------------------------------
        # Build a diverse set of seed sequences (all on the simplex)
        # --------------------------------------------------------------
        seed_seqs = []

        # Geometric families (r = 1 - c/n)
        for c in np.linspace(0.5, 5.0, 12):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_geom(n, r))

        # Power‑law families
        for p in np.linspace(0.0, 2.2, 12):
            seed_seqs.append(_power(n, p))

        # Combined geometric × power families
        for c in np.linspace(0.5, 4.0, 8):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 8):
                seed_seqs.append(_geom_power(n, r, p))

        # Symmetric geometric
        for c in np.linspace(0.5, 4.5, 8):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_sym_geom(n, r))

        # Smooth sine window (good baseline)
        seed_seqs.append(_sine_window(n))

        # Dirichlet samples (different concentration parameters)
        for alpha in [0.2, 0.5, 0.8, 2.0, 5.0]:
            # Dirichlet draw – may be a bit heavy for very large n,
            # but only a few samples are taken.
            a = np.random.gamma(alpha, 1.0, size=n)
            a_sum = a.sum()
            if a_sum > 0:
                seed_seqs.append(a / a_sum)

        # Random soft‑max seeds
        for _ in range(5):
            v_rand = np.random.randn(n)
            a_rand = _softmax(v_rand)
            seed_seqs.append(a_rand)

        # If a previous construction is present, reuse it
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                a_prev = np.asarray(prev, dtype=np.float64)
                if a_prev.ndim == 1:
                    if a_prev.shape[0] != n:
                        # resize by truncation / zero‑padding
                        a_adj = np.zeros(n, dtype=np.float64)
                        m = min(a_prev.shape[0], n)
                        a_adj[:m] = a_prev[:m]
                        a_prev = a_adj
                    if a_prev.sum() > 0:
                        a_prev = a_prev / a_prev.sum()
                        seed_seqs.append(a_prev)
            except Exception:
                pass

        # --------------------------------------------------------------
        # Quick scan to obtain an initial best sequence
        # --------------------------------------------------------------
        for idx, a in enumerate(seed_seqs):
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best from seed {idx} = {cur_val:.6f}")

        # --------------------------------------------------------------
        # Refinement – run a long‑run Adam optimiser on the current best
        # --------------------------------------------------------------
        # Ensure we have a starting point
        if global_best_seq is None:
            # fallback uniform distribution
            global_best_seq = np.ones(n, dtype=np.float64) / n
            print("[fallback] using uniform start")

        # Run the dedicated refinement loop (time‑aware)
        refined_seq, refined_val = _refine_sequence(
            global_best_seq,
            n,
            deadline,
            start_beta=5.0,
            beta_mul=1.02,
            lr0=0.8,
        )
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[overall] new global best after refinement (len={n}) = {refined_val:.6f}")

        # If still time remains, attempt a few perturbed restarts from the
        # best sequence found so far.
        while time.time() < deadline:
            # create a perturbed logits vector
            noisy_logits = np.log(global_best_seq + 1e-30) + 0.02 * np.random.randn(n)
            perturbed_seq = _softmax(noisy_logits)
            seq2, val2 = _refine_sequence(
                perturbed_seq,
                n,
                deadline,
                start_beta=5.0,
                beta_mul=1.02,
                lr0=0.8,
            )
            if val2 < global_best_val:
                global_best_val = val2
                global_best_seq = seq2.copy()
                print(f"[perturb] improved global best to {val2:.6f}")

        # --------------------------------------------------------------
        # End of this length – move on to the next if time permits
        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] final uniform sequence")

    # Normalise to sum‑to‑one (evaluation is scale‑invariant)
    s = np.sum(global_best_seq)
    if s > 0:
        global_best_seq = global_best_seq / s

    # Convert to plain Python list
    result = global_best_seq.tolist()
    print(f"[final] best objective (fast) = {global_best_val:.6f}")
    return result
