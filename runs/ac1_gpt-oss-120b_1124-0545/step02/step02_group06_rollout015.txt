def _conv_max_and_arg(seq: np.ndarray) -> tuple[float, int]:
    """Return (max(conv(seq,seq)), argmax) using an FFT‑based convolution."""
    n = seq.size
    size = 2 * n - 1
    # next power of two for a fast real FFT
    m = 1 << ((size - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:size]
    arg = int(np.argmax(conv))
    return float(conv[arg]), arg


def _score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective:
        2 * n * max(conv(seq,seq)) / (sum(seq) ** 2)
    The sequence is assumed to be non‑negative and already normalised
    (sum(seq)=1).  This matches the behaviour of `evaluate_sequence`.
    """
    n = seq.size
    max_b, _ = _conv_max_and_arg(seq)
    # sum is 1 after normalisation, but we keep the formula for safety
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _make_shape(
    n: int,
    p: float = 0.5,
    r: float = 0.5,
    eps: float = 0.5,
) -> np.ndarray:
    """
    Build a discrete sequence approximating
        a_i ∝ (n‑1‑i)^r / (i+eps)^p.

    Parameters
    ----------
    n   : length of the sequence
    p   : exponent on the left side (default 0.5)
    r   : exponent on the right side (default 0.5)
    eps : small shift to avoid division by zero at i=0
    """
    i = np.arange(n, dtype=float)
    rev = n - 1 - i  # (n‑1‑i)

    # rev**r is zero at the last index, as desired
    left = (i + eps) ** (-p)          # (i+eps)^(-p)
    right = rev ** r                  # (n‑1‑i)^r
    seq = left * right
    # enforce non‑negativity (already true) and normalise to unit sum
    total = seq.sum()
    if total == 0:
        # fallback to uniform if something went wrong
        seq = np.ones(n, dtype=float) / n
    else:
        seq = seq / total
    return seq
