"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The evaluation is scale‑invariant, so we work on the probability simplex
Σa_i = 1.  The algorithm consists of three stages:

1.  Analytic families – a dense grid over the two‑parameter family
    a_i ∝ ((i+δ)*(n‑1‑i+δ))^{-p} (and its sum‑variant) is explored.
    The best member serves as a strong start.  The previously found
    construction `height_sequence_1` is also considered.

2.  Refined stochastic optimisation – a simulated‑annealing loop
    combines projected sub‑gradient steps (targeting the current
    maximal convolution entry) with occasional random mass‑transfer
    and “window‑flattening’’ moves.  The temperature and step‑size are
    annealed, a Metropolis acceptance rule is used, and a restart with
    jitter is performed after prolonged stagnation.

3.  The best sequence encountered before the time budget expires is
    returned.  All operations respect the non‑negativity and simplex
    constraints, and the convolution is computed by FFT for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper: smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_candidate(a: np.ndarray, fft_size: int) -> tuple[float, float]:
    """
    Compute the objective value for a candidate a (already on the simplex)
    using an FFT‑based convolution.  Returns (score, max_conv_value).
    """
    n = a.shape[0]
    # FFT convolution (linear, zero‑padded)
    A = np.fft.rfft(a, n=fft_size)
    conv = np.fft.irfft(A * A, n=fft_size)[: 2 * n - 1]
    max_b = conv.max()
    # sum(a) should be 1, but we keep the exact expression for safety
    s = a.sum()
    if s < 1e-12:
        return np.inf, np.inf
    score = float(2.0 * n * max_b / (s * s))
    return score, max_b


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index i to a different
    random index j (preserving the total mass).  Afterwards the vector
    is projected back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def flatten_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Choose a short contiguous window and redistribute its total mass
    almost uniformly (with a small random jitter).  The result is again
    projected onto the simplex.
    """
    n = a.shape[0]
    # window length between 2 % and 10 % of the sequence length
    L = max(2, int(n * (0.02 + 0.08 * np.random.rand())))
    start = np.random.randint(0, n - L + 1)
    total = a[start:start + L].sum()
    base = total / L
    jitter = step_scale * total * (np.random.rand(L) - 0.5)
    new_vals = np.maximum(base + jitter, 0.0)
    a_new = a.copy()
    a_new[start:start + L] = new_vals
    return project_simplex(a_new)


def gradient_move(a: np.ndarray, step_scale: float, fft_size: int) -> np.ndarray:
    """
    Projected sub‑gradient step w.r.t. the currently maximal convolution
    entry.  The sub‑gradient is g_i = 2·a_{k_max‑i} for indices where it
    exists.  After the step the vector is projected onto the simplex.
    """
    n = a.shape[0]
    # Obtain the index k_max of the peak of a*a
    A = np.fft.rfft(a, n=fft_size)
    conv = np.fft.irfft(A * A, n=fft_size)[: 2 * n - 1]
    k_max = int(conv.argmax())

    # Build the sub‑gradient
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k_max - idx]

    # Gradient descent step followed by projection
    a_new = a - step_scale * g
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_simplex(a_new)
    return a_new


def generate_product_shape(p: float, offset: float, n: int) -> np.ndarray:
    """Symmetric shape a_i ∝ ((i+offset)*(n‑1‑i+offset))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    return a / a.sum()


def generate_sum_shape(p: float, offset: float, n: int) -> np.ndarray:
    """Symmetric shape a_i ∝ (i+offset)^{-p} + (n‑1‑i+offset)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    return a / a.sum()

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     n: int = 2000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    the evaluation function.  The routine combines an analytic grid
    search with a simulated‑annealing loop (gradient + random moves) and
    respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # FFT size for linear convolution (next power of two ≥ 2·n‑1)
    fft_size = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1 – Analytic start: grid over the product/sum families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Helper to evaluate and possibly update the global best
    def consider(candidate: np.ndarray, remark: str = ""):
        nonlocal best_seq, best_score
        score, _ = evaluate_candidate(candidate, fft_size)
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"[grid] new best {remark}score = {score:.6f}")
        return score

    # Uniform candidate (always worthwhile)
    uniform = np.ones(n) / n
    consider(uniform, "uniform ")

    # Include the previous best construction if available
    if "height_sequence_1" in globals():
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.shape[0] == n:
            hs_norm = hs / hs.sum()
            consider(hs_norm, "prev ")

    # Parameter grid
    p_vals = np.linspace(0.20, 0.60, 21)          # 0.20 … 0.60
    off_vals = np.linspace(0.10, 2.00, 20)       # 0.10 … 2.00

    for p in p_vals:
        for off in off_vals:
            # product shape
            cand = generate_product_shape(p, off, n)
            consider(cand, f"prod(p={p:.3f},δ={off:.3f}) ")
            # sum shape
            cand2 = generate_sum_shape(p, off, n)
            consider(cand2, f"sum(p={p:.3f},δ={off:.3f}) ")

            # early‑exit if we are about to run out of time
            if time.time() > deadline - 30.0:   # keep ~30 s for optimisation
                break
        if time.time() > deadline - 30.0:
            break

    # If nothing succeeded (should not happen) fall back to uniform
    if best_seq is None:
        best_seq = uniform.copy()
        best_score, _ = evaluate_candidate(best_seq, fft_size)

    a = best_seq.copy()
    cur_score = best_score
    print(f"[init] start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – Stochastic optimisation (simulated annealing)
    # ------------------------------------------------------------------
    # Annealing schedules (linear)
    init_temp = 1.0
    final_temp = 1e-4
    init_step = 0.30
    final_step = 5e-5

    iteration = 0
    last_improve_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = min(1.0, elapsed / (budget_s - 10.0))

        # temperature controls acceptance of uphill moves
        temp = init_temp * (1.0 - progress) + final_temp * progress
        # step size – also decays with iteration (Polyak–like)
        base_step = init_step * (1.0 - progress) + final_step * progress
        step_scale = base_step / np.sqrt(iteration + 1)

        # ------------------------------------------------------------------
        # Choose a move type
        r = np.random.rand()
        if r < 0.70:
            cand = gradient_move(a, step_scale, fft_size)
        elif r < 0.90:
            cand = random_move(a, step_scale * 0.5)
        else:
            cand = flatten_move(a, step_scale * 0.5)

        cand_score, _ = evaluate_candidate(cand, fft_size)

        # Metropolis acceptance (compare to current state, not only best)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score

        # Record global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            last_improve_time = time.time()
            print(f"[{iteration:07d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # Stagnation restart
        if iteration % 20000 == 0 and (time.time() - last_improve_time) > 30.0:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            cur_score, _ = evaluate_candidate(a, fft_size)
            print(f"[{iteration:07d}] restart after stagnation")
            last_improve_time = time.time()

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best_score={best_score:.6f}, time={total_time:.1f}s")
    # Return the best sequence (as a plain Python list)
    return best_seq.tolist()
