Optimise a step-function to minimise evaluate_sequence.
The search consists of three stages:
1) Parameterised power‑law + uniform mix: a_i ∝ (i+1+δ)⁻ᵅ,
   optionally mixed with a tiny uniform component.
   A coarse‑to‑fine grid over (α,δ,λ) is explored using a fast
   FFT‑based convolution to compute the objective.
2) Smoothed gradient descent on the exact objective.
   The gradient of a smooth approximation of max(conv) is obtained
   via the weights w_k = exp(γ (c_k/ max_c − 1)) and the correlation
   w ∗ a.  After each step the vector is projected onto the probability
   simplex.
3) Simple random mass‑exchange moves are interleaved to escape plateaus.

The whole routine respects the 1000 s budget and returns the best sequence
found as a plain Python list.
"""

import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """
    Fast maximum of the self‑convolution using a real FFT.
    The sequence is assumed to sum to one.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    # rfft returns the non‑negative frequencies of a real‑valued signal
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """
    Exact score used by evaluate_sequence when the sequence already
    sums to one (the denominator becomes one).
    """
    n = seq.shape[0]
    return 2.0 * n * _conv_max(seq)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if v.sum() == 1 and (v >= 0).all():
        return v.copy()
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        # all entries will be zero – fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / len(v))
    w /= w_sum
    return w


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised power‑law a_i ∝ (i+1+offset)^(-α), α>0.
    The result always sums to one.
    """
    i = np.arange(1, n + 1, dtype=np.float64) + offset
    a = i ** (-alpha)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


def _mixed_sequence(power: np.ndarray, lam: float) -> np.ndarray:
    """
    Convex combination of a power‑law sequence and the uniform distribution.
    lam∈[0,1] is the weight of the uniform component.
    Both components are already normalised, therefore the mixture is too.
    """
    if lam <= 0:
        return power.copy()
    if lam >= 1:
        return np.full_like(power, 1.0 / power.shape[0])
    n = power.shape[0]
    uniform = np.full(n, 1.0 / n)
    return (1.0 - lam) * power + lam * uniform


def _smooth_gradient_step(seq: np.ndarray,
                          step: float,
                          gamma: float) -> np.ndarray:
    """
    One project‑gradient step on the smooth approximation
         F_γ(a) = Σ_k exp(γ (c_k/max_c – 1)),
    where c = a∗a and max_c = max(c).
    The gradient is 2·correlation(w, a) with
        w_k = exp(γ (c_k/max_c – 1)).
    Afterwards the vector is projected back onto the simplex.
    """
    n = seq.shape[0]
    # compute self‑convolution (FFT)
    L = _next_pow2(2 * n - 1)
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_c = conv.max()
    if max_c == 0:
        # degenerate case – return the current sequence unchanged
        return seq.copy()

    # weight concentrating on the largest entries
    w = np.exp(gamma * (conv / max_c - 1.0))

    # correlation w ⋆ a (valid mode) gives ∂F/∂a up to the factor 2
    grad = 2.0 * np.correlate(w, seq, mode='valid')
    # keep us on the affine hyperplane Σ a_i = 1
    grad -= grad.mean()
    # gradient step
    new_seq = seq - step * grad
    # enforce non‑negativity & re‑normalise
    new_seq = _project_to_simplex(new_seq)
    return new_seq

   if it improves the objective.  This helps escape plateaus.

The routine respects a 1000 s budget (leaving a 5 s safety margin) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x*."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """
    Maximum entry of the self‑convolution of *seq* (which is assumed to sum
    to 1).  Computed with a real FFT for speed.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """
    Exact score used by ``evaluate_sequence`` when the sequence already
    satisfies Σ a_i = 1 (the denominator becomes 1):
        score = 2·n·max(conv).
    """
    n = seq.shape[0]
    return 2.0 * n * _conv_max(seq)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex {x ≥ 0, Σ x = 1}.
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if v.sum() == 1.0 and (v >= 0).all():
        return v.copy()

    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:                         # all entries will be zero
        return np.full_like(v, 1.0 / len(v))

    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / len(v))
    w /= w_sum
    return w


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised power‑law sequence a_i ∝ (i+1+offset)^(-α), α>0.
    The result always satisfies Σ a_i = 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64) + offset
    a = i ** (-alpha)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


def _mixed_sequence(power: np.ndarray, lam: float) -> np.ndarray:
    """
    Convex combination of a power‑law vector and the uniform distribution.
    ``lam`` ∈ [0,1] is the weight of the uniform part.
    Both arguments are already normalised, so the mixture is also
    normalised.
    """
    if lam <= 0.0:
        return power.copy()
    if lam >= 1.0:
        n = power.shape[0]
        return np.full(n, 1.0 / n)
    n = power.shape[0]
    uniform = np.full(n, 1.0 / n)
    return (1.0 - lam) * power + lam * uniform


def _smooth_gradient_step(seq: np.ndarray, step: float, gamma: float) -> np.ndarray:
    """
    One projected gradient step on the smooth surrogate of the max‑convolution.
    The surrogate is
        F_γ(a) = Σ_k exp(γ (c_k / max(c) – 1)),
    where c = a*a.  Its gradient is
        ∇F_γ = 2·corr(w, a)   with   w_k = exp(γ (c_k / max(c) – 1)).
    After the step the iterate is projected back onto the simplex.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_c = conv.max()
    if max_c == 0.0:
        return seq.copy()

    w = np.exp(gamma * (conv / max_c - 1.0))          # weights concentrate on the large entries
    grad = 2.0 * np.correlate(w, seq, mode='valid')   # correlation = derivative of the surrogate
    grad -= grad.mean()                               # keep step in the hyperplane Σa_i = 1
    new_seq = seq - step * grad
    new_seq = _project_to_simplex(new_seq)
    return new_seq


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the given time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a few seconds for safety
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Choose a convenient length.
    # ------------------------------------------------------------------
    n = 3000                     # enough to obtain a near‑optimal constant,
                                 # still cheap for many FFTs.

    # ------------------------------------------------------------------
    # 1. Coarse grid over (α, δ, λ).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_params = (0.0, 0.0, 0.0)   # α, δ, λ

    alphas = np.arange(0.45, 0.62, 0.02)          # 0.45 … 0.60
    deltas = np.arange(0.0, 21.0, 2.0)           # 0 … 20 (step 2)
    lams = np.array([0.0, 0.05, 0.10, 0.15, 0.20])  # mix with uniform

    for alpha in alphas:
        for delta in deltas:
            power = _power_sequence(n, alpha, offset=delta)
            for lam in lams:
                if time.time() >= deadline:
                    break
                seq = _mixed_sequence(power, lam)
                cur_score = _score(seq)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = seq.copy()
                    best_params = (alpha, delta, lam)
                    print(f"[coarse] α={alpha:.3f}, δ={delta:.1f}, λ={lam:.2f} → score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Fine grid around the best point found so far.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)
        best_score = _score(best_seq)
        best_params = (0.5, 0.0, 0.0)

    best_alpha, best_delta, best_lam = best_params

    fine_alphas = np.linspace(max(0.01, best_alpha - 0.01),
                              best_alpha + 0.01, 21)    # step ≈ 0.001
    fine_deltas = np.linspace(max(0.0, best_delta - 2.0),
                              best_delta + 2.0, 21)     # step ≈ 0.2
    fine_lams = np.linspace(max(0.0, best_lam - 0.02),
                            min(0.5, best_lam + 0.02), 9)  # step ≈ 0.005

    for alpha in fine_alphas:
        for delta in fine_deltas:
            power = _power_sequence(n, alpha, offset=delta)
            for lam in fine_lams:
                if time.time() >= deadline:
                    break
                seq = _mixed_sequence(power, lam)
                cur_score = _score(seq)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = seq.copy()
                    best_alpha, best_delta, best_lam = alpha, delta, lam
                    print(f"[fine]   α={alpha:.5f}, δ={delta:.2f}, λ={lam:.3f} → score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed gradient descent with occasional random swaps.
    # ------------------------------------------------------------------
    step = 0.15
    gamma = 5.0                     # start with mild smoothing
    step_decay = 0.9997
    gamma_mul = 1.01
    gamma_max = 300.0
    it = 0

    seq = best_seq.copy()
    best_seq = seq.copy()
    best_score = _score(best_seq)

    while time.time() < deadline:
        it += 1

        # one projected gradient step on the smooth surrogate
        seq = _smooth_gradient_step(seq, step, gamma)

        # random mass exchange every few dozen iterations
        if it % 50 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(step * seq[i], 0.5 * seq[i])
            cand = seq.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = _project_to_simplex(cand)
            seq = cand

        cur_score = _score(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={it}, α≈{best_alpha:.5f}, δ≈{best_delta:.2f}, λ≈{best_lam:.3f}, score={best_score:.6f}")

        # update schedules
        step = max(step * step_decay, 1e-7)
        gamma = min(gamma * gamma_mul, gamma_max)

    print(f"[final] best_score={best_score:.6f}, α={best_alpha:.5f}, δ={best_delta:.2f}, λ={best_lam:.3f}")

    # Return a plain Python list – compatible with the evaluation harness.
    return best_seq.tolist()
```
