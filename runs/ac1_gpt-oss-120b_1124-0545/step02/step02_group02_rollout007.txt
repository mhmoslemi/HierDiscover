```python
def solve_lp_for_g(f):
    n = len(f)
    # Build A_ub as dense
    # shape (2n-1, n+1)
    k_len = 2*n-1
    A_ub = np.zeros((k_len, n+1), dtype=float)
    # Fill A_ub[:, :n] with convolution coefficients (f reversed in diag)
    for k in range(k_len):
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        # i indexes f, j = k - i
        # Set A_ub[k, j] = f[i]
        for i in range(lo, hi+1):
            j = k - i
            A_ub[k, j] = f[i]
        # last column: -1 for M variable
        A_ub[k, -1] = -1.0

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Objective: minimize M
    c = np.zeros(n+1)
    c[-1] = 1.0

    # Solve
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=np.zeros(k_len),
                                 A_eq=A_eq, b_eq=b_eq,
                                 bounds=bounds, method='highs')
    if not res.success:
        # fallback: return f itself
        return f, np.max(np.convolve(f,f))
    g=res.x[:n]
    M_opt=res.x[-1]
    # Ensure non-negativity minor corrections
    g = np.maximum(g, 0.0)
    # renorm to sum 1 (due to rounding errors)
    s = g.sum()
    if s>0:
        g = g / s
    else:
        g = f.copy()
        M_opt = np.max(np.convolve(g,g))
    return g, M_opt
```

"""
Propose a low‑score step‑function using a two‑phase optimisation:

1. **Parametric seeding** – explore a small family of power‑law sequences
   (a_i ∝ (i+1+δ)^{-α}) to obtain a good initial shape.

2. **Alternating linear‑programming refinement** –  for a fixed length `n`
   repeatedly solve the linear programme

        minimise   M
        subject to g ≥ 0 , Σ g = 1
                    (f * g)(k) ≤ M   for all k

   (where `f` is the current sequence).  The optimal `g` reduces the peak of
   the cross‑convolution with `f`.  We move the current sequence towards `g`
   by a convex combination `f ← (1‑t)·f + t·g` with a decreasing step size `t`
   and repeat.  After a few LP updates a short projected‑gradient descent
   cleans up the remaining local noise.

The whole routine respects the 1000 s budget, uses only NumPy/SciPy,
and returns a plain Python list that can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised power‑law sequence a_i ∝ (i+1+offset)^{-α}.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _score_from_sequence(seq: np.ndarray) -> float:
    """
    Exact score used by ``evaluate_sequence`` for a normalised sequence
    (i.e. sum(seq) == 1).

    score = 2 * n * max(conv)          (sum(seq) == 1 → denominator = 1)
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)          # length 2n‑1
    return float(2.0 * n * conv.max())


def _projected_gradient_step_multi(seq: np.ndarray,
                                   step: float,
                                   top_k: int = 3) -> np.ndarray:
    """
    One sub‑gradient step that reduces the largest entries of the self‑convolution.
    After the step the vector is projected onto the simplex {x ≥ 0, Σ x = 1}.
    """
    n = seq.size
    conv = np.convolve(seq, seq)

    if top_k >= conv.size:
        top_idx = np.arange(conv.size)
    else:
        top_idx = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(seq)
    for k in top_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += seq[k - i_range]

    # make the step tangent to Σx = 1
    grad -= grad.mean()

    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total > 0.0:
        new_seq /= total
    else:
        new_seq = seq.copy()
    return new_seq


def _solve_lp_optimal_g(f: np.ndarray):
    """
    Solve the linear programme

        minimise   M
        subject to g ≥ 0 , Σ g = 1
                    (f * g)(k) ≤ M   for all k = 0 … 2n‑2

    Returns the optimal g (as a 1‑D ndarray) and the optimal bound M.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # Build sparse matrix A_ub such that A_ub @ [g; M] ≤ 0
    data = []
    row_ind = []
    col_ind = []

    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i                     # index of g
            data.append(f[i])
            row_ind.append(k)
            col_ind.append(j)
        # coefficient of -M
        data.append(-1.0)
        row_ind.append(k)
        col_ind.append(n)                # last column is M

    A_ub = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n + 1))
    b_ub = np.zeros(rows)

    # Equality Σ g = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: g_i ≥ 0, M ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Objective: minimise M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    if not res.success:
        # Fallback – return the current vector unchanged
        return f.copy(), np.max(np.convolve(f, f))

    g = np.maximum(res.x[:n], 0.0)
    M_opt = res.x[-1]

    # Normalise g (numerical drift)
    s = g.sum()
    if s > 0.0:
        g /= s
    else:
        g = f.copy()
        M_opt = np.max(np.convolve(g, g))

    return g, M_opt


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm combines a power‑law seed,
    alternating linear‑programming refinement and a short projected‑gradient
    polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # 1. Choose a length that keeps the LP tractable.
    # --------------------------------------------------------------
    n = 1500                         # 1500 yields ~4.5 M matrix entries (~36 MiB)

    # --------------------------------------------------------------
    # 2. Seeding – coarse + fine grid over power‑law exponent α.
    # --------------------------------------------------------------
    best_f = None
    best_score = np.inf
    best_alpha = 0.5

    # coarse scan
    for alpha in np.arange(0.45, 0.71, 0.04):
        if time.time() >= deadline:
            break
        f = _power_sequence(n, alpha)
        sc = _score_from_sequence(f)
        if sc < best_score:
            best_score, best_f, best_alpha = sc, f.copy(), alpha
            print(f"[coarse] α={alpha:.4f}, score={sc:.6f}")

    # fine scan around the best α
    fine_alphas = np.linspace(best_alpha - 0.02, best_alpha + 0.02, 81)
    for alpha in fine_alphas:
        if time.time() >= deadline:
            break
        f = _power_sequence(n, alpha)
        sc = _score_from_sequence(f)
        if sc < best_score:
            best_score, best_f, best_alpha = sc, f.copy(), alpha
            print(f"[fine]   α={alpha:.5f}, score={sc:.6f}")

    # optional integer offset search (δ = 0 … 4)
    for delta in range(0, 5):
        if time.time() >= deadline:
            break
        f = _power_sequence(n, best_alpha, offset=float(delta))
        sc = _score_from_sequence(f)
        if sc < best_score:
            best_score, best_f, best_alpha = sc, f.copy(), best_alpha
            print(f"[offset] δ={delta}, score={sc:.6f}")

    # --------------------------------------------------------------
    # 3. Alternating LP refinement.
    # --------------------------------------------------------------
    f = best_f
    mix = 0.5                         # initial convex‑combination weight
    lp_iter = 0
    max_lp_iters = 12                # enough for noticeable progress

    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1

        # Solve LP: give the best partner g for the current f
        g, M_opt = _solve_lp_optimal_g(f)

        # Convex update f ← (1‑t)·f + t·g
        f = (1.0 - mix) * f + mix * g
        # f stays normalised because both operands sum to 1
        cur_score = _score_from_sequence(f)

        if cur_score < best_score:
            best_score, best_f = cur_score, f.copy()
            print(f"[LP   ] iter={lp_iter}, mix={mix:.3f}, score={cur_score:.6f}")

        # slowly decay the mixing factor, avoiding too aggressive jumps
        mix = max(mix * 0.8, 0.05)

    # --------------------------------------------------------------
    # 4. Fine‑grained projected‑gradient polishing.
    # --------------------------------------------------------------
    step = 0.25
    decay = 0.9999
    grad_iter = 0
    while time.time() < deadline and grad_iter < 300000:
        grad_iter += 1
        f = _projected_gradient_step_multi(f, step, top_k=5)

        # occasional random exchange to escape tiny plateaus
        if grad_iter % 97 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(step * f[i], f[i] * 0.4)
            cand = f.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand /= s
                f = cand

        cur_score = _score_from_sequence(f)
        if cur_score < best_score:
            best_score, best_f = cur_score, f.copy()
            print(f"[grad ] iter={grad_iter}, step={step:.5f}, score={cur_score:.6f}")

        step = max(step * decay, 1e-7)

    # --------------------------------------------------------------
    # 5. (Optional) enlarge the sequence length to a more conventional size.
    #    Upsampling by linear interpolation keeps the shape while increasing n.
    # --------------------------------------------------------------
    final_len = 5000
    if final_len > n:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, final_len)
        f_big = np.interp(x_new, x_old, best_f)
        f_big = np.maximum(f_big, 0.0)
        f_big /= f_big.sum()
        # a few cheap gradient steps on the up‑sampled vector
        step_big = 0.15
        for _ in range(2000):
            f_big = _projected_gradient_step_multi(f_big, step_big, top_k=3)
            step_big = max(step_big * 0.99995, 1e-7)
        best_f = f_big

    # --------------------------------------------------------------
    # 6. Return the best sequence found (as plain Python list).
    # --------------------------------------------------------------
    final_score = _score_from_sequence(best_f)
    print(f"[final] best_score={final_score:.6f}, length={len(best_f)}")
    return best_f.tolist()
