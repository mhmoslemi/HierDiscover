"""
Propose a low‑value step function via Adam‑optimised smoothed max‑autocorrelation.

The algorithm works on the probability simplex (∑a = 1) and treats the
objective 2·n·max(conv(a,a)) as a smooth approximation
fₚ(a) = (∑ₖ (conv(a,a)ₖ)ᵖ)^{1/p}.  For a fixed p the gradient is
    ∇fₚ = 2·conv_rev(w)   with   wₖ ∝ (conv(a,a)ₖ)^{p‑1}.
Adam (adaptive moment estimation) is used to take steps in the negative
gradient direction; the iterate is projected onto the non‑negative simplex
after every step.  The smoothness parameter p is slowly increased so that
the optimisation finally focuses on the true maximum.  Symmetry (a = rev(a))
is enforced after each projection, a tiny random “mass‑transfer’’ move is
applied with low probability, and a restart is triggered when no
improvement is seen for a while.  The best sequence found within the time
budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution of u and v using FFT of length ``size``.
    ``size`` must be >= len(u)+len(v)-1.
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a random fraction of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    n : int, optional
        Length of the sequence (default 8000).  A previous construction may be
        up‑sampled if the global variable ``height_sequence_1`` is present.

    Returns
    -------
    list[float]
        The best sequence (list of Python floats) found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # tiny safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 8000))

    # --------------------------------------------------------------
    # initialise a starting distribution
    # --------------------------------------------------------------
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None:
        try:
            seq0 = np.asarray(seq0, dtype=float)
            if seq0.size != n:
                # linear interpolation onto the new grid
                x_old = np.linspace(0.0, 1.0, seq0.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a = np.interp(x_new, x_old, seq0)
                a = np.clip(a, 0.0, None)
            else:
                a = np.clip(seq0, 0.0, None)
        except Exception:
            a = np.random.rand(n)
    else:
        a = np.random.rand(n)

    # normalise to the simplex
    if a.sum() == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / a.sum()

    # enforce symmetry (helps the optimiser and is known to be optimal for many n)
    a = (a + a[::-1]) / 2.0

    # --------------------------------------------------------------
    # pre‑compute FFT length (power of two)
    # --------------------------------------------------------------
    size = _next_pow2(3 * n - 2)   # enough for both convolutions

    # --------------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                     # base learning rate
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    # --------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------
    # initial exact score (objective = 2·n·max_b because sum(a)=1)
    ra = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(ra * ra, n=size)[: 2 * n - 1]
    max_b = conv_full.max()
    best_score = 2.0 * n * max_b
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # hyper‑parameters for the outer loop
    rand_prob = 0.08          # probability of a random transfer move
    rand_scale = 0.05         # size of random transfer
    stagnation_time = 30.0    # seconds without improvement → restart
    last_improve = start
    it = 0

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- 1. autocorrelation and exact score -----
        ra = np.fft.rfft(a, n=size)
        conv_full = np.fft.irfft(ra * ra, n=size)[: 2 * n - 1]
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ----- 2. adaptive smoothness parameter p -----
        # increase p slowly; p controls how close we are to the true max.
        p = min(32, 4 + it // 200000)       # 4 → 5 → … → 32 over the run

        # ----- 3. compute the smoothed gradient -----
        # 3a) weight vector w (soft‑max of the autocorrelation)
        pow_b = np.power(conv_full, p, dtype=np.float64)
        sum_pow = pow_b.sum()
        if sum_pow > 0.0:
            w = np.power(conv_full, p - 1, dtype=np.float64)
            w /= sum_pow ** ((p - 1) / p)   # normalise as described in the analysis
        else:
            # fallback to a strict max‑mask (behaves like sub‑gradient)
            w = (conv_full == max_b).astype(np.float64)

        # 3b) convolution a_rev * w  → gradient components
        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        w_fft = np.fft.rfft(w, n=size)
        conv_rw = np.fft.irfft(rev_fft * w_fft, n=size)[: 3 * n - 2]
        grad = 2.0 * conv_rw[n - 1 : n - 1 + n]   # central n entries

        # ----- 4. Adam update (projected onto the simplex) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- 5. occasional random mass‑transfer move (exploration) -----
        if np.random.rand() < rand_prob:
            a = _random_transfer(a, rand_scale)

        # ----- 6. projection onto the non‑negative simplex -----
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.ones_like(a) / n
        else:
            a /= s

        # ----- 7. enforce symmetry (keeps the search space small) -----
        a = (a + a[::-1]) / 2.0

        # ----- 8. restart if we have been stuck for a while -----
        if time.time() - last_improve > stagnation_time:
            a = np.random.rand(n)
            a /= a.sum()
            a = (a + a[::-1]) / 2.0
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{it:7d}] restart (stagnation)")

        it += 1

        # occasional status output
        if it % 50000 == 0:
            print(f"[{it:7d}] current best = {best_score:.6f}")

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    # return the best sequence found (as plain Python floats)
    return best_seq.tolist()
