"""
Hybrid optimisation for the step‑function evaluation problem.

The routine
* builds a diverse collection of initial sequences (geometric, power‑law,
  combined families, symmetric shapes, sine windows and a few random draws);
* runs a two‑phase Adam optimiser on the soft‑max parametrisation.
  – A warm‑up phase uses a smoothly increasing log‑sum‑exp weight
    (softmax over the convolution values) to drive the shape toward a flat
    autocorrelation.
  – A fine‑tuning phase switches to an exact sub‑gradient (one‑hot weight)
    of the current maximum, allowing precise reduction of the peak.
* finishes with a cheap stochastic local‑search that repeatedly perturbs the
  best sequence while staying on the probability simplex.
All work is performed inside the allotted time budget; the best sequence
found is returned as a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using an FFT of length a power of two.
    The result has length len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Parametric families (used as seeds)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (beta - 1))
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


# ----------------------------------------------------------------------
# Optimiser for a single length
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    warmup_iters: int,
    max_iters: int,
    init_lr: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑softmax parametrisation.
    Returns the best sequence produced during this run and its objective value.
    """
    # hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # initialise
    v = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(v)            # first moment
    v2 = np.zeros_like(v)           # second moment
    t = 0

    best_seq = a0.copy()
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                           # stays on the simplex

        # fast evaluation (sum(a)=1, so obj = 2*n*max(conv))
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(
                f"[adam] n={n:5d} it={it:5d} best={cur_val:.6f}"
            )

        # ---- weight vector for the gradient -------------------------
        if it <= warmup_iters:
            # smoothed max (log‑sum‑exp) – beta grows geometrically
            beta = 5.0 * (1.02 ** it)
            shift = cur_max
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                continue
            w /= w_sum
        else:
            # exact sub‑gradient: one‑hot at the maximal index
            w = np.zeros_like(conv)
            idx = int(np.argmax(conv))
            w[idx] = 1.0

        # ---- gradient wrt a (see derivation in the comments of the
        #      original script) ------------------------------------
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # ---- convert to gradient wrt logits (softmax) ---------------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update --------------------------------------------
        lr = init_lr / np.sqrt(t)          # simple decay
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic local refinement (simple perturb‑and‑project)
# ----------------------------------------------------------------------


def _local_refine(a: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex that accepts only improving moves.
    Returns the (possibly) improved sequence and its value.
    """
    best_seq = a.copy()
    conv = _conv_fft(best_seq, best_seq)
    best_val = 2.0 * n * np.max(conv)

    # step‑size schedule – start relatively large, cool down slowly
    step_scale = 0.02

    while time.time() < deadline:
        # pick two distinct positions
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # propose moving a small amount from i to j
        delta = step_scale * np.random.rand() * min(best_seq[i], 0.5)
        if delta <= 0:
            continue

        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # non‑negativity is already guaranteed by the construction
        conv_c = _conv_fft(cand, cand)
        cand_val = 2.0 * n * np.max(conv_c)

        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand
            print(f"[refine] new best={best_val:.6f}")

        # geometric cooling of the step size
        step_scale *= 0.99995
        if step_scale < 1e-6:
            step_scale = 1e-6

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects the time budget (default 1000 s)
    and returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    # keep a small safety margin for the runner that calls us
    deadline = time.time() + budget_s - 3.0

    # ------------------------------------------------------------------
    # We start with a moderately large length; increasing length generally
    # brings the attainable bound nearer to the theoretical limit 1.
    # ------------------------------------------------------------------
    lengths = [8000, 6000, 4000]  # fall‑backs if time runs out

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        # --------------------------------------------------------------
        # Build a modest but diverse set of seed sequences.
        # --------------------------------------------------------------
        seed_seq_list = []

        # uniform baseline (useful as a sanity check)
        seed_seq_list.append(np.ones(n, dtype=np.float64) / n)

        # geometric families (r = 1 - c/n)
        for c in np.linspace(0.5, 5.0, 8):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seq_list.append(_geom(n, r))

        # pure power‑law families
        for p in np.linspace(0.0, 2.2, 8):
            seed_seq_list.append(_power(n, p))

        # combined geometric × power families (limited sampling)
        for c in np.linspace(0.5, 4.0, 4):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 4):
                seed_seq_list.append(_geom_power(n, r, p))

        # symmetric geometric
        for c in np.linspace(0.5, 4.5, 5):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seq_list.append(_sym_geom(n, r))

        # sine window (smooth, centred)
        seed_seq_list.append(_sine_window(n))

        # Beta‑shaped families (U‑shaped and hump‑shaped)
        for a_param in (0.5, 1.0, 2.0):
            for b_param in (0.5, 1.0, 2.0):
                seed_seq_list.append(_beta_shape(n, a_param, b_param))

        # A few random Dirichlet draws (concentration 0.5 – 5.0)
        for conc in (0.3, 1.0, 3.0):
            dir_sample = np.random.dirichlet([conc] * n)
            seed_seq_list.append(dir_sample)

        # --------------------------------------------------------------
        # Quick scan with the cheap objective to get a quick global best.
        # --------------------------------------------------------------
        for s in seed_seq_list:
            cur_val = 2.0 * n * np.max(_conv_fft(s, s))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = s.copy()
                print(f"[scan] len={n} new global best={cur_val:.6f}")

        # --------------------------------------------------------------
        # Adam optimisation – each seed runs its own two‑phase Adam.
        # --------------------------------------------------------------
        for idx, s0 in enumerate(seed_seq_list):
            if time.time() > deadline:
                break

            # numbers of iterations are chosen so that we stay within budget
            warmup_iters = min(3000, n // 2)          # focus on smoothed max
            max_iters = max(8000, 2 * n)               # allow a long fine‑tune

            seq, val = _adam_optimize(
                a0=s0,
                n=n,
                deadline=deadline,
                warmup_iters=warmup_iters,
                max_iters=max_iters,
                init_lr=0.6,
            )
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[adam‑global] len={n} seed={idx} best={val:.6f}")

        # --------------------------------------------------------------
        # Local stochastic refinement around the best candidate for this length.
        # --------------------------------------------------------------
        if global_best_seq is not None and time.time() < deadline:
            refined_seq, refined_val = _local_refine(
                a=global_best_seq,
                n=n,
                deadline=deadline,
            )
            if refined_val < global_best_val:
                global_best_val = refined_val
                global_best_seq = refined_seq.copy()
                print(f"[overall‑refine] len={n} new best={refined_val:.6f}")

    # ------------------------------------------------------------------
    # Final fallback – should never be needed but guarantees a return.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] using uniform distribution")

    # Return as plain Python list of floats
    return global_best_seq.tolist()
