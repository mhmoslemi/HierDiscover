"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm uses three ideas that improve on the previous search:

1. **Rich analytic start** – a short sweep of power‑law families
   (including the classic inverse‑sqrt “arcsine”) is evaluated and the
   best is taken as the initial point.  If the global variable
   ``height_sequence_1`` (the best sequence found so far) is present it
   is also added to the pool.

2. **Smoothed‑max projected gradient descent** – we minimise a smooth
   approximation of the max‑convolution term

        Sβ(a) = (1/β)·log Σk exp(β·(a∗a)k)

   whose gradient is a simple weighted correlation
   `g = 2·correlate(p, a, mode='valid')` with the soft‑max weights
   `p`.  The update `a ← a – η·g` is projected onto the probability
   simplex and symmetrised (the optimum is expected to be symmetric).

   The smoothing parameter `β` and the step size `η` follow linear
   schedules: early iterations use a small `β` (smooth objective) and
   a relatively large step; later iterations increase `β` (approaching
   the true max) and shrink the step.

3. **Exploration & restart** – occasional random mass‑transfer moves
   (Metropolis acceptance) help escape tiny local minima, and a simple
   stagnation detector injects a small random jitter when no improvement
   has been seen for a while.

All operations stay in NumPy (SciPy is only used for the fast
correlation).  The routine respects the 1000 s budget, prints progress,
and finally returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np
from scipy.signal import correlate  # fast cross‑correlation (valid mode)


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a small amount of probability mass from a random index
    to another random index and re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    # amount limited by what is available at the source
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def initialise_candidates(n: int) -> list[np.ndarray]:
    """
    Build a short list of analytically motivated, normalised candidates.
    """
    cand = []

    # 1. Uniform
    cand.append(np.ones(n) / n)

    # 2. Classic arcsine (inverse‑sqrt product)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(a / a.sum())

    # 3. Pure inverse‑sqrt product (no 0.5 offset)
    a = 1.0 / np.sqrt((i + 1) * (n - i))
    cand.append(a / a.sum())

    # 4. Mixed inverse‑sqrt sums with several exponents
    for p in (0.3, 0.5, 0.7, 1.0):
        a = (i + 1) ** (-p) + (n - i) ** (-p)
        cand.append(a / a.sum())

    # 5. Pure power‑law families (single‑sided) – useful for fine tuning
    for p in np.linspace(0.2, 1.5, 66):
        a = (i + 1) ** (-p) + (n - i) ** (-p)      # symmetric sum of two tails
        cand.append(a / a.sum())

    return cand


def smoothed_max_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the gradient of the smooth max term
        Sβ(a) = (1/β)·log Σ exp(β·(a∗a))
    using the soft‑max weights.  The returned vector has length n.
    """
    # full convolution a∗a (length 2n‑1)
    conv = np.convolve(a, a)
    # numeric stabilisation
    max_c = conv.max()
    # soft‑max weights (size 2n‑1)
    with np.errstate(over='ignore'):
        w = np.exp(beta * (conv - max_c))
    p = w / w.sum()                     # Σ p = 1
    # gradient = 2·correlate(p, a)  (valid mode, length n)
    grad = 2.0 * correlate(p, a, mode='valid')
    return grad

    start_time = time.time()
    # keep a tiny safety margin so we never overrun the time limit
    deadline = start_time + budget_s - 0.5

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1 – construct a strong initial point
    # ------------------------------------------------------------------
    candidates = initialise_candidates(n)

    # if the harness supplied the previous best, use it as an extra seed
    if 'height_sequence_1' in globals():
        try:
            hs = np.asarray(height_sequence_1, dtype=float)
            if hs.shape[0] != n:
                # truncate or pad to the required length
                if hs.shape[0] > n:
                    hs = hs[:n]
                else:
                    hs = np.pad(hs, (0, n - hs.shape[0]), 'constant')
            hs = np.maximum(hs, 0.0)
            if hs.sum() > 0:
                hs = hs / hs.sum()
                candidates.append(hs)
        except Exception:
            pass  # ignore malformed global

    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates):
        # because all candidates are already normalised Σa = 1,
        # the objective reduces to 2·n·max(conv(a,a))
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"[init] candidate {idx:03d} : score = {score:.6f}")

    a = best_seq.copy()
    cur_score = best_score
    print(f"[init] chosen start score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – optimisation loop (smoothed‑max gradient + occasional random)
    # ------------------------------------------------------------------
    # schedule parameters
    beta_start = 2.0          # low smoothing at the beginning
    beta_end = 200.0          # approach true max in the end
    step_start = 0.4          # relatively aggressive initial step
    step_end = 1e-5           # fine‑grained refinements later
    temp_start = 0.08         # temperature for Metropolis acceptance
    temp_end = 1e-7

    momentum = np.zeros_like(a)   # for a simple exponential moving‑average
    momentum_coeff = 0.9

    iteration = 0
    last_improve = start_time

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 5.0)   # a smooth [0,1] indicator
        progress = min(max(progress, 0.0), 1.0)

        # linear schedules
        beta = beta_start + (beta_end - beta_start) * progress
        step = step_start + (step_end - step_start) * progress
        temp = temp_start + (temp_end - temp_start) * progress

        # ------------------------------------------------------------------
        # Gradient step using the smoothed max
        # ------------------------------------------------------------------
        grad = smoothed_max_gradient(a, beta)

        # momentum (exponential moving average of the gradient)
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * grad

        a_new = a - step * momentum
        a_new = project_simplex(a_new)
        # enforce symmetry – the optimum is expected to be symmetric
        a_new = (a_new + a_new[::-1]) * 0.5

        # occasional random local perturbation (helps escaping very flat plateaus)
        if np.random.rand() < 0.015:
            a_new = random_move(a_new, step * 0.5)

        # ------------------------------------------------------------------
        # Evaluate the true objective (exact max, not the smoothed one)
        # ------------------------------------------------------------------
        conv_new = np.convolve(a_new, a_new)
        new_score = 2.0 * n * conv_new.max()

        # Metropolis acceptance (allows uphill moves when temperature > 0)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability that decays with temperature
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                last_improve = time.time()
                print(f"[{iteration:06d}] NEW BEST = {best_score:.6f} (t={elapsed:.1f}s)")

        # ------------------------------------------------------------------
        # Stagnation handling – jitter around the best known point
        # ------------------------------------------------------------------
        if iteration % 50000 == 0 and (time.time() - last_improve) > 30.0:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            a = (a + a[::-1]) * 0.5
            momentum[:] = 0.0
            print(f"[{iteration:06d}] RESTART (stagnation) – jitter added")

        # occasional lightweight progress report
        if iteration % 8000 == 0:
            print(f"[{iteration:06d}] progress {progress*100:5.1f}% "
                  f"best {best_score:.6f} cur {cur_score:.6f}")

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, "
          f"time={total_time:.1f}s")
    # Return the best sequence found (as a plain Python list)
    return best_seq.tolist()
