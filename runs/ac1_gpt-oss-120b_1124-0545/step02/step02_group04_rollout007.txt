"""
Propose a step‑function that (approximately) minimizes

    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

under the constraints a_i ≥ 0 and sum(a)=√(2 n).

The algorithm is a deterministic projected sub‑gradient descent that
repeatedly reduces the current maximal convolution entry:

* Project the current vector onto the ℓ₁‑simplex (sum fixed, non‑negative).
* Find the index i* where the self‑convolution attains its maximum.
* The gradient of (a*a)_[i*] is 2·a reversed and shifted; a feasible
  direction that respects the sum‑constraint is d = -grad + mean(grad)·1.
* Take the largest step η that keeps all coordinates non‑negative,
  project back to the simplex and accept the move if the maximal
  convolution value decreases.
* If no improvement is observed for a long time a small random
  perturbation is applied to escape stagnation, and the whole process
  may be restarted near the best solution found so far.

The routine runs for at most ``budget_s`` seconds (default 1000) and
returns the best sequence encountered.  Intermediate progress is printed
as “new best …”.  All heavy work (convolution, projection) uses NumPy
and FFTs, so even for a few thousand coordinates the loop executes many
thousands of iterations within the time budget.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector `v` onto the simplex {x ≥ 0, sum(x)=s}.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]                 # sort in descending order
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Only happens when s == 0
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, nfft: int) -> float:
    """
    Return the maximal entry of the linear convolution a*a.
    The convolution is performed by FFT for speed.
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the evaluation
    function.  The routine respects a time budget (default 1000 s) and
    returns the best sequence found as a plain Python list.
    """
    # --------------------------------------------------------------
    # Initialise RNGs
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # Load a starting point if one is already available
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        n = a.size
    else:
        n = 2500                     # a modest size if nothing else is given
        a = np.random.exponential(scale=1.0, size=n)

    # Normalise to the conventional ℓ₁‑norm √(2 n)
    target_sum = math.sqrt(2.0 * n)
    a = a / a.sum() * target_sum

    # Length for the FFT (next power of two ≥ 2 n‑1)
    nfft = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Initial evaluation
    cur_max = _max_convolution(a, nfft)
    best_seq = a.copy()
    best_val = cur_max
    print(f"Initial max‑convolution: {best_val:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin
    iteration = 0
    no_improve = 0
    RESTART_AFTER = 50000                           # iterations without gain

    # --------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # ------- current convolution and the worst index ----------
        conv = np.fft.irfft(np.fft.rfft(a, n=nfft) ** 2, n=nfft)[: 2 * n - 1]
        max_idx = int(conv.argmax())
        cur_max = float(conv.max())

        # ------- track best ----------
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            print(f"[{iteration:7d}] new best: {best_val:.6f}")
            no_improve = 0

        # ------- gradient of the active quadratic term ----------
        idx = np.arange(n)
        partner = max_idx - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[partner[mask]]          # ∇(a*a)_{max_idx}

        # Projected feasible direction (sum‑preserving)
        lam = grad.mean()
        d = -grad + lam                               # sum(d) = 0

        # Largest admissible step before some coordinate hits zero
        neg = d < -1e-15
        if np.any(neg):
            eta_max = np.min(-a[neg] / d[neg])
        else:
            eta_max = target_sum                     # effectively unbounded

        # Try a few step‑size factors
        improved = False
        for factor in (0.5, 0.2, 0.1, 0.05, 0.02):
            eta = factor * eta_max
            a_candidate = a + eta * d
            # bring back to the simplex (exact sum, non‑negative)
            a_candidate = _simplex_projection(a_candidate, target_sum)
            new_max = _max_convolution(a_candidate, nfft)
            if new_max < cur_max - 1e-12:
                a = a_candidate
                cur_max = new_max
                improved = True
                break

        if not improved:
            no_improve += 1
            # occasional stochastic shake‑up to leave flat regions
            if iteration % 2000 == 0 and iteration > 0:
                a = best_seq + np.random.normal(scale=target_sum * 0.001, size=n)
                a = np.maximum(a, 0.0)
                a = _simplex_projection(a, target_sum)
                cur_max = _max_convolution(a, nfft)
                no_improve = 0

        # Restart if we have been stuck for a long time
        if no_improve >= RESTART_AFTER:
            a = best_seq + np.random.normal(scale=target_sum * 0.01, size=n)
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a, target_sum)
            cur_max = _max_convolution(a, nfft)
            no_improve = 0

        iteration += 1

    # --------------------------------------------------------------
    # A few gentle smoothing passes – they sometimes shave off a
    # fractional amount of the maximum without hurting the sum.
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)   # sum = 1
    for _ in range(3):

        smooth = _simplex_projection(smooth, target_sum)
        val = _max_convolution(smooth, nfft)
        if val < best_val:
            best_val = val
            best_seq = smooth

    print(f"Finished. Best max‑convolution: {best_val:.6f}")
    # --------------------------------------------------------------
    return best_seq.tolist()
