"""
Propose a high‑quality non‑negative step‑function sequence.

The algorithm combines:
  1. A low‑dimensional parametric family (tails + central plateau) that
     captures the known “root‑law” shape but adds a tunable plateau.
  2. A broad random search inside this family to obtain a very good
     starting point.
  3. A refinement phase that alternates
     – projected gradient steps on the active (maximal) convolution entry,
     – small random mass‑transfer moves,
     – a simple simulated‑annealing acceptance rule.
The whole search respects a 1000 s budget and returns the best sequence
encountered (as a plain list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------
def _conv_max_and_arg(seq: np.ndarray) -> tuple[float, int]:
    """Return (max(conv(seq,seq)), argmax) using an FFT‑based convolution."""
    n = seq.size
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())          # next power of two
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[:size]
    arg = int(np.argmax(conv))
    return float(conv[arg]), arg


def _score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective used by the harness:
        2 * n * max(conv(seq,seq)) / (sum(seq) ** 2)
    seq is assumed to be non‑negative; its sum must be > 0.
    """
    n = seq.size
    max_b, _ = _conv_max_and_arg(seq)
    s = seq.sum()
    if s < 0.01:                     # mirrors the safety check in evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _project_onto_simplex(x: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x >= 0, sum x = 1}.
    Simple clipping + renormalisation works because we never produce
    negative values after the gradient step that are large in magnitude.
    """
    x = np.maximum(x, 0.0)
    total = x.sum()
    if total > 0.0:
        x /= total
    else:                             # safety fallback – return uniform
        x[:] = 1.0 / x.size
    return x


def _tail_plateau_sequence(n: int,
                           p: float,
                           alpha: float,
                           L: int) -> np.ndarray | None:
    """
    Build a symmetric sequence consisting of two power‑law tails of length L
    and a flat plateau in the middle.
    - Tail:   a_i ∝ (i + alpha + 1) ** (-p)   for i = 0 .. L‑1
    - Plateau: constant value chosen so that the whole vector sums to 1.
    The shape is forced to be non‑negative; if the plateau would become
    negative we return None (the chosen tail is too heavy).
    """
    if L < 0 or 2 * L > n:
        return None
    # left tail values (including i = 0)
    i = np.arange(L, dtype=float)
    left = (i + alpha + 1.0) ** (-p)          # +1 avoids a singularity at i=0
    tail_sum = left.sum()

    # mass that must be placed on the plateau
    remaining = n - 2 * L
    if remaining <= 0:
        return None
    plateau_val = (1.0 - 2.0 * tail_sum) / remaining
    if plateau_val <= 0.0:
        return None                           # tails too heavy for this L

    # assemble the full vector
    seq = np.empty(n, dtype=float)
    seq[:L] = left
    seq[L:n - L] = plateau_val
    seq[n - L:] = left[::-1]                 # mirror the left tail

    # final safety projection (numerical noise)
    return _project_onto_simplex(seq)


def _gradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    One gradient descent step on the active convolution entry.
    The gradient of the entry (k_max) w.r.t. a_j is a_{k_max-j}
    (up to a factor of 2 which does not affect the direction).
    After a step we project back onto the simplex.
    """
    n = seq.size
    _, k_max = _conv_max_and_arg(seq)

    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)

    grad = np.zeros_like(seq)
    grad[idx] = seq[k_max - idx]            # gradient of the active entry
    grad -= grad.mean()                     # keep the sum unchanged

    new_seq = seq - step * grad
    return _project_onto_simplex(new_seq)


def _random_mass_transfer(seq: np.ndarray) -> np.ndarray:
    """
    Transfer a tiny random fraction of the mass from one index to another.
    The amount is drawn proportionally to the source value, guaranteeing
    that no entry becomes negative.
    """
    n = seq.size
    i, j = np.random.choice(n, 2, replace=False)
    # fraction of the source mass that we move (0.5 % – 1 %)
    eps = seq[i] * np.random.uniform(0.0005, 0.0100)
    eps = min(eps, seq[i])
    cand = seq.copy()
    cand[i] -= eps
    cand[j] += eps
    return _project_onto_simplex(cand)


# ----------------------------------------------------------------------
# Main entry point – the function demanded by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # 5 s safety margin

    # --------------------------------------------------------------
    # 0) decide on the length (use the global one if provided)
    # --------------------------------------------------------------
    try:
        init_global = np.array(height_sequence_1, dtype=float)   # type: ignore
        if init_global.size == 0 or init_global.sum() == 0:
            raise ValueError
        n = init_global.size
        print(f"[info] using global height_sequence_1 (n={n})")
    except Exception:
        n = 5000
        print(f"[info] falling back to n = {n}")

    # --------------------------------------------------------------
    # 1) Start from a decent deterministic baseline (the classic root law)
    # --------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=float)
    root = 1.0 / np.sqrt(idx)
    root /= root.sum()
    best_seq = root.copy()
    best_score = _score(best_seq)
    print(f"[baseline] root‑law score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Randomised search in the low‑dimensional tail+plateau family
    # --------------------------------------------------------------
    max_trials = 4000
    for trial in range(max_trials):
        if time.time() >= deadline:
            break

        # sample parameters
        p = np.random.uniform(0.35, 0.70)          # tail exponent
        alpha = np.random.uniform(0.0, 5.0)        # tail shift
        L = np.random.randint(1, n // 2)          # tail length

        cand = _tail_plateau_sequence(n, p, alpha, L)
        if cand is None:
            continue

        cand_score = _score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[shape] trial={trial:4d} p={p:.4f} α={alpha:.4f} L={L:4d}"
                  f" → score={cand_score:.6f}")

    # --------------------------------------------------------------
    # 3) Refinement – gradient + simulated annealing
    # --------------------------------------------------------------
    seq = best_seq.copy()
    cur_score = best_score
    it = 0

    # annealing schedule (linear in elapsed time)
    T0, Tf = 1.0, 1e-4
    # starting step size for the gradient move
    step = 0.30
    step_decay = 0.99995

    while time.time() < deadline:
        it += 1

        # ----- choose a move -------------------------------------------------
        if np.random.rand() < 0.65:                     # most of the time random transfer
            cand = _random_mass_transfer(seq)
        else:                                            # occasional gradient step
            cand = _gradient_step(seq, step)

        cand_score = _score(cand)

        # ----- acceptance rule (simulated annealing) -----------------------
        if cand_score < cur_score:
            accept = True
        else:
            elapsed = time.time() - start_time
            fraction = elapsed / (deadline - start_time)
            T = T0 * (1.0 - fraction) + Tf * fraction
            prob = math.exp(-(cand_score - cur_score) / T) if T > 0 else 0.0
            accept = np.random.rand() < prob

        if accept:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[improve] iter={it:6d} score={best_score:.6f}")

        # ----- decay the gradient step size ---------------------------------
        if it % 400 == 0:
            step = max(step * step_decay, 1e-7)

        # early exit when target is reached
        if best_score < 1.503:
            print("[target] reached target bound, stopping early")
            break

    print(f"[final] iterations={it}, best_score={best_score:.6f}")
    return best_seq.tolist()
