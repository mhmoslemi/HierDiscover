def propose_candidate(seed=..., budget_s=1000, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
    else:
        n=2500
        base = np.full(n, 1.0)

    S = math.sqrt(2.0 * n)

    # optionally start with edge-heavy shape:
    i = np.arange(n)
    a0 = 1.0/(i+1) + 1.0/(n-i)   # symmetric
    # Also could use random mixing:
    alpha = 0.5
    a = alpha*a0 + (1-alpha)*base
    # Normalize:
    a = a / a.sum() * S

    # Clip to <=1000 just for safe:
    a = np.clip(a, 0, 1000)

    # main loop
    best_seq = a.copy()
    # compute current objective:
    conv = np.convolve(a, a)
    best_score = float(conv.max())   # because normalized sum => objective = max(conv)

    # also maintain time
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # parameters
    t = 0
    # schedule for step size eta: start large, decay geometric.
    eta_start = 0.5   # maybe
    eta_end = 0.01

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        eta = eta_start * ((eta_end/eta_start) ** frac)

        # compute conv
        conv = np.convolve(a, a)
        max_val = conv.max()
        # mask of max indices within tolerance (maybe absolute diff < 1e-9)
        tolerance = 1e-12
        mask = (conv >= max_val - tolerance).astype(float)
        # this gives weight w_i = 1 for max indices else 0
        w = mask
        # compute gradient as earlier
        # Compute sigma:
        # cumsum_w = np.cumsum(w)
        # sigma = cumsum_w[n-1:2*n-1] - np.concatenate(([0.0], cumsum_w[:n-1]))
        cumsum_w = np.cumsum(w)
        sigma = cumsum_w[n-1:2*n-1] - np.concatenate(([0.0], cumsum_w[:n-1]))
        # first term: conv of w with reversed a
        a_rev = a[::-1]
        conv_wa = np.convolve(w, a_rev)
        first = conv_wa[n-1:n-1+n]

        grad = first + a * sigma   # shape n

        # exponentiated gradient update:
        # to avoid overflow/underflow, we can shift gradients
        # Apply a_i = a_i * exp(-eta * (grad - grad.mean()))  (mean subtraction ensures scale invariance)
        g = grad - np.mean(grad)
        a = a * np.exp(-eta * g)
        # re-normalize to sum S
        # note that exponents may propagate zeros, but we have positive a
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, S / len(a))
        else:
            a = a * (S / a_sum)

        # compute new conv and maybe update best
        conv = np.convolve(a, a)
        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print("[{:6d}] New best {:.6f} t={:.1f}s eta={:.5f}".format(t, best_score, elapsed, eta))

        # random perturbation occasionally
        if t % 400 == 0 and t>0:
            # random mixing
            rnd = np.random.exponential(scale=1.0, size=n)
            rnd = rnd / rnd.sum() * S
            mix = np.random.rand()
            a_pert = mix * best_seq + (1-mix) * rnd
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * (S / a_pert.sum())
            conv_p = np.convolve(a_pert, a_pert)
            score_p = float(conv_p.max())
            if score_p < best_score:
                best_score = score_p
                best_seq = a_pert.copy()
                print("[{:6d}] New best after perturbation {:.6f}".format(t, best_score))
            if score_p < cur_score:
                a = a_pert

        # occasionally restart if stagnation
        # we could track last_improve like previous code.

        t += 1

    return best_seq.tolist()
