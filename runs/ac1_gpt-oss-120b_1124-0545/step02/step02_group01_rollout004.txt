"""
Step‑function search for the 2·n·max(conv(a,a)) / (sum(a)²) objective.

The algorithm works in two phases for each candidate length:

1. **Adam on a smooth max** – we parametrize the step heights by a
   log‑softmax vector `v` (so a = softmax(v) lies on the probability
   simplex).  The loss is a smoothed maximum of the auto‑convolution
   obtained with the log‑sum‑exp trick.  The smoothing parameter β is
   increased quickly (β←β·1.05) so the optimisation soon focuses on the
   true max.  Adam updates keep the iterates on the simplex automatically.

2. **Projected sub‑gradient refinement** – starting from the best Adam
   result we perform a mirror‑descent style update that uses the exact
   sub‑gradient of the true max (the index where the convolution attains
   the maximum).  After each step we project back onto the simplex by
   normalising; the learning rate decays as 1/√t.

Both phases are time‑aware and stop as soon as the global deadline is
reached.  Several seed families (geometric, power‑law, combined,
symmetric, sine‑window and random) are tried for each length (4000 and
8000).  The best sequence seen across all runs is returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Parametric families for seeds (all returned already normalised to sum 1)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max of the auto‑convolution.
    Returns the best sequence (as a numpy array) encountered and its
    objective value (2·n·max(conv) because sum(a)=1).
    """
    # initialise log‑softmax parameters
    v = np.log(seed + 1e-30)

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # hyper‑parameters
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    # smoothing schedule (beta grows quickly)
    beta = 5.0
    beta_mul = 1.05

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                       # stays on the simplex

        # true objective (fast version – sum(a) == 1)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta *= beta_mul
        shift = np.max(conv)                   # shift = max(conv) ⇒ safe exponent
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:        # numerical safety
            continue
        w /= w_sum

        # ----- gradient w.r.t a (∇_a Σ w_k·conv_k) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # ----- convert to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient refinement (exact max, mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Refine a sequence using the true sub‑gradient of max(conv).
    We use an exponentiated‑gradient (mirror‑descent) update that keeps
    the vector on the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # Sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Mirror‑descent (exponentiated‑gradient) step
        # lr schedule decays like 1/√t, capped to avoid too large steps
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # for numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a modest collection of diverse starting points for length `n`.
    All seeds are already normalised to sum 1.
    """
    seeds: list[np.ndarray] = []

    # uniform baseline
    seeds.append(_uniform(n))

    # geometric families (r = 1 - c/n)
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power mix
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # smooth sine window (often a surprisingly good baseline)
    seeds.append(_sine_window(n))

    # a few random soft‑max vectors
    rng = np.random.default_rng(12345)
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse previously discovered sequence if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given time
    budget and returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # possibly re‑use a previously computed construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start with 4000, then 8000 if time permits
    # ------------------------------------------------------------------
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous)

        # quick scan of raw seeds
        for s in seeds:
            conv = _conv_fft(s, s)
            val = 2.0 * n * np.max(conv)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1: Adam on a smoothed max for each seed
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline,
                                      max_iters=3000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall] n={n} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2: Exact sub‑gradient refinement from the current best
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_ref, val_ref = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=15000)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall] after sub‑grad n={n} best={val_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    # Final sanity check – evaluate via the provided function (optional)
    try:
        from __main__ import evaluate_sequence  # noqa: F401
    except Exception:
        # evaluate_sequence is already in the global namespace in the runner
        pass

    # Return as plain Python list
    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
```

This implementation expands on the previous approach by:

* accelerating the smoothing schedule (β grows faster) so the Adam optimiser
  focuses on the true max earlier;
* employing a dedicated mirror‑descent sub‑gradient stage that works on the
  exact max of the convolution;
* using a compact but diverse set of seeds and re‑using any previously
  discovered construction;
* being fully time‑aware and reporting progress.

It respects the 1000 s budget, works on the probability simplex,
and returns a list of non‑negative heights ready for `evaluate_sequence`.
