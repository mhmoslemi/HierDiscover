def compute_gradient(a, alpha):
    conv = np.fft.irfft(np.fft.rfft(a, nfft) * np.fft.rfft(a, nfft), n=nfft)[:2*n-1] # conv_i
    # compute softmax weights

    exp_weights = np.exp(alpha * (conv - max_conv))
    sum_weights = exp_weights.sum()
    w = exp_weights / sum_weights  # length 2n-1

    # compute w_rev
    w_rev = w[::-1]

    # convolution of a and w_rev via FFT
    n_fft = 1 << ((len(a) + len(w_rev) - 1).bit_length())
    grad_full = np.fft.irfft(np.fft.rfft(a, n_fft) * np.fft.rfft(w_rev, n_fft), n=n_fft)

    offset = len(w_rev) - 1  # = 2n-2 
    grad = 2 * grad_full[offset - np.arange(len(a))]
    return grad

def _gradient_and_max(a, alpha, nfft1, nfft2):
    n = a.shape[0]
    # compute conv for smoothing
    af = np.fft.rfft(a, n=nfft1)
    conv_full = np.fft.irfft(af * af, n=nfft1)[:2*n-1]  # conv
    max_c = conv_full.max()
    # compute softmax weights
    # shift for numeric stability
    exp_args = alpha * (conv_full - max_c)
    # to avoid overflow/underflow:
    np.clip(exp_args, -700, 700, out=exp_args)  # np.exp limit
    exp_vals = np.exp(exp_args)
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp

    # reverse w
    rev_w = w[::-1]

    # convolution a with rev_w
    af2 = np.fft.rfft(a, n=nfft2)
    wr = np.fft.rfft(rev_w, n=nfft2)
    conv_ab = np.fft.irfft(af2 * wr, n=nfft2)[: (n + len(rev_w) - 1)]

    offset = len(rev_w) - 1  # = 2n-2
    grad = 2.0 * conv_ab[offset - np.arange(n)]

    return grad, max_c

"""
Gradient‑based annealing for the max‑self‑convolution objective.

The algorithm treats the objective  max(conv(a,a))  (the sum of a is kept
fixed) and minimises a smooth approximation of the max‑function
(log‑sum‑exp).  The gradient of the smooth max can be written in closed
form using two FFT‑based convolutions, so each iteration costs only
O(n log n).  A decreasing temperature controls a stochastic step size
and a Metropolis acceptance rule, while a simplex projection enforces
the ℓ₁‑normalisation Σa = √(2 n) and non‑negativity.  The routine runs
until the time budget (≈1000 s) is exhausted and returns the best
sequence found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector v onto the simplex {x ≥ 0, Σx = s}.
    Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT of length nfft.
    Returns the full convolution vector (length 2·len(a)‑1).
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(
    a: np.ndarray,
    alpha: float,
    nfft1: int,
    nfft2: int,
) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the smooth max  (log‑sum‑exp) of the
    self‑convolution of a, together with the current maximal entry.
    Both convolutions are performed via FFT.
    """
    n = a.size

    # ----- 1) self‑convolution (for the weights) -----
    conv = _max_convolution_fft(a, nfft1)          # length 2n‑1
    max_c = conv.max()

    # weights w_i = exp(alpha*(conv_i - max_c)) / Σ
    # numeric stabilisation – clip exponents
    exp_arg = alpha * (conv - max_c)
    np.clip(exp_arg, -700, 700, out=exp_arg)       # avoid overflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ----- 2) gradient = 2 * (a * reverse(w))_{offset - k} -----
    rev_w = w[::-1]                                # length 2n‑1
    # convolution a (length n) with rev_w (length 2n‑1)
    af = np.fft.rfft(a, n=nfft2)
    wf = np.fft.rfft(rev_w, n=nfft2)
    full = np.fft.irfft(af * wf, n=nfft2)
    full = full[: n + rev_w.size - 1]              # keep only the linear part
    offset = rev_w.size - 1                         # = 2n‑2
    grad = 2.0 * full[offset - np.arange(n)]

    return grad, max_c


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises

        E(a) = 2·n·max(conv(a,a)) / (sum(a)²)

    (equivalently minimise max(conv) after fixing Σa = √(2 n)).
    The routine runs for at most ``budget_s`` seconds and returns the
    best sequence discovered.
    """
    # ------------------------------------------------------------------
    # reproducibility
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # problem size ----------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        n = a.size
    else:
        n = 3000                               # a modest yet expressive size
        a = np.random.exponential(scale=1.0, size=n)

    # normalise to the conventional ℓ₁‑norm √(2 n)
    target_sum = math.sqrt(2.0 * n)
    a = _simplex_projection(a, target_sum)

    # FFT lengths (powers of two for efficiency)
    nfft1 = 1 << ((2 * n - 1).bit_length())          # for a*a
    nfft2 = 1 << ((3 * n - 2).bit_length())          # for a * rev(w)

    # ------------------------------------------------------------------
    # initial evaluation
    cur_max = _max_convolution_fft(a, nfft1).max()
    best_seq = a.copy()
    best_score = cur_max

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    iteration = 0

    # annealing schedule
    T_start = 1.0
    T_end = 1e-5

    # smoothing (log‑sum‑exp) parameter schedule
    alpha_start = 10.0
    alpha_end = 200.0

    # base step size (fraction of the target sum)
    step_start = target_sum * 1e-3
    step_end = target_sum * 1e-5

    # noise amplitude (decays to zero)
    noise_start = target_sum * 5e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)

        # temperature for Metropolis rule
        temperature = T_start * ((T_end / T_start) ** frac)

        # schedule parameters
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        step = step_start + (step_end - step_start) * frac
        noise_amp = noise_start * (1.0 - frac)

        # ----- gradient of the smooth max -----
        grad, conv_max = _smooth_max_gradient(a, alpha, nfft1, nfft2)

        # ----- stochastic projected gradient step -----
        a_prop = a - step * grad
        if noise_amp > 0:
            a_prop += noise_amp * np.random.randn(n)
        a_prop = _simplex_projection(a_prop, target_sum)

        # ----- evaluate the genuine objective (max convolution) -----
        new_max = _max_convolution_fft(a_prop, nfft1).max()

        # ----- Metropolis acceptance -----
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            try:
                prob = math.exp(-(new_max - cur_max) / (temperature + 1e-12))
            except OverflowError:
                prob = 0.0
            if random.random() < prob:
                accept = True

        if accept:
            a = a_prop
            cur_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_seq = a.copy()
                print(f"[{iteration:7d}] new best max_conv = {best_score:.6f}"
                      f" (time {elapsed:.1f}s)")

        # occasional global perturbation to escape flat plateaux
        if iteration % 50000 == 0 and iteration > 0:
            a += np.random.normal(scale=target_sum * 1e-3, size=n)
            a = _simplex_projection(a, target_sum)
            cur_max = _max_convolution_fft(a, nfft1).max()

        # stop early if we already beat the target bound
        if best_score < 1.503:
            print(f"Target bound reached (score {best_score:.6f}) – stopping.")
            break

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best vector as a plain Python list
    return best_seq.tolist()
