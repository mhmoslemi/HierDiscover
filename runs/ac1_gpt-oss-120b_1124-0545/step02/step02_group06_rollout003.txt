def proj_simplex(v):
    # returns projection onto simplex {x_i >=0, sum x_i = 1}
    # (not restricted to dimension)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1

    w = np.maximum(v - theta, 0)

    # v: np array (any shape)
    # returns projection
    if v.ndim != 1: v = v.ravel()
    n = v.shape[0]
    u = np.sort(v)[::-1]

"""
Propose a non‑negative step‑function sequence that minimises
`evaluate_sequence`.  The method consists of:

1. initialisation (global solution or 1/√(i+1) root law);
2. random sampling of several parametric families (power‑law,
   exponential, two‑level step, logistic) – fast FFT based scoring;
3. projected gradient descent on a soft‑max approximation of the
   convolution maximum, with a temperature schedule (γ) that
   gradually focuses the optimisation on the true max.  The iterate
   is kept on the probability simplex via Euclidean projection.
The whole search respects the given time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _initial_root_sequence(n: int) -> np.ndarray:
    """Root law a_i ∝ 1/√(i+1) normalised to unit sum."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _power_law_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """a_i ∝ (i+α+1)^‑p, normalised to unit sum."""
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _two_level_sequence(n: int, split: float, ratio: float) -> np.ndarray:
    """
    Two‑level step: first `split` proportion of indices get weight `ratio`,
    the rest get weight 1, then normalised.
    """
    split_idx = int(np.clip(split * n, 1, n - 1))
    a = np.ones(n, dtype=float)
    a[:split_idx] *= ratio
    a /= a.sum()
    return a


def _logistic_sequence(n: int, mu: float, scale: float) -> np.ndarray:
    """a_i ∝ 1 / (1 + exp((i‑mu)/scale)), normalised."""
    idx = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((idx - mu) / scale))
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _score(seq: np.ndarray) -> float:
    """
    Fast version of the objective:
        2 * n * max(conv(seq,seq)) / (sum(seq)**2)
    The sequence is assumed non‑negative and already normalised
    (sum≈1) – this mirrors `evaluate_sequence` without its safety
    checks.
    """
    n = seq.size
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:size]
    max_b = float(conv.max())
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0, sum x_i = 1 }.
    """
    if v.ndim != 1:
        v = v.ravel()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]  # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Stable soft‑max of γ·conv."""
    # conv is a non‑negative probability distribution (sum≈1)
    # we compute w_i = exp(gamma*conv_i) / Σ exp(gamma*conv)
    # using the log‑sum‑exp trick.
    z = gamma * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()
    return w


def _gradient_softmax(seq: np.ndarray, w: np.ndarray, m: int) -> np.ndarray:
    """
    Compute the gradient of Σ_i Σ_j a_i a_j w_{i+j}
    w.r.t. the sequence `a` (which is `seq`), using FFT.
    Resulting gradient is 2 * (rev(seq) * w)_{n‑1 … n‑1+n-1}.
    """
    n = seq.size
    # reverse the sequence
    rev = seq[::-1]
    # FFT of rev and w (both zero‑padded to length m)
    F_rev = np.fft.rfft(rev, n=m)
    F_w = np.fft.rfft(w, n=m)
    # convolution (real part)
    conv_rev = np.fft.irfft(F_rev * F_w, n=m)
    # extract the relevant n entries
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad
