def _local_random_search(seq, deadline, rng, best_score):

"""
Improved search for a low‑evaluation step‑function.

The algorithm proceeds in three phases:

1️⃣  Systematic exploration of analytic families.
    • Power‑law shapes a_i = (i+δ)^(-p) possibly with an additional
      exponential factor exp(-c·i/n).  The parameters (p,δ,c) are
      swept on a fine grid around the known near‑optimal exponent p≈0.5.
    • Extended beta‑like shapes a_i = (i+1)^α·(n‑i)^β with α∈[-1,2],
      β∈[0,2] in small steps.
    • A few classic families (constant, linear, triangular, sinusoidal)
      are retained for completeness.
    Every raw shape is normalised (max < 0.9, Σa=1) – the scaling does not
    influence the evaluation because the objective is homogeneous.
    Candidates whose cheap Fourier‑based estimate 2·n·max(conv) beats the
    current best by >5 % are sent to the official ``evaluate_sequence``.

2️⃣  Sub‑gradient refinement.
    Starting from the best candidate, we minimise the exact objective
    L(a)=2·n·max_k(a∗a)_k via a multiplicative (exponentiated‑gradient)
    update.  The step‑size decays exponentially, occasional block
    reversals and a random “mass‑shuffle’’ move are injected to escape
    plateaus.  The best point seen is remembered.

3️⃣  Simple local random‑search (hill‑climbing).
    With the remaining budget we repeatedly transfer a small random
    amount of mass between two positions, accept the move if the exact
    score improves, and continue.  This often yields a final polishing
    step that pushes the bound below the previous 1.5665‑level.

All helper routines are top‑level (no closures) and the implementation
uses only ``numpy`` and the provided ``evaluate_sequence``.  The routine
conforms to the required API and respects the 1000 s budget.
"""

def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """
    Scale an array so that its maximum is ≤0.9 and its sum is at least 0.01.
    The scaling does not affect the final objective value.
    """
    a = np.asarray(a, dtype=float)
    if a.size == 0:
        raise ValueError("empty sequence")
    max_v = a.max()
    if max_v > 0.0:
        a = a * (0.9 / max_v)      # keep values below the 1000‑cap
    total = a.sum()
    if total < 0.01:
        if total > 0.0:
            a = a * (0.01 / total)
        else:
            a = a + (0.01 / a.size)
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using an FFT of power‑of‑two length.
    The returned array has length 2·len(a)‑1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)
    return conv[: 2 * n - 1]


def _gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of L(a)=max_k (a∗a)_k at the active index k.
    Returns g with g_i = 2·a_{k‑i} if 0≤k‑i<n, otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k - i
    g[i] = 2.0 * a[j]
    return g

def _candidate_power_law(n: int):
    """Yield power‑law (optionally multiplied by a mild exponential) shapes."""
    i = np.arange(n, dtype=float)
    p_vals = np.linspace(0.45, 0.55, 21)        # fine sweep around 0.5
    delta_vals = np.array([0.0, 0.5, 1.0, 2.0, 4.0])
    c_vals = np.array([0.0, 0.2, 0.5, 1.0])    # exponential tail strength
    for p in p_vals:
        for delta in delta_vals:
            # avoid division by zero when p>0 and delta==0
            if delta == 0.0 and p > 0.0:
                continue
            base = (i + delta) ** (-p)
            for c in c_vals:
                if c == 0.0:
                    cand = base
                else:
                    cand = base * np.exp(-c * i / n)
                yield cand


def _candidate_beta_like(n: int):

    i = np.arange(1, n + 1, dtype=float)
    alphas = np.linspace(-1.0, 2.0, 13)   # -1 … 2 in steps of 0.25
    betas = np.linspace(0.0, 2.0, 9)     # 0 … 2 in steps of 0.25
    for α in alphas:
        for β in betas:
            cand = (i ** α) * ((n - i + 1) ** β)
            yield cand


def _candidate_analytic(n: int):
    """A small set of classic analytic shapes for coverage."""
    # constant
    yield np.ones(n, dtype=float)
    # linear decreasing / increasing
    yield np.arange(n, 0, -1, dtype=float)
    yield np.arange(1, n + 1, dtype=float)
    # triangular mountain
    t = np.arange(1, n + 1, dtype=float)
    yield np.minimum(t, n - t + 1)
    # sine‑squared bump
    yield np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2


def _enumerate_candidates(n: int):
    """Yield raw (un‑normalised) candidate vectors for length *n*."""
    # order of families is chosen to favour cheap but promising shapes first
    for raw in _candidate_analytic(n):
        yield raw
    for raw in _candidate_beta_like(n):
        yield raw
    for raw in _candidate_power_law(n):
        yield raw

def _subgradient_optimize(start_seq: np.ndarray,
                          deadline: float,
                          start_time: float,
                          rng: np.random.Generator,
                          target_score: float = 1.48) -> np.ndarray:
    """
    Exponentiated‑gradient (multiplicative) sub‑gradient descent.
    The input sequence must already be normalised (non‑negative,
    Σa_i = 1).  The routine returns the best sequence found before the
    deadline.
    """
    a = start_seq.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # step‑size schedule – exponential decay
    start_eta, end_eta = 0.8, 0.001

    iteration = 0
    while time.time() < deadline:
        # ---- evaluate current point (cheap approximation) -----------------
        conv = _conv_fft(a)
        cur_approx = 2.0 * n * conv.max()
        # verify exact score only when promising
        if cur_approx < best_score * 1.05:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"Sub‑grad new best score={best_score:.6f}")
                if best_score <= target_score:
                    print("[*] Target reached – exiting sub‑gradient phase.")
                    break

        # ---- sub‑gradient step -----------------------------------------
        k_star = int(conv.argmax())
        g = _gradient_at_max(a, k_star)
        g_max = g.max()
        if g_max == 0.0:
            break

        # time‑dependent step size (exponential decay)
        elapsed = time.time() - start_time
        total = deadline - start_time
        eta = start_eta * (0.5 ** (elapsed / total)) + end_eta

        factor = np.exp(-eta * g / g_max)          # element‑wise
        a_new = a * factor
        s = a_new.sum()
        if s == 0.0:
            a_new = a.copy()
        else:
            a_new = a_new / s                       # simplex projection

        # occasional random block reversal (escape plateaus)
        if rng.random() < 0.01:
            pos = rng.integers(0, n)
            length = rng.integers(1, min(256, n - pos) + 1)
            a_new[pos:pos + length] = a_new[pos:pos + length][::-1]

        # occasional “mass‑shuffle’’ move
        if rng.random() < 0.01:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a_new[i] > 0.0:
                eps = rng.uniform(0.0, 0.5) * a_new[i]
                a_new[i] -= eps
                a_new[j] += eps

        a = a_new
        iteration += 1

    return best_a


def _local_random_search(seq: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         current_best: float) -> np.ndarray:
    """
    Very cheap hill‑climbing: repeatedly move a small random amount of mass
    from one position to another and keep the move if it improves the
    exact score.
    """
    a = seq.copy()
    n = a.shape[0]
    best_score = current_best
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # transfer up to 5 % of the source mass
        eps = rng.uniform(0.0, 0.05) * a[i]
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        a_new /= a_new.sum()
        # cheap proxy to cut most useless trials
        conv = _conv_fft(a_new)
        approx = 2.0 * n * conv.max()
        if approx >= best_score * 0.98:
            continue
        exact = evaluate_sequence(a_new.tolist())
        if exact < best_score:
            best_score = exact
            a = a_new
            print(f"[{time.time() - start_time:.1f}s] "
                  f"Random‑search new best={best_score:.6f}")
    return a

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient list that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget (≈1000 s) and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any
    # --------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(np.asarray(hs1, dtype=float))
        seq = seq / seq.sum()
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from "
                  f"height_sequence_1 → score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – systematic grid scan of analytic families
    # --------------------------------------------------------------
    # Lengths to explore – small lengths are cheap probes,
    # the longest length (8192) gave the previous best.
    lengths = [1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.1f}s] Scanning length {n}")
        for raw in _enumerate_candidates(n):
            if time.time() >= deadline:
                break
            # normalise and enforce Σa=1 for a fair comparison
            seq = _normalise_seq(raw)
            seq = seq / seq.sum()
            # cheap Fourier‑based upper bound
            conv = _conv_fft(seq)
            approx_score = 2.0 * n * conv.max()
            # evaluate precisely only if promising
            if approx_score < best_score * 1.10:
                exact = evaluate_sequence(seq.tolist())
                if exact < best_score:
                    best_score = exact
                    best_seq = seq.tolist()
                    print(f"[{time.time() - start_time:.1f}s] "
                          f"New best (len={n}) score={best_score:.6f}")
                # early exit if we are already very close to target
                if best_score <= 1.48:
                    break

    # ------------------------------------------------------------------
    # Guard: if nothing passed the filter, start from a random vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8192
        best_seq = rng.random(n).tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time() - start_time:.1f}s] Random start → "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement on the best length
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)
    cur = cur / cur.sum()                         # enforce Σa=1

    refined = _subgradient_optimize(cur, deadline, start_time, rng)

    # ------------------------------------------------------------------
    # Phase 3 – cheap local random search for final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        refined = _local_random_search(refined, deadline,
                                       start_time, rng, best_score)

    # final verification – keep the absolute best
    final_score = evaluate_sequence(refined.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = refined.tolist()
        print(f"[{time.time() - start_time:.1f}s] Final exact best "
              f"score={best_score:.6f}")

    print(f"[{time.time() - start_time:.1f}s] Finished – best score={best_score:.6f}")

    # Return a plain Python list of non‑negative floats
    return [float(x) for x in best_seq]
