    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, N)

def project_onto_simplex(v):
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last positive
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

        # gradient via convolution with reversed weights
        w_rev = w[::-1]
        grad_full = _conv_fft(w_rev, a)
        # slice required part

        grad = 2.0 * grad_slice

        # gradient descent step
        a_new = a - step * grad
        # projection onto simplex
        a_new = _project_simplex(a_new)

        # occasionally random swap
        if iter % 100 == 0:
            i, j = np.random.randint(0, n, size=2)

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm works on a single length‑n probability vector (sum = 1) and
explicitly minimises the quantity

    score = 2 * n * max( a * a )                (· denotes linear convolution)

which is exactly the value returned by ``evaluate_sequence`` for a
properly normalised vector.

The optimisation proceeds in three phases:

1️⃣  **Initialisation** – start from a simple power‑law (the best known
    shape from earlier runs) or from the global variable
    ``height_sequence_1`` if it is available.

2️⃣  **Smoothed‑max gradient descent** – each iteration evaluates the
    full self‑convolution via an FFT (O(n log n)).  The maximum is
    approximated by a smooth function
        w_i ∝ (c_i / max(c))^p
    where ``c`` is the convolution vector and ``p`` is slowly increased.
    The gradient of the smoothed maximum is computed analytically
    (again using an FFT) and a projected gradient step keeps the vector
    on the probability simplex.

3️⃣  **Exploration & final selection** – occasional random mass swaps
    help escape plateaus; the best vector observed during the run is
    returned.

The whole search respects the 1000 s budget, uses only NumPy (no
external LP solvers) and returns a plain ``list[float]``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised non‑negative sequence a_i ∝ (i+1+offset)^{-α}.
    The result sums to 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (fast for the modest sizes we use).
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()               # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    conv = np.fft.irfft(Z, N)[:n]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum(x_i) = 1}
    """
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho value
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_sequence(seq: np.ndarray) -> float:
    """
    Exact score used by `evaluate_sequence` for a vector that already
    satisfies sum(seq) == 1.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)          # NumPy’s direct convolution (fast enough)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  The function respects the given time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Safety margin – leave a few seconds for the final return
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Choose the length and an initial vector
    # --------------------------------------------------------------
    try:
        # the harness may provide a previously discovered vector
        global height_sequence_1
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim != 1 or init.size == 0:
            raise ValueError
        n = init.size
        a = np.maximum(init, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            raise ValueError
    except Exception:
        # fall back to a modest length and a power‑law start
        n = 3000                       # moderate size – good speed / quality trade‑off
        a = _power_sequence(n, alpha=0.5045)   # exponent close to the previous best

    # Keep the best observed vector
    best_seq = a.copy()
    best_score = _score_from_sequence(a)

    # --------------------------------------------------------------
    # 2. Parameters for the smoothed‑max descent
    # --------------------------------------------------------------
    step = 0.25                     # initial step size (will decay)
    step_decay = 0.9999
    p = 1.0                         # smoothing exponent (will increase)
    p_inc = 0.001
    max_p = 40.0

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- current convolution and score -----
        conv = _conv_fft(a, a)                # length 2*n-1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[iter {iteration:6d}] score={cur_score:.6f}  (p={p:.3f}, step={step:.6f})")

        # ----- build smoothed weights for the max -----
        if max_conv == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            rel = conv / max_conv               # values in [0,1]
            w = np.power(rel, p)                # emphasize large entries
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

        # ----- gradient of the smoothed max -----
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        w_rev = w[::-1]
        grad_full = _conv_fft(w_rev, a)       # length 3*n-2
        # extract the slice that corresponds to i = 0..n-1
        grad_slice = grad_full[n - 1: 2 * n - 1][::-1]
        grad = 2.0 * grad_slice                # shape (n,)

        # ----- projected gradient step -----
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        # occasional random local move to help escape flat regions
        if iteration % 100 == 0:
            i, j = np.random.choice(n, size=2, replace=False)
            eps = min(a_new[i], a_new[i] * 0.5)
            if eps > 0:
                a_tmp = a_new.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                a_tmp = _project_simplex(a_tmp)
                a_new = a_tmp

        a = a_new

        # ----- schedule updates -----
        step = max(step * step_decay, 1e-7)
        p = min(p + p_inc, max_p)

    # --------------------------------------------------------------
    # 3. Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f}  (n={n})")
    return best_seq.tolist()
