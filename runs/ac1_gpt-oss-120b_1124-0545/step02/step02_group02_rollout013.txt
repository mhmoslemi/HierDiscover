Implementation code:

```
def _project_simplex(v):
    """
    Euclidean projection of v onto probability simplex {x >=0, sum x = 1}.
    """
    # algorithm: sort v descending, find rho such that v_i - (cumulative sum -1)/i > 0 ...
    # Implementation from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function sequence with a low evaluation score.

The search consists of three stages:

1. **Parametric families** –  A grid search over a power‑law family
   a_i ∝ (i+1+δ)^{-α} (with α≈0.5) is performed to obtain a good
   starting point.  The best sequence (normalised to sum 1) is kept.

2. **Smoothed max‑gradient descent** –  Starting from the best power‑law
   vector we minimise the exact score
   `2·n·max(conv(a,a))` while keeping a on the probability simplex.
   The non‑smooth “max’’ is approximated by a softmax with an
   annealing temperature λ.  Gradients are obtained by FFT‑based
   convolution, a momentum term is used, and the iterate is projected
   onto the simplex after every step.  λ and the step size are
   increased / decreased gradually so that the optimisation first
   explores a large region and then focuses on the true maximum.

3. **Local random swaps** –  Every few hundred iterations a tiny
   random mass exchange (i↔j) is tried; if it improves the score the
   swap is kept.  This helps the optimiser escape tiny plateaus.

The whole routine respects a user‑provided time budget (default
1000 s).  The returned list is a non‑negative sequence whose sum is
>0.01, suitable for the supplied `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """Return a normalised non‑negative power‑law sequence a_i ∝ (i+1+offset)^(-α)."""
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _score_from_sequence(seq: np.ndarray) -> float:
    """Exact score used by `evaluate_sequence` for a normalised sequence (sum == 1)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implementation of the algorithm by Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_conv_self(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length = 2·len(seq)-1)."""
    fa = np.fft.rfft(seq, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    conv_len = 2 * seq.shape[0] - 1
    return conv_full[:conv_len]


def _fft_conv_weight_seq(weight: np.ndarray, seq: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear convolution weight * seq (full length) via FFT.
    Returns the whole convolution (length = len(weight)+len(seq)-1).
    """
    fa = np.fft.rfft(seq, n=fft_len)
    fw = np.fft.rfft(weight, n=fft_len)
    conv_full = np.fft.irfft(fw * fa, n=fft_len)
    return conv_full  # caller will slice the needed part


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The algorithm runs for at most ``budget_s``
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ----------------- 1. choose a length ---------------------------------
    n = 5000                                        # good trade‑off between speed and quality

    # ----------------- 2. parametric search (power‑law) -------------------
    best_seq = None
    best_score = np.inf
    best_alpha = 0.5
    best_offset = 0.0

    # coarse scan
    for alpha in np.arange(0.48, 0.701, 0.02):
        if time.time() >= deadline:
            break
        seq = _power_sequence(n, alpha)
        score = _score_from_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_alpha = alpha
            best_offset = 0.0
            print(f"[coarse] α={alpha:.4f}, score={score:.6f}")

    # fine scan around the best α
    fine_alphas = np.linspace(best_alpha - 0.02, best_alpha + 0.02, 81)
    for alpha in fine_alphas:
        if time.time() >= deadline:
            break
        seq = _power_sequence(n, alpha)
        score = _score_from_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_alpha = alpha
            print(f"[fine]   α={alpha:.5f}, score={score:.6f}")

    # integer offset search (δ = 0 … 5)
    for delta in range(0, 6):
        if time.time() >= deadline:
            break
        seq = _power_sequence(n, best_alpha, offset=float(delta))
        score = _score_from_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_offset = float(delta)
            print(f"[offset] δ={delta}, score={score:.6f}")

    # ----------------- 3. smoothed max gradient descent --------------------
    # FFT parameters (next power of two ≥ 3n-2)
    conv_grad_len = 3 * n - 2
    fft_len = 1 << ((conv_grad_len - 1).bit_length())

    # optimiser hyper‑parameters
    lam = 5.0            # initial soft‑max temperature
    lam_factor = 1.02    # geometric increase per iteration
    lam_max = 200.0

    step = 0.2           # base step size
    step_decay = 0.9995  # geometric decay
    beta = 0.9           # momentum coefficient

    momentum = np.zeros_like(best_seq)
    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1

        # 1) self‑convolution via FFT
        conv = _fft_conv_self(best_seq, fft_len)          # length 2n-1
        max_conv = conv.max()

        # 2) soft‑max weights over convolution entries
        shifted = conv - max_conv                         # stabilise exponentials
        exp_vals = np.exp(lam * shifted)
        w = exp_vals / exp_vals.sum()                     # length 2n-1, sums to 1

        # 3) gradient of the smoothed max (2 * w * a convolution)
        grad_full = _fft_conv_weight_seq(w, best_seq, fft_len)  # length ~3n-2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]           # slice = n entries

        # 4) keep the iterate on the affine hyperplane Σa = 1
        grad -= grad.mean()

        # 5) momentum update and projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        new_seq = best_seq - step * momentum
        new_seq = _project_simplex(new_seq)

        # 6) evaluate
        cur_score = _score_from_sequence(new_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            last_improve_iter = iteration
            print(f"[improve] iter={iteration:6d}, λ={lam:.2f}, step={step:.5f}, score={best_score:.6f}")

        else:
            # keep the previous sequence if no improvement
            best_seq = best_seq

        # 7) occasional random tiny mass exchange (helps escape tiny plateaus)
        if iteration % 70 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(step * best_seq[i], 0.3 * best_seq[i])
            cand = best_seq.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = _project_simplex(cand)
            cand_score = _score_from_sequence(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve_iter = iteration
                print(f"[swap]   iter={iteration:6d}, score={best_score:.6f}")

        # 8) adapt hyper‑parameters
        step = max(step * step_decay, 1e-7)
        lam = min(lam * lam_factor, lam_max)

        # safety – stop if we have not improved for a long while
        if iteration - last_improve_iter > 20000:
            # restart from a fresh random point (keeps exploration alive)
            seq = np.random.rand(n)
            seq = _project_simplex(seq)
            best_seq = seq.copy()
            last_improve_iter = iteration
            step = 0.2
            lam = 5.0
            momentum.fill(0.0)
            print(f"[restart] iter={iteration}, resetting optimiser")

    # ----------------------------------------------------------------------
    # Final report
    # ----------------------------------------------------------------------
    final_score = _score_from_sequence(best_seq)
    print(f"[final] best_score={final_score:.6f}, α={best_alpha:.5f}, δ={best_offset}")

    # Return as plain Python list (required by the harness)
    return best_seq.tolist()
