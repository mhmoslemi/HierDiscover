    K = w.shape[0]
    n = a.shape[0]
    N = 1 << ((K + n - 1).bit_length())

    A = np.fft.rfft(a, n=N)

import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based, no loops)
# ----------------------------------------------------------------------


def _normalize_sum_one(a: np.ndarray) -> np.ndarray:
    """Scale the vector to have Σa_i = 1 (preserving non‑negativity)."""
    s = a.sum()
    if s <= 0.0:
        a = np.ones_like(a) / a.size
    else:
        a = a / s
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ∗ a using an FFT of length the next power‑of‑two.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)
    return conv[: 2 * n - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation w ⋆ a (lags 0 … len(a)‑1).
    w has length 2·len(a)‑1, a has length len(a).
    Returns an array of length len(a) where entry i = Σ_j w[i+j]·a[j].
    """
    K = w.shape[0]
    n = a.shape[0]
    N = 1 << ((K + n - 1).bit_length())
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a, n=N)
    corr = np.fft.irfft(W * np.conj(A), n=N)
    return corr[:n]


# ----------------------------------------------------------------------
# Phase 1 – systematic exploration of parametric families
# ----------------------------------------------------------------------


def _explore_parametric_families(n: int, deadline: float, rng: np.random.Generator):
    """
    Scan several analytic families (exponential, power‑law, beta‑like,
    mixed) and return the best candidate (as a NumPy array) together
    with its cheap‑approximation score 2·n·max(a∗a).
    """
    best_seq = None
    best_score = float("inf")

    def _approx_score(seq):
        conv = _conv_fft(seq)
        return 2.0 * n * conv.max()

    # 1. Pure exponential a_i = r^i  (r = exp(-c / n))
    for c in np.linspace(0.2, 5.0, 80):
        r = np.exp(-c / n)
        seq = r ** np.arange(n, dtype=np.float64)
        seq = _normalize_sum_one(seq)
        sc = _approx_score(seq)
        if sc < best_score:
            best_seq, best_score = seq, sc

    # 2. Power‑law a_i = 1/(i+1)^p
    for p in np.linspace(0.1, 3.0, 60):
        i = np.arange(1, n + 1, dtype=np.float64)
        seq = 1.0 / (i ** p)
        seq = _normalize_sum_one(seq)
        sc = _approx_score(seq)
        if sc < best_score:
            best_seq, best_score = seq, sc

    # 3. Beta‑like a_i = (i+1)^α·(n‑i)^β
    alphas = np.linspace(0.0, 2.0, 9)
    betas = np.linspace(0.0, 2.0, 9)
    for α in alphas:
        for β in betas:
            i = np.arange(1, n + 1, dtype=np.float64)
            seq = (i ** α) * ((n - i + 1) ** β)
            seq = _normalize_sum_one(seq)
            sc = _approx_score(seq)
            if sc < best_score:
                best_seq, best_score = seq, sc

    # 4. Mixed exponential × power‑law  a_i = r^i / (i+1)^p
    for c in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0):
        r = np.exp(-c / n)
        for p in (0.5, 1.0, 1.5):
            i = np.arange(1, n + 1, dtype=np.float64)
            seq = (r ** (i - 1)) / (i ** p)
            seq = _normalize_sum_one(seq)
            sc = _approx_score(seq)
            if sc < best_score:
                best_seq, best_score = seq, sc

    # 5. Random draws inside a modest parameter box (adds diversity)
    for _ in range(200):
        if time.time() >= deadline:
            break
        family = rng.integers(0, 3)
        if family == 0:  # exponential
            c = rng.uniform(0.1, 5.0)
            r = np.exp(-c / n)
            seq = r ** np.arange(n, dtype=np.float64)
        elif family == 1:  # power law
            p = rng.uniform(0.05, 4.0)
            i = np.arange(1, n + 1, dtype=np.float64)
            seq = 1.0 / (i ** p)
        else:  # beta‑like
            α = rng.uniform(0.0, 2.5)
            β = rng.uniform(0.0, 2.5)
            i = np.arange(1, n + 1, dtype=np.float64)
            seq = (i ** α) * ((n - i + 1) ** β)
        seq = _normalize_sum_one(seq)
        sc = _approx_score(seq)
        if sc < best_score:
            best_seq, best_score = seq, sc

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max (log‑sum‑exp) mirror descent on the full vector
# ----------------------------------------------------------------------


def _smooth_max_optimize(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         target_score: float = 1.503) -> np.ndarray:
    """
    Mirror‑descent on the smoothed L∞ surrogate.
    The sequence stays on the probability simplex (non‑negative, sum = 1).
    Returns the best exact sequence observed before the deadline.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    start_time = time.time()
    it = 0

    while time.time() < deadline:
        # periodic exact evaluation
        if it % 7 == 0:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score, best_a = exact, a.copy()
                print(f"[{time.time() - start_time:.1f}s] New best (smooth) score={best_score:.6f}")
                if best_score <= target_score:
                    break

        # ----- smoothed max via log‑sum‑exp -----
        conv = _conv_fft(a)                     # length 2n‑1
        max_c = conv.max()
        eps = max(1e-8, max_c * 0.02)            # 2 % of the current peak
        offs = conv - max_c                     # ≤ 0
        w = np.exp(offs / eps)
        w /= w.sum()                            # weights sum to 1

        # gradient of the surrogate: 2·(w ⋆ a)
        grad = 2.0 * _corr_fft(w, a)

        # ----- exponentiated‑gradient (mirror) step -----
        elapsed = time.time() - start_time
        total = deadline - start_time
        eta = 0.5 * (1.0 - elapsed / total) + 0.001 * (elapsed / total)

        g_max = np.max(np.abs(grad))
        if g_max == 0.0:
            break
        grad_norm = grad / g_max
        factor = np.exp(-eta * grad_norm)
        a = a * factor
        s = a.sum()
        a = a / s if s > 0 else np.ones_like(a) / n

        # occasional random jitter to escape plateaus
        if rng.random() < 0.015:
            noise = rng.normal(scale=0.01, size=n)
            a = a * (1.0 + noise)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        it += 1

    return best_a.tolist()


# ----------------------------------------------------------------------
# Public entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient list that minimises the
    ``evaluate_sequence`` bound.

    The search consists of a fast parametric sweep (Phase 1) followed by
    a smooth‑max mirror‑descent refinement (Phase 2).  The routine obeys
    the given time budget and returns the best sequence encountered.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalize_sum_one(np.asarray(hs1, dtype=float))
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score, best_seq = sc, seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from height_sequence_1 → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exhaustive scan of analytic families (different lengths)
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]   # longest gives most flexibility
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.1f}s] Exploring length {n}")
        cand_seq, _ = _explore_parametric_families(n, deadline, rng)
        if cand_seq is not None:
            exact = evaluate_sequence(cand_seq.tolist())
            if exact < best_score:
                best_score, best_seq = exact, cand_seq.tolist()
                print(f"[{time.time() - start_time:.1f}s] New best after Phase 1 (n={n}) score={best_score:.6f}")
            if best_score <= 1.50:   # early stop if already very good
                break

    # ------------------------------------------------------------------
    # Guard: fall back to a random simplex point if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8192
        a_rand = rng.random(n)
        best_seq = _normalize_sum_one(a_rand).tolist()
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max mirror descent refinement on the best length
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)
    cur = _normalize_sum_one(cur)          # guarantee Σ=1 for the surrogate
    refined = _smooth_max_optimize(cur, deadline, rng)

    # ------------------------------------------------------------------
    # Final verification – keep the absolute best of all candidates
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(refined)
    if final_score < best_score:
        best_score, best_seq = final_score, refined

    print(f"[{time.time() - start_time:.1f}s] Finished – best score={best_score:.6f}")

    # Return a plain list of non‑negative floats as required by the API.
    return [float(x) for x in best_seq]
