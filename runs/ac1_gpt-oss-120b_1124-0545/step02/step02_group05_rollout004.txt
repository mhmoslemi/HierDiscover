def solve_lp(a: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    n = a.size
    b = cp.Variable(n, nonneg=True)
    conv = cp.conv(b, a)  # shape (2n-1)
    constraints = [conv <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=False, max_iters=5000, verbose=False)
    except cp.error.SolverError:
        # fallback to SCS
        prob.solve(solver=cp.SCS, warm_start=False, max_iters=2000, verbose=False)
    if b.value is None:
        return None
    b_opt = np.array(b.value).flatten()
    s = b_opt.sum()
    if s == 0:
        return None
    b_scaled = b_opt * (target_sum / s)
    return b_scaled

"""
Iterative LP+Gradient descent optimizer for minimizing max self‑convolution.
Uses a linear program to compute a partner vector g that satisfies f*g <= M,
mixes it into the current f, and then refines the result with a smooth
max‑gradient (log‑sum‑exp) Adam optimizer that works with FFTs for speed.
Both phases respect a time budget and keep the vector on the probability simplex
(ℓ₁ = √(2n)). The best sequence seen is returned.
"""

import time
import math
import random
import numpy as np
import cvxpy as cp

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {w ≥ 0, Σ w = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: equal distribution
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _solve_lp(a: np.ndarray, M: float, target_sum: float):
    """Solve   max Σ b_j   s.t. b ≥ 0 , a * b ≤ M .   Return b scaled to target_sum."""
    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)
    # Linear convolution a * b (a constant, b variable)
    conv_expr = cp.conv(b, a)  # shape (2n‑1)
    constraints = [conv_expr <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=False, max_iters=3000, verbose=False)
    except (cp.error.SolverError, Exception):
        prob.solve(solver=cp.SCS, warm_start=False, max_iters=2000, verbose=False)
    if b.value is None:
        return None
    b_opt = np.array(b.value).flatten()
    total = b_opt.sum()
    if total == 0:
        return None
    return b_opt * (target_sum / total)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function height vector that minimises
    `evaluate_sequence`.  The routine first performs a few linear‑programming
    updates (the “g‑step” from the paper) and then refines the result with a
    projected Adam optimiser on a log‑sum‑exp smooth approximation of the max
    self‑convolution.  The best sequence found within the time budget is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Starting point
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        n = a0.size
    else:
        n = 2500
        a0 = np.full(n, 1.0, dtype=float)

    target_sum = math.sqrt(2.0 * n)

    # Normalise to the required ℓ₁‑norm
    a0 = a0 / a0.sum() * target_sum

    # Create a couple of simple alternatives and keep the best as the initial iterate
    uniform = np.full(n, target_sum / n)
    rand = np.random.exponential(scale=1.0, size=n)
    rand = rand / rand.sum() * target_sum

    starters = [a0, uniform, rand]
    best_seq = None
    best_score = np.inf
    for cand in starters:
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    a = best_seq.copy()

    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # keep 5 s safety margin

    # --------------------------------------------------------------
    # Phase 1 – Linear‑programming updates (≈ 30 % of the budget)
    lp_deadline = start_time + (budget_s * 0.3) - 2.0
    iter_lp = 0
    max_lp_iters = 30          # generous upper bound – will stop earlier by time

    # Mixing schedule: start relatively aggressive, decay to a small step
    t0, t_min = 0.3, 0.02

    while time.time() < lp_deadline and iter_lp < max_lp_iters:
        # current max convolution
        fa = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fa * fa, n=2 * n - 1)
        M = float(conv.max())

        b_scaled = _solve_lp(a, M, target_sum)
        if b_scaled is None:
            # LP failed – abort the LP phase
            break

        # mixing parameter follows a simple exponential decay
        t = t0 * ((t_min / t0) ** (iter_lp / max_lp_iters))
        a_new = (1.0 - t) * a + t * b_scaled

            best_seq = a_new.copy()

        iter_lp += 1

    # Phase 2 – Smoothed Adam gradient descent (remainder of the budget)
    # Hyper‑parameters
    eps_start, eps_end = 0.2, 0.004      # smoothing
    lr_start = target_sum * 0.001
    lr_end   = target_sum * 0.00002

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    t_iter = 0
    last_improve = 0

    # we will use a simple schedule for the Adam step size as above.

    while time.time() < deadline:
        # fraction of second half of the budget
        elapsed = time.time() - start_time
        # we map the whole elapsed fraction [0,1] to schedule, but with a slower start
        frac = elapsed / (deadline - start_time)
        eps = eps_start * ((eps_end / eps_start) ** frac)
        lr  = lr_start * ((lr_end / lr_start) ** frac)

        fa = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fa * fa, n=2 * n - 1)
        cur_score = float(conv.max())
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = t_iter
            print(f"[GD {t_iter:06d}] new best max(conv) = {best_score:.6f} eps={eps:.4f}")

        conv_shift = conv - conv.max()
        # stable exponentiation
        with np.errstate(over='ignore'):
            w_raw = np.exp(conv_shift / eps)
        w_sum = w_raw.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w_raw / w_sum

        # Compute sigma = Σ_{i=j}^{j+n‑1} w_i   (sliding window of length n)
        cumsum = np.cumsum(w)
        sigma = cumsum[n-1:2*n-1] - np.concatenate(([0.0], cumsum[:n-1]))

        # First term = (w * reversed a)  (convolution)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=2 * n - 1)
        fa_rev = np.fft.rfft(a_rev, n=2 * n - 1)
        conv_wa = np.fft.irfft(fw * fa_rev, n=2 * n - 1)
        first = conv_wa[n-1:n-1+n]   # length n

        # full gradient of the smooth objective
        grad = first + a * sigma

        # Adam update (minimise)
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = a - step

        a_new = _simplex_projection(a_new, target_sum)

            fa_pert = np.fft.rfft(a_pert, n=2 * n - 1)
            conv_pert = np.fft.irfft(fa_pert * fa_pert, n=2 * n - 1)
            score_pert = float(conv_pert.max())
            if score_pert < best_score:
                best_score = score_pert
                best_seq = a_pert.copy()
                print(f"[GD {t_iter}] new best after perturb = {best_score:.6f}")

            if score_pert < cur_score:
                a = a_pert
                continue

        # restart if we have been stagnant for a while
        if t_iter - last_improve > 4000:
            # re‑initialise from a blend of best and fresh random noise
            noise = np.random.exponential(scale=1.0, size=n)
            noise = noise / noise.sum() * target_sum
            a = 0.5 * best_seq + 0.5 * noise
            a = _simplex_projection(a, target_sum)
            last_improve = t_iter
            print(f"[GD {t_iter}] restart after stagnation")

        a = a_new

    # --------------------------------------------------------------
    # Return the best sequence seen.
    final_val = evaluate_sequence(best_seq.tolist())
    print(f"Final best max(conv) = {final_val:.6f}")
    return best_seq.tolist()

"""
Iterative LP + smoothed Adam optimiser for the max‑self‑convolution problem.

*Phase 1* – a few linear‑programming updates (the “g‑step” from the paper):
  maximise Σ b_j subject to b≥0 and (a * b)_i ≤ M, where M is the current
  maximum of a * a.  The solution is rescaled to the required ℓ₁‑norm and
  mixed into the current vector.

*Phase 2* – a projected Adam descent on a log‑sum‑exp approximation of
  max(a * a).  Convolutions are performed with FFTs for speed, the iterate is
  projected onto the simplex {x≥0, Σx = √(2n)} after each step, and occasional
  random perturbations are used to escape plateaus.

Both phases respect a 1000 s budget and keep track of the best sequence seen.
The function returns that sequence as a plain Python list.
"""

import time
import math
import random
import numpy as np
import cvxpy as cp


def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex { w ≥ 0 , Σ w = s }."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):         # fallback – uniform distribution
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _solve_lp(a: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve   max Σ b_j   s.t. b≥0 , a * b ≤ M .
    Return a feasible b scaled so that Σ b = target_sum.
    """
    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)

    # linear convolution a * b  (a is constant, b is the variable)
    conv = cp.conv(b, a)                     # length 2n‑1
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [conv <= M])

    try:
        prob.solve(solver=cp.ECOS, max_iters=3000, warm_start=False, verbose=False)
    except Exception:                        # fall back to a slower open‑source solver
        prob.solve(solver=cp.SCS, max_iters=2000, warm_start=False, verbose=False)

    if b.value is None:
        return None

    b_opt = np.array(b.value).flatten()
    total = b_opt.sum()
    if total == 0:
        return None
    return b_opt * (target_sum / total)
