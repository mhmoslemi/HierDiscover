"""
Non‑negative step‑function optimisation. 
The algorithm builds a diverse set of parametric seeds (geometric, power‑law,
Beta‑distribution, Gaussian, sine window), evaluates them fast, and
runs a constrained Adam optimiser (log‑softmax) on the best few seeds.
A final refinement stage uses the exact max‑gradient with projection onto
the probability simplex. The whole search respects the supplied time
budget and returns the best sequence found.
"""

import time
import numpy as np

# helper functions: next_pow2, conv_fft, softmax
def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# Projection onto the simplex (non‑negative, sum = 1)
def _proj_simplex(v):
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive, return uniform
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # ensure sum exactly 1 (tiny numerical drift)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0 / n)
    return w / w_sum

# Parametric families -------------------------------------------------
def _uniform_seq(n):
    return np.full(n, 1.0 / n)

def _sine_window(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s

def _gaussian_seq(n, sigma):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    s = a.sum()
    return a / s

def _beta_seq(n, alpha, beta):
    # Avoid exact 0/1 which would raise Inf for alpha<1 or beta<1
    eps = 1e-12
    i = np.arange(n, dtype=np.float64) + 0.5  # offset to stay away from 0,1
    t = i / (n + 0.0)  # roughly (0,1)
    # Clip to (eps, 1-eps) for safety
    t = np.clip(t, eps, 1 - eps)
    a = (t ** (alpha - 1)) * ((1 - t) ** (beta - 1))
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        # fallback to uniform
        return _uniform_seq(n)
    return a / s

def _geom_seq(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform_seq(n)

def _power_seq(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform_seq(n)

def _geom_power_seq(n, r, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform_seq(n)

def _sym_geom_seq(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform_seq(n)

# Fast objective (sum=1) -------------------------------------------------
def _objective_fast(a, n):
    # a is assumed to sum to 1 (or close)
    conv = _conv_fft(a, a)
    cur_max = np.max(conv)
    return float(2.0 * n * cur_max)

# Adam optimiser -------------------------------------------------------
def _adam_optimize(a0, n, deadline,
                   max_iters=15000,
                   lr0=0.6,
                   beta1=0.9,
                   beta2=0.999,
                   eps=1e-12,
                   beta_start=5.0,
                   beta_mul=1.02):
    """
    Run Adam on the log‑softmax parametrisation starting from a0 (already
    on the simplex). Returns the best point seen.
    """
    # initialise logits
    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best = a0.copy()
    best_val = _objective_fast(best, n)
    t = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
        # smoothing weight
        beta = beta_start * (beta_mul ** it)
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        # central n entries
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # softmax gradient
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best, best_val

def _subgradient_refine(a0, n, deadline,
                       max_iters=5000,
                       eta0=0.2,
                       decay=0.99,
                       tol=1e-12):
    """
    Projected sub‑gradient descent on the TRUE max (no smoothing).
    """
    a = a0.copy()
    best = a.copy()
    best_val = _objective_fast(best, n)
    eta = eta0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        idxs = np.where(np.abs(conv - max_val) <= 1e-12)[0]
        # Build sub‑gradient: average of gradients from all active indices
        grad = np.zeros_like(a)
        for idx in idxs:
            # grad_i = 2 * a_{idx - i} where index in range
            low = max(0, idx - (n - 1))
            high = min(idx, n - 1)
            # i from low..high => j = idx-i
            # vectorized:
            i_range = np.arange(low, high + 1)
            j_range = idx - i_range
            grad[i_range] += 2.0 * a[j_range]
        grad /= len(idxs)  # average
        # gradient step
        a_next = a - eta * grad
        # project onto simplex
        a_next = _proj_simplex(a_next)

        cur_val = _objective_fast(a_next, n)
        if cur_val < best_val - tol:
            best_val = cur_val
            best = a_next.copy()
        # simple acceptance rule (always accept projected point)
        a = a_next
        eta *= decay  # slowly shrink step
    return best, best_val

"""
Non‑negative step‑function optimisation.

The algorithm builds a wide collection of parametric “seed” sequences
(geometric, power‑law, beta‑distribution, Gaussian, sine‑window, etc.),
evaluates them quickly, and runs a constrained Adam optimiser on the
most promising seeds.  A final refinement stage uses the exact max‑gradient
with a projection onto the probability simplex.  The whole search respects
the supplied time budget (default 1000 s) and returns the best sequence
found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – always sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    # Find the last index where the constraint is active
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive -> uniform point
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Re‑normalise to avoid tiny drift
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sine_pow_seq(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 1) / (n + 1)) ** p
    s = a.sum()
    if s == 0.0 or not np.isfinite(s):
        return _uniform_seq(n)
    return a / s


def _gaussian_seq(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        return _uniform_seq(n)
    return a / s


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1} (shifted by 0.5 to avoid 0/1)."""
    eps = 1e-12
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / (n + 0.0)
    t = np.clip(t, eps, 1.0 - eps)
    a = (t ** (alpha - 1.0)) * ((1.0 - t) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0 or not np.isfinite(s):
        return _uniform_seq(n)
    return a / s


def _geom_seq(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return _uniform_seq(n)
    return a / s


def _power_seq(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0.0:
        return _uniform_seq(n)
    return a / s


def _geom_power_seq(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1.0)) / (idx ** p)
    s = a.sum()
    if s == 0.0:
        return _uniform_seq(n)
    return a / s


def _sym_geom_seq(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return _uniform_seq(n)
    return a / s


# ----------------------------------------------------------------------
# Objective (fast variant – works because vectors sum to 1)
# ----------------------------------------------------------------------


def _objective_fast(a: np.ndarray, n: int) -> float:
    """
    Returns 2 * n * max(conv(a, a)).  The function assumes Σa = 1.
    """
    conv = _conv_fft(a, a)
    cur_max = np.max(conv)
    return float(2.0 * n * cur_max)


# ----------------------------------------------------------------------
# Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5000,
    lr0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
    beta_start: float = 5.0,
    beta_mul: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Constrained Adam on the simplex.  a0 must already be a probability
    vector (non‑negative, sums to one).  Returns the best point found
    together with its fast objective value.
    """
    v = np.log(a0 + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    best_seq = a0.copy()
    best_val = _objective_fast(best_seq, n)
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                       # stays on the simplex
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[adam] n={n} it={it:4d} new best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        beta = beta_start * (beta_mul ** it)
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient wrt a (smooth loss) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]          # central n entries

        # ---- convert to gradient wrt logits v (softmax) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact max‑gradient refinement (projected sub‑gradient descent)
# ----------------------------------------------------------------------


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3000,
    eta0: float = 0.2,
    decay: float = 0.99,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max (no smoothing).
    The step size eta is geometrically decayed.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective_fast(best_seq, n)
    eta = eta0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        # indices attaining the maximum (usually a single index)
        idxs = np.where(np.abs(conv - max_val) <= 1e-12)[0]

        # sub‑gradient: average over all active indices
        grad = np.zeros_like(a)
        for idx in idxs:
            lo = max(0, idx - (n - 1))
            hi = min(idx, n - 1)
            i_range = np.arange(lo, hi + 1)
            j_range = idx - i_range
            grad[i_range] += 2.0 * a[j_range]
        grad /= len(idxs)

        # descent step + projection onto simplex
        a_next = _proj_simplex(a - eta * grad)

        cur_val = _objective_fast(a_next, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_next.copy()
            print(f"[refine] n={n} it={it:4d} new best={cur_val:.6f}")

        a = a_next
        eta *= decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # leave a tiny safety margin

    # Primary lengths – try the larger one first; fall back to a smaller one
    lengths = [8000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        # --------------------------------------------------------------
        # Assemble a diverse pool of seeds
        # --------------------------------------------------------------
        seeds = []

        # basic deterministic families
        seeds.append(_uniform_seq(n))
        seeds.append(_sine_window(n))
        for p in [0.3, 0.5, 0.7, 0.9, 1.1, 1.5]:
            seeds.append(_sine_pow_seq(n, p))

        # Gaussian – a handful of widths
        for sigma in [n / 8.0, n / 6.0, n / 4.0]:
            seeds.append(_gaussian_seq(n, sigma))

        # Beta family – coarse grid in (α,β)
        alphas = np.linspace(0.2, 2.0, 7)
        betas = np.linspace(0.2, 2.0, 7)
        for a_ in alphas:
            for b_ in betas:
                seeds.append(_beta_seq(n, a_, b_))

        # arcsine (α=β=0.5) – historically a strong candidate
        seeds.append(_beta_seq(n, 0.5, 0.5))

        # geometric / power families (similar to the previous baseline)
        c_vals = np.linspace(0.5, 5.0, 7)
        for c in c_vals:
            r = 1.0 - c / n
            if r > 0.0:
                seeds.append(_geom_seq(n, r))

        p_vals = np.linspace(0.0, 2.2, 7)
        for p in p_vals:
            seeds.append(_power_seq(n, p))

        for c in np.linspace(0.5, 4.0, 5):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 5):
                seeds.append(_geom_power_seq(n, r, p))

        for c in np.linspace(0.5, 4.5, 5):
            r = 1.0 - c / n
            if r > 0.0:
                seeds.append(_sym_geom_seq(n, r))

        # a few random softmax seeds
        for _ in range(5):
            v_rand = np.random.randn(n)
            seeds.append(_softmax(v_rand))

        # possibly reuse a previously computed solution
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                a_prev = np.asarray(prev, dtype=np.float64)
                if a_prev.shape == (n,):
                    a_prev = a_prev / a_prev.sum()
                    seeds.append(a_prev)
                else:
                    adj = np.zeros(n, dtype=np.float64)
                    k = min(len(a_prev), n)
                    adj[:k] = a_prev[:k]
                    adj = adj / adj.sum()
                    seeds.append(adj)
            except Exception:
                pass

        # --------------------------------------------------------------
        # Quick scan – keep the best few seeds
        # --------------------------------------------------------------
        seed_vals = [(a, _objective_fast(a, n)) for a in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [a for a, _ in seed_vals[:5]]
        best_local_seq = top_seeds[0].copy()
        best_local_val = _objective_fast(best_local_seq, n)
        print(f"[scan] length={n} best initial value={best_local_val:.6f}")

        # --------------------------------------------------------------
        # Adam optimisation on the top seeds
        # --------------------------------------------------------------
        for idx, s in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline)
            print(f"[adam] length={n} seed={idx} final val={val:.6f}")
            if val < best_local_val:
                best_local_val = val
                best_local_seq = seq.copy()

        # --------------------------------------------------------------
        # Exact max‑gradient refinement
        # --------------------------------------------------------------
        if time.time() < deadline:
            seq_ref, val_ref = _subgradient_refine(best_local_seq, n, deadline)
            print(f"[refine] length={n} refined val={val_ref:.6f}")
            if val_ref < best_local_val:
                best_local_val = val_ref
                best_local_seq = seq_ref.copy()

        # --------------------------------------------------------------
        # Update global best
        # --------------------------------------------------------------
        if best_local_val < global_best_val:
            global_best_val = best_local_val
            global_best_seq = best_local_seq.copy()
            print(f"[global] new best length={n} value={global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform_seq(n)
        print("[fallback] using uniform distribution")

    # Return plain Python list as required by the platform
    return global_best_seq.tolist()
