def _project_simplex(v):

    """Project onto probability simplex (sum=1, non-neg)."""

    n = len(v)

    # sort descending

    u = np.sort(v)[::-1]

    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]

def _gradient_step(seq, step, top_k):
    n = seq.shape[0]
    conv = _conv_fft(seq)
    # top indices
    if top_k >= conv.size:
        top_indices = np.arange(conv.size)
    else:
        top_indices = np.argpartition(-conv, top_k-1)[:top_k]
    grad = np.zeros_like(seq)
    for k in top_indices:
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        i_range = np.arange(lo, hi+1)
        grad[i_range] += seq[k - i_range]
    grad *= 2.0
    grad -= grad.mean()   # zero mean to preserve sum after step
    new_seq = seq - step * grad
    # projection onto simplex
    new_seq = _project_simplex(new_seq)
    return new_seq, conv.max()

def _project_simplex(v):
    # works with any shape
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
A fast gradient‑based search for a non‑negative step function that minimises
``evaluate_sequence``.

The algorithm works in three stages:

1.  **Parametric start** – a power‑law family a_i ∝ (i+1+δ)^{-α} is
    explored on a fairly large length `n`.  A coarse then a fine grid over the
    exponent α (and a small integer offset) are scanned using a cheap
    FFT‑based convolution to obtain an initial sequence.

2.  **Projected gradient descent** – the objective 2·n·max(conv(a,a))
    is approximated by the sub‑gradient of the max‑entry of the self‑convolution.
    In each iteration the `top_k` largest convolution entries are identified,
    their sub‑gradient is accumulated, a step is taken and the vector is
    projected onto the probability simplex {x≥0, Σx=1}.  The step size decays
    slowly; occasional random mass swaps help escape flat regions.

3.  **Random restarts** – the whole procedure is repeated from a few random
    power‑law seeds (different α and δ) for a short budget; the best result
    over all restarts is kept.

All convolutions are performed with FFT (O(n log n)) which makes it possible
to work with lengths up to ≈ 2·10⁴ while staying well inside the 1000 s
overall budget.  The final sequence is returned as a plain Python list – the
harness will call ``evaluate_sequence`` exactly once after the function returns.

The implementation is completely self‑contained, uses only NumPy and the
standard library and respects the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Produce a normalised non‑negative sequence proportional to
    (i+1+offset)^{-α} for i=0,…,n‑1.

    Parameters
    ----------
    n : int
        Length of the sequence.
    alpha : float
        Exponent (α>0).
    offset : float, optional
        Small shift that can be used to reduce the first coefficient.

    Returns
    -------
    np.ndarray
        Length‑`n` array with sum equal to 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Self‑convolution using the FFT (real‑valued).  Returns the full
    length‑(2n‑1) vector.
    """
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values; clip to zero.
    conv[conv < 0.0] = 0.0
    return conv


def _score_fft(seq: np.ndarray) -> float:
    """
    Exact value of the objective used by ``evaluate_sequence`` for a
    normalised sequence (Σseq == 1).  The denominator is therefore 1.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the O(n) algorithm of
    Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho value
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_step(seq: np.ndarray, step: float, top_k: int) -> np.ndarray:
    """
    One projected sub‑gradient step that reduces the largest entries of the
    self‑convolution.

    Parameters
    ----------
    seq : np.ndarray
        Current probability vector (sum = 1, non‑negative).
    step : float
        Step size.
    top_k : int
        Number of largest convolution entries that are used to build the
        sub‑gradient.

    Returns
    -------
    np.ndarray
        Updated probability vector (still on the simplex).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)

    # indices of the `top_k` largest convolution values
    if top_k >= conv.size:
        top_idx = np.arange(conv.size)
    else:
        top_idx = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(seq)
    for k in top_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=int)
        # a_{k-i} contribution for each i
        grad[i_range] += seq[k - i_range]

    # the true sub‑gradient of max(conv) is 2·a_{k‑i}
    grad *= 2.0

    # make the step tangent to the affine hyperplane Σx = 1
    grad -= grad.mean()

    new_seq = seq - step * grad
    # projection onto the simplex guarantees feasibility
    new_seq = _project_simplex(new_seq)
    return new_seq


def _run_gradient(
    seq: np.ndarray,
    time_budget: float,
    *,
    max_iters: int = 10 ** 9,
    top_k: int = 20,
    step0: float = 0.4,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """
    Runs projected gradient descent until either the time budget expires or the
    maximum number of iterations is reached.  Returns the best sequence seen
    and its score.

    Parameters
    ----------
    seq : np.ndarray
        Starting point (already normalised).
    time_budget : float
        Available time in seconds.
    max_iters : int, optional
        Upper bound on iterations (default = very large).
    top_k : int, optional
        Number of top convolution entries used for the sub‑gradient.
    step0 : float, optional
        Initial step size.
    decay : float, optional
        Multiplicative decay applied to the step size each iteration.

    Returns
    -------
    (np.ndarray, float)
        The best sequence found and its corresponding score.
    """
    start = time.time()
    best_seq = seq.copy()
    best_score = _score_fft(seq)

    step = step0
    iteration = 0

    while iteration < max_iters and (time.time() - start) < time_budget:
        iteration += 1

        # gradient step
        seq = _gradient_step(seq, step, top_k)

        # occasional random mass exchange to help escape plateaus
        if iteration % 200 == 0:
            i, j = np.random.choice(seq.size, 2, replace=False)
            eps = min(step * seq[i], seq[i] * 0.3)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

        cur_score = _score_fft(seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        step = max(step * decay, 1e-8)

    return best_seq, best_score
