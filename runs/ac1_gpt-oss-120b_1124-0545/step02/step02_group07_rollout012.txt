"""
Propose a low‑score step‑function sequence.

Algorithm
---------

1. **Systematic scan of parametric families**  
   For several lengths (256 … 8192) we generate a variety of analytic
   shapes (exponential, power‑law, Gaussian, classic window functions,
   sine‑squared, beta‑like, valley, etc.).  For each shape the exact score

        score = 2·n·max(a∗a) / (∑a)²

   is computed with a fast FFT‑based convolution.  The best sequence seen
   so far is stored.

2. **Local optimisation** – a hybrid simulated‑annealing / exponentiated‑
   gradient hill‑climber.  Starting from the best sequence of phase 1 we
   repeatedly apply one of three random moves:

   * transfer a small amount of mass from one index to another,
   * a multiplicative “exponentiated‑gradient’’ step using the active
     convolution index,
   * a random block reversal (helps escaping plateaus).

   After each move we renormalise the total sum (scaling does not affect
   the objective) and keep the move if it improves the score, otherwise
   we accept it with a temperature‑controlled probability.  The current
   best sequence is kept as the global answer.

The whole routine respects a user‑supplied time budget (default 1000 s)
and prints progress information.  The returned list contains only
non‑negative floats, ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # smallest power of two ≥ 2n‑1
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)
    return conv[: 2 * n - 1]                     # keep the linear part


def _gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of L(a)=max_k (a∗a)_k at the active index k.
    Returns vector g with g[i] = 2·a[k‑i] when 0≤k‑i<n, otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _score_fast(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective used by `evaluate_sequence`.
    Returns 2·n·max(conv) / (sum(seq)²).  Values are clipped to the same
    range as `evaluate_sequence`, but all generated sequences stay well
    below the caps, so this is a faithful proxy.
    """
    a = np.asarray(seq, dtype=float)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    n = a.shape[0]
    conv = _conv_fft(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic shape generators (return raw, un‑normalised arrays)
# ----------------------------------------------------------------------


def _gen_constant(n: int) -> np.ndarray:
    return np.ones(n, dtype=float)


def _gen_linear(n: int) -> list[np.ndarray]:
    return [np.arange(1, n + 1, dtype=float),
            np.arange(n, 0, -1, dtype=float)]


def _gen_quadratic(n: int) -> list[np.ndarray]:
    return [np.square(np.arange(1, n + 1, dtype=float)),
            np.square(np.arange(n, 0, -1, dtype=float))]


def _gen_triangular(n: int) -> np.ndarray:
    t = np.arange(1, n + 1, dtype=float)
    return np.minimum(t, n - t + 1)


def _gen_valley(n: int, sigma_factor: float = 0.15) -> np.ndarray:
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    return 1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) / sigma) ** 2)


def _gen_sine_squared(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    return np.sin(np.pi * i / (n + 1)) ** 2


def _gen_sine(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    return np.sin(np.pi * i / (n + 1))


def _gen_exponential(n: int, c: float) -> np.ndarray:
    """a_i = exp(-c·i/n)  (i = 0 … n‑1)."""
    return np.exp(-c * np.arange(n, dtype=float) / n)


def _gen_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    return 1.0 / (i ** p)


def _gen_beta_like(n: int, α: float, β: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    return (i ** α) * ((n - i + 1) ** β)


def _gen_gaussian(n: int, sigma_factor: float) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = sigma_factor * n
    return np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)


def _gen_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))


def _gen_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))


def _gen_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return (0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))


# ----------------------------------------------------------------------
# Phase 1 – exhaustive scan over families
# ----------------------------------------------------------------------


def _scan_families(deadline: float, rng: np.random.Generator) -> tuple[list[float], float]:
    """Returns (best_sequence, best_score) after a systematic scan."""
    best_seq = None
    best_score = float("inf")
    start_time = time.time()

    # lengths to explore – short lengths are cheap, long ones give finer granularity
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.1f}s] Scanning length {n}")

        # ---------- constant ----------
        seq = _gen_constant(n)
        score = _score_fast(seq)
        if score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"  New best (const) score={best_score:.6f}")

        # ---------- linear ----------
        for seq in _gen_linear(n):
            score = _score_fast(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"  New best (linear) score={best_score:.6f}")

        # ---------- quadratic ----------
        for seq in _gen_quadratic(n):
            score = _score_fast(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"  New best (quadratic) score={best_score:.6f}")

        # ---------- triangular ----------
        seq = _gen_triangular(n)
        score = _score_fast(seq)
        if score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"  New best (triangular) score={best_score:.6f}")

        # ---------- window functions ----------
        for gen in (_gen_hann, _gen_hamming, _gen_blackman):
            seq = gen(n)
            score = _score_fast(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"  New best ({gen.__name__}) score={best_score:.6f}")

        # ---------- sine / sine‑squared ----------
        for seq in (_gen_sine(n), _gen_sine_squared(n)):
            score = _score_fast(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"  New best (sine) score={best_score:.6f}")

        # ---------- valleys ----------
        for sigma_fac in (0.08, 0.12, 0.18):
            seq = _gen_valley(n, sigma_fac)
            score = _score_fast(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"  New best (valley σ={sigma_fac}) score={best_score:.6f}")

        # ---------- exponentials (fine grid on c) ----------
        for c in np.linspace(0.5, 3.5, 31):    # step≈0.1
            seq = _gen_exponential(n, c)
            score = _score_fast(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"  New best (exp c={c:.2f}) score={best_score:.6f}")

        # ---------- power‑law ----------
        for p in np.linspace(0.3, 2.5, 23):
            seq = _gen_power_law(n, p)
            score = _score_fast(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"  New best (power p={p:.2f}) score={best_score:.6f}")

        # ---------- Gaussian ----------
        for sigma_fac in (0.08, 0.12, 0.18, 0.25):
            seq = _gen_gaussian(n, sigma_fac)
            score = _score_fast(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"  New best (gauss σ={sigma_fac:.2f}) score={best_score:.6f}")

        # ---------- beta‑like ----------
        for α in (0.5, 1.0, 1.5):
            for β in (0.5, 1.0, 1.5):
                seq = _gen_beta_like(n, α, β)
                score = _score_fast(seq)
                if score < best_score:
                    best_score, best_seq = score, seq.copy()
                    print(f"  New best (beta α={α},β={β}) score={best_score:.6f}")

        # ---------- mixed exponential × power ----------
        for c in (0.8, 1.2, 1.6, 2.0):
            for p in (-0.5, 0.0, 0.5, 1.0):
                exp_part = _gen_exponential(n, c)
                pow_part = np.power(np.arange(1, n + 1, dtype=float), p)
                seq = exp_part * pow_part
                score = _score_fast(seq)
                if score < best_score:
                    best_score, best_seq = score, seq.copy()
                    print(f"  New best (mix c={c:.1f},p={p:.1f}) score={best_score:.6f}")

    return best_seq.tolist() if best_seq is not None else None, best_score


# ----------------------------------------------------------------------
# Phase 2 – hill‑climbing / simulated annealing refinement
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                start_time: float) -> np.ndarray:
    """
    Local search starting from `start_seq`.  Random mass‑transfer,
    block‑reversal, and exponentiated‑gradient steps are mixed.
    Returns the best sequence found before `deadline`.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score_fast(best_a)
    cur_score = best_score

    while time.time() < deadline:
        # --------------------------------------------------------------
        # pick a move type
        # --------------------------------------------------------------
        move = rng.integers(0, 3)

        if move == 0:
            # ----- transfer a small chunk from i to j -----------------
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.05          # up to 5 % of source
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

        elif move == 1:
            # ----- exponentiated‑gradient step on the active convolution -----
            conv = _conv_fft(a)
            k_star = int(conv.argmax())
            g = _gradient_at_max(a, k_star)
            g_max = g.max()
            if g_max == 0.0:
                continue
            # linear decay of η from 0.5 → 0.001 over the remaining time
            elapsed = time.time() - start_time
            total = deadline - start_time
            eta = 0.5 * (1.0 - elapsed / total) + 0.001 * (elapsed / total)
            factor = np.exp(-eta * g / g_max)
            a_new = a * factor

        else:
            # ----- random block reversal (helps escape plateaus) ----------
            start_idx = rng.integers(0, n)
            block_len = rng.integers(1, min(256, n - start_idx) + 1)
            a_new = a.copy()
            a_new[start_idx:start_idx + block_len] = a_new[start_idx:start_idx + block_len][::-1]

        # keep the total sum unchanged (scaling does not affect the score)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new = a_new * (a.sum() / s)

        new_score = _score_fast(a_new)

        # --------------------------------------------------------------
        # acceptance rule (simulated annealing)
        # --------------------------------------------------------------
        if new_score < cur_score:
            a, cur_score = a_new, new_score
        else:
            # temperature linearly decays from 0.05 → 0.0
            t_frac = (deadline - time.time()) / (deadline - start_time)
            temperature = 0.05 * t_frac
            accept_prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < accept_prob:
                a, cur_score = a_new, new_score

        # keep the global best
        if new_score < best_score:
            best_score, best_a = new_score, a_new.copy()
            print(f"[{time.time() - start_time:.1f}s] Hill‑climb new best score={best_score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            print("Target bound reached – leaving hill‑climb.")
            break

    return best_a


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient list that minimises
    `evaluate_sequence`.  The function respects the time budget; after the
    budget expires it returns the best sequence found so far.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1      # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Warm‑start from a known construction if available
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = np.asarray(hs1, dtype=float)
        score = _score_fast(seq)
        if score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from height_sequence_1 → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exhaustive parametric scan
    # ------------------------------------------------------------------
    seq1, score1 = _scan_families(deadline, rng)
    if seq1 is not None and score1 < best_score:
        best_score, best_seq = score1, np.asarray(seq1, dtype=float)
        print(f"[{time.time() - start_time:.1f}s] Phase 1 best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Guard: if nothing was produced, initialise randomly
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8192
        best_seq = rng.random(n)
        best_score = _score_fast(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – local refinement (hill‑climb + simulated annealing)
    # ------------------------------------------------------------------
    cur = best_seq / best_seq.sum()          # normalise to unit sum (any scale works)
    refined = _hill_climb(cur, deadline, rng, start_time)

    # ------------------------------------------------------------------
    # Final verification using the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(refined.tolist())
    if final_score < best_score:
        best_score, best_seq = final_score, refined.copy()

    print(f"[{time.time() - start_time:.1f}s] Finished – best score={best_score:.6f}")

    # Return a plain list of non‑negative floats as required by the API
    return [float(x) for x in best_seq]
