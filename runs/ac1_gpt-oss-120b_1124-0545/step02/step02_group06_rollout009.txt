Projected gradient onto simplex algorithm: There is known O(n) method: sort descending, compute threshold. But we can also use the standard method from Duchi et al. There's a simple implementation.

def _project_simplex(v):
    # ensure sum(v) = 1, v_i >= 0
    # works for any vector.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_onto_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive; return uniform?
        return np.full(n, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step‑function sequence that minimises the
`evaluate_sequence` objective.

The algorithm has three phases:

1. **Parametric search** – random sampling of three simple families
   (power‑law, exponential, beta‑like) and keep the best candidate.
2. **Smooth‑max projected gradient descent** – minimise a smooth
   approximation of `max(conv(a,a))` using an exponentiated‑gradient
   (mirror‑descent) on the probability simplex.  The smoothness parameter
   β is increased gradually, turning the objective into the true
   max‑norm.
3. **Fine‑grained annealing** – occasional random mass transfers with a
   classic simulated‑annealing acceptance rule to escape shallow local
   minima.

All operations are carried out with FFT‑based convolutions, and each
candidate is projected onto the simplex (non‑negative, unit sum) after
every update.  The routine respects the supplied time budget (default
1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex:
    {x | x_i >= 0, sum x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_root_sequence(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1), normalised to unit sum."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _power_law_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """a_i ∝ (i+α+1)^(-p), normalised to unit sum."""
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _exp_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to unit sum."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _beta_sequence(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{p-1} (n-i)^{q-1}, normalised."""
    i = np.arange(1, n + 1, dtype=float)          # 1 … n
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _conv_self(seq: np.ndarray, size: int) -> np.ndarray:
    """
    Full self‑convolution (seq * seq) using FFT.
    Returns an array of length 2*len(seq)-1.
    """
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * seq.size - 1]
    return conv


def _correlate_seq_weight(seq: np.ndarray, weight: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the correlation  g_k = Σ_i seq_i * weight_{i+k}
    (valid mode, length = len(seq)).
    Implemented via FFT using reversed weight.
    """
    n = seq.size
    rev = weight[::-1]
    fa = np.fft.rfft(seq, n=size)
    fw = np.fft.rfft(rev, n=size)
    conv = np.fft.irfft(fa * fw, n=size)
    # conv[t] contains Σ_i seq_i * rev_{t-i}
    # Desired g_k = Σ_i seq_i * weight_{i+k}
    # This equals conv[n-1 : n-1+n][::-1]
    start = n - 1
    end = start + n
    g = conv[start:end][::-1] * 2.0
    return g


def _fast_score(seq: np.ndarray, size_self: int) -> float:
    """Fast version of evaluate_sequence (no clipping, sum=1 assumed)."""
    n = seq.size
    conv = _conv_self(seq, size_self)
    max_b = conv.max()
    return 2.0 * n * max_b  # sum(seq)==1 by construction

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a non‑negative step‑function sequence that minimises the
    provided `evaluate_sequence`.  The function obeys the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    size_self = _next_pow2(2 * n - 1)          # FFT size for self‑conv
    best_seq = init.copy()
    best_score = _fast_score(best_seq, size_self)

    # allocate roughly 20 % of the time budget to this phase

    families = ("power", "exp", "beta")
    trials = 0
    while time.time() < deadline_param:

    step_decay = 0.9998

        if it % 2500 == 0:
