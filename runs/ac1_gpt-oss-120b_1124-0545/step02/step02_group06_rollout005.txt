def random_move(seq, i_max):
    n=seq.size
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    # pick i from active region where seq[i]>0
    active_idx = np.where(seq[lo:hi+1] > 0)[0] + lo
    if active_idx.size == 0: return seq.copy()
    i = np.random.choice(active_idx)
    # choose j from outside support
    # combine two ranges: [0, lo-1] and [hi+1, n-1]
    candidates = []
    if lo > 0:
        candidates.append((0, lo-1))
    if hi < n-1:
        candidates.append((hi+1, n-1))
    # choose a range then random index
    r = np.random.choice(len(candidates))
    start, end = candidates[r]
    j = np.random.randint(start, end+1)
    eps = seq[i] * np.random.uniform(0.001, 0.1)
    eps = min(eps, seq[i])
    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    # renorm:
    total = new_seq.sum()
    if total > 0:
        new_seq /= total
    return new_seq
```

"""
Propose a high‑quality non‑negative step‑function sequence.

The search combines three ideas:

1. **Initialisation** – start from the provided global construction or the
   classic 1/√(i+1) “root” law (normalised to unit sum).

2. **Targeted projected gradient descent** – at each iteration we locate the
   index `i_max` where the self‑convolution `a*a` is maximal.
   The gradient of this active entry w.r.t. the coefficients is
   `∂c_i/∂a_j = a_j + a_{i‑j}` (when `i‑j` is a valid index, otherwise 0).
   A small step opposite to this gradient is taken, the vector is clipped to
   non‑negativity and projected back onto the simplex (unit sum).

3. **Exploratory random mass transfers** – with probability 0.3 a random
   “mass‑move” is performed: a fraction of the mass from a position that
   participates in the active convolution is moved to a far‑away index
   (outside the active support).  This helps escape local minima.

Both move types are combined inside a simulated‑annealing loop: worsening
steps may be accepted with a temperature that cools linearly to a tiny value.
The best sequence encountered (according to the exact `evaluate_sequence`
objective) is kept and returned.

The algorithm runs inside the supplied time budget (default 1000 s) and
logs progress throughout the run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _initial_root_sequence(n: int) -> np.ndarray:
    """Root law a_i ∝ 1/√(i+1) normalised to unit sum."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _power_law_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Generate a normalised sequence a_i ∝ (i + α + 1)^‑p.
    Adding 1 avoids division by zero when alpha = 0.
    """
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _conv_full(seq: np.ndarray, m: int) -> np.ndarray:
    """Full linear convolution of seq with itself using a fixed FFT size m."""
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * seq.size - 1]
    return conv


def _conv_max_and_arg(seq: np.ndarray, m: int) -> tuple[float, int]:
    """
    Return (max(conv(seq,seq)), argmax) using the pre‑allocated FFT size m.
    """
    conv = _conv_full(seq, m)
    arg = int(np.argmax(conv))
    return float(conv[arg]), arg


def _score(seq: np.ndarray, m: int) -> float:
    """
    Fast evaluation of the objective:
        2 * n * max(conv(seq,seq)) / (sum(seq) ** 2)

    The sequence is assumed to be non‑negative and already normalised
    (sum ≈ 1), therefore the denominator is ≈1.
    """
    n = seq.size
    max_b, _ = _conv_max_and_arg(seq, m)
    s = seq.sum()
    if s < 0.01:               # mirrors the safety check in evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=target_sum}."""
    # Implementation of the classic algorithm by Wang & Carreira‑Perpinán
    if target_sum <= 0:
        raise ValueError("target_sum must be positive")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_step(seq: np.ndarray, i_max: int, step: float) -> np.ndarray:
    """
    One projected gradient step that attempts to decrease the active
    convolution entry at `i_max`.

    Gradient for a_j (where j contributes to i_max) is:
        grad_j = a_j + a_{i_max - j}
    (the second term is ignored when i_max - j is out of bounds).
    """
    n = seq.size
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    rev = i_max - idx                     # always in‑range because of lo/hi

    grad = np.zeros_like(seq)
    # add a_j term
    grad[idx] += seq[idx]
    # add a_{i_max - j} term
    grad[idx] += seq[rev]

    # gradient descent step
    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)

    # project back onto the simplex (unit sum)
    total = new_seq.sum()
    if total > 0:
        new_seq /= total
    else:                     # fallback – should never happen
        new_seq = seq.copy()
    return new_seq


def _random_mass_move(seq: np.ndarray, i_max: int) -> np.ndarray:
    """
    Randomly transfer a modest fraction of mass from a coefficient that
    participates in the active convolution to a coefficient far away
    (outside the active support).  This creates diversity and helps escape
    local minima.
    """
    n = seq.size
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)

    # Choose a source position with positive mass inside the active block
    active_pos = np.where(seq[lo : hi + 1] > 0.0)[0]
    if active_pos.size == 0:
        return seq.copy()
    src = lo + np.random.choice(active_pos)

    # Choose a destination outside the active block
    parts = []
    if lo > 0:
        parts.append((0, lo - 1))
    if hi < n - 1:
        parts.append((hi + 1, n - 1))
    if not parts:
        # No outside positions – fallback to a small local shuffle
        dst = (src + np.random.randint(1, n)) % n
    else:
        start, end = parts[np.random.randint(len(parts))]
        dst = np.random.randint(start, end + 1)

    # Transfer a random fraction of the source mass
    eps = seq[src] * np.random.uniform(0.001, 0.05)
    eps = min(eps, seq[src])

    new_seq = seq.copy()
    new_seq[src] -= eps
    new_seq[dst] += eps

    # Re‑normalise to unit sum
    total = new_seq.sum()
    if total > 0:
        new_seq /= total
    else:
        new_seq = seq.copy()
    return new_seq

                best_seq = cand.copy()
