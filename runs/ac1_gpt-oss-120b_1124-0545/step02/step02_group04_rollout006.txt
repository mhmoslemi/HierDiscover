def _peak_pair_move(a, max_idx, target_sum, temperature):
   n=len(a)
   i_range = np.arange(max_idx+1)
   partner = max_idx - i_range
   mask = (partner>=0)&(partner<n)
   i_valid = i_range[mask]
   p_valid = partner[mask]
   # compute product
   prod = a[i_valid] * a[p_valid]
   # find index with max product
   idx_max = np.argmax(prod)
   i_src = i_valid[idx_max]
   j_src = p_valid[idx_max]
   # destination index: random among indices not involved
   excluded = set(i_valid) | set(p_valid)
   all_idx = np.arange(n)
   candidates = np.setdiff1d(all_idx, np.array(list(excluded)), assume_unique=True)
   if candidates.size == 0:
        dst = np.random.randint(n)
   else:
        dst = np.random.choice(candidates)
   max_delta = min(a[i_src], a[j_src])
   delta = max_delta * temperature * 0.5 * np.random.random()
   if delta < 1e-12:
        return a.copy()
   a_new = a.copy()
   a_new[i_src] = max(0.0, a_new[i_src] - delta)
   a_new[j_src] = max(0.0, a_new[j_src] - delta)
   a_new[dst] += 2*delta
   # enforce non-negativity by max(0)
   a_new = np.maximum(a_new, 0.0)
   # project onto simplex
   a_new = _simplex_projection(a_new, target_sum)
   # enforce symmetry maybe after projection elsewhere
   return a_new

"""
propose_candidate – Simulated‑annealing optimizer for the max‑convolution
objective  E(a)=2·n·max(conv(a,a))/sum(a)² .

The routine

* works on the probability simplex  {x≥0, sum(x)=√(2·n)}  (projection in
  O(n log n));
* uses a fast FFT based evaluation of the linear convolution;
* provides several low‑level moves (random mass transfer, swap,
  peak‑reduction, pair‑peak reduction and a neighbourhood‑smoothing move);
* gradually cools the temperature and accepts uphill moves with the usual
  Metropolis probability;
* starts from a small set of hand‑crafted shapes (uniform, V‑shape,
  quadratic, exponential, …) and keeps the best one as the seed;
* runs for at most ``budget_s`` seconds and returns the best sequence found.

The algorithm is deliberately written without any file or network I/O,
uses only NumPy and the already‑imported ``evaluate_sequence`` for final
validation, and stays well within the 2‑CPU limit.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project ``v`` onto the simplex {x ≥ 0, sum(x)=s}.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length ``nfft``."""
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)
    # The linear convolution lives in the first 2·len(a)-1 entries.
    return conv[: 2 * a.size - 1]


def _max_conv_and_index(a: np.ndarray, nfft: int):
    """Return (max_value, argmax, full convolution) for ``a*a``."""
    conv = _conv_fft(a, nfft)
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return max_val, max_idx, conv


# ----------------------------------------------------------------------
def _random_transfer_move(a: np.ndarray, target_sum: float, temperature: float) -> np.ndarray:
    """Transfer a small random amount of mass from one index to another."""
    n = a.size
    i, j = np.random.randint(n, size=2)
    if i == j:
        return a
    max_delta = min(a[i], target_sum * 0.04)
    delta = max_delta * temperature * random.random()
    if delta < 1e-12:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return _simplex_projection(a_new, target_sum)


def _swap_move(a: np.ndarray) -> np.ndarray:
    """Swap two entries – this changes only the multiset of values."""
    n = a.size
    i, j = np.random.randint(n, size=2)
    if i == j:
        return a
    a_new = a.copy()
    a_new[i], a_new[j] = a_new[j], a_new[i]
    return a_new


def _peak_reduction_move(a: np.ndarray, max_idx: int,
                         target_sum: float, temperature: float) -> np.ndarray:
    """
    Move a small amount of mass from a coordinate that heavily contributes
    to the current maximal convolution entry to a coordinate that contributes
    little (or not at all). Followed by simplex projection.
    """
    n = a.size
    idx = np.arange(n)
    partner = max_idx - idx
    valid = (partner >= 0) & (partner < n)

    # contribution of each coordinate to the maximal entry
    contrib = np.zeros_like(a)
    contrib[valid] = a[idx[valid]] * a[partner[valid]]

    # source – one of the top contributors
    top_k = 10
    src_candidates = np.argpartition(-contrib, top_k)[:top_k]
    src = src_candidates[np.argmax(contrib[src_candidates])]

    # destination – preferably a coordinate that does NOT contribute
    noncontrib = np.where(~valid)[0]
    if noncontrib.size > 0:
        dst = int(np.random.choice(noncontrib))
    else:
        low_k = 10
        dst_candidates = np.argpartition(contrib, low_k)[:low_k]
        dst = int(np.random.choice(dst_candidates))

    max_delta = min(a[src], target_sum * 0.02)
    delta = max_delta * temperature * random.random()
    if delta < 1e-12:
        return a
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    return _simplex_projection(a_new, target_sum)


def _peak_pair_move(a: np.ndarray, max_idx: int,
                    target_sum: float, temperature: float) -> np.ndarray:
    """
    Reduce the dominant product term that defines the current maximum.
    Two indices that form the largest product a[i]·a[j] for i+j=max_idx are
    both decreased and the mass is moved to a far‑away index.
    """
    n = a.size
    i_range = np.arange(max_idx + 1)
    partner = max_idx - i_range
    mask = (partner >= 0) & (partner < n)
    i_valid = i_range[mask]
    j_valid = partner[mask]

    # product of each contributing pair
    prod = a[i_valid] * a[j_valid]
    if prod.size == 0:
        return a
    idx_max = np.argmax(prod)
    i_src = i_valid[idx_max]
    j_src = j_valid[idx_max]

    # pick a destination that is not involved in the max‑sum
    occupied = np.unique(np.concatenate((i_valid, j_valid)))
    cand = np.setdiff1d(np.arange(n), occupied, assume_unique=True)
    if cand.size == 0:
        dst = np.random.randint(n)
    else:
        dst = int(np.random.choice(cand))

    max_delta = min(a[i_src], a[j_src])
    delta = max_delta * temperature * 0.5 * random.random()
    if delta < 1e-12:
        return a

    a_new = a.copy()
    a_new[i_src] = max(0.0, a_new[i_src] - delta)
    a_new[j_src] = max(0.0, a_new[j_src] - delta)
    a_new[dst] += 2.0 * delta
    a_new = np.maximum(a_new, 0.0)
    return _simplex_projection(a_new, target_sum)


def _smooth_move(a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Replace a short random segment by its average value – a cheap way to
    flatten local peaks while preserving the total mass.
    """
    n = a.size
    seg_len = random.randint(3, 9)          # length of the segment
    start = random.randint(0, n - seg_len)
    segment = a[start:start + seg_len]
    avg = float(segment.mean())
    a_new = a.copy()
    a_new[start:start + seg_len] = avg
    return _simplex_projection(a_new, target_sum)


# ----------------------------------------------------------------------
def _initial_candidates(n: int, target_sum: float):
    """
    Produce several deterministic starting vectors (uniform, V‑shape,
    quadratic, exponential, linear ramps).  Each candidate is projected onto
    the simplex of sum ``target_sum``.
    """
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    cand = []

    # 1) Uniform
    cand.append(np.ones(n))

    # 2) V‑shape (low in the centre, high at the ends)
    vshape = np.abs(i - mid) + 0.5
    cand.append(vshape)

    # 3) Quadratic (parabolic, also low in centre)
    quad = (i - mid) ** 2 + 0.5
    cand.append(quad)

    # 4) Exponential decay from the left side
    exp = np.exp(-0.001 * i)
    cand.append(exp)

    # 5) Linear decreasing (left‑to‑right ramp)
    lin_desc = np.linspace(1.0, 0.0, n)
    cand.append(lin_desc)

    # 6) Linear increasing (right‑to‑left ramp)
    lin_asc = np.linspace(0.0, 1.0, n)
    cand.append(lin_asc)

    # 7) If a previous construction exists, use it (padded/truncated).
    if "height_sequence_1" in globals():
        old = np.array(height_sequence_1, dtype=np.float64)
        if old.size < n:
            # pad with zeros at the end
            old = np.pad(old, (0, n - old.size), mode='constant')
        elif old.size > n:
            old = old[:n]
        cand.append(old)

    # project everything onto the simplex
    projected = [_simplex_projection(c, target_sum) for c in cand]
    return projected


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    # ------------------------------------------------------------------
    # RNG seeding
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size – we deliberately use a power‑of‑two length for FFT
    # efficiency.  If a previous construction is present we honour its length,
    # otherwise we pick a moderately large default.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 4096                      # a power of two, good compromise

    target_sum = math.sqrt(2.0 * n)

    # FFT length – next power of two ≥ 2·n‑1
    nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Generate a handful of deterministic seeds and keep the best one.
    candidates = _initial_candidates(n, target_sum)
    best_seq = None
    best_score = float('inf')
    for cand in candidates:
        max_val, _, _ = _max_conv_and_index(cand, nfft)
        # with the simplex scaling, the objective equals the maximal entry
        if max_val < best_score:
            best_score = max_val
            best_seq = cand.copy()
    # safety fallback
    if best_seq is None:
        best_seq = np.random.exponential(scale=1.0, size=n)
        best_seq = _simplex_projection(best_seq, target_sum)

    # ------------------------------------------------------------------
    # Simulated‑annealing state
    a = best_seq.copy()
    cur_score, cur_max_idx, _ = _max_conv_and_index(a, nfft)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin
    iteration = 0
    last_improve_iter = 0

    # temperature schedule (exponential decay)
    T_start = 1.0
    T_end = 1e-5

    # Relative frequencies of the five move types
    move_weights = np.array([0.25, 0.10, 0.25, 0.25, 0.15])
    move_weights = move_weights / move_weights.sum()
    cum_weights = np.cumsum(move_weights)

    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temperature = T_start * ((T_end / T_start) ** frac)

        # --------------------------------------------------------------
        # Decide which elementary move to apply
        r = random.random()
        if r < cum_weights[0]:
            a_prop = _random_transfer_move(a, target_sum, temperature)
        elif r < cum_weights[1]:
            a_prop = _swap_move(a)
        elif r < cum_weights[2]:
            a_prop = _peak_reduction_move(a, cur_max_idx, target_sum, temperature)
        elif r < cum_weights[3]:
            a_prop = _peak_pair_move(a, cur_max_idx, target_sum, temperature)
        else:
            a_prop = _smooth_move(a, target_sum)

        # --------------------------------------------------------------
        # Evaluate new candidate (max entry of the self‑convolution)
        new_score, new_max_idx, _ = _max_conv_and_index(a_prop, nfft)

        # --------------------------------------------------------------
        # Acceptance rule (Metropolis)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (temperature + 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            cur_max_idx = new_max_idx
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                last_improve_iter = iteration
                print(f"[{iteration:7d}]  New best max‑conv = {best_score:.6f} (time {elapsed:.1f}s)")

        # --------------------------------------------------------------
        # Periodic escape: add a tiny Gaussian jitter to the current best
        if iteration % 40000 == 0 and iteration > 0:
            a = best_seq + np.random.normal(scale=target_sum * 0.0008, size=n)
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a, target_sum)
            cur_score, cur_max_idx, _ = _max_conv_and_index(a, nfft)

        # --------------------------------------------------------------
        # If nothing improved for a long stretch, restart from the best with
        # a moderate perturbation – helps to escape deep local minima.
        if iteration - last_improve_iter > 250000:
            a = best_seq + np.random.normal(scale=target_sum * 0.0015, size=n)
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a, target_sum)
            cur_score, cur_max_idx, _ = _max_conv_and_index(a, nfft)
            last_improve_iter = iteration

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # (The external ``evaluate_sequence`` will perform the final scaling checks.)
    return best_seq.tolist()
