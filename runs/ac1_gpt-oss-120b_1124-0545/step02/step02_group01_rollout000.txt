def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # Determine lengths
    lengths = [2000, 4000, 8000]  # start with 2000 for quick shape, then larger

    best_seq = None
    best_val = float('inf')
    best_logits = None  # store logits for SA

    prev_best = None

    for n in lengths:
        if time.time() > deadline:
            break

        # Generate seeds
        seeds = []
        # uniform
        seeds.append(np.ones(n)/n)

        # triangular shape
        half = n//2
        left = np.arange(1, half+1)
        if n%2==0:
            right = left[::-1]
        else:
            right = np.arange(half,0,-1)
        tri = np.concatenate([left,right])
        seeds.append(tri/tri.sum())

        # Gaussian with varying sigma
        xs = np.arange(n)
        for sigma_frac in [0.1,0.2,0.3,0.4]:
            sigma = sigma_frac * n
            g = np.exp(-0.5*((xs - (n-1)/2)/sigma)**2)
            seeds.append(g/g.sum())

        # Hann window
        hann = 0.5 * (1 - np.cos(2*np.pi*xs/(n-1)))
        seeds.append(hann/hann.sum())

        # Blackman window
        black = 0.42 - 0.5*np.cos(2*np.pi*xs/(n-1)) + 0.08*np.cos(4*np.pi*xs/(n-1))
        seeds.append(black/black.sum())

        # random Dirichlet samples
        for _ in range(5):
            a = np.random.rand(n)
            a = a/ a.sum()
            seeds.append(a)

        # geometric families
        for c in np.linspace(0.5,5.0,8):
            r = 1 - c / n
            if r>0:
                geom = r**np.arange(n)
                seeds.append(geom/geom.sum())

        # symmetric geometric
        for c in np.linspace(0.5,4.5,6):
            r = 1 - c/n
            if r>0:
                a = r**np.arange(n)
                a = a + r**np.arange(n-1,-1,-1)
                seeds.append(a/a.sum())

        # Transfer previous best via upsample

            up = _upsample(prev_best, n)
            seeds.append(up)

            val = _compute_val(a)

                best_val = val; best_seq = a.copy()

        # Determine per seed max iters based on remaining time.
        # We'll allocate roughly (deadline - now)/ (len(seeds)*2) maybe.
        remaining = deadline - time.time()

"""
Multi‑stage optimisation for the step‑function bound.

The algorithm proceeds in three stages:

1. **Seed generation** – a handful of diverse, analytically‑motivated
   probability vectors (uniform, triangular, Hann, Blackman, Gaussian,
   geometric, symmetric geometric, random Dirichlet).  All are normalised
   to sum = 1.

2. **Adam optimisation on the simplex** – each seed is refined by
   gradient descent on the smoothed maximum of the autocorrelation
   (`a*a`).  The parameters are the unconstrained logits `v`
   (`a = softmax(v)`).  A log‑sum‑exp smoothing parameter `β` is
   increased geometrically, and an Adam update keeps the iterates on
   the probability simplex.  The routine is fully time‑aware.

3. **Simulated annealing fine‑tuning** – starting from the best sequence
   found by Adam, we perform a temperature‑controlled random walk on the
   logit space, accepting moves that improve the objective or with a
   Metropolis probability.  This often pushes the bound a little farther.

All convolution work is performed with an FFT‑based routine, so even for
lengths up to 8000 the inner loop is fast.  The final sequence is returned
as a plain Python list of floats; it is non‑negative and normalised (the
evaluation function rescales internally, so any positive scaling works).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _compute_val(a: np.ndarray) -> float:
    """Internal version of evaluate_sequence for a normalised (sum=1) vector."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a longer length, keeping the sum = 1."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new /= seq_new.sum()
    return seq_new


# ----------------------------------------------------------------------
# Adam optimiser for a single length
# ----------------------------------------------------------------------


def _adam_optimize_one_seed(
    v0: np.ndarray,
    n: int,
    deadline: float,
    best_seq_ref,
    best_val_ref,
) -> tuple[np.ndarray, float]:
    """
    Run Adam on a single seed (logits `v0`).  The routine updates
    ``best_seq_ref`` and ``best_val_ref`` in‑place when a new global best
    is discovered.  Returns the final (logit, value) pair for the seed.
    """
    v = v0.copy()
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    t = 0

    # hyper‑parameters
    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    beta_start = 0.5          # smoothing start
    beta_mul = 1.015          # smoothing growth per iteration
    max_iters = 15000

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)

        # objective for this iterate
        cur_val = _compute_val(a)
        if cur_val < best_val_ref[0]:
            best_val_ref[0] = cur_val
            best_seq_ref[0] = a.copy()
            print(f"[Adam] len={n:5d} it={it:4d} new global best={cur_val:.6f}")

        # smoothed‑max weights
        beta = beta_start * (beta_mul ** it)
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]        # central n entries

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    # final value for this seed (not needed by caller but handy)
    final_a = _softmax(v)
    final_val = _compute_val(final_a)
    return v, final_val


def _optimise_length(
    n: int,
    seed_vectors: list[np.ndarray],
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation from a collection of seeds for a given length.
    Returns the best sequence found (as a NumPy array) and its objective value.
    """
    best_seq = [None]      # mutable container for in‑place updates
    best_val = [float("inf")]

    # Try each seed
    for idx, a0 in enumerate(seed_vectors):
        if time.time() > deadline:
            break
        # initialise logits from the seed
        v0 = np.log(a0 + 1e-30)
        _adam_optimize_one_seed(v0, n, deadline, best_seq, best_val)

    if best_seq[0] is None:
        # fallback – uniform distribution
        best_seq[0] = np.ones(n, dtype=np.float64) / n
        best_val[0] = _compute_val(best_seq[0])

    return best_seq[0], best_val[0]


# ----------------------------------------------------------------------
# Simulated annealing fine‑tuning
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple temperature‑controlled random walk on the logit space.
    Starts from ``a_start`` (already normalised) and returns a possibly
    improved sequence together with its objective.
    """
    n = len(a_start)
    logits = np.log(a_start + 1e-30)
    best_seq = a_start.copy()
    best_val = _compute_val(best_seq)

    T0 = 0.05
    T_min = 1e-6
    total_iters = 20000

    for i in range(total_iters):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T0 * (0.99 ** i), T_min)
        sigma = 0.01 * (T / T0)  # noise amplitude shrinks with temperature

        prop_logits = logits + sigma * np.random.randn(n)
        prop_a = _softmax(prop_logits)
        prop_val = _compute_val(prop_a)

        # Metropolis acceptance
        accept = False
        if prop_val < best_val:
            accept = True
        else:
            # probability to accept worsening move
            prob = np.exp(-(prop_val - best_val) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            logits = prop_logits
            if prop_val < best_val:
                best_val = prop_val
                best_seq = prop_a.copy()
                print(f"[SA] iter={i:5d} new best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects a time budget (default
    1000 s) and returns the best sequence it finds.

    The method combines diverse analytical seeds, Adam optimisation on the
    simplex, up‑sampling across increasing lengths, and a final simulated
    annealing fine‑tune.
    """
    np.random.seed(seed)
    # keep a small safety margin for the caller
    deadline = time.time() + budget_s - 5.0

    # Length schedule – start modest, then increase if time permits.
    lengths = [2000, 4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")
    prev_best = None  # best sequence from the previous length (for upsampling)

    for n in lengths:
        if time.time() > deadline:
            break

        # ------------------------------
        # Construct a modest but diverse set of seeds
        # ------------------------------
        seeds = []

        # 1) Uniform
        seeds.append(np.ones(n, dtype=np.float64) / n)

        # 2) Simple triangular (symmetric)
        half = n // 2
        left = np.arange(1, half + 1, dtype=np.float64)
        if n % 2 == 0:
            right = left[::-1]
        else:
            right = np.arange(half, 0, -1, dtype=np.float64)
        tri = np.concatenate([left, right])
        seeds.append(tri / tri.sum())

        # 3) Hann window
        i = np.arange(n, dtype=np.float64)
        hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        seeds.append(hann / hann.sum())

        # 4) Blackman window
        black = (
            0.42
            - 0.5 * np.cos(2 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4 * np.pi * i / (n - 1))
        )
        seeds.append(black / black.sum())

        # 5) Gaussian families (different widths)
        xs = np.arange(n, dtype=np.float64)
        for sigma_frac in (0.12, 0.20, 0.30):
            sigma = sigma_frac * n
            g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
            seeds.append(g / g.sum())

        # 6) Random Dirichlet draws (few)
        for _ in range(3):
            rnd = np.random.rand(n) + 1e-12
            seeds.append(rnd / rnd.sum())

        # 7) Geometric families
        for c in np.linspace(0.5, 5.0, 5):
            r = 1.0 - c / n
            if r > 0.0:
                geom = r ** np.arange(n, dtype=np.float64)
                seeds.append(geom / geom.sum())

        # 8) Symmetric geometric
        for c in np.linspace(0.5, 4.5, 4):
            r = 1.0 - c / n
            if r > 0.0:
                a = r ** np.arange(n, dtype=np.float64)
                a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
                seeds.append(a / a.sum())

        # 9) Upsample previous best if available
        if prev_best is not None:
            up = _upsample(prev_best, n)
            seeds.append(up)

        # ------------------------------------------------------------------
        # Quick scan to capture any obvious winner before heavy optimisation
        # ------------------------------------------------------------------
        for a in seeds:
            val = _compute_val(a)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = a.copy()
                print(f"[scan] len={n} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Adam optimisation from each seed (time‑aware)
        # ------------------------------------------------------------------
        best_seq_len, best_val_len = _optimise_length(n, seeds, deadline)

        if best_seq_len is not None and best_val_len < global_best_val:
            global_best_val = best_val_len
            global_best_seq = best_seq_len.copy()
            print(f"[overall] len={n} improved global best to {global_best_val:.6f}")

        # remember for the next length stage
        prev_best = best_seq_len.copy()

    # ----------------------------------------------------------------------
    # Final simulated‑annealing refinement (if time permits)
    # ----------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        refined_seq, refined_val = _simulated_annealing(global_best_seq, deadline)
        if refined_val < global_best_val:
            global_best_seq = refined_seq
            global_best_val = refined_val
            print(f"[final] SA improved global best to {global_best_val:.6f}")

    # ----------------------------------------------------------------------
    # Defensive fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] using uniform distribution")

    # return a plain Python list as required by the harness
    return global_best_seq.tolist()
