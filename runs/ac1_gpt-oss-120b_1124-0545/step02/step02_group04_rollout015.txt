"""
Propose a step function (non‑negative vector) that minimizes

    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm combines three ideas:

1. **Projected gradient descent on a smooth surrogate**
   – Use the log‑sum‑exp of the convolution entries as a smooth
     approximation of the max.  The gradient can be written as a
     correlation of the soft‑max weights with the current vector.
   – Adam‑style adaptive steps are used and the iterate is projected
     onto the ℓ₁‑simplex {x≥0, sum(x)=√(2n)} at each step.

2. **Periodic linear‑programming “boost”**
   – With the current vector f we solve a small LP:
     maximise  Σ g_j  subject to  f * g ≤ M   and  g ≥ 0,
     where M = max(conv(f,f)).
   – The LP solution g₀ is scaled to the required ℓ₁‑norm and mixed
     with f (convex combination).  This move provably never increases
     the maximal convolution entry and often reduces it significantly.

3. **Occasional random perturbations** to escape plateaus.

All operations are performed with FFTs (O(n log n)) and the LP is
invoked only a few times, so the total runtime stays well below the
1000 s budget.  The best sequence seen is returned as a plain list of
floats.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x≥0, sum(x)=s} (Wang‑Carreira‑Perpiñán)."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length nfft."""
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max with temperature beta."""
    shift = x.max()
    e = np.exp(beta * (x - shift))
    return e / e.sum()


def _lp_boost(f: np.ndarray, M: float, target_sum: float,
              mix: float = 0.2) -> np.ndarray:
    """
    Solve the LP   maximise Σ g_j   s.t.  f * g ≤ M , g≥0.
    Returns a new vector after mixing f with the scaled LP solution.
    """
    n = f.size
    # Build Toeplitz matrix for the convolution constraints.
    # First column = f padded with n-1 zeros, first row = [f0,0,…,0].
    c = np.concatenate([f, np.zeros(n - 1)])
    r = np.concatenate([f[:1], np.zeros(n - 1)])
    A = toeplitz(c, r)                     # shape (2n‑1, n)

    # Linear program: maximise sum(g)  ⇔ minimise –sum(g)
    c_obj = -np.ones(n)                    # objective coefficients
    b = np.full(2 * n - 1, M)              # upper bound on every convolution entry
    bounds = [(0, None)] * n                # g ≥ 0

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success or res.x.sum() <= 0:
        return f.copy()                     # fallback – keep the old vector

    g0 = res.x
    scale = target_sum / g0.sum()
    g = g0 * scale

    # convex combination and projection (the projection restores the exact sum)
    new_f = (1.0 - mix) * f + mix * g
    new_f = _simplex_projection(new_f, target_sum)
    return new_f


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine runs for at most ``budget_s`` seconds and
    returns the best sequence found.
    """
    # ------------------------------------------------------------------
    # Initialise RNGs
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2500                     # default length
        a = np.random.exponential(scale=1.0, size=n)

    target_sum = math.sqrt(2.0 * n)
    a = np.maximum(a, 0.0)
    a = a / a.sum() * target_sum

    # FFT length – must cover 3*n‑2 for the correlation used in the gradient
    nfft = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # temperature for the smooth‑max (log‑sum‑exp)
    beta_start = 0.5
    beta_max = 120.0                       # very high → close to true max
    beta_inc_per_sec = (beta_max - beta_start) / max(budget_s - 20, 1)

    # LP boosting schedule
    last_lp = time.time()
    lp_interval = 30.0                     # seconds between LP calls

    # evaluation bookkeeping
    best_seq = a.copy()
    best_score = float("inf")

    start = time.time()
    deadline = start + budget_s - 5.0       # safety margin
    it = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Current convolution, maximal entry, and (approximate) objective
        conv = _conv_self(a, nfft)          # length 2n‑1
        M = conv.max()
        # since sum(a)=target_sum, the true objective equals M
        cur_score = float(M)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{it:7d}] New best score: {best_score:.6f}  "
                  f"time {time.time() - start:.1f}s")

        # ------------------------------------------------------------------
        # Decide which move to perform
        move = random.random()

        if move < 0.65:
            # ---------- Gradient descent on smooth max (Adam) ----------
            elapsed = time.time() - start
            beta = min(beta_max, beta_start + beta_inc_per_sec * elapsed)

            # soft‑max weights over the convolution entries
            p = _softmax(conv, beta)

            # correlation p ⋆ a (valid part) ⇒ gradient of the smoothed max
            a_rev = a[::-1]
            pf = np.fft.rfft(p, n=nfft)
            ar = np.fft.rfft(a_rev, n=nfft)
            corr_full = np.fft.irfft(pf * np.conj(ar), n=nfft)
            grad = 2.0 * corr_full[n - 1:n - 1 + n]   # shape (n,)

            # Adam update (minimise → move opposite to gradient)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))
            lr = 0.6 * target_sum / math.sqrt(it + 1)   # decreasing step size
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = _simplex_projection(a, target_sum)

        elif move < 0.90:
            # ---------- Occasionally perform the LP boost ----------
            if time.time() - last_lp >= lp_interval:
                a = _lp_boost(a, M, target_sum, mix=0.25)
                last_lp = time.time()
            else:
                # a tiny random perturbation to keep the dynamics alive
                a = a + np.random.normal(scale=target_sum * 5e-4, size=n)
                a = _simplex_projection(a, target_sum)

        else:
            # ---------- Pure random mass transfer (exploration) ----------
            i, j = np.random.randint(n, size=2)
            if i != j:
                max_delta = min(a[i], target_sum * 0.03)
                delta = max_delta * random.random()
                a[i] -= delta
                a[j] += delta
                a = _simplex_projection(a, target_sum)

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    return best_seq.tolist()
