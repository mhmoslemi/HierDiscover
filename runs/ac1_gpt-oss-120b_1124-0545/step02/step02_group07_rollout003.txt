def _subgradient_optimize(start_seq, deadline, rng, target_score=1.48):
    a = start_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    start_time = time.time()
    start_eta = 0.5
    end_eta = 0.001
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        cur_score = 2 * n * max_val
        exact = evaluate_sequence(a.tolist())
        if exact < best_score:
            best_score = exact
            best_a = a.copy()
            print(f"[{time.time() - start_time:.1f}s] New best score={best_score:.6f}")
            if best_score <= target_score:
                break
        # gradient for all active indices
        active = np.where(conv >= max_val - 1e-12)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi+1)
            j = k - i
            g[i] += 2.0 * a[j]
        g_max = np.max(np.abs(g))
        if g_max == 0:
            break
        # step size schedule linear in elapsed time
        elapsed = time.time() - start_time
        total = deadline - start_time
        eta = start_eta * (1.0 - elapsed/total) + end_eta * (elapsed/total)
        factor = np.exp(-eta * g / g_max)
        a_new = a * factor
        a_sum = a_new.sum()
        if a_sum == 0:
            a_new = a.copy()
        else:
            a_new = a_new / a_sum
        if rng.random() < 0.01:
            start_idx = rng.integers(0, n)
            block_len = rng.integers(1, min(256, n - start_idx) + 1)
            a_new[start_idx:start_idx+block_len] = a_new[start_idx:start_idx+block_len][::-1]
        a = a_new
    return best_a.tolist()

```python
"""
Propose a candidate step‑function sequence using a two‑stage global‑local search.
Stage 1: a compact parametric representation (block‑constant) is optimised
         by differential evolution on a reduced length (2048).  The
         best prototype is up‑sampled to the full length (8192) and
         evaluated.
Stage 2: the resulting full‑length vector is refined by a sub‑gradient
         descent on the max‑convolution objective, with an adaptive
         step‑size, occasional random block reversals and a tolerance for
         multiple active convolution entries.
The algorithm respects the 1000‑second budget and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global state for the DE callback (required because SciPy cannot receive
# a closure or lambda according to the competition rules).
# ----------------------------------------------------------------------
_DE_DEADLINE = None
_DE_n_small = None
_DE_block_cnt = None

def _de_time_callback(xk, convergence):
    """Callback for SciPy DE that stops when the global deadline is reached."""
    if _DE_DEADLINE is not None and time.time() >= _DE_DEADLINE:
        return True
    return False

def _de_objective(params, n, block_cnt):
    """
    Objective for the differential‑evolution stage.
    `params` contains a non‑negative height for each block.
    The function builds a block‑constant prototype of length ``n``,
    normalises it to sum = 1 and returns the peak of its self‑convolution.
    """
    block_size = n // block_cnt
    h = np.maximum(params, 0.0)
    seq = np.repeat(h, block_size)
    # handle a possible remainder
    rem = n - block_size * block_cnt
    if rem > 0:
        seq = np.concatenate([seq, h[:rem]])
    total = seq.sum()
    if total == 0.0:
        # all zeros – fall back to a tiny uniform bump
        seq = np.full_like(seq, 1.0 / seq.size)
        total = seq.sum()
    seq = seq / total
    conv = _conv_fft(seq)
    return conv.max()

def _de_objective_wrapper(params):
    """Thin wrapper required by SciPy (no closure)."""
    return _de_objective(params, _DE_n_small, _DE_block_cnt)

def _conv_fft(a):
    """Linear convolution a*a using an FFT of a power‑of‑two length."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())       # next power of two ≥ 2n‑1
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)
    return conv[: 2 * n - 1]

def _normalise_seq(a):

def _gradient_at_max_multi(a, conv, eps=1e-12):
    """Sub‑gradient of L(a)=max_k (a∗a)_k when several k are tied."""

        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g

    The function returns the best sequence (as a plain Python list) found

        cur_score = 2.0 * n * conv.max()
        exact = evaluate_sequence(a.tolist())
        if exact < best_score:

            best_a = a.copy()
            print(f"[{time.time() - start_time:.1f}s] New best score={best_score:.6f}")

                print("[+] Target score reached – stopping sub‑gradient stage.")
                break

        g = _gradient_at_max_multi(a, conv)
        g_max = np.max(g)
        if g_max == 0.0:
            break

        elapsed = time.time() - start_time
        total = deadline - start_time
        eta = start_eta * (1.0 - elapsed / total) + end_eta * (elapsed / total)

        a_new = a * factor
        s = a_new.sum()
        if s == 0.0:
            a_new = a.copy()
        else:
            a_new = a_new / s

        if rng.random() < 0.01:

            block_len = rng.integers(1, min(256, n - start_idx) + 1)
            a_new[start_idx:start_idx + block_len] = a_new[start_idx:start_idx + block_len][::-1]

def _build_seq_from_params(params, n, block_cnt):

    if total == 0.0:

    return seq
