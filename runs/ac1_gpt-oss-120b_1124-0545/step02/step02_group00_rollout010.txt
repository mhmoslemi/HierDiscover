def flatten_move(a, step_scale):
   # get convolution and index of max
   b = np.convolve(a,a)
   k_max = int(np.argmax(b))
   # compute contributions a_i * a_{k_max-i}
   lo = max(0, k_max - (len(a)-1))
   hi = min(len(a)-1, k_max)
   i_vals = np.arange(lo, hi+1)
   contributions = a[i_vals] * a[k_max - i_vals]
   # find index of largest contribution
   idx_max = np.argmax(contributions)
   i = i_vals[idx_max]
   j = k_max - i
   # transfer small mass epsilon from i and j to neighbors
   eps = step_scale * min(a[i], a[j]) * 0.5
   if eps <= 0:
       return a  # unchanged
   a_new = a.copy()
   a_new[i] -= eps
   a_new[j] -= eps
   # distribute eps equally to neighbors
   # choose directions: i-1,i+1 and j-1,j+1 (if in bounds)
   # distribute quarter to each neighbor if valid, else adjust.
   for idx in [i-1, i+1, j-1, j+1]:
       if 0 <= idx < len(a):
           a_new[idx] += eps/4
   # project onto simplex (to correct slight sum errors)
   a_new = project_simplex(a_new)
   return a_new

def solve_lp(f, M):
    n = len(f)
    rows = 2*n-1
    A = np.empty((rows, n), dtype=np.float64)
    b = np.full(rows, M, dtype=np.float64)
    # Build A
    for k in range(rows):
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # compute f_j where j = k-i
        # We'll fill A[k, i_min:i_max+1] = f[k - np.arange(i_min,i_max+1)]
        idx_i = np.arange(i_min, i_max+1)
        A[k, i_min:i_max+1] = f[k - idx_i]  # vectorized
    # Solve
    c = -np.ones(n)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'time_limit': 10}) # but time_limit maybe not allowed?
    if not res.success:
        return None
    return res.x

"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines a rich set of analytically‑motivated initial
candidates (Beta‑shapes, inverse‑sqrt, uniform) with a projected
sub‑gradient descent, occasional flattening moves, and a small number
of linear‑programming refinements (the LP step described in the
statement).  A simple simulated‑annealing acceptance rule lets the
search escape local minima.  The routine respects the time budget,
keeps the best sequence seen and returns it after the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming support (SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog  # type: ignore
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative array a.
    The array is assumed to sum to 1; the function returns
    2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return math.inf
    n = a.shape[0]
    # Use FFT for an O(n log n) full convolution
    size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    A = np.fft.rfft(a, size)
    b = np.fft.irfft(A * A, size)[: 2 * n - 1]
    max_b = b.max()
    return float(2.0 * n * max_b / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def gradient_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Projected sub‑gradient step w.r.t. the currently maximal convolution entry."""
    n = a.shape[0]
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k_max - idx]

    a_new = a - scale * g
    a_new = np.clip(a_new, 0.0, None)
    return project_simplex(a_new)


def flatten_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    A cheap move that takes a little mass from the pair (i,j) that
    contributes most to the peak convolution and spreads it to neighbours.
    """
    n = a.shape[0]
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    i_vals = np.arange(lo, hi + 1, dtype=int)
    j_vals = k_max - i_vals
    # contributions a_i * a_j
    contrib = a[i_vals] * a[j_vals]
    max_idx = int(np.argmax(contrib))
    i = int(i_vals[max_idx])
    j = int(j_vals[max_idx])

    eps = scale * min(a[i], a[j]) * 0.5
    if eps <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= eps
    a_new[j] -= eps

    # distribute eps to valid neighbours
    share = eps / 4.0
    for idx in (i - 1, i + 1, j - 1, j + 1):
        if 0 <= idx < n:
            a_new[idx] += share
    return project_simplex(a_new)


def lp_refine(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP described in the statement:
        maximise  Σ g_i
        s.t.    g_i ≥ 0
                (f * g)_k ≤ M  for every k
    Returns the optimal g (as a dense vector) or None on failure.
    """
    if not _SCIPY_OK:                     # pragma: no cover
        return None
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix A_ub (rows x n)
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # indices i_min..i_max inclusive
        idx_i = np.arange(i_min, i_max + 1, dtype=int)
        A[k, i_min : i_max + 1] = f[k - idx_i]

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximise Σ g_i  → minimise −Σ g_i
    bounds = [(0, None)] * n

    # ‘highs’ is fast and robust for this size
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if not res.success:
        return None
    return res.x


def beta_candidate(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) shape on a discrete grid."""
    i = np.arange(n, dtype=np.float64)
    # centre at 0.5 offset to avoid zeros at the boundaries for p<1
    a = ((i + 0.5) ** (p - 1.0)) * ((n - i - 0.5) ** (p - 1.0))
    # Guard against infinities when p<0.5
    a = np.where(np.isfinite(a), a, 0.0)
    return a / a.sum()


def arcsine_candidate(n: int, offset: float) -> np.ndarray:
    """1/√((i+offset)*(n-1-i+offset)) shape (continuous arcsine analogue)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + offset) * (n - 1 - i + offset))
    a = np.where(np.isfinite(a), a, 0.0)
    return a / a.sum()


def initialise_candidates(n: int) -> list[np.ndarray]:
    """Return a list of analytically motivated normalised candidates."""
    candidates: list[np.ndarray] = []

    # 1. Uniform
    candidates.append(np.ones(n) / n)

    # 2. Beta‑shapes
    for p in np.linspace(0.2, 2.0, 10):
        candidates.append(beta_candidate(n, p))

    # 3. Arcsine‑like families with different offsets
    for off in (0.0, 0.2, 0.5, 1.0, 2.0):
        candidates.append(arcsine_candidate(n, off))

    # 4. Mixed inverse‑sqrt sums
    i = np.arange(n, dtype=np.float64)
    for p in (0.3, 0.5, 0.7, 1.0, 1.5):
        a = (i + 1.0) ** (-p) + (n - i) ** (-p)
        candidates.append(a / a.sum())

    # 5. Pure inverse‑sqrt product (no 0.5 offset)
    a = 1.0 / np.sqrt((i + 1.0) * (n - i))
    candidates.append(a / a.sum())

    # 6. If a previous good construction is available, add it
    if "height_sequence_1" in globals():
        try:
            hs = np.asarray(height_sequence_1, dtype=float)
            if hs.ndim == 1 and hs.size > 0:
                candidates.append(hs / hs.sum())
        except Exception:
            pass

    return candidates

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2).  The routine builds a strong analytic
    start, performs projected sub‑gradient descent with occasional
    flattening/random moves and a few LP refinements, and returns the best
    sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a small safety margin
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # parameters
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1 – pick the best analytic seed
    # ------------------------------------------------------------------
    cand_pool = initialise_candidates(n)
    best_seq = None
    best_score = math.inf

    print(f"[init] evaluating {len(cand_pool)} analytic seeds (n={n})")
    for idx, a in enumerate(cand_pool):
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        print(f"[init] candidate {idx:02d} score = {sc:.6f}")
    a = best_seq.copy()
    print(f"[init] chosen start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – optimisation loop
    # ------------------------------------------------------------------
    init_temp = 2.0
    final_temp = 0.001
    init_step = 0.8
    final_step = 5e-5

    iteration = 0
    last_improve = start_time
    lp_calls = 0
    max_lp_calls = 12                     # keep the total LP workload modest

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 10.0)   # smooth [0,1] scale

        # linear annealing schedules
        temp = init_temp * (1.0 - progress) + final_temp * progress
        step_scale = init_step * (1.0 - progress) + final_step * progress

        # ------------------------------------------------------------------
        # pick a move type
        # ------------------------------------------------------------------
        r = np.random.rand()
        if r < 0.75:
            cand = gradient_move(a, step_scale)          # main engine
        elif r < 0.90:
            cand = random_move(a, step_scale * 0.5)      # exploratory mass swap
        else:
            cand = flatten_move(a, step_scale * 0.3)     # targeted peak reduction

        cand_score = compute_score(cand)

        # Metropolis acceptance (lower is better, but occasional uphill allowed)
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            if np.random.rand() < math.exp(-(cand_score - best_score) / max(temp, 1e-12)):
                accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # ------------------------------------------------------------------
        # occasional LP refinement (once every few thousand iterations)
        # ------------------------------------------------------------------
        if iteration % 5000 == 0 and lp_calls < max_lp_calls and time.time() < deadline:
            # compute current peak M
            b_full = np.convolve(a, a)
            M = b_full.max()
            g = lp_refine(a, M)
            if g is not None:
                sum_g = g.sum()
                if sum_g > 1.0 + 1e-12:
                    # normalise g to a probability vector
                    g_norm = g / sum_g
                    # mixing fraction – we cannot take t larger than (sum_g-1)/sum_g
                    t_mix = min(0.5, (sum_g - 1.0) / sum_g)
                    f_new = (1.0 - t_mix) * a + t_mix * g_norm
                    f_new = project_simplex(f_new)
                    new_score = compute_score(f_new)
                    if new_score < best_score:
                        a = f_new
                        best_seq = f_new.copy()
                        best_score = new_score
                        last_improve = time.time()
                        print(f"[LP {lp_calls}] improved to {best_score:.6f}")
                    else:
                        # accept with temperature as a fallback
                        if np.random.rand() < math.exp(-(new_score - best_score) / max(temp, 1e-12)):
                            a = f_new
                    lp_calls += 1
                else:
                    # no gain possible from LP – treat as exhausted
                    lp_calls = max_lp_calls

        # ------------------------------------------------------------------
        # restart if we have been stuck for a while
        # ------------------------------------------------------------------
        if iteration % 20000 == 0 and (time.time() - last_improve) > 30.0:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            print(f"[{iteration:07d}] restart after stagnation")

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, LP calls={lp_calls}, best score={best_score:.6f}, time={total_time:.1f}s")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
