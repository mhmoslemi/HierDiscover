def simplex_projection(v):
    """Project onto simplex {x >=0, sum x = 1}."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def compute_conv(seq):
    n = len(seq)
    size = 2*n-1
    m = 1 << ((size-1).bit_length())
    # compute FFT of seq padded to m
    f = np.fft.rfft(seq, n=m)
    conv_full = np.fft.irfft(f * f, n=m)[:size]
    return conv_full

def loss_and_weights(conv, alpha):
    max_c = conv.max()
    exp_shift = np.exp(alpha * (conv - max_c))
    sumexp = exp_shift.sum()
    loss = max_c + np.log(sumexp) / alpha
    w = exp_shift / sumexp
    return loss, w

"""
A fast projected‑gradient optimizer for the “max‑self‑convolution” problem.

The algorithm works directly on the probability simplex (non‑negative entries
with sum = 1).  It minimises a smooth surrogate of the objective

        L(a) = max_k (a * a)[k]

by using a soft‑max (log‑sum‑exp) with a temperature α that is slowly
increased, turning the surrogate into the true max.  Each iteration performs

1.   Full convolution a*a (FFT) – needed for the loss and for the active‐set
     index when α is large.
2.   Soft‑max weights w_k ∝ exp(α·conv_k) (stable by subtracting the max).
3.   Gradient of the surrogate:
         ∂L/∂a_i = 2 * Σ_k w_k a_{k-i}
     which is computed as a convolution of the reversed weight vector with a.
4.   A back‑tracking line search followed by projection onto the simplex.
5.   Simple schedule for the step size and for α (increase α, decay step).

Occasionally a tiny random jitter is added to escape plateaus.  The best
sequence encountered (according to the exact ``evaluate_sequence`` metric)
is stored and returned.  The whole search respects the supplied time budget
(≤ 1000 s) and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x ≥ 0, Σx = 1})
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the closest point to v in the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution of a vector with itself
# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a*a (full convolution) using a real FFT."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for speed
    m = 1 << ((size - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:size]
    return conv


# ----------------------------------------------------------------------
# Compute smooth (log‑sum‑exp) loss and its soft‑max weights
# ----------------------------------------------------------------------
def _softmax_loss_and_weights(conv: np.ndarray, alpha: float):
    """
    Return (loss, weights) where
        loss = (1/α) * log Σ_k exp(α * conv_k)
    and   weights_k = exp(α * conv_k) / Σ exp(α * conv_k)
    The implementation is numerically stable.
    """
    max_c = conv.max()
    shifted = conv - max_c                      # avoid overflow
    exp_shift = np.exp(alpha * shifted)
    sumexp = exp_shift.sum()
    # loss = max_c + (1/α) * log(sumexp)
    loss = max_c + (np.log(sumexp) / alpha)
    weights = exp_shift / sumexp
    return loss, weights


# ----------------------------------------------------------------------
# Gradient of the smooth loss w.r.t. the sequence a
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  L where L uses the soft‑max weights w.
    Formula: grad_i = 2 * Σ_k w_k * a_{k-i}
    (indices outside [0,n‑1] are ignored).
    This is implemented via one FFT‑based convolution.
    """
    n = a.shape[0]
    # reverse the weight vector (needed for the shift pattern)
    w_rev = w[::-1]

    size2 = w_rev.shape[0] + n - 1               # = 3n‑2
    m2 = 1 << ((size2 - 1).bit_length())
    fw = np.fft.rfft(w_rev, n=m2)
    fa = np.fft.rfft(a, n=m2)
    conv_rev = np.fft.irfft(fw * fa, n=m2)[:size2]

    # indices 2n‑2‑i (i=0…n‑1) give the required sums
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv_rev[idx]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` (equivalently, the maximal entry of its
    self‑convolution).  The routine respects the supplied time budget and
    returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin in case of overhead
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1) Initialisation – use global construction if present,
    #    otherwise fall back to a simple power‑law shape.
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore
        if init.size == 0 or init.sum() == 0:
            raise ValueError
        n = init.size
        print(f"[init] using supplied height_sequence_1 (n={n})")
    except Exception:
        # fallback: a reasonably spread distribution (inverse‑square‑root)
        n = 5000
        idx = np.arange(n, dtype=float)
        init = 1.0 / np.sqrt(idx + 1.0)
        init /= init.sum()
        print(f"[init] fallback to power‑law seed (n={n})")

    # enforce feasibility
    init = np.maximum(init, 0.0)
    init_sum = init.sum()
    if init_sum == 0:
        init = np.full(n, 1.0 / n)
    else:
        init /= init_sum

    seq = init.copy()
    conv = _self_convolution_fft(seq)
    best_score = 2.0 * n * conv.max()
    best_seq = seq.copy()
    # initial smooth loss (α will be tuned)
    alpha = 1.0
    loss, weights = _softmax_loss_and_weights(conv, alpha)

    # hyper‑parameters
    step_size = 0.5          # initial gradient step
    step_decay = 0.99995     # multiplicative decay per iteration
    alpha_mul = 1.02         # geometric increase of α each iteration
    max_alpha = 2000.0
    iter_cnt = 0
    report_every = 500

    print(f"[baseline] score={best_score:.6f} loss={loss:.6f}")

    # --------------------------------------------------------------
    # 2) Main optimisation loop (smooth → hard max)
    # --------------------------------------------------------------
    while time.time() < deadline:
        iter_cnt += 1

        # ----------------------------------------------------------
        # gradient of the current smooth loss
        # ----------------------------------------------------------
        grad = _gradient_from_weights(seq, weights)

        # ----------------------------------------------------------
        # back‑tracking line search (max 5 reductions)
        # ----------------------------------------------------------
        step = step_size
        for _ in range(5):
            cand = _project_onto_simplex(seq - step * grad)
            conv_cand = _self_convolution_fft(cand)
            loss_cand, weights_cand = _softmax_loss_and_weights(conv_cand, alpha)
            if loss_cand <= loss:
                break
            step *= 0.5
        else:
            # If no improvement, perform a tiny random jitter
            jitter = np.random.normal(scale=1e-5, size=n)
            cand = _project_onto_simplex(seq + jitter)
            conv_cand = _self_convolution_fft(cand)
            loss_cand, weights_cand = _softmax_loss_and_weights(conv_cand, alpha)

        # ----------------------------------------------------------
        # accept the candidate
        # ----------------------------------------------------------
        seq = cand
        conv = conv_cand
        loss = loss_cand
        weights = weights_cand

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iter_cnt:6d} score={best_score:.6f} loss={loss:.6f} α={alpha:.1f}")

        # ----------------------------------------------------------
        # schedule updates
        # ----------------------------------------------------------
        alpha = min(alpha * alpha_mul, max_alpha)
        step_size *= step_decay

        # occasional cheap random perturbation to shake the system
        if iter_cnt % 1000 == 0:
            eps = np.random.uniform(0.0, 0.01)
            i, j = np.random.choice(n, 2, replace=False)
            cand2 = seq.copy()
            transfer = eps * cand2[i]
            cand2[i] -= transfer
            cand2[j] += transfer
            seq = _project_onto_simplex(cand2)

        # periodic report
        if iter_cnt % report_every == 0:
            elapsed = time.time() - start_time
            print(f"[report] iter={iter_cnt} elapsed={elapsed:.1f}s best_score={best_score:.6f}")

        # early exit if we already beat the target comfortably
        if best_score < 1.45:
            break

    # --------------------------------------------------------------
    # 3) Final hard‑max refinement (few projected‑gradient steps on the true max)
    # --------------------------------------------------------------
    # only if there is time left
    while time.time() < deadline:
        # find active convolution index
        k_max = np.argmax(conv)
        # gradient of the active entry (ordered pairs)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idxs = np.arange(lo, hi + 1, dtype=int)
        grad_hard = np.zeros_like(seq)
        grad_hard[idxs] = seq[k_max - idxs]          # derivative 2*a_{k_max-i} collapsed into factor 2 via step
        # keep on simplex
        grad_hard -= grad_hard.mean()
        # small step
        step_hard = 0.02
        cand_hard = seq - step_hard * grad_hard
        cand_hard = _project_onto_simplex(cand_hard)
        conv_hard = _self_convolution_fft(cand_hard)
        score_hard = 2.0 * n * conv_hard.max()
        if score_hard < best_score:
            best_score = score_hard
            best_seq = cand_hard.copy()
            seq = cand_hard
            conv = conv_hard
            print(f"[hard] iter={iter_cnt} score={best_score:.6f}")
        else:
            # no improvement – reduce step and continue a few more attempts
            step_hard *= 0.7
        iter_cnt += 1
        if step_hard < 1e-6:
            break

    print(f"[final] iterations={iter_cnt}, best_score={best_score:.6f}")
    # Return plain Python list as required by the harness
    return best_seq.tolist()
