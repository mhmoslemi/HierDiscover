def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np, math, random
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    best_seq = None
    best_score = np.inf

    # Determine n: try to get large n for better optimum but within time
    # We'll try n = 8000 (or 10000) as long as conv O(n log n) doesn't dominate.
    # Let's set n = min(10000, 2 * int(math.sqrt(budget_s)*100))? Hmm.
    # We'll just pick n = 8000.

    n = 8000

    target_sum = math.sqrt(2 * n)

    # define projection function (simple but used earlier)
    def simplex_projection(v, s):
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        ind = np.arange(1, len(v)+1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            return np.full_like(v, s / len(v))
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

    # initial candidates list
    candidates = []

    # uniform
    uniform = np.full(n, target_sum / n)
    candidates.append(uniform)

    # exponential
    exp_vec = np.random.exponential(scale=1.0, size=n)
    exp_vec = exp_vec / exp_vec.sum() * target_sum
    candidates.append(exp_vec)

    # triangular (tent)
    # linear increase then decrease
    half = n // 2
    tent = np.concatenate([np.linspace(0, 1, half, endpoint=False),
                           np.linspace(1, 0, n - half, endpoint=False)])
    tent = tent / tent.sum() * target_sum
    candidates.append(tent)

    # beta distribution shape to get peaked shape (like a^a)
    # Use parameters alpha=0.5 beta=0.5 (U-shaped) maybe not good. Actually shape around 2.
    # We'll sample a Beta distribution and scale.
    try:
        import scipy.stats as stats
        beta = stats.beta.rvs(a=0.5, b=0.5, size=n, random_state=seed+1)
        beta = beta / beta.sum() * target_sum
        candidates.append(beta)
    except Exception:
        pass

    # previous known sequence if exists
    if 'height_sequence_1' in globals():
        seq_prev = np.array(height_sequence_1, dtype=float)
        if len(seq_prev) != n:
            # reinterpolate: either pad or truncate, then renorm
            if len(seq_prev) > n:
                seq_prev = seq_prev[:n]
            else:
                extra = np.random.exponential(scale=1.0, size=n - len(seq_prev))
                seq_prev = np.concatenate([seq_prev, extra])
        seq_prev = seq_prev / seq_prev.sum() * target_sum
        candidates.append(seq_prev)

    # function for evaluating max conv (since sum is normalized)
    def evaluate_max(conv):
        # conv is numpy array of convolution values
        return float(conv.max())

    # gradient descent optimizer
    # hyperparameters
    total_iters = 0

    # We'll allocate time budget among candidates adaptively.
    # For each candidate, run gradient descent until time limit or certain iteration count.
    # We'll interleave candidates: run few iterations per candidate, keep best overall.

    # We'll store each candidate's state: a, m, v (Adam moments)
    states = []
    for cand in candidates:
        a = cand.copy()
        # ensure nonnegative sum via projection (just in case)
        a = simplex_projection(a, target_sum)
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        states.append((a, m, v))

    # set schedule parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Surrogate schedule parameters
    p_start = 2.0
    p_end = 60.0  # large p approximates max.

    lr_start = target_sum * 0.001
    lr_end = target_sum * 0.00002

    # Main loop
    step_counter = 0
    while time.time() < deadline:
        # compute time fraction
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        # update p and lr based on frac
        p = p_start * ((p_end / p_start) ** frac)  # exponential schedule
        lr = lr_start * ((lr_end / lr_start) ** frac)

        # iterate over each candidate state
        new_states = []
        for (a, m, v) in states:
            # compute convolution
            conv = np.convolve(a, a)
            cur_score = float(conv.max())
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{step_counter:6d}] New best score {best_score:.6f} (p={p:.1f})")

            # compute surrogate weight w = conv^{p-1}
            # to avoid overflow for large p, we normalize
            # compute max conv value
            cmax = conv.max()
            # subtract log to keep numbers moderate? We'll compute w = (conv / cmax)^{p-1}
            # Then gradient scaled by cmax^{p-1}
            if cmax == 0:
                w = np.zeros_like(conv)
                weight_factor = 0.0
            else:
                # compute pow safely
                # compress values that are below small threshold to zero to avoid underflow
                ratio = conv / cmax
                # we can clip ratio to [0,1]
                # compute pow
                w = np.power(ratio, p - 1, where=ratio>0)
                weight_factor = (cmax ** (p - 1))
            # gradient of surrogate = p * (sum conv^p)^{1/p - 1} * (2 * conv(w) convolve a)
            # Compute sum conv^p
            # sum_pow = np.sum(conv ** p)  # may overflow
            # more stable: use scaling: sum_pow = cmax**p * np.sum(ratio**p)
            sum_pow = (cmax ** p) * np.sum(np.power(ratio, p, where=ratio>0))
            # compute scaling factor
            if sum_pow == 0:
                scale = 0.0
            else:
                scale = p * (sum_pow) ** (1.0 / p - 1.0) * 2.0 * weight_factor
                # Actually after factoring we have: gradient = 2 * p * (sum c^p)^{1/p -1} * (conv^{p-1} convolve a)
                # But we already included weight_factor; we need to compute conv_{w} * a part.
                # We'll compute conv_w_a = np.convolve(w, a[::-1]) slice length n
            # Compute convolution w * a_rev
            conv_w_a = np.convolve(w, a[::-1])
            first = conv_w_a[(n-1):(n-1+n)]
            # gradient = scale * first
            grad = scale * first

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** (step_counter + 1))
            v_hat = v / (1.0 - beta2 ** (step_counter + 1))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a_new = np.maximum(a_new, 0.0)  # nonnegative
            a_new = simplex_projection(a_new, target_sum)

            # occasionally perturb lengths
            if step_counter % 400 == 0 and step_counter > 0:
                # random perturbation mixing best_seq
                rnd = np.random.exponential(scale=1.0, size=n)
                rnd = rnd / rnd.sum() * target_sum
                mix = np.random.rand()
                a_pert = mix * best_seq + (1 - mix) * rnd
                a_pert = simplex_projection(a_pert, target_sum)
                # evaluate
                conv_pert = np.convolve(a_pert, a_pert)
                score_pert = float(conv_pert.max())
                if score_pert < best_score:
                    best_score = score_pert
                    best_seq = a_pert.copy()
                    print(f"[{step_counter:6d}] New best after perturb {best_score:.6f}")
                # accept if improves current a (maybe)
                if score_pert < cur_score:
                    a_new = a_pert

            new_states.append((a_new, m, v))
            step_counter += 1
            if time.time() >= deadline:
                break
        states = new_states
        # if stagnated: maybe restart some candidates
        if step_counter % 5000 == 0:
            # random restart for half of the candidates
            for i in range(len(states)):
                if random.random() < 0.5:
                    a0 = np.random.exponential(scale=1.0, size=n)
                    a0 = a0 / a0.sum() * target_sum
                    m0 = np.zeros_like(a0)
                    v0 = np.zeros_like(a0)
                    states[i] = (a0, m0, v0)

    # final return
    return best_seq.tolist()

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, random, math
    np.random.seed(seed)
    random.seed(seed)

    # Choose n. Possibly use existing height_sequence_1 if length > maybe 5000.
    if 'height_sequence_1' in globals():
        n0 = len(height_sequence_1)
        # We'll keep n same for simplicity.
        n = n0
        base = np.asarray(height_sequence_1, dtype=float)
    else:
        # Use 2500 default
        n = 2500
        base = np.full(n, 1.0)

    # Normalization sum
    target_sum = math.sqrt(2.0 * n)

    # Ensure base normalized
    base = base / base.sum() * target_sum

    # Helper functions

    # Simplex projection (if needed)
    def simplex_projection(v, s):
        # same as earlier
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        ind = np.arange(1, v.size + 1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            return np.full_like(v, s / v.size)
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        return w

    # convolution max using FFT
    def max_conv(a):
        n_local = a.shape[0]
        mfft = 1 << ((2 * n_local - 1).bit_length())
        A = np.fft.rfft(a, mfft)
        conv = np.fft.irfft(A * A, mfft)[:2 * n_local - 1]
        return float(conv.max())

    # Phase 1: gradient descent on p-norm surrogate
    # Options: initialize from uniform, random exponential, maybe from base.
    uniform = np.full(n, target_sum / n)
    exp_vec = np.random.exponential(scale=1.0, size=n)
    exp_vec = exp_vec / exp_vec.sum() * target_sum
    # Use a mix for initial a
    a = (uniform + exp_vec) / 2.0
    a = a / a.sum() * target_sum

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t_iter = 0

    # schedule parameters for surrogate p and learning rate
    p_start = 2.0
    p_end = 80.0
    lr_start = target_sum * 0.001
    lr_end = target_sum * 0.00002

    start_time = time.time()
    deadline = start_time + budget_s - 20.0  # keep 20 sec for final annealing

    best_seq = a.copy()
    best_score = max_conv(best_seq)

    # gradient descent loop
    while time.time() < deadline:
        # compute schedule fractions
        elapsed = time.time() - start_time
        total_time = deadline - start_time
        frac = elapsed / total_time
        p = p_start * ((p_end / p_start) ** frac)
        lr = lr_start * ((lr_end / lr_start) ** frac)

        # compute conv and surrogate gradients
        # conv
        n_local = n
        mfft = 1 << ((2 * n_local - 1).bit_length())
        A = np.fft.rfft(a, mfft)
        conv = np.fft.irfft(A * A, mfft)[:2 * n_local - 1]

        # compute smooth surrogate weights: w_i = conv_i^{p-1}
        cmax = conv.max()
        if cmax == 0:
            break  # degenerate
        ratio = conv / cmax
        # avoid divide by zero for zeros
        ratio = np.where(ratio > 0, ratio, 0.0)
        w = np.power(ratio, p - 1, where=ratio > 0)
        # compute factor
        sum_pow = np.sum(np.power(ratio, p, where=ratio > 0))
        # Because scaling by cmax factor cancels out
        factor = 2.0 * (sum_pow ** (1.0 / p - 1.0)) if sum_pow > 0 else 0.0

        a_rev = a[::-1]
        conv_w_a = np.fft.irfft(np.fft.rfft(w, mfft) * np.fft.rfft(a_rev, mfft), mfft)[:2 * n_local - 1]

        first = conv_w_a[(n_local - 1):(n_local - 1 + n_local)]

        grad = factor * first

        m_hat = m / (1.0 - beta1 ** t_iter)

def propose_candidate(seed=42, budget_s=1000):

   # Choose n

   # Initial a: uniform or random exponential normalized.

   a = a / a.sum() * target_sum

   best_seq = a.copy()
   best_score = compute_max_conv(a)

   start = time.time()
   deadline = start + budget_s - 5.0

   iteration = 0
   mixing_t = 0.2  # step size

   while time.time() < deadline:
        # compute M = max conv of current a
        M = max_conv(a)

        # Build A matrix as sparse
        rows = []
        cols = []
        data = []

        # For each constraint k in 0..2n-2
        for k in range(2*n - 1):
            # j range such that 0 <= j < n and 0 <= k-j < n -> j in [max(0, k - (n-1)), min(n-1, k)]
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            for j in range(j_start, j_end+1):
                i = k - j
                rows.append(k)
                cols.append(j)
                data.append(a[i])  # coefficient f_i = a[i]

        # also optional g <= 1000 (not needed)
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        try:

        except cp.error.SolverError:
            # fallback to faster solver like OSQP? Not linear? We'll ignore.
            break

        if g.value is None:
            break

        g_opt = np.array(g.value).flatten()
        # Scale to same L1 sum as a
        sum_g = g_opt.sum()
        if sum_g == 0:

        g_scaled = g_opt * (target_sum / sum_g)

        a_new = (1 - mixing_t) * a + mixing_t * g_scaled

            print(f"[Iter {iteration}] new best max(conv) {best_score:.6f}")

        # optional random perturbation if stagnation
        iteration += 1

   return best_seq.tolist()

"""
Propose a step‑function (non‑negative vector) that minimises
    E(a) = 2·n·max(conv(a,a)) / (∑a)² .

The algorithm consists of two stages:

1️⃣  Gradient descent on a smooth p‑norm approximation of the max‑conv
    (p is gradually increased, Adam‑style moments are used and the
    iterate is projected onto the ℓ₁‑simplex to keep the sum fixed
    to √(2·n)).  This stage is fast and quickly finds a good “shape”.

2️⃣  Simulated‑annealing on the ordering / tiny mass transfers of the
    entries found in stage 1.  Since max(conv) is just the maximum
    inner‑product of the vector with its shifted copies, re‑ordering the
    same multiset of values can lower the peak further.  Swaps and
    small transfers are accepted with a Metropolis rule and the best
    vector seen is returned.

The whole routine respects the 1000 s budget and prints occasional
progress reports.  All helper functions are top‑level and there are
no closures or file I/O.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex { w ≥ 0 , ∑ w = s }."""
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all equal
        return np.full_like(v, s / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)) using a real FFT (full convolution)."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine runs a p‑norm smooth gradient descent followed
    by a simulated‑annealing permutation / mass‑transfer phase.
    """
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # problem size -------------------------------------------------
    # use the existing construction if it exists, otherwise a safe default
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base = np.asarray(height_sequence_1, dtype=float)
    else:
        n = 2500                         # default, works well within the budget
        base = np.full(n, 1.0)

    target_sum = math.sqrt(2.0 * n)      # we keep ∑a = √(2n)  ⇒  evaluate = max(conv)

    # --------------------------------------------------------------
    # initialise a few starters and pick the best as a start point
    uniform = np.full(n, target_sum / n)
    exp_vec = np.random.exponential(scale=1.0, size=n)
    exp_vec = exp_vec / exp_vec.sum() * target_sum
    starters = [base, uniform, exp_vec]

    best_seq = None
    best_score = np.inf
    for cand in starters:
        cand = _simplex_projection(cand, target_sum)
        sc = _max_convolution(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # current iterate for the gradient phase
    a = best_seq.copy()

    # Adam moments -------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t_iter = 0

    # surrogate‑p‑norm schedule
    p_start, p_end = 2.0, 80.0          # p grows → better approximation of max
    lr_start = target_sum * 0.001
    lr_end   = target_sum * 0.00002

    start_time   = time.time()
    # allocate ~65 % of the time budget to the gradient stage
    gd_deadline   = start_time + budget_s * 0.65
    final_deadline = start_time + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # Gradient descent (p‑norm surrogate)
    # --------------------------------------------------------------
    while time.time() < gd_deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (gd_deadline - start_time)          # 0 → 1
        # smooth schedules
        p  = p_start * ((p_end / p_start) ** frac)
        lr = lr_start * ((lr_end / lr_start) ** frac)

        # full convolution (FFT – cheap)
        n_local = a.shape[0]
        m_fft = 1 << ((2 * n_local - 1).bit_length())
        A = np.fft.rfft(a, n=m_fft)
        conv = np.fft.irfft(A * A, n=m_fft)[:2 * n_local - 1]

        # ---- p‑norm surrogate -------------------------------------------------
        cmax = conv.max()
        if cmax == 0.0:
            break
        ratio = conv / cmax                         # ∈[0,1]
        # w = ratio^{p‑1}
        w = np.power(ratio, p - 1, where=ratio > 0)

        # sum of ratio^{p}
        sum_ratio_pow = np.sum(np.power(ratio, p, where=ratio > 0))
        if sum_ratio_pow == 0.0:
            factor = 0.0
        else:
            factor = 2.0 * p * (sum_ratio_pow ** (1.0 / p - 1.0))

        # convolution w * a_rev (only the central slice needed)
        a_rev = a[::-1]
        Aw = np.fft.rfft(w, n=m_fft)
        Ar = np.fft.rfft(a_rev, n=m_fft)
        conv_w_a = np.fft.irfft(Aw * Ar, n=m_fft)[:3 * n_local - 2]
        # slice corresponding to the gradient (same as in the original code)
        first = conv_w_a[(n_local - 1):(n_local - 1 + n_local)]

        grad = factor * first
        # optional clipping to keep steps sane
        grad = np.clip(grad, -10.0 * target_sum, 10.0 * target_sum)

        # ---- Adam update ----------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _simplex_projection(a_new, target_sum)

        a = a_new

        # record progress
        cur_score = _max_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[GD {t_iter:6d}] New best max(conv) = {best_score:.6f}")

    # --------------------------------------------------------------
    # Simulated annealing on permutations / tiny transfers
    # --------------------------------------------------------------
    # work with the best vector found so far
    a_curr = best_seq.copy()
    cur_score = best_score

    # temperature schedule – start a bit above the current excess over 1
    T_start = max(0.1, (cur_score - 1.0) * 0.5)
    T_end   = 1e-4

    # remaining time for SA
    sa_deadline = final_deadline
    it_sa = 0
    while time.time() < sa_deadline:
        # linearly decreasing temperature
        elapsed_sa = time.time() - (sd := sa_deadline - (final_deadline - start_time))
        # the expression above is just a cheap way to reuse time – we
        # compute the fraction directly:
        frac_sa = (time.time() - start_time) / (final_deadline - start_time)
        T = T_start * ((T_end / T_start) ** frac_sa)

        # choose move type
        if random.random() < 0.7:                 # pure swap
            i, j = random.sample(range(n), 2)
            a_cand = a_curr.copy()
            a_cand[i], a_cand[j] = a_cand[j], a_cand[i]
        else:                                      # small mass transfer
            i, j = random.sample(range(n), 2)
            if a_curr[i] == 0:
                continue
            delta = random.random() * a_curr[i] * 0.3   # move up to 30 % of source
            a_cand = a_curr.copy()
            a_cand[i] -= delta
            a_cand[j] += delta

        cand_score = _max_convolution(a_cand)

        delta_score = cand_score - cur_score
        if delta_score < 0 or random.random() < math.exp(-delta_score / (T + 1e-12)):
            a_curr = a_cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_curr.copy()
                print(f"[SA {it_sa:7d}] New best max(conv) = {best_score:.6f}")

        it_sa += 1

    # --------------------------------------------------------------
    # Return the best sequence (as a plain Python list)
    # --------------------------------------------------------------
    return best_seq.tolist()
