"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The search uses a smooth‑max approximation of the L∞‑peak and an
Adam‑style projected gradient descent on the simplex.  A few
analytically motivated initialisations (uniform, inverse‑sqrt,
mixed‑power families and, if present, the previously best sequence)
seed the optimisation.  The temperature β of the log‑sum‑exp
approximation is increased linearly during the run, which first
provides a smooth landscape for escaping shallow plateaus and later
behaves like a true sub‑gradient of the max‑entry.  After each update
the vector is projected onto the probability simplex (non‑negative,
sum = 1) and symmetrised (a ↔ a[::-1]) – symmetry is known to be
advantageous for this problem.  Random jitter and restarts avoid
stagnation.  The routine respects a user‑supplied time budget (default
1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑2 ≥ x."""
    return 1 << (x - 1).bit_length()


def convolve_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


def conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y via FFT."""
    L = len(x) + len(y) - 1
    size = _next_pow2(L)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[:L]


# ----------------------------------------------------------------------
# Random perturbation (tiny mass transfer)
# ----------------------------------------------------------------------


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    delta = step_scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------


def initialise_candidates(n: int) -> list[np.ndarray]:
    """Return a modest pool of analytically motivated (and normalised) candidates."""
    candidates = []

    # 1. uniform
    candidates.append(np.ones(n) / n)

    i = np.arange(n, dtype=np.float64)

    # 2. arcsine‑like (inverse‑sqrt product with half‑offset)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    candidates.append(a / a.sum())

    # 3. mixed inverse‑sqrt powers
    for p in (0.3, 0.5, 0.7, 1.0):
        a = (i + 1) ** (-p) + (n - i) ** (-p)
        candidates.append(a / a.sum())

    # 4. pure inverse‑sqrt product (no offset)
    a = 1.0 / np.sqrt((i + 1) * (n - i))
    candidates.append(a / a.sum())

    # 5. previous best if present
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.shape[0] == n:
            candidates.append(seq / seq.sum())
        else:
            # simple rescaling/truncation to match length n
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                # pad with zeros
                seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
            # re‑normalise
            if seq.sum() > 0:
                candidates.append(seq / seq.sum())

    # 6. random Dirichlet start
    candidates.append(np.random.dirichlet(np.ones(n)))

    return candidates


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function under the given time budget.
    Returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    n = int(kwargs.get("n", 2000))

    # ---- 1. initialise ----
    cand_pool = initialise_candidates(n)

    best_seq = None
    best_score = np.inf

    def exact_score(a: np.ndarray) -> float:
        """Exact objective (scale‑invariant) for a normalised array."""
        conv = convolve_self_fft(a)
        return float(2.0 * n * conv.max())

    print("[init] evaluating analytical candidates")
    for idx, a in enumerate(cand_pool):
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            continue
        a = a / a.sum()
        sc = exact_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        print(f"[init] candidate {idx:02d} score = {sc:.6f}")

    if best_seq is None:
        raise RuntimeError("Failed to obtain a valid initial candidate.")

    # ---- 2. optimisation loop (Adam + smooth‑max) ----
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters
    max_iters = 200_000                     # hard upper bound
    init_lr = 0.05
    final_lr = 5e-5
    beta_target = 30.0                       # final smoothing temperature
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    enforce_symmetry = True
    last_improve = time.time()
    iter_cnt = 0

    while time.time() < deadline and iter_cnt < max_iters:
        # progress fraction (0‑1)
        frac = iter_cnt / max_iters

        # ----- convolution of current iterate -----
        conv = convolve_self_fft(a)

        # ----- smooth‑max weights (log‑sum‑exp) -----
        beta = 1.0 + (beta_target - 1.0) * frac
        max_conv = conv.max()
        # shifting for numerical stability
        w = np.exp(beta * (conv - max_conv))
        w /= w.sum()                         # shape (2·n‑1,)

        # ----- gradient of the smooth max -----
        # correlation w ⋆ a (valid part) = convolution(w, a_rev)[n‑1 : n‑1+n]
        a_rev = a[::-1]
        corr_full = conv_full_fft(w, a_rev)  # length = (2n‑1)+n‑1
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----- Adam update -----
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        lr = init_lr * (1.0 - frac) + final_lr * frac
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random jitter
        if np.random.rand() < 0.01:
            a_new = random_move(a_new, step_scale=lr)

        # enforce symmetry (helps to keep the peak low)
        if enforce_symmetry:
            a_new = (a_new + a_new[::-1]) * 0.5

        # projection onto simplex (non‑negative, sum = 1)
        a_new = project_simplex(a_new)

        # ----- evaluate exact objective for possible improvement -----
        conv_new = convolve_self_fft(a_new)
        score_new = float(2.0 * n * conv_new.max())

        if score_new + 1e-12 < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            last_improve = time.time()
            print(f"[iter {iter_cnt:07d}] new best = {best_score:.6f}")

        # accept the move (always continue from a_new)
        a = a_new

        # ----- stagnation handling -----
        if time.time() - last_improve > 30.0:
            # slight jitter around the current best and reset Adam stats
            jitter = np.random.rand(n) * 1e-4
            a = best_seq + jitter
            a = project_simplex(a)
            if enforce_symmetry:
                a = (a + a[::-1]) * 0.5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[iter {iter_cnt:07d}] restart (stagnation)")

    total_time = time.time() - start_time
    print(f"[done] iterations={iter_cnt}, best_score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
