Project onto simplex algorithm:

def simplex_projection(v):
    # ensure nonnegative sum = 1
    n=v.shape[0]
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)-1
    ind=np.arange(n)+1
    cond=u - cssv/ind > 0
    rho=np.where(cond)[0][-1]
    theta=cssv[rho]/(rho+1)
    w=np.maximum(v-theta,0)
    return w

def _simplex_projection(v):
    """Projection of vector v onto the probability simplex {x >=0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Now gradient function:

```
def _gradient_top_k(seq, top_k):
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    if top_k >= conv.size:
        k_indices = np.arange(conv.size)
    else:
        k_indices = np.argpartition(-conv, top_k-1)[:top_k]
    grad = np.zeros_like(seq)
    for k in k_indices:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1)
        grad[idx] += 2 * seq[k - idx]
    return grad
```

def _projected_gradient_step(seq, step, top_k=5):
    grad = _gradient_top_k(seq, top_k)
    new_seq = seq - step * grad
    # enforce sum=1 and nonneg via simplex projection
    new_seq = _simplex_projection(new_seq)
    return new_seq

"""
propose_candidate
=================

Search for a non‑negative step‑function (a list of heights) that minimizes the
`evaluate_sequence` score.

Algorithm
---------

1. **Coarse grid**  
   Choose a length `n` (5000 works well).  Scan a 2‑D grid of the two
   parameters of a power‑law family  

        a_i ∝ (i+1+δ)^{-α} ,   i = 0 … n‑1

   where the exponent α∈[0.30,0.80] and the offset δ≥0.  The grid is fairly
   dense (α step 0.005, δ step 0.2).  The best pair (α,δ) is kept.

2. **Fine grid**  
   Around the best pair a finer search (α step ≈5·10⁻⁴, δ step 0.05) is
   performed to locate a more accurate starting point.

3. **Projected gradient descent**  
   The score is `2·n·max(conv(a,a))` after normalising `sum(a)=1`.  The
   gradient of a single convolution entry `(a*a)[k]` is known analytically:

        ∂(a*a)[k]/∂a_j = 2·a_{k‑j}   (when the index exists)

   To reduce the *maximum* entry we take the sub‑gradient formed by the
   `top_k` largest convolution values (here `top_k=10`).  After a step we
   project back onto the probability simplex by clipping to non‑negative and
   renormalising.  A simple Adam‑style adaptive step‑size and occasional
   random mass swaps are used to escape plateaus.

4. **Time‑aware loop**  
   All work stops a few seconds before the 1000 s budget so the function can
   return the best sequence found.

The combination of a dense parameter sweep and a stronger gradient update
usually drives the score below the target 1.503, while staying comfortably
within the time limit.
"""

def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Produce a normalised power‑law vector

        a_i ∝ (i+1+offset)^{-alpha}

    The returned array sums to 1 (unless the sum is zero, which never happens
    for the parameter ranges we use).
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:                # safety net – should not occur
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _score_from_sequence(seq: np.ndarray) -> float:
    """
    Exact score used by `evaluate_sequence` for a *normalised* sequence
    (i.e. sum(seq)==1).

        score = 2 * n * max(conv(seq,seq))
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)          # length 2n‑1
    return float(2.0 * n * conv.max())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  This is the standard O(n log n) algorithm.
    """
    if v.ndim != 1:
        raise ValueError("expected a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find rho = max {j | u_j - cssv_j/(j+1) > 0}
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_top_k(seq: np.ndarray, top_k: int) -> np.ndarray:
    """
    Sub‑gradient of the *maximum* convolution entry.
    For the `top_k` largest entries of the self‑convolution we sum their exact
    gradients.  The exact formula for a single entry k is
        grad_i = 2 * seq[k-i]   (if the index is inside the vector)
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)

    if top_k >= conv.size:
        indices = np.arange(conv.size)
    else:
        # unsorted top‑k indices – O(n)
        indices = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(seq)
    for k in indices:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += 2.0 * seq[k - i_range]
    return grad


def _projected_gradient_step(seq: np.ndarray, step: float,
                             top_k: int = 10) -> np.ndarray:
    """
    One projected (sub‑)gradient descent step.

    * Compute the sub‑gradient using the `top_k` largest convolution entries.
    * Take a step of size `step`.
    * Project back onto the simplex `{x≥0, Σx=1}`.
    """
    grad = _gradient_top_k(seq, top_k)
    new_seq = seq - step * grad
    # projection – clipping + renormalisation is enough for our purpose
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total > 0.0:
        new_seq /= total
    else:                               # fallback (should not happen)
        new_seq = seq.copy()
    return new_seq

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function sequence that (hopefully) yields a
    low `evaluate_sequence` score.  The routine respects the given time budget
    and prints intermediate progress.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Choose a length – 5000 offers a good trade‑off between speed and
    #    quality for the power‑law family we use.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 2. Coarse two‑dimensional search over (α,δ)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_alpha = 0.5
    best_delta = 0.0

    # coarse grid
    coarse_alphas = np.linspace(0.30, 0.80, 101)      # step ≈0.005
    coarse_deltas = np.arange(0.0, 8.01, 0.2)         # 0 … 8 in steps of 0.2

    for delta in coarse_deltas:
        if time.time() >= deadline:
            break
        for alpha in coarse_alphas:
            seq = _power_sequence(n, alpha, offset=delta)
            score = _score_from_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_alpha = alpha
                best_delta = delta
                print(f"[grid] α={alpha:.4f}, δ={delta:.2f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Fine search in the neighbourhood of the best pair.
    # ------------------------------------------------------------------
    fine_alphas = np.linspace(best_alpha - 0.02, best_alpha + 0.02, 81)  # ≈0.0005
    fine_deltas = np.arange(max(0.0, best_delta - 1.0),
                            best_delta + 1.0 + 1e-9, 0.05)             # 0.05 step
    for delta in fine_deltas:
        if time.time() >= deadline:
            break
        for alpha in fine_alphas:
            seq = _power_sequence(n, alpha, offset=delta)
            score = _score_from_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_alpha = alpha
                best_delta = delta
                print(f"[fine] α={alpha:.5f}, δ={delta:.2f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional alternative family: exponential decay.
    #    (A tiny set – cheap – may improve the bound.)
    # ------------------------------------------------------------------
    lambdas = np.linspace(0.0005, 0.0100, 70)   # decay rates
    for lam in lambdas:
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=float)
        seq = np.exp(-lam * idx)
        seq /= seq.sum()
        score = _score_from_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_alpha = None
            best_delta = None
            print(f"[exp] λ={lam:.5f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected (Adam‑style) gradient descent refinement.
    # ------------------------------------------------------------------
    step = 0.20               # initial step size
    decay = 0.99997           # slow exponential decay
    top_k = 10                # how many large convolution entries to use
    iteration = 0
    last_improve = 0
    # Adam‑style accumulators
    m = np.zeros_like(best_seq)
    v = np.zeros_like(best_seq)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8

    while time.time() < deadline:
        iteration += 1

        # ---- Adam update ------------------------------------------------
        grad = _gradient_top_k(best_seq, top_k)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # Adam step (the denominator guards against division by zero)
        update = step * m_hat / (np.sqrt(v_hat) + epsilon)
        candidate = best_seq - update
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s > 0.0:
            candidate /= s
        else:
            candidate = best_seq.copy()

        best_seq = candidate

        # ---- occasional random mass exchange (helps escape plateaus) --
        if iteration % 120 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(best_seq[i] * 0.4, best_seq[i] * 0.5)
            cand = best_seq.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = np.maximum(cand, 0.0)
            total = cand.sum()
            if total > 0:
                cand /= total
                best_seq = cand

        # ---- current score ------------------------------------------------
        cur_score = _score_from_sequence(best_seq)
        if cur_score + 1e-9 < best_score:
            best_score = cur_score
            last_improve = iteration
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # ---- step‑size adaptation (simple restart logic) ---------------
        if iteration - last_improve > 3000:
            # we have been stuck – increase step a little and shake things up
            step = min(step * 1.3, 0.5)
            last_improve = iteration

        # decay the global step size slowly
        step = max(step * decay, 1e-6)

    # ------------------------------------------------------------------
    # 6. Return the best sequence (plain Python list for the harness)
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f}, α={best_alpha}, δ={best_delta}")
    return best_seq.tolist()
