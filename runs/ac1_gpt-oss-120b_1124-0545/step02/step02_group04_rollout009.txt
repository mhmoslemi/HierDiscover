def _smooth_max_and_gradient(a, alpha, target_sum):
    n = a.size
    # compute conv (full length 2n-1)
    nfft = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(Af * Af, n=nfft)[:2*n-1]
    # compute softmax
    max_c = conv.max()
    diff = conv - max_c
    exp_terms = np.exp(alpha * diff)
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp
    # compute gradient via convolution
    a_rev = a[::-1]
    # convolution length m = n + (2n-1) -1 = 3n-2
    mfft = 1 << ((3*n - 2).bit_length())
    A_rev = np.fft.rfft(a_rev, n=mfft)
    W = np.fft.rfft(w, n=mfft)
    conv_rev = np.fft.irfft(A_rev * W, n=mfft)[:3*n-2]
    start = n-1
    grad = 2 * conv_rev[start:start+n]
    L = max_c + np.log(sum_exp) / alpha
    return L, grad, conv

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The evaluation reduces to `max(conv(a,a))` when the ℓ₁‑norm is fixed to
√(2 n).  The algorithm uses a smoothed “soft‑max” of the convolution to obtain a
differentiable surrogate of the objective.  At each iteration we

1. compute the full self‑convolution by FFT,
2. form soft‑max weights `w` (≈ the arg‑max indicator for large α),
3. evaluate the gradient   ∇L = 2·(a_rev * w)[n‑1:n‑1+n]   (cross‑correlation),
4. take a projected gradient step onto the simplex
   {x ≥ 0, Σ x = √(2 n)},
5. accept the step if it improves the true objective, otherwise accept
   with a decreasing temperature (simulated‑annealing).

The schedule for the soft‑max temperature α, the learning rate and the SA
temperature is exponential in the elapsed fraction of the time budget.
The routine performs many short restarts (a few seconds each) with a small
random perturbation of the current best solution, which helps to escape
local minima.  The whole search runs for at most the supplied time budget
(≤ 1000 s) and returns the best sequence found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Projects vector `v` onto the simplex {x ≥ 0, Σ x = s}
    (Wang & Carreira‑Perpiñán 2013, O(n log n)).
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # s = 0 edge‑case
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length `nfft`."""
    Af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(Af * Af, n=nfft)
    # keep only the genuine linear part (length 2·len(a)‑1)
    return conv[: 2 * a.size - 1]


def _max_convolution(a: np.ndarray, nfft: int) -> float:
    """Maximum entry of the linear self‑convolution of `a`."""
    conv = _conv_fft(a, nfft)
    return float(conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimizes
    the evaluation function.  The routine respects the 1000 s time budget
    and returns the best sequence found.
    """
    # --------------------------------------------------------------
    # initialise RNGs
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # problem size and initial point
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500                      # modest default
        a0 = np.random.exponential(1.0, size=n)

    target_sum = math.sqrt(2.0 * n)   # Σ a = √(2 n)

    # project initial vector onto the simplex (ensures feasibility)
    a0 = _simplex_projection(a0, target_sum)

    # pre‑compute FFT sizes (next powers of two)
    nfft = 1 << ((2 * n - 1).bit_length())       # for a*a
    mfft = 1 << ((3 * n - 2).bit_length())       # for a_rev * w

    # --------------------------------------------------------------
    # keep the globally best solution
    best_seq = a0.copy()
    best_score = _max_convolution(best_seq, nfft)
    print(f"[init] n={n}, initial max‑conv={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # 5 s safety margin
    overall_iter = 0

    # ------------------------------------------------------------------
    # hyper‑parameter schedules (exponential in the elapsed fraction)
    def schedule(frac, start, end):
        """exponential interpolation between `start` and `end`."""
        return start * ((end / start) ** frac)

    # ------------------------------------------------------------------
    # main loop – many short restarts
    while time.time() < deadline:
        # start each restart from the current global best plus a tiny jitter
        a = best_seq + np.random.normal(scale=target_sum * 0.001, size=n)
        a = np.maximum(a, 0.0)
        a = _simplex_projection(a, target_sum)

        # current true score
        cur_score = _max_convolution(a, nfft)

        # inner loop for this restart (≈ 5 s)
        restart_start = time.time()
        while time.time() < deadline and (time.time() - restart_start) < 5.0:
            elapsed = time.time() - start_time
            frac = elapsed / (deadline - start_time)

            # schedules
            alpha = schedule(frac, 5.0, 25.0)          # soft‑max sharpness
            lr    = schedule(frac, 0.5, 0.05)         # learning rate
            temp  = schedule(frac, 0.1, 0.001)        # SA temperature

            # ------------------------------------------------------
            # 1) convolution of the current iterate
            conv_cur = _conv_fft(a, nfft)
            max_cur = conv_cur.max()

            # 2) soft‑max weights (smooth approximation of the arg‑max)
            diff = conv_cur - max_cur               # ≤ 0, zero at the max entry
            exp_terms = np.exp(alpha * diff)        # safe because diff ≤ 0
            w = exp_terms / exp_terms.sum()         # Σ w = 1, concentrated on the peak

            # 3) gradient of the smooth objective:
            #    grad = 2·(a_rev * w)[n‑1 : n‑1+n]
            a_rev = a[::-1]
            A_rev = np.fft.rfft(a_rev, n=mfft)
            W = np.fft.rfft(w, n=mfft)
            corr_full = np.fft.irfft(A_rev * W, n=mfft)[: 3 * n - 2]
            start_idx = n - 1
            grad = 2.0 * corr_full[start_idx:start_idx + n]

            # 4) projected gradient step
            a_prop = a - lr * grad
            a_prop = _simplex_projection(a_prop, target_sum)

            # 5) evaluate the true (non‑smoothed) objective
            new_score = _max_convolution(a_prop, nfft)

            # ------------------------------------------------------
            # acceptance rule (deterministic improvement + SA)
            if new_score < cur_score:
                accept = True
            else:
                prob = math.exp(-(new_score - cur_score) / (temp + 1e-12))
                accept = random.random() < prob

            if accept:
                a = a_prop
                cur_score = new_score
                # keep the best seen so far
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[{overall_iter:7d}] new best score: {best_score:.6f} "
                          f"(time {elapsed:.1f}s)")

            overall_iter += 1

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list
    return best_seq.tolist()
