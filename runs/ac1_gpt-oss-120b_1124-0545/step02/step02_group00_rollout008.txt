def exp_gradient_move(a, step_scale):
    # compute gradient g_i = 2 * a_{kmax-i}
    n = len(a)
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))
    lo = max(0, k_max - (n-1))
    hi = min(n-1, k_max)
    idx = np.arange(lo, hi+1)
    # counterpart indices
    g = np.zeros_like(a)
    g[idx] = 2 * a[k_max - idx]
    # multiplicative update:
    a_new = a * np.exp(-step_scale * g)
    # normalize to simplex:
    a_new = a_new / a_new.sum()
    # ensure nonnegative -> expo positive
    return a_new

A stochastic optimizer for the step‑function problem.

The objective (scale‑invariant) is

    score(a) = 2·n·max(conv(a,a)) / (sum(a)²)

We work on the simplex Σa_i = 1, a_i ≥ 0.  The algorithm consists of

1.  A rich set of analytically‑motivated initial candidates
    (uniform, arcsine‑like, random power‑law families and the
    previously best known construction if it is available).

2.  An iterative search that mixes several move types:
    – projected sub‑gradient descent on the *active* convolution entry,
    – exponentiated‑gradient (mirror descent) on the same sub‑gradient,
    – a smoothed‑max gradient (softmax over all convolution entries)
      whose “temperature’’ β is annealed,
    – cheap random mass‑transfer moves for exploration.

   After each move the vector is symmetrised (the optimal solution is
   known to be symmetric), projected back onto the probability simplex
   and optionally accepted with a Metropolis rule.

3.  Simple annealing schedules for the step size, the Metropolis
    temperature and the soft‑max parameter β, together with occasional
    jitter‑restarts when progress stalls.

The routine respects the 1000 s budget, prints progress and returns the
best sequence it discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level – no closures)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact (scale‑invariant) objective for a non‑negative array a."""
    s = a.sum()
    if s < 1e-12:                     # should never happen
        return np.inf
    b = np.convolve(a, a)             # linear (zero‑padded) convolution
    max_b = b.max()
    return float(2.0 * len(a) * max_b / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending order
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force left‑right symmetry and renormalise onto the simplex."""
    a = (a + a[::-1]) * 0.5
    return project_simplex(a)


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return symmetrize(a_new)


def subgradient_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Projected sub‑gradient step for the current maximal convolution entry.
    """
    n = a.shape[0]
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)

    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k_max - idx]          # sub‑gradient of max(conv)

    a_new = a - step_scale * g
    a_new = project_simplex(a_new)
    return symmetrize(a_new)


def exp_gradient_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Mirror‑descent / exponentiated‑gradient step for the same sub‑gradient.
    """
    n = a.shape[0]
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)

    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k_max - idx]

    # multiplicative update, then re‑normalise
    a_new = a * np.exp(-step_scale * g)
    a_new = project_simplex(a_new)
    return symmetrize(a_new)


def smooth_gradient_move(a: np.ndarray, beta: float, step_scale: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Uses the cross‑correlation a ⋆ w, where w_k ∝ exp(β·B_k).
    """
    n = a.shape[0]

    # 1) convolution a*a
    B = np.convolve(a, a)

    # 2) soft‑max weights (stable computation)
    maxB = B.max()
    w = np.exp(beta * (B - maxB))      # all ≤ 1, avoids overflow
    Z = w.sum()
    if Z == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / Z                       # now Σ w = 1

    # 3) correlation a ⋆ w  (lag k = i)
    # np.correlate returns lags from -(len(w)-1) .. (len(a)-1)
    corr = np.correlate(a, w, mode='full')
    offset = len(w) - 1
    g = 2.0 * corr[offset: offset + n]   # gradient of soft‑max w.r.t. a

    # 4) step and projection
    a_new = a - step_scale * g
    a_new = project_simplex(a_new)
    return symmetrize(a_new)


def initialise_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a moderate pool of analytically motivated candidates,
    plus a random parametric family.  All candidates are normalised to
    sum = 1.
    """
    candidates = []

    # 0. Previously known best construction, if present
    if 'height_sequence_1' in globals():
        h = np.array(globals()['height_sequence_1'], dtype=float)
        if h.size == n:
            candidates.append(project_simplex(h))

    # 1. Uniform
    candidates.append(np.ones(n) / n)

    # 2. Classic arcsine‑like (inverse‑sqrt product with 0.5 offset)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    candidates.append(a / a.sum())

    # 3. Pure product without 0.5 offset (parameter 0.0)
    a = 1.0 / np.sqrt((i + 1) * (n - i))
    candidates.append(a / a.sum())

    # 4. Mixed inverse‑sqrt sums with several exponents
    for p in (0.3, 0.5, 0.7, 1.0):
        a = (i + 1) ** (-p) + (n - i) ** (-p)
        candidates.append(a / a.sum())

    # 5. Randomised power‑law / product families (20 samples)
    rng = np.random.default_rng(12345)
    for _ in range(20):
        p = rng.uniform(0.2, 1.2)                # exponent
        alpha = rng.uniform(0.0, 2.0)            # offset
        a = 1.0 / ((i + alpha + 1) * (n - i + alpha)) ** p
        # avoid infinities – already safe because denominator >0
        a = a / a.sum()
        candidates.append(a)

    # 6. Symmetrised random Dirichlet draws (10 samples)
    for _ in range(10):
        raw = rng.random(n)
        raw = (raw + raw[::-1]) * 0.5           # enforce symmetry early
        a = project_simplex(raw)
        candidates.append(a)

    # ensure symmetry of all candidates before returning
    return [symmetrize(c) for c in candidates]


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Stochastic optimisation of the step‑function heights.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds; the routine stops shortly before the limit.
    kwargs : dict
        Optional keyword ``n`` specifying the length of the sequence
        (default 2000).

    Returns
    -------
    list[float]
        The best non‑negative sequence found (sum does not have to be 1;
        the evaluator is scale‑invariant).
    """
    np.random.seed(seed)
    start = time.time()
    # leave a 0.5 s safety cushion before the hard deadline
    deadline = start + budget_s - 0.5

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1 – initialise from a small analytical / random pool
    # ------------------------------------------------------------------
    cand_pool = initialise_candidates(n)
    best_seq = None
    best_score = np.inf

    for idx, a in enumerate(cand_pool):
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        print(f"[init] cand {idx:02d} → score {sc:.6f}")

    a = best_seq.copy()
    print(f"[init] chosen start → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – optimisation loop
    # ------------------------------------------------------------------
    # annealing / step‑size schedules
    init_temp = 1.0
    final_temp = 1e-3

    init_step = 0.7          # fairly aggressive at the beginning
    final_step = 1e-5

    beta_start = 0.5
    beta_end = 200.0

    iteration = 0
    last_improve = start

    while time.time() < deadline:
        now = time.time()
        elapsed = now - start
        # progress in [0,1]
        progress = min(1.0, elapsed / (budget_s - 5.0))

        # linear schedules for temperature and step size
        temp = init_temp * (1.0 - progress) + final_temp * progress
        step_scale = init_step * (1.0 - progress) + final_step * progress

        # exponential schedule for β (soft‑max temperature)
        beta = beta_start * (beta_end / beta_start) ** progress

        # ------------------------------------------------------------------
        #  choose move type
        # ------------------------------------------------------------------
        r = np.random.rand()
        if r < 0.65:
            cand = subgradient_move(a, step_scale)
        elif r < 0.85:
            cand = exp_gradient_move(a, step_scale)
        elif r < 0.95:
            # occasional smooth‑max step (costlier, so rarer)
            cand = smooth_gradient_move(a, beta, step_scale * 0.8)
        else:
            # cheap random transfer
            cand = random_move(a, step_scale * 0.5)

        cand_score = compute_score(cand)

        # Metropolis‑style acceptance (allows uphill moves)
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            # probability decays with temperature
            prob = np.exp(-(cand_score - best_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = now
                print(f"[{iteration:07d}] new best = {best_score:.6f}  t={elapsed:.1f}s")

        # ------------------------------------------------------------------
        #  stagnation handling – jitter around current best if stuck
        # ------------------------------------------------------------------
        if (now - last_improve) > 25.0 and iteration % 5000 == 0:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = symmetrize(a)
            print(f"[{iteration:07d}] restart due to stagnation (t={elapsed:.1f}s)")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    # return a plain Python list; scaling is irrelevant for the evaluator
    return best_seq.tolist()
