"""
Projected Adam with smooth max approximation for minimizing 2·n·max(conv(a,a)).
The algorithm works on the probability simplex (∑a=1, a_i≥0). It uses a
log‑sum‑exp surrogate for the max, computes its gradient via FFT and
updates the vector with Adam. Occasional random “mass‑transfer” moves
and a stagnation‑restart heuristic explore the landscape.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of ``u`` and ``v`` using FFT of length ``size``."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a random small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # renormalise to keep the simplex
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a_new) / n
    else:
        a_new /= s
    return a_new

    The algorithm uses a smooth‑max surrogate and Adam on the simplex,
    together with random mass‑transfer moves and a stagnation‑restart
    heuristic.
    """
    np.random.seed(seed)
    start = time.time()
    # tiny safety margin to avoid overshooting the budget
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size (can be overridden via kwargs)
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # initialise a feasible point on the simplex
    # --------------------------------------------------------------
    try:
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            seq0 = np.asarray(seq0, dtype=float)
            if seq0.size != n:
                # linear interpolation to required length
                x_old = np.linspace(0.0, 1.0, seq0.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a = np.interp(x_new, x_old, seq0)
            else:
                a = seq0.copy()
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.random.rand(n)
                a /= a.sum()
            else:
                a /= s
        else:
            raise KeyError
    except Exception:
        a = np.random.rand(n)
        a /= a.sum()

    # --------------------------------------------------------------
    # pre‑compute FFT lengths (powers of two) – they stay constant
    # --------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a * a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * weight vector (len 2n‑1)

    # --------------------------------------------------------------
    # Adam bookkeeping
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_size = 0.03                # base learning rate for Adam
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    # --------------------------------------------------------------
    # smooth‑max surrogate parameters
    # --------------------------------------------------------------
    lam = 0.5                        # start λ
    lam_max = 20.0
    lam_factor = 1.001               # increase λ slowly each iteration

    # --------------------------------------------------------------
    # exploration / restart settings
    # --------------------------------------------------------------
    rand_prob = 0.08                 # probability of a random mass‑transfer move
    rand_scale = 0.04                # size of that transfer (relative to source mass)
    stagnation_time = 45.0           # seconds without improvement → restart
    last_improve = start

    # --------------------------------------------------------------
    # bookkeeping for the best sequence seen so far
    # --------------------------------------------------------------
    conv_init = _conv_fft(a, a, size1)
    max_b = conv_init.max()
    best_score = 2.0 * n * max_b
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    it = 0

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---------- 1. evaluate current objective ----------
        conv_full = _conv_fft(a, a, size1)        # length 2n‑1
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ---------- 2. decide which move to take ----------
        if np.random.rand() < rand_prob:
            # cheap exploratory mass‑transfer
            a = _random_transfer(a, rand_scale)
        else:
            # ---- smooth‑max gradient step (Adam) ----
            # build soft‑max weights from the convolution values
            shifted = conv_full - max_b               # ≤ 0
            w = np.exp(lam * shifted)                # weights ≤ 1
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.ones_like(w) / w.size
            else:
                p = w / w_sum

            # gradient = 2 * (a_rev * p) (central n entries)
            a_rev = a[::-1]
            grad_full = _conv_fft(a_rev, p, size2)   # length 3n‑2
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # Adam update (gradient descent)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - step_size * m_hat / (np.sqrt(v_hat) + eps)

            # project back onto simplex (non‑negative + Σ=1)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.ones_like(a) / n
            else:
                a /= s

        # ---------- 3. update smoothing parameter ----------
        lam = min(lam_max, lam * lam_factor)

        # ---------- 4. stagnation restart ----------
        if time.time() - last_improve > stagnation_time:
            # perturb the best sequence slightly and reset Adam state
            a = best_seq + 0.02 * np.random.rand(n)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.ones_like(a) / n
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = time.time()
            print(f"[{it:7d}] restart due to stagnation (perturb best)")

        it += 1

    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
