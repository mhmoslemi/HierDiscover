def propose_candidate(seed=..., budget_s=1000, n=6000,...):

    Set np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Initialize z: random normal shape n.
    z = np.random.randn(n) * 0.01  # small
    # best tracking
    best_seq = None
    best_score = float('inf')
    # Adam params
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0

    # schedule for beta_temp (smoothness)
    beta_temp = 5.0

    # precompute FFT size for conv length (len a * a) = 2n-1
    size1 = _next_pow2(2 * n - 1)
    # For gradient conv of a_rev with w (weight length 2n-1) => conv length n+(2n-1)-1 = 3n-2
    size2 = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        # compute a via softmax
        max_z = np.max(z)  # for numerical stability
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # compute convolution a * a
        conv_full = _max_convolution(a, size1)  # shape 2n-1

        # compute smooth max using softmax of conv*beta_temp
        conv_scaled = conv_full * beta_temp
        # numerical stable softmax
        max_c = np.max(conv_scaled)
        w_exp = np.exp(conv_scaled - max_c)
        w = w_exp / w_exp.sum()   # shape same as conv_full

        # evaluate objective exact for tracking
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{t}] new best {best_score:.6f}")

        # compute gradient wrt a: grad_a = 2 * conv(a_rev, w) central part
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, w, size2)  # length n + (2n-1) -1 = 3n-2
        # central part (indices n-1 to n-1+n-1)
        grad_a = 2.0 * grad_full[n-1:n-1+n]   # shape n

        # gradient wrt z
        dot_aw = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot_aw)

        # Adam update

"""
Propose a low‑value step‑function using a smooth‑max Adam optimiser.

The objective is   2·n·max(conv(a,a)) / (Σa)² .
Because scaling cancels, we keep a on the probability simplex (Σa=1) and
optimise the smoothed maximum of the convolution.
The vector a is parametrised as a soft‑max of an unconstrained vector *z*,
ensuring non‑negativity and the simplex constraint.  At each iteration we

1. evaluate the full convolution a*a (FFT, O(n log n));
2. compute a soft‑max over β·conv to obtain a smooth approximation of the
   max (weights w_r ≈ 1 for the largest entries, 0 elsewhere);
3. obtain the exact gradient of the smooth max w.r.t. a:
        ∂/∂a  (log‑sum‑exp) = 2·conv(a_rev, w)   (central n entries);
4. back‑propagate through the soft‑max to get ∂/∂z;
5. perform an Adam update on z.

A two‑phase scheme is used:
* a coarse optimisation on a small problem (n≈2 k) runs for ~40 % of the time;
* the best coarse solution is linearly interpolated to the target size and
  refined for the remaining budget.

Occasional random perturbations and a simple stagnation‑restart keep the
search from being trapped in flat plateaus.  The best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT of length ``size`` (must be ≥ len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _max_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a * a (full convolution) using a pre‑computed FFT size."""
    return _conv_fft(a, a, size)


# ----------------------------------------------------------------------
# Optimisation core (single‑resolution)
# ----------------------------------------------------------------------


def _optimize_one_size(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Run the Adam‑based smooth‑max optimiser for a given length ``n``
    for at most ``time_limit`` seconds.
    Returns the best probability vector found and its (exact) score.
    """
    start = time.time()
    deadline = start + time_limit

    # ----- initialise a (probability vector) -------------------------
    if init_a is not None:
        a0 = np.asarray(init_a, dtype=float)
        if a0.size != n:
            # linear interpolation if the size does not match
            x_old = np.linspace(0.0, 1.0, a0.size, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            a0 = np.interp(x_new, x_old, a0)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0.0:
            a0 = np.random.rand(n)
        a0 /= a0.sum()
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # parameter vector for soft‑max, small random perturbation around log(a0)
    eps = 1e-12
    z = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2                     # initial learning‑rate for z

    # FFT sizes (stay constant)
    size1 = _next_pow2(2 * n - 1)        # for a*a
    size2 = _next_pow2(3 * n - 2)        # for a_rev * weight

    best_score = float("inf")
    best_seq = None
    it = 0
    last_improve = start

    # smoothing temperature for the soft‑max over the convolution
    beta_temp = 5.0
    beta_max = 250.0

    while time.time() < deadline:
        it += 1

        # ----- a = softmax(z) (on the simplex) -----------------------
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)            # shift for numeric stability
        a = exp_z / exp_z.sum()               # Σa = 1, a ≥ 0

        # ----- convolution a*a ---------------------------------------
        conv_full = _max_convolution(a, size1)            # length 2n‑1
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max                     # exact objective (Σa=1)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # ----- smooth‑max weights ------------------------------------
        #   w_r = exp(β·conv_r) / Σ exp(β·conv)
        conv_scaled = conv_full * beta_temp
        max_c = np.max(conv_scaled)
        w_exp = np.exp(conv_scaled - max_c)
        w = w_exp / w_exp.sum()                     # Σw = 1

        # ----- gradient w.r.t. a (exact for the smooth max) ---------
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, w, size2)      # length = 3n‑2
        # central n entries correspond to ∂/∂a_k
        grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- back‑prop through the soft‑max (∂/∂z) -----------------
        dot_aw = np.dot(a, grad_a)                  # scalar
        grad_z = a * (grad_a - dot_aw)              # shape (n,)

        # ----- Adam update on z --------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules ------------------------------------------------
        # increase the smooth‑max temperature slowly
        if beta_temp < beta_max:
            beta_temp = min(beta_max, beta_temp * 1.001)

        # decay learning‑rate a little every 10 k iterations
        if it % 10000 == 0:
            lr *= 0.95

        # ----- occasional random perturbation -------------------------
        if np.random.rand() < 0.02:                  # 2 % chance
            z += 0.1 * np.random.randn(n)

        # ----- simple stagnation restart -------------------------------
        if (time.time() - last_improve) > 30.0:      # 30 s without improvement
            # re‑initialise from the current best with a small jitter
            if best_seq is not None:
                a_jitter = best_seq.copy()
                a_jitter += 0.01 * np.random.rand(n)
                a_jitter = np.clip(a_jitter, 0.0, None)
                a_jitter /= a_jitter.sum()
                z = np.log(a_jitter + eps)
                last_improve = time.time()
                print(f"[{it:7d}] restart (stagnation)")
            else:
                # pure random restart if we have no good candidate yet
                a_rand = np.random.rand(n)
                a_rand /= a_rand.sum()
                z = np.log(a_rand + eps)
                last_improve = time.time()
                print(f"[{it:7d}] restart (random)")

    total = time.time() - start
    print(f"[phase done] n={n}, best_score={best_score:.6f}, it={it}, time={total:.1f}s")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises
        2·n·max(conv(a,a)) / (Σa)²
    within the allotted time budget.  The routine performs a
    coarse optimisation on a smaller grid and then refines the result
    on the requested size (default 6000).  The best sequence found is
    returned as a list of Python floats.
    """
    np.random.seed(seed)
    start_total = time.time()
    deadline_total = start_total + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Target problem size
    # --------------------------------------------------------------
    n_target = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # Coarse phase (≈40 % of the time)
    # --------------------------------------------------------------
    # Choose a modest size for the fast exploration phase.
    n_coarse = min(2048, n_target)   # at most 2048, never larger than target

    time_coarse = 0.4 * budget_s
    # If the coarse size equals the target, we just run one phase.
    if n_coarse == n_target:
        best_seq, best_score = _optimize_one_size(
            n=n_target,
            time_limit=budget_s,
            init_a=None,
        )
        return best_seq.tolist()

    # Use an optional warm start from a previously supplied sequence.
    init_seq = globals().get("height_sequence_1", None)

    # Run the coarse optimiser.
    best_coarse, _ = _optimize_one_size(
        n=n_coarse,
        time_limit=time_coarse,
        init_a=init_seq,
    )
    # Upsample to the target resolution (linear interpolation).
    x_old = np.linspace(0.0, 1.0, n_coarse, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_target, endpoint=False)
    upsampled = np.interp(x_new, x_old, best_coarse)
    upsampled = np.clip(upsampled, 0.0, None)
    if upsampled.sum() == 0.0:
        upsampled = np.random.rand(n_target)
    upsampled /= upsampled.sum()

    # --------------------------------------------------------------
    # Fine‑tuning phase (remaining budget)
    # --------------------------------------------------------------
    time_left = deadline_total - time.time()
    if time_left < 5.0:                     # not enough time → fallback
        return upsampled.tolist()

    best_fine, final_score = _optimize_one_size(
        n=n_target,
        time_limit=time_left,
        init_a=upsampled,
    )

    total_elapsed = time.time() - start_total
    print(f"[done] total time={total_elapsed:.1f}s, final score={final_score:.6f}")
    # Return the best sequence discovered (the fine‑tuned one).
    return best_fine.tolist()
