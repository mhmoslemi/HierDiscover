def _optimise_one_seed(a0, n, deadline):
    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    t=0
    beta1=0.9; beta2=0.999; eps=1e-12
    beta_start=1.0
    beta_mul=1.015
    lr0=0.8
    best_seq_local = a0.copy()
    best_val_local = 2.0*n*np.max(_conv_fft(a0, a0))
    max_iters = 15000
    for it in range(max_iters):
        if time.time()>deadline: break
        t+=1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2.0*n*cur_max
        if cur_val < best_val_local:
            best_val_local = cur_val
            best_seq_local = a.copy()
        # smoothed max
        beta = beta_start * (beta_mul**it)
        shift = cur_max
        w = np.exp(beta*(conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w /= w_sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0*full[n-1:n-1+n]
        # softmax grad
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam
        m = beta1*m + (1-beta1)*grad_v
        v2 = beta2*v2 + (1-beta2)*(grad_v**2)
        m_hat = m/(1-beta1**t)
        v2_hat = v2/(1-beta2**t)
        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat)+eps)
    return best_seq_local, best_val_local

"""
Propose a non‑negative step‑function (list of heights) that minimises
evaluate_sequence().  The algorithm

1. builds a rich collection of seed vectors (geometric, power‑law,
   geometric×power, symmetric geometric, sine‑window, beta‑distribution,
   random, and mixtures);
2. runs an Adam optimiser on the log‑softmax parametrisation of each seed.
   A smoothed max (log‑sum‑exp) loss is used, with a slowly increasing
   smoothing factor β; the optimiser works on the simplex automatically.
3. keeps the best sequence found over all runs;
4. finally performs a few hard‑max sub‑gradient steps (β→∞) to polish
   the incumbent;
5. respects the supplied time budget (default 1000 s) and returns the
   best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution and simplex utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; the result always sums to one."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric progression a_i = r**i, normalised to sum 1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric × power a_i = r**i / (i+1)^p, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (i - 1)) / (i ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric shape (mirrored)."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sine_window(n: int) -> np.ndarray:
    """Pure sine window (positive half‑sine)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑distribution shape on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a simplex vector (sum(a)=1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _optimise_one_seed(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a single seed.
    Returns the best vector (on the simplex) and its objective value.
    """
    v = np.log(a0 + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    t = 0

    # Adam hyper‑parameters
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # β‑schedule for the smoothed max
    beta_start = 1.0
    beta_mul = 1.015

    best_a = a0.copy()
    best_val = _objective(best_a)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex

        # true objective (fast version)
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[seed‑opt] n={n:5d} it={it:5d} new best={cur_val:.6f}")

        # ----- smoothed‑max weight vector -----
        beta = beta_start * (beta_mul ** it)
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient w.r.t. a (smooth loss) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]        # central n entries

        # ----- convert to gradient w.r.t. v (softmax) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_a, best_val


def _hardmax_refinement(a_start: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Final polish: sub‑gradient steps with the true (hard) max.
    Simple SGD with a decreasing step size.
    """
    v = np.log(a_start + 1e-30)
    best_a = a_start.copy()
    best_val = _objective(best_a)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[hard‑max] n={n:5d} it={it:4d} improved={cur_val:.6f}")

        # indicator weights for the active constraints
        eps_eq = 1e-15
        active = np.where(conv >= cur_max - eps_eq)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # simple decreasing learning rate
        lr = 0.5 / np.sqrt(it + 1)
        v = v - lr * grad_v

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed‑generation
# ----------------------------------------------------------------------


def _build_seeds(n: int) -> list[np.ndarray]:
    """Create a diverse collection of initial probability vectors."""
    seeds = []

    # 1) Geometric families
    for c in np.linspace(0.1, 5.0, 15):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_geom(n, r))

    # 2) Pure power‑law families
    for p in np.linspace(0.0, 3.0, 15):
        seeds.append(_power(n, p))

    # 3) Geometric × power families
    for c in np.linspace(0.2, 4.0, 8):
        r = 1.0 - c / n
        if r <= 0:
            continue
        for p in np.linspace(0.0, 3.0, 8):
            seeds.append(_geom_power(n, r, p))

    # 4) Symmetric geometric
    for c in np.linspace(0.2, 4.0, 8):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_sym_geom(n, r))

    # 5) Smooth sine window
    seeds.append(_sine_window(n))

    # 6) Beta‑distribution shapes (skewed left/right)
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        for beta in [0.5, 1.0, 2.0, 3.0, 5.0]:
            seeds.append(_beta_shape(n, alpha, beta))

    # 7) A few random softmax vectors
    for _ in range(6):
        v_rand = np.random.randn(n)
        seeds.append(_softmax(v_rand))

    # 8) Random mixtures of two previously generated seeds
    if len(seeds) >= 2:
        for _ in range(5):
            i, j = np.random.choice(len(seeds), size=2, replace=False)
            w = np.random.rand()
            mix = w * seeds[i] + (1.0 - w) * seeds[j]
            mix /= mix.sum()
            seeds.append(mix)

    # 9) Include the last known good construction if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            a_prev = np.asarray(prev, dtype=np.float64)
            if a_prev.ndim == 1:
                if a_prev.shape != (n,):
                    # truncate or pad to the required length
                    adj = np.zeros(n, dtype=np.float64)
                    m = min(a_prev.shape[0], n)
                    adj[:m] = a_prev[:m]
                    a_prev = adj
                a_prev = a_prev / a_prev.sum()
                seeds.append(a_prev)
        except Exception:
            pass

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects a time budget (default
    1000 s) and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # try a few lengths – longer vectors give more freedom but cost more
    lengths = [4000, 8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _build_seeds(n)

        # Quick scan of the raw seeds (no optimisation)
        for a in seeds:
            val = _objective(a)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = a.copy()
                print(f"[seed‑scan] len={n:5d} val={val:.6f}")

        # Optimise each seed independently (Adam)
        for idx, a0 in enumerate(seeds):
            if time.time() > deadline:
                break
            best_a, best_val = _optimise_one_seed(a0, n, deadline)
            if best_a is not None and best_val < global_best_val:
                global_best_val = best_val
                global_best_seq = best_a.copy()
                print(f"[global] new best from seed {idx} len={n} val={best_val:.6f}")

        # Final hard‑max polishing on the current global best of this length
        if global_best_seq is not None:
            refined_a, refined_val = _hardmax_refinement(
                global_best_seq, n, deadline
            )
            if refined_a is not None and refined_val < global_best_val:
                global_best_val = refined_val
                global_best_seq = refined_a.copy()
                print(f"[global] after hard‑max len={n} val={refined_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.full(n, 1.0 / n)
        print("[fallback] using uniform distribution")

    # Return as a plain Python list of floats
    return global_best_seq.tolist()
