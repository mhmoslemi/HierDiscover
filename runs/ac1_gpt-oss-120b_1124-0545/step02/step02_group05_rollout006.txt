def propose_candidate(...):
    import time, math, random, numpy as np, cvxpy as cp, scipy.linalg as la

    if 'height_sequence_1' in globals():
        base_seq = np.asarray(height_sequence_1, dtype=float)
        n = len(base_seq)
    else:
        n = 2500
    target_sum = math.sqrt(2.0 * n)

    # helper: normalization to target_sum
    def normalize(v):
        if v.sum() == 0:
            return np.full_like(v, target_sum / len(v))
        return v / v.sum() * target_sum

    # generate initial candidates
    candidates = []

    # uniform
    uniform = np.full(n, target_sum / n)
    candidates.append(uniform)

    # triangular
    tri = np.concatenate([np.arange(1, n//2+1), np.arange(n//2, 0, -1)])[:n]
    # can't guarantee length exact; we handle even n.
    tri = tri.astype(float)
    tri = normalize(tri)
    candidates.append(tri)

    # raised cosine (Hann)
    i = np.arange(n)
    hann = np.sin(np.pi * i / (n - 1)) ** 2
    hann = normalize(hann)
    candidates.append(hann)

    # gaussian with sigma = n/8, n/5, n/3
    for sigma in [n/8, n/5, n/3]:

        gauss = normalize(gauss)
        candidates.append(gauss)

"""
Search for a non‑negative step‑function (height sequence) that minimises

    E(a) = 2·n·max(conv(a, a)) / (∑a)²

using a fast exponentiated‑gradient optimiser on a log‑sum‑exp surrogate,
augmented by occasional Frank‑Wolfe‑style linear‑program updates.
A variety of analytically‑motivated warm‑starts (uniform, Hann, triangular,
Gaussian, exponential and the previously found best) are evaluated first,
and the best of them seeds the optimiser.

The algorithm respects a 1000 s time budget, constantly records the best
sequence found and finally returns it as a plain Python list.
"""

# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using a power‑of‑two FFT."""
    n = len(u) + len(v) - 1
    fsize = 1 << (n - 1).bit_length()          # next power of two
    fu = np.fft.rfft(u, fsize)
    fv = np.fft.rfft(v, fsize)
    conv = np.fft.irfft(fu * fv, fsize)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a."""
    return _conv_fft(a, a)


def _normalize(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm = target_sum."""
    s = v.sum()
    if s <= 0:
        return np.full_like(v, target_sum / v.size)
    return v / s * target_sum

# Linear‑program helper (only used if cvxpy is available)
try:
    import cvxpy as cp
    import scipy.linalg as sla
except Exception:          # pragma: no cover
    cp = None
    sla = None


def _lp_direction(a: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """Solve  max ∑b   s.t.  b≥0,  b≤target_sum,  conv(a,b) ≤ M .
    Returns the optimal b (or None on failure)."""
    if cp is None or sla is None:          # cvxpy not installed
        return None

    n = a.size
    # Toeplitz convolution matrix:  (2n‑1) × n
    first_col = np.concatenate([a, np.zeros(n - 1)])
    first_row = np.concatenate([a[:1], np.zeros(n - 1)])
    A = sla.toeplitz(first_col, first_row)        # dense but moderate size

    b = cp.Variable(n)
    constraints = [
        b >= 0,
        b <= target_sum,
        A @ b <= M + 1e-10
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-7, reltol=1e-6, feastol=1e-8)
    except Exception:          # pragma: no cover
        return None

    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None

    b_opt = np.asarray(b.value).ravel()
    b_opt = np.maximum(b_opt, 0.0)            # numerical safety
    return b_opt

    uniform = np.full(n, target_sum / n)
    candidates.append(uniform)

    # 2) triangular (peak in the centre)
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(float)
    candidates.append(_normalize(tri, target_sum))

    # 3) Hann (raised cosine)
    i = np.arange(n)
    hann = np.sin(np.pi * i / (n - 1)) ** 2
    candidates.append(_normalize(hann, target_sum))

    # 4) Gaussian families
    centre = (n - 1) / 2.0
    for sigma in (n / 8.0, n / 5.0, n / 3.0):
        gauss = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
        candidates.append(_normalize(gauss, target_sum))

    # 5) random exponential (already used in the reference code)
    rand_exp = np.random.exponential(scale=1.0, size=n)
    candidates.append(_normalize(rand_exp, target_sum))

    # 6) previous best, if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=float)
        candidates.append(_normalize(prev, target_sum))

    # pick the best of the warm‑starts (according to max(conv))
    best_seq = None
    best_score = np.inf
    for cand in candidates:
        conv = _conv_self(cand)
        score = float(conv.max())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    print(f"Initial best max(conv) = {best_score:.6f}")

    # --------------------------------------------------------------
    # Re‑parameterise via logits (soft‑max) to keep positivity & sum exact
    eps_small = 1e-12
    logits = np.log(best_seq + eps_small)               # shape (n,)

    # Adam moments for the logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    t_iter = 0

    # schedules for smoothing (ε) and learning‑rate
    eps_start, eps_end = 0.5, 0.001            # final smoothing smaller than before
    lr_start = target_sum * 0.001
    lr_end = target_sum * 0.00002

    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # tiny safety margin
    it = 0
    last_improve = 0

    best_overall_seq = best_seq.copy()
    best_overall_score = best_score

    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---- schedules -------------------------------------------------
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        eps = eps_start * ((eps_end / eps_start) ** frac)
        lr = lr_start * ((lr_end / lr_start) ** frac)

        # ---- current a (soft‑max) --------------------------------------
        max_log = np.max(logits)                     # for numerical stability
        exp_logits = np.exp(logits - max_log)
        a = target_sum * exp_logits / exp_logits.sum()

        # ---- evaluate current max(conv) --------------------------------
        conv = _conv_self(a)
        cur_max = float(conv.max())
        if cur_max < best_overall_score:
            best_overall_score = cur_max
            best_overall_seq = a.copy()
            last_improve = it
            print(f"[{it:6d}] New best max(conv) = {best_overall_score:.6f} "
                  f"(t={elapsed:.1f}s, eps={eps:.4f})")

        # ---- smooth‑max weights ----------------------------------------
        conv_shift = conv - conv.max()
        w_raw = np.exp(conv_shift / eps)
        w_sum = w_raw.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w_raw / w_sum                       # shape (2n‑1,)

        # ---- gradient w.r.t. a (derived from the surrogate) -----------
        # sigma_i = Σ_{k=i}^{i+n‑1} w_k
        cumsum_w = np.concatenate(([0.0], np.cumsum(w)))   # length 2n
        sigma = cumsum_w[n:2 * n] - cumsum_w[:n]            # length n

        # first term = Σ_{t=0}^{n‑1} w_{i+t} a_t  (vectorised via convolution)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)                      # length 3n‑2
        first = conv_wa[n - 1:n - 1 + n]                    # slice of length n

        grad_a = first + a * sigma                         # ∂L/∂a

        # ---- gradient w.r.t. logits (soft‑max chain rule) ------------
        s = a / target_sum                                 # current probability vector
        mean = np.dot(s, grad_a)
        grad_log = target_sum * s * (grad_a - mean)        # ∂L/∂logits

        # ---- Adam update on logits ------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        logits -= step

        # ---- occasional Frank‑Wolfe style LP step --------------------
        if it % 3000 == 0 and it > 0:          # not too frequent (≈ every few secs)
            b_dir = _lp_direction(a, cur_max, target_sum)
            if b_dir is not None:
                sum_b = b_dir.sum()
                if sum_b > 1e-12:
                    # scale to the required ℓ₁‑norm
                    g = b_dir * (target_sum / sum_b)

                    # line‑search over mixing coefficients
                    best_mix_score = cur_max
                    best_mix_a = a
                    for mix in (0.1, 0.25, 0.5, 0.75, 1.0):
                        a_mix = (1.0 - mix) * a + mix * g
                        conv_mix = _conv_self(a_mix)
                        score_mix = float(conv_mix.max())
                        if score_mix < best_mix_score:
                            best_mix_score = score_mix
                            best_mix_a = a_mix.copy()

                    if best_mix_score < cur_max:          # improvement found
                        a = best_mix_a
                        # re‑initialise logits and Adam moments from the new a
                        logits = np.log(a + eps_small)
                        m = np.zeros_like(logits)
                        v = np.zeros_like(logits)
                        t_iter = 0
                        cur_max = best_mix_score
                        if cur_max < best_overall_score:
                            best_overall_score = cur_max
                            best_overall_seq = a.copy()
                            last_improve = it
                            print(f"[{it:6d}] LP‑mix improved max(conv) = {best_overall_score:.6f}")

        # ---- random perturbation (helps escape plateaus) -------------
        if it % 500 == 0 and it > 0:
            rnd = np.random.exponential(scale=1.0, size=n)
            rnd = _normalize(rnd, target_sum)
            mix = np.random.rand()
            pert = mix * best_overall_seq + (1.0 - mix) * rnd
            pert = _normalize(pert, target_sum)

            conv_pert = _conv_self(pert)
            score_pert = float(conv_pert.max())
            if score_pert < best_overall_score:
                best_overall_score = score_pert
                best_overall_seq = pert.copy()
                print(f"[{it:6d}] Perturbation yielded new best = {best_overall_score:.6f}")

        # ---- restart if stagnated for a long while --------------------
        if it - last_improve > 4000:
            noise = np.random.exponential(scale=1.0, size=n)
            noise = _normalize(noise, target_sum)
            a = 0.5 * best_overall_seq + 0.5 * noise
            logits = np.log(a + eps_small)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            t_iter = 0
            last_improve = it
            print(f"[{it:6d}] Restart due to stagnation")

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    return best_overall_seq.tolist()
