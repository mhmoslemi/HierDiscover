def _smooth_max_gradient_step(seq, step, gamma=200.0):
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # compute weight as exp(gamma*(conv - max_val)). This makes w_max = 1.
    w = np.exp(gamma * (conv - max_val))

    if w_sum == 0:

    else:

    else:
        new_seq = seq.copy()
    return new_seq

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  The algorithm consists of three stages:

1. **Exploratory families** –  generate candidates from several
   parameterised families:
   * power‑law  a_i ∝ (i+1+δ)^(-α)
   * beta‑law   a_i ∝ (i+δ)^(p‑1) (n+1‑i+δ)^(q‑1)   (U‑shaped includes the
     arcsine distribution p=q=½)
   * a few hand‑crafted offsets.

   For each length n (4000…8000) a coarse grid is scanned; the
   best candidate (lowest exact score 2·n·max(conv)) is kept.

2. **Refinement with a smoothed max‑gradient** – starting from the best
   candidate we minimise the smooth approximation
   L = Σ_k w_k (a*a)[k]  with   w_k ∝ exp(γ·((a*a)[k]‑max)),
   i.e. a log‑sum‑exp of the convolution entries.  
   The gradient of L has a closed‑form expression derived from the
   convolution and is applied with a simple projected‑gradient step
   (project onto the simplex {x≥0, Σx=1}).  An Adam‑style adaptive
   step‑size is used and the temperature γ is slowly increased so that
   the smoothed objective approaches the true max.

3. **Escapes & finalisation** – occasional random mass swaps
   (every ~100 iterations) help to leave plateaus.  The best sequence
   observed throughout the run is returned.

All heavy work is done with NumPy; the exact score is computed from
the convolution (no costly calls to ``evaluate_sequence`` are needed
during the search).  The routine respects the 1000 s wall‑time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    a_i ∝ (i+1+offset)^(-α)   ,   Σ a_i = 1
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _beta_sequence(
    n: int, p: float, q: float, offset: float = 0.0
) -> np.ndarray:
    """
    a_i ∝ (i+offset)^(p‑1) · (n+1‑i+offset)^(q‑1)   ,   Σ a_i = 1.
    (When p = q = 0.5 this is the discrete arcsine law.)
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    left = idx ** (p - 1.0)
    right = (n + 1.0 - idx) ** (q - 1.0)
    a = left * right
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _score_from_sequence(seq: np.ndarray) -> float:
    """
    Exact score used by ``evaluate_sequence`` for a normalised sequence
    (Σ seq == 1).  Returns 2·n·max(conv) where conv = seq * seq.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)  # length 2n‑1
    return float(2.0 * n * conv.max())


def _smooth_max_step(
    seq: np.ndarray,
    step: float,
    gamma: float,
    m1: np.ndarray,
    v1: np.ndarray,
    beta1: float,
    beta2: float,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One Adam‑style projected gradient step using a smoothed max
    (log‑sum‑exp) of the self‑convolution.

    Returns (new_seq, updated_m1, updated_v1).
    """
    n = seq.size
    conv = np.convolve(seq, seq)                # length 2n‑1
    max_val = conv.max()
    # soft‑max weights; the largest entry gets weight 1, others are
    # exponentially smaller.
    w = np.exp(gamma * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    # gradient of Σ w_k (a*a)[k]  with respect to a.
    # grad = a + (w ⊛ reverse(a))[n‑1 : n‑1+n]
    rev = seq[::-1]
    tmp = np.convolve(w, rev)                # length 3n‑2
    grad = seq + tmp[n - 1 : n - 1 + n]

    # Adam update of the (unprojected) parameters
    m1 = beta1 * m1 + (1.0 - beta1) * grad
    v1 = beta2 * v1 + (1.0 - beta2) * (grad * grad)

    m_hat = m1 / (1.0 - beta1)
    v_hat = v1 / (1.0 - beta2)

    update = step * m_hat / (np.sqrt(v_hat) + eps)
    new_seq = seq - update

    # projection onto the simplex {x ≥ 0, Σ x = 1}
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq /= s
    else:
        new_seq = seq.copy()

    return new_seq, m1, v1


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a few seconds as safety margin
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Exploration of parametric families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_n = None

    # candidate lengths (trade‑off between quality and speed)
    length_candidates = [4000, 5000, 6000, 8000]

    for n in length_candidates:
        if time.time() >= deadline:
            break

        # ---- uniform baseline (just for reference) ----
        seq = np.full(n, 1.0 / n)
        score = _score_from_sequence(seq)
        if score < best_score:
            best_score, best_seq, best_n = score, seq.copy(), n

        # ---- power‑law family ----
        alphas = np.linspace(0.45, 0.70, 14)  # step ≈0.018
        for alpha in alphas:
            if time.time() >= deadline:
                break
            seq = _power_sequence(n, alpha)
            score = _score_from_sequence(seq)
            if score < best_score:
                best_score, best_seq, best_n = score, seq.copy(), n

        # ---- fine tuning around the best α for this n ----
        if best_seq is not None:
            # use the last α that gave improvement (stored in best_seq)
            # we keep a small window around it
            window = np.linspace(alpha - 0.02, alpha + 0.02, 41)
            for a in window:
                if time.time() >= deadline:
                    break
                seq = _power_sequence(n, a)
                score = _score_from_sequence(seq)
                if score < best_score:
                    best_score, best_seq, best_n = score, seq.copy(), n

        # ---- beta‑law (U‑shaped) family ----
        p_vals = np.linspace(0.2, 2.0, 10)
        q_vals = np.linspace(0.2, 2.0, 10)
        for p in p_vals:
            if time.time() >= deadline:
                break
            # we only scan a subset of q to keep time reasonable
            for q in q_vals:
                if time.time() >= deadline:
                    break
                seq = _beta_sequence(n, p, q)
                score = _score_from_sequence(seq)
                if score < best_score:
                    best_score, best_seq, best_n = score, seq.copy(), n

        # ---- arcsine (p=q=0.5) – a notable special case ----
        seq = _beta_sequence(n, 0.5, 0.5)
        score = _score_from_sequence(seq)
        if score < best_score:
            best_score, best_seq, best_n = score, seq.copy(), n

    # If a global reference is available, also try it as a starting point
    if "height_sequence_1" in globals():
        ref = np.asarray(height_sequence_1, dtype=float)
        if ref.ndim == 1 and ref.size > 0:
            # normalise to unit sum (the objective is scale‑invariant)
            s = ref.sum()
            if s > 0.0:
                ref = ref / s
                score = _score_from_sequence(ref)
                if score < best_score:
                    best_score, best_seq, best_n = score, ref.copy(), ref.size

    # ------------------------------------------------------------------
    # 2. Refinement with smoothed max‑gradient (Adam)
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely, but guard against an empty result
        raise RuntimeError("No candidate sequence was generated.")

    seq = best_seq.copy()
    n = seq.size

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    m1 = np.zeros_like(seq)
    v1 = np.zeros_like(seq)

    step = 0.5          # initial learning rate
    step_decay = 0.9996
    gamma = 150.0       # start with a moderate smoothing
    gamma_inc = 1.02    # slowly increase to sharpen the max

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current exact score (no extra overhead – we already have conv)
        conv = np.convolve(seq, seq)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, seq.copy()
            print(
                f"[improve] iter={iteration}, n={n}, score={best_score:.6f}"
            )

        # gradient descent step
        seq, m1, v1 = _smooth_max_step(
            seq,
            step,
            gamma,
            m1,
            v1,
            beta1,
            beta2,
        )

        # occasional random mass exchange to help escaping plateaus
        if iteration % 97 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(step * seq[i], seq[i] * 0.3)
            cand = seq.copy()
            cand[i] = max(0.0, cand[i] - eps)
            cand[j] += eps
            total = cand.sum()
            if total > 0.0:
                cand /= total
                seq = cand

        # adapt learning‑rate and smoothing parameter
        step = max(step * step_decay, 1e-6)
        if iteration % 2000 == 0:
            gamma *= gamma_inc

    print(f"[final] best_score={best_score:.6f}, n={best_n}")
    return best_seq.tolist()
