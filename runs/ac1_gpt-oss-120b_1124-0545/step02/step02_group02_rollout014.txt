def _smooth_gradient_step(seq, step, lam=30):
    n = seq.size
    conv = np.convolve(seq, seq)
    maxc = conv.max()
    # compute weights based on softmax over conv
    exp_vals = np.exp(lam * (conv - maxc))
    w = exp_vals / exp_vals.sum()
    # gradient direction
    grad = 2.0 * np.correlate(w, seq, mode='valid')
    # orthogonal projection to hyperplane sum=0
    grad -= grad.mean()
    new_seq = seq - step * grad
    # project onto nonnegative simplex
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total > 0:
        new_seq /= total
    else:
        new_seq = seq.copy()
    return new_seq
```

"""
Propose a step‑function sequence that minimises the evaluation metric.

Algorithm
---------
1. **Parametric seeding** –  Scan a two‑parameter family
       a_i ∝ (i+1+δ)^{-α} .
   The scan uses a moderately sized support (n≈5000) and keeps the best
   normalised candidate.

2. **Masked sub‑gradient descent** –  The objective is
       F(a) = max_k (a∗a)_k .
   The exact sub‑gradient of the max is the sum of the gradients of all
   active constraints (indices where the convolution attains its maximum).
   We therefore build a binary mask `w` that marks those active indices and
   compute the gradient via an FFT‑based correlation
       g = 2·corr(w, a) .
   The gradient is projected onto the hyper‑plane ∑a_i = 1, a_i ≥ 0
   (simple orthogonal projection followed by normalisation).  A simple
   step‑size schedule together with occasional random mass swaps enables
   the iterate to leave plateaus.

3. **Time‑bounded loop** –  All work stops a few seconds before the
   supplied 1000 s budget, guaranteeing that a sequence is returned.

The code is fully vectorised, uses only NumPy and the already‑imported
`evaluate_sequence`, and respects the non‑negative, sum‑> 0.01
constraints required by the evaluator.
"""

import time

# Helper utilities (top‑level – no closures)

def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """Return a normalised (sum = 1) sequence a_i ∝ (i+1+offset)^{-α}."""
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:                     # fallback – should never happen
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _fft_len(required: int) -> int:
    """Smallest power‑of‑two ≥ required (for efficient FFTs)."""
    return 1 << (required - 1).bit_length()


def _self_convolution(seq: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution of `seq` using real FFT.
    Returns length (2·len(seq)‑1) array.
    """
    n = seq.shape[0]
    L = _fft_len(2 * n - 1)
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    return conv


def _metric(seq: np.ndarray) -> float:
    """
    Exact score used by ``evaluate_sequence`` for a normalised sequence
    (∑seq == 1):  score = 2·n·max(conv).
    """
    n = seq.shape[0]
    conv = _self_convolution(seq)
    return float(2.0 * n * conv.max())


def _masked_subgradient_step(seq: np.ndarray, step: float,
                             rel_tol: float = 0.001) -> np.ndarray:
    """
    One projected sub‑gradient step.
    Active set is defined by ``conv >= max(conv)·(1‑rel_tol)``.
    The gradient is g = 2·corr(mask, seq) (mask is binary, 0/1).
    After the step the vector is projected onto the simplex {x≥0,∑x=1}.
    """
    n = seq.shape[0]

    # --- convolution and active mask -----------------------------------
    conv = _self_convolution(seq)
    max_c = conv.max()
    # indices whose value is within ``rel_tol`` of the maximum.
    mask = (conv >= max_c * (1.0 - rel_tol)).astype(float)

    # --- gradient via correlation (FFT) ---------------------------------
    # correlation of mask with seq == convolution of mask with reversed seq
    L = _fft_len(mask.shape[0] + n - 1)
    fft_mask = np.fft.rfft(mask, n=L)
    fft_seq_rev = np.fft.rfft(seq[::-1], n=L)           # reversed seq
    corr_full = np.fft.irfft(fft_mask * fft_seq_rev, n=L)

    # the valid part (length n) starts at offset n‑1
    start = n - 1
    grad = 2.0 * corr_full[start : start + n]

    # --- projection onto the affine hyperplane ∑x = 1 ------------------
    grad -= grad.mean()                     # make step tangent to the simplex

    # --- gradient step + projection onto the simplex -------------------
    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)      # non‑negativity
    total = new_seq.sum()
    if total > 0.0:
        new_seq /= total
    else:
        new_seq = seq.copy()
    return new_seq


def _random_swap(seq: np.ndarray, scale: float = 0.2) -> np.ndarray:
    """
    Perform a tiny random exchange of mass between two distinct indices.
    The amount moved is bounded by ``scale·a_i`` and the result is
    re‑normalised to keep the sum equal to one.
    """
    n = seq.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    if seq[i] == 0.0:
        return seq
    eps = min(scale * seq[i], np.random.rand() * seq[i] * 0.5)
    cand = seq.copy()
    cand[i] -= eps
    cand[j] += eps
    cand = np.maximum(cand, 0.0)
    total = cand.sum()
    if total > 0.0:
        cand /= total
    else:
        cand = seq.copy()
    return cand


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time limit and
    returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a few seconds for final output
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Determine a sensible length.
    # ------------------------------------------------------------------
    n = 5000                                 # trade‑off between speed and quality

    # ------------------------------------------------------------------
    # 2. Coarse+fine grid search in the (α,δ) family.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_alpha = 0.5
    best_delta = 0.0

    alphas = np.linspace(0.48, 0.55, 71)      # step ≈0.001
    deltas = np.arange(0.0, 5.01, 0.5)       # integer and half‑integer shifts

    for alpha in alphas:
        for delta in deltas:
            if time.time() >= deadline:
                break
            seq = _power_sequence(n, alpha, offset=float(delta))
            score = _metric(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_alpha = alpha
                best_delta = delta
                print(f"[grid] α={alpha:.5f}, δ={delta:.2f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Accept a previously computed construction if it is better.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is provided by the evaluation harness.
        prev = np.array(height_sequence_1, dtype=float)
        if prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0.0:
                prev /= prev.sum()               # normalise to sum = 1
                cand_score = _metric(prev)
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = prev.copy()
                    print(f"[seed] used height_sequence_1, score={cand_score:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 4. Main optimisation loop – masked sub‑gradient descent.
    # ------------------------------------------------------------------
    step = 0.20                               # initial step size
    decay = 0.9999
    min_step = 1e-7
    no_improve = 0
    iter_cnt = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iter_cnt += 1
        # masked sub‑gradient step
        best_seq = _masked_subgradient_step(best_seq, step, rel_tol=0.001)

        cur_score = _metric(best_seq)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            last_improve_iter = iter_cnt
            no_improve = 0
            print(f"[improve] it={iter_cnt}, α≈{best_alpha:.5f}, "
                  f"δ≈{best_delta:.2f}, score={best_score:.6f}")
        else:
            no_improve += 1

        # occasional random mass swap to escape flat plateaus
        if iter_cnt % 500 == 0:
            cand = _random_swap(best_seq, scale=step)
            cand_score = _metric(cand)
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                last_improve_iter = iter_cnt
                no_improve = 0
                print(f"[swap] it={iter_cnt}, score={best_score:.6f}")

        # step‑size schedule
        step = max(step * decay, min_step)

        # safety break if we have stagnated for a long time
        if no_improve > 50000 and step <= min_step:
            break

    # ------------------------------------------------------------------
    # 5. Final sanity check and return.
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f}, length={n}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
