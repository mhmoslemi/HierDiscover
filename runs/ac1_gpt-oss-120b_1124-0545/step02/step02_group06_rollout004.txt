def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, math

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    # Initialize
    try:
        init = np.array(height_sequence_1, dtype=float)
        if init.size == 0:
            raise Exception
    except Exception:
        # Choose n=4000 (or 5000 if time allow)
        n = 4000
        init = _initial_root_sequence(n)

    # Ensure sum=1
    init = np.maximum(init, 0.0)
    total = init.sum()
    if total == 0:
        init = np.ones_like(init) / init.size
    else:
        init = init / total

    n = init.shape[0]

    # parameters
    total_time = deadline - start
    gamma_start = 100.0      # start small
    gamma_end = 50000.0
    step = 0.5
    step_decay = 0.9995

    # precompute FFT length m >= 3n-2.
    m = 1 << ((3*n - 2 - 1).bit_length())

    # helper for simplex projection
    def project_simplex(v):
        # Duchi et al.
        if np.all(v >= 0) and np.isclose(v.sum(), 1):
            return v
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    best_seq = init.copy()
    # compute best_score
    # compute conv and max
    F_init = np.fft.rfft(init, n=m)
    conv_init_full = np.fft.irfft(F_init * F_init, n=m)[:2*n-1]
    max_init = conv_init_full.max()
    best_score = 2.0 * n * max_init / (best_seq.sum()**2)
    cur_seq = init.copy()
    cur_score = best_score
    iteration = 0

    print(f"[init] n={n}, best_score={best_score:.6f}")

    while time.time() < deadline:
        iteration += 1

        # compute a*a convolution
        F = np.fft.rfft(cur_seq, n=m)
        conv_full = np.fft.irfft(F * F, n=m)[:2*n-1]
        max_conv = conv_full.max()

        # compute softmax weights
        shift = max_conv
        # using gamma schedule based on elapsed time fraction
        elapsed = time.time() - start
        frac = min(1.0, elapsed / total_time)
        gamma = gamma_start + frac * (gamma_end - gamma_start)

        exp_term = np.exp(gamma * (conv_full - shift))
        w = exp_term / exp_term.sum()

        # gradient = 2 * conv(p, a_rev) slice
        a_rev = cur_seq[::-1]
        F_w = np.fft.rfft(w, n=m)
        F_a_rev = np.fft.rfft(a_rev, n=m)
        conv_rev_full = np.fft.irfft(F_w * F_a_rev, n=m)[:len(w) + len(a_rev) - 1]
        grad = 2.0 * conv_rev_full[n-1:n-1+n]

        # propose new seq
        cand = cur_seq - step * grad
        cand = project_simplex(cand)

        # evaluate cand
        # compute conv for cand quickly
        F_cand = np.fft.rfft(cand, n=m)
        conv_cand_full = np.fft.irfft(F_cand * F_cand, n=m)[:2*n-1]
        max_cand = conv_cand_full.max()
        cand_score = 2.0 * n * max_cand / (cand.sum()**2)  # sum=1 after projection

        # acceptance: use simple descent with occasional random acceptance
        if cand_score < cur_score:
            accept = True
        else:
            # SA temperature schedule
            T0, Tf = 1.0, 0.001
            T = T0 * (1 - frac) + Tf * frac
            prob = math.exp(-(cand_score - cur_score) / T) if T > 0 else 0.0
            accept = np.random.rand() < prob

        if accept:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")

        # step decay periodically
        if iteration % 500 == 0:
            step = max(step * step_decay, 1e-6)

        # occasional random perturbation to jump out of local minima
        if iteration % 2000 == 0:
            # random reweight a few indices
            i, j = np.random.choice(n, 2, replace=False)
            eps = cur_seq[i] * np.random.uniform(0.001, 0.01)
            cand2 = cur_seq.copy()
            cand2[i] = max(cand2[i] - eps, 0)
            cand2[j] += eps
            cand2 = project_simplex(cand2)
            # compute score quickly using conv computed or recompute...
            F_c2 = np.fft.rfft(cand2, n=m)
            conv_c2 = np.fft.irfft(F_c2 * F_c2, n=m)[:2*n-1]
            cand2_score = 2.0 * n * conv_c2.max()
            if cand2_score < cur_score:
                cur_seq = cand2
                cur_score = cand2_score
                if cand2_score < best_score:
                    best_score = cand2_score
                    best_seq = cand2.copy()
                    print(f"[random_improve] iter={iteration:6d}, score={best_score:.6f}")

        # early stop if target achieved
        if best_score < 1.503:
            break

    print(f"[final] it={iteration}, best_score={best_score:.6f}")
    return best_seq.tolist()

"""
Propose a high‑quality non‑negative step‑function sequence.

Algorithm
---------
1. Initialise from a known construction (`height_sequence_1`) or fall back
   to the classic root law a_i ∝ 1/√(i+1) (normalised to sum = 1).

2. Shape optimisation – sample many power‑law sequences
   a_i ∝ (i+α+1)⁻ᵖ (p≈0.5, α≥0) and keep the best one according to the
   true objective (fast FFT convolution).

3. Smooth‑max gradient descent.
   * The objective  maxₖ (a*a)[k] is approximated by the
     log‑sum‑exp L(a)= (1/γ) log Σₖ exp(γ·(a*a)[k]).
   * ∂L/∂aᵢ = 2·Σₖ pₖ a_{k−i} where pₖ are the soft‑max weights.
     The sum is computed by a single FFT‑based convolution.
   * After a gradient step we project onto the probability simplex
     (non‑negative, unit sum).
   * γ grows linearly from a modest value to a large one, turning the
     smooth surrogate into the true max.
   * A tiny simulated‑annealing component allows occasional uphill moves.

The whole search runs inside the given time budget and returns the best
sequence encountered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _initial_root_sequence(n: int) -> np.ndarray:
    """Root law a_i ∝ 1/√(i+1) normalised to unit sum."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _power_law_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Generate a normalised sequence a_i ∝ (i + α + 1)^‑p.
    The '+1' avoids a division by zero when α=0.
    """
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _fft_len_for_conv(len_a: int, len_b: int) -> int:
    """Return the smallest power‑of‑two length that can contain the full
    convolution of two real vectors of lengths len_a and len_b."""
    need = len_a + len_b - 1
    return 1 << ((need - 1).bit_length())


def _conv_full(a: np.ndarray, b: np.ndarray, m: int) -> np.ndarray:
    """
    Full convolution of real vectors a and b using an FFT of length m.
    Returns a real array of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, n=m)
    fb = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(fa * fb, n=m)
    return conv[:len(a) + len(b) - 1]


def _self_conv_max(a: np.ndarray, m: int) -> float:
    """
    Compute max_k (a*a)[k] using an FFT of length m.
    """
    conv = _conv_full(a, a, m)
    return float(conv.max())


def _score_from_max(n: int, max_conv: float) -> float:
    """Objective value given max convolution entry (sum of a is 1)."""
    return 2.0 * n * max_conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: re‑normalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum

# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function sequence that minimises the
    provided `evaluate_sequence`.  The function respects the time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------

        if total == 0.0:

        init /= total
        n = init.size
        print(f"[init] using global height_sequence_1 (n={n})")
    except Exception:
        # fallback – a reasonably large n (5000) works well
        n = 5000
        init = _initial_root_sequence(n)

    # --------------------------------------------------------------
    # 2) Quick shape optimisation – random power‑law sampling
    # --------------------------------------------------------------
    best_seq = init.copy()
    m_conv = _fft_len_for_conv(n, n)               # length for a*a
    best_max = _self_conv_max(best_seq, m_conv)
    best_score = _score_from_max(n, best_max)
    print(f"[baseline] score={best_score:.6f}")

    # limited number of trials; stop early if time runs out
    max_trials = 2000
    for trial in range(max_trials):

            break
        p = np.random.uniform(0.35, 0.70)
        alpha = np.random.uniform(0.0, 5.0)

        cand_max = _self_conv_max(cand, m_conv)

    # --------------------------------------------------------------
    # 3) Smooth‑max gradient descent with simplex projection
    # --------------------------------------------------------------

    cur_seq = best_seq.copy()          # always sum to 1

    cur_score = best_score

    # parameters for the annealing / soft‑max schedule
    total_time = deadline - start_time

        iteration += 1

        # ----- compute self‑convolution and soft‑max weights -----
        conv_full = _conv_full(cur_seq, cur_seq, m_conv)            # length 2n‑1
        cur_max = float(conv_full.max())

        elapsed = time.time() - start_time

        gamma = gamma_start + frac * (gamma_end - gamma_start)

        # soft‑max (numerically stable)

        w = np.exp(gamma * (conv_full - shift))
        w /= w.sum()                                              # pₖ

        # ----- gradient of the smooth‑max -----
        a_rev = cur_seq[::-1]
        # convolution w * a_rev  (full length = (2n‑1)+n‑1 = 3n‑2)
        conv_rev_full = _conv_full(w, a_rev, m_grad)
        # gradient_i = 2 * conv_rev_full[n-1+i]
        grad = 2.0 * conv_rev_full[n - 1:n - 1 + n]

        cand_seq = cur_seq - step * grad
        cand_seq = _project_simplex(cand_seq)

        # ----- evaluate candidate -----
        cand_conv = _conv_full(cand_seq, cand_seq, m_conv)
        cand_max = float(cand_conv.max())
        cand_score = _score_from_max(n, cand_max)

            cur_score = cand_score
            cur_max = cand_max
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()

    # Return a plain Python list – the harness will call evaluate_sequence()
