Implementation:

```
def _softmax_weights(conv, tau):
    # conv is np.array of size 2n-1
    if tau < 1e-12:
        # treat as hard max; for tau extremely small set weight to 1 at max index
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    shifted = conv - np.max(conv)
    exp_vals = np.exp(shifted / tau)
    return exp_vals / exp_vals.sum()
```

    if cut <=0: cut=1

    a = np.empty(n)
    a[:cut] = (idx[:cut] + 1.0) ** (-p1)
    a[cut:] = (idx[cut:] + 1.0) ** (-p2)
    a /= a.sum()
    return a

Propose a high‑quality non‑negative step‑function sequence.

Algorithm
---------
1. **Initialisation**

   * If a global `height_sequence_1` is present, normalise and use it instead.

   * Keep the best (lowest evaluation score) among a few‑thousand trials.

   * Starting from the best candidate, minimise the smoothed objective
       F_τ(a) = τ·log( Σ_k exp( (a*a)_k / τ) )

   * Occasionally inject a tiny random perturbation to avoid stagnation.
   * Keep track of the best sequence seen during the refinement.

def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * b using a power‑of‑two FFT."""
    n = a.size + b.size - 1
    m = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    fb = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(fa * fb, n=m)[:n]
    return conv


def _score(seq: np.ndarray) -> float:
    """Fast objective  2·n·max(conv(seq,seq))  (sum(seq) is assumed = 1)."""
    n = seq.size
    conv = _conv_full(seq, seq)
    max_conv = conv.max()
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Analytic families of candidate sequences
# ----------------------------------------------------------------------
def _initial_root_sequence(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)  – normalised to unit L1."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _power_law_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """a_i ∝ (i+α+1)^‑p   with unit L1."""
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _piecewise_power_law_sequence(
    n: int, p1: float, p2: float, cut_frac: float
) -> np.ndarray:
    """Two‑segment power law.
    First `cut` entries use exponent p1, the rest use p2.
    """
    cut = max(1, int(cut_frac * n))
    idx = np.arange(n, dtype=float)
    a = np.empty_like(idx)
    a[:cut] = (idx[:cut] + 1.0) ** (-p1)
    a[cut:] = (idx[cut:] + 1.0) ** (-p2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max weights of a convolution and gradient w.r.t. the sequence
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Return weights w_k ∝ exp(conv_k / τ).  Handles τ≈0 safely."""
    if tau < 1e-12:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    shifted = conv - conv.max()                 # for numerical stability
    exp_vals = np.exp(shifted / tau)
    w = exp_vals / exp_vals.sum()
    return w


def _grad_from_weights(seq: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max smoothed max‑convolution:
        g_i = 2 * Σ_k w_k * seq_{k-i}   (out‑of‑range terms are 0).
    Implemented via a single FFT convolution with the reversed weight vector.
    """
    n = seq.size
    w_rev = w[::-1]                     # reverse because of index shift
    size = n + w_rev.size - 1
    m = 1 << ((size - 1).bit_length())
    fa = np.fft.rfft(seq, n=m)
    fw = np.fft.rfft(w_rev, n=m)
    conv_rev = np.fft.irfft(fa * fw, n=m)[:size]        # length = n+2n-2 = 3n-2
    offset = len(w) - 1                               # = 2n-2
    grad = 2.0 * conv_rev[offset - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found (as a list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the harness
    deadline = start_time + budget_s - 4.0

    # --------------------------------------------------------------
    # 1) Initialise a baseline sequence
    # --------------------------------------------------------------
    try:
        # `height_sequence_1` may be provided by the surrounding framework
        init = np.array(height_sequence_1, dtype=float)  # type: ignore
        init = np.maximum(init, 0.0)
        if init.size == 0 or init.sum() == 0.0:
            raise ValueError
        init /= init.sum()
        n = init.size
        print(f"[init] using global height_sequence_1 (n={n})")
    except Exception:
        n = 5000                     # a good compromise between quality & speed
        init = _initial_root_sequence(n)
        print(f"[init] fallback to root sequence (n={n})")

    best_seq = init.copy()
    best_score = _score(best_seq)
    print(f"[baseline] score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Coarse search over a few analytic families
    # --------------------------------------------------------------
    max_trials = 6000
    for trial in range(max_trials):
        if time.time() >= deadline:
            break

        # Randomly pick a family
        family = np.random.choice(["power", "piecewise"])
        if family == "power":
            p = np.random.uniform(0.35, 0.70)
            alpha = np.random.uniform(0.0, 12.0)
            cand = _power_law_sequence(n, p, alpha)
        else:  # piecewise
            p1 = np.random.uniform(0.30, 0.65)
            p2 = np.random.uniform(0.45, 0.75)
            cut_frac = np.random.uniform(0.05, 0.45)   # early‑segment length
            cand = _piecewise_power_law_sequence(n, p1, p2, cut_frac)

        cand_score = _score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(
                f"[shape] trial={trial:4d} fam={family} → score={cand_score:.6f}"
            )

    # --------------------------------------------------------------
    # 3) Fine‑grained mirror‑descent optimisation (soft‑max smoothing)
    # --------------------------------------------------------------
    seq = best_seq.copy()
    cur_score = best_score

    # Temperature and step‑size schedules (linear cooling)
    tau_start = 0.20
    tau_end = 1e-6
    step_start = 0.60
    step_end = 1e-4

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- current convolution and objective ----
        conv = _conv_full(seq, seq)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={it:7d} score={best_score:.6f}")

        # Target reached – we can stop early
        if best_score < 1.503:
            print("[target] reached required bound")
            break

        # ---- compute soft‑max weights for the current temperature ----
        elapsed = time.time() - start_time
        frac = (elapsed) / (deadline - start_time)   # 0 … 1
        tau = tau_start * (1.0 - frac) + tau_end * frac
        step = step_start * (1.0 - frac) + step_end * frac

        w = _softmax_weights(conv, tau)

        # ---- gradient of the smoothed max ----
        grad = _grad_from_weights(seq, w)

        # ---- multiplicative (mirror) update on the simplex ----
        # limit the exponent to avoid overflow
        delta = -step * grad
        delta = np.clip(delta, -20.0, 20.0)
        seq = seq * np.exp(delta)

        # projection onto the probability simplex (non‑negative, sum=1)
        total = seq.sum()
        if total > 0.0:
            seq /= total
        else:                             # numerical safeguard
            seq = np.full_like(seq, 1.0 / n)

        # ---- occasional random jitter to escape possible plateaus ----
        if it % 2000 == 0:
            jitter = 0.01 * (np.random.rand(n) - 0.5)
            seq = np.maximum(seq + jitter, 0.0)
            seq /= seq.sum()

    print(f"[final] iterations={it}, best_score={best_score:.6f}")

    # Return a plain Python list (the harness expects this)
    return best_seq.tolist()
