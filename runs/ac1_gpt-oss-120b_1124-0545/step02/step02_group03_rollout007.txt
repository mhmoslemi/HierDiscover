```
def project_to_simplex(v):
    # ensure nonnegative entries sum to 1
    # works for any vector.
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    # algorithm: sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_to_simplex(v):
    # Returns projection of v onto the simplex (non-negative sum to 1)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # if all <=0, set uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑value step function.

The implementation improves on the reference sub‑gradient method by
* using a smooth approximation of the *max* operation (soft‑max) which
  yields a dense gradient even when only a few convolution entries are
  active,
* employing the Adam adaptive‑learning‑rate optimiser on the probability
  simplex (after each update we simply clip to ≥0 and renormalise),
* slowly increasing the soft‑max “temperature’’ so the optimiser starts
  with a smooth landscape and ends with a sharp approximation of the true
  max,
* adding occasional random mass‑transfer moves to escape plateaus and a
  simple stagnation‑restart heuristic.

All heavy work is done with FFT‑based convolutions, thus each iteration
costs two O(n log n) FFTs regardless of the chosen length `n`.  The code
runs for up to the allotted 1000 s and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT of the given length."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _max_convolution(a: np.ndarray, size1: int) -> np.ndarray:
    """Full linear convolution a*a."""
    return _conv_fft(a, a, size1)


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # renormalise to keep sum = 1
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds. The function stops a little before the limit.
    kwargs : dict
        Optional overrides, e.g. ``n`` to change the length of the sequence.

    Returns
    -------
    list[float]
        The best non‑negative sequence found (already normalised to sum=1).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5       # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # ------------------------------------------------------------------
    # initialise a starting distribution (reuse previous construction if
    #   available, otherwise random)
    # ------------------------------------------------------------------
    try:
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            a = np.asarray(seq0, dtype=float)
            if a.size != n:
                # linear interpolation to the required grid
                x_old = np.linspace(0, 1, a.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, a)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.random.rand(n)
                a /= a.sum()
            else:
                a /= s
        else:
            raise KeyError
    except Exception:
        a = np.random.rand(n)
        a /= a.sum()

    # ------------------------------------------------------------------
    # pre‑compute FFT lengths (they stay constant)
    # ------------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a*a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * * (softmax) vector

    # ------------------------------------------------------------------
    # optimiser state (Adam)
    # ------------------------------------------------------------------
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05         # base learning rate

    # soft‑max temperature schedule
    beta_temp = 5.0
    beta_max = 200.0

    best_score = np.inf

        max_b = conv_a.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:

            best_seq = a.copy()

        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
