"""
Non‑negative step‑function optimizer.

The routine builds a diverse collection of parametric seeds, then
optimises each seed with Adam on the log‑softmax parametrisation using
a smooth approximation of the max‑convolution (log‑sum‑exp).  The
smoothing parameter β is increased linearly up to a modest value
(≈200) so the gradient never collapses to a single index.  After the
gradient phase a cheap local “mass‑swap’’ refinement is performed.
All work respects the supplied time budget (default 1000 s) and the
best sequence found (as a plain list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using the FFT.
    Returns a vector of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones_like(a) / n


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _triangular(n: int) -> np.ndarray:
    half = np.arange(1, (n + 1) // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        a = np.concatenate([half, half[::-1][1:]])
    else:
        a = np.concatenate([half, half[-2::-1]])
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _cosine_bell(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * (2 * i / (n - 1) - 1.0))) / 2.0
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


# ----------------------------------------------------------------------
# Optimisation of a single seed
# ----------------------------------------------------------------------


def _optimise_one_seed(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑softmax parametrisation.
    Returns the best sequence (as a numpy array) and its objective value.
    """
    # Hyper‑parameters
    lr0 = 0.5          # base learning rate
    beta1 = 0.9
    beta2 = 0.99
    eps_adam = 1e-12

    beta_start = 1.0          # initial smoothing beta
    beta_max = 200.0          # clamp for numerical stability
    beta_inc = (beta_max - beta_start) / max_iters

    v = np.log(a0 + 1e-30)    # logits for softmax
    m = np.zeros_like(v)      # first moment
    v2 = np.zeros_like(v)     # second moment

    best_seq = a0.copy()
    conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(conv)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[optim] n={n:5d} it={it:5d} new best={cur_val:.6f}")

        # ---------- smooth‑max weights ----------
        beta = min(beta_max, beta_start + beta_inc * it)
        shift = cur_max                               # stability shift
        w = np.exp(beta * (conv - shift))            # exponentials
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---------- gradient w.r.t. a ----------
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                  # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]      # central n entries

        # ---------- gradient w.r.t. logits ----------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val

# Simple local refinement: tiny mass swaps
# ----------------------------------------------------------------------


def _local_refine(seq: np.ndarray,
                  n: int,
                  deadline: float,
                  attempts: int = 3000) -> tuple[np.ndarray, float]:
    """
    Perform cheap stochastic pairwise swaps of a tiny amount of mass.
    Only runs while time permits.
    """
    a = seq.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    # weight for choosing a source index – square favours large entries
    src_weights = a ** 2 + 1e-30
    src_weights /= src_weights.sum()

    for _ in range(attempts):
        if time.time() > deadline:
            break

        i = np.random.choice(n, p=src_weights)
        j = np.random.randint(n)
        if i == j:
            continue

        # move a *relative* amount; keep it tiny so that the landscape stays smooth
        delta = min(0.02 * a[i], 5e-4)   # at most 2 % of a[i] or 5e‑4 absolute mass
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv_new = _conv_fft(a_new, a_new)
        new_val = 2.0 * n * np.max(conv_new)

        if new_val < best_val:
            a = a_new
            conv = conv_new
            best_val = new_val
            # refresh source probabilities because the distribution changed
            src_weights = a ** 2 + 1e-30
            src_weights /= src_weights.sum()
            # occasional progress report
            print(f"[refine] n={n:5d} improved to {best_val:.6f}")

    return a, best_val

                seeds.append(prev_seq / prev_seq.sum())
            else:
                # linear interpolation / truncation to the new size
                x_old = np.linspace(0, 1, prev_seq.shape[0])
                x_new = np.linspace(0, 1, n)
                interp = np.interp(x_new, x_old, prev_seq)
                if interp.sum() == 0:
                    interp = np.ones_like(interp) / n
                else:
                    interp = interp / interp.sum()
                seeds.append(interp)

        # --------------------------------------------------------------
        # Fast pre‑scan to get a quick global baseline
        # --------------------------------------------------------------
        for a in seeds:
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] n={n:5d} new global baseline={cur_val:.6f}")

        # --------------------------------------------------------------
        # Optimise each seed independently (time‑aware)
        # --------------------------------------------------------------
        for idx, a0 in enumerate(seeds):
            if time.time() > deadline:
                break
            best_seq_local, best_val_local = _optimise_one_seed(
                a0, n, deadline, max_iters=15000
            )
            if best_seq_local is not None and best_val_local < global_best_val:
                global_best_val = best_val_local
                global_best_seq = best_seq_local.copy()
                print(f"[overall] n={n:5d} seed={idx:2d} improved global best={global_best_val:.6f}")

        # --------------------------------------------------------------
        # Local stochastic refinement on the best sequence for this length
        # --------------------------------------------------------------
        if global_best_seq is not None:
            refined_seq, refined_val = _local_refine(
                global_best_seq, n, deadline, attempts=3000
            )
            if refined_val < global_best_val:
                global_best_val = refined_val
                global_best_seq = refined_seq.copy()
                print(f"[overall] n={n:5d} after refinement global best={global_best_val:.6f}")

        # keep current best for possible interpolation to the next length
        prev_seq = global_best_seq

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] using uniform distribution")

    # Convert to plain Python list as required by the platform
    return global_best_seq.tolist()
