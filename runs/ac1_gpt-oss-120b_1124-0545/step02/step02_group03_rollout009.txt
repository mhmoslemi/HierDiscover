"""
search for a low‑value step‑function using a smoothed max‑autocorrelation
gradient descent (Adam) together with occasional random “mass‑transfer”
perturbations and stagnation‑restarts.

The sequence is kept on the probability simplex (∑a = 1).  At each
iteration we

  1. compute the full linear convolution a*a,
  2. form a soft‑max weighting of the convolution entries (parameter β
     controls the smoothness, it is increased gradually so that the
     weights concentrate on the true maximal entries),
  3. obtain the gradient of the log‑sum‑exp of the convolution
     (which is a smooth approximation of ∇max (a*a)),
  4. apply an Adam step and project back onto the simplex,
  5. with a small probability perform a random transfer of mass
     between two coordinates to escape flat plateaus,
  6. restart from a fresh random distribution if no improvement has been
     observed for a while.

The best sequence seen during the allotted time is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution u*v using FFT of given length (size ≥ len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return _project_simplex(a_new)


# ----------------------------------------------------------------------
# Main optimisation routine required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence``.  The algorithm runs for at most ``budget_s``
    seconds and prints intermediate progress.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5       # safety margin

    # --------------------------------------------------------------
    # problem size (allow user override)
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # initialise a probability vector (sum = 1)
    # --------------------------------------------------------------
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None:
        a0 = np.asarray(seq0, dtype=float).ravel()
        if a0.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, a0.size, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            a0 = np.interp(x_new, x_old, a0)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0.0:
            a = np.random.rand(n)
            a /= a.sum()
        else:
            a = a0 / a0.sum()
    else:
        a = np.random.rand(n)
        a /= a.sum()

    # --------------------------------------------------------------
    # pre‑compute FFT lengths (powers of two) – they stay constant
    # --------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a*a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * weight vector

    # --------------------------------------------------------------
    # optimisation bookkeeping
    # --------------------------------------------------------------
    conv_full = _conv_fft(a, a, size1)
    max_b = conv_full.max()
    best_score = 2.0 * n * max_b          # because sum(a)=1
    best_seq = a.copy()
    print(f"[init] n={n}, init score={best_score:.6f}")

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.05                     # base learning rate for Adam

    # soft‑max (log‑sum‑exp) temperature schedule
    beta = 5.0                    # start “soft” – low beta
    beta_max = 150.0
    beta_inc = 1.001             # geometric increase per 100 iterations

    # random‑move parameters
    rand_prob = 0.14
    rand_scale = 0.07

    # stagnation detection
    stagnation_restart = 30.0            # seconds without improvement
    last_improve = start

    it = 0

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---- 1. current convolution and score --------------------
        conv_full = _conv_fft(a, a, size1)
        max_b = conv_full.max()
        sum_a = a.sum()
        cur_score = 2.0 * n * max_b / (sum_a * sum_a)

        # ---- 2. keep best ----------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ---- 3. decide what move to apply -----------------------
        if np.random.rand() < rand_prob:
            # cheap random perturbation
            a = _random_transfer(a, rand_scale)
        else:
            # ----- 3a. soft‑max gradient step (Adam) -------------
            # increase the temperature slowly
            if it % 100 == 0 and beta < beta_max:
                beta = min(beta_max, beta * beta_inc)

            # soft‑max weights (log‑sum‑exp gradient)
            # subtract max to avoid overflow
            diff = conv_full - max_b          # ≤ 0
            w = np.exp(beta * diff)           # weight for each convolution entry
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of log‑sum‑exp w.r.t. a   = 2 * (a_rev * w)
            g_full = _conv_fft(a[::-1], w, size2)          # length 3n‑2
            grad = 2.0 * g_full[n - 1 : n - 1 + n]          # central n entries

            # Adam update (descent)
            it1 = it + 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it1)
            v_hat = v / (1.0 - beta2 ** it1)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_simplex(a)

        # ---- 4. stagnation restart -------------------------------
        if time.time() - last_improve > stagnation_restart:
            # fresh random start, re‑initialise moments
            a = np.random.rand(n)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{it:7d}] restart due to stagnation")

        it += 1

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
