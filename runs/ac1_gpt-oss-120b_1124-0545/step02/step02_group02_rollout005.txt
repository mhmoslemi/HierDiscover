"""
Improved step‑function search.

The algorithm consists of three stages:

1. **Parametric sweep** –  a three‑parameter family  
   a_i ∝ (i+1+δ)^(-α)·exp(-λ·i)   (α>0, λ≥0, integer δ≥0) is scanned on a
   coarse grid and then locally refined.  The best vector (sum = 1) is kept.

2. **Projected sub‑gradient descent** –  starting from the best parameter
   candidate we minimise  max(a*a)  (the numerator of the evaluation
   function) with a classic sub‑gradient step.  The gradient of the
   *k*‑th convolution entry is   ∂(a*a)[k]/∂a_i = 2·a_{k-i}  (or a_i
   when i=k−i).  We take the sub‑gradient of the *top‑k* largest
   convolution entries, take a step, and project back onto the simplex
   {x≥0, ∑x=1}.  Momentum and occasional random mass swaps help to
   escape flat regions.

3. **Result** –  the best sequence seen during the whole run is returned.
   All numbers are non‑negative and the total mass is 1, guaranteeing a
   finite score from ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (onto {x ≥ 0, sum(x)=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of candidate sequences
# ----------------------------------------------------------------------
def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^(-α), normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-alpha)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _exp_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _combined_sequence(n: int, alpha: float, lam: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^(-α)·exp(-lam·i), normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = (idx ** (-alpha)) * np.exp(-lam * (idx - 1))
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Scoring utilities (exact score = 2·n·max(conv) because sum=1)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Exact score used by ``evaluate_sequence`` for a normalised sequence."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)          # length 2·n‑1
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the maximum convolution entry (top‑k variant)
# ----------------------------------------------------------------------
def _subgradient(seq: np.ndarray, top_idx: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of the sum of the convolution entries indexed by
    ``top_idx``.  Implements the factor‑2 rule (∂(a*a)[k]/∂a_i = 2·a_{k-i}).
    """
    n = seq.shape[0]
    grad = np.zeros_like(seq)
    for k in top_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        j = k - i
        mask = i != j
        # off‑diagonal contributions (factor 2)
        grad[i[mask]] += 2.0 * seq[j[mask]]
        # diagonal contributions (i == j) have factor 1
        diag = i[~mask]
        grad[diag] += seq[diag]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a ~1000 s time budget, prints progress and finally
    returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a few seconds for the final return
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 0. Basic configuration
    # --------------------------------------------------------------
    n = 5000                     # length of the step function (good speed/quality)

    # --------------------------------------------------------------
    # 1. Coarse grid search over the 3‑parameter family
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_params = None

    # coarse ranges
    alphas = np.arange(0.35, 0.71, 0.025)                # α
    lams   = np.arange(0.0, 0.0065, 0.0005)              # λ (exponential decay)
    offsets = range(0, 6)                               # δ

    for alpha in alphas:
        for lam in lams:
            for delta in offsets:
                if time.time() >= deadline:
                    break
                seq = _combined_sequence(n, alpha, lam, offset=float(delta))
                sc = _score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    best_params = (alpha, lam, delta)
                    print(f"[grid] α={alpha:.3f}, λ={lam:.4f}, δ={delta}, score={sc:.6f}")

    # --------------------------------------------------------------
    # 2. Fine‑grained search around the best parameters found above
    # --------------------------------------------------------------
    if best_params is None:        # safety, should never happen

        best_score = _score(best_seq)
        best_params = (0.5, 0.0, 0)

    fine_alphas = np.linspace(max(0.1, a0 - 0.02), a0 + 0.02, 41)   # step ≈ 0.001
    fine_lams   = np.linspace(max(0.0, l0 - 0.001), l0 + 0.001, 21) # step ≈ 0.0001
    fine_offsets = range(max(0, d0 - 2), d0 + 3)

    for alpha in fine_alphas:
        for lam in fine_lams:
            for delta in fine_offsets:
                if time.time() >= deadline:
                    break
                seq = _combined_sequence(n, alpha, lam, offset=float(delta))
                sc = _score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    best_params = (alpha, lam, delta)
                    print(f"[fine] α={alpha:.5f}, λ={lam:.5f}, δ={delta}, score={sc:.6f}")

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient descent refinements
    # --------------------------------------------------------------
    # initialise step‑size schedule and momentum
    step = 0.3
    decay = 0.99994
    momentum = np.zeros_like(best_seq)

    cur_seq = best_seq.copy()
    cur_score = best_score
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        iteration += 1

        # compute convolution and the indices of the largest entries
        conv = np.convolve(cur_seq, cur_seq)
        max_val = conv.max()
        top_k = min(7, conv.size)          # use a few top entries
        top_idx = np.argpartition(-conv, top_k - 1)[:top_k]

        # sub‑gradient for the selected entries
        grad = _subgradient(cur_seq, top_idx)

        # momentum update
        momentum = 0.9 * momentum + 0.1 * grad

        # tentative update and projection onto the simplex
        cand_seq = cur_seq - step * momentum
        cand_seq = _project_simplex(cand_seq)

        # evaluate candidate
        cand_score = _score(cand_seq)

        # accept only if we get an improvement (monotone descent)
        if cand_score < cur_score - 1e-12:
            cur_seq = cand_seq
            cur_score = cand_score
            no_improve = 0
            # update global best
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()
                print(f"[improve] iter={iteration}, score={best_score:.6f}")
        else:
            # no progress – shrink step size a bit more aggressively
            step *= 0.5
            no_improve += 1

        # occasional random mass exchange to escape plateaus
        if iteration % 120 == 0:

            rnd_seq = cur_seq.copy()

            rnd_seq[j] += eps
            rnd_seq = _project_simplex(rnd_seq)
            rnd_score = _score(rnd_seq)
            if rnd_score < best_score:
                best_score = rnd_score
                best_seq = rnd_seq.copy()
                cur_seq = rnd_seq
                cur_score = rnd_score
                print(f"[rand‑swap] iter={iteration}, score={best_score:.6f}")

        # decay step size (but never let it vanish completely)
        step = max(step * decay, 1e-7)

        # safety break if we have done a huge number of iterations

    print(f"[final] best_score={best_score:.6f}, params={best_params}, iterations={iteration}")
