"""
Propose a step‑function (list of non‑negative heights) that minimizes

    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of two phases:

1.  **LP‑balancing phase** – a Frank‑Wolfe / cutting‑plane style iteration.
    For the current vector `f` we compute `M = max(conv(f,f))`.  Keeping `M`
    fixed we solve the linear program

        maximise   sum(g)
        subject to  conv(f,g) ≤ M      (linear in g)
                    g ≥ 0

    The solution `g` tries to place as much mass as possible in positions
    where the current convolution is far from the bound `M`.  After scaling
    `g` to the prescribed ℓ₁‑norm we move a fraction `t` towards `g` and
    repeat.  The LP is solved with `scipy.optimize.linprog` using a sparse
    Toeplitz constraint matrix; a few dozen iterations already reduce the
    objective substantially.

2.  **Simulated‑annealing refinement** – a lightweight SA/hill‑climbing loop
    (the one from the reference implementation) runs on the best vector
    found by the LP phase for the remaining time.  It uses several elementary
    moves (mass transfer, swapping, peak‑reduction and a sub‑gradient step)
    and a temperature schedule to escape local minima.

Both phases respect the simplex constraint `sum(a) = sqrt(2n)` by means of
the O(n log n) projection of Wang & Carreira‑Perpiñán.  The maximal
convolution entry is evaluated internally with an FFT (O(n log n)) while
the final sequence is returned as a plain Python list for the external
`evaluate_sequence` call.

The routine runs for at most ``budget_s`` seconds and prints progress.
"""
import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector `v` onto the simplex {x ≥ 0, sum(x)=s}.
    Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, nfft: int) -> float:
    """
    Return the maximal entry of the linear convolution a*a.
    Uses an FFT of length `nfft` (next power of two larger than
    2*len(a)-1) for speed.
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray, row_idx: np.ndarray, col_idx: np.ndarray) -> csr_matrix:
    """
    Construct the sparse matrix that represents the linear map
    g ↦ conv(f, g) (length 2n‑1).  The rows and columns pattern is
    pre‑computed once; only the data (which are copies of f[i]) changes.
    """
    data = np.repeat(f, f.size)               # each f[i] repeated n times
    return csr_matrix((data, (row_idx, col_idx)),
                      shape=(2 * f.size - 1, f.size))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises the evaluation function.  The routine uses the LP‑balancing
    phase followed by a simulated‑annealing refinement and respects the
    1000 s time budget.
    """
    # ------------------------ initialisation ---------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # Problem size ------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        f = np.array(height_sequence_1, dtype=np.float64)
    else:
        # choose a moderate size that still gives a good search space
        n = 1500
        f = np.random.exponential(scale=1.0, size=n)

    target_sum = math.sqrt(2.0 * n)                # normalisation for the objective
    f = f / f.sum() * target_sum                   # simplex projection (exact)
    f = _simplex_projection(f, target_sum)

    # FFT length for fast convolution -----------------------------------------
    nfft = 1 << ((2 * n - 1).bit_length())

    # Pre‑compute the row/column pattern of the Toeplitz convolution matrix
    col_idx = np.tile(np.arange(n), n)                     # 0,1,…,n‑1 repeated n times
    row_idx = np.repeat(np.arange(n), n) + col_idx        # i + j for every pair (i,j)

    # ------------------------ keep best solution -------------------------------
    cur_score = _max_convolution(f, nfft)
    best_f = f.copy()
    best_score = cur_score
    print(f"[init] n={n}, score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0       # small safety margin
    iteration = 0

    # ------------------------------ LP phase ---------------------------------
    # We dedicate roughly the first half of the budget to LP updates.
    lp_deadline = start_time + budget_s * 0.55

    while time.time() < lp_deadline:
        # recompute maximum convolution entry
        M = _max_convolution(f, nfft)  # M = max(conv(f,f))

        # build the constraint matrix A (conv(f,g) ≤ M)
        A = _build_conv_matrix(f, row_idx, col_idx)

        # linear program: maximise sum(g)  ⇔  minimise -sum(g)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M + 1e-12)  # tiny slack for numerical safety

                      method='highs', options={"presolve": True})

            eps = target_sum * 1e-5
            f += np.random.normal(scale=eps, size=n)
            f = np.maximum(f, 0.0)
            f = _simplex_projection(f, target_sum)
            cur_score = _max_convolution(f, nfft)
            continue

        sum_g = g.sum()
        if sum_g <= 0:
            # degenerate solution – make a small random move
            f += np.random.normal(scale=target_sum * 1e-5, size=n)
            f = np.maximum(f, 0.0)
            f = _simplex_projection(f, target_sum)
            cur_score = _max_convolution(f, nfft)
            continue

        # scale g to the correct ℓ₁‑norm and mix it with the current f
        g = g * (target_sum / sum_g)
        t = 0.2                         # step size towards the LP direction
        f_new = (1.0 - t) * f + t * g
        f_new = _simplex_projection(f_new, target_sum)

        new_score = _max_convolution(f_new, nfft)

        # accept if we made progress; otherwise keep the old f
        if new_score < cur_score:
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f.copy()
                print(f"[LP  {iteration:5d}]  new best score: {best_score:.6f} (time {time.time()-start_time:.1f}s)")

        iteration += 1

    # --------------------------- SA refinement phase ---------------------------
    # Use the best vector from the LP stage as the starting point.
    a = best_f.copy()
    cur_score = best_score
    # temperature schedule (exponential cooling)
    T_start = 1.0
    T_end = 1e-6

    # main SA loop – runs until the overall deadline is reached
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)          # 0 → 1
        temperature = T_start * ((T_end / T_start) ** frac)

        move_type = random.random()
        a_prop = a.copy()

        # ---- 1) random transfer ------------------------------------------------
        if move_type < 0.30:
            i, j = np.random.randint(n, size=2)
            if i != j:
                max_delta = min(a_prop[i], target_sum * 0.04)
                delta = max_delta * temperature * random.random()
                if delta > 1e-12:
                    a_prop[i] -= delta
                    a_prop[j] += delta

        # ---- 2) swap ------------------------------------------------------------
        elif move_type < 0.55:
            i, j = np.random.randint(n, size=2)
            if i != j:
                a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        # ---- 3) peak‑reduction --------------------------------------------------
        elif move_type < 0.80:
            conv_full = np.convolve(a, a)
            max_idx = int(np.argmax(conv_full))
            # determine contributions of each coordinate to the peak
            idx = np.arange(n)
            partner = max_idx - idx
            mask = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[idx[mask]] * a[partner[mask]]

            # source = a high‑contributor, destination = low‑contributor
            src = np.argmax(contrib)
            # choose a destination that does NOT contribute (or a low‑contributor)
            non_contrib = np.where(~mask)[0]
            if non_contrib.size:
                dst = int(np.random.choice(non_contrib))
            else:
                dst = int(np.argmin(contrib))

            max_delta = min(a_prop[src], target_sum * 0.02)
            delta = max_delta * temperature * random.random()
            if delta > 1e-12:
                a_prop[src] -= delta
                a_prop[dst] += delta

        # ---- 4) sub‑gradient step ------------------------------------------------
        else:
            conv_full = np.convolve(a, a)
            max_idx = int(np.argmax(conv_full))
            idx = np.arange(n)
            partner = max_idx - idx
            mask = (partner >= 0) & (partner < n)
            grad = np.zeros_like(a)
            grad[mask] = a[partner[mask]]
            # sub‑gradient (sign opposite to increase) – step size scales with temperature
            eta = target_sum * 0.20 * temperature
            a_prop -= eta * grad

        # enforce feasibility
        a_prop = _simplex_projection(a_prop, target_sum)

        # evaluate (fast FFT version)
        new_score = _max_convolution(a_prop, nfft)

        # simulated‑annealing acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (temperature + 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = a.copy()
                print(f"[SA  {iteration:7d}]  new best score: {best_score:.6f} (time {elapsed:.1f}s)")

        # occasional global perturbation to avoid stagnation
        if iteration % 25000 == 0 and iteration > 0:
            a = best_f + np.random.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a, target_sum)
            cur_score = _max_convolution(a, nfft)

        iteration += 1

    # -------------------------------------------------------------------------
    # Return the best sequence discovered
    print(f"[final] best score = {best_score:.6f}, n = {n}")
    return best_f.tolist()
