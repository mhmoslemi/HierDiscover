def project_onto_simplex(v):
    '''Projects vector v onto the probability simplex {x >=0, sum=1}.'''
    # algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _grad_step(seq, step):
    n = len(seq)
    # compute max conv index
    # compute conv via fft
    m = 1 << ((2*n-2).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    k_max = int(conv.argmax())
    # gradient
    # grad_i = 2 * seq[k_max - i] for valid indices i, else 0
    # compute using vectorized moves
    lo = max(0, k_max - (n-1))
    hi = min(k_max, n-1)
    # indices within support
    grad = np.zeros(n)
    # i in [lo, hi]
    idx = np.arange(lo, hi+1)
    rev = k_max - idx
    grad[idx] = 2.0 * seq[rev]
    # Project onto simplex after gradient descent
    new_seq = seq - step * grad
    # project onto simplex
    proj_seq = _project_onto_simplex(new_seq)
    return proj_seq

def _project_onto_simplex(v):
    """Project onto simplex {x >=0, sum x = 1}."""
    # ensure array
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all non-positive, return uniform distribution
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # ensure sum exactly 1 (numerical)
    w /= w.sum()
    return w

"""
High‑performance construction of a non‑negative step sequence.

The algorithm works in three phases within the allowed time budget:

1. **Exploration of simple parametric families** – random power‑law
   sequences a_i ∝ (i+α+1)^‑p, as well as “flattened’’ versions where
   the first L entries are forced to the value at index L
   (a_i = min( (i+1)^‑p , (L+1)^‑p )).  Thousands of candidates are
   generated and the best one (according to the fast‑FFT score) is kept.

2. **Projected sub‑gradient descent** – the objective
   f(a)=2·n·max(conv(a,a)) is piecewise linear.  The sub‑gradient of
   the active convolution index k* is
        g_i = 2·a_{k*‑i}   (if 0≤i≤k* and k*‑i<n, else 0).
   A step a←a−η·g is taken and the result is projected onto the
   probability simplex (non‑negative, unit sum) using the O(n)
   algorithm of Duchi et al.  A simple schedule for η (geometric decay)
   together with occasional random mass‑transfer moves yields a monotone
   decrease of the score.

3. **Polishing & early exit** – every improvement is printed.
   If the target bound 1.503 is reached the search stops early.

The code is fully self‑contained, uses only NumPy and SciPy (for the
optional linear‑programming fallback) and respects the 1000 s budget.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # optional, used only for a quick LP warm‑start

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _initial_root_sequence(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)  (sum normalised to 1)."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _power_law_sequence(n: int, p: float, alpha: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+α+1)^‑p  (normalised to unit sum)."""
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _plateau_power_law_sequence(n: int, p: float, L: int) -> np.ndarray:
    """
    Power‑law tail with a flat plateau of length L.
    For i ≤ L set a_i = (L+1)^‑p, otherwise a_i = (i+1)^‑p.
    The result is normalised to sum 1.
    """
    tail = (np.arange(n, dtype=float) + 1.0) ** (-p)
    if L >= n:
        a = np.full(n, tail[-1])
    else:
        plateau_val = tail[L]
        a = np.minimum(tail, plateau_val)
    a /= a.sum()
    return a


def _conv_max_and_arg(seq: np.ndarray) -> tuple[float, int]:
    """
    Return (max(conv(seq,seq)), argmax) using a real FFT.
    """
    n = seq.size
    size = 2 * n - 1
    # next power of two for speed
    m = 1 << ((size - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:size]
    arg = int(np.argmax(conv))
    return float(conv[arg]), arg


def _score(seq: np.ndarray) -> float:
    """
    Fast surrogate for `evaluate_sequence`.
    The sequence is assumed non‑negative and already normalised to sum 1,
    therefore the exact expression 2·n·max(conv) is sufficient.
    """
    n = seq.size
    max_b, _ = _conv_max_and_arg(seq)
    # safety replicate of evaluate_sequence's guard against tiny sums
    if seq.sum() < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (seq.sum() ** 2)


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values are non‑positive – fall back to uniform distribution
        return np.full(n, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= w_sum
    return w


def _subgradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    One sub‑gradient step for the objective 2·n·max(conv(seq,seq)).
    The sub‑gradient is averaged over all currently active (maximal)
    convolution indices.
    """
    n = seq.size
    max_b, _ = _conv_max_and_arg(seq)
    # recompute the whole convolution to locate all active indices
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:size]

    # tolerance to catch ties arising from floating point noise
    tol = max_b * 1e-12
    active = np.where(conv >= max_b - tol)[0]

    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        rev = k - idx
        grad[idx] += 2.0 * seq[rev]      # factor 2 comes from the derivative
    if active.size > 0:
        grad /= active.size               # average over the active set

    new_seq = seq - step * grad
    return _project_onto_simplex(new_seq)


def _random_mass_transfer(seq: np.ndarray, eps_frac: float = 0.005) -> np.ndarray:
    """Move a small fraction of mass from a random i to a random j."""
    n = seq.size
    i, j = np.random.choice(n, 2, replace=False)
    eps = seq[i] * np.random.uniform(eps_frac * 0.5, eps_frac)
    eps = min(eps, seq[i])
    cand = seq.copy()
    cand[i] -= eps
    cand[j] += eps
    return _project_onto_simplex(cand)


def _lp_improvement(seq: np.ndarray, M: float) -> np.ndarray:
    """
    One iteration of the linear‑programming improvement described in the
    statement.  Solve max Σb_j  s.t.  b≥0  and  conv(seq,b) ≤ M.
    The optimiser returns an extreme point; we then renormalise it.
    """
    n = seq.size
    # Build the Toeplitz matrix representing conv(seq,·)
    # Each row k (0≤k<2n‑1) contains the coefficients of b that affect (seq*b)_k
    A = np.zeros((2 * n - 1, n), dtype=float)
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        indices = np.arange(lo, hi + 1, dtype=int)
        A[k, indices] = seq[k - indices]
    # Linear program: maximize c·b  (c = ones)  with  A·b ≤ M
    c = -np.ones(n)                     # linprog does minimisation → use -c
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return seq
    b = res.x
    # renormalise to unit sum (the original routine rescales to √(2n), but for us
    # any scaling leaves the objective unchanged – we keep unit sum)
    if b.sum() == 0.0:
        return seq
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function sequence that (approximately)
    minimises `evaluate_sequence`.  The routine respects the given
    time budget and prints progress messages.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1) Choose a problem size (large enough for a good constant,
    #    small enough to keep each iteration cheap)
    # ------------------------------------------------------------------
    n = 4000                     # 4000 gives a good trade‑off (≈10 k FFT size)
    print(f"[info] Using length n={n}")

    # ------------------------------------------------------------------
    # 2) Initialise – try the global construction if it exists,
    #    otherwise fall back to the classical root law.
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` may be supplied by the evaluation harness
        global_seq = np.array(height_sequence_1, dtype=float)  # type: ignore
        global_seq = np.maximum(global_seq, 0.0)
        if global_seq.size == n:
            init = global_seq / global_seq.sum()
            best_seq = init.copy()
            best_score = _score(best_seq)
            print(f"[init] global height_sequence_1 score={best_score:.6f}")
        else:
            raise ValueError
    except Exception:
        best_seq = _initial_root_sequence(n)
        best_score = _score(best_seq)
        print(f"[init] root law baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Random exploration of simple parametric families
    # ------------------------------------------------------------------
    shape_deadline = start_time + (budget_s * 0.30)   # 30 % of the budget
    trials = 0
    while time.time() < shape_deadline:
        trials += 1
        # pick a family
        family = np.random.choice(["power", "plateau"], p=[0.65, 0.35])
        p = np.random.uniform(0.15, 2.0)          # exponent
        if family == "power":
            alpha = np.random.uniform(0.0, 8.0)
            cand = _power_law_sequence(n, p, alpha)
        else:  # plateau family
            L = np.random.randint(0, n)          # plateau length
            cand = _plateau_power_law_sequence(n, p, L)

        # very occasionally mix two random families (adds diversity)
        if np.random.rand() < 0.07:
            # second random shape
            p2 = np.random.uniform(0.15, 2.0)
            family2 = np.random.choice(["power", "plateau"])
            if family2 == "power":
                alpha2 = np.random.uniform(0.0, 8.0)
                shape2 = _power_law_sequence(n, p2, alpha2)
            else:
                L2 = np.random.randint(0, n)
                shape2 = _plateau_power_law_sequence(n, p2, L2)
            w = np.random.rand()
            cand = w * cand + (1 - w) * shape2
            cand = _project_onto_simplex(cand)

        score = _score(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[explore] trial={trials:5d} fam={family} p={p:.4f} "
                  f"{'L=' + str(L) if family == 'plateau' else 'α=' + f'{alpha:.2f}'} "
                  f"→ score={score:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 4) Refinement – projected sub‑gradient descent + occasional
    #    random mass transfers and a single LP warm‑start.
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    cur_score = best_score
    step = 0.7
    step_min = 1e-6
    decay = 0.99993
    iter_cnt = 0
    last_improve = time.time()

    while time.time() < deadline:
        iter_cnt += 1

        # ----- sub‑gradient step -----
        cand = _subgradient_step(seq, step)
        cand_score = _score(cand)

        # ----- accept if better; otherwise perform a small random move -----
        if cand_score < cur_score:
            seq, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"[grad] iter={iter_cnt:6d} score={best_score:.6f}")
                last_improve = time.time()
        else:
            # occasional random perturbation to escape flat regions
            if np.random.rand() < 0.12:
                cand2 = _random_mass_transfer(seq)
                cand2_score = _score(cand2)
                if cand2_score < cur_score:
                    seq, cur_score = cand2, cand2_score
                    if cand2_score < best_score:
                        best_score, best_seq = cand2_score, cand2.copy()
                        print(f"[rand] iter={iter_cnt:6d} score={best_score:.6f}")
                        last_improve = time.time()

        # ---- step‑size schedule ----
        if iter_cnt % 500 == 0:
            step = max(step * decay, step_min)

        # ---- occasional LP step (once every few thousand iters) ----
        if iter_cnt % 4000 == 0:
            # compute current maximal convolution value M
            M, _ = _conv_max_and_arg(seq)
            lp_candidate = _lp_improvement(seq, M)
            lp_score = _score(lp_candidate)
            if lp_score < cur_score - 1e-7:   # take only a clear improvement
                seq, cur_score = lp_candidate, lp_score
                if lp_score < best_score:
                    best_score, best_seq = lp_score, lp_candidate.copy()
                    print(f"[LP] iter={iter_cnt:6d} score={best_score:.6f}")
                    last_improve = time.time()

        # ---- restart if stagnation > 30 s ----
        if time.time() - last_improve > 30.0:
            # re‑initialise from a fresh random power‑law
            p = np.random.uniform(0.2, 1.8)
            alpha = np.random.uniform(0.0, 6.0)
            seq = _power_law_sequence(n, p, alpha)
            cur_score = _score(seq)
            last_improve = time.time()
            print(f"[restart] iter={iter_cnt:6d} new p={p:.3f} α={alpha:.3f}")

        # ---- early exit on target ----
        if best_score < 1.503:
            break

    print(f"[final] iterations={iter_cnt}, best_score={best_score:.6f}")
    # The returned list must contain plain Python floats
    return best_seq.tolist()
