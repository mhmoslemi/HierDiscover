Propose a non‑negative step‑function sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The search proceeds in three stages:

1. **Initialisation** – use the provided global construction if present,
   otherwise fall back to the classic root law a_i ∝ 1/√(i+1).

2. **Broad random search** – explore three flexible families of densities:
   * power‑law   a_i ∝ (i+α)^‑p
   * beta‑law    a_i ∝ x^{a‑1}(1‑x)^{b‑1}   (x = (i+½)/n)
   * Dirichlet   a_i ∝ Gamma(γ) samples
   Every candidate is evaluated with a fast FFT‑based score and the best
   one is kept.

3. **Refinement** – a projected gradient descent that reduces the
   supremum of the self‑convolution.  The active set consists of the
   K largest convolution entries; the gradient is the (sub‑)gradient of
   the maximum over that set.  After each step we project onto the
   probability simplex (non‑negative, unit sum).  Occasionally a random
   mass‑transfer move is performed to escape shallow plateaus.  The
   step size is adapted automatically.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time

import numpy as np

# ----------------------------------------------------------------------
# Simple helper utilities
# ----------------------------------------------------------------------
def _initial_root_sequence(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)  (unit‑sum)."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _power_law_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """a_i ∝ (i+α+1)^‑p  (unit‑sum)."""
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _beta_sequence(n: int, a_param: float, b_param: float) -> np.ndarray:
    """
    a_i ∝ x^{a‑1} (1‑x)^{b‑1} with x centred in the i‑th bin.
    (unit‑sum).  Small epsilons avoid 0^negative‑exponent.
    """
    eps = 1e-12
    # centre of each bin
    x = (np.arange(n, dtype=float) + 0.5) / n
    x = np.clip(x, eps, 1.0 - eps)
    a = (x ** (a_param - 1.0)) * ((1.0 - x) ** (b_param - 1.0))
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _dirichlet_sequence(n: int, gamma_shape: float) -> np.ndarray:
    """Sample Gamma(gamma_shape) i.i.d. and normalise (unit‑sum)."""
    v = np.random.gamma(gamma_shape, 1.0, size=n)
    s = v.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return v / s


def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Linear convolution seq * seq (real FFT, length 2·n‑1)."""
    n = seq.size
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:size]
    return conv


def _score(seq: np.ndarray) -> float:
    """
    Fast objective: 2·n·max(conv) / (sum(seq)^2)
    The sequence is assumed non‑negative and reasonably bounded
    (the clipping performed inside `evaluate_sequence` never triggers).
    """
    n = seq.size
    conv = _conv_full(seq)
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:                 # safety net matching `evaluate_sequence`
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_of_max(conv: np.ndarray, seq: np.ndarray,
                        eps: float = 1e-4) -> np.ndarray:
    """
    Approximate sub‑gradient of max(conv) with respect to seq.
    All convolution entries within `eps` of the current maximum are
    treated as active and their individual gradients are summed.
    """
    n = seq.size
    max_val = conv.max()
    # active set: entries whose value is within eps·max_val of the maximum
    active = np.where(conv >= max_val - eps * max_val)[0]
    grad = np.zeros_like(seq)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += seq[k - idx]          # contribution from this k
    return grad

                seq = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    print(f"[sa] iter={it:6d} T={T:.4f} → best={best_score:.6f}")

        # periodic step‑size decay (prevent step from stagnating at large values)
        if it % 500 == 0:
            step = max(step * step_decay, min_step)

        # early stop if we have reached the target bound
        if best_score < 1.503:
            break

    total_time = time.time() - start_time
    print(f"[final] it={it}, best_score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
