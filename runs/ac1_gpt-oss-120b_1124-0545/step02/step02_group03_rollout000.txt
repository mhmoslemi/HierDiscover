def project_simplex(v):
    """Project onto probability simplex (sum to 1, nonnegative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
A multiscale Adam‑based optimizer for the max‑autocorrelation problem.

The algorithm works on a sequence a that lives on the probability simplex
(Σa = 1, a_i ≥ 0).  For a given length n it repeatedly:

    1. evaluates the full convolution b = a * a,
    2. builds a smooth approximation of   max(b)   via a soft‑max
       with a temperature that is slowly increased,
    3. computes the exact gradient of  Σ_i w_i·b_i  (w is the soft‑max)
       – the gradient is 2·conv(w, a_rev)  restricted to the central n
       entries,
    4. performs an Adam update and projects back onto the simplex.

The optimizer is run on a hierarchy of grid sizes (128 → 256 → … → N),
each level being initialised by linear interpolation of the solution from
the previous level.  Simple random “mass‑transfer’’ moves and occasional
re‑starts guard against stagnation.  The best sequence found before the
deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Helper utilities (no closures, top‑level only)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of u and v using a padded FFT of length ``size``."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    # keep the linear‑convolution part only
    return w[: len(u) + len(v) - 1]


def _max_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full convolution a*a (length 2·len(a)‑1)."""
    return _conv_fft(a, a, size)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | Σx = 1, x_i ≥ 0}.
    """
    if np.all(v <= 0):
        # uniform fallback – avoids division by zero
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0

    if not np.any(cond):
        # uniform fallback (should not happen)
        return np.full_like(v, 1.0 / v.size)

    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

            a_rev = a[::-1]
            conv_wr = _conv_fft(w, a_rev, size2)          # length 3n‑2
            grad = 2.0 * conv_wr[n - 1 : n - 1 + n]       # central n entries

            # ----------------------------------------------------------
            #  4) Adam update (minimise, hence step opposite to grad)
            # ----------------------------------------------------------
            it1 = it + 1
            m = beta1 * m + (1.0 - beta1) * grad
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it1)
            v_hat_corr = v_hat / (1.0 - beta2 ** it1)

            step = base_lr * m_hat / (np.sqrt(v_hat_corr) + eps)
            a = a - step
            a = _project_simplex(a)

            # ----------------------------------------------------------
            #  5) Occasionally perform a cheap random mass‑transfer
            # ----------------------------------------------------------
            if np.random.rand() < 0.015:
                # pick donor and receiver
                i, j = np.random.randint(n, size=2)
                while i == j:
                    j = np.random.randint(n)
                delta = 0.07 * a[i] * np.random.rand()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

            # ----------------------------------------------------------
            #  6) Keep track of the best sequence seen so far
            # ----------------------------------------------------------
            cur_score = 2.0 * n * max_b               # sum(a)=1 → denominator =1
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                last_global_improve = time.time()
                print(f"[n={n:5d} it={it:7d}] new GLOBAL best = {best_score:.6f}")

            if cur_score < local_best_score - 1e-12:
                local_best_score = cur_score
                local_best_seq = a.copy()
                last_improve = time.time()

            # ----------------------------------------------------------
            #  7) Simple stagnation detection → random restart
            # ----------------------------------------------------------
            if time.time() - last_improve > 20.0:
                a = np.random.rand(n)
                a = _project_simplex(a)
                m.fill(0.0)
                v_hat.fill(0.0)
                last_improve = time.time()
                print(f"[n={n:5d} it={it:7d}] restart (stagnation)")

            it += 1

        # --------------------------------------------------------------
        #  End of current resolution – keep the best local solution
        # --------------------------------------------------------------
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
        print(f"[level n={n}] best score this level = {local_best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best_score={best_score:.6f}, time={total_time:.1f}s, length={len(best_seq)}")
    # ``evaluate_sequence`` caps numbers at 1000; our entries are ≤1, so just return them.
    return best_seq.tolist()
