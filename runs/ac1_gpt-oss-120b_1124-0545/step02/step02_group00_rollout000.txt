"""
Propose a non‑negative step function that (empirically) beats the
previous 1.515386 bound.

The algorithm has three phases:

1. **Fast parametric sweep** – generate a few families of “U‑shaped”
   sequences (inverse‑product, inverse‑sum, and a mixture with the
   uniform distribution).  The families are controlled by a small set
   of continuous parameters (exponent and offset).  All candidates are
   projected onto the probability simplex (∑a_i = 1) and the best one
   (lowest score) is kept.

2. **Momentum based sub‑gradient descent** – for the current sequence
   `a`, locate the index `k_max` where the convolution `a*a` attains its
   maximum.  The sub‑gradient of the peak w.r.t. `a` is
   `g_i = 2·a_{k_max‑i}` (zero outside the overlapping range).  A
   momentum term `m ← β·m + (1‑β)·g` smooths the updates.
   The step `a ← a – η·m` is taken and the result is projected back onto
   the simplex.

3. **Exploration & restarts** – with a small probability a cheap random
   mass‑transfer move is performed; if no improvement is seen for a
   while the current state is jittered and the momentum is reset.

Both the temperature and the step‑size follow an exponential schedule
so that early iterations are exploratory and later ones perform
fine‑grained refinement.  The whole routine respects the 1000 s budget
and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative vector a.
    The sequence is assumed to lie on the probability simplex
    (sum(a) = 1) but the formula works for any positive sum.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = np.convolve(a, a)          # linear convolution (zero‑padded)
    max_b = b.max()
    return float(2.0 * len(a) * max_b / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    Implementation follows the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Transfer a small amount of mass from a random index i to a random
    index j and project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    # ensure i != j
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    # delta is a fraction of the source mass
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def gradient_step_with_momentum(a: np.ndarray,
                                momentum: np.ndarray,
                                step_scale: float,
                                beta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    One projected sub‑gradient step using a simple momentum term.
    Returns the new vector and the updated momentum.
    """
    n = a.shape[0]
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # build sub‑gradient g_i = 2 * a_{k_max-i}
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k_max - idx]

    # momentum update
    m_new = beta * momentum + (1.0 - beta) * g
    a_new = a - step_scale * m_new
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_simplex(a_new)
    return a_new, m_new


def parametric_candidates(n: int) -> list[np.ndarray]:
    """
    Generate a modest collection of analytically motivated candidates.
    Families:
        • product‑inverse power: a_i ∝ ((i+c)*(n‑1‑i+c))^{‑p/2}
        • sum‑inverse power:   a_i ∝ (i+c)^{‑p} + (n‑1‑i+c)^{‑p}
        • mixture of the classic arcsine shape with uniform mass.
    All candidates are normalised to sum to 1.
    """
    i = np.arange(n, dtype=np.float64)
    candidates = []

    # 1) product‑inverse power (U‑shape, “arcsine” family)
    for p in np.linspace(0.2, 1.2, 11):
        for c in (0.0, 0.1, 0.5, 1.0, 2.0):
            # avoid zero in denominator
            a = ((i + c + 0.5) * (n - i - c - 0.5)) ** (-p / 2.0)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            candidates.append(a)

    # 2) sum‑inverse power (two tails added)
    for p in np.linspace(0.2, 1.5, 13):
        for c in (0.0, 0.5, 1.0):
            a = (i + c + 1.0) ** (-p) + (n - 1 - i + c) ** (-p)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            candidates.append(a)

    # 3) mixture of classic arcsine shape with uniform component
    base = ((i + 0.5) * (n - i - 0.5)) ** (-0.5)
    base /= base.sum()
    uniform = np.ones(n) / n
    for w in np.linspace(0.5, 1.0, 11):
        a = w * base + (1.0 - w) * uniform
        a /= a.sum()
        candidates.append(a)

    # 4) if a previous best is available via a global variable, include it
    try:
        from __main__ import height_sequence_1  # type: ignore
        a = np.asarray(height_sequence_1, dtype=np.float64)
        if a.ndim == 1 and a.size > 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            candidates.append(a)
    except Exception:
        pass

    return candidates

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function of length ``n`` (default 2000) whose
    evaluation value is as low as possible.  The routine first scans a
    small parametric family, then refines the best candidate with a
    momentum‑based projected sub‑gradient descent and occasional random
    mass‑transfer moves.  The best sequence found before the time budget
    expires is returned.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin so we finish cleanly before the hard deadline
    deadline = start + budget_s - 0.5

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1 – quick parametric sweep
    # ------------------------------------------------------------------
    cand_pool = parametric_candidates(n)
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(cand_pool):
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        if idx % 10 == 0:
            print(f"[init] candidate {idx:03d} score = {sc:.6f}")

    # ensure we have a starting point
    a = best_seq.copy()
    print(f"[init] chosen start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – optimisation loop (momentum sub‑gradient + random perturbations)
    # ------------------------------------------------------------------
    # annealing / step‑size schedules (exponential decay)
    init_temp = 5.0
    final_temp = 0.001
    init_step = 0.3
    final_step = 5e-5
    beta = 0.9                     # momentum decay

    momentum = np.zeros_like(a)
    iteration = 0
    last_improve = start

    # Main optimisation loop
    while time.time() < deadline:
        elapsed = time.time() - start
        # progress ∈ [0,1]
        progress = elapsed / (budget_s - 5.0)
        progress = min(max(progress, 0.0), 1.0)

        # exponential schedules
        temp = init_temp * (final_temp / init_temp) ** progress
        step_scale = init_step * (final_step / init_step) ** progress

        # decide move type
        if np.random.rand() < 0.85:
            # momentum sub‑gradient step
            cand, momentum = gradient_step_with_momentum(
                a, momentum, step_scale, beta)
        else:
            # cheap random mass transfer
            cand = random_move(a, step_scale * 0.5)

        cand_score = compute_score(cand)

        # Metropolis‑style acceptance (lower is better)
        if cand_score < best_score or np.random.rand() < math.exp(
                -(cand_score - best_score) / max(temp, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f} "
                      f"(t={elapsed:.1f}s)")

        # periodic stagnation restart
        if iteration % 50000 == 0 and (time.time() - last_improve) > 30.0:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            momentum = np.zeros_like(a)
            print(f"[{iteration:07d}] restart due to stagnation")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, "
          f"time={total_time:.1f}s")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
