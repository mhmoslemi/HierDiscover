"""
Propose a step‑function that minimizes
    2·n·max(conv(a,a)) / (∑a)²
using a fast projected Adam optimizer on a log‑sum‑exp surrogate of the
self‑convolution, followed by a tiny hill‑climbing phase.
The optimizer works on the ℓ₁‑simplex (∑a = √(2·n)) and never produces
negative heights.  Convolutions are evaluated with an FFT based O(n log n)
routine, allowing many thousands of iterations inside the 1000 s budget.
"""
import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project non‑negative vector v onto the simplex { w ≥ 0 , ∑ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using an FFT (real‑to‑real)."""
    size = a.size + b.size - 1
    # next power‑of‑two for efficient FFT
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)[:size]
    # small negative artefacts from rounding are harmless for the max
    return conv

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)                 # keep non‑negative
        a_new = _simplex_projection(a_new, target_sum)

        a = a_new

        # ---- occasional random perturbations -------------------------------
        if it % 400 == 0 and it > 0:
            # blend the best known solution with a fresh random vector
            rnd = np.random.exponential(scale=1.0, size=n)
            rnd = rnd / rnd.sum() * target_sum
            mix = random.random()
            a_pert = mix * best_seq + (1.0 - mix) * rnd
            a_pert = _simplex_projection(np.maximum(a_pert, 0.0), target_sum)
            conv_pert = _full_convolve(a_pert, a_pert)
            score_pert = float(conv_pert.max())
            if score_pert < best_score:
                best_score = score_pert
                best_seq = a_pert.copy()
                last_improve = it
                print(f"[{it:6d}] New best after perturbation: {best_score:.6f}")
            if score_pert < cur_score:
                a = a_pert

        # ---- restart if stagnation -----------------------------------------
        if it - last_improve > 5000:
            noise = np.random.exponential(scale=1.0, size=n)
            noise = noise / noise.sum() * target_sum
            a = 0.5 * best_seq + 0.5 * noise
            a = _simplex_projection(a, target_sum)
            last_improve = it

        it += 1

    # ----------------------------------------------------------------------
    # Tiny hill‑climbing phase (uses remaining time)
    remaining = deadline - time.time()
    if remaining > 1.0:
        attempts = int(remaining * 50)          # roughly 50 tiny moves per second
        for _ in range(attempts):
            conv = _full_convolve(a, a)
            max_val = conv.max()
            # pick a random index among all maximal positions
            max_idxs = np.where(conv == max_val)[0]
            k = random.choice(max_idxs.tolist())

            # indices that contribute to this shift
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_candidates = np.arange(i_start, i_end + 1)
            j_candidates = k - i_candidates
            contrib = a[i_candidates] * a[j_candidates]
            i_src = i_candidates[np.argmax(contrib)]

            # destination – the current smallest entry (away from the peak)
            i_dst = int(np.argmin(a))
            if i_src == i_dst:
                continue

            # small transfer of mass
            delta = min(a[i_src] * 0.05, target_sum * 0.001)
            if delta <= 0:
                continue

            a_trial = a.copy()
            a_trial[i_src] -= delta
            a_trial[i_dst] += delta
            if a_trial[i_src] < 0:
                continue

            new_conv = _full_convolve(a_trial, a_trial)
            new_max = new_conv.max()
            if new_max + 1e-12 < best_score:
                best_score = new_max
                best_seq = a_trial.copy()
                a = a_trial
                print(f"[Hill] Improved max(conv) = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best sequence found (plain Python list)
    return best_seq.tolist()
