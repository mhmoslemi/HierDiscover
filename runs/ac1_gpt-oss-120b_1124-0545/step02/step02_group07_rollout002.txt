def _smooth_step(a, epsilon, eta):
    n = len(a)
    conv = _conv_fft(a) # length 2n-1

    # compute weight w = exp(conv/epsilon - max(conv)/epsilon) for numeric stability
    max_conv = conv.max()
    scaled = (conv - max_conv) / epsilon
    w = np.exp(scaled)
    sum_w = w.sum()
    # compute gradient via convolution of a and w_rev
    # w_rev = w[::-1]
    # compute conv2 = np.convolve(a, w_rev) using FFT for speed
    # conv2 length len(a)+len(w_rev)-1 = n + 2n-1 -1 = 3n-2
    # Use FFT of size appropriate power of two.
    total_len = len(a) + len(w) - 1
    N = 1 << ((total_len).bit_length())
    fa = np.fft.rfft(a, n=N)
    fw_rev = np.fft.rfft(w[::-1], n=N)
    conv2 = np.fft.irfft(fa * fw_rev, n=N)[:total_len]
    # compute gradient g_i = 2 * conv2[2*n-2 - i] / sum_w
    idx_start = 2 * n - 2
    g = 2.0 * conv2[idx_start - np.arange(n)] / sum_w
    # update a via exponentiated gradient
    factor = np.exp(-eta * g / (np.max(np.abs(g)) + 1e-12))
    a_new = a * factor
    a_new /= a_new.sum()
    return a_new, conv, np.log(sum_w) * epsilon

def _smooth_optimize(start_seq, deadline, rng, target_score=1.503):

    a = start_seq.copy()
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    start_time = time.time()
    total_budget = deadline - start_time

    iteration = 0

    while time.time() < deadline:
        # compute conv via FFT
        N = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(fft_a * fft_a, n=N)[:2*n-1]

        max_conv = conv.max()
        # evaluate score approximated (exact uses 2n*max_conv)
        approx_score = 2.0 * n * max_conv

        # compute exact at some interval (every 10 iterations) to reduce calls
        if iteration % 20 == 0:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}s] New best in smooth opt: {best_score:.6f}")
                if best_score <= target_score:
                    break

        # Check if approx score already worse than best_score * 1.2 maybe skip
        # else continue

        # compute smoothing epsilon
        elapsed = time.time() - start_time
        frac = elapsed / total_budget
        eps_factor = 0.5 * (1 - frac) + 0.02 * frac  # linearly decrease from 0.5 to 0.02

        fft_a2 = np.fft.rfft(a, n=N2)
        conv2 = np.fft.irfft(fft_a2 * fft_w_rev, n=N2)
        # conv2 length = n + len(w_rev) -1 = 3n-2 ; we need portion up to 2n-2 maybe

        # exponentiated gradient update
        max_abs_g = np.max(np.abs(g)) + 1e-12

        a_new = a * factor

        if a_new_sum == 0.0:

            delta = min(a_new[i], 0.1 * a_new[i])
            a_new[i] -= delta
            a_new[j] += delta

        a = a_new
        iteration += 1

Algorithm
---------
1. **Analytic families** – generate many parametric shapes (constant,
   linear, exponential, power‑law, beta‑like, plateau‑exponential,
   double‑exponential, …) for a few lengths (2048‑8192).  Each raw shape
   is normalised to have unit ℓ₁‑norm, its convolution is obtained via an
   FFT and a cheap homogeneous score `2·n·max(conv)` is computed.
   The best few candidates (default 5) are kept for refinement.

2. **Smoothed max‑convolution descent** – starting from a candidate,
   minimise the convex objective  

        L(a) = max_k (a∗a)_k                         (∑a = 1, a≥0)

   by approximating the maximum with a log‑sum‑exp smoothing:

        L_ε(a) = ε·log( Σ_k exp((a∗a)_k / ε) ).

   For a given ε the gradient can be written in closed form and is
   computed efficiently with two FFTs (one for a∗a and one for
   a convolved with the soft‑max weights).  An exponentiated‑gradient
   update keeps the iterate non‑negative; a simple ℓ₁‑normalisation
   projects back onto the simplex.  The smoothing parameter ε is
   decreased linearly from 0.5·max(conv) to 0.02·max(conv) while the
   step‑size η decays from 0.5 to 0.001.  Occasional random perturbations
   (2 % chance) help escaping plateaus.

   The optimiser runs for the allotted time slice and returns the best
   exact sequence found (checked with the official evaluator).

3. **Random restarts** – if time remains, new random simplex points are
   generated and refined with the same smoothed optimiser.

The whole procedure respects the global 1000 s budget and always returns
the best sequence discovered.  All numbers are non‑negative floats, and
the returned list is ready for `evaluate_sequence`.  The code is pure
NumPy and runs on a single thread (well within the 2‑CPU limit).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """
    Scale the array so that its maximum is ≤ 0.9 and its total sum is ≥ 0.01.
    The scaling does NOT affect the evaluation score because the objective
    is homogeneous.
    """
    a = np.asarray(a, dtype=float)
    max_v = a.max()
    if max_v > 0.0:
        a = a * (0.9 / max_v)          # keep values comfortably below 1
    total = a.sum()
    if total < 0.01:
        if total > 0.0:
            a = a * (0.01 / total)
        else:  # all zeros – give a tiny uniform bump
            a = a + (0.01 / a.size)
    return a


def _candidate_shapes_ext(n: int):
    """
    Yield a stream of raw (unnormalised) shape vectors of length `n`.
    The shapes include classic families and a few newly added ones.
    """
    # 1. constant
    yield np.ones(n, dtype=float)

    # 2. linear (decreasing / increasing)
    yield np.arange(n, 0, -1, dtype=float)
    yield np.arange(1, n + 1, dtype=float)

    # 3. quadratic (decreasing / increasing)
    yield np.square(np.arange(n, 0, -1, dtype=float))
    yield np.square(np.arange(1, n + 1, dtype=float))

    # 4. triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    yield np.minimum(t, n - t + 1)

    # 5. valley (large at ends, small in centre)
    centre = (n - 1) / 2.0
    yield 1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2))

    # 6. sine‑squared shape
    yield np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2

    # 7. classical exponential decays
    for r in (0.95, 0.9, 0.8, 0.7, 0.6, 0.5):
        yield np.power(r, np.arange(n, dtype=float))

    # 8. “slow” exponential decays a_i = exp(-c·i / n)
    for c in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0):
        r = np.exp(-c / n)
        yield np.power(r, np.arange(n, dtype=float))

    # 9. power‑law decays
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        yield 1.0 / (i ** p)

    # 10. beta‑like shapes a_i = (i+1)^α·(n‑i)^β
    for alpha in (0.5, 1.0, 2.0):
        for beta in (0.5, 1.0, 2.0):
            i = np.arange(1, n + 1, dtype=float)
            yield (i ** alpha) * ((n - i + 1) ** beta)

    # 11. plateau‑exponential (flat start, then decay)
    for plateau_frac in (0.1, 0.25, 0.5):
        p = int(plateau_frac * n)
        for r in (0.5, 0.7, 0.9):
            tail = np.power(r, np.arange(n - p, dtype=float))
            yield np.concatenate([np.ones(p, dtype=float), tail])

    # 12. double‑exponential (two different decay rates)
    for split_frac in (0.3, 0.5, 0.7):
        split = int(split_frac * n)
        for r1 in (0.9, 0.7):
            for r2 in (0.5, 0.3):
                left = np.power(r1, np.arange(split, dtype=float))
                right = np.power(r2, np.arange(n - split, dtype=float))
                yield np.concatenate([left, right])


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a∗a using an FFT of length a power‑of‑two.
    Returns the full array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)
    return conv[: 2 * n - 1]


def _exp_gradient_step(a: np.ndarray,
                      conv: np.ndarray,
                      epsilon: float,
                      eta: float,
                      N_grad: int) -> np.ndarray:
    """
    Perform a single smoothed exponentiated‑gradient update.
    Parameters
    ----------
    a         : current simplex‑normalised vector (sum = 1)
    conv      : a∗a (already computed, length 2n‑1)
    epsilon   : smoothing parameter (positive)
    eta       : step size
    N_grad    : FFT length for the gradient convolution (≥ 3n‑2, power‑of‑two)
    Returns
    -------
    a_new     : updated vector, again normalised to sum = 1
    """
    n = a.shape[0]
    max_conv = conv.max()

    # soft‑max weights w_k = exp((conv_k - max)/epsilon)  (stable)
    w = np.exp((conv - max_conv) / epsilon)
    sum_w = w.sum()
    w_rev = w[::-1]                     # needed for the gradient

    # FFTs for the weighted convolution (a * w_rev)
    fft_a = np.fft.rfft(a, n=N_grad)
    fft_w = np.fft.rfft(w_rev, n=N_grad)
    conv2 = np.fft.irfft(fft_a * fft_w, n=N_grad)
    conv2 = conv2[: 3 * n - 2]          # exact length we need

    idx_start = 2 * n - 2               # mapping index where g_i lives
    idxs = idx_start - np.arange(n)
    g = 2.0 * conv2[idxs] / sum_w       # smoothed sub‑gradient

    # exponentiated‑gradient update (keep a ≥ 0)
    max_abs = np.max(np.abs(g)) + 1e-12
    factor = np.exp(-eta * g / max_abs)
    a_new = a * factor
    total = a_new.sum()
    if total == 0.0:
        a_new = a.copy()
    else:
        a_new = a_new / total
    return a_new


def _smooth_optimize(start_seq: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator,
                     target_score: float = 1.503) -> np.ndarray:
    """
    Smoothed exponentiated‑gradient optimisation of the max‑convolution
    objective.  The input `start_seq` must already be on the simplex
    (non‑negative, sum = 1).  The routine returns the best sequence found
    before the deadline (as a plain Python list of floats).
    """
    a = start_seq.copy()
    n = a.shape[0]

    # Pre‑compute FFT lengths
    N_conv = 1 << ((2 * n - 1).bit_length())      # for a∗a
    N_grad = 1 << ((3 * n - 2).bit_length())      # for gradient

    # Track the globally best sequence (exact evaluation)
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    start_time = time.time()
    total_budget = max(deadline - start_time, 1e-9)

    iteration = 0
    while time.time() < deadline:
        # ----- compute a∗a -------------------------------------------------
        fft_a = np.fft.rfft(a, n=N_conv)
        conv = np.fft.irfft(fft_a * fft_a, n=N_conv)
        conv = conv[: 2 * n - 1]          # length 2n‑1
        max_conv = conv.max()

        # ----- keep exact best occasionally (every 20 iters) -------------
        if iteration % 20 == 0:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:.1f}s] New best (smooth) score={best_score:.6f}")
                if best_score <= target_score:
                    print("Target reached – terminating smoothed optimisation.")
                    break

        # ----- smoothing parameter ε (decrease linearly) ---------------
        elapsed = time.time() - start_time
        frac = elapsed / total_budget
        eps_factor = 0.5 * (1.0 - frac) + 0.02 * frac   # 0.5 → 0.02
        epsilon = max(1e-15, max_conv * eps_factor)

        # ----- step‑size η (decrease linearly) -------------------------
        eta = 0.5 * (1.0 - frac) + 0.001 * frac

        # ----- perform one exponentiated‑gradient step -----------------
        a = _exp_gradient_step(a, conv, epsilon, eta, N_grad)

        # ----- occasional random perturbation ---------------------------
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            delta = min(a[i], 0.1 * a[i])
            a[i] -= delta
            a[j] += delta
            # re‑normalise (still on simplex)
            a = a / a.sum()

        iteration += 1

    # Final exact check for the last iterate
    final_exact = evaluate_sequence(a.tolist())
    if final_exact < best_score:
        best_score = final_exact
        best_a = a.copy()
        print(f"[{time.time() - start_time:.1f}s] Final smoothed best score={best_score:.6f}")

    return best_a.tolist()


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient list that minimises `evaluate_sequence`.
    The function respects the given time budget (≈1000 s) and returns the
    best sequence discovered so far.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(np.asarray(hs1, dtype=float))
        seq = seq / seq.sum()                 # simplex normalisation
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from "
                  f"height_sequence_1 → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic scan of analytic families
    # ------------------------------------------------------------------
    # We'll examine a couple of lengths; the longest (8192) gave the best
    # historical results, but we also probe a mid‑size length for cheap.
    lengths = [4096, 8192]
    top_candidates = []          # list of (approx_score, seq_array)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.1f}s] Scanning analytic families, n={n}")

        for raw in _candidate_shapes_ext(n):
            if time.time() >= deadline:
                break
            seq = _normalise_seq(raw)
            # normalise to unit ℓ₁ – the score is homogeneous
            seq = seq / seq.sum()
            conv = _conv_fft(seq)
            approx_score = 2.0 * n * conv.max()
            # Keep only promising candidates (within a small factor)
            if approx_score < best_score * 1.10:
                # store the raw numpy array for later refinement
                top_candidates.append((approx_score, seq.copy()))
            # retain at most 8 best candidates overall
            if len(top_candidates) > 12:
                top_candidates.sort(key=lambda x: x[0])
                top_candidates = top_candidates[:12]

    # Ensure we have something to start from
    if not top_candidates:
        n = 8192
        a0 = rng.random(n)
        a0 = a0 / a0.sum()
        top_candidates = [(2.0 * n * _conv_fft(a0).max(), a0)]

    # Sort candidates by the cheap homogeneous score
    top_candidates.sort(key=lambda x: x[0])
    # Keep only the very best few for the expensive refinement phase
    top_candidates = top_candidates[:5]

    # ------------------------------------------------------------------
    # Phase 2 – refined optimisation on the best candidates
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        # allocate an equal time slice to each candidate (but at least 5 s)
        per_cand = max(5.0, remaining / len(top_candidates))
        for approx, cand_arr in top_candidates:
            sub_deadline = min(deadline, time.time() + per_cand)
            refined = _smooth_optimize(cand_arr, sub_deadline, rng)
            # exact evaluation of the refined sequence
            exact = evaluate_sequence(refined)
            if exact < best_score:
                best_score = exact
                best_seq = refined
                print(f"[{time.time() - start_time:.1f}s] Updated global best (refine) "
                      f"score={best_score:.6f}")
            if best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # Phase 3 – random restarts (if any time left)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0 and best_score > 1.503:
        # Use the length of the current best (or fall back to 8192)
        cur_len = len(best_seq) if best_seq is not None else 8192
        a_rand = rng.random(cur_len)
        a_rand = a_rand / a_rand.sum()
        sub_deadline = min(deadline, time.time() + 15.0)
        refined = _smooth_optimize(a_rand, sub_deadline, rng)
        exact = evaluate_sequence(refined)
        if exact < best_score:
            best_score = exact
            best_seq = refined
            print(f"[{time.time() - start_time:.1f}s] Random restart improved "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity – return a sequence that obeys the API guarantees
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution over the chosen length
        n = 8192
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # Apply the generic normaliser one last time (caps values, avoids tiny sum)
    best_seq_np = _normalise_seq(np.asarray(best_seq, dtype=float))
    print(f"[{time.time() - start_time:.1f}s] Finished – best score={best_score:.6f}")
    return [float(x) for x in best_seq_np.tolist()]
