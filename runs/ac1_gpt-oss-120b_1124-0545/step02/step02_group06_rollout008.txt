def _simplex_projection(v):
    # ensure sum(v) = 1, v_i >= 0
    # using algorithm from https://arxiv.org/pdf/1101.6081.pdf
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _gradient_step(seq, step):
    n = len(seq)
    max_b, k = _conv_max_and_arg(seq)
    lo = max(0, k-(n-1))
    hi = min(k, n-1)
    idx = np.arange(lo, hi+1)
    grad = np.zeros_like(seq)
    grad[idx] = seq[k-idx]
    grad = grad - grad.mean()  # enforce zero sum component so step doesn't change sum
    cand = seq - step * grad
    cand = np.maximum(cand, 0.0)
    total = cand.sum()
    if total > 0:
        cand /= total
    else:
        cand = seq.copy()
    return cand
```

Propose a non‑negative step‑function sequence with a low evaluation score.

Algorithm
---------
1. **Candidate generation** – multiple starting shapes are created:
   * the classic root law a_i ∝ 1/√(i+1);
   * uniform and random Dirichlet vectors;
   * truncated / shifted power‑law families a_i ∝ (i+α+1)^‑p
     with p≈0.5, α in a wide range;
   * (if available) the globally supplied `height_sequence_1`.

2. **Local optimisation** – for every candidate a projected gradient
   descent on the quantity `max(conv(a,a))` is performed.
   * The active shift `k* = argmax conv` defines the sub‑gradient
     `g_j = 2·a_{k*‑j}` (zero outside the overlap).
   * After a gradient step we clip to non‑negative values and
     renormalise to unit L¹‑norm (equivalent to the scaling used in
     `evaluate_sequence`).
   * A simple back‑tracking line search is used to guarantee a
     monotone decrease of the objective.
   * Occasionally a random mass‑transfer move is tried to escape
     shallow local minima.

3. **Time‑bounded search** – the whole procedure runs until the
   allotted budget (≈1000 s) is exhausted, constantly remembering the
   best sequence encountered.

The code is deliberately CPU‑only, uses only NumPy and the standard
library, and respects all constraints of the evaluation harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _initial_root_sequence(n: int) -> np.ndarray:
    """Root law a_i ∝ 1/√(i+1) normalised to unit sum."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _power_law_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Generate a normalised sequence a_i ∝ (i+α+1)^‑p.
    The extra '+1' prevents a division by zero when alpha=0.
    """
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _conv_max_and_arg(seq: np.ndarray) -> tuple[float, int]:
    """
    Return (max(convolution(seq,seq)), argmax) using a real FFT.
    """
    n = seq.size
    size = 2 * n - 1
    # next power of two for a fast real FFT
    m = 1 << ((size - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:size]
    arg = int(np.argmax(conv))
    return float(conv[arg]), arg


def _score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective used by the harness.
    The sequence is assumed non‑negative and already normalised to
    unit sum (this matches the scaling employed inside the search).
    """
    n = seq.size
    max_b, _ = _conv_max_and_arg(seq)
    s = seq.sum()
    if s < 0.01:                     # mirrors the safety check in evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip negative entries, then renormalise to unit L¹‑norm."""
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total > 0.0:
        seq = seq / total
    else:  # should never happen, but keep a valid fallback
        seq = np.full_like(seq, 1.0 / seq.size)
    return seq


def _projected_gradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    Single projected gradient step that attempts to lower the maximal
    entry of the self‑convolution.
    """
    n = seq.size
    _, kmax = _conv_max_and_arg(seq)

    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    indices = np.arange(lo, hi + 1, dtype=int)

    # sub‑gradient of the active convolution entry (factor 2 omitted – absorbed into step)
    grad = np.zeros_like(seq)
    grad[indices] = seq[kmax - indices]

    # keep the update on the simplex (zero‑mean direction)
    grad -= grad.mean()

    cand = seq - step * grad
    cand = _normalize(cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The function respects a wall‑clock budget and returns the best
    sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Choose a handful of sequence lengths (including the global one)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000]
    if "height_sequence_1" in globals():
        lengths.append(len(height_sequence_1))
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # 2) Main search over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[search] starting length n={n}")

        # --------------------------------------------------------------
        # 2a) Build an initial pool of seeds
        # --------------------------------------------------------------
        seeds = []

        # classic root law
        seeds.append(_initial_root_sequence(n))

        # uniform distribution
        seeds.append(np.full(n, 1.0 / n))

        # random Dirichlet (fully dense)
        seeds.append(np.random.dirichlet(np.ones(n)))

        # global construction, trimmed / padded to the current length
        if "height_sequence_1" in globals():
            try:
                tmp = np.array(height_sequence_1, dtype=float)
                if tmp.size > n:
                    tmp = tmp[:n]
                elif tmp.size < n:
                    tmp = np.pad(tmp, (0, n - tmp.size), constant_values=0.0)
                tmp = _normalize(tmp)
                seeds.append(tmp)
            except Exception:
                pass

        # a few hand‑crafted power‑law seeds
        for _ in range(6):
            p = np.random.uniform(0.45, 0.55)      # zoom around the critical ½
            alpha = np.random.uniform(0.0, 10.0)   # allow a noticeable shift
            seeds.append(_power_law_sequence(n, p, alpha))

        # a small deterministic grid around the root law
        for p in np.linspace(0.48, 0.52, 5):
            for alpha in [0.0, 1.0, 3.0, 7.0]:
                seeds.append(_power_law_sequence(n, p, alpha))

        # --------------------------------------------------------------
        # 2b) Optimise each seed independently
        # --------------------------------------------------------------
        for si, init_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            seq = init_seq.copy()
            cur_score = _score(seq)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[init] n={n} seed={si} score={cur_score:.6f}")

            # ---- projected gradient descent with back‑tracking line search ----
            step = 0.5                     # initial step size
            no_improve = 0                 # counter for stagnation

            # a hard limit on iterations per seed; the outer time check is
            # the real stop condition.
            for it in range(800):
                if time.time() >= deadline:
                    break

                # line search – try progressively smaller steps until improvement
                improved = False
                trial_step = step
                for _ in range(12):
                    cand = _projected_gradient_step(seq, trial_step)
                    cand_score = _score(cand)
                    if cand_score < cur_score:
                        improved = True
                        break
                    trial_step *= 0.5

                if improved:
                    seq = cand
                    cur_score = cand_score
                    step = trial_step * 1.2      # gently increase after a win
                    no_improve = 0

                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = seq.copy()
                        print(f"[improve] n={n} seed={si} it={it} score={best_score:.6f}")

                    if best_score < 1.503:       # target reached – stop immediately
                        print("[target] score under 1.503, exiting early")
                        return best_seq.tolist()
                else:
                    # no progress – shrink step and try a random mass move occasionally
                    step *= 0.7
                    no_improve += 1

                    if np.random.rand() < 0.15:
                        i, j = np.random.choice(n, 2, replace=False)
                        eps = min(seq[i], seq[i] * np.random.uniform(5e-4, 5e-2))
                        cand = seq.copy()
                        cand[i] -= eps
                        cand[j] += eps
                        cand = _normalize(cand)
                        cand_score = _score(cand)
                        if cand_score < cur_score:
                            seq = cand
                            cur_score = cand_score
                            if cand_score < best_score:
                                best_score = cand_score
                                best_seq = cand.copy()
                                print(f"[rand] n={n} seed={si} it={it} score={best_score:.6f}")
                            no_improve = 0

                if step < 1e-8 or no_improve > 100:
                    # step too tiny or stuck for a while – give up on this seed
                    break

    # ------------------------------------------------------------------
    # 3) Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n)
        best_score = _score(best_seq)
        print("[fallback] returning uniform sequence")

    print(f"[final] best_score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
