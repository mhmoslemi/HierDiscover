"""
Propose a candidate step‑function sequence (list of non‑negative heights).

Algorithm
---------
* Initialise a feasible sequence (preferably from the global `height_sequence_1`,
  otherwise uniform random) and normalise it to unit ℓ₁‑norm.
* Repeatedly minimise a *smooth* approximation of the peak of the linear
  convolution `a * a`.  The smooth max is realised by a soft‑max weighting
  `w_k ∝ exp((b_k−max(b))/τ)`, where `b = a*a`.  The gradient of the
  smooth objective w.r.t. `a` is `‑2·(a_rev ∗ w)`.  After a gradient step we
  project back onto the probability simplex (clip to ≥0, renormalise).
* With a small probability we perform a random “mass‑transfer’’ move
  (move a fraction of the mass from one index to another) to escape flat
  plateaux.
* If no improvement is observed for a configurable amount of time we
  restart from a fresh random point.
* The best sequence seen (lowest value of
  `evaluate_sequence`) is returned.  The whole search respects the
  1000‑second budget.

The method improves on the plain sub‑gradient approach by using a
soft‑max weighting (a differentiable approximation of the max) which gives
non‑zero gradients even when many convolution entries are close to the
maximum.  Adaptive step‑sizes and occasional random perturbations help to
avoid stagnation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT.  ``size`` must be ≥ len(u)+len(v)‑1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size1: int) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    return _conv_fft(a, a, size1)


def _smooth_gradient_step(
    a: np.ndarray,
    conv_full: np.ndarray,
    a_rev: np.ndarray,
    size2: int,
    step: float,
    tau: float,
) -> np.ndarray:
    """
    Perform one projected gradient descent step for the smooth
    approximation of `max(conv(a, a))`.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1, non‑negative).
    conv_full : np.ndarray
        Full convolution a*a (length 2*n‑1).
    a_rev : np.ndarray
        Reversed copy of ``a``.
    size2 : int
        FFT size for the convolution a_rev * weight (≥ 3*n‑2).
    step : float
        Step size (positive).
    tau : float
        Smoothing temperature; larger values give a flatter weight
        distribution, smaller values approach the hard max.

    Returns
    -------
    np.ndarray
        Updated probability vector (still on the simplex).
    """
    max_b = conv_full.max()
    # Soft‑max weighting; shift by max_b for numerical stability.
    # The exponent is negative for all entries except those attaining the max.
    weight = np.exp((conv_full - max_b) / tau)
    # Normalise the weight to keep the gradient magnitude bounded.
    w_sum = weight.sum()
    if w_sum == 0.0:
        weight = np.ones_like(weight) / weight.size
    else:
        weight /= w_sum

    # Gradient = -2 * (a_rev ∗ weight)   (central n entries)
    g_full = _conv_fft(a_rev, weight, size2)
    n = a.shape[0]
    g = g_full[n - 1 : n - 1 + n]          # central part, length n

    d = -2.0 * g                           # descent direction
    # Remove any component that changes the total mass (project to hyperplane Σd=0)
    d -= d.mean()

    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a random fraction of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

    tau_decay = 0.9995          # multiplicative decay per iteration
    stagnation_seconds = 30.0   # restart if no improvement for this long
    last_improve = start
    it = 0

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # 1. current convolution and objective
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b           # because sum(a)=1

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # 2. decide the move
        if np.random.rand() < rand_prob:
            # cheap random perturbation
            a = _random_transfer(a, rand_scale)
        else:
            # smooth gradient descent step
            # update smoothing temperature (annealing)
            tau = max(tau_factor_initial * max_b, 1e-12)
            # apply exponential decay to the temperature factor
            tau_factor_initial *= tau_decay
            # adaptive step size (1/√it) – guarantees diminishing steps
            step = base_step / np.sqrt(it + 1.0)

            a = _smooth_gradient_step(
                a,
                conv_full,
                a_rev,
                size2,
                step,
                tau,
            )
            a_rev = a[::-1]   # keep the reversed copy up‑to‑date

        # 3. occasional restart if we are stuck
        if (it % 50000 == 0) and (time.time() - last_improve > stagnation_seconds):
            a = np.random.rand(n)
            a /= a.sum()
            a_rev = a[::-1].copy()
            # reset temperature schedule
            tau_factor_initial = max(0.3, tau_factor_initial)
            print(f"[{it:7d}] restart due to stagnation")

        it += 1

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
