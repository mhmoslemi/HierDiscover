"""
Gradient‑descent + soft‑max optimisation for the step‑function problem.

The evaluation function (after normalising the ℓ₁‑norm to √(2 n)) is simply
the maximal entry of the self‑convolution.  We therefore minimise
    max_k (a * a)_k
subject to a_i ≥ 0 and Σ a_i = √(2 n).

Algorithm
---------
* Initialise a with a mixture of the known good construction (if available)
  and a fresh random exponential vector, projected onto the simplex.
* Re‑express the objective as a smooth log‑sum‑exp:
      Fβ(a) = (1/β)·log Σ_k exp(β·c_k)   where c = a∗a .
  Its gradient is 2·(w∗a) where w is the soft‑max over the convolution entries.
  Both the convolution and the gradient are obtained by a pair of FFTs,
  i.e. O(n log n) per iteration.
* Use the Adam optimiser (first‑ and second‑moment adaptivity) together with
  an exponential schedule for β (starting very smooth, ending sharp) and a
  linearly decreasing learning‑rate.
* After each Adam step the vector is projected back onto the probability
  simplex {x ≥ 0, Σ x = √(2 n)} using the O(n log n) algorithm of
  Wang & Carreira‑Perpiñán.
* The best vector seen so far is remembered.  If no improvement occurs for
  a while a small Gaussian perturbation of the current best is performed to
  escape stagnation.
* When the time budget (default 1000 s) is almost exhausted a brief
  hill‑climbing phase (random mass transfers) is executed for polishing.

The routine returns the list of heights achieving the lowest maximal
convolution entry that was discovered within the allotted time.
"""
import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = s}  (Wang‑Carreira‑Perpiñán)."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Should only happen when s == 0
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, nfft: int) -> float:
    """Return the maximal entry of the linear convolution a*a using FFT."""
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _gradient_logsumexp(a: np.ndarray, nfft: int, beta: float,
                       target_sum: float) -> np.ndarray:
    """
    Compute the gradient of the smoothed objective
        Fβ(a) = (1/β)·log Σ_k exp(β·c_k)   with c = a∗a.
    The gradient equals 2·(w∗a) where w is the soft‑max over c.
    """
    # FFT of a
    Af = np.fft.rfft(a, n=nfft)
    # convolution c = a*a
    c = np.fft.irfft(Af * Af, n=nfft)[: 2 * a.size - 1]
    max_c = c.max()
    # soft‑max weights (numerically stable)
    exp_w = np.exp(beta * (c - max_c))
    w = exp_w / exp_w.sum()
    # FFT of the weights
    Wf = np.fft.rfft(w, n=nfft)
    # gradient = 2 * ifft(Af * Wf)  (first n entries correspond to ∂F/∂a)
    grad_full = np.fft.irfft(Af * Wf, n=nfft)
    grad = 2.0 * np.real(grad_full[: a.size])
    return grad, max_c


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the evaluation
    function.  Returns a plain Python list of floats.
    """
    # ------------------------------------------------------------------
    # RNG initialisation
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base = np.array(height_sequence_1, dtype=np.float64)
        base = base / base.sum() * math.sqrt(2.0 * n)
    else:
        n = 2500                     # modest default
        base = np.random.exponential(scale=1.0, size=n)

    # Normalise the base vector to the conventional ℓ₁‑norm √(2 n)
    target_sum = math.sqrt(2.0 * n)
    base = base / base.sum() * target_sum

    # Additional random start (same length) – will be mixed with the base
    rnd = np.random.exponential(scale=1.0, size=n)
    rnd = rnd / rnd.sum() * target_sum

    # Blend the two starts (average) and project onto the simplex
    a = _simplex_projection(0.5 * base + 0.5 * rnd, target_sum)

    # ------------------------------------------------------------------
    # FFT length (next power of two ≥ 2 n‑1)
    nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Adam optimiser state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Hyper‑parameter schedules
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    total_time = deadline - start_time

    # β goes from ~0.1 (very smooth) to ~20 (sharp) linearly in time
    beta_start, beta_end = 0.1, 20.0
    # learning‑rate decays from ≈0.5 to ≈0.001 (absolute)
    lr_start, lr_end = 0.5, 0.001

    # Tracking best solution
    cur_max = _max_convolution(a, nfft)
    best_seq = a.copy()
    best_score = cur_max
    last_improve_iter = 0
    iter_cnt = 0
    max_no_improve = 3000      # after this many iter without progress we perturb

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Update schedules
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac   # linear decay

        # --------------------------------------------------------------
        # Gradient of the smoothed objective
        grad, cur_max = _gradient_logsumexp(a, nfft, beta, target_sum)

        # Adam update
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _simplex_projection(a_new, target_sum)

        # --------------------------------------------------------------
        # Evaluate new candidate (max entry of the self‑convolution)
        new_max = _max_convolution(a_new, nfft)

        # Accept the move (always accept improvement, otherwise keep current)
        if new_max < cur_max:
            a = a_new
            cur_max = new_max
            # record best overall
            if new_max < best_score:
                best_score = new_max
                best_seq = a.copy()
                last_improve_iter = iter_cnt
                print(f"[{iter_cnt:7d}] New best max‑conv {best_score:.6f} (t={elapsed:.1f}s)")
        else:
            # Even if not better we still keep the projected point; this
            # simplifies the code and works well with the Adam dynamics.
            a = a_new
            cur_max = new_max

        # --------------------------------------------------------------
        # Stagnation handling – occasional random kick‑off
        if iter_cnt - last_improve_iter > max_no_improve:
            # perturb around the best solution seen so far
            noise = np.random.normal(scale=target_sum * 0.02, size=n)
            a = best_seq + noise
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a, target_sum)
            # reset Adam statistics
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = iter_cnt
            print(f"[{iter_cnt:7d}] Perturbation applied (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # Final polishing: a short hill‑climbing phase (random transfers)
    # --------------------------------------------------------------
    polishing_deadline = time.time() + min(5.0, budget_s * 0.01)  # ≤5 s
    a = best_seq.copy()
    cur_max = _max_convolution(a, nfft)

        if i == j:
            continue

        a[i] -= delta
        a[j] += delta
        a = _simplex_projection(a, target_sum)
        new_max = _max_convolution(a, nfft)
        if new_max < cur_max:
            cur_max = new_max
            best_seq = a.copy()
            best_score = cur_max
            print(f"[polish] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return best sequence as a plain list of floats
    return best_seq.tolist()
