assistantfinal**Solution Overview**

The score returned by `evaluate_sequence` is  

```
score = 2·n·max(conv(a,a)) / (∑a)²
```

Because the function is scale‑invariant we can work on the probability
simplex `∑a = 1`.  
For a vector `a` the only non‑differentiable part of the objective is the
`max` over the convolution entries.  

The sub‑gradient of `max(conv(a,a))` can be obtained analytically:
for every index `k` attaining the maximal value we need the gradient of the
corresponding convolution entry.  This gradient is the (linear) convolution
of the reversed vector with the indicator of the active set.

The previous baseline used a simple projected sub‑gradient step with a
\(1/\sqrt{t}\) step‑size schedule.  In practice the convergence stalls long
before the optimum is reached.  The new algorithm improves on that in three
ways

1. **Adam–style adaptive step** – we keep first‑ and second‑moment estimates
   of the sub‑gradient, which yields larger steps in directions that have
   consistently small gradients and shrinks steps where the gradient varies
   a lot.

2. **Exponentiated‑gradient (mirror descent) update** – the simplex constraint
   is handled naturally by re‑normalising the exponentiated parameters
   (`a_i ← a_i·exp(−η·g_i)`).  This avoids the crude clipping‑and‑renormalise
   step of the baseline.

3. **More aggressive exploration** – a higher probability of a *mass‑transfer*
   perturbation and an adaptive restart mechanism that fires when no
   improvement has been seen for a configurable amount of time.

All heavy lifting (convolution, FFT, mask creation) stays unchanged and
remains `O(n log n)`.  The code stays completely self‑contained, uses only
`numpy` and the standard library and respects the 1000 s budget.

```python
"""
Propose a low‑value step‑function.

The algorithm is a projected Adam / exponentiated‑gradient optimizer for the
objective 2·n·max(conv(a,a))/(∑a)² (∑a is kept equal to 1).  It repeatedly

  1. evaluates the full convolution a*a,
  2. builds a mask of the (almost) maximal convolution entries,
  3. computes the sub‑gradient of the max via a second convolution,
  4. updates the probabilities with an Adam‑style adaptive step (mirror‑descent
     on the simplex), and
  5. occasionally performs a random “mass‑transfer’’ move to escape plateaus.

A simple stagnation‑restart is triggered when no improvement is observed
for a while.  The best sequence seen during the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of two real vectors using FFT of length *size*."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a * a (full linear convolution) using pre‑computed FFT size."""
    return _conv_fft(a, a, size)


def _grad_convolution(a_rev: np.ndarray, mask: np.ndarray, size: int) -> np.ndarray:
    """
    Convolution a_rev * mask (full) and extract the central *n* entries.
    The result is the gradient of a single convolution entry with respect to a.
    """
    full = _conv_fft(a_rev, mask, size)
    n = a_rev.shape[0]
    # central part corresponds to indices n‑1 … 2·n‑2‑1
    return full[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_exp_update(a: np.ndarray,
                     grad: np.ndarray,
                     m: np.ndarray,
                     v: np.ndarray,
                     it: int,
                     lr: float,
                     beta1: float,
                     beta2: float,
                     eps: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform one Adam‑style update on the simplex using exponentiated
    gradient (mirror descent).  Returns the new a, and the updated moment
    vectors m and v.
    """
    # Adam moments
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    # bias correction
    m_hat = m / (1.0 - beta1 ** (it + 1))
    v_hat = v / (1.0 - beta2 ** (it + 1))

    # adaptive learning rate per coordinate (mirror‑descent style)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    # exponentiated update (keeps positivity)
    a = a * np.exp(-step)

    # re‑project onto the simplex (simple normalisation)
    a_sum = a.sum()
    if a_sum == 0.0:
        # very unlikely – fall back to uniform
        a = np.ones_like(a) / a.shape[0]
    else:
        a /= a_sum
    return a, m, v


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small mass from a random index i to another index j."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # problem size (can be overridden via kwargs)
    n = int(kwargs.get("n", 6000))

    # ------------------------------------------------------------------
    # initialise a feasible distribution (probability simplex)
    # ------------------------------------------------------------------
    try:
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            seq0 = np.asarray(seq0, dtype=float)
            if seq0.size != n:
                # linear interpolation to required length
                x_old = np.linspace(0, 1, seq0.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, seq0)
                a = np.clip(a, 0.0, None)
                if a.sum() == 0.0:
                    a = np.random.rand(n)
                a /= a.sum()
            else:
                a = np.clip(seq0, 0.0, None)
                a /= a.sum()
        else:
            raise KeyError
    except Exception:
        a = np.random.rand(n)
        a /= a.sum()

    # ------------------------------------------------------------------
    # FFT sizes (constant throughout the run)
    # ------------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a * a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * mask

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    conv_full = _full_convolution(a, size1)
    max_b = conv_full.max()
    best_score = 2.0 * n * max_b
    best_seq = a.copy()
    print(f"[init] n={n}, score={best_score:.6f}")

    # Adam / EG parameters
    lr = 0.05                          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # exploration / restart parameters
    rand_prob = 0.18                   # probability of a random transfer move
    rand_scale = 0.03                  # relative size of the random transfer
    tolerance_factor = 1e-7           # active set tolerance
    stagnation_restart = 30.0          # seconds without improvement → restart
    last_improve = start
    it = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1) evaluate current objective
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # 2) decide whether to do a random mass transfer
        if np.random.rand() < rand_prob:
            a = _random_transfer(a, rand_scale)
        else:
            # 3) build mask of (almost) maximal convolution entries
            tol = max(tolerance_factor * max_b, 1e-12)
            mask = (conv_full >= max_b - tol).astype(float)

            # 4) compute sub‑gradient via a_rev * mask
            a_rev = a[::-1]
            g = _grad_convolution(a_rev, mask, size2)

            # gradient of max(conv) w.r.t. a is -2 * g
            subgrad = -2.0 * g

            # 5) Adam / exponentiated‑gradient step
            a, m, v = _adam_exp_update(
                a,
                subgrad,
                m,
                v,
                it,
                lr,
                beta1,
                beta2,
                eps,
            )

        # 6) restart if we have been stuck for a while
        if (it % 25000 == 0) and (time.time() - last_improve > stagnation_restart):
            a = np.random.rand(n)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{it:7d}] restart due to stagnation")

        it += 1

    # ------------------------------------------------------------------
    # final reporting
    # ------------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
```
