"""
Gradient‐descent optimizer for the max–self‑convolution objective.

The algorithm works on the simplex
    S = { a_i ≥ 0 , Σ a_i = √(2 n) }.
It minimises the smoothed objective

    F_α(a) = (1/α)·log Σ_i exp(α·(a*a)[i]) ,

which converges to the true max when α → ∞.
For a given α the gradient is

    ∇F_α(a) = 2·conv( w , a_rev )[n‑1 : n‑1+n]

where
    w_i = exp(α·(a*a)[i]) / Σ_j exp(α·(a*a)[j])
    a_rev = a reversed,
    conv(·,·) is the linear convolution.

Each iteration:
    1. Compute the self‑convolution by FFT.
    2. Build soft‑max weights w (shifted by the current maximum to avoid overflow).
    3. Obtain the gradient via another FFT‑based convolution.
    4. Take an additive step with a step‑size that decays exponentially in time.
    5. Project back onto the probability simplex (exact ℓ₁‑norm).
    6. Keep the best vector seen (according to the true max‑convolution).

The temperature‑like schedule controls α (grows from ≈0.5 to ≈200)
and the step‑size (shrinks from ≈10 % of the total mass to ≈0.1 %).
Occasional random perturbations are injected when progress stalls.

The routine respects the 1000 s budget and returns the best sequence found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto the simplex {x ≥ 0, Σ x = s} (Wang & Carreira‑Perpiñán)."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Return the full linear convolution a*a using FFT of length nfft."""
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)
    # keep only the first 2·len(a)‑1 entries (linear convolution)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, conv: np.ndarray,
                     w: np.ndarray, nfft: int) -> np.ndarray:
    """
    Compute ∇F_α(a) = 2·conv(w, a_rev)[n‑1:n‑1+len(a)].
    All convolutions are performed via FFT for speed.
    """
    a_rev = a[::-1]
    w_fft = np.fft.rfft(w, n=nfft)
    a_rev_fft = np.fft.rfft(a_rev, n=nfft)
    full = np.fft.irfft(w_fft * a_rev_fft, n=nfft)
    # length of the true linear convolution is (2n‑1)+(n)‑1 = 3n‑2
    # we need the central n entries starting at index n‑1
    start = a.size - 1
    grad = 2.0 * full[start: start + a.size]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    E(a) = 2·n·max(conv(a,a)) / (Σ a)².
    The routine runs for at most `budget_s` seconds and returns the best
    sequence discovered.
    """
    # ------------------------------------------------------------------
    # RNG initialisation
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2500                     # default length if no starter is provided
        a = np.random.exponential(scale=1.0, size=n)

    # Normalise to Σ a = √(2 n) – this is the conventional scaling used in the
    # reference code.  The objective then equals max(conv(a,a)).
    target_sum = math.sqrt(2.0 * n)
    a = a / a.sum() * target_sum

    # ------------------------------------------------------------------
    # FFT length – must accommodate convolutions of length 3n‑2 (for the
    # gradient) and 2n‑1 (for the self‑convolution).  Choose the next power
    # of two for efficiency.
    needed = 3 * n - 2
    nfft = 1 << (needed - 1).bit_length()

    # ------------------------------------------------------------------
    # Scheduling parameters (exponential schedules)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # α grows from α_start → α_end
    alpha_start = 0.5
    alpha_end = 200.0

    # step size shrinks from step_start → step_end
    step_start = target_sum * 0.12
    step_end = target_sum * 0.0012

    # keep track of the current best
    conv_cur = _max_convolution_fft(a, nfft)
    cur_score = float(conv_cur.max())
    best_seq = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    iteration = 0
    no_improve = 0          # counter for stagnation detection

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)   # ∈[0,1]

        # ---- schedules -------------------------------------------------
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)
        step = step_start * ((step_end / step_start) ** frac)

        # ---- forward pass: self‑convolution ---------------------------
        conv = _max_convolution_fft(a, nfft)          # length 2n‑1
        max_conv = float(conv.max())

        # update global best
        if max_conv < best_score:
            best_score = max_conv
            best_seq = a.copy()
            no_improve = 0
            print(f"[{iteration:7d}] New best max‑conv: {best_score:.6f}  "
                  f"(t={elapsed:.1f}s) α={alpha:.2f} step={step:.5f}")
        else:
            no_improve += 1

        # ---- soft‑max weights -----------------------------------------
        # shift by max to keep exponentials ≤ 1
        shifted = conv - max_conv                # ≤ 0
        w_exp = np.exp(alpha * shifted)          # stable
        w = w_exp / w_exp.sum()                  # shape (2n‑1,)

        # ---- gradient of the smoothed objective ----------------------
        grad = _gradient_softmax(a, conv, w, nfft)

        # ---- gradient step + simplex projection -----------------------
        a_new = a - step * grad
        a_new = _simplex_projection(a_new, target_sum)

        # ---- simple acceptance rule (always accept the projected step) --
        a = a_new
        iteration += 1

        # ---- occasional random perturbation to escape plateaus --------
        if no_improve >= 4000:
            # reset with a small Gaussian jitter around the current best
            jitter = np.random.normal(scale=target_sum * 0.001, size=n)
            a = best_seq + jitter
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a, target_sum)
            no_improve = 0
            print(f"[{iteration:7d}] Perturbation applied (stagnation)")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list).
    return best_seq.tolist()
