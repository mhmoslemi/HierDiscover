def _project_simplex(v):
    """Projects vector v onto the probability simplex."""
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v):
    # ensures sum to 1, v may be negative
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all u - something <=0, then return uniform
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Projected gradient descent with momentum on the probability simplex,
symmetrisation and a multi‑stage seed search.

The algorithm works as follows:

1. Generate a diverse collection of seed sequences for a chosen length
   (geometric, power‑law, geometric × power, symmetric geometric,
   sinusoidal windows, Hann, triangular and a Gaussian‐like shape).
   All seeds are normalised to sum to one.

2. Pick the best seed (lowest true objective) by a fast FFT based evaluation.

3. Run a custom optimiser that works directly on the coefficients:
   * loss is a smoothed max of the auto‑convolution (log‑sum‑exp with an
     increasing smoothing factor β);
   * the gradient w.r.t. the coefficients is obtained by a single FFT‑based
     convolution;
   * a momentum term is used and the step size decays as 1/√t;
   * after each update the vector is symmetrised (a←(a+a[::-1])/2) and
     projected onto the probability simplex (non‑negative, sum=1).

4. When the improvement stalls a tiny random jitter is injected to help
   escaping shallow local minima.

5. If time permits a short stochastic local‑search (pairwise mass swaps)
   is performed for fine‑tuning.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projects an arbitrary vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform distribution
        return np.ones_like(v) / n
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
    return a / a.sum()


def _triangular_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _gaussian_like(n: int, sigma_frac: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


# ----------------------------------------------------------------------
# Core optimiser (projected gradient with momentum)
# ----------------------------------------------------------------------


def _optimise_seq(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with momentum on the simplex.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_fft(a, a)
    cur_max = conv.max()
    best_val = 2.0 * n * cur_max

    # momentum buffer
    m = np.zeros_like(a)

    # smoothing and step‑size schedules
    beta = 5.0
    beta_mul = 1.02
    alpha0 = 0.5
    momentum = 0.9

    t = 0
    stagnation = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # --- evaluate current objective ---
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            stagnation = 0
            print(f"[opt] len={n:5d} it={t:5d} new best={best_val:.6f}")
        else:
            stagnation += 1

        # --- smoothed‑max weights ---
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            beta *= beta_mul
            continue
        w /= w_sum

        # --- gradient w.r.t. a ---
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad = 2.0 * full[n - 1 : n - 1 + n]      # central n entries

        # --- momentum update ---
        m = momentum * m + (1.0 - momentum) * grad

        # --- step size ---
        alpha = alpha0 / np.sqrt(t)

        # --- gradient step on the simplex ---
        a = a - alpha * m

        # enforce symmetry (the optimal solution is expected to be symmetric)
        a = (a + a[::-1]) * 0.5

        # projection onto the probability simplex
        a = _project_simplex(a)

        # --- smoothing parameter update ---
        beta *= beta_mul

        # --- occasional jitter to escape plateaus ---
        if stagnation >= 2000:
            eps = 0.005
            a += eps * np.random.randn(n)
            a = _project_simplex(a)
            stagnation = 0

        # early exit if we already beat the target
        if best_val <= 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap local fine‑tuning via pairwise mass swaps
# ----------------------------------------------------------------------


def _local_search(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """Random pairwise swaps that keep the sum fixed; accepts only improvements."""
    a = a0.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        # transfer a small amount of mass from i to j
        delta = 0.001 * a[i]  # proportional perturbation
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # stay on the simplex automatically; just need non‑negativity
        if a_new[i] < 0.0:
            continue
        cur_val = 2.0 * n * _conv_fft(a_new, a_new).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            a = a_new
            print(f"[local] it={it:5d} new best={best_val:.6f}")
    return best_a, best_val

# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    """
    np.random.seed(seed)
    # small safety margin to avoid accidental timeout
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Lengths to explore – start modest, then go larger if time permits
    # ------------------------------------------------------------------
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

        if time.time() > deadline:
            break

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        seeds = []

        # pure power‑law families
        for p in np.linspace(0.0, 2.2, 12):
            seeds.append(_power(n, p))

        # combined geometric × power families
        for c in np.linspace(0.5, 4.0, 8):

            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 8):
                seeds.append(_geom_power(n, r, p))

        # symmetric geometric
        for c in np.linspace(0.5, 4.5, 8):
            r = 1.0 - c / n
            if r > 0.0:
                seeds.append(_sym_geom(n, r))

        # classic windows
        seeds.append(_sine_window(n))
        seeds.append(_hann_window(n))
        seeds.append(_triangular_window(n))
        seeds.append(_gaussian_like(n, sigma_frac=0.15))

        # a few random points (softmax of Gaussian logits)
        for _ in range(5):
            v_rand = np.random.randn(n)
            a_rand = np.exp(v_rand - np.max(v_rand))
            a_rand /= a_rand.sum()
            seeds.append(a_rand)

                    seeds.append(a_prev)

                    seeds.append(adj)

    return global_best_seq.tolist()
