"""
Propose a step‑function (non‑negative sequence) with a very low
evaluation score.  The algorithm combines:

1.  A small analytic pool (including the previous best construction
    if the global ``height_sequence_1`` is present) and picks the best
    one as a start point.
2.  An Adam‑style projected gradient descent on the current maximal
    convolution entry.  The sub‑gradient of ``max(conv(a,a))`` w.r.t.
    ``a`` is known analytically, so each iteration is cheap.
3.  A simple Metropolis acceptance rule with a linear temperature
    schedule, together with occasional random “mass‑swap’’ moves to
    escape plateaus.
4.  Restarts with a tiny jitter if the search stalls for more than
    a few seconds.

The search runs until the supplied time budget (default 1000 s) is
exhausted and returns the best sequence found.  All vectors are kept
on the probability simplex (non‑negative, sum = 1), which makes the
objective scale‑invariant and matches ``evaluate_sequence``’s
requirements.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another
    and re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def max_convolution(a: np.ndarray):
    """Return (max_value, argmax_index) of the linear convolution a*a."""
    b = np.convolve(a, a)
    idx = int(np.argmax(b))
    return float(b[idx]), idx

    if "height_sequence_1" in globals():
        tmp = base.copy()
        tmp = np.maximum(tmp, 0.0)
        tmp = project_simplex(tmp)          # ensure exact sum = 1
        cand_pool.append(tmp)

    # b) uniform
    cand_pool.append(np.ones(n) / n)

    # c) arcsine‑like product (the classic start)
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand_pool.append(a / a.sum())

    # d) product with a few different exponents
    for p in (0.6, 0.7, 0.8, 0.9, 1.0, 1.2):
        a = ((i + 0.5) * (n - i - 0.5)) ** (-p)
        cand_pool.append(a / a.sum())

    # e) sum of two inverse‑power tails
    for p in (0.3, 0.5, 0.7, 1.0, 1.3):
        a = (i + 1) ** (-p) + (n - i) ** (-p)
        cand_pool.append(a / a.sum())

    # pick the best candidate according to the true objective
    best_seq = None
    best_max = np.inf
    for idx, a in enumerate(cand_pool):
        mx, _ = max_convolution(a)
        print(f"[init] candidate {idx:02d} max(conv) = {mx:.8f}")
        if mx < best_max:
            best_max = mx
            best_seq = a.copy()
    print(f"[init] chosen start max(conv) = {best_max:.8f}")

    # ------------------------------------------------------------------
    # 2 – optimisation loop (Adam‑style projected sub‑gradient)
    # ------------------------------------------------------------------
    a = best_seq.copy()                # always kept on the simplex
    m = np.zeros_like(a)               # first moment
    v = np.zeros_like(a)               # second moment
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps = 1e-8

    # linear annealing schedules
    init_lr, final_lr = 0.5, 1e-4
    init_temp, final_temp = 1.0, 0.001

    iteration = 0
    last_improve = start
    best_overall_max = best_max
    best_overall_seq = a.copy()

    while time.time() < deadline:
        elapsed = time.time() - start
        # progress ∈ [0,1] (smooth, leaves a 10‑second head‑room)
        progress = elapsed / (budget_s - 10.0)

        # schedules
        lr = init_lr * (1.0 - progress) + final_lr * progress
        temp = init_temp * (1.0 - progress) + final_temp * progress

        # ------------------------------------------------------------------
        # compute current maximal convolution entry and its index
        # ------------------------------------------------------------------
        max_val, k_max = max_convolution(a)

        # track global best
        if max_val < best_overall_max:
            best_overall_max = max_val
            best_overall_seq = a.copy()
            last_improve = time.time()
            L = 2.0 * n * max_val          # objective (sum(a)=1)
            print(f"[{iteration:07d}] new best L = {L:.6f} (t={elapsed:.1f}s)")

        # ------------------------------------------------------------------
        # sub‑gradient of the maximal entry (a simple piecewise linear map)
        # ------------------------------------------------------------------
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        g[idx_range] = 2.0 * a[k_max - idx_range]

        # ------------------------------------------------------------------
        # Adam update (projected onto the simplex)
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

        # ------------------------------------------------------------------
        # Metropolis acceptance (lower is always accepted)
        # ------------------------------------------------------------------
        cand_max, _ = max_convolution(a_candidate)
        accept = False
        if cand_max < max_val:
            accept = True
        else:
            # occasional uphill moves governed by temperature
            prob = np.exp(-(cand_max - max_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
        else:
            # with a tiny probability perform a random mass transfer
            if np.random.rand() < 0.02:
                a = random_move(a, lr * 0.5)

        # ------------------------------------------------------------------
        # restart if we have been stuck for a while
        # ------------------------------------------------------------------
        if iteration % 50000 == 0 and (time.time() - last_improve) > 30.0:
            jitter = np.random.rand(n) * 1e-4
            a = best_overall_seq + jitter
            a = project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0
            print(f"[{iteration:07d}] restart after stagnation")

        iteration += 1

    total_time = time.time() - start
    final_L = 2.0 * n * best_overall_max
    print(f"[done] iterations={iteration}, best L={final_L:.6f}, time={total_time:.1f}s")
    return best_overall_seq.tolist()
