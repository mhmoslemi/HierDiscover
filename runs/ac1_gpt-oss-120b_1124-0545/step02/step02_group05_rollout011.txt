def propose_candidate(seed=0, budget_s=1000):

        score = float(conv.max())
        if score < best_score:
            best_score = score
            best_q = start.copy()
    q = best_q.copy()

    start_time = time.time()
    deadline = start_time + budget_s - 5
    it = 0
    while time.time() < deadline:
        # compute conv
        conv = np.convolve(q, q)
        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_q = q.copy()
            print(f"[{it}] New best {best_score:.6f}")
        # smoothing exponent schedule
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        p = 2 + 18 * frac  # start at 2, go to 20
        # compute weights w = conv**(p-1)
        # avoiding zeros
        eps = 1e-12
        w = np.power(conv + eps, p-1)
        w_sum = w.sum()

        q = q * exp_factor
        # projection onto simplex (sum to 1)

"""
Exponentiated‑gradient optimisation of the self‑convolution peak.

The evaluation for a normalised step‑function reduces to
    score = max(conv(a,a))                      (because ∑a = √(2·n))

We therefore work with a probability vector `q` (∑q = 1, q≥0) and
optimise the smoothed maximum of `conv(q,q)` using an
entropy‑mirror‑descent (multiplicative‑weights) update:

    w_k ∝ conv(q,q)_k^{p-1}          (p slowly grows → true max)
    g = 2·(w * q_rev)               gradient of the smooth max
    q ← q ⊙ exp(−η·g)                multiplicative update
    q ← simplex_projection(q,1)      keep ∑q = 1

The schedule for the exponent `p` and the stepsize `η` is
gradually tightened; occasional random mixes with a fresh Dirichlet
vector help to escape plateaux.  The best vector found is finally
scaled to the required ℓ₁‑norm √(2·n) and returned.

The algorithm runs for the full time budget (default 1000 s) and
records the best sequence seen.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all entries equal
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv(q: np.ndarray) -> np.ndarray:
    """Self‑convolution of a probability vector (direct method)."""
    return np.convolve(q, q)


# ----------------------------------------------------------------------
def _exponential_gradient_opt(
    q0: np.ndarray,
    target_sum: float,
    time_deadline: float,
    *,
    p_start: float = 2.0,
    p_end: float = 80.0,
    lr_start: float = 0.5,
    lr_end: float = 0.001,
    max_grad: float = 20.0,
    perturb_interval: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation on the simplex.
    Returns the best probability vector (∑=1) and its max‑convolution value.
    """
    n = q0.size
    q = q0.copy()
    best_q = q.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0

    while time.time() < time_deadline:
        # ---------- current objective ----------
        conv = _conv(q)
        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_q = q.copy()
            last_improve = it
            print(
                f"[{it:5d}] New best max(conv) = {best_score:.8f} (t={time.time() - (time_deadline - 1000):.1f}s)"
            )

        # ---------- smoothing schedule ----------
        elapsed = time.time() - (time_deadline - 1000)
        frac = elapsed / (time_deadline - (time_deadline - 1000))
        p = p_start + (p_end - p_start) * frac          # exponent for the p‑norm
        lr = lr_start * ((lr_end / lr_start) ** frac)   # stepsize

        # ---------- weight vector (smooth max) ----------
        # w_k ∝ conv_k^{p-1}   – avoid overflow / NaNs
        eps = 1e-14
        w_raw = np.power(conv + eps, p - 1.0)
        w_sum = w_raw.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_raw / w_sum

        # ---------- gradient of the smooth max ----------
        # g_i = 2 * Σ_k w_k * q_{k-i}
        q_rev = q[::-1]
        grad_full = np.convolve(w, q_rev)               # length = 3n-2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]       # size n
        # clip to avoid overflow in the exponent
        grad = np.clip(grad, -max_grad, max_grad)

        # ---------- multiplicative (mirror‑descent) update ----------
        factor = np.exp(-lr * grad)
        q = q * factor
        q = _simplex_projection(q, 1.0)                 # keep on the simplex

        # ---------- occasional random perturbations ----------
        if it % perturb_interval == 0 and it > 0:
            # mix the current best with a fresh Dirichlet sample
            rnd = np.random.dirichlet(np.ones(n))
            mix = np.random.rand()
            q_pert = mix * q + (1.0 - mix) * rnd
            q_pert = _simplex_projection(q_pert, 1.0)

            conv_pert = _conv(q_pert)
            score_pert = float(conv_pert.max())
            if score_pert < best_score:
                best_score = score_pert
                best_q = q_pert.copy()
                last_improve = it
                print(
                    f"[{it:5d}] New best after perturbation: {best_score:.8f}"
                )
            if score_pert < cur_score:
                q = q_pert

        # ---------- restart if stagnating ----------
        if it - last_improve > 5000:
            # re‑initialise from a blend of the best and noise
            noise = np.random.dirichlet(np.ones(n))
            q = 0.5 * best_q + 0.5 * noise
            q = _simplex_projection(q, 1.0)
            last_improve = it

        it += 1

    return best_q, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function height vector that minimises
    `evaluate_sequence`.  The routine builds an exponentiated‑gradient
    optimiser on the probability simplex, progressively tightening a
    p‑norm smooth‑max surrogate of the self‑convolution peak.  The best
    sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size and initial vector
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base_vec = np.asarray(height_sequence_1, dtype=float)
    else:
        n = 2500
        base_vec = np.full(n, 1.0)

    # Normalise to the required ℓ₁‑norm (∑a = √(2·n))
    target_sum = math.sqrt(2.0 * n)
    base_vec = base_vec / base_vec.sum() * target_sum

    # Work with probabilities (∑q = 1)
    uniform_q = np.full(n, 1.0 / n)
    rand_q = np.random.dirichlet(np.ones(n))

    starters = [base_vec / target_sum, uniform_q, rand_q]
    best_q = None
    best_score = float("inf")
    for cand in starters:
        conv = _conv(cand)
        score = float(conv.max())
        if score < best_score:
            best_score = score
            best_q = cand.copy()

    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # Run the exponentiated‑gradient optimiser
    final_q, final_score = _exponential_gradient_opt(
        best_q,
        target_sum,
        deadline,
        p_start=2.0,
        p_end=80.0,
        lr_start=0.5,
        lr_end=0.001,
        max_grad=20.0,
        perturb_interval=400,
    )

    # ------------------------------------------------------------------
    # Scale the best probability vector back to the required ℓ₁‑norm
    best_sequence = (final_q * target_sum).tolist()

    # (Optional) final verification – prints the achieved score
    # evaluate_sequence is assumed to be available in the environment.
    try:
        import __main__

        score = __main__.evaluate_sequence(best_sequence)
        print(f"Final score reported by evaluate_sequence: {score:.8f}")
    except Exception:
        # In case evaluate_sequence is not importable in this context.
        pass

    return best_sequence
