def _project_onto_simplex(v):
    # ensure non-negative and sum1
    # Duchi et al algorithm.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # Determine initial sequence
    try:
        init = np.array(height_sequence_1, dtype=float)
        n = init.shape[0]
        a = init.copy()
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total <= 0:
            a = _power_sequence(5000, 0.5)
            n = a.shape[0]
        else:
            a /= total
    except Exception:
        n = 5000
        a = _power_sequence(n, 0.5)
    # Ensure sum = 1
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    best_seq = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()
    print(f"[init] n={n}, score={best_score:.6f}")

    # Parameters
    lr = 0.2
    lr_decay = 0.99995
    lr_min = 1e-7

    sigma = 0.5          # smoothing factor multiplier of max conv
    sigma_decay = 0.99995
    sigma_min = 0.02

    iteration = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        iteration += 1

        # 1) self‑convolution
        conv = _conv_fft(a, a)
        max_conv = conv.max()

        # 2) smoothing parameter
        sigma = max(sigma * sigma_decay, sigma_min)
        eps = max(sigma * max_conv, 1e-12)

        # 3) softmax weights over convolution entries
        expo = conv / eps
        # Clip for numerical stability
        expo = np.clip(expo, a_min=None, a_max=50.0)
        w = np.exp(expo)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # 4) gradient of smoothed max
        w_rev = w[::-1]
        v_full = _conv_fft(a, w_rev)   # length 3n-2
        idx = 2 * n - 2 - np.arange(n)
        v = v_full[idx]
        grad = v + a

        # momentum
        momentum = 0.9 * momentum + grad

        # 5) gradient step and simplex projection
        a = a - lr * momentum
        # project onto simplex {x>=0, sum=1}
        a = _project_onto_simplex(a)

        # decay step size
        lr = max(lr * lr_decay, lr_min)

        # occasional random mass exchange
        if iteration % 100 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.5, a[i] * lr)   # small chunk
            if delta > 0:

"""
Optimisation of a step‑function (non‑negative sequence) for the
``evaluate_sequence`` objective.

The algorithm works in three stages:

1. **Initialisation** –  use the best known construction (the global
   ``height_sequence_1`` if it exists, otherwise a power‑law
   `a_i ∝ (i+1)^{-0.5}`) and normalise it to a probability vector.

2. **Smoothed max‑gradient descent** –  minimise the
   ``max(conv)`` where ``conv = a∗a``.
   The non‑smooth maximum is approximated by a soft‑max
   `ε·log Σ exp(conv/ε)`.  Its gradient can be written as

       ∇f(a) = v + a  with   v_j = Σ_i w_{i+j} a_i ,

   where `w` are the soft‑max weights.  The computation of `v` is a
   convolution and is performed efficiently with FFTs.  After each
   gradient step the vector is projected onto the probability simplex
   (`a_i≥0, Σa_i=1`).

3. **Fine‑grained refinement** –  a small random mass exchange is
   performed periodically to escape shallow plateaus.  The best sequence
   (lowest exact score) seen during the run is kept and returned.

The whole routine respects the 1000 s time budget, uses only NumPy,
and returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).

    Parameters
    ----------
    v : np.ndarray
        Input vector (any real values).

    Returns
    -------
    np.ndarray
        Projected vector (non‑negative, sum‑1).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the largest ρ where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Convolution of two real vectors using FFT (fast for lengths ≥ ~1e3).

    Parameters
    ----------
    x, y : np.ndarray
        Real 1‑D input arrays.

    Returns
    -------
    np.ndarray
        Full convolution (length = len(x)+len(y)-1).
    """
    size = x.size + y.size - 1
    # next power of two for efficient FFT
    N = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised power‑law sequence a_i ∝ (i+1+offset)^{-α}.

    Parameters
    ----------
    n      : int   – length of the sequence.
    alpha  : float – exponent (α>0).
    offset : float – optional shift of the index.

    Returns
    -------
    np.ndarray
        Length‑`n` array with sum equal to 1 and all entries ≥ 0.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  A few seconds are kept as safety margin.
    kwargs : dict
        Unused – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, sum > 0.01) discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Initialise the probability vector
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be supplied by the surrounding harness
        init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init.size
        a = np.maximum(init, 0.0)
        total = a.sum()
        if total > 0.0:
            a /= total
        else:
            raise ValueError
    except Exception:
        # fallback: a power‑law with exponent ≈0.5 (near the known optimum)
        n = 5000
        a = _power_sequence(n, alpha=0.5)

    # ensure we are on the simplex
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    # keep the best sequence seen so far
    conv = _conv_fft(a, a)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smoothed max‑gradient descent with annealing
    # ------------------------------------------------------------------
    lr = 0.2                # initial step size
    lr_decay = 0.99995
    lr_min = 1e-7

    sigma = 0.5             # smoothing factor multiplier of max(conv)
    sigma_decay = 0.99995
    sigma_min = 0.02        # never go below this to avoid overflow

    iteration = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        iteration += 1

        # (a) self‑convolution and max entry
        conv = _conv_fft(a, a)
        max_conv = conv.max()

        # (b) smoothing parameter ε = σ·max(conv)
        sigma = max(sigma * sigma_decay, sigma_min)
        eps = max(sigma * max_conv, 1e-12)

        # (c) soft‑max weights over convolution entries
        #    w_i ∝ exp(conv_i / ε)
        expo = conv / eps
        # clip to keep exp() in a safe range
        expo = np.clip(expo, a_min=None, a_max=50.0)
        w = np.exp(expo)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # (d) gradient of the smoothed objective:
        #     ∇f = v + a,   v_j = Σ_i w_{i+j} a_i
        w_rev = w[::-1]                     # reversed weight vector
        v_full = _conv_fft(a, w_rev)        # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)    # pick the slice giving v_j
        v = v_full[idx]

        grad = v + a

        # (e) momentum (helps to traverse flat regions)
        momentum = 0.9 * momentum + grad

        # (f) gradient step + projection onto simplex
        a = a - lr * momentum
        a = _project_onto_simplex(a)

        # (g) decay the learning rate
        lr = max(lr * lr_decay, lr_min)

        # (h) occasional random mass‑swap to escape tiny plateaus
        if iteration % 100 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            # move a modest fraction of mass from i to j
            delta = min(a[i] * 0.5, a[i] * lr)
            if delta > 0.0:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

        # (i) exact score (the objective used by evaluate_sequence)
        conv_now = _conv_fft(a, a)
        cur_score = 2.0 * n * conv_now.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f}, iterations={iteration}")
    return best_seq.tolist()
