def conv_fft(x, y):
    n = x.shape[0] + y.shape[0] - 1
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    c = np.fft.irfft(X * Y, N)[:n]
    return c

"""
Propose candidate step‑function heights.

The algorithm works in three layers:

1. **Parameterisation** – the non‑negative vector `a` with sum
   `√(2·n)` is written as `a = S·softmax(z)`.  The free variables are
   the logits `z ∈ ℝⁿ`.  This automatically respects the simplex
   constraint and avoids a projection step.

2. **Smooth max surrogate** – for a temperature `ε` the objective
   `F_ε(z) = ε·log Σ_i exp(conv_i/ε)` approximates the true quantity
   `max(conv)`.  An analytical gradient w.r.t. `z` is derived from
   `∂F/∂a = Σ_i w_i·∂conv_i/∂a` where `w` is the softmax over the
   convolution entries.  The gradient is computed with three FFT‑based
   convolutions, i.e. in `O(n log n)` time.

3. **Annealed L‑BFGS optimisation** – a short L‑BFGS‑B run is performed
   for a decreasing schedule of ε values (`[1.0,…,0.001]`).  The
   solution of one stage is used as the start for the next, progressively
   sharpening the surrogate.  Several random restarts (uniform and
   exponential initialisations) are performed until the time budget
   expires.  The best sequence encountered (according to the true
   `max(conv)`) is returned.

The routine prints periodic progress reports and stays well within the
1000 s time limit.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    shift = np.max(z)
    e = np.exp(z - shift)
    return e / e.sum()


# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a real array of length len(x)+len(y)-1.
    """
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    c = np.fft.irfft(X * Y, N)[:n]
    return c


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Given a non‑negative vector a and a temperature eps,
    returns (value, grad) where value = ε·log Σ exp(conv/ε)
    and grad = ∂value/∂a (shape = a.shape).
    """
    n = a.size
    # full self‑convolution
    conv = _conv_fft(a, a)                     # length 2n‑1
    m = conv.max()
    # stable softmax over the shifted entries
    shifted = conv - m
    # avoid overflow/underflow – for very small eps many terms vanish
    exp_shift = np.exp(shifted / eps)
    sum_exp = exp_shift.sum()
    if sum_exp == 0 or not np.isfinite(sum_exp):
        # fall back to a hard max (one‑hot)
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        sum_exp = 1.0
    else:
        w = exp_shift / sum_exp

    # ----- gradient w.r.t. a ------------------------------------------------
    # sigma_j = sum_{i=j}^{j+n-1} w_i   (sliding window of length n)
    cumsum_w = np.cumsum(w)
    sigma = cumsum_w[n - 1:2 * n - 1] - np.concatenate(([0.0], cumsum_w[:n - 1]))
    # first term = Σ_{t=0}^{n‑1} w_{j+t} * a_t   (vectorised via a convolution)
    a_rev = a[::-1]
    conv_wa = _conv_fft(w, a_rev)              # length 3n‑2
    first = conv_wa[n - 1:n - 1 + n]           # slice of length n
    grad_a = first + a * sigma

    # smooth objective value
    value = float(m + eps * math.log(sum_exp))

    return value, grad_a


# ----------------------------------------------------------------------
def _obj_and_grad(z: np.ndarray, eps: float, target_sum: float, n: int):
    """
    Returns (f, g) where f = smoothed max of the self‑convolution of the
    vector a = target_sum·softmax(z) and g = ∂f/∂z.
    """
    # a lives on the simplex automatically
    p = _softmax(z)
    a = target_sum * p

    # smooth objective and grad w.r.t. a
    f, grad_a = _smooth_max_and_grad(a, eps)

    # chain rule: ∂a_i/∂z_j = a_i·(δ_{ij} – p_j)
    dot = float(np.dot(grad_a, a))          # scalar = a·∇_a f
    grad_z = a * grad_a - dot * p           # shape (n,)

    return f, grad_z


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function height vector that minimises
    `evaluate_sequence`.  The search uses an annealed L‑BFGS optimisation
    on a smooth surrogate of the max‑convolution, starting from several
    random restarts.  The best sequence found before the time budget expires
    is returned.
    """
    # ---- reproducibility --------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # ---- problem size ------------------------------------------------------
    if "height_sequence_1" in globals():
        base_vec = np.asarray(height_sequence_1, dtype=float)
        n = base_vec.size
    else:
        n = 2500                     # default size, works well within budget

    target_sum = math.sqrt(2.0 * n)          # ℓ₁‑norm we keep fixed

    # ---- helper to turn a (non‑negative) vector into logits ----------------
    def logits_from_a(a_vec: np.ndarray) -> np.ndarray:
        """Return logits z such that softmax(z) = a_vec / target_sum."""
        # protect against exact zeros
        eps = 1e-15
        p = a_vec / target_sum
        p = np.maximum(p, eps)
        p /= p.sum()
        return np.log(p)

    # ---- initialise a pool of starting points --------------------------------
    starters = []

    # uniform distribution
    uniform = np.full(n, target_sum / n)
    starters.append(logits_from_a(uniform))

    # random exponential vector
    rand_exp = np.random.exponential(scale=1.0, size=n)
    rand_exp = rand_exp / rand_exp.sum() * target_sum
    starters.append(logits_from_a(rand_exp))

    # optional reuse of the previous construction (if present)
    if "height_sequence_1" in globals():
        base = base_vec.copy()
        # normalise to required ℓ₁‑norm
        if base.sum() > 0:
            base = base / base.sum() * target_sum
        else:
            base = uniform.copy()
        starters.append(logits_from_a(base))

    # ----------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # safety margin

    best_seq = None
    best_val = np.inf
    it_total = 0
    restart_id = 0

    # define a decreasing schedule for the smoothing temperature ε
    eps_schedule = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    # ----------------------------------------------------------------------
    while time.time() < deadline:
        # ----- pick a starting logit vector --------------------------------
        if restart_id < len(starters):
            z0 = starters[restart_id].copy()
        else:
            # fresh random start after we exhausted the predefined ones
            a_rand = np.random.exponential(scale=1.0, size=n)
            a_rand = a_rand / a_rand.sum() * target_sum
            z0 = logits_from_a(a_rand)

        restart_id += 1

        # ----- run the annealing loop over ε --------------------------------
        z_cur = z0.copy()
        for eps in eps_schedule:
            # time check before entering a new stage
            if time.time() >= deadline:
                break

            # L‑BFGS‑B options – a modest number of iterations per stage.
            # Later (smaller ε) stages are given a few more iterations.
            stage_iter = 80 if eps <= 0.01 else 50
            options = {"maxiter": stage_iter, "disp": False}

            # minimise the smooth surrogate
            res = minimize(_obj_and_grad,
                           z_cur,
                           args=(eps, target_sum, n),
                           method='L-BFGS-B',
                           jac=True,
                           options=options)

            z_cur = res.x  # use the result as start for the next temperature

            # evaluate the genuine objective (max self‑convolution)
            a_cur = target_sum * _softmax(z_cur)
            conv_cur = _conv_fft(a_cur, a_cur)
            max_conv = float(conv_cur.max())

            it_total += 1
            if max_conv < best_val:
                best_val = max_conv
                best_seq = a_cur.copy()
                elapsed = time.time() - start_time
                print(f"[{it_total:6d}] New best max(conv)={best_val:.6f} "
                      f" (t={elapsed:.1f}s, eps={eps:.4f})")

        # after a full temperature schedule, optionally perform a short Adam
        # refinement to escape tiny local plateaus
        if time.time() < deadline and best_seq is not None:
            a_ref = best_seq.copy()
            # simple Adam on the original (non‑softened) max
            lr = target_sum * 0.0003
            beta1, beta2 = 0.9, 0.999
            m = np.zeros_like(a_ref)
            v = np.zeros_like(a_ref)
            t = 0
            for _ in range(200):
                if time.time() >= deadline:
                    break
                t += 1
                # compute gradient of the hard max (sub‑gradient at the argmax)
                conv = _conv_fft(a_ref, a_ref)
                idx_max = int(conv.argmax())
                # sub‑gradient of max is exactly the derivative of conv at idx_max
                # ∂conv_idx_max/∂a_j = a_{idx_max-j} + a_{j-idx_max} (when in range)
                grad_sub = np.zeros_like(a_ref)
                # contribution from the first term
                j_start = max(0, idx_max - (n - 1))
                j_end = min(n - 1, idx_max)
                grad_sub[j_start:j_end + 1] = a_ref[idx_max - np.arange(j_start, j_end + 1)]
                # contribution from the second term
                # this is symmetric; we can add the same as above because conv is symmetric
                # The above already accounts for both sides (the formula is correct)
                grad = grad_sub

                # Adam step
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad ** 2)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

                a_ref = a_ref - step
                a_ref = np.maximum(a_ref, 0.0)
                # re‑project onto the simplex (simple scaling works because all entries stay non‑negative)
                a_ref = a_ref / a_ref.sum() * target_sum

                # evaluate
                conv = _conv_fft(a_ref, a_ref)
                cur_val = float(conv.max())
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a_ref.copy()
                    elapsed = time.time() - start_time
                    print(f"[{it_total:6d}] Adam refined best max(conv)={best_val:.6f} "
                          f" (t={elapsed:.1f}s)")

        # safety: stop if we already beat the target comfortably
        if best_val < 1.50:
            break

    # ----------------------------------------------------------------------
    if best_seq is None:
        # fall back to a uniform vector (should never happen)
        best_seq = np.full(n, target_sum / n)

    # final sanity check – clamp to the limits used inside evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0).tolist()
    return best_seq
