            (β grows from 1 to 200 during the run).
        *   Compute the gradient of f_β analytically:
                ∂f/∂a_i = 4·n· Σ_k w_k·a_{k-i},
            where w_k = softmax_β((a∗a)_k).
            The Σ is a correlation that is evaluated efficiently by FFT.
        *   Take a step in the negative gradient direction,
            normalise the step size by the ∞‑norm of the gradient,
            and project the result onto the probability simplex
            (non‑negative + Σa_i=1).
        *   Occasionally (every ~20 iterations) compute the exact
            score with ``evaluate_sequence`` and keep the best vector.
        *   Random block reversals (1 % chance) are injected to escape
            plateaus.
3.  After the gradient phase perform a brief simulated‑annealing style
    local search that moves a small random amount of mass between two
    positions; the exact evaluator is consulted only when the cheap
    surrogate suggests a genuine improvement.
4.  The best sequence found before the global deadline (≈1000 s) is
    returned.  All numbers are non‑negative floats, and the routine
    respects the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top level, no closures
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the O(n log n) algorithm of (Duchi et al., 2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index of the cutoff
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """
    Scale the array so that its maximum is 0.9 and its sum is at least 0.01.
    Scaling does not affect the evaluation function because it is
    homogeneous.
    """
    a = np.asarray(a, dtype=float)
    max_v = a.max()
    if max_v > 0.0:
        a = a * (0.9 / max_v)
    total = a.sum()
    if total < 0.01:
        if total > 0.0:
            a = a * (0.01 / total)
        else:
            a = a + (0.01 / a.size)
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using an FFT of power‑of‑two length.
    Returned length = 2·len(a)‑1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)
    return conv[: 2 * n - 1]


def _correlation_w_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation C[l] = Σ_t w[t]·a[t‑l]  (zero‑padding outside range).
    For the gradient we only need C[l] for l = 0,…,n‑1.
    """
    L = w.shape[0]
    n = a.shape[0]
    N = 1 << ((L + n - 1).bit_length())   # next power of two
    # FFT of the two vectors (zero‑padded to length N)
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a, n=N)
    # correlation via IFFT(W * conj(A))
    C_full = np.fft.irfft(W * np.conj(A), n=N)
    return C_full[:n]                     # first n entries are the needed ones


def _smooth_obj_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Compute the smooth surrogate objective f_β(a) and its gradient.
    The surrogate is
        f_β(a) = 2·n·(1/β)·log Σ_k exp(β·(a∗a)_k)
    Gradient:
        ∂f/∂a_i = 4·n· Σ_k w_k·a_{k‑i},
    where w_k = softmax_β((a∗a)_k).
    """
    n = a.shape[0]
    # convolution a∗a
    s = _conv_fft(a)                     # length 2·n‑1
    # log‑sum‑exp in a numerically stable way
    scaled = beta * s
    max_scaled = scaled.max()
    exp_shifted = np.exp(scaled - max_scaled)
    sum_exp = exp_shifted.sum()
    log_sum_exp = max_scaled + np.log(sum_exp)   # = log Σ exp(beta·s)
    f = 2.0 * n * (log_sum_exp / beta)           # smooth objective value

    # softmax weights
    w = exp_shifted / sum_exp                     # length 2·n‑1, sum to 1

    # gradient via correlation (see derivation in the docstring)
    corr = _correlation_w_a(w, a)                 # length n
    g = 4.0 * n * corr

    return f, g


def _smooth_descent(start_a: np.ndarray,
                    start_time: float,
                    deadline: float,
                    rng: np.random.Generator,
                    length: int) -> np.ndarray:
    """
    Projected smooth gradient descent on the log‑sum‑exp surrogate.
    Returns the best vector (as a NumPy array) discovered before the deadline.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # schedule parameters
    beta_start, beta_end = 1.0, 200.0          # inverse temperature
    lr_start, lr_end = 0.5, 0.001             # step‑size for gradient descent
    max_iters = 2000                          # safety cap; real stop is time

    # momentum term (simple Nesterov‑like)
    velocity = np.zeros_like(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # progress fraction for schedules
        frac = (now - start_time) / (deadline - start_time)
        beta = beta_start + frac * (beta_end - beta_start)
        lr = lr_start + (1.0 - frac) * (lr_end - lr_start)

        # smooth objective & gradient
        f_smooth, grad = _smooth_obj_and_grad(a, beta)

        # normalise step using ∞‑norm of gradient (so the maximal coordinate
        # changes by at most lr)
        g_max = np.max(np.abs(grad))
        if g_max == 0.0:
            break
        step = lr / g_max

        # Nesterov style update
        velocity = 0.9 * velocity + (1.0 - 0.9) * grad
        a_new = a - step * velocity
        a_new = _project_to_simplex(a_new)

        # occasional random block reversal (helps escape plateaus)
        if rng.random() < 0.01:
            start_idx = rng.integers(0, n)
            block_len = rng.integers(1, min(256, n - start_idx) + 1)
            a_new[start_idx:start_idx + block_len] = \
                a_new[start_idx:start_idx + block_len][::-1]

        a = a_new

        # Exact evaluation only every ~30 iterations (cheap surrogate is cheap)
        if it % 30 == 0:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(f"[{now - start_time:.1f}s] "
                      f"New best (len={length}) exact score={best_score:.6f}")

                # early stop if we already beat the target
                if best_score <= 1.48:
                    print("Target already reached – ending descent.")
                    break

    return best_a


def _local_random_search(a: np.ndarray,
                         start_time: float,
                         deadline: float,
                         rng: np.random.Generator,
                         length: int) -> np.ndarray:
    """
    Simple simulated‑annealing style hill‑climb that moves a small amount
    of probability mass between two random positions.  The cheap surrogate
    (log‑sum‑exp) is used to pre‑filter candidates; the exact evaluator is
    called only when a genuine improvement is likely.
    """
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # Parameters for the random walk
    max_trials = 2000
    for t in range(max_trials):
        now = time.time()
        if now >= deadline:
            break

        # propose a move: transfer ε·a[i] from i → j
        i, j = rng.integers(0, length, size=2)
        if i == j:
            continue
        eps = rng.random() * 0.05          # up to 5 % of the source mass
        delta = eps * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection onto simplex is unnecessary because we kept the sum
        # but rounding errors may accumulate – re‑project occasionally.
        if t % 200 == 0:
            cand = _project_to_simplex(cand)

        # cheap surrogate check
        beta_tmp = 150.0                    # relatively high to mimic max
        f_surr, _ = _smooth_obj_and_grad(cand, beta_tmp)
        # If the surrogate improved by more than 10⁻⁴, evaluate precisely
        if f_surr < (2.0 * length * np.max(_conv_fft(best_a))):   # rough guard
            exact = evaluate_sequence(cand.tolist())
            if exact < best_score:
                best_score = exact
                best_a = cand
                print(f"[{now - start_time:.1f}s] "
                      f"Random‑search improvement (len={length}) → score={best_score:.6f}")

                if best_score <= 1.48:
                    break
    return best_a


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient list that minimises
    `evaluate_sequence`.  The function respects the time budget; after the
    budget expires it returns the best sequence found so far.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(np.asarray(hs1, dtype=float))
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from "
                  f"height_sequence_1 → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to try.  Smaller lengths are cheaper, larger ones give
    # a richer search space; we allocate time roughly proportionally.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.1f}s] Optimising length {n}")

        # ------------------------------------------------------------------
        # A few different initialisations for robustness
        # ------------------------------------------------------------------
        initialisations = []

        # (a) Exponential decay with parameter tuned around the best found so far
        for c in (1.5, 2.0, 2.5):
            r = np.exp(-c / n)
            init = np.power(r, np.arange(n, dtype=float))
            init /= init.sum()
            initialisations.append(init)

        # (b) Power‑law decay
        for p in (0.8, 1.0, 1.2):
            idx = np.arange(1, n + 1, dtype=float)
            init = 1.0 / (idx ** p)
            init /= init.sum()
            initialisations.append(init)

        # (c) Uniform random vector
        for _ in range(2):
            init = rng.random(n)
            init /= init.sum()
            initialisations.append(init)

        # ------------------------------------------------------------------
        # Run the smooth‑gradient descent from each start
        # ------------------------------------------------------------------
        for start_vec in initialisations:
            if time.time() >= deadline:
                break

            refined = _smooth_descent(start_vec, start_time, deadline, rng, n)

            # Local random search to fine‑tune
            refined = _local_random_search(refined, start_time, deadline, rng, n)

            score = evaluate_sequence(refined.tolist())
            if score < best_score:
                best_score = score
                best_seq = refined.tolist()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"New global best (len={n}) score={best_score:.6f}")

                if best_score <= 1.48:       # reaching the target early saves time
                    break

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a simple exponential shape
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8192
        r = np.exp(-2.0 / n)
        best_seq = (np.power(r, np.arange(n, dtype=float)) / n).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"[{time.time() - start_time:.1f}s] Finished – best score={best_score:.6f}")
    # Ensure API compliance (list of non‑negative floats)
    return [float(x) for x in best_seq]
