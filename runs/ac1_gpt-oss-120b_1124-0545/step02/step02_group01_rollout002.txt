import numpy as np
import time

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _generate_seeds(n):
    seeds = []
    # geometric
    for c in np.linspace(0.5, 5.0, 12):
        r = 1.0 - c / n
        if r > 0:
            idx = np.arange(n)
            a = r**idx
            seeds.append(a / a.sum())
    # power law
    for p in np.linspace(0.0, 2.2, 12):
        idx = np.arange(1, n+1)
        a = 1.0 / (idx**p)
        seeds.append(a / a.sum())
    # geom*power
    for c in np.linspace(0.5, 4.0, 8):
        r = 1.0 - c / n
        if r <= 0:
            continue
        for p in np.linspace(0.0, 2.0, 8):
            idx = np.arange(1, n+1)
            a = (r**(idx-1)) / (idx**p)
            seeds.append(a / a.sum())
    # symmetric geometric
    for c in np.linspace(0.5, 4.5, 8):
        r = 1.0 - c / n
        if r > 0:
            a = r**np.arange(n) + r**np.arange(n-1,-1,-1)
            seeds.append(a / a.sum())
    # sine window
    i = np.arange(n)

    # Kaiser windows (beta parameter)

    for sigma in np.linspace(0.2, 0.5, 5):
        i = np.arange(n)
        mu = (n-1)/2
        a = np.exp(-0.5*((i-mu)/sigma/(n-1))**2)
        seeds.append(a / a.sum())
    # Random seeds
    for _ in range(5):
        v = np.random.randn(n)
        a = _softmax(v)
        seeds.append(a)
    # previous global if exists
    prev = globals().get('height_sequence_1')
    if prev is not None:
        try:
            a_prev = np.asarray(prev, dtype=np.float64)
            # Normalize length
            if a_prev.shape != (n,):
                # Pad or truncate
                a_adj = np.zeros(n, dtype=np.float64)
                m = min(a_prev.shape[0], n)
                a_adj[:m] = a_prev[:m]
                a_prev = a_adj
            a_prev = a_prev / a_prev.sum()
            seeds.append(a_prev)
        except Exception:
            pass
    return seeds

def _adam_optimize(n, seeds, deadline):
    best_seq = None
    best_val = float('inf')
    # hyperparams
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    beta_start = 2.0
    beta_mul = 1.05
    beta_max = 500.0
    # limit iterations per seed to avoid infinite loops. We'll break by time.
    for s_idx, a0 in enumerate(seeds):
        # logits init
        v = np.log(a0 + 1e-30)
        m = np.zeros_like(v)
        v2 = np.zeros_like(v)
        t = 0
        while True:
            if time.time() > deadline:
                return best_seq, best_val
            t += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)
            cur_max = np.max(conv)
            cur_val = 2.0 * n * cur_max
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[adam] n={n} seed={s_idx} it={t} val={cur_val:.6f}")
            # smoothed max weights
            beta = beta_start * (beta_mul ** t)
            if beta > beta_max:
                beta = beta_max
            shift = cur_max
            # compute w; avoid overflow
            # exp(beta*(conv - shift)) => for conv <= shift, exponent <=0.
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                continue
            w /= w_sum
            # gradient:w.r.t a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)   # length 3n-2
            grad_a = 2.0 * full[n-1:n-1+n]
            # gradient w.r.t v
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v**2)
            m_hat = m / (1.0 - beta1**t)
            v2_hat = v2 / (1.0 - beta2**t)
            lr = lr0 / np.sqrt(t)
            v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)
    return best_seq, best_val

def _local_search(n, a_start, start_val, deadline):
    """Simple simulated-annealing style search that moves mass between two entries."""
    best_seq = a_start.copy()
    best_val = start_val
    cur_seq = a_start.copy()
    cur_val = start_val
    # Temperature schedule
    # Initial temp relative to best_val
    T0 = 0.01 * best_val
    # We'll decay temperature geometrically
    decay = 0.9995
    T = T0
    iteration = 0
    # pre-allocate buffer for candidate
    while time.time() < deadline:
        iteration += 1
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # ensure we move from index with larger mass to another
        if cur_seq[i] < 1e-12:
            continue
        # step size proportional to current mass on i
        step_factor = 0.05  # up to 5% of source mass
        delta = step_factor * cur_seq[i] * np.random.rand()
        # apply move
        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # no need to renormalize; sum preserved
        # compute objective
        conv = _conv_fft(cand, cand)
        cand_val = 2.0 * n * np.max(conv)
        # acceptance
        if cand_val < cur_val or np.random.rand() < np.exp(-(cand_val - cur_val) / max(T, 1e-12)):
            cur_seq = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                print(f"[local] it={iteration} new best={best_val:.6f}")
        # temperature decay
        T *= decay
        # occasionally increase temperature modestly to avoid being stuck
        if iteration % 5000 == 0:
            T = max(T, T0 * 0.1)
    return best_seq, best_val

"""
Non‑negative step‑function optimisation.

The algorithm works in three stages:

1.  **Seed creation** – a large collection of simple analytic families
    (geometric, power‑law, geometric×power, symmetric geometric,
    sine, Kaiser, Gaussian) and a few random logits.  Every seed is
    normalised to a probability distribution (sum = 1).

2.  **Fast pre‑screen** – the seeds are evaluated with a cheap FFT
    convolution and the best ‑~20 candidates are kept for a more
    expensive optimiser.

3.  **Adam on the simplex** – each selected seed is optimised by Adam
    on the log‑softmax parametrisation, using a smoothed‑max loss that
    gradually approaches the true max (log‑sum‑exp with an increasing
    temperature β).  The optimiser runs until the global time budget is
    exhausted or the deadline is reached.

4.  **Local simulated‑annealing refinement** – starting from the best
    sequence obtained by Adam, a simple SA moves a small amount of mass
    from one index to another (preserving the sum).  The temperature is
    slowly decreased; any improvement is kept, otherwise a move is
    accepted with probability exp(−Δ/T).  This stage uses the exact
    objective 2·n·max(conv(a,a)) (sum = 1) and runs for the remaining
    time.

The routine respects the supplied time budget and returns the best
sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Basic utilities (FFT convolution, softmax, etc.)
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (outputs a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ------------------------------------------------------------
# Seed generation
# ------------------------------------------------------------

def _generate_seeds(n: int) -> list[np.ndarray]:
    """Create a rich set of diverse initial probability vectors."""
    seeds: list[np.ndarray] = []

    # --- geometric families ------------------------------------------------
    for c in np.linspace(0.5, 5.0, 12):
        r = 1.0 - c / n
        if r > 0.0:
            a = r ** np.arange(n, dtype=np.float64)
            seeds.append(a / a.sum())

    # --- pure power‑law ----------------------------------------------------
    for p in np.linspace(0.0, 2.2, 12):
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / (idx ** p)
        seeds.append(a / a.sum())

    # --- combined geometric × power ----------------------------------------
    for c in np.linspace(0.5, 4.0, 8):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 8):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (r ** (idx - 1)) / (idx ** p)
            seeds.append(a / a.sum())

    # --- symmetric geometric ------------------------------------------------
    for c in np.linspace(0.5, 4.5, 8):
        r = 1.0 - c / n
        if r > 0.0:
            a = r ** np.arange(n, dtype=np.float64)
            a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
            seeds.append(a / a.sum())

    # --- sine window (smooth baseline) ------------------------------------
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(a / a.sum())

    # --- Kaiser windows (requires scipy) ---------------------------------
    try:
        from scipy.special import i0
        for beta in np.linspace(1.0, 12.0, 8):
            x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
            w = i0(beta * np.sqrt(1.0 - x * x))
            seeds.append(w / w.sum())
    except Exception:
        pass  # SciPy not available – skip this family

    # --- Gaussian‑like windows ---------------------------------------------
    for sigma in np.linspace(0.2, 0.5, 5):
        i = np.arange(n, dtype=np.float64)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / (sigma * (n - 1))) ** 2)
        seeds.append(a / a.sum())

    # --- random initial points (log‑softmax of Gaussian logits) ----------
    for _ in range(5):
        v = np.random.randn(n)
        seeds.append(_softmax(v))

    # --- previous best (if available) ------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            a_prev = np.asarray(prev, dtype=np.float64)
            if a_prev.ndim == 1:
                if a_prev.shape[0] != n:
                    # Truncate or pad with zeros
                    adj = np.zeros(n, dtype=np.float64)
                    m = min(a_prev.shape[0], n)
                    adj[:m] = a_prev[:m]
                    a_prev = adj
                if a_prev.sum() == 0:
                    a_prev = np.ones(n, dtype=np.float64) / n
                else:
                    a_prev = a_prev / a_prev.sum()
                seeds.append(a_prev)
        except Exception:
            pass

    # Shuffle a bit to avoid any ordering bias
    np.random.shuffle(seeds)
    return seeds


# ------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parameters)
# ------------------------------------------------------------

def _adam_optimize(
    n: int,
    seeds: list[np.ndarray],
    deadline: float,
) -> tuple[np.ndarray | None, float]:
    """
    Run a time‑aware Adam optimiser on each seed.
    Returns the best sequence found (as a probability vector) and its value.
    """
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Hyper‑parameters (tuned for this problem)
    lr0 = 0.8          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta_start = 2.0   # start of the smoothing parameter β
    beta_mul = 1.05    # geometric increase per iteration
    beta_max = 500.0   # cap to avoid numerical overflow

    # Process seeds one‑by‑one; each inner loop terminates when the
    # global deadline is reached.
    for s_idx, a0 in enumerate(seeds):
        # Initialise log‑softmax parameters (logits)
        v = np.log(a0 + 1e-30)
        m = np.zeros_like(v)      # first momentum
        v2 = np.zeros_like(v)     # second momentum
        t = 0

        while True:
            if time.time() > deadline:
                return best_seq, best_val

            t += 1
            # Probability vector (always on the simplex)
            a = _softmax(v)

            # Exact objective (sum = 1 ⇒ 2·n·max(conv))
            conv = _conv_fft(a, a)
            cur_max = np.max(conv)
            cur_val = 2.0 * n * cur_max

            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[adam] n={n:5d} seed={s_idx:2d} it={t:5d} val={cur_val:.6f}")

            # ---------- smoothed‑max (log‑sum‑exp) ----------
            beta = beta_start * (beta_mul ** t)
            if beta > beta_max:
                beta = beta_max
            shift = cur_max                     # improves numerical stability
            w = np.exp(beta * (conv - shift))   # weights ∝ exp(β·(c−max))
            w_sum = w.sum()
            if w_sum == 0.0:
                # Degenerate case – skip the update
                continue
            w /= w_sum

            # Gradient of the smooth loss w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]  # central n entries

            # Convert to gradient w.r.t. logits v (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            lr = lr0 / np.sqrt(t)   # gentle decay
            v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ------------------------------------------------------------
# Simple simulated‑annealing refinement (moves mass between two bins)
# ------------------------------------------------------------

def _local_search(
    n: int,
    a_start: np.ndarray,
    start_val: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Refine a candidate by repeatedly moving a small fraction of mass
    from one index to another.  Acceptance follows a Metropolis rule
    with exponential cooling.
    """
    best_seq = a_start.copy()
    best_val = start_val

    cur_seq = a_start.copy()
    cur_val = start_val

    # Temperature schedule (relative to the current best)
    T = 0.01 * best_val          # initial temperature ≈ 1 % of the objective
    decay = 0.9995               # geometric decay per iteration

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # Choose two distinct positions uniformly at random
        i, j = np.random.choice(n, 2, replace=False)

        # Only move mass from a non‑zero entry
        if cur_seq[i] < 1e-14:
            continue

        # Propose moving a random fraction (up to 5 %) of a[i] to a[j]
        step_frac = 0.05
        delta = step_frac * cur_seq[i] * np.random.rand()

        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta

        # Exact objective (sum = 1)
        conv = _conv_fft(cand, cand)
        cand_val = 2.0 * n * np.max(conv)

        # Metropolis acceptance
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(cand_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_seq = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                print(f"[local] it={iteration:6d} new best={best_val:.6f}")

        # Cool down
        T *= decay
        # Occasionally boost temperature a little to avoid stagnation
        if iteration % 5000 == 0:
            T = max(T, 0.001 * best_val)

    return best_seq, best_val

    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # Short fallback for extremely tiny budgets
    if budget_s < 5:
        n_small = 200
        return [1.0 / n_small] * n_small

    # ----------------------------------------------------------------
    # Choose a length.  8000 is a good trade‑off between flexibility
    # and computational cost; the objective is essentially scale‑free.
    # ----------------------------------------------------------------
    n = 8000

    # ----------------------------------------------------------------
    # 1️⃣  Build a large pool of diverse seeds.
    # ----------------------------------------------------------------
    all_seeds = _generate_seeds(n)

    # ----------------------------------------------------------------
    # 2️⃣  Quick scan – keep only the most promising seeds for Adam.
    # ----------------------------------------------------------------
    seed_vals = []
    for a in all_seeds:
        conv = _conv_fft(a, a)
        val = 2.0 * n * np.max(conv)
        seed_vals.append((a, val))
        if val < float("inf"):  # always true, just a placeholder
            pass

    # Sort by the cheap objective and retain the top‑k (k≈20)
    seed_vals.sort(key=lambda x: x[1])
    k_keep = min(20, len(seed_vals))
    selected_seeds = [a for a, _ in seed_vals[:k_keep]]

    # Keep the overall best from the quick scan as a fallback
    global_best_seq, global_best_val = seed_vals[0]

    # ----------------------------------------------------------------
    # 3️⃣  Adam optimisation on the selected seeds (time‑aware).
    # ----------------------------------------------------------------
    best_seq, best_val = _adam_optimize(n, selected_seeds, deadline)

    # Update global best if Adam improved it
    if best_seq is not None and best_val < global_best_val:
        global_best_seq = best_seq.copy()
        global_best_val = best_val

    # ----------------------------------------------------------------
    # 4️⃣  Local simulated‑annealing refinement (uses remaining time).
    # ----------------------------------------------------------------
    if time.time() < deadline:
        # Ensure we have a valid starting point
        if best_seq is None:
            best_seq = global_best_seq.copy()
            conv = _conv_fft(best_seq, best_seq)
            best_val = 2.0 * n * np.max(conv)
        best_seq, best_val = _local_search(n, best_seq, best_val, deadline)
        if best_val < global_best_val:
            global_best_seq = best_seq.copy()
            global_best_val = best_val

    # Return the final candidate as a list of floats
    return global_best_seq.tolist()
