def softmax_weights(conv, beta):
    max_val = conv.max()
    exp_vals = np.exp(beta * (conv - max_val))
    w = exp_vals / exp_vals.sum()

def _smooth_gradient_descent(a0, deadline, rng, target_score=1.48):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # schedule parameters
    start_eta = 0.5
    end_eta = 0.001
    start_beta = 0.5
    end_beta = 150.0  # high enough
    while time.time() < deadline:
        # compute convolution via FFT
        N_conv = 1 << ((2*n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=N_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=N_conv)[:2*n-1]
        max_conv = conv.max()
        approx_score = 2.0 * n * max_conv   # sum a =1
        # optionally compute exact if improvement
        if approx_score < best_score * 0.99:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(f"New best exact {best_score}")
                if best_score <= target_score:
                    break
        # Compute beta schedule
        elapsed = time.time() - (deadline - (budget_s-0.1))
        total = deadline - (deadline - (budget_s-0.1))
        beta = start_beta * (end_beta / start_beta) ** (elapsed / total)
        # Compute softmax weights
        # Subtract max for stability
        w_exp = np.exp(beta * (conv - max_conv))
        w = w_exp / w_exp.sum()
        # Compute gradient g via correlation
        # Compute FFT of w
        N_corr = 1 << (( (2*n-1) + n - 1 ).bit_length())

        # scaling for update
        g_max = g.max()
        if g_max == 0:
            break
        # step size schedule

    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    best_seq = None
    best_score = float('inf')
    best_approx = float('inf')

    # generate candidate shapes

        seq = _normalise_seq(raw)
        # compute approx_score using FFT conv
        conv = _conv_fft(seq)  # using our own fast method

    for _ in range(20):
        raw = rng.random(n)
        seq = _normalise_seq(raw)
        conv = _conv_fft(seq)
        approx = 2 * n * conv.max()
        seeds.append((approx, seq))

    # sort by approx
    seeds.sort(key=lambda x: x[0])
    # keep top K seeds
    top_seeds = [seq for approx, seq in seeds[:10]]
    # iterate refinement
    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        a = seed_seq / seed_seq.sum()

"""
Fast “soft‑max” gradient descent for the max‑convolution bound.

The evaluation function is homogeneous:
    evaluate(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

Hence we can restrict ourselves to L¹‑normalised non‑negative vectors
a ∈ ℝⁿ₊, Σaᵢ = 1, and minimise
        L(a) = maxₖ (a∗a)ₖ .
Direct sub‑gradient on the hard max converges slowly because only a single
constraint is active.  This implementation replaces the hard max by the
smooth “log‑sum‑exp’’ approximation
        L_β(a) = log ∑ₖ exp(β·(a∗a)ₖ) / β
with a temperature β that is annealed from a small to a large value.
The gradient of L_β is
        ∂L_β/∂a_i = 2·∑ₖ wₖ a_{k‑i},
where wₖ are the soft‑max weights.
We perform an exponentiated‑gradient update
        a ← a·exp(‑η·g/‖g‖∞)
and re‑normalise to Σaᵢ = 1.  The step size η decays linearly.
Occasional random mass transfers help escape plateaus.

A cheap systematic scan of a few analytic families provides several
promising seeds (exponential, power‑law, beta‑like, etc.).  Each seed is
refined with the smoother gradient descent within the remaining time
budget.  The best exact result (checked with the official
`evaluate_sequence`) is returned.

The whole search respects the 1000 s budget; if it finishes early the
current best sequence is emitted.
"""

def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """Scale a to satisfy the evaluator’s guards (max < 0.9, sum ≥ 0.01)."""
    a = np.asarray(a, dtype=float)
    max_v = a.max()
    if max_v > 0.0:
        a = a * (0.9 / max_v)
    total = a.sum()
    if total < 0.01:
        if total > 0.0:
            a = a * (0.01 / total)
        else:                                   # all zeros – give a tiny bump
            a = a + (0.01 / a.size)
    return a


def _candidate_shapes_ext(n: int):
    """
    Analytic families used for warm‑starts.
    Each returned array is *raw* (not normalised); the caller will normalise.
    """
    cand = []

    # 1. constant
    cand.append(np.ones(n, dtype=float))

    # 2. linear (decreasing / increasing)
    cand.append(np.arange(n, 0, -1, dtype=float))
    cand.append(np.arange(1, n + 1, dtype=float))

    # 3. quadratic (decreasing / increasing)
    cand.append(np.square(np.arange(n, 0, -1, dtype=float)))
    cand.append(np.square(np.arange(1, n + 1, dtype=float)))

    # 4. triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    cand.append(np.minimum(t, n - t + 1))

    # 5. valley (large at ends, small in centre)
    centre = (n - 1) / 2.0
    cand.append(1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2)))

    # 6. sine‑squared shape
    cand.append(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2)

    # 7. classic exponentials
    for r in (0.95, 0.9, 0.8, 0.7, 0.6, 0.5):
        cand.append(np.power(r, np.arange(n, dtype=float)))

    # 8. “slow” exponentials a_i = exp(-c·i/n)  (c from 0.5 to 4)
    for c in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0):
        r = np.exp(-c / n)
        cand.append(np.power(r, np.arange(n, dtype=float)))

    # 9. power‑law decays
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        cand.append(1.0 / (i ** p))

    # 10. beta‑like shapes a_i = (i+1)^α·(n‑i)^β
    for alpha in (0.5, 1.0, 2.0):
        for beta in (0.5, 1.0, 2.0):
            i = np.arange(1, n + 1, dtype=float)
            cand.append((i ** alpha) * ((n - i + 1) ** beta))

    # 11. continuous Beta‑pdf shapes (more symmetric)
    for alpha in (0.5, 1.0, 2.0):
        for beta in (0.5, 1.0, 2.0):
            x = (np.arange(n, dtype=float) + 0.5) / n
            cand.append((x ** (alpha - 1)) * ((1 - x) ** (beta - 1)))

    return cand


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a∗a using FFT.
    Returns a full vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())           # smallest power‑of‑two ≥ 2n‑1
    fa = np.fft.rfft(a, n=N)
    conv_full = np.fft.irfft(fa * fa, n=N)
    return conv_full[: 2 * n - 1]


def _smooth_gradient_descent(start_seq: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             target_score: float = 1.503) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the soft‑max approximation of the max‑convolution.
    start_seq must satisfy Σa = 1 and a_i ≥ 0.
    Returns the best L¹‑normalised sequence discovered before `deadline`.
    """
    a = start_seq.copy()
    n = a.shape[0]

    # keep the best *exact* score found so far
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # schedule parameters
    eta_start, eta_end = 0.5, 0.001
    beta_start, beta_end = 0.5, 200.0          # soft‑max temperature
    total_allowed = deadline - time.time()     # remaining time (updates each loop)

    # main loop
    while time.time() < deadline:
        # ------- FFT of a (single forward transform, reused below) ----------
        N = 1 << ((3 * n - 2).bit_length())   # ≥ 3n‑2, sufficient for both ops
        A = np.fft.rfft(a, n=N)

        # ------- convolution a∗a (linear) ---------------------------------
        conv_full = np.fft.irfft(A * A, n=N)
        conv = conv_full[: 2 * n - 1]          # length 2n‑1
        max_conv = conv.max()

        # ------- cheap surrogate score (exact = 2·n·max_conv) -------------
        approx_score = 2.0 * n * max_conv
        # evaluate exactly only if promising
        if approx_score < best_score * 0.99:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(f"[{time.time():.1f}s] New exact best = {best_score:.6f}")
                if best_score <= target_score:
                    print("Target reached – early exit from gradient phase.")
                    break

        # ------- soft‑max weights -----------------------------------------
        # β grows from beta_start to beta_end (exponential schedule)
        elapsed = time.time() - (deadline - total_allowed)
        fraction = min(1.0, elapsed / total_allowed)
        beta = beta_start * (beta_end / beta_start) ** fraction
        # stable softmax
        shifted = conv - max_conv
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:               # numerical safety
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # ------- gradient via correlation a⋆w (valid part) -----------------
        W = np.fft.rfft(w, n=N)
        corr_full = np.fft.irfft(A * np.conj(W), n=N)   # linear correlation a⋆w
        start = (2 * n - 1) - n          # = n‑1
        g = 2.0 * corr_full[start:start + n]              # length n

        # ------- exponentiated‑gradient step -------------------------------
        g_max = g.max()
        if g_max == 0.0:                 # degenerate – stop
            break

        eta = eta_start * (1.0 - fraction) + eta_end * fraction
        factor = np.exp(-eta * g / g_max)   # ∈ (0,1]
        a_new = a * factor
        s = a_new.sum()
        if s == 0.0:                     # protect against zero vector
            a_new = a.copy()
        else:
            a_new = a_new / s

        # occasional random mass transfer (1 % chance) to jump out of plateaus
        if rng.random() < 0.01:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            # transfer a small random fraction of the mass at i to j
            transferred = min(a_new[i], rng.random() * 0.01)
            a_new[i] -= transferred
            a_new[j] += transferred

        a = a_new

    return best_a.tolist()


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient list that minimises
    `evaluate_sequence`.  The routine respects the time budget,
    explores a few analytic families, refines the most promising seeds
    with a soft‑max gradient descent, and finally returns the best
    sequence found.
    """
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.1
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(np.asarray(hs1, dtype=float))
        seq = seq / seq.sum()
        w = _smooth_gradient_descent(seq, deadline, rng)
        sc = evaluate_sequence(w)
        if sc < best_score:
            best_score, best_seq = sc, w
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from height_sequence_1 → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap scan of analytic families (length 8192)
    # ------------------------------------------------------------------
    n = 8192
    raw_candidates = _candidate_shapes_ext(n)
    # also add a few random vectors
    for _ in range(15):
        raw_candidates.append(rng.random(n))

    # keep the top‑k candidates according to the cheap surrogate
    top_k = 12
    scored = []
    for raw in raw_candidates:
        if time.time() >= deadline:
            break
        seq = _normalise_seq(raw)
        # cheap surrogate: 2·n·max(a∗a)
        conv = _conv_fft(seq)
        approx = 2.0 * n * conv.max()
        scored.append((approx, seq))
    scored.sort(key=lambda x: x[0])
    seeds = [seq / seq.sum() for _, seq in scored[:top_k]]

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑gradient refinement of each seed
    # ------------------------------------------------------------------
    for idx, seed in enumerate(seeds):
        if time.time() >= deadline:
            break
        # give each seed a slice of the remaining time (but never less than 5 s)
        remaining = deadline - time.time()
        per_seed_deadline = time.time() + max(5.0, remaining / (len(seeds) - idx))
        refined = _smooth_gradient_descent(seed, per_seed_deadline, rng)
        exact = evaluate_sequence(refined)
        if exact < best_score:
            best_score, best_seq = exact, refined
            print(f"[{time.time() - start_time:.1f}s] New global best after seed {idx} → {best_score:.6f}")
            if best_score <= 1.503:
                print("Target reached – terminating search.")
                break

    # ------------------------------------------------------------------
    # If still not satisfied, try a few pure random restarts.
    # ------------------------------------------------------------------
    while time.time() < deadline and best_score > 1.503:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        rand = rng.random(n)
        rand = rand / rand.sum()
        per_restart_deadline = time.time() + min(30.0, remaining / 3.0)
        refined = _smooth_gradient_descent(rand, per_restart_deadline, rng)
        exact = evaluate_sequence(refined)
        if exact < best_score:
            best_score, best_seq = exact, refined
            print(f"[{time.time() - start_time:.1f}s] Random restart improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity: ensure we return a list of plain Python floats.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector (should never happen)
        best_seq = [1.0 / n] * n
    print(f"[{time.time() - start_time:.1f}s] Finished – best score={best_score:.6f}")
    return [float(x) for x in best_seq]
