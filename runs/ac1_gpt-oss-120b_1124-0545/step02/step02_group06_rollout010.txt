def _project_onto_simplex(v):
    # returns vector w of same shape with nonnegative entries sum to 1.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step‑function sequence by multi‑start projected gradient descent.

The algorithm consists of three stages:

1. **Random start generation** – draw many candidate shapes (power‑law,
   exponential, or the historic `height_sequence_1`) and keep the best
   according to a fast FFT‑based score.

2. **Refinement** – from each start run a projected gradient descent that
   minimises the L2 distance between the self‑convolution and the perfectly
   uniform distribution (the theoretical lower bound).  The gradient is
   obtained analytically:  ∇‖a*a‑u‖² = 2·( (a_rev) * (a*a‑u) ), where `a_rev`
   is the reversed vector.  FFT is used for both the convolution and the
   gradient.  After each step the vector is projected onto the probability
   simplex (non‑negative, sum = 1).

3. **Multi‑restart** – the refinement is performed repeatedly from different
   random starts until the time budget (default 1000 s) is exhausted.
   The best sequence encountered is returned.

All operations stay inside the time budget, use only NumPy and SciPy
(`scipy.optimize.linprog` is *not* required), and avoid any I/O.
"""


import time
import math
import numpy as np
from scipy.optimize import linprog  # not used for the main loop, but allowed


# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _initial_root_sequence(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1), normalised to sum = 1."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _power_law_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """a_i ∝ (i+α+1)^‑p, normalised to sum = 1."""
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _initial_root_sequence(n)
    return a / s


def _exp_sequence(n: int, beta: float) -> np.ndarray:
    """a_i ∝ exp(-beta·i), normalised to sum = 1."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-beta * idx)
    a /= a.sum()
    return a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()

    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise to exactly 1
    w_sum = w.sum()
    if w_sum <= 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


def _conv_self(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full self‑convolution (length 2·len(seq)‑1) using FFT."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    return conv


def _conv_error_grad(seq: np.ndarray, error: np.ndarray,
                     fft_len_err: int) -> np.ndarray:
    """
    Gradient of ½·‖seq*seq - target‖² w.r.t. `seq`.
    With error = (seq*seq - target), the gradient is
        2 * (error  *  rev(seq))
    where * denotes linear convolution.
    """
    rev = seq[::-1]
    f_err = np.fft.rfft(error, n=fft_len_err)
    f_rev = np.fft.rfft(rev, n=fft_len_err)
    full = np.fft.irfft(f_err * f_rev, n=fft_len_err)
    # slice corresponds to indices n‑1 … n‑1+len(seq)‑1
    start = seq.size - 1
    grad = 2.0 * full[start : start + seq.size]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    evaluation function.  The routine respects the given time budget
    (`budget_s` seconds) and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 0) Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000                                      # can be changed if desired
    conv_len = 2 * n - 1
    target_val = 1.0 / conv_len                   # uniform target for a*a

    # FFT lengths (next power of two for speed)
    fft_len_self = 1 << ((conv_len - 1).bit_length())
    fft_len_grad = 1 << ((conv_len + n - 2).bit_length())

    # ------------------------------------------------------------------
    # 1) Initialise – try to use the historic construction if present.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore
        init = np.maximum(init, 0.0)
        if init.size == 0 or init.sum() == 0.0:
            raise ValueError
        init /= init.sum()
        print(f"[init] using provided height_sequence_1 (n={init.size})")
    except Exception:
        init = _initial_root_sequence(n)
        print(f"[init] fallback to root sequence (n={n})")

    best_seq = init.copy()
    best_score = 2.0 * n * _conv_self(best_seq, fft_len_self).max()
    print(f"[baseline] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Random shape exploration (power‑law / exponential)
    # ------------------------------------------------------------------
    max_trials = 2000
    for trial in range(max_trials):
        if time.time() >= deadline:
            break
        which = np.random.rand()
        if which < 0.6:
            # power‑law shape
            p = np.random.uniform(0.3, 0.8)
            alpha = np.random.uniform(0.0, 5.0)
            cand = _power_law_sequence(n, p, alpha)
        else:
            # exponential shape
            beta = np.random.uniform(0.0005, 0.02)
            cand = _exp_sequence(n, beta)

        score = 2.0 * n * _conv_self(cand, fft_len_self).max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[shape] trial={trial:4d} score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) Multi‑restart projected gradient descent on the L2 error
    #    between a*a and the uniform distribution.
    # ------------------------------------------------------------------
    # Parameters for the optimisation
    max_outer = 50                     # number of independent starts (adjusted by time)
    max_inner = 5000                   # max inner iterations per start
    step_initial = 0.5
    momentum_beta = 0.9
    step_decay = 0.9999

    outer = 0
    while time.time() < deadline and outer < max_outer:
        outer += 1

        # ----- choose a fresh starting point -----
        if outer == 1:
            a = best_seq.copy()                # reuse the best shape found above
        else:
            # randomised start (re‑sample a shape)
            if np.random.rand() < 0.5:
                p = np.random.uniform(0.3, 0.85)
                alpha = np.random.uniform(0.0, 6.0)
                a = _power_law_sequence(n, p, alpha)
            else:
                beta = np.random.uniform(0.0005, 0.02)
                a = _exp_sequence(n, beta)
        a = _project_onto_simplex(a)             # ensure feasible (sum = 1)

        # ----- inner gradient descent -----
        step = step_initial
        momentum = np.zeros_like(a)

        for it in range(max_inner):
            if time.time() >= deadline:
                break

            # self‑convolution and current score
            conv = _conv_self(a, fft_len_self)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # compute error w.r.t. uniform target
            error = conv - target_val

            # gradient of ½·‖error‖²
            grad = _conv_error_grad(a, error, fft_len_grad)

            # momentum update
            momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad

            # tentative update and projection onto simplex
            cand = a - step * momentum
            cand = _project_onto_simplex(cand)

            # evaluate candidate
            conv_cand = _conv_self(cand, fft_len_self)
            cand_score = 2.0 * n * conv_cand.max()

            if cand_score < cur_score:
                # accept improvement
                a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a.copy()
                    print(f"[improve] outer={outer:2d} it={it:5d} score={best_score:.6f}")

                # slightly enlarge the step after success
                step *= 1.02
            else:
                # reject – back‑off step size
                step *= 0.5
                if step < 1e-8:
                    break

            # gradual decay to keep steps reasonable
            if (it + 1) % 200 == 0:
                step = max(step * step_decay, 1e-6)

            # early exit if we have already beaten the target bound
            if best_score < 1.45:
                break

        # after a run, give a tiny random shake to escape possible plateaus
        if time.time() < deadline:
            shake = np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a + shake)

    # ------------------------------------------------------------------
    # 4) Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f} n={n}")
    return best_seq.tolist()
