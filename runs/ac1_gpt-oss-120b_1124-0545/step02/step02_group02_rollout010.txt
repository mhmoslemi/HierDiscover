def simplex_projection(v):
    # Compute projection onto simplex {x >=0, sum(x)=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    start = time.time()
    deadline = start + budget_s - 5.0
    n = 5000
    # Phase 1: parameter search
    best_seq = None
    best_score = np.inf
    # simple uniform baseline
    seq_uniform = np.full(n, 1.0 / n)
    score_uniform = _score(seq_uniform)
    best_seq = seq_uniform.copy()
    best_score = score_uniform
    print(f"Uniform baseline score={best_score:.6f}")
    
    alphas = np.arange(0.45, 0.601, 0.005)  # 0.5 +- 0.05
    betas = [0.0, 2e-4, 5e-4, 1e-3]

    total = len(alphas) * len(betas) * len(offsets)

    for alpha in alphas:

                    break
                seq = _shape_sequence(n, alpha, beta, float(offs))
                # compute conv via FFT

def _simplex_projection(v):
    # Project onto simplex {x >= 0, sum = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero? fallback uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function with a very low `evaluate_sequence` score.

The algorithm works in three stages:

1. **Parametric sweep** – a 3‑parameter family of monotone decreasing
   sequences is examined:
        a_i ∝ (i+δ)^(-α)·exp(-β·(i+δ))
   where α∈[0.45,0.60], β∈{0,2e‑4,5e‑4,1e‑3} and δ is an integer shift.
   For each choice the exact score `2·n·max(conv)` (the denominator is 1
   after normalisation) is evaluated with a fast FFT‑based convolution.
   The best candidate is kept.

2. **Projected Adam optimisation** – starting from the best parametric
   candidate we minimise the maximal self‑convolution using a
   sub‑gradient of the `max` functional.
   The gradient is built from all convolution indices that are tied for
   the maximum (within a tiny tolerance) and the update is performed
   with the Adam optimiser.  After each step the vector is projected onto
   the probability simplex (non‑negative, sum = 1) by a simple
   clipping+renormalisation.

3. **Escaping plateaux** – every few hundred iterations a tiny random
   mass exchange between two coordinates is attempted; if it improves
   the score the change is kept.

The whole routine respects the 1000 s budget and prints progress.
The returned list can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _shape_sequence(n: int,
                    alpha: float,
                    beta: float,
                    offset: float) -> np.ndarray:
    """
    Build a monotone decreasing sequence

        a_i ∝ (i+offset)^(-α)·exp(-β·(i+offset))

    and normalise it to sum = 1.
    """
    idx = np.arange(n, dtype=float) + offset
    a = idx ** (-alpha)
    if beta != 0.0:
        a *= np.exp(-beta * idx)
    s = a.sum()
    if s == 0.0:
        a.fill(1.0 / n)
    else:
        a /= s
    return a


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Self‑convolution of ``seq`` using an FFT (real‑valued, fast).
    Returns the full convolution of length 2·n‑1.
    """
    n = seq.size
    n_fft = 1 << (int(np.ceil(np.log2(2 * n - 1))))   # next power of two
    f = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(f * f, n=n_fft)[: 2 * n - 1]
    # numerical noise may produce tiny negative values – clip to zero
    return np.maximum(conv, 0.0)


def _score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv)  (the denominator is 1 because sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project ``v`` onto the probability simplex {x ≥ 0, sum(x)=1}
    by clipping negatives and renormalising.
    """
    w = np.maximum(v, 0.0)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w.fill(1.0 / v.shape[0])
    return w


def _subgradient(seq: np.ndarray,
                 conv: np.ndarray,
                 max_val: float,
                 tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv)`` with respect to ``seq``.
    All indices whose convolution value is within ``tol`` of the maximum
    are treated as active.
    """
    n = seq.size
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=int)
        # contribution of a_i·a_{k-i}
        grad[i_range] += seq[k - i_range]
    # keep the update on the affine hyperplane sum(seq)=1
    grad -= grad.mean()
    return grad


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s``
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1️⃣  Choose a length – 5000 offers a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 2️⃣  Parametric sweep (α,β,δ) – keep the best candidate.
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n)               # uniform baseline
    best_score = _score(best_seq)
    print(f"[baseline] uniform score = {best_score:.6f}")

    alphas = np.arange(0.45, 0.601, 0.005)       # 0.45 … 0.60
    betas = [0.0, 2e-4, 5e-4, 1e-3]              # exponential decay strength
    offsets = np.arange(0, 6, 1)                 # integer shift

    for alpha in alphas:
        for beta in betas:
            for offs in offsets:
                if time.time() >= deadline:
                    break
                seq = _shape_sequence(n, alpha, beta, float(offs))
                sc = _score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[param] α={alpha:.4f}, β={beta:.5f}, δ={offs} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Adam‑based projected gradient descent on the simplex.
    # ------------------------------------------------------------------
    m = np.zeros_like(best_seq)                 # 1st moment
    v_vec = np.zeros_like(best_seq)             # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.05
    it = 0
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_fft(best_seq)
        max_val = conv.max()

        # sub‑gradient of the maximum convolution entry
        grad = _subgradient(best_seq, conv, max_val, tol=1e-12)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_vec / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        candidate = _simplex_projection(best_seq - step)

        # evaluate the new point
        cand_score = _score(candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate
            print(f"[improve] iter={it:<6d} score={best_score:.6f}")

        # occasional random mass swap to escape flat regions
        if it % 100 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps_swap = min(best_seq[i] * 0.2, best_seq[i] * 0.5)
            tmp = best_seq.copy()
            tmp[i] -= eps_swap
            tmp[j] += eps_swap
            tmp = _simplex_projection(tmp)
            swap_score = _score(tmp)
            if swap_score < best_score:
                best_score = swap_score
                best_seq = tmp
                print(f"[swap]   iter={it:<6d} score={best_score:.6f}")

        # slowly decay the learning rate for stability
        if it % 5_000 == 0:
            base_lr *= 0.9

    print(f"[final]  best_score={best_score:.6f} after {it} iterations")
    return best_seq.tolist()
