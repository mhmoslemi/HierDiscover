def greedy_sidon(n, max_trials=200):
    best_set = set()
    best_len = 0
    for t in range(max_trials):
        # random order of indices
        perm = np.random.permutation(n)
        S = []
        sums = set()
        for idx in perm:
            # compute candidate sums with existing S
            conflict = False
            for s in S:
                if (idx + s) in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # also need to check self sum idx+idx not colliding
            if (2*idx) in sums:
                continue
            # If okay, add
            # Add self sum
            sums.add(2*idx)
            for s in S:
                sums.add(idx + s)
            S.append(idx)
        # track best
        if len(S) > best_len:
            best_len = len(S)
            best_set = set(S)
            # early exit if we reached theoretical max (floor solution)
            if best_len >= int(math.floor((math.sqrt(8*n-7)-1)/2)):
                break
    return list(best_set)

"""
propose_candidate – hybrid Adam‑descent + greedy local refinement.

The algorithm works in three stages:

1. **Initialisation** – start from an existing construction if it exists,
   otherwise from a uniform vector or a random exponential vector.  The
   ℓ₁–norm is fixed to `sqrt(2·n)` (the objective is scale invariant).

2. **Smoothed Adam optimisation** – minimise a log‑sum‑exp smoothing of the
   maximum self‑convolution.  The smoothing parameter `eps` is driven from
   `0.5` down to `1e‑5`, so the algorithm gradually turns into a (projected)
   sub‑gradient descent for the true max.  Adam moments are kept and the
   iterate is projected onto the probability simplex after every step.

3. **Greedy local search** – starting from the best sequence found by the
   Adam phase, repeatedly move a small amount of mass from a randomly chosen
   index to another one (keeping the ℓ₁‑norm fixed).  Moves are accepted only
   if they strictly improve the maximum convolution value.  This simple hill
   climbing often reduces the objective a few percent further.

All computation respects the 1000 s budget.  Progress (new best values and
timings) is printed to stdout.  The returned list contains non‑negative
heights whose sum equals `sqrt(2·n)`, ready to be fed to `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np


def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all entries equal
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray) -> float:
    """Return max_{t} (a * a)[t] (the objective after ℓ₁‑normalisation)."""
    return float(np.convolve(a, a).max())


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  Returns the best sequence found before the time
    budget expires.
    """
    # ----------------------------------------------------------------------
    # Setup
    np.random.seed(seed)
    random.seed(seed)

    if "height_sequence_1" in globals():
        base_seq = np.asarray(height_sequence_1, dtype=float)
    else:
        base_seq = None

    n = len(base_seq) if base_seq is not None else 2500
    target_sum = math.sqrt(2.0 * n)          # ∑a = √(2n)

    # ----------------------------------------------------------------------
    # Several cheap starters – pick the best as initial point
    def _normalise(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0:
            v[:] = target_sum / n
        else:
            v = v / s * target_sum
        return v

    uniform = np.full(n, target_sum / n)
    rand_exp = np.random.exponential(scale=1.0, size=n)
    rand_exp = _normalise(rand_exp)

    starters = [uniform, rand_exp]
    if base_seq is not None:
        starters.append(_normalise(base_seq))

    best_seq = None
    best_score = np.inf
    for cand in starters:
        sc = _max_convolution(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[init] Best start score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Timing
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ----------------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (log‑sum‑exp)
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    t_it = 0

    # schedules (more aggressive than the previous code)
    eps_start, eps_end = 0.5, 1e-5          # smoothing schedule
    lr_start = target_sum * 0.001           # learning‑rate schedule
    lr_end   = target_sum * 0.0002

    last_improve = 0
    it = 0
    perturb_every = 150          # more frequent perturbations
    stagnant_limit = 2000        # restart earlier than before

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)

        eps = eps_start * ((eps_end / eps_start) ** frac)
        lr  = lr_start  * ((lr_end  / lr_start ) ** frac)

        # current objective
        conv = np.convolve(a, a)
        cur_score = float(conv.max())

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = it
            print(f"[{it:6d}] New best  max(conv) = {best_score:.6f} "
                  f"(t={elapsed:.1f}s, eps={eps:.4e})")

        # ----------- smooth max weights (log‑sum‑exp) -------------
        conv_shift = conv - conv.max()
        # stability: large negative values may underflow → use 1‑hot fallback
        with np.errstate(over='ignore', under='ignore'):
            w_raw = np.exp(conv_shift / eps)
        w_sum = w_raw.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w_raw / w_sum

        # ---------- gradient of the smoothed objective ----------
        # sigma = moving sum of w over a window of length n
        cumsum_w = np.cumsum(w)
        sigma = cumsum_w[n-1:2*n-1] - np.concatenate(([0.0], cumsum_w[:n-1]))

        a_rev = a[::-1]
        conv_wa = np.convolve(w, a_rev)        # length 3n‑2, but we only need n entries
        first = conv_wa[(n-1):(n-1+n)]

        grad = first + a * sigma                # size n

        # -------------------- Adam update -----------------------
        t_it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_it)
        v_hat = v / (1.0 - beta2 ** t_it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _simplex_projection(a_new, target_sum)
        a = a_new

        # ------------------- occasional perturbation -------------
        if it % perturb_every == 0 and it > 0:
            # blend best solution with fresh noise
            rnd = np.random.exponential(scale=1.0, size=n)
            rnd = _normalise(rnd)
            mix = random.random()
            a_pert = mix * best_seq + (1.0 - mix) * rnd
            a_pert = _simplex_projection(a_pert, target_sum)

            score_pert = _max_convolution(a_pert)
            if score_pert < best_score:
                best_score = score_pert
                best_seq = a_pert.copy()
                last_improve = it
                print(f"[{it:6d}] New best after perturbation: {best_score:.6f}")

            if score_pert < cur_score:
                a = a_pert

        # ------------------- stagnation restart -----------------
        if it - last_improve > stagnant_limit:
            noise = np.random.exponential(scale=1.0, size=n)
            noise = _normalise(noise)
            a = 0.5 * best_seq + 0.5 * noise
            a = _simplex_projection(a, target_sum)
            last_improve = it
            print(f"[{it:6d}] Restarted after stagnation (no improvement for {stagnant_limit} it)")

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – greedy local refinement (hill climbing)
    # Continue only if we have a few seconds left.
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("[local] Starting greedy local search")
        a_local = best_seq.copy()
        cur_score = best_score
        nonzero_inds = np.where(a_local > 1e-12)[0]
        # parameters for the hill climbing
        move_ratio = 0.15             # at most this fraction of source mass is moved
        temp_start = 0.02 * cur_score
        temp_end = 0.0
        ls_start = time.time()
        ls_deadline = ls_start + remaining - 1.0
        ls_it = 0
        while time.time() < ls_deadline:
            # annealing temperature (linear decay)
            ls_elapsed = time.time() - ls_start
            ls_frac = ls_elapsed / (ls_deadline - ls_start)
            temperature = temp_start * (1.0 - ls_frac) + temp_end

            # pick a source index with positive mass
            i = random.choice(nonzero_inds)
            ai = a_local[i]
            if ai <= 0:
                # should be rare – rebuild the list and continue
                nonzero_inds = np.where(a_local > 1e-12)[0]
                continue

            # choose destination (different)
            j = random.randrange(n)
            while j == i:
                j = random.randrange(n)

            # propose moving a fraction of mass
            delta = ai * random.random() * move_ratio
            if delta <= 0:
                continue

            a_new = a_local.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # renormalise tiny drift
            s = a_new.sum()
            if abs(s - target_sum) > 1e-9:
                a_new *= target_sum / s

            new_score = _max_convolution(a_new)

            # accept if better, otherwise maybe uphill with small prob
            if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
                a_local = a_new
                cur_score = new_score
                # update non‑zero list
                if a_new[i] <= 1e-12:
                    nonzero_inds = np.delete(nonzero_inds, np.where(nonzero_inds == i))
                if a_new[j] > 1e-12 and j not in nonzero_inds:
                    nonzero_inds = np.append(nonzero_inds, j)

                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_local.copy()
                    print(f"[local {ls_it:6d}] New best = {best_score:.6f}")

            ls_it += 1

    print(f"[final] Best max(conv) = {best_score:.6f}")
    return best_seq.tolist()
