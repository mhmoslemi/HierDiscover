def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

assistantfinal**Solution Overview**

The goal is to find a non‑negative sequence `a` (step‑function) that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}.
\]

Because the objective is *scale invariant* we restrict the search to the
probability simplex  

\[
\sum_i a_i = 1,\qquad a_i\ge 0 .
\]

The algorithm consists of three phases

1. **Analytic start** – a small pool of analytically motivated candidates
   (uniform, inverse‑sqrt “arcsine”, mixed power laws).  
   The best of them becomes the initial point.

2. **Smoothed projected gradient descent** – the nondifferentiable
   `max` is replaced by a softmax (log‑sum‑exp) with a temperature
   parameter `α`.  
   For a given `α` the gradient is

   \[
   \nabla_{\!a}\;M_\alpha(a)=2\;(a^{\rm rev}*w),\qquad
   w_k\propto\exp\!\bigl(\alpha\,( (a*a)_k-M(a) )\bigr),
   \]

   where `a_rev` is the reversed vector and `*` denotes convolution.
   The gradient step is projected back onto the simplex.
   The temperature `α` is increased linearly during the run, turning the
   smoothed objective into the exact `max`.

3. **Exploration & restarts** – with a small probability a cheap random
   mass‑transfer move is tried; if the search stagnates for more than
   30 s a jitter around the current best solution is applied.

All convolutions are performed with the FFT (complexity
\(O(n\log n)\)), which makes the method fast enough for sequences of a
few thousand entries.  The whole routine respects the 1000 s budget,
prints progress and finally returns the best sequence found.

```python
"""
Simulated‑annealing + smoothed gradient descent for
minimising  2·n·max(conv(a,a)) / (sum(a)^2)  on the simplex.
The implementation works with FFT‑based convolutions, a soft‑max
approximation of the infinity norm and occasional random perturbations.
It is able to handle sequences of several thousand entries within the
1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple random mass‑transfer move (exploration)
# ----------------------------------------------------------------------
def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Exact sub‑gradient step (uses only the current maximal entry)
# ----------------------------------------------------------------------
def gradient_step_exact(a: np.ndarray, step: float, L: int) -> np.ndarray:
    """Projected sub‑gradient descent w.r.t. the exact max entry."""
    n = a.shape[0]
    Fa = np.fft.fft(a, L)
    conv = np.fft.ifft(Fa * Fa).real[: 2 * n - 1]
    k_max = int(np.argmax(conv))

    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)

    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k_max - idx]

    a_new = a - step * g
    a_new = np.clip(a_new, 0.0, None)
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Smoothed gradient step (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------
def gradient_step_smooth(a: np.ndarray,
                         step: float,
                         alpha: float,
                         L: int,
                         n: int) -> np.ndarray:
    """One descent step on the smoothed max‑convolution objective."""
    # ----- convolution a*a (FFT) -----
    Fa = np.fft.fft(a, L)
    conv = np.fft.ifft(Fa * Fa).real[: 2 * n - 1]

    # ----- soft‑max weights -----
    max_c = conv.max()
    # shift to avoid overflow
    exp_arg = alpha * (conv - max_c)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum

    # ----- gradient = 2 * (a_rev * w_norm) -----
    a_rev = a[::-1]
    Fa_rev = np.fft.fft(a_rev, L)
    Fw = np.fft.fft(w_norm, L)
    grad_full = np.fft.ifft(Fa_rev * Fw).real
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]

    # ----- update & project -----
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, None)
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Helper to compute the exact objective (fast, FFT based)
# ----------------------------------------------------------------------
def objective_maxconv(a: np.ndarray, L: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    Fa = np.fft.fft(a, L)
    conv = np.fft.ifft(Fa * Fa).real[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic candidate pool
# ----------------------------------------------------------------------
def initialise_candidates(n: int) -> list[np.ndarray]:
    """Return a short list of analytically motivated probability vectors."""
    candidates = []

    # 1) Uniform
    candidates.append(np.ones(n) / n)

    # 2) Inverse‑sqrt “arcsine‑like” with 0.5 offset
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    candidates.append(a / a.sum())

    # 3) Mixed inverse‑sqrt powers
    for p in (0.3, 0.5, 0.7, 1.0):
        a = (i + 1) ** (-p) + (n - i) ** (-p)
        candidates.append(a / a.sum())

    # 4) Pure inverse‑sqrt product (no offset)
    a = 1.0 / np.sqrt((i + 1) * (n - i))
    candidates.append(a / a.sum())

    return candidates


# ----------------------------------------------------------------------
# Main entry point required by the external harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function of length ``n`` that minimises the
    evaluation function.  The routine combines analytic start points,
    a smoothed projected gradient descent (with an annealed soft‑max
    temperature), occasional random moves and restarts.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    n = int(kwargs.get("n", 3000))          # default length
    # FFT length (next power of two ≥ 2·n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 1 – Build and evaluate analytic start pool
    # ------------------------------------------------------------------
    cand_pool = initialise_candidates(n)

    # If a previously found sequence exists, add it as a candidate
    hs1 = globals().get("height_sequence_1")
    if isinstance(hs1, list) and len(hs1) == n:
        a = np.array(hs1, dtype=float)
        if a.sum() > 0:
            cand_pool.append(a / a.sum())

    best_seq = None
    best_val = np.inf

    for idx, a in enumerate(cand_pool):
        val = objective_maxconv(a, L)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[init] candidate {idx:02d} score = {val:.6f}")

    # Fallback – should never happen
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_val = objective_maxconv(best_seq, L)

    a = best_seq.copy()
    print(f"[init] chosen start score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2 – Optimisation loop (smoothed gradient + exploration)
    # ------------------------------------------------------------------
    # annealing / schedule parameters
    init_temp = 1.0
    final_temp = 0.001
    init_step = 0.20
    final_step = 1e-5
    init_alpha = 0.2
    final_alpha = 8.0

    iteration = 0
    last_improve = start_time

    while time.time() < deadline:
        now = time.time()
        elapsed = now - start_time
        progress = min(elapsed / (budget_s - 10.0), 1.0)

        # Linear schedules
        temp = init_temp * (1.0 - progress) + final_temp * progress
        step = init_step * (1.0 - progress) + final_step * progress
        alpha = init_alpha * (1.0 - progress) + final_alpha * progress

        # ------------------------------------------------------------------
        # Choose move type
        # ------------------------------------------------------------------
        if np.random.rand() < 0.80:                     # gradient‑type moves
            if np.random.rand() < 0.65:                 # smoothed gradient
                cand = gradient_step_smooth(a, step, alpha, L, n)
            else:                                       # exact max‑subgradient
                cand = gradient_step_exact(a, step, L)
        else:                                            # cheap random perturbation
            cand = random_move(a, step * 0.5)

        # ------------------------------------------------------------------
        # Evaluate candidate
        # ------------------------------------------------------------------
        cand_val = objective_maxconv(cand, L)

        # Metropolis–Hastings acceptance (allows occasional uphill moves)
        accept = False
        if cand_val < best_val:
            accept = True
            best_val = cand_val
            best_seq = cand.copy()
            last_improve = now
            print(f"[{iteration:06d}] new best = {best_val:.6f} (t={elapsed:.1f}s)")
        else:
            # probability to accept a worse move
            prob = math.exp(-(cand_val - best_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand

        # ------------------------------------------------------------------
        # Stagnation handling – jitter and restart
        # ------------------------------------------------------------------
        if iteration % 20000 == 0 and (now - last_improve) > 30.0:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            print(f"[{iteration:06d}] restart after stagnation")

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best score={best_val:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
```
