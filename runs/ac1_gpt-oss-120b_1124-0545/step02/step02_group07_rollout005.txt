"""
Propose a step‑function sequence with a low bound on `evaluate_sequence`.

Algorithm
---------

1️⃣  **Exploratory scan (≈ 15 % of the budget)**
    * Generate a diverse collection of analytic families (constant,
      linear, exponential, power‑law, beta‑like, sine‑squared, mixtures).
    * Normalise each shape (max < 0.9, Σ≥0.01) – scaling does not affect the
      objective because the evaluator is homogeneous.
    * Compute the *linear* convolution `a*a` with an FFT (fast, O(n log n)).
    * Keep the best shape by the cheap surrogate score  
          `2·n·max(a*a)`.
      Only if the surrogate looks promising the full `evaluate_sequence`
      is called.

2️⃣  **Smoothed Adam optimisation**
    * Work with a fixed length `n` (the length of the best shape from step 1)
      and enforce Σ aᵢ = 1 via a soft‑max parametrisation:
          a = softmax(x) .
    * Objective to minimise is the *maximum* convolution entry.
      We replace `max` by a smooth approximation (log‑sum‑exp) with a
      temperature β that is increased linearly from 1.0 to 200.0.
    * Gradient of the smooth maximum:
          g = 2·( w  *  a_rev )_[n‑1 : 2n‑1] ,
      where `w` are the soft‑max weights of the convolution
      (size `2n‑1`) and `a_rev = a[::-1]`.  The convolution is performed
      with a second FFT.
    * The gradient w.r.t. the logits `x` is obtained via the chain rule
          ∂L/∂x = a ∘ ( g  –  ⟨a , g⟩ ) .
    * Adam (β₁=0.9, β₂=0.999) with a linearly decaying step‑size
      (0.08 → 0.001) updates `x`.  A tiny clip keeps the logits in a
      numerically safe range.
    * Every few seconds the exact evaluator is called; the best sequence
      seen so far is stored.  Random Gaussian perturbations of the logits
      (1 % chance) help to escape plateaus.

The optimisation is fully deterministic apart from the seeded RNG.
The whole routine respects the supplied time budget (≈1000 s) and
returns the best sequence found.
"""

# ----------------------------------------------------------------------
# Convolution utilities (FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)
    return conv[: 2 * n - 1]


def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (length = len(a)+len(b)-1) using FFT."""
    n, m = a.shape[0], b.shape[0]
    N = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)
    return conv[: n + m - 1]

# Normalisation / helper utilities

def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """Scale to max < 0.9 and Σ ≥ 0.01 (scaling does not affect the score)."""
    a = np.asarray(a, dtype=float)
    max_v = a.max()
    if max_v > 0.0:
        a = a * (0.9 / max_v)
    total = a.sum()
    if total < 0.01:
        if total > 0.0:
            a = a * (0.01 / total)
        else:
            a = a + (0.01 / a.size)
    return a


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()

def _candidate_shapes_ext(n: int):
    """Yield raw (unnormalised) shape vectors of length `n`."""
    cand = []

    # constant
    cand.append(np.ones(n, dtype=float))

    # linear decreasing / increasing
    cand.append(np.arange(n, 0, -1, dtype=float))
    cand.append(np.arange(1, n + 1, dtype=float))

    # quadratic
    cand.append(np.square(np.arange(n, 0, -1, dtype=float)))
    cand.append(np.square(np.arange(1, n + 1, dtype=float)))

    # triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    cand.append(np.minimum(t, n - t + 1))

    # valley (large at ends, small in centre)
    centre = (n - 1) / 2.0
    cand.append(1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2)))

    # sine‑squared
    cand.append(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2)

    # classic exponential decays
    for r in (0.95, 0.9, 0.8, 0.7, 0.6, 0.5):
        cand.append(np.power(r, np.arange(n, dtype=float)))

    # near‑unit exponential decays a_i = exp(-c·i/n)
    for c in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0):
        r = np.exp(-c / n)
        cand.append(np.power(r, np.arange(n, dtype=float)))

    # power‑law decays
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        cand.append(1.0 / (i ** p))

    # beta‑like shapes a_i = (i+1)^α·(n‑i)^β
    for alpha in (0.5, 1.0, 2.0):
        for beta in (0.5, 1.0, 2.0):
            i = np.arange(1, n + 1, dtype=float)
            cand.append((i ** alpha) * ((n - i + 1) ** beta))

    # random mixtures of two exponentials
    rng = np.random.default_rng(12345)
    for _ in range(6):
        c1 = rng.uniform(0.3, 4.0)
        c2 = rng.uniform(0.3, 4.0)
        split = rng.integers(1, n - 1)
        r1 = np.exp(-c1 / n)
        r2 = np.exp(-c2 / n)
        part = np.concatenate(
            [np.power(r1, np.arange(split, dtype=float)),
             np.power(r2, np.arange(n - split, dtype=float))])
        cand.append(part)

    return cand

def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a  smoothmax_β (a*a) .
    Returns a vector `g` of length n: g_i = 2·∑_j w_{i+j}·a_j,
    where w are the soft‑max weights of the convolution.
    """
    n = a.shape[0]

    # convolution a*a
    conv = _conv_fft(a)                     # length 2n-1
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    exp_w = np.exp(beta * (conv - max_c))
    w = exp_w / exp_w.sum()                # length 2n-1, ∑ w = 1

    # gradient = 2 * (w * a_rev)_{n-1 … 2n-2}
    a_rev = a[::-1]
    conv_full = _conv_full_fft(w, a_rev)   # length 3n-2
    g = 2.0 * conv_full[n - 1 : 2 * n - 1]
    return g


def _adam_optimize(start_a: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   target_score: float = 1.503) -> list[float]:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    `start_a` must be a probability vector (Σa=1, a≥0).
    Returns the best sequence (as a plain Python list) discovered.
    """
    n = start_a.shape[0]
    # initialise logits from a (avoid log(0))
    eps = 1e-30
    x = np.log(start_a + eps)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # schedules
    start_lr, end_lr = 0.08, 0.001        # learning‑rate schedule
    beta_start, beta_end = 1.0, 200.0    # smoothing‑temperature schedule

    best_a = start_a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    start_time = time.time()
    last_eval = start_time
    eval_interval = 4.0                  # seconds between exact evaluations
    t = 0

    while time.time() < deadline:
        # -------------------------------------------------------------
        # current point (soft‑max ensures feasibility)
        # -------------------------------------------------------------
        a = _softmax(x)                     # Σa = 1, a ≥ 0

        # *smoothed* gradient
        elapsed = time.time() - start_time
        total = deadline - start_time
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        g = _smooth_gradient(a, beta)

        # gradient wrt logits (mirror‑descent on simplex)
        dot_ag = np.dot(a, g)
        grad_x = a * (g - dot_ag)

        # -------------------------------------------------------------
        # Adam step
        # -------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = start_lr + (end_lr - start_lr) * (elapsed / total)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        x = x - step
        # keep logits in a safe numerical range
        np.clip(x, -50.0, 50.0, out=x)

        # occasional random perturbation (helps escape local minima)
        if rng.random() < 0.01:
            x += rng.normal(scale=0.5, size=n)

        # -------------------------------------------------------------
        # occasional exact evaluation
        # -------------------------------------------------------------
        if time.time() - last_eval >= eval_interval:
            cur_seq = a.tolist()
            exact = evaluate_sequence(cur_seq)
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"New best exact score={best_score:.6f}")
                if best_score <= target_score:
                    print("Target reached – terminating optimisation.")
                    break
            last_eval = time.time()

    return best_a.tolist()

# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient list that minimises `evaluate_sequence`.
    The routine respects the given time budget and returns the best sequence
    found (as a plain list of floats).  It combines a fast analytic scan with
    a smoothed Adam optimisation on the max‑convolution objective.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously known construction, if available
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(np.asarray(hs1, dtype=float))
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from "
                  f"height_sequence_1 → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic scan of analytic families
    # ------------------------------------------------------------------
    # Allocate roughly 15 % of the total budget for the scan.
    scan_deadline = start_time + 0.15 * (deadline - start_time)
    lengths = [2048, 4096, 8192]
    for n in lengths:
        if time.time() >= scan_deadline:
            break
        print(f"[{time.time() - start_time:.1f}s] Scanning length {n}")
        for raw in _candidate_shapes_ext(n):
            if time.time() >= scan_deadline:
                break
            seq = _normalise_seq(raw)
            # cheap surrogate score (no call to the expensive evaluator)
            conv = _conv_fft(seq)
            surrogate = 2.0 * n * conv.max()
            # we only invoke the official evaluator when the surrogate is promising
            if surrogate < best_score * 1.05:
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    print(f"[{time.time() - start_time:.1f}s] "
                          f"New best (len={n}) score={best_score:.6f}")
                if best_score <= 1.503:
                    break

    # ------------------------------------------------------------------
    # Guard: if nothing was accepted, start from a simple exponential shape
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8192
        r = np.exp(-2.0 / n)
        best_seq = (np.power(r, np.arange(n, dtype=float))).tolist()
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – smoothed Adam optimisation (main work)
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)
    cur = cur / cur.sum()                     # enforce Σ=1
    refined = _adam_optimize(cur, deadline, rng)

    # final verification – keep absolute best
    final_score = evaluate_sequence(refined)
    if final_score < best_score:
        best_score = final_score
        best_seq = refined

    print(f"[{time.time() - start_time:.1f}s] Finished – best score={best_score:.6f}")

    # Ensure we return plain Python floats (the API requires this)
    return [float(x) for x in best_seq]
