    # sort results
    results.sort(key=lambda x: x[1])
    top_k = results[:5]  # adjust based on remaining time
    # long runs
    for idx, (seed_seq, seed_val) in enumerate(top_k):
        if time.time() > deadline:
            break
        # compute allowed remaining time
        # run optimizer longer
        best_seq, best_val = _optimise_seed(seed_seq, n, deadline,
                                            max_iters=8000,
                                            switch_iter=3000)
        if best_val < global_best_val:
            global_best_seq, global_best_val = best_seq, best_val
            print("[overall] new global best after long refinement idx={} val={}".format(idx, best_val))
    # fallback uniform if not found
    if global_best_seq is None:
        global_best_seq = np.full(n, 1.0/n)
    return global_best_seq.tolist()

"""
Propose a low‑autocorrelation non‑negative step sequence.

The method works in two phases (all on the probability simplex
via a soft‑max parametrisation).

1️⃣  **Seed generation** – a diverse set of simple families
    (geometric, power‑law, combined, symmetric, sine window,
    Gaussian, random Dirichlet, and very sparse “comb” vectors)
    is built for the target length (default 8000).  All seeds are
    normalised to sum‑to‑one.

2️⃣  **Optimisation** – each seed is refined with a projected
    Adam optimiser.  The loss is the *max* of the self‑convolution
    (the quantity we finally want to minimise).  For the first
    part of the run a smooth approximation (log‑sum‑exp) is used;
    after a configurable number of iterations the weight collapses
    to a hard max, giving a true sub‑gradient of the objective.
    Adam’s adaptive moments keep the iterates on the simplex
    automatically.  Short runs (≈3000 it) are performed for all
    seeds; the best few are then continued for many more iterations
    (up to the remaining time budget).

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list of floats.
All entries are non‑negative and sum to one, so the call
``evaluate_sequence`` will never be rejected.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, stable soft‑max, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors via FFT.
    Returns the full result of length len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (result sums to 1, all entries >0)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Seed families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``alpha`` (scalar)."""
    draws = np.random.gamma(alpha, 1.0, size=n)
    if draws.sum() == 0:
        draws = np.ones(n)
    return draws / draws.sum()


def _random_sparse(n: int, k: int) -> np.ndarray:
    """Create a sparse vector with exactly ``k`` random non‑zero positions."""
    a = np.zeros(n, dtype=np.float64)
    indices = np.random.choice(n, size=min(k, n), replace=False)
    values = np.random.rand(len(indices)) + 1e-3   # avoid exact zero
    a[indices] = values
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Optimisation of a *single* seed (Adam on the log‑softmax parameters)
# ----------------------------------------------------------------------


def _optimise_seed(
    init_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 8000,
    switch_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of one starting vector ``init_a``.
    Returns the best sequence (as a numpy array) and its raw objective value
    ``2 * n * max(conv(a, a))`` (sum(a) is 1 by construction).
    """
    # Hyper‑parameters
    lr0 = 0.6          # base learning rate (Adam)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    # Smoothing schedule for the early phase
    beta_start = 0.1   # low temperature – smooth loss
    beta_mul = 1.05

    # Initialise log‑softmax parameters
    v = np.log(np.clip(init_a, 1e-30, None))
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    t = 0  # iteration counter (1‑based for bias‑correction)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1

        # ---------- probability vector (always on the simplex) ----------
        a = _softmax(v)

        # ---------- current objective ----------
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(
                f"[opt] len={n:5d} it={it:5d} new_best={cur_val:.6f}"
            )

        # ---------- choose weighting for the gradient ----------
        if it < switch_iter:
            # smooth log‑sum‑exp approximation of the max
            beta = beta_start * (beta_mul ** it)
            shift = cur_max
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0 or np.isnan(w_sum):
                # fallback to hard max (numerical under‑flow)
                w = np.zeros_like(conv)
                max_idxs = np.flatnonzero(conv >= cur_max - 1e-12)
                w[max_idxs] = 1.0 / len(max_idxs)
            else:
                w /= w_sum
        else:
            # hard max – a genuine sub‑gradient of the true max‑loss
            w = np.zeros_like(conv)
            max_idxs = np.flatnonzero(conv >= cur_max - 1e-12)
            w[max_idxs] = 1.0 / len(max_idxs)

        # ---------- gradient w.r.t. a ----------
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]         # central n entries

        # ---------- convert to gradient w.r.t. the logits ----------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        # learning‑rate schedule: decay like 1/√t
        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # Return the best interior point we have seen
    if best_seq is None:
        best_seq = _softmax(v)  # fallback
        best_val = 2.0 * n * np.max(_conv_fft(best_seq, best_seq))
    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generator for a chosen length
# ----------------------------------------------------------------------


def _generate_seeds(n: int) -> list[np.ndarray]:
    """Produce a modest, diversified collection of normalised seeds."""
    seeds = []

    # 1. Geometric families (r = 1 - c/n)
    c_vals = np.linspace(0.5, 5.0, 12)
    for c in c_vals:
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # 2. Pure power‑law families
    p_vals = np.linspace(0.0, 2.2, 12)
    for p in p_vals:
        seeds.append(_power(n, p))

    # 3. Combined geometric × power
    for c in np.linspace(0.5, 4.0, 8):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 8):
            seeds.append(_geom_power(n, r, p))

    # 4. Symmetric geometric
    for c in np.linspace(0.5, 4.5, 8):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # 5. Smooth sine window
    seeds.append(_sine_window(n))

    # 6. Gaussian windows (various widths)
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        seeds.append(_gaussian(n, sigma))

    # 7. Random Dirichlet samples (different concentrations)
    for alpha in (0.1, 0.3, 0.7, 1.0, 2.0, 5.0):
        seeds.append(_random_dirichlet(n, alpha))

    # 8. Very sparse “comb” vectors (few non‑zeros)
    for k in (2, 3, 5, 10, 20):
        seeds.append(_random_sparse(n, k))

    # 9. Small random logits (softmax of Gaussian)
    for _ in range(5):
        v_rand = np.random.randn(n)
        seeds.append(_softmax(v_rand))

    # 10. If a previous solution is available, reuse it (scaled if needed)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            a_prev = np.asarray(prev, dtype=np.float64)
            if a_prev.shape == (n,):
                seeds.append(a_prev / a_prev.sum())
            else:
                # simple resizing – truncate / repeat
                if a_prev.size < n:
                    a_adj = np.repeat(a_prev, int(np.ceil(n / a_prev.size)))[:n]
                else:
                    a_adj = a_prev[:n]
                seeds.append(a_adj / a_adj.sum())
        except Exception:
            pass

    # Ensure all seeds are proper probability vectors (protect against zeros)
    cleaned = []
    eps = 1e-30
    for s in seeds:
        s = np.asarray(s, dtype=np.float64).ravel()
        s = np.clip(s, eps, None)
        s = s / s.sum()
        cleaned.append(s)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # Target length – 8000 is a good compromise between quality and speed
    n = 8000
    print(f"[info] target length = {n}")

    # ------------------------------------------------------------------
    # Phase 1 – short optimisations for a broad set of seeds
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n)
    short_iters = 3000          # enough to get out of the “flat” region
    switch_iter = 1500          # when to switch to hard‑max gradient

    results = []                # holds (best_seq, best_val) for every seed
    global_best_seq = None
    global_best_val = float("inf")

    for idx, seed_a in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"[phase1] seed {idx+1}/{len(seeds)}")
        best_seq, best_val = _optimise_seed(
            seed_a,
            n,
            deadline,
            max_iters=short_iters,
            switch_iter=switch_iter,
        )
        results.append((best_seq, best_val))

        if best_val < global_best_val:
            global_best_val = best_val
            global_best_seq = best_seq.copy()
            print(f"[overall] new global best (phase 1) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – take the most promising seeds and run a longer refinement
    # ------------------------------------------------------------------
    # Sort by the best value attained in phase 1
    results.sort(key=lambda t: t[1])
    top_k = 5
    long_iters = 12000          # further refinement (subject to remaining budget)

    for idx, (seed_a, seed_val) in enumerate(results[:top_k]):
        if time.time() > deadline:
            break
        print(f"[phase2] refining top‑{idx+1} (seed val={seed_val:.6f})")
        best_seq, best_val = _optimise_seed(
            seed_a,
            n,
            deadline,
            max_iters=long_iters,
            switch_iter=switch_iter,
        )
        if best_val < global_best_val:
            global_best_val = best_val
            global_best_seq = best_seq.copy()
            print(f"[overall] new global best (phase 2) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        print("[fallback] using uniform distribution")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Convert to plain Python list (evaluate_sequence expects a list)
    final_seq = global_best_seq.tolist()
    print(f"[final] best objective ≈ {global_best_val:.6f} (raw 2*n*max(conv))")
    return final_seq
