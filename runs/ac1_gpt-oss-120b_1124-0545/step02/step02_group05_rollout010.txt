"""
Hybrid sub‑gradient / simulated‑annealing optimizer.

The objective of ``evaluate_sequence`` for a normalized vector (∑a = √(2·n))
is simply ``max(conv(a,a))``.  The algorithm starts from the best known
construction (the global variable ``height_sequence_1`` if present) and
performs a large number of cheap sub‑gradient steps that directly target
the current maximal convolution entry.  After each step we project back
onto the ℓ₁‑simplex (non‑negative entries with the required sum).  To avoid
getting stuck in a flat region we interleave occasional random
perturbations and a simple Metropolis acceptance rule (simulated annealing).
The whole routine respects a 1000 s budget, keeps the best sequence found,
and returns it as a plain Python list.
"""

import time
import math
import random
import numpy as np

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector v onto the simplex { w ≥ 0 , ∑ w = s }.
    Implements the O(n log n) algorithm from:
    Wang & Carreira‑Perpiñán (2013) “Projection onto the probability simplex”.
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    # make a copy to avoid mutating the input
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.nonzero(rho_arr > 0)[0]
    if len(rho) == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, s / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function height vector that minimises
    `evaluate_sequence`.  The method combines a direct sub‑gradient descent
    on the exact max‑convolution with a temperature‑controlled Metropolis
    acceptance and occasional random re‑initialisations.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise the sequence (use the best known construction if available)
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=float)
    else:
        # fallback – a moderately sized uniform vector
        n = 2500
        a = np.full(n, 1.0, dtype=float)

    n = len(a)
    target_sum = math.sqrt(2.0 * n)

    # Normalise to the required ℓ₁‑norm and ensure non‑negativity
    a = np.maximum(a, 0.0)
    a = a / a.sum() * target_sum

    # ------------------------------------------------------------------
    # 2.  Helper to evaluate the current max‑convolution
    def _max_conv(x: np.ndarray) -> float:
        # using np.convolve (O(n²) but fast for n≈2500)
        return np.convolve(x, x).max()

    cur_max = _max_conv(a)
    best_a = a.copy()
    best_val = cur_max
    print(f"[init] n={n}, initial max(conv) = {cur_max:.6f}")

    # ------------------------------------------------------------------
    # 3.  Optimisation parameters
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin
    it = 0
    # step size schedule – larger at the beginning, decays as 1/√it
    step0 = target_sum * 0.05
    # temperature for Metropolis acceptance – linear cooling
    T_start = cur_max * 0.5
    T_end = cur_max * 0.01
    # after how many iterations we inject a random perturbation
    perturb_interval = 800
    # how long we tolerate stagnation before a restart
    stagnation_limit = 5000
    last_improve_it = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- current convolution and its arg‑max
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        cur_max = float(conv[max_idx])

        # ---- sub‑gradient of the (non‑smooth) max‑convolution
        #   ∂ max(conv) / ∂ a_i = 2 * a_{max_idx - i}  when the index is valid
        grad = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx = np.arange(lo, hi + 1)                     # indices that actually appear
        grad[idx] = 2.0 * a[max_idx - idx]

        # ---- step size (diminishing)
        step = step0 / math.sqrt(it + 1.0)

        # ---- gradient descent step (moving opposite to the sub‑gradient)
        a_candidate = a - step * grad
        a_candidate = _simplex_projection(a_candidate, target_sum)

        # ---- evaluate candidate
        cand_max = _max_conv(a_candidate)

        # ---- Metropolis acceptance (allows occasional uphill moves)
        # temperature linear from T_start to T_end over the whole run
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = T_start * (1.0 - frac) + T_end * frac
        accept = cand_max <= cur_max or (np.random.rand() < math.exp(-(cand_max - cur_max) / max(T, 1e-12)))
        if accept:
            a = a_candidate
            cur_max = cand_max

        # ---- keep the best ever seen
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            last_improve_it = it
            print(f"[{it:6d}] New best  max(conv) = {best_val:.6f} (t={elapsed:.1f}s)")

        # ---- periodic random perturbations to escape flat regions
        if it % perturb_interval == 0 and it > 0:
            # blend the best known solution with fresh exponential noise
            noise = np.random.exponential(scale=1.0, size=n)
            noise = noise / noise.sum() * target_sum
            mix = np.random.rand()
            a_pert = mix * best_a + (1.0 - mix) * noise
            a_pert = _simplex_projection(a_pert, target_sum)

            pert_max = _max_conv(a_pert)
            if pert_max < best_val:
                best_val = pert_max
                best_a = a_pert.copy()
                a = a_pert
                last_improve_it = it
                print(f"[{it:6d}] New best after perturbation: {best_val:.6f}")

            elif pert_max < cur_max:
                a = a_pert
                cur_max = pert_max

        # ---- restart if stagnation persists for a long while
        if it - last_improve_it > stagnation_limit:
            # re‑initialise from the current best plus a burst of noise
            noisy = np.random.exponential(scale=1.0, size=n)
            noisy = noisy / noisy.sum() * target_sum
            a = 0.5 * best_a + 0.5 * noisy
            a = _simplex_projection(a, target_sum)
            cur_max = _max_conv(a)
            last_improve_it = it
            print(f"[{it:6d}] Restarted after stagnation (max(conv)={cur_max:.6f})")

        it += 1

    # ------------------------------------------------------------------
    print(f"[final] Best max(conv) found = {best_val:.6f} after {it} iterations")
    return best_a.tolist()
