def _smooth_max_and_grad(seq, beta):
    conv = np.convolve(seq, seq)
    max_conv = conv.max()
    # weighting
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    w_norm = w / w_sum
    # gradient of softmax max approx
    grad = 2 * np.correlate(w_norm, seq, mode='full')[:len(seq)]
    # smooth objective value approx as max_conv + np.log(w_sum)/beta
    f = max_conv + np.log(w_sum)/beta
    return f, grad

```
def _simplex_projection(v):
    # v: np.array
    # return projection onto simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function sequence minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Candidate families** –  For several lengths `n` we generate
   power‑law sequences   a_i ∝ (i+1+δ)^(-α)   (δ = 0,…,5).  A coarse
   grid of exponents `α` is scanned, followed by a fine grid around the
   best exponent for that length.  The best of all candidates (lowest
   exact score `2·n·max(conv)`) is kept.

2. **Smooth‑max gradient descent** –  Starting from the best candidate we
   minimise the (convex) objective `max(a*a)` using a smoothed
   approximation:
        f(a) = max(conv) + log(∑exp(β·(conv−max))) / β .
   Its gradient is
        ∇f = 2·corr(w, a) ,    w_i ∝ exp(β·(conv_i−max)),
   where `corr` denotes the (full) cross‑correlation.
   An Adam optimiser is applied; after each step the vector is projected
   onto the simplex {x ≥ 0, ∑x = 1}.  The smoothing parameter `β` is
   slowly increased from 5 to 200 so the optimisation moves from a
   smooth landscape to the true max‑objective.

3. **Random mass swaps** –  Every few thousand iterations a tiny random
   exchange of mass between two positions is performed to escape flat
   plateaus.

The loop runs until the time budget (≈ 1000 s) is exhausted.  The best
sequence found (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Generate a non‑negative sequence a_i ∝ (i+1+offset)^{-α} and normalise
    it to have sum equal to 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _score_from_sequence(seq: np.ndarray) -> float:
    """
    Exact score used by ``evaluate_sequence`` for a normalised vector
    (sum(seq) == 1).  The denominator is 1, so
            score = 2 * n * max(conv) .
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)      # length 2n‑1
    return float(2.0 * n * conv.max())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        S = {x ∈ ℝⁿ | x_i ≥ 0, ∑ x_i = 1 } .
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sort decreasing
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_and_grad(seq: np.ndarray, beta: float):
    """
    Smoothed version of max(conv) and its gradient.

    Parameters
    ----------
    seq : np.ndarray
        Non‑negative vector with sum 1 (length n).
    beta : float
        Smoothing parameter (β > 0).  Larger β makes the approximation
        tighter to the true maximum.

    Returns
    -------
    f_val : float
        Smooth approximation to max(conv).
    grad : np.ndarray
        Gradient of the smooth approximation w.r.t. ``seq`` (same shape
        as ``seq``).  The factor 2·n of the final score is omitted – it
        does not influence the optimisation direction.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)                # length 2n‑1
    max_conv = conv.max()
    # softmax weights (shifted by max_conv to avoid overflow)
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    w_norm = w / w_sum

    # gradient of the soft‑max approximation: 2 * correlation(w_norm, seq)
    # np.correlate returns (full) correlation; the first n entries are
    # exactly the needed values.
    grad = 2.0 * np.correlate(w_norm, seq, mode='full')[:n]

    # smooth objective value (useful for monitoring)
    f_val = max_conv + np.log(w_sum) / beta
    return f_val, grad


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The function respects the supplied time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a few seconds margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Optional warm‑start from a previously stored best sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_n = None
    best_alpha = None
    best_offset = 0.0

    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.maximum(cand, 0.0)
                cand_sum = cand.sum()
                if cand_sum > 0.0:
                    cand = cand / cand_sum
                    cur = _score_from_sequence(cand)
                    best_seq = cand.copy()
                    best_score = cur
                    best_n = cand.size
                    print(f"[init] height_sequence_1 length={best_n} score={cur:.6f}")
        except Exception:
            pass   # ignore any problem and continue with fresh search

    # ------------------------------------------------------------------
    # 1. Search over several lengths and power‑law parameters
    # ------------------------------------------------------------------
    # lengths are chosen to balance quality and speed
    lengths = [2000, 4000, 6000, 8000, 12000, 20000]

    for n in lengths:
        if time.time() >= deadline:
            break
        # Coarse exponent scan
        alphas_coarse = np.arange(0.45, 0.66, 0.02)
        for alpha in alphas_coarse:
            if time.time() >= deadline:
                break
            seq = _power_sequence(n, alpha)
            score = _score_from_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_n = n
                best_alpha = alpha
                best_offset = 0.0
                print(f"[coarse] n={n} α={alpha:.4f} score={score:.6f}")

        # Fine tune around the best α found for this length
        if best_n == n and best_alpha is not None:
            low = max(0.4, best_alpha - 0.03)
            high = min(0.8, best_alpha + 0.03)
            alphas_fine = np.linspace(low, high, 81)   # step ≈0.00075
            for alpha in alphas_fine:
                if time.time() >= deadline:
                    break
                seq = _power_sequence(n, alpha)
                score = _score_from_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_alpha = alpha
                    best_offset = 0.0
                    print(f"[fine]   n={n} α={alpha:.5f} score={score:.6f}")

        # Small integer offset search (δ = 0 … 5)
        if best_n == n and best_alpha is not None:
            for delta in range(0, 6):
                if time.time() >= deadline:
                    break
                seq = _power_sequence(n, best_alpha, offset=float(delta))
                score = _score_from_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_offset = float(delta)
                    print(f"[offset] n={n} δ={delta} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. If nothing was found (unlikely) fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_n = 2000
        best_seq = np.full(best_n, 1.0 / best_n, dtype=np.float64)
        best_score = _score_from_sequence(best_seq)
        print(f"[fallback] using uniform length={best_n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine the best candidate with smoothed‑max Adam optimisation
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    n = seq.shape[0]

    # Adam state
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.05          # base learning rate (Adam scales it internally)

    # smoothing parameter schedule
    beta_smooth = 5.0
    beta_max = 200.0
    beta_growth = (beta_max / beta_smooth) ** (1.0 / 150000)  # exponential growth

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # smoothed objective and gradient (ignores the constant factor 2·n)
        _, grad = _smooth_max_and_grad(seq, beta_smooth)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        seq = seq - step

        # projection onto the simplex (ensures sum = 1, non‑negative)
        seq = _simplex_projection(seq)

        # occasional random mass exchange to break plateaus
        if iteration % 5000 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps_swap = min(0.005 * seq[i], seq[i] * 0.5)
            cand = seq.copy()
            cand[i] -= eps_swap
            cand[j] += eps_swap
            cand = _simplex_projection(cand)
            seq = cand

        # evaluate the exact (non‑smoothed) score
        cur_score = _score_from_sequence(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration:,} β={beta_smooth:.1f} score={best_score:.6f}")

        # increase smoothing parameter (approach the true max)
        if beta_smooth < beta_max:
            beta_smooth *= beta_growth
            if beta_smooth > beta_max:
                beta_smooth = beta_max

    print(f"[final] best_score={best_score:.6f} length={best_n} α={best_alpha} δ={best_offset}")
    # Return as a plain Python list as required by the harness
    return best_seq.tolist()
