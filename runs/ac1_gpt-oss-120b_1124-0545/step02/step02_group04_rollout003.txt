"""
Projected gradient descent on a smooth approximation of max(conv(a,a)).

The algorithm works on the ℓ₁‑simplex
    {x ≥ 0,  Σ x = √(2 n)},
because the evaluation function is scale‑invariant:
for Σa = √(2 n) the objective equals max(conv(a,a)).
We replace the non‑smooth max by a soft‑max
   sα(a)= (1/α)·log Σ exp(α·conv_i)
with a temperature α that is slowly increased.
Its gradient can be written analytically as a convolution,
so each iteration only needs a few FFTs.
After a gradient step we project back onto the simplex.
The learning rate is exponentially decayed.
The best vector seen is kept and returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector v onto the simplex {x >= 0, sum(x)=s}.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – occurs only when s == 0
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, nfft: int) -> float:
    """
    Return max entry of the linear convolution a*a using an FFT of length nfft.
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the evaluation metric.
    Uses projected gradient descent on a smooth approximation of the max‑convolution.
    """
    # ------------------------------------------------------------------
    # initialise RNGs
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # pick a starting sequence – use the previously found construction if present
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        n = a.size
    else:
        n = 2500                     # reasonable default size
        a = np.random.exponential(scale=1.0, size=n)

    # target ℓ₁‑norm (the normalisation used in the problem statement)
    target_sum = math.sqrt(2.0 * n)

    # project onto the simplex to enforce feasibility
    a = _simplex_projection(a, target_sum)

    # pre‑compute FFT length for the self‑convolution (size 2 n‑1)
    nfft_conv = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # keep the best solution discovered
    best_seq = a.copy()
    best_score = _max_convolution(a, nfft_conv)

    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # couple of seconds safety margin

    # schedule parameters
    eta_start = target_sum * 0.2      # initial step size (large)
    eta_end   = target_sum * 0.001    # final step size (tiny)
    alpha_start = 0.5                 # start with a smooth surrogate
    alpha_end   = 30.0                # focus more on the true max later

    iteration = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)   # 0 → 1 over the run

        # exponential decay for the learning rate, linear increase for α
        eta   = eta_start * ((eta_end / eta_start) ** frac)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # --------------------------------------------------------------
        # 1) compute the self‑convolution of a (via FFT)
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]

        # --------------------------------------------------------------
        # 2) smooth‑max (soft‑max) and weight vector
        max_c = conv.max()
        # stability: subtract the maximum before exponentiation
        exp_vals = np.exp(alpha * (conv - max_c))
        sum_exp = exp_vals.sum()
        # the smooth approximation (unused for the update, kept for diagnostics)
        smooth_val = max_c + math.log(sum_exp) / alpha    # = soft‑max
        w = exp_vals / sum_exp    # normalised weights, ∑w = 1

        # --------------------------------------------------------------
        # 3) gradient of the smooth‑max w.r.t. a
        #    grad = 2 * (a_rev * w) restricted to the original n entries
        a_rev = a[::-1]                       # reversed a
        len_conv = a_rev.size + w.size - 1    # = 3n‑2
        nfft_grad = 1 << (len_conv.bit_length())
        # FFT‑based convolution a_rev * w
        af_rev = np.fft.rfft(a_rev, n=nfft_grad)
        wf = np.fft.rfft(w, n=nfft_grad)
        conv_rev_full = np.fft.irfft(af_rev * wf, n=nfft_grad)[:len_conv]
        start_idx = a_rev.size - 1           # position of the first valid gradient entry
        grad = 2.0 * conv_rev_full[start_idx:start_idx + n]

        # --------------------------------------------------------------
        # 4) gradient step followed by simplex projection
        a_candidate = a - eta * grad
        a_candidate = _simplex_projection(a_candidate, target_sum)

        # --------------------------------------------------------------
        # 5) evaluate the (true) objective – max convolution entry
        cur_score = _max_convolution(a_candidate, nfft_conv)

        # --------------------------------------------------------------
        # keep the best one seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            print(f"[{iteration:7d}] New best score: {best_score:.6f} (time {elapsed:.1f}s)")

        # accept the candidate (always move – the method is a descent direction)
        a = a_candidate

        # --------------------------------------------------------------
        # occasional random perturbation to escape stagnation
        if iteration % 5000 == 0 and iteration > 0:
            a = best_seq + np.random.normal(scale=target_sum * 0.005, size=n)
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a, target_sum)
            # recompute score after perturbation (but do not overwrite best_seq)
            cur_score = _max_convolution(a, nfft_conv)

        iteration += 1

    # ------------------------------------------------------------------
    # return the best sequence as a plain Python list
    return best_seq.tolist()
