"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm:

1.  Initialise a long sequence (default length 5000) from the
    analytically‑motivated “arcsine” shape (a_i ∝ 1/sqrt((i+½)(n‑i‑½))).
    If the global variable ``height_sequence_1`` exists it is also tried
    as a starting point.

2.  Re‑scale the vector to the probability simplex (non‑negative, sum = 1).

3.  Repeatedly perform *projected gradient descent* on a smooth
    approximation of the max‑convolution.  For a vector a we compute
    the full autocorrelation b = a⋆a (via an FFT).  The smoothed
    maximum is

        s_α(b) = (1/α)·log Σ_k exp(α·b_k)

    whose gradient w.r.t. a is
        ∇_a s_α = 2·(w ⋆ a_rev) ,

    where w_k = exp(α·b_k)/Σ_j exp(α·b_j) and a_rev is a reversed.
    The true objective (with Σa = 1) is f(a)=2 n·s_α(b); we thus use
    ∇f = 2 n·∇s_α.

    The smoothing parameter α is increased from 1 to a few hundred
    (exponential schedule) – a *continuation* method that first
    moves quickly on a smooth landscape and later focuses on the
    true L∞‑peak.

4.  Each gradient step is projected back onto the simplex by simple
    clipping and renormalisation (very cheap).  A tiny probability
    (≈2 %) of a random “mass‑swap’’ is added to escape shallow
    minima.  An annealed Metropolis acceptance rule allows occasional
    uphill moves early on.

5.  The best sequence seen during the allotted wall‑time (default
    1000 s) is returned.

All heavy convolutions are performed with FFTs (O(n log n)), making the
search feasible even for n≈5000 and thousands of iterations.
"""

import time
import numpy as np

# Simple utilities
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Clip to non‑negative entries and renormalise to sum = 1.
    If the vector becomes all zeros (numerical accident) we fall back to
    the uniform distribution.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    else:
        # extremely unlikely, but keep something feasible
        return np.full_like(v, 1.0 / v.shape[0])


def compute_score_fft(a: np.ndarray, fft_len: int) -> float:
    """
    Fast version of the evaluation metric using FFT.
    Assumes a is already non‑negative with sum≈1.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_b = conv_full.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def arcsine_candidate(n: int) -> np.ndarray:
    """
    Classic arcsine (inverse‑sqrt product) shape:
        a_i ∝ 1 / sqrt((i+0.5)·(n-i-0.5))
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a = a / a.sum()
    return a


def random_mass_swap(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a small random amount of mass from one index to another.
    Used as a cheap exploratory move.
    """
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_onto_simplex(a_new)

        sc = compute_score_fft(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
        print(f"[init] candidate {idx:02d} score = {sc:.6f}")

    a = best_seq.copy()
    curr_score = best_score
    print(f"[init] chosen start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    last_improve_time = start_time

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 0.5)   # in [0,1]

        # --- schedules -------------------------------------------------
        # exponential continuation for α
        alpha = alpha_start * ((alpha_end / alpha_start) ** progress)
        # linear annealing for step magnitude
        step_scale = step_start * (1.0 - progress) + step_end * progress

        # --- compute current autocorrelation ----------------------------
        fa = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_b = conv_full.max()

        # --- smooth max weights -----------------------------------------
        shift = conv_full - max_b                # max becomes zero
        # avoid overflow/underflow
        exp_arg = np.clip(alpha * shift, -700, 700)
        w_raw = np.exp(exp_arg)
        w = w_raw / w_raw.sum()                 # soft‑max weights, sum = 1

        # --- gradient of the smoothed objective ------------------------
        # g_i = 2 * Σ_j w_{i+j} a_j   (i = 0…n‑1)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, fft_len)
        far = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(fw * far, fft_len)
        # extract the relevant slice (indices n‑1 … n‑1+n‑1)
        g_slice = grad_full[n - 1 : n - 1 + n]   # Σ_j w_{i+j} a_j
        grad = 4.0 * n * g_slice                 # ∇f = 2 n·2·Σ… = 4 n·Σ…

        # Normalise the gradient direction to have unit Euclidean norm.
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            direction = grad / grad_norm
        else:
            direction = np.zeros_like(a)

        # --- propose a new point ----------------------------------------
        if np.random.rand() < 0.02:                     # occasional random swap
            cand = random_mass_swap(a, step_scale * 0.5)
        else:
            cand = a - step_scale * direction
            cand = project_onto_simplex(cand)

        # --- evaluate candidate (fast) ----------------------------------
        cand_score = compute_score_fft(cand, fft_len)

        # Metropolis‑style acceptance (temperature mirrors step size)
        temp = max(step_scale, 1e-12)   # temperature proportional to step magnitude
        accept = False
        if cand_score < curr_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - curr_score) / temp)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            curr_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve_time = time.time()
                print(
                    f"[{iteration:07d}] new best = {best_score:.6f} (t={elapsed:.1f}s)"
                )

        # occasional restart if stuck for >30 s without improvement
        if (iteration % 20000 == 0) and (time.time() - last_improve_time > 30.0):
            jitter = np.random.rand(n) * 1e-4
            a = project_onto_simplex(best_seq + jitter)
            print(f"[{iteration:07d}] restart after stagnation")

        iteration += 1

    total_time = time.time() - start_time
    print(
        f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s"
    )
    return best_seq.tolist()
