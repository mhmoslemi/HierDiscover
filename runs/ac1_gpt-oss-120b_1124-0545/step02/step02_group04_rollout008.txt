"""
propose_candidate
=================

Search for a non‑negative step function that minimizes

    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a hybrid of simulated‑annealing and a deterministic
projected gradient descent on the current maximal convolution entry.
It repeatedly:

* computes the index *k* where the self‑convolution attains its maximum,
* forms the sub‑gradient g_i = a_{k-i} (zero if the partner index is out of range),
* tries a back‑tracked gradient step a ← Π_S (a – α·g) (Π_S = simplex projection),
* occasionally performs cheap random mass‑transfer or “peak‑reduction” moves,
* accepts every improving move and otherwise accepts with a Metropolis probability
  that depends on an exponential cooling schedule.

The search is performed many times with random restarts until the allotted
time budget (default 1000 s) is exhausted.  The best sequence seen so far is
returned as a plain ``list[float]``.  All helper functions are top‑level and
contain no closures or lambdas, complying with the competition constraints.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector ``v`` onto the simplex {x ≥ 0, sum(x) = s}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – should happen only when s == 0
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution_and_index(a: np.ndarray, nfft: int) -> tuple[float, int]:
    """
    Return the maximal entry of the linear convolution a*a and the index where
    it occurs, using an FFT of length ``nfft`` (the next power of two ≥ 2·len(a)-1).
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    max_idx = int(np.argmax(conv))
    return float(conv[max_idx]), max_idx


# ----------------------------------------------------------------------
def _peak_reduction_move(a: np.ndarray, max_idx: int,
                         target_sum: float, temperature: float) -> np.ndarray:
    """
    Transfer a small amount of mass from a “source” coordinate that heavily
    contributes to the current maximal convolution entry to a “destination”
    coordinate that contributes little.  The move is followed by a simplex
    projection to restore feasibility.
    """
    n = a.size
    idx = np.arange(n)
    partner = max_idx - idx
    valid = (partner >= 0) & (partner < n)

    # contribution of each coordinate to the maximal entry
    contrib = np.zeros_like(a)
    contrib[valid] = a[idx[valid]] * a[partner[valid]]

    # source: among the top contributors
    top_k = 8
    src_candidates = np.argpartition(-contrib, top_k)[:top_k]
    src = src_candidates[np.argmax(contrib[src_candidates])]

    # destination: preferably a coordinate that does not appear in the sum pair
    noncontrib = np.where(~valid)[0]
    if noncontrib.size > 0:
        dst = int(np.random.choice(noncontrib))
    else:
        low_k = 8
        dst_candidates = np.argpartition(contrib, low_k)[:low_k]
        dst = int(np.random.choice(dst_candidates))

    max_delta = min(a[src], target_sum * 0.02)
    delta = max_delta * temperature * random.random()
    if delta < 1e-12:
        return a.copy()

    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = _simplex_projection(a_new, target_sum)
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Hybrid simulated‑annealing / projected‑gradient search for a step‑function
    that minimises the evaluation function.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence found.
    """
    # ------------------------------------------------------------------
    # Initialise RNGs
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size ----------------------------------------------------
    # Use the length of the known construction if available, otherwise a
    # moderate default.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2048                      # power of two – convenient for FFT
        a0 = np.random.exponential(scale=1.0, size=n)

    # Normalise to the conventional ℓ₁‑norm √(2 n); the objective is invariant
    # under positive scaling, but keeping this normalisation makes the
    # temperature schedule independent of the chosen n.
    target_sum = math.sqrt(2.0 * n)
    a0 = a0 / a0.sum() * target_sum

    # FFT length (next power of two that can hold a full linear convolution)
    nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Global best bookkeeping
    best_seq = a0.copy()
    best_M, _ = _max_convolution_and_index(best_seq, nfft)

    # ------------------------------------------------------------------
    start_time = time.time()
    # Keep a small safety margin for the external harness
    deadline = start_time + budget_s - 5.0
    # Annealing parameters (global schedule)
    T_start = 1.0
    T_end = 1e-5

    # ------------------------------------------------------------------
    # Helper that runs a local optimisation starting from ``a_start``
    def _local_search(a_start: np.ndarray, time_limit: float) -> None:
        nonlocal best_seq, best_M, target_sum, nfft, deadline

        a = a_start.copy()
        cur_M, cur_idx = _max_convolution_and_index(a, nfft)

        # Simple per‑run schedule – based on elapsed time of the whole run
        while time.time() < time_limit:
            # ----------------------------------------------------------------
            # Temperature (exponential decay over the whole remaining budget)
            elapsed = time.time() - start_time
            frac = elapsed / (deadline - start_time)
            temperature = T_start * ((T_end / T_start) ** frac)

            # ----------------------------------------------------------------
            # Choose a move type
            move_type = random.random()
            a_prop = a.copy()

            if move_type < 0.30:
                # 1) Random transfer of a small amount of mass i → j
                i, j = np.random.randint(n, size=2)
                if i != j:
                    max_delta = min(a_prop[i], target_sum * 0.04)
                    delta = max_delta * temperature * random.random()
                    if delta > 1e-12:
                        a_prop[i] -= delta
                        a_prop[j] += delta

            elif move_type < 0.55:
                # 2) Simple swap of two coordinates (permutes the multiset)
                i, j = np.random.randint(n, size=2)
                if i != j:
                    a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

            elif move_type < 0.80:
                # 3) Peak‑reduction move targeting the current maximal entry
                a_prop = _peak_reduction_move(a, cur_idx, target_sum, temperature)

            else:
                # 4) Projected sub‑gradient step on the maximal entry
                #    grad_i = a_{cur_idx - i} for valid partner indices
                partner = cur_idx - np.arange(n)
                mask = (partner >= 0) & (partner < n)
                grad = np.zeros_like(a)
                grad[mask] = a[partner[mask]]
                # Back‑tracking line search to guarantee descent
                alpha = target_sum * 0.30 * temperature
                for _ in range(5):   # at most five halvings
                    a_candidate = a - alpha * grad
                    a_candidate = _simplex_projection(a_candidate, target_sum)
                    new_M, _ = _max_convolution_and_index(a_candidate, nfft)
                    if new_M <= cur_M:
                        a_prop = a_candidate
                        cur_M = new_M
                        break
                    alpha *= 0.5
                else:
                    # line search failed – fall back to a tiny random move
                    i, j = np.random.randint(n, size=2)
                    if i != j:
                        delta = min(a_prop[i], target_sum * 0.001) * random.random()
                        a_prop[i] -= delta
                        a_prop[j] += delta

            # ----------------------------------------------------------------
            # Project if we have not already projected
            if not np.isclose(np.sum(a_prop), target_sum):
                a_prop = _simplex_projection(a_prop, target_sum)

            # ----------------------------------------------------------------
            # Evaluate
            new_M, new_idx = _max_convolution_and_index(a_prop, nfft)

            # ----------------------------------------------------------------
            # Acceptance rule (Metropolis)
            accept = False
            if new_M < cur_M:
                accept = True
            else:
                prob = math.exp(-(new_M - cur_M) / (temperature + 1e-12))
                if random.random() < prob:
                    accept = True

            if accept:
                a = a_prop
                cur_M = new_M
                cur_idx = new_idx
                # Update global best
                if cur_M < best_M:
                    best_M = cur_M
                    best_seq = a.copy()
                    elapsed_total = time.time() - start_time
                    print(f"[{elapsed_total:7.1f}s] New global best M = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – multiple random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Allocate a slice of the remaining budget to a fresh local search.
        # The slice shrinks as we approach the deadline to guarantee termination.
        remaining = deadline - time.time()
        # At least a few seconds per restart; otherwise we break.
        if remaining < 8.0:
            break
        slice_budget = min(30.0, remaining * 0.6)   # keep some time for later restarts

        # Initialise a fresh random point (exponential distribution works well)
        a_start = np.random.exponential(scale=1.0, size=n)
        a_start = a_start / a_start.sum() * target_sum

        _local_search(a_start, time.time() + slice_budget)

        # Occasionally inject the best‑so‑far vector with a tiny perturbation to
        # explore its neighbourhood more thoroughly.
        if random.random() < 0.25:
            pert = best_seq + np.random.normal(scale=target_sum * 0.0008, size=n)
            pert = np.maximum(pert, 0.0)
            pert = _simplex_projection(pert, target_sum)
            # Run a short focused search around the perturbed point.
            _local_search(pert, time.time() + min(12.0, deadline - time.time()))

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    return best_seq.tolist()
