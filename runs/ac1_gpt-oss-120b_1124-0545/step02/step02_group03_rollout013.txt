def propose_candidate(seed=..., budget_s=..., **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = int(kwargs.get('n', 6000))

    def init_arcsine():
        # compute arcsine shape via Beta(0.5,0.5)
        ix = np.arange(n) + 0.5
        x = ix / n
        a = 1.0 / np.sqrt(x * (1 - x))
        a = a / a.sum()
        return a

    try:
        seq0 = globals().get('height_sequence_1', None)
        if seq0 is not None:
            seq0 = np.asarray(seq0, dtype=float)
            if seq0.size != n:
                # interpolate
                pass
            else:
                a0 = np.clip(seq0, 0, None)
                if a0.sum() == 0:
                    a0 = init_arcsine()
                else:
                    a0 = a0 / a0.sum()
                a = a0
        else:
            raise KeyError
    except Exception:
        a = init_arcsine() # start with arcsine shape

    # ensure a positive
    a = np.clip(a, 1e-15, None)
    a = a / a.sum()
    x = np.log(a)  # latent logits

    # FFT sizes
    size1 = _next_pow2(2*n - 1)
    size2 = _next_pow2(3*n - 2)

    # Adam
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.5

    # tolerance for mask
    tolerance_factor = 1e-8

    # restart parameters
    stagnation_limit = 30.0
    last_improve = start

    best_seq = None
    best_score = np.inf

    it = 0

    while time.time() < deadline:
        it += 1

        # compute probabilities a
        # Use stable softmax of x:
        x_shift = np.max(x)
        exp_x = np.exp(x - x_shift)
        a = exp_x / np.sum(exp_x)

        # compute convolution
        conv_full = _conv_fft(a, a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # stagnation restart
        if time.time() - last_improve > stagnation_limit:
            # restart to arcsine shape
            a = init_arcsine()
            x = np.log(a)
            m = np.zeros(n)
            v = np.zeros(n)
            last_improve = time.time()
            print(f"[{it:7d}] restart due to stagnation")
            continue

        # compute mask for active max entries
        tol = max(tolerance_factor * max_b, 1e-12)
        mask = (conv_full >= max_b - tol).astype(float)

        # compute gradient wrt a (descent direction)
        a_rev = a[::-1]
        grad_a = 2.0 * _conv_fft(a_rev, mask, size2)[n - 1 : n - 1 + n]  # positive gradient (to increase loss)
        # We need gradient of loss: it's grad_a (since loss = max_b).
        # Now compute gradient wrt x (since a = softmax(x)).
        # For loss L(a), dL/dx = a * (dL/da - <a, dL/da>)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = base_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random perturbation
        if np.random.rand() < 0.03:
            # random transfer and update x accordingly
            a = _random_transfer(a, 0.05)
            x = np.log(a)

    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    if best_seq is None:
        # fallback, return uniform
        best_seq = np.ones(n) / n
    return best_seq.tolist()

"""
Propose a low‑value step‑function.
-------------------------------------------------
Algorithm
---------
* Initialise the probability vector on the simplex with the
  Beta(½,½) (“arcsine”) shape – this shape is known to give a
  fairly flat self‑convolution.
* Parameterise the vector as a soft‑max of unrestricted logits
  ``x`` (a = softmax(x)).  This guarantees non‑negativity and the
  sum‑to‑one constraint automatically.
* At each iteration:
    1. Compute the linear convolution ``b = a * a`` by FFT.
    2. Identify the (almost) active indices where ``b`` attains its
       maximum (a tolerance‑scaled mask).
    3. The sub‑gradient of the objective ``max(b)`` w.r.t. ``a`` is
       ``g = 2·conv_rev(mask)`` (the same expression used in the
       reference implementation).
    4. Convert the gradient to the logit space:  gradₓ = a ⊙ (g − ⟨a,g⟩).
    5. Perform an Adam update on the logits ``x`` (step size decays
       like 1/√iteration).  This provides momentum and adaptive
       scaling, which speeds up convergence compared with plain
       sub‑gradient descent.
    6. With a small probability apply a random “mass‑transfer”
       move to escape flat plateaus; after such a move the logits
       are re‑initialised from the perturbed vector.
    7. If no improvement has been seen for a while, restart from the
       arcsine distribution (and reset the Adam moments).
* Keep track of the best sequence seen (according to the exact
  ``evaluate_sequence`` objective, which in our normalised setting
  equals ``2·n·max(b)``) and return it when the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT (real input, real output)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Move a small amount of probability mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    if a[i] == 0.0:
        return a
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def _arcsine_initial(n: int) -> np.ndarray:
    """Beta(½,½) density on a uniform grid, normalised to sum = 1."""
    # mid‑points of the intervals
    xs = (np.arange(n) + 0.5) / n
    # avoid division by zero – xs∈(0,1)
    a = 1.0 / np.sqrt(xs * (1.0 - xs))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low value of `evaluate_sequence`.
    Uses an Adam‑optimised sub‑gradient method on the max‑autocorrelation
    functional, starting from the analytic arcsine distribution.
    """
    np.random.seed(seed)
    start = time.time()
    dead = start + budget_s - 0.5                # tiny safety margin

    # --------------------------------------------------------------
    # problem size (can be overridden by the harness)
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # initialise a distribution (logits) on the simplex
    # --------------------------------------------------------------
    try:
        # If a previous construction is present we upscale it; otherwise use arcsine.
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            seq0 = np.asarray(seq0, dtype=float)
            if seq0.size != n:
                # linear interpolation onto the new grid
                x_old = np.linspace(0, 1, seq0.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, seq0)
                a = np.clip(a, 0.0, None)
                if a.sum() == 0.0:
                    a = _arcsine_initial(n)
                else:
                    a = a / a.sum()
            else:
                a = np.clip(seq0, 0.0, None)
                if a.sum() == 0.0:
                    a = _arcsine_initial(n)
                else:
                    a = a / a.sum()
        else:
            raise KeyError
    except Exception:
        a = _arcsine_initial(n)

    # latent logits – soft‑max of these yields the probability vector
    # add a tiny epsilon to avoid log(0)
    eps = 1e-15
    a = np.clip(a, eps, None)
    a = a / a.sum()
    x = np.log(a)

    # --------------------------------------------------------------
    # pre‑compute FFT lengths (powers of two) – they stay constant
    # --------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a * a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * mask

    # --------------------------------------------------------------
    # Adam state
    # --------------------------------------------------------------
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-12
    base_lr = 0.5                     # base learning rate (decayed later)

    # --------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------
    best_score = np.inf
    best_seq = None
    it = 0
    last_improve = start
    stagnation_sec = 30.0            # seconds without improvement → restart
    rand_prob = 0.04                 # probability of a random mass‑transfer move
    rand_scale = 0.06                # size of the random transfer
    tolerance_factor = 1e-8          # for building the active mask

    # ------------------------------------------------------------------
    while time.time() < dead:
        it += 1

        # ----- 1. current probability vector (soft‑max) -----
        x_shift = np.max(x)
        exp_x = np.exp(x - x_shift)
        a = exp_x / exp_x.sum()

        # ----- 2. convolution and current objective -----
        conv_full = _conv_fft(a, a, size1)          # length 2n‑1
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b                # sum(a)=1 ⇒ objective = 2·n·max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ----- 3. stagnation handling -----
        if time.time() - last_improve > stagnation_sec:
            # restart from the analytic arcsine shape
            a = _arcsine_initial(n)
            a = np.clip(a, eps, None)
            a = a / a.sum()
            x = np.log(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{it:7d}] restart due to stagnation")
            continue

        # ----- 4. occasional random mass transfer (exploration) -----
        if np.random.rand() < rand_prob:
            a = _random_transfer(a, rand_scale)
            # re‑initialise logits after the move
            a = np.clip(a, eps, None)
            a = a / a.sum()
            x = np.log(a)
            # reset Adam moments to avoid stale directions
            m.fill(0.0)
            v.fill(0.0)
            continue

        # ----- 5. sub‑gradient of the max‑convolution -----
        tol = max(tolerance_factor * max_b, 1e-12)
        mask = (conv_full >= max_b - tol).astype(float)   # shape (2n‑1)

        # gradient w.r.t. a (positive direction of increase)
        a_rev = a[::-1]
        grad_a = 2.0 * _conv_fft(a_rev, mask, size2)[n - 1 : n - 1 + n]

        # ----- 6. chain rule to logits x -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)        # ∂L/∂x  (L = max_b)

        # ----- 7. Adam step on the logits -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / (np.sqrt(it) + 1.0)         # decay ∝ 1/√iteration
        x = x - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.ones(n) / n
    return best_seq.tolist()
