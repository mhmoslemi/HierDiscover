"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The code builds a fast optimisation loop that works on the
scale‑invariant simplex (∑a_i = 1).  Three ingredients are combined:

1. **Analytic start** – a short pool of parametric families
   (including the classic arcsine shape) is scanned and the best
   candidate is taken as the initial point.

2. **Projected‑simplex stochastic descent** – the current maximal
   convolution entry (and a few neighbours) determines a sub‑gradient.
   A step is taken, the vector is projected back onto the probability
   simplex and Metropolis acceptance with a linearly annealed temperature
   is used.  In addition to pure gradient steps we occasionally
   perform tiny random mass transfers and a low‑pass smoothing move,
   which help the search escape shallow local minima.

3. **Optional LP boost** – when SciPy is available, every few
   hundred‑thousand iterations a small linear programme is solved:
   maximise  ∑g_i  subject to  a * g ≤ M  (pointwise) and g ≥ 0,
   where M = max(a*a).  The LP solution is renormalised and mixed
   into the current iterate, guaranteeing a non‑increasing objective.

All operations are implemented with FFT‑based convolution
(`O(n log n)`) and a fast Euclidean projection onto the simplex,
so the loop can execute many‑hundreds of thousands of iterations
within the 1000 s budget.

The function returns the best sequence found.
"""

import time
import numpy as np

# Optional SciPy imports (LP part)
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _HAVE_SCIPY = True
except Exception:  # pragma: no cover
    _HAVE_SCIPY = False


# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback – uniform distribution
    return np.ones_like(v) / v.shape[0]


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear (zero‑padded) convolution a*a using a power‑of‑2 FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # eliminate tiny negative artefacts from floating point noise
    return np.maximum(conv, 0.0)


def _score(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies ∑a = 1."""
    # 2 * n * max(conv(a,a))  (the denominator is 1)
    max_b = _conv_fft(a).max()
    return float(2.0 * a.shape[0] * max_b)


def _random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return _project_simplex(a_new)


def _smoothing_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Low‑pass (Gaussian‑like) smoothing move."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    smooth = np.convolve(a, kernel, mode="same")
    a_new = (1.0 - scale) * a + scale * smooth
    return _project_simplex(a_new)


def _gradient_move_multi(a: np.ndarray, step: float, eps: float = 1e-3) -> np.ndarray:
    """
    Projected sub‑gradient descent step.
    The sub‑gradient is averaged over all convolution entries whose
    value is within (1‑eps) of the current maximum.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_b = conv.max()
    thresh = max_b * (1.0 - eps)
    ks = np.where(conv >= thresh)[0]
    if ks.size == 0:                     # safety fallback
        ks = np.array([int(np.argmax(conv))])

    g = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution 2 * a_{k-i}
        g[idx] += 2.0 * a[k - idx]
    g /= ks.size

    a_new = a - step * g
    return _project_simplex(a_new)


# ----------------------------------------------------------------------
# Linear‑programming helper (optional, used when SciPy is present)
# ----------------------------------------------------------------------


def _solve_lp_active(a: np.ndarray, M: float, active_ks: np.ndarray):
    """
    Solve the LP:
        maximise   ∑ g_i
        subject to a * g ≤ M  (only for rows in active_ks)
                  g_i ≥ 0
    Returns the optimal g vector or None on failure.
    """
    n = a.shape[0]
    m = active_ks.size
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_ks):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # coefficients are a_{k-i}
        for i in range(lo, hi + 1):
            rows.append(row_idx)
            cols.append(i)
            data.append(a[k - i])

    A = _sp.csr_matrix((data, (rows, cols)), shape=(m, n))
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -∑g_i  → maximise ∑g_i
    bounds = [(0, None)] * n

    res = _opt.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    return None

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length ``n`` (default 2000)
    that minimises the evaluation function.  The routine respects the
    time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    n = int(kwargs.get("n", 2000))

    # --------------------------------------------------------------
    # 1 – initialise from a small analytic pool (including possible
    #     seed from the global variable ``height_sequence_1``)
    # --------------------------------------------------------------
    candidates = []

    # use global seed if available and of matching length
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)) and len(global_seq) == n:
        arr = np.asarray(global_seq, dtype=np.float64)
        arr = _project_simplex(arr)          # ensure proper normalisation
        candidates.append(arr)

    # uniform
    candidates.append(np.ones(n, dtype=np.float64) / n)

    # classic arcsine‑like product (p = 0.5, δ = 0.5)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 1 - i + 0.5))
    candidates.append(a / a.sum())

    # product family  a_i ∝ ((i+δ)*(n‑1‑i+δ))^{‑p}
    for p in np.linspace(0.35, 1.15, 10):
        δ = 0.5
        a = ((i + δ) * (n - 1 - i + δ)) ** (-p)
        candidates.append(a / a.sum())

    # sum family  a_i ∝ (i+δ)^{‑p} + (n‑1‑i+δ)^{‑p}
    for p in np.linspace(0.35, 1.15, 10):
        δ = 0.5
        a = (i + δ) ** (-p) + (n - 1 - i + δ) ** (-p)
        candidates.append(a / a.sum())

    best_seq = None
    best_val = np.inf
    for idx, cand in enumerate(candidates):
        val = _score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        print(f"[init] candidate {idx:02d} score = {val:.6f}")

    a = best_seq.copy()
    a_val = best_val
    print(f"[init] chosen start score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2 – main optimisation loop (projected stochastic descent,
    #     occasional smoothing, random swaps and optional LP boost)
    # --------------------------------------------------------------
    # linear schedules for temperature and step size
    init_temp = 2.0
    final_temp = 0.001
    init_step = 0.7
    final_step = 1e-5

    iter_cnt = 0
    last_improve = start_time

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 10.0)           # [0,1] (smooth)

        # linear annealing
        temp = init_temp * (1.0 - progress) + final_temp * progress
        step = init_step * (1.0 - progress) + final_step * progress

        # ----------------------------------------------------------
        # pick a move type
        # ----------------------------------------------------------
        r = np.random.rand()
        if r < 0.70:
            cand = _gradient_move_multi(a, step, eps=1e-3)
        elif r < 0.90:
            cand = _random_move(a, step * 0.5)
        else:
            cand = _smoothing_move(a, step * 0.5)

        cand_val = _score(cand)

        # ----------------------------------------------------------
        # Metropolis acceptance (using the current state a_val)
        # ----------------------------------------------------------
        accept = False
        if cand_val < a_val:
            accept = True
        else:
            prob = np.exp(-(cand_val - a_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            a_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iter_cnt:07d}] new best = {best_val:.6f} (t={elapsed:.1f}s)")

        # ----------------------------------------------------------
        # optional LP boost – every 200 k iterations (if SciPy present)
        # ----------------------------------------------------------
        if _HAVE_SCIPY and (iter_cnt % 200000 == 0) and iter_cnt > 0:
            # recompute convolution and active constraints
            full_conv = _conv_fft(a)
            M = full_conv.max()
            # take all indices whose value is within 0.5 % of the max
            active = np.where(full_conv >= M * (1 - 0.005))[0]
            if active.size > 0:
                g_vec = _solve_lp_active(a, M, active)
                if g_vec is not None and g_vec.sum() > 0:
                    g_norm = g_vec / g_vec.sum()
                    mix_t = 0.12                     # small mixing factor
                    a = (1.0 - mix_t) * a + mix_t * g_norm
                    a = _project_simplex(a)
                    a_val = _score(a)
                    if a_val < best_val:
                        best_val = a_val
                        best_seq = a.copy()
                        last_improve = time.time()
                        print(f"[{iter_cnt:07d}] LP mix → best = {best_val:.6f}")

        # ----------------------------------------------------------
        # simple stagnation restart
        # ----------------------------------------------------------
        if iter_cnt % 50000 == 0 and (time.time() - last_improve) > 30.0:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = _project_simplex(a)
            a_val = _score(a)
            print(f"[{iter_cnt:07d}] restart after stagnation (t={elapsed:.1f}s)")

        iter_cnt += 1

    total = time.time() - start_time
    print(f"[done] iterations={iter_cnt}, best_score={best_val:.6f}, time={total:.1f}s")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
