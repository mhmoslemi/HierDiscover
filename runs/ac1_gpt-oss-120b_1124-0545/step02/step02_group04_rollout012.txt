"""
Propose a step‑function (non‑negative vector) that minimises
    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm works in two stages:

1.  A smooth approximation of the “max‑convolution’’ is optimised by
    projected gradient descent.
    •  The objective  f(a)= (1/τ)·log Σ exp(τ·conv_k(a))  is an
       increasingly accurate surrogate of  max(conv)  as τ grows.
    •  Its gradient can be written in closed form and is evaluated
       efficiently with two FFTs (O(n log n)).
    •  After each gradient step the iterate is projected onto the
       probability simplex {x≥0, Σx = √(2n)}.
    •  τ is increased exponentially from a small value (τ≈0.5) to a
       moderate value (τ≈20) – a homotopy that first smooths the landscape
       and then focuses on the true maximum.

2.  When the time budget is almost exhausted a short simulated‑annealing
    fine‑tuner (random mass transfers and “peak‑reduction’’ moves) is run
    for a few seconds to escape possible plateaus.

The routine respects the 1000 s limit, prints a new best value whenever
the bound improves and finally returns the best sequence found as a
plain list of floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector v onto the simplex {x ≥ 0, Σx = s}.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – should only happen when s = 0
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def _max_convolution(a: np.ndarray) -> float:
    """Return max entry of the linear convolution a*a (FFT‑based)."""
    n = a.size
    nfft = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
    return float(conv.max())


def _gradient_logmax(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of   f(a)= (1/τ)·log Σ exp(τ·conv_k(a))
    with respect to a, assuming a ≥ 0.
    Returns a vector of length n.
    """
    n = a.size
    # FFT size for convolutions with the softmax weights (length 2n‑1)
    nfft = _next_pow2(3 * n - 2)          # for a * w_rev (size 3n‑2)

    # ----- convolution a*a (to get conv) -----
    A = np.fft.rfft(a, n=nfft)
    conv_full = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]

    # ----- softmax weights -----
    max_c = np.max(conv_full)
    # scale with τ, subtract max for numerical stability
    exp_vals = np.exp(tau * (conv_full - max_c))
    w = exp_vals / exp_vals.sum()                     # length 2n‑1, Σ w = 1

    # ----- gradient computation -----
    # The derivative of each conv entry w.r.t. a_i is
    #   2·a_{k‑i}  (except when 2i = k, where the factor is 1).
    # Hence   grad = 2·sum_k w_k·a_{k‑i} – a_i·w_{2i}.
    w_rev = w[::-1]                                   # reversed weights
    W_rev = np.fft.rfft(w_rev, n=nfft)
    conv_rev = np.fft.irfft(A * W_rev, n=nfft)[: 3 * n - 2]

    # g_raw[i] = Σ_k a_k·w_{i+k}  = conv_rev[2n‑2‑i]   (i = 0..n‑1)
    start = 2 * n - 2
    g_raw = conv_rev[start : start - n : -1]          # length n

    # diagonal correction (indices where 2i is a valid conv index)
    diag_idx = 2 * np.arange(n)
    diag = a * w[diag_idx]                            # length n

    grad = 2.0 * g_raw - diag
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  Returns the best sequence (list of floats) found within the
    allotted time budget.
    """
    # ------------------------------------------------------------------
    # 1) Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # Decide on problem size.  If a previous construction is present we
    # reuse its length as a convenient starting point; otherwise we pick
    # a modest size that still leaves enough degrees of freedom.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        # use the existing vector as a warm‑start (scaled later)
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)            # safety – non‑negative
    else:
        n = 3000                          # default length
        a = np.random.exponential(scale=1.0, size=n)

    target_sum = math.sqrt(2.0 * n)      # ℓ₁‑norm we must maintain

    # Project onto the simplex so that Σa = √(2n) from the very beginning.
    a = _simplex_projection(a, target_sum)

    # ------------------------------------------------------------------
    # 2) Parameters for the homotopy / gradient descent
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # small safety margin

    # τ controls the smoothness of the log‑sum‑exp surrogate.
    tau = 0.5
    tau_max = 20.0

    # Base step size (absolute, later normalised by gradient norm)
    step_scale = target_sum * 0.5

    # Keep track of the best solution seen so far.
    best_a = a.copy()
    best_score = _max_convolution(best_a)
    cur_score = best_score

    # Counters and timing helpers
    it = 0
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Main optimisation loop (smooth gradient descent)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- gradient of smoothed max -----
        grad = _gradient_logmax(a, tau)

        # Adaptive step size: inverse to gradient norm.
        grad_norm = np.linalg.norm(grad) + 1e-12
        lr = step_scale / grad_norm

        # ----- tentative update and projection -----
        a_new = a - lr * grad
        a_new = _simplex_projection(a_new, target_sum)

            # Temperature decays linearly with remaining time.

            if random.random() < prob:

                cur_score = new_score

        # ----- update τ (make the surrogate sharper) -----

        # ----- occasional random perturbation if we are stuck -----
        if it % 50000 == 0 and it > 0:

                a = np.maximum(a, 0.0)
                a = _simplex_projection(a, target_sum)
                cur_score = _max_convolution(a)

        it += 1

    # ------------------------------------------------------------------
    # 4) Short simulated‑annealing refinement (optional, uses <≈5 s)
    # ------------------------------------------------------------------
    # This phase re‑uses the elementary moves from the original code.
    # It can shave a few hundredths off the bound if time permits.
    refine_deadline = time.time() + min(5.0, max(0.0, deadline - time.time()))
    nfft_sa = _next_pow2(2 * n - 1)

    while time.time() < refine_deadline:
        # Choose a simple move at random
        move = random.random()
        a_prop = best_a.copy()

        if move < 0.35:

            i, j = np.random.randint(n, size=2)
            if i != j:
                max_delta = min(a_prop[i], target_sum * 0.04)
                delta = max_delta * random.random()
                a_prop[i] -= delta
                a_prop[j] += delta
        elif move < 0.55:
            # swap two entries

            if i != j:
                a_prop[i], a_prop[j] = a_prop[j], a_prop[i]
        elif move < 0.80:
            # peak‑reduction move (target current maximal conv entry)
            conv = np.convolve(best_a, best_a)
            max_idx = int(np.argmax(conv))
            # determine contributions for the peak
            idx = np.arange(n)
            partner = max_idx - idx
            valid = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(best_a)
            contrib[valid] = best_a[idx[valid]] * best_a[partner[valid]]
            src_candidates = np.argpartition(-contrib, 10)[:10]

            delta = max_delta * random.random()
            a_prop[src] -= delta

        else:

            conv = np.convolve(best_a, best_a)

            idx = np.arange(n)
            partner = max_idx - idx

        # Project back to the simplex and evaluate

        cand_score = _max_convolution(a_prop)

        if cand_score < best_score:

            best_a = a_prop.copy()
            elapsed = time.time() - start_time
            print(f"[refine] New best: {best_score:.6f} (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # 5) Return the best sequence found.
    # ------------------------------------------------------------------
    return best_a.tolist()
