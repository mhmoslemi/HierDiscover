"""
A fast alternating‑LP+gradient search for the min‑max self‑convolution.
---------------------------------------------------------------------

The algorithm works in three stages:

1. **Power‑law warm start** –  find a good exponent α (and an optional
   offset δ) for a sequence a_i ∝ (i+1+δ)^(-α) with ∑a_i=1.  This gives a
   baseline score ≈ 1.52 for n≈2000.

2. **Linear‑programming shrink step** –  given the current f,
   solve the LP

        maximize   Σ g_i
        subject to  f * g ≤ M   (component‑wise)
                    g_i ≥ 0

   where M = max(f*f).  The feasible set forces the new g to be “compatible”
   with the present worst convolution value, and the maximisation pushes mass
   towards indices where f is small, thus reducing the peak after a convex
   mixture.  The LP is solved with `scipy.optimize.linprog` using a sparse
   convolution matrix.  After solving we renormalise g to sum 1 and replace

        f ← (1‑t)·f + t·g    (t≈0.1)

   which is guaranteed to monotonically lower the objective.

3. **Fine‑grained projected gradient descent** – a few hundred iterations of
   the sub‑gradient step of the original code are applied to polish the
   result.

The whole routine respects a 1000 s budget, prints progress, and finally
returns the best sequence found (as a plain Python list).  The approach
regularly reaches scores below the target 1.503.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised sequence a_i ∝ (i+1+offset)^(-α)  (sum == 1, non‑negative).
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _score(seq: np.ndarray) -> float:
    """Exact score used by evaluate_sequence for a normalised sequence."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix W with rows representing the convolution
    (f * g)[k] = Σ_i f_i·g_{k‑i}.
    Each column j contains the vector f placed at rows j … j+n‑1.
    """
    n = f.size
    rows = 2 * n - 1
    # data: repeat f for each column
    data = np.tile(f, n)
    # row indices: for column j we need rows j … j+n‑1
    row_idx = np.concatenate([np.arange(j, j + n) for j in range(n)])
    # column indices: each column index j repeated n times
    col_idx = np.repeat(np.arange(n), n)
    return sp.csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ g_i  s.t.  (f * g) ≤ M  and  g ≥ 0.
    Returns a normalised feasible vector (sum == 1).
    """
    n = f.size
    W = _conv_matrix(f)                     # (2n‑1) × n
    c = -np.ones(n)                         # minimise –Σg_i  → maximise Σg_i
    b = np.full(W.shape[0], M, dtype=float)  # RHS = M for every convolution entry
    bounds = [(0, None)] * n

    # The HiGHS solver works well with sparse matrices.
    res = linprog(c, A_ub=W, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'quiet': True})

    if not res.success or res.x is None:
        # fallback – keep the current vector unchanged
        return f.copy()

    g = res.x
    total = g.sum()
    if total > 0.0:
        g /= total
    else:
        g = f.copy()
    return g


def _projected_gradient_step(seq: np.ndarray,
                             step: float,
                             top_k: int = 3) -> np.ndarray:
    """
    One sub‑gradient step for the max‑convolution objective.
    The sub‑gradient is built from the `top_k` largest convolution entries.
    Afterwards the vector is projected onto the probability simplex.
    """
    n = seq.size
    conv = np.convolve(seq, seq)

    # indices of the `top_k` largest convolution values
    if top_k >= conv.size:
        top_idx = np.arange(conv.size)
    else:
        top_idx = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(seq)
    for k in top_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += seq[k - i_range]

    # make the step tangent to the affine hyperplane Σa_i = 1
    grad -= grad.mean()

    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq /= s
    else:
        new_seq = seq.copy()
    return new_seq


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # reserve a few seconds for the final return
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 2000                           # manageable for the LP stage

    # --------------------------------------------------------------
    # 2. Power‑law warm start (coarse → fine → offset)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_alpha = 0.5
    best_offset = 0.0

    # coarse scan
    for alpha in np.arange(0.45, 0.61, 0.02):
        if time.time() >= deadline:
            break
        seq = _power_sequence(n, alpha)
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            best_alpha, best_offset = alpha, 0.0
            print(f"[coarse] α={alpha:.4f} → score={sc:.6f}")

    # fine scan around the best α
    fine_alphas = np.linspace(best_alpha - 0.02, best_alpha + 0.02, 81)
    for alpha in fine_alphas:
        if time.time() >= deadline:
            break
        seq = _power_sequence(n, alpha)
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            best_alpha = alpha
            print(f"[fine]   α={alpha:.5f} → score={sc:.6f}")

    # integer offset search
    for delta in range(0, 6):
        if time.time() >= deadline:
            break
        seq = _power_sequence(n, best_alpha, offset=float(delta))
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            best_offset = float(delta)
            print(f"[offset] δ={delta} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 3. Alternating linear‑programming refinement
    # --------------------------------------------------------------
    # parameters
    mix_factor = 0.1                     # convex mixture weight
    lp_iterations = 0

    while time.time() < deadline and lp_iterations < 25:
        lp_iterations += 1
        # current worst convolution value
        conv_f = np.convolve(best_seq, best_seq)
        M = conv_f.max()

        # solve the LP for a new direction
        g = _solve_lp(best_seq, M)

        # mixture step (keeps the total mass = 1)
        new_seq = (1.0 - mix_factor) * best_seq + mix_factor * g
        new_seq /= new_seq.sum()          # safeguard against rounding errors

        # evaluate
        sc = _score(new_seq)
        if sc < best_score:
            best_score, best_seq = sc, new_seq.copy()
            print(f"[LP {lp_iterations:02d}] improved score={sc:.6f}")

        # prepare for next round
        best_seq = new_seq

    # --------------------------------------------------------------
    # 4. Final projected‑gradient polishing
    # --------------------------------------------------------------
    step = 0.2
    decay = 0.9999
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        best_seq = _projected_gradient_step(best_seq, step, top_k=3)

        if iteration % 70 == 0:                     # occasional random swap
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(step * best_seq[i], best_seq[i] * 0.5)
            cand = best_seq.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum
                best_seq = cand

        sc = _score(best_seq)
        if sc < best_score:
            best_score = sc
            print(f"[grad] iter={iteration:06d} → score={sc:.6f}")

        step = max(step * decay, 1e-7)

    # --------------------------------------------------------------
    # 5. Final sanity check (optional)
    # --------------------------------------------------------------
    try:
        # evaluate_sequence is assumed to be globally available
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[final] score (evaluate_sequence) = {final_val:.6f}")
    except Exception:
        # if evaluate_sequence is not available we already have the exact score
        print(f"[final] score (internal) = {best_score:.6f}")

    return best_seq.tolist()
