def simplex_projection(v):
    # Projects v onto the simplex {x >=0, sum x = 1}
    # Implementation from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    # v: np.ndarray
    # Project onto simplex sum=1, v>=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

        return np.maximum(v, 0)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

"""
Propose a non‑negative step‑function (sequence) that minimises
evaluate_sequence().  The method consists of:

1.  Load an existing construction if available, otherwise start from a
    uniform distribution (sum = 1).

2.  Run a few independent restarts.  Each restart creates a random
    starting shape (Dirichlet, power‑law or a simple two‑piece linear
    profile) and then performs a *projected gradient descent* on the
    squared L2‑distance between the self‑convolution and the ideal flat
    target 1/(2·n‑1).  The gradient can be written as a convolution and
    is computed efficiently with FFTs.  After every step the new vector
    is projected on the probability simplex (non‑negative, sum = 1).

3.  Keep the best sequence seen over all restarts.

4.  As a final polish a short simulated‑annealing phase (random mass
    transfers) is applied to the current best sequence to escape any
    remaining shallow local minima.

The whole search respects a user supplied time budget (default 1000 s)
and always returns the best sequence discovered before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures, no lambdas
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Small utility: next power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (full, length 2·n‑1) using an FFT of size
    a power of two for speed.  The input is assumed to be a 1‑D real
    array.
    """
    n = a.size
    size = 2 * n - 1
    m = _next_pow_two(size)
    freq = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(freq * freq, n=m)[:size]
    return conv


def _grad_squared_error(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the squared L2 error
        J(a) = Σ_k ( (a*a)[k] – 1/(2·n‑1) )²
    with respect to a (the sequence).  The formula reduces to a
    convolution that is evaluated by FFTs.
    """
    n = a.size
    target = 1.0 / (2 * n - 1)
    diff = conv - target                     # length 2·n‑1
    diff_rev = diff[::-1]                    # reversed version
    # convolution a * diff_rev  (full length 3·n‑2)
    size = n + (2 * n - 1) - 1                # = 3·n‑2
    m = _next_pow_two(size)
    Fa = np.fft.rfft(a, n=m)
    Fb = np.fft.rfft(diff_rev, n=m)
    conv_rev = np.fft.irfft(Fa * Fb, n=m)[:size]   # length 3·n‑2
    # the needed slice is conv_rev[ (2·n‑2) - i ] for i = 0 … n‑1
    idx = (2 * n - 2) - np.arange(n)               # descending indices
    grad = 4.0 * conv_rev[idx]                     # factor 4 from derivation
    return grad


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_fast(a: np.ndarray) -> float:
    """
    Fast surrogate for evaluate_sequence(): 2·n·max(conv(a,a))
    The sum of a is assumed to be 1, which is enforced by the
    optimisation loops.
    """
    n = a.size
    conv = _full_convolution(a)
    max_c = conv.max()
    return 2.0 * n * max_c


def _power_law_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Generate a normalised power‑law a_i ∝ (i+α+1)^(-p) .
    """
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _gradient_descent(
    init_seq: np.ndarray,
    max_iters: int,
    eta0: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the squared‑error objective.
    Returns the best sequence seen during the run together with its
    score (2·n·max(conv)).
    """
    n = init_seq.size
    a = init_seq.copy()
    best_seq = a.copy()
    conv = _full_convolution(a)
    best_score = 2.0 * n * conv.max()
    eta = eta0
    it = 0
    while it < max_iters and time.time() < deadline:
        # gradient of J(a)
        grad = _grad_squared_error(a, conv)

        # take a step and project back onto the simplex
        cand = a - eta * grad
        cand = _simplex_projection(cand)

        # evaluate candidate
        conv_cand = _full_convolution(cand)
        cand_score = 2.0 * n * conv_cand.max()

        # accept if we improve; otherwise keep the old point
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
        if cand_score < (2.0 * n * conv.max()):
            a = cand
            conv = conv_cand
        else:
            # if no improvement shrink the step size a little
            eta *= 0.7

        # modest decay of the step size – helps convergence
        eta *= 0.9995
        it += 1

    return best_seq, best_score

        n = 5000
        init = np.full(n, 1.0 / n)          # uniform start
        print(f"[init] fallback to uniform (n={n})")

    best_seq = init.copy()
    best_score = _score_fast(best_seq)
    print(f"[baseline] score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Multiple restarts with projected gradient descent
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline and restart < 12:
        # ---- pick a random initial shape ---------------------------------
        mode = restart % 3
        if mode == 0:
            # Dirichlet‑like (Gamma) start – concentration < 1 gives
            # spiky but diverse shapes
            x = np.random.gamma(0.5, 1.0, size=n)
            start_seq = x / x.sum()
        elif mode == 1:
            # random power‑law
            p = np.random.uniform(0.35, 0.70)
            alpha = np.random.uniform(0.0, 5.0)
            start_seq = _power_law_sequence(n, p, alpha)
        else:
            # two‑piece linear profile (more mass at the ends)
            cut = np.random.randint(1, n - 1)
            left = np.linspace(np.random.uniform(0.2, 1.0),
                               np.random.uniform(0.0, 0.3),
                               cut, endpoint=False)
            right = np.linspace(np.random.uniform(0.0, 0.3),
                                np.random.uniform(0.2, 1.0),
                                n - cut, endpoint=False)
            start_seq = np.concatenate([left, right])
            start_seq = np.maximum(start_seq, 0.0)
            start_seq /= start_seq.sum()

        # ---- gradient descent on the L2 deviation -----------------------
        max_iters = 6000
        eta0 = 0.8
        seq, score = _gradient_descent(start_seq, max_iters, eta0, deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[restart {restart:02d}] improved to {best_score:.6f}")

        restart += 1

        # early exit if we have already crossed the target bound
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # 3) Final polish – simple simulated annealing on the current best
    # --------------------------------------------------------------
    seq = best_seq.copy()
    cur_score = best_score
    it = 0
    T0, Tf = 0.8, 1e-4               # linear cooling schedule

    while time.time() < deadline:
        it += 1
        # random mass transfer between two positions
        i, j = np.random.choice(n, 2, replace=False)
        eps = seq[i] * np.random.uniform(1e-4, 5e-3)
        cand = seq.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = _simplex_projection(cand)

        conv = _full_convolution(cand)
        cand_score = 2.0 * n * conv.max()

        if cand_score < cur_score:
            accept = True
        else:
            elapsed = time.time() - start_time
            fraction = (elapsed) / (deadline - start_time)
            T = T0 * (1.0 - fraction) + Tf * fraction
            prob = math.exp(-(cand_score - cur_score) / T) if T > 0 else 0.0
            accept = np.random.rand() < prob

        if accept:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[anneal] iter={it:7d} score={best_score:.6f}")

        # optional early stop
        if best_score < 1.503:
            break

    print(f"[final] best_score={best_score:.6f} after {it} annealing steps")
    # return plain Python list as required by the harness
    return best_seq.tolist()
