Propose a candidate step‑function using a *linear‑programming based mixing* scheme
followed by a short sub‑gradient fine‑tuning.

1.  Start from an existing construction (if any) or a uniform vector whose
    ℓ₁‑norm is √(2·n).  The ℓ₁‑norm is kept fixed for the whole search.

2.  **LP mixing step** – given the current vector `a` let

        maximise   1ᵀb
        subject to  conv(a,b) ≤ M   (component‑wise)
                    b ≥ 0

    The convolution constraint is linear in `b`; it is built as a dense
    matrix `A` with shape (2·n‑1, n).  The optimiser (SciPy HiGHS) returns a

    maximum self‑convolution and usually pushes it down.

3.  The LP step is repeated while time permits (a few dozen times).  The best
    vector seen so far is stored.

4.  **Sub‑gradient refinement** – with the best vector as a warm‑start a
    projected Adam optimiser minimises the exact max‑convolution.  The
    (sub‑)gradient of `max_k conv(a,a)[k]` is obtained from the indices where
    the maximum is attained (simple closed‑form, no smoothing).

5.  Finally the best vector found is returned as a plain Python list.  All
    entries are non‑negative and the ℓ₁‑norm equals √(2·n), guaranteeing that
    `evaluate_sequence` returns exactly the maximal convolution value.
"""

import time
import math
import random
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the probability simplex {w ≥ 0, ∑w = s}."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(v: np.ndarray) -> float:
    """Return max(conv(v,v))."""
    conv = np.convolve(v, v)
    return float(conv.max())


# ----------------------------------------------------------------------
def _build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that (A @ b)[k] = conv(a, b)[k] for k = 0..2n‑2.
    A has shape (2n‑1, n) and is dense.
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes a shifted copy of a
    for i in range(n):
        A[i:i + n, i] = a
    return A


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the function f(a) = max(conv(a,a))
    (the exact max, not a smoothed version).
    """
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    max_inds = np.flatnonzero(conv == max_val)

    grad = np.zeros_like(a)
    for k in max_inds:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        # contribution 2 * a[j] for each i
        grad[i_range] += 2.0 * a[j_range]

    # average over all maximising indices (helps when several ties)
    grad /= max(1, len(max_inds))
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine first performs a few LP‑mixing steps
    and then a short Adam‑style sub‑gradient descent.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base = np.asarray(height_sequence_1, dtype=float)
    else:
        n = 2500                     # safe default when no prior solution
        base = np.full(n, 1.0)

    target_sum = math.sqrt(2.0 * n)          # ℓ₁‑norm we keep fixed

    # normalise the starting vector
    base = np.maximum(base, 0.0)
    s = base.sum()
    if s == 0:
        a = np.full(n, target_sum / n)
    else:
        a = base / s * target_sum

    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = _max_convolution(a)
    print(f"[init] max(conv) = {best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # margin for safety

    # ------------------------------------------------------------------
    # 1) Linear‑programming mixing phase
    t_mix = 0.3          # mixing coefficient
    lp_iter = 0
    last_improve = 0
    max_lp_iters = 30    # a few LP solves are enough – they are the expensive part

    while time.time() < deadline and lp_iter < max_lp_iters:
        M = _max_convolution(a)

        # build the convolution matrix A (A @ b = conv(a, b))
        A = _build_convolution_matrix(a)

        # linear programme: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        # solve with HiGHS (fast dense LP solver)
        res = opt.linprog(c, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'presolve': True, 'tol': 1e-9})

        if not res.success:
            # fallback: a random perturbation and continue
            print(f"[lp {lp_iter}] LP failed – random perturbation")
            rnd = np.random.exponential(scale=1.0, size=n)
            rnd = rnd / rnd.sum() * target_sum
            a = (1 - t_mix) * a + t_mix * rnd
            a = _simplex_projection(a, target_sum)
            lp_iter += 1
            continue

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b < 1e-12:
            break

        # scale to the required ℓ₁‑norm
        scale = target_sum / sum_b
        g = b_opt * scale

        # mixing step
        a_new = (1 - t_mix) * a + t_mix * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _simplex_projection(a_new, target_sum)

        new_score = _max_convolution(a_new)

        if new_score < best_score - 1e-9:
            best_score = new_score
            best_a = a_new.copy()
            last_improve = lp_iter
            print(f"[lp {lp_iter:03d}] New best max(conv) = {best_score:.6f}")

        # Accept the new point if it improves the current max
        if new_score < M:
            a = a_new
        else:
            # shrink the mixing parameter a little if we are not improving
            t_mix = max(t_mix * 0.9, 0.05)

        # occasional restart if we stall
        if lp_iter - last_improve > 15:
            rnd = np.random.exponential(scale=1.0, size=n)
            rnd = rnd / rnd.sum() * target_sum
            a = 0.5 * best_a + 0.5 * rnd
            a = _simplex_projection(a, target_sum)
            last_improve = lp_iter

        lp_iter += 1

    # ------------------------------------------------------------------
    # 2) Sub‑gradient / Adam fine‑tuning
    # (few thousand Adam steps with a schedule)
    a = best_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    t = 0
    max_iter = 8000
    it = 0
    last_imp = 0

    while time.time() < deadline and it < max_iter:
        # schedule of learning rate (linear decay)
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        lr = 0.5 * (1.0 - frac) + 0.01 * frac   # from 0.5 down to 0.01

        # current objective
        cur = _max_convolution(a)
        if cur < best_score - 1e-9:
            best_score = cur
            best_a = a.copy()
            last_imp = it
            print(f"[sg {it:05d}] New best (sub‑grad) max(conv) = {best_score:.6f}")

        # sub‑gradient of the exact max
        grad = _subgradient(a)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # projected Adam step
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _simplex_projection(a_new, target_sum)

        a = a_new
        it += 1

        # occasional random restart if no improvement for a while
        if it - last_imp > 2000:
            rnd = np.random.exponential(scale=1.0, size=n)
            rnd = rnd / rnd.sum() * target_sum
            a = 0.5 * best_a + 0.5 * rnd
            a = _simplex_projection(a, target_sum)
            last_imp = it

    # ------------------------------------------------------------------
    # Return the best sequence found (plain Python list)
    return best_a.tolist()
```
