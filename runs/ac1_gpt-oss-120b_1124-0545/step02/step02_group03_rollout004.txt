assistantfinal**Solution Overview**

The objective after fixing the scale (`∑a = 1`) is  

```
F(a) = 2·n·max( a * a )
```

i.e. we have to minimise the largest value of the self‑convolution of a non‑negative
vector `a`.  
The function is **scale‑invariant** and the feasible region is the probability simplex.
A direct sub‑gradient on the *hard* max is prone to plateaus.  
Instead we smooth the max with a soft‑max (log‑sum‑exp) and use an **exponentiated‑gradient
mirror‑descent** on the simplex:

```
w = soft‑max_β ( a * a )                # β controls the sharpness
g = 4·n·conv( a_rev , w )                # ∂F/∂a (up to the soft‑max)
a ← a ∘ exp( -α·g )                     # ∘ = elementwise product
a ← a / ∑a                               # projection onto the simplex
```

`β` is increased during the run so that the optimisation first follows a smooth
approximation of the max and later concentrates on the true peak.  
A tiny probability of a random “mass‑transfer’’ move and a simple stagnation‑restart
prevent the search from getting stuck.

The algorithm is fast (three FFTs per iteration) and works with any length `n`.
It starts from a collection of heuristics (uniform, sine, triangular, exponential
and the previously‑found best sequence) and keeps the best vector encountered.

```python
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT of length ``size`` (must be ≥ len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _max_convolution(a: np.ndarray, size1: int) -> np.ndarray:
    """Return a * a (full convolution) using the pre‑computed FFT size."""
    return _conv_fft(a, a, size1)


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Exponentiated‑gradient optimisation of the L∞ autocorrelation.
    A soft‑max (temperature β) smoothes the max‑convolution, the gradient
    is obtained by a second convolution, and a multiplicative update
    (mirror‑descent on the simplex) drives the distribution to lower
    peak values.  Occasional random mass‑transfer moves and a stagnation
    restart keep the search from stalling.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # ------------------------------------------------------------------
    # build a few initial candidates (uniform, sine, triangular, …)
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.ones(n) / n)

    # sine‑shaped (non‑negative)
    x = np.arange(n)
    a_sine = np.sin(np.pi * (x + 0.5) / n)
    a_sine = np.maximum(0.0, a_sine)
    a_sine /= a_sine.sum()
    candidates.append(a_sine)

    # triangular bump
    a_tri = 1.0 - np.abs((x - (n - 1) / 2) / ((n - 1) / 2))
    a_tri = np.maximum(0.0, a_tri)
    a_tri /= a_tri.sum()
    candidates.append(a_tri)

    # exponential decay
    a_exp = np.exp(-x / (n / 10.0))
    a_exp /= a_exp.sum()
    candidates.append(a_exp)

    # previous best if supplied by the harness
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None:
        seq0 = np.asarray(seq0, dtype=float)
        if seq0.size != n:
            # linear interpolation to new grid
            x_old = np.linspace(0, 1, seq0.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            seq0 = np.interp(x_new, x_old, seq0)
        seq0 = np.clip(seq0, 0.0, None)
        seq0_sum = seq0.sum()
        if seq0_sum == 0.0:
            seq0 = np.ones(n) / n
        else:
            seq0 = seq0 / seq0_sum
        candidates.append(seq0)

    # random start (fallback)
    rand_vec = np.random.rand(n)
    rand_vec = np.clip(rand_vec, 0.0, None)
    rand_vec /= rand_vec.sum()
    candidates.append(rand_vec)

    # ------------------------------------------------------------------
    # helper to evaluate the (scaled) objective: 2·n·max(conv(a,a))
    # ------------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)

    def _score(a_vec: np.ndarray) -> float:
        conv_full = _max_convolution(a_vec, size1)
        return 2.0 * n * conv_full.max()

    # pick the best among the seeds
    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            cand = np.ones_like(cand) / n
        else:
            cand = cand / s
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[init] n={n}, best seed score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    a = best_seq.copy()
    size2 = _next_pow2(3 * n - 2)   # for a_rev * w
    it = 0
    init_step = 0.5                # base step for the multiplicative update
    rand_prob = 0.03               # probability of a random mass‑transfer move
    rand_scale = 0.02
    stagnation_seconds = 30.0
    last_improve = start

    beta = 0.5                     # initial soft‑max sharpness
    beta_max = 200.0
    beta_inc = 1e-4                # increase per iteration (≈ 200 % over 2 M iters)

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1) current convolution and score
        conv_full = _max_convolution(a, size1)
        cur_score = 2.0 * n * conv_full.max()

        # keep the best seen so far
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # 2) occasional random move (exploration)
        if np.random.rand() < rand_prob:
            a = _random_transfer(a, rand_scale)
        else:
            # 3) soft‑max weights for the current autocorrelation
            shift = conv_full.max()                # for numerical stability
            exp_vals = np.exp(beta * (conv_full - shift))
            w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

            # 4) gradient of the smoothed objective
            a_rev = a[::-1]
            conv_grad_full = _conv_fft(a_rev, w, size2)   # length 3n‑2
            # central n entries give the gradient components
            g = conv_grad_full[n - 1 : n - 1 + n]
            grad = 4.0 * n * g                     # ∇F (up to the soft‑max)

            # 5) multiplicative (exponentiated) update on the simplex
            step = init_step / np.sqrt(it + 1.0)   # decaying step size
            max_grad = grad.max()
            if max_grad > 0.0:
                alpha = step / max_grad           # scale so that max update ≈ step
            else:
                alpha = 0.0
            a = a * np.exp(-alpha * grad)         # element‑wise
            s = a.sum()
            if s == 0.0:
                a = np.ones_like(a) / n            # fallback to uniform
            else:
                a = a / s

        # 6) stagnation restart – re‑inject a tiny amount of noise
        if time.time() - last_improve > stagnation_seconds:
            a = best_seq.copy()
            a += 0.01 * np.random.rand(n) * a     # 1 % relative noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = time.time()
            print(f"[{it:7d}] restart (stagnation)  time={last_improve-start:.1f}s")

        # increase the soft‑max temperature
        beta = min(beta_max, beta + beta_inc)

        it += 1

    # ------------------------------------------------------------------
    # final report
    # ------------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
```
