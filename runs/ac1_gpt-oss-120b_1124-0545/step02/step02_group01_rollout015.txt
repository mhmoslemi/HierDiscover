Optimises a non‑negative step‑function to minimise
2*n*max(conv(a,a))/(sum(a)**2) using a mixture of
Adam (smooth max) and exponentiated‑gradient descent.
Seeds include geometric, power‑law, windows and random
initialisations.  Time budget is respected and the best
sequence is returned as a plain list.

def _next_pow2(x: int) -> int:
    """return smallest power of two >= x"""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """full linear convolution via real FFT"""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """stable softmax, returns distribution that sums to 1"""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    # From https://arxiv.org/pdf/1309.1541.pdf
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _power(n: int, p: float) -> np.ndarray:

    w = (1 - np.cos(2 * np.pi * i / (n - 1))) ** p
    s = w.sum()
    return w / s if s != 0 else np.full(n, 1.0 / n)

def _gaussian_window(n: int, sigma_frac: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)

def _kaiser_window(n: int, beta: float = 8.0) -> np.ndarray:
    if _has_scipy:
        w = _windows.kaiser(n, beta)
        s = w.sum()
        return w / s if s != 0 else np.full(n, 1.0 / n)
    else:
        # fall back to sine window
        return _sine_window(n)

def _linear_ramp(n: int) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64)
    s = a.sum()
    return a / s

def _inverse_ramp(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s

def _random_simplex(n: int) -> np.ndarray:
    # sample Dirichlet(1,...,1) via exponential
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()

def _build_seeds(n: int) -> List[np.ndarray]:
    seeds: List[np.ndarray] = []
    # geometric families
    for c in np.linspace(0.5, 5.0, 10):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_geom(n, r))
    # power families
    for p in np.linspace(0.0, 2.2, 12):
        seeds.append(_power(n, p))
    # combined geometric * power
    for c in np.linspace(0.5, 4.0, 6):
        r = 1.0 - c / n
        if r <= 0:
            continue
        for p in np.linspace(0.0, 2.0, 6):
            seeds.append(_geom_power(n, r, p))
    # symmetric geometric
    for c in np.linspace(0.5, 4.5, 6):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_sym_geom(n, r))
    # smooth windows
    seeds.append(_sine_window(n))
    seeds.append(_raised_cosine(n, p=0.5))
    seeds.append(_raised_cosine(n, p=1.0))
    seeds.append(_gaussian_window(n, sigma_frac=0.15))
    seeds.append(_gaussian_window(n, sigma_frac=0.25))
    # kaiser windows with various betas
    if _has_scipy:
        for beta in [3.0, 6.0, 12.0]:
            seeds.append(_kaiser_window(n, beta))
    # linear ramps
    seeds.append(_linear_ramp(n))
    seeds.append(_inverse_ramp(n))
    # random draws
    for _ in range(8):
        seeds.append(_random_simplex(n))
    # previously discovered sequence if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            a_prev = np.asarray(prev, dtype=np.float64)
            if a_prev.ndim == 1 and a_prev.size > 0:
                if a_prev.shape[0] != n:
                    # Resize by truncation/padding then renorm
                    tmp = np.zeros(n, dtype=np.float64)
                    m = min(a_prev.shape[0], n)
                    tmp[:m] = a_prev[:m]
                    a_prev = tmp
                a_prev = np.maximum(a_prev, 0.0)
                if a_prev.sum() == 0.0:
                    a_prev = _random_simplex(n)
                else:
                    a_prev = a_prev / a_prev.sum()
                seeds.append(a_prev)
        except Exception:
            pass
    return seeds

# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(seed_a: np.ndarray, n: int, deadline: float,
                   max_iters: int = 12000) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (log‑sum‑exp)."""
    v = np.log(seed_a + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_a = seed_a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))
    # hyper‑parameters
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta_start = 0.5
    beta_mul = 1.025
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # smooth‑max weight vector
        beta = beta_start * (beta_mul ** it)
        shift = cur_max                     # numerical shift
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum) or np.isinf(w_sum):
            # skip this iteration if overflow/underflow occurs
            continue
        w /= w_sum
        # gradient w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)         # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries
        # back‑prop to logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
    return best_a, best_val

def _expgrad_optimize(seed_a: np.ndarray, n: int, deadline: float,
                      max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the true max."""
    a = seed_a.copy()
    best_a = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    lr0 = 0.4
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # sub‑gradient of the max (choose the peak index)
        k = int(np.argmax(conv))
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # exponentiated‑gradient step
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * grad)   # stay non‑negative
        # project back onto the simplex
        a = _project_simplex(a)
    return best_a, best_val

def _run_optimisers_for_seed(seed_a: np.ndarray, n: int, deadline: float,
                             global_best_val: float) -> tuple[np.ndarray, float]:
    """Run both Adam and exponentiated‑gradient, keep the better result."""
    best_seq = None
    best_val = float("inf")
    # Adam phase – relatively long
    seq_adam, val_adam = _adam_optimize(seed_a, n, deadline, max_iters=12000)
    if val_adam < best_val:
        best_val = val_adam
        best_seq = seq_adam
    # Exponential‑gradient phase – short refinement
    seq_exp, val_exp = _expgrad_optimize(seq_adam, n, deadline,
                                         max_iters=6000)
    if val_exp < best_val:
        best_val = val_exp
        best_seq = seq_exp
    # also try starting directly from the original seed with exp‑grad
    seq_exp2, val_exp2 = _expgrad_optimize(seed_a, n, deadline,
                                           max_iters=6000)
    if val_exp2 < best_val:
        best_val = val_exp2
        best_seq = seq_exp2
    # Global comparison (allow early exit if already worse than global best)
    if best_val >= global_best_val:
        # no improvement over known global best, keep the old best
        return None, None
    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a buffer

    # Choose a single length that historically gives good results.
    # If time permits we also try a second, larger length.
    lengths = [8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"[main] Optimising length {n}")

        # ------------------------------------------------------------------
        # Generate seeds
        # ------------------------------------------------------------------
        seed_seqs = _build_seeds(n)

        # ------------------------------------------------------------------
        # Quick scan to get an initial reference point
        # ------------------------------------------------------------------
        for a in seed_seqs:
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best (len={n}) = {global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Optimise each seed – time‑aware loop
        # ------------------------------------------------------------------
        for idx, seed_a in enumerate(seed_seqs):
            if time.time() > deadline:
                break
            # Run the hybrid optimiser; early‑exit if it cannot beat global best
            seq, val = _run_optimisers_for_seed(seed_a, n, deadline, global_best_val)
            if seq is not None and val is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[optim] len={n} seed={idx} new global best = {global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Local perturbation stage – explore neighbourhood of the current best
        # ------------------------------------------------------------------
        if global_best_seq is not None:
            for pert in range(12):
                if time.time() > deadline:
                    break
                # multiplicative Gaussian perturbation (keep positivity)
                noise = np.random.randn(n) * 0.05   # 5 % noise
                a_pert = global_best_seq * np.exp(noise)
                a_pert = _project_simplex(a_pert)
                # short refinement
                a_ref, val_ref = _expgrad_optimize(a_pert, n, deadline, max_iters=3000)
                if val_ref < global_best_val:
                    global_best_val = val_ref
                    global_best_seq = a_ref.copy()
                    print(f"[pert] len={n} pert={pert} improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("[fallback] using uniform distribution")

    # Return list of floats
    return global_best_seq.tolist()

"""
optimiser for the step‑function benchmark.

The routine builds a diverse set of initial “seeds” (geometric,
power‑law, Gaussian, sine/kaiser windows, random Dirichlet draws and
any previously‑found solution).  For each seed a two‑stage optimiser
is run:

* Adam on a log‑sum‑exp smoothed version of the max‑convolution
  (the standard “soft‑max” surrogate).  The smoothing parameter β
  starts small (β≈0.5) and is increased geometrically; Adam’s learning
  rate decays as 1/√t.

* A short “exponentiated‑gradient” (mirror‑descent) phase that uses
  the true sub‑gradient of the max‑convolution (the peak index) and
  projects back onto the probability simplex after each update.

The best sequence across all seeds and a final perturbation‑refinement
stage is kept.  The loop respects a user supplied time budget (default
1000 s) and returns the best sequence as a plain list of floats.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Optional SciPy windows for extra seeds
# ----------------------------------------------------------------------
try:
    from scipy.signal import windows as _scipy_windows
    _has_scipy = True
except Exception:          # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the lengths used)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x)=1}.  Implementation follows
    https://arxiv.org/pdf/1309.1541.pdf.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families (all return a vector that sums to 1)
# ----------------------------------------------------------------------
def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _raised_cosine(n: int, power: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (1 - np.cos(2 * np.pi * i / (n - 1))) ** power
    s = w.sum()
    return w / s if s != 0 else np.full(n, 1.0 / n)


def _gaussian_window(n: int, sigma_frac: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _kaiser_window(n: int, beta: float = 8.0) -> np.ndarray:
    if _has_scipy:
        w = _scipy_windows.kaiser(n, beta)
        s = w.sum()
        return w / s if s != 0 else np.full(n, 1.0 / n)
    # fall‑back to a sine window
    return _sine_window(n)


def _linear_ramp(n: int) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _inverse_ramp(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


def _build_seeds(n: int) -> List[np.ndarray]:
    """Collect a modest, diverse list of initial probability vectors."""
    seeds: List[np.ndarray] = []

    # geometric families
    for c in np.linspace(0.5, 5.0, 10):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # pure power‑law families
    for p in np.linspace(0.0, 2.2, 12):
        seeds.append(_power(n, p))

    # combined geometric × power families
    for c in np.linspace(0.5, 4.0, 6):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 6):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.5, 6):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # smooth windows
    seeds.append(_sine_window(n))
    seeds.append(_raised_cosine(n, power=0.5))
    seeds.append(_raised_cosine(n, power=1.0))
    seeds.append(_gaussian_window(n, sigma_frac=0.15))
    seeds.append(_gaussian_window(n, sigma_frac=0.25))
    if _has_scipy:
        for beta in (3.0, 6.0, 12.0):
            seeds.append(_kaiser_window(n, beta))

    # linear ramps
    seeds.append(_linear_ramp(n))
    seeds.append(_inverse_ramp(n))

    # a few random simplex points
    for _ in range(8):
        seeds.append(_random_simplex(n))

    # possibly reuse a previously discovered best sequence
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            a_prev = np.asarray(prev, dtype=np.float64)
            if a_prev.ndim == 1 and a_prev.size > 0:
                if a_prev.shape[0] != n:
                    tmp = np.zeros(n, dtype=np.float64)
                    m = min(a_prev.shape[0], n)
                    tmp[:m] = a_prev[:m]
                    a_prev = tmp
                a_prev = np.maximum(a_prev, 0.0)
                if a_prev.sum() == 0.0:
                    a_prev = _random_simplex(n)
                else:
                    a_prev = a_prev / a_prev.sum()
                seeds.append(a_prev)
        except Exception:          # pragma: no cover
            pass

    # limit the number of seeds to keep the run inside the time budget
    np.random.shuffle(seeds)
    return seeds[:30]      # 30 seeds are usually enough for a good coverage


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(seed_a: np.ndarray, n: int, deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smooth approximation of the max.
    Returns the best sequence seen and its true objective value.
    """
    v = np.log(seed_a + 1e-30)          # logits → softmax keeps us on simplex
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_a = seed_a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))

    # hyper‑parameters
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta_start = 0.5
    beta_mul = 1.025

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stay on simplex
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # --------- smooth‑max weights ----------
        beta = beta_start * (beta_mul ** it)
        shift = cur_max                         # shift to avoid overflow
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum) or np.isinf(w_sum):
            continue
        w /= w_sum

        # --------- gradient w.r.t. a ----------
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)             # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # --------- back‑prop to logits ----------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # --------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_a, best_val


def _expgrad_optimize(seed_a: np.ndarray, n: int, deadline: float,
                      max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the true max‑convolution.
    The sub‑gradient is taken at the peak index of the convolution.
    """
    a = seed_a.copy()
    best_a = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    lr0 = 0.4

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # sub‑gradient of the max (peak index)
        k = int(np.argmax(conv))
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # exponentiated‑gradient step
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * grad)      # stays non‑negative
        a = _project_simplex(a)         # back onto simplex

    return best_a, best_val


def _optimise_one_seed(seed_a: np.ndarray, n: int, deadline: float,
                       global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Hybrid optimisation: Adam → short exp‑grad refinement.
    Returns (best_seq, best_val) if an improvement over global_best_val
    is achieved, otherwise (None, None).
    """
    # First stage – Adam (fairly long)
    seq_adam, val_adam = _adam_optimize(seed_a, n, deadline,
                                        max_iters=8000)

    # Second stage – refinement with exponentiated‑gradient
    seq_ref, val_ref = _expgrad_optimize(seq_adam, n, deadline,
                                         max_iters=4000)

    # a third attempt: start the exp‑grad directly from the original seed
    seq_ref2, val_ref2 = _expgrad_optimize(seed_a, n, deadline,
                                           max_iters=4000)

    # choose the best among the three runs
    candidates = [(seq_adam, val_adam), (seq_ref, val_ref), (seq_ref2, val_ref2)]
    best_seq, best_val = min(candidates, key=lambda kv: kv[1])

    if best_val < global_best_val:
        return best_seq, best_val
    return None, None


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    # keep a small safety margin for the surrounding harness
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Lengths we will try – start with 8000 (the best length observed so far)
    # ------------------------------------------------------------------
    lengths = [8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"[main] Optimising length {n}")

        # ------------------------------------------------------------------
        # 1) generate seeds
        # ------------------------------------------------------------------
        seed_seqs = _build_seeds(n)

        # ------------------------------------------------------------------
        # 2) quick scan with the fast internal evaluator to get a starter
        # ------------------------------------------------------------------
        for a in seed_seqs:
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best (len={n}) = {global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) hybrid optimisation for every seed (time‑aware)
        # ------------------------------------------------------------------
        for idx, seed_a in enumerate(seed_seqs):
            if time.time() > deadline:
                break
            # avoid spending time on obviously bad seeds
            if 2.0 * n * np.max(_conv_fft(seed_a, seed_a)) >= global_best_val * 1.05:
                # seed is already >5 % worse than current best – skip it
                continue
            seq, val = _optimise_one_seed(seed_a, n, deadline, global_best_val)
            if seq is not None and val is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[optim] len={n} seed={idx} new global best = {global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 4) neighbourhood search around the current best (perturb + refine)
        # ------------------------------------------------------------------
        if global_best_seq is not None:
            for pert in range(12):
                if time.time() > deadline:
                    break
                # multiplicative Gaussian noise (~5 % relative)
                noise = np.random.randn(n) * 0.05
                a_pert = global_best_seq * np.exp(noise)
                a_pert = _project_simplex(a_pert)
                # short refinement with exponentiated‑gradient
                a_ref, val_ref = _expgrad_optimize(a_pert, n, deadline,
                                                   max_iters=3000)
                if val_ref < global_best_val:
                    global_best_val = val_ref
                    global_best_seq = a_ref.copy()
                    print(f"[pert] len={n} pert={pert} improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never happen)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("[fallback] using uniform distribution")

    # Return as plain Python list
    return global_best_seq.tolist()
