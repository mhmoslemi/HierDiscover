def propose_candidate(...):
    import time, numpy as np, math, random, scipy.optimize

    # define n, target_sum, base vector.
    # define projection onto simplex used for baseline: ignore because we will param via softmax.

    def compute_conv(a):
        return np.convolve(a, a)

    def objective_and_grad(y, eps):
        # compute a via softmax
        maxy = np.max(y)
        exp_y = np.exp(y - maxy)  # for numerical stability
        sum_exp = np.sum(exp_y)
        a = exp_y / sum_exp * target_sum  # shape (n,) float
        # compute conv
        conv = np.convolve(a, a)
        # compute smoothed max L = eps * log sum exp(conv/eps)
        # but we can use np.logaddexp.reduce
        # more stable: compute max conv
        max_conv = np.max(conv)
        # use log-sum-exp
        conv_shift = conv - max_conv
        w_raw = np.exp(conv_shift / eps)
        w = w_raw / np.sum(w_raw)  # shape (2n-1,)
        # compute gradient w.r.t a:
        # gradient from earlier: grad = first + a * sigma
        # where sigma = moving sum of w over length n; compute these as before.
        n_ = len(a)
        # compute sigma (size n)
        cumsum_w = np.cumsum(w)
        # sigma[t] = sum_{i=t}^{t+n-1} w_i
        sigma = cumsum_w[n_-1:2*n_-1] - np.concatenate(([0.0], cumsum_w[:n_-1]))
        # compute first term: conv_wa = conv(w, a_rev)
        a_rev = a[::-1]
        conv_wa = np.convolve(w, a_rev)
        first = conv_wa[(n_-1):(n_-1+n_)]
        grad_a = first + a * sigma

        # chain rule for softmax: da/dy_i = a_i * (δ_{ij} - a_j/target_sum)

"""
Propose a step‑function that minimizes the evaluation metric
E(a) = 2·n·max(conv(a,a)) / (∑ a)².

The algorithm consists of three stages:

1.  **Soft‑max (log‑sum‑exp) optimisation** –  the non‑negative
    vector is parametrised by soft‑max variables `y`.  For a given
    temperature `eps` the objective is
        L(y) = eps·log(∑ exp(conv(a(y))/eps))
    whose gradient w.r.t. `y` can be written analytically.
    A sequence of decreasing temperatures is processed, each time
    using a limited‑memory BFGS (L‑BFGS‑B) optimiser.  Several
    initialisations are tried (uniform, exponential, the existing
    construction and a random start).

2.  **Projection on the simplex** – the mapping `y → a` is performed
    by a soft‑max, which automatically yields a non‑negative vector
    with the required ℓ₁‑norm `∑ a = √(2·n)`.

3.  **Simple hill‑climbing refinement** – after the continuous
    optimisation a fast coordinate‑wise local search moves a tiny
    amount of mass from a random source index to a random target
    index whenever the maximum entry of the self‑convolution is
    reduced.  The search runs until the time budget is exhausted.

The whole routine respects a 1000 s wall‑clock limit; the best
sequence found is returned as a plain Python list.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax_to_a(y: np.ndarray, target_sum: float) -> np.ndarray:
    """Convert unconstrained logits y to a probability vector a with ℓ₁‑norm = target_sum."""
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)            # shift for numerical stability
    sum_exp = np.sum(exp_y)
    a = exp_y / sum_exp * target_sum
    return a


# ----------------------------------------------------------------------
def _objective_and_grad(y: np.ndarray, target_sum: float, eps: float):
    """
    Smooth approximation of the max‑convolution objective:
        L(y) = eps·log(∑ exp(conv(a(y))/eps))

    Returns the scalar L and its gradient w.r.t. y.
    """
    # ----- a = soft‑max(y) with fixed ℓ₁‑norm -----
    a = _softmax_to_a(y, target_sum)

    # ----- convolution a*a -----
    conv = np.convolve(a, a)                     # length 2n‑1
    max_conv = np.max(conv)

    # ----- weights for the log‑sum‑exp -----
    conv_shift = conv - max_conv
    # exp arguments are ≤ 0, so overflow is not an issue
    w_raw = np.exp(conv_shift / eps)
    w_sum = np.sum(w_raw)
    if w_sum == 0 or not np.isfinite(w_sum):
        # hard max fallback: one‑hot on the maximal index
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    else:
        w = w_raw / w_sum

    n = a.size

    # ----- gradient of the smoothed objective w.r.t. a -----
    # sigma[t] = Σ_{i=t}^{t+n‑1} w_i   (moving window sum of length n)
    cumsum_w = np.cumsum(w)
    sigma = cumsum_w[n - 1:2 * n - 1] - np.concatenate(([0.0], cumsum_w[:n - 1]))

    # first term = (w ⋆ a_rev)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    conv_wa = np.convolve(w, a_rev)      # length 3n‑2
    first = conv_wa[n - 1:n - 1 + n]

    grad_a = first + a * sigma           # ∂L/∂a  (vector of length n)

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)               # <∂L/∂a , a>
    grad_y = a * (grad_a - dot / target_sum)

    # ----- scalar objective value -----
    L = max_conv + eps * np.log(w_sum)    # = eps·log‑sum‑exp(conv/eps)

    return L, grad_y


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget and returns the best sequence found.
    """
    # --------------------------------------------------------------
    # Basic settings
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # --------------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base_vec = np.asarray(height_sequence_1, dtype=float)
    else:
        n = 2500                     # fallback length
        base_vec = np.full(n, 1.0)

    target_sum = math.sqrt(2.0 * n)          # ℓ₁‑norm that makes the objective = max(conv)

    # Normalise the supplied base vector (if any)
    base_vec = base_vec / np.sum(base_vec) * target_sum

    # --------------------------------------------------------------
    # Prepare several starting points (soft‑max logits)
    starters = []

    # 1) the existing construction
    starters.append(_softmax_to_a(np.log(np.where(base_vec > 0, base_vec, 1e-12)), target_sum))

    # 2) uniform distribution
    uniform = np.full(n, target_sum / n)
    starters.append(_softmax_to_a(np.log(np.where(uniform > 0, uniform, 1e-12)), target_sum))

    # 3) exponential random vector
    rand_exp = np.random.exponential(scale=1.0, size=n)
    rand_exp = rand_exp / rand_exp.sum() * target_sum
    starters.append(_softmax_to_a(np.log(np.where(rand_exp > 0, rand_exp, 1e-12)), target_sum))

    # 4) a random start
    rand_logits = np.random.normal(size=n)
    starters.append(_softmax_to_a(rand_logits, target_sum))

    # keep the best found so far
    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Temperature schedule for the smoothed max
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]

    # L‑BFGS options – keep them modest to stay within the time budget
    lbfgs_opts = {"maxiter": 200, "disp": False}

    # --------------------------------------------------------------
    # Main optimisation loop over the different initialisations
    for idx, a0 in enumerate(starters):
        if time.time() > deadline:
            break
        # initialise logits from a0 (soft‑max inverse)
        y = np.log(np.where(a0 > 0, a0, 1e-12))

        for eps in eps_schedule:
            if time.time() > deadline:
                break

            # minimise the smooth objective for the current eps
            res = minimize(fun=_objective_and_grad,
                           x0=y,
                           args=(target_sum, eps),
                           method="L-BFGS-B",
                           jac=True,
                           options=lbfgs_opts)

            y = res.x
            a = _softmax_to_a(y, target_sum)

            # evaluate the true objective (max of the self‑convolution)
            cur_score = float(np.convolve(a, a).max())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(f"[init {idx:02d}] eps={eps:.5f} → new best max(conv)={best_score:.6f} (t={elapsed:.1f}s)")

        # ----------------------------------------------------------
        # Optional small random perturbation after each full schedule
        # (helps to jump out of very flat plateaus)
        if time.time() < deadline:
            perturb = np.random.exponential(scale=1.0, size=n)
            perturb = perturb / perturb.sum() * target_sum
            mix = np.random.rand()
            a_pert = mix * a + (1.0 - mix) * perturb
            a_pert = _simplex_projection(a_pert, target_sum)   # ensure exact ℓ₁‑norm
            score_pert = float(np.convolve(a_pert, a_pert).max())
            if score_pert < best_score:
                best_score = score_pert
                best_seq = a_pert.copy()
                elapsed = time.time() - start_time
                print(f"[init {idx:02d}] post‑perturb → new best max(conv)={best_score:.6f} (t={elapsed:.1f}s)")

    # --------------------------------------------------------------
    # Simple hill‑climbing refinement (mass swaps) – works directly on a
    if best_seq is None:
        # fallback – should never happen
        best_seq = np.full(n, target_sum / n)
        best_score = float(np.convolve(best_seq, best_seq).max())

    a_cur = best_seq.copy()
    cur_score = best_score
    it = 0
    # step size for swaps (a small fraction of the total ℓ₁‑norm)
    delta_frac = 0.001

    while time.time() < deadline and it < 200000:
        # pick a donor (with some mass) and a receiver
        i = np.random.randint(n)
        if a_cur[i] <= 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = min(a_cur[i], target_sum * delta_frac)
        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # because we move mass only, the ℓ₁‑norm stays unchanged
        # evaluate
        new_score = float(np.convolve(a_new, a_new).max())
        if new_score < cur_score:
            a_cur = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_cur.copy()
                elapsed = time.time() - start_time
                print(f"[hill {it:06d}] improvement → max(conv)={best_score:.6f} (t={elapsed:.1f}s)")
        it += 1

    # --------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    return best_seq.tolist()
