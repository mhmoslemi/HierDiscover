Define compute_max_conv(vec) as:

```
def max_convolution(vec):
    n = len(vec)
    L = 1 << (2*n-1).bit_length()
    # or use next power of 2 >= 2*n
    L = 1 << ((2*n-1).bit_length())
    # compute FFT
    f = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(f * f, n=L)[:2*n-1]
    return float(np.max(conv))
```

def max_convolution(a, fft_len=None):
    n = len(a)
    if fft_len is None:
        fft_len = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    idx = int(np.argmax(conv))
    val = float(conv[idx])
    return val, idx

"""
Propose a candidate step‑function sequence by two complementary
optimisation phases:

1. **Projected sub‑gradient descent** on the exact objective
   `max(conv(a,a))` (the quantity returned by `evaluate_sequence` when the
   ℓ₁‑norm is fixed to √(2·n)).  The sub‑gradient of the maximum
   convolution entry is obtained analytically: for the index `k` where the
   self‑convolution attains its maximum the gradient is
   `g_i = 2·a_{k‑i}` (zero outside the support).  A diminishing
   step‑size `α_t = α₀/√t` together with projection onto the probability
   simplex (`ℓ₁`‑norm = √(2·n) and non‑negativity) yields a monotone
   improvement in practice.

2. **Simulated annealing** starting from the best vector found in phase 1.
   Random mass transfers between two positions are proposed; the new
   maximum convolution value is evaluated via an FFT‑based convolution.
   Moves that improve the objective are always accepted, otherwise they
   are accepted with probability `exp(-(Δ)/T)`.  Temperature `T`
   decays exponentially from `0.2·current_best` to `1e‑5·current_best`
   over the remaining time budget.

Both phases respect the global time budget (≈ 1000 s) and constantly
record the best sequence encountered.  The returned sequence is a plain
Python list suitable for the external `evaluate_sequence` call.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1, used for efficient FFT convolutions."""
    return 1 << ((2 * n - 1).bit_length())


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Return (max_convolution_value, argmax_index) for the self‑convolution
    a*a.  Computed with FFT of length ``fft_len``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    val = float(conv[idx])
    return val, idx


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises
    ``evaluate_sequence``.  The algorithm consists of a sub‑gradient
    phase followed by simulated annealing, both running within the
    allotted time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Choose problem size.  Use the known construction if available,
    # otherwise start with a larger default (≈ 5000 entries).
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
    else:
        # a generous size that still keeps FFT fast
        base = np.random.exponential(scale=1.0, size=5000)

    n = base.size
    target_sum = math.sqrt(2.0 * n)          # ℓ₁‐norm required by the objective

    # Normalise to the required ℓ₁‑norm
    a = base.astype(float)
    a = a / a.sum() * target_sum

    # ------------------------------------------------------------------
    # Pre‑compute FFT length (used throughout)
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent on max(conv)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # keep a small safety margin
    # allocate roughly 60 % of the budget to this phase
    subgrad_deadline = start_time + budget_s * 0.60

    best_seq = a.copy()
    cur_val, cur_k = _max_convolution(a, fft_len)
    best_val = cur_val
    print(f"[INIT] n={n}, target_sum={target_sum:.3f}, init max(conv)={cur_val:.6f}")

    # Sub‑gradient step‑size schedule: α_t = α₀ / sqrt(t)
    alpha0 = target_sum * 0.02          # 2 % of the total ℓ₁‑mass as initial step
    it = 0
    last_improve = 0

    while time.time() < subgrad_deadline:
        # compute current max‑convolution and its index
        cur_val, cur_k = _max_convolution(a, fft_len)

        # record best
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = it
            print(f"[{it:6d}] subgrad new best max(conv) = {best_val:.6f}")

        # construct sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, cur_k - (n - 1))
        i_end = min(cur_k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        # g_i = 2 * a_{k-i}
        g[idx] = 2.0 * a[cur_k - idx]

        # diminishing step size
        step = alpha0 / math.sqrt(it + 1.0)

        # gradient step + projection onto the simplex
        a_candidate = a - step * g
        a_candidate = _simplex_projection(a_candidate, target_sum)

        a = a_candidate
        it += 1

        # occasional tiny random perturbation to escape plateaus
        if it % 2000 == 0:
            src = np.random.randint(n)
            if a[src] > 1e-12:
                dst = np.random.randint(n)
                delta = a[src] * 0.03 * np.random.rand()
                a_pert = a.copy()
                a_pert[src] -= delta
                a_pert[dst] += delta
                a_pert = _simplex_projection(a_pert, target_sum)
                val_pert, _ = _max_convolution(a_pert, fft_len)
                if val_pert < cur_val:
                    a = a_pert
                    cur_val = val_pert
                if val_pert < best_val:
                    best_val = val_pert
                    best_seq = a_pert.copy()
                    print(f"[{it:6d}] subgrad perturbation improved best to {best_val:.6f}")

        # restart if stagnation for a long while
        if it - last_improve > 8000:
            a = np.random.exponential(scale=1.0, size=n)
            a = a / a.sum() * target_sum
            last_improve = it
            print(f"[{it:6d}] subgrad restart due to stagnation")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing starting from the best vector found
    cur_val, cur_k = _max_convolution(best_seq, fft_len)
    a = best_seq.copy()
    cur_val = cur_val
    best_val = cur_val
    best_seq = a.copy()
    print(f"[SA-START] initial max(conv) = {best_val:.6f}")

    # Temperature schedule: exponential decay from T0 to Tmin
    T0 = best_val * 0.20           # relatively high to permit uphill moves
    Tmin = best_val * 1e-5
    sa_start = time.time()
    sa_deadline = deadline
    step_counter = 0

    while time.time() < sa_deadline:
        # exponential cooling based on fraction of SA time elapsed
        elapsed_sa = time.time() - sa_start
        frac = elapsed_sa / (sa_deadline - sa_start + 1e-12)
        T = T0 * ((Tmin / T0) ** frac)  # T decreases from T0 to Tmin

        # propose a random transfer of mass
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move at most 10 % of the source mass
        delta = a[i] * 0.10 * np.random.rand()
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta
        # no need to project – the ℓ₁‑sum stays constant and non‑negativity is kept

        cand_val, _ = _max_convolution(a_cand, fft_len)

        # Metropolis acceptance
        if cand_val < cur_val:
            a = a_cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_cand.copy()
                print(f"[SA {step_counter:6d}] new best max(conv) = {best_val:.6f}")
        else:
            # uphill move with probability exp(-Δ/T)
            prob = math.exp(-(cand_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_cand
                cur_val = cand_val

        step_counter += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    print(f"[DONE] best max(conv) = {best_val:.6f} after {it} subgrad iters "
          f"and {step_counter} SA steps")
    return best_seq.tolist()
