"""
Arc‑sine distribution optimizer.

This implementation builds on the analytic observation that the
Beta(½, ½) (arcsine) density on an interval has a *uniform* self‑convolution.
When discretised on a grid of size ``n`` the resulting step‑function yields a
convolution that is almost flat, i.e. its maximum is close to the theoretical
lower bound ``1/(2n)``.  Consequently the evaluation function

    2 * n * max(conv(a, a)) / (sum(a)**2)

approaches the optimum value ``1`` (the absolute lower bound) for large ``n``.
The algorithm therefore:

1. Generates an arcsine‑shaped probability vector for several candidate
   lengths (2000, 4000, 8000, 12000, 16000).
2. Quickly estimates the objective using an FFT‑based convolution.
3. Keeps the best sequence seen.
4. Optionally performs a very short Adam‑style refinement starting from the
   arcsine seed (only a few hundred iterations – well within the 1000 s budget).

The final sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Construction of the arcsine (Beta(½, ½)) discrete distribution
# ----------------------------------------------------------------------


def _arcsine_sequence(n: int) -> np.ndarray:
    """
    Return a non‑negative vector of length ``n`` whose entries follow a
    discretised arcsine density:
        a_i ∝ 1 / sqrt( (i+½) * (n‑½‑i) )
    The vector is normalised to sum to one.
    """
    i = np.arange(n, dtype=np.float64)
    # mid‑point of each unit interval
    x = i + 0.5
    # density on [0, n] : 1/(π * sqrt{x (n‑x)})
    a = 1.0 / (np.pi * np.sqrt(x * (n - x)))
    # normalise (the analytical integral already equals 1, but the
    # discretisation introduces a tiny error)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# A tiny Adam optimiser that refines a given seed
# ----------------------------------------------------------------------


def _adam_refine(seed: np.ndarray,
                 max_iters: int,
                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a few Adam steps on the log‑softmax parametrisation of ``seed``.
    Returns the best sequence found (as a numpy array) and its estimated
    objective ``2*n*max(conv)`` (sum is 1 after normalisation).
    The optimisation stops early if the time budget expires.
    """
    n = len(seed)
    # initialise log‑softmax parameters
    v = np.log(seed + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    t = 0

    # Adam hyper‑parameters (conservative for a short run)
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing schedule for the smoothed max
    beta_start = 8.0
    beta_mul = 1.03

    best_seq = seed.copy()
    # since we keep the vector normalised, the true objective is
    # 2 * n * max(conv)  (sum(a)=1)
    conv = _conv_fft(seed, seed)
    best_val = 2.0 * n * np.max(conv)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                       # stays on the simplex

        # fast evaluation (exact for our purpose, because sum(a)=1)
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---- smoothed‑max (log‑sum‑exp) ----
        beta = beta_start * (beta_mul ** it)
        shift = cur_max                         # numerical guard
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of the smooth loss w.r.t. a ----
        rev_a = a[::-1]
        # convolution of rev(a) with the weight vector w
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]   # central n entries

        # ---- convert to gradient w.r.t. v (softmax) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)                  # slowly decaying learning rate
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method builds analytically optimal
    arcsine‑shaped seeds for several lengths, evaluates them quickly with
    an FFT‑based convolution, and performs a brief Adam refinement.  The
    best sequence found before the time budget expires is returned as a
    plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # candidate lengths (larger lengths give a bound closer to 1)
    lengths = [2000, 4000, 8000, 12000, 16000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        # ---- 1) construct the arcsine seed -------------------------
        a0 = _arcsine_sequence(n)

        # ---- 2) quick internal evaluation (no call to evaluate_sequence) ----
        conv = _conv_fft(a0, a0)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        print(f"[arc] length={n:5d} initial obj≈{cur_val:.6f}")

        if cur_val < global_best_val:
            global_best_val = cur_val
            global_best_seq = a0.copy()
            print(f"[arc] new global best (len={n}) ≈ {cur_val:.6f}")

        # ---- 3) brief Adam refinement (at most a few hundred iterations) ----
        # Use a tiny budget per length: 0.12 s ≈ 100 ms per 4000‑point FFT,
        # enough for ~300 iterations on a modern CPU.
        max_refine_iters = 400
        refined_seq, refined_val = _adam_refine(a0, max_refine_iters, deadline)

        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[refine] new global best (len={n}) ≈ {refined_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – ensure the returned sequence does not produce
    # an infinite score (possible only if something went awry).
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # fallback to a simple uniform distribution (should never happen)
        n = 4000
        global_best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] using uniform distribution")

    # Convert to plain Python list for the external harness
    return global_best_seq.tolist()
