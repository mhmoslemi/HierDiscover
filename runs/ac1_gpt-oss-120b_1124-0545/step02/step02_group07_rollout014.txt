"""
Fast search for a low‑evaluation step‑function.

The evaluation
    E(a) = 2·n·max_k (a∗a)_k / (∑a_i)²
is homogeneous: scaling the whole vector does not change the score.
For a pure (truncated) geometric sequence a_i = r^i the score depends only on
the decay parameter c where r = exp(-c/n).  Analytically

        E(c) = 2·c²·exp(-c) / (1‑exp(-c))² .

E(c) is decreasing for c ≳ 2.5 and tends to 0 as c→∞, so a simple scan over
c quickly finds a vector with a score far below the target 1.503.
The code below searches several families (geometric, stretched‑exponential,
power‑law × exponential) on a few lengths, keeps the best candidate by the
exact evaluator and finally returns it.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a∗a using an FFT of power‑of‑two length.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)
    return conv[: 2 * n - 1]


def _estimate_score(a: np.ndarray) -> float:
    """
    Fast homogenous score 2·n·max(a∗a)/ (∑a)².
    Identical to evaluate_sequence for non‑negative inputs that stay within
    the clamping limits.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def _gen_exponential(c: float, n: int) -> np.ndarray:
    """geometric shape a_i = exp(-c·i/n)  (i = 0,…,n‑1)."""
    i = np.arange(n, dtype=float)
    return np.exp(-c * i / n)


def _gen_stretched_exponential(c: float, p: float, n: int) -> np.ndarray:
    """a_i = exp(-c·(i/n)^p)."""
    t = np.arange(n, dtype=float) / n
    return np.exp(-c * (t ** p))


def _gen_power_exponential(c: float, alpha: float, n: int) -> np.ndarray:
    """a_i = i^{-α}·exp(-c·i/n)  (i starts at 1)."""
    i = np.arange(1, n + 1, dtype=float)
    return (i ** (-alpha)) * np.exp(-c * (i - 1) / n)


def _refine_subgradient(start_seq: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 5000) -> np.ndarray:
    """
    A short exponentiated‑gradient refinement.
    The sequence is already normalised (∑=1).  The routine stops
    when the time budget expires.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _estimate_score(best)

    start_eta, end_eta = 0.5, 0.001
    it = 0
    while time.time() < deadline and it < max_iter:
        # convolution and active index
        conv = _conv_fft(a)
        k_star = int(conv.argmax())
        # sub‑gradient of the active term
        g = np.zeros_like(a)
        lo = max(0, k_star - (n - 1))
        hi = min(k_star, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k_star - i
        g[i] = 2.0 * a[j]

        gmax = g.max()
        if gmax == 0.0:
            break

        # linear schedule for η
        elapsed = time.time() - (deadline - 1000.0)
        total = deadline - (deadline - 1000.0)
        eta = start_eta * (1.0 - elapsed / total) + end_eta * (elapsed / total)

        factor = np.exp(-eta * g / gmax)   # multiplicative update
        a = a * factor
        a = a / a.sum()                    # simplex projection

        # occasional random shuffle to escape plateaus
        if rng.random() < 0.01:
            s = rng.integers(0, n)
            l = rng.integers(1, min(256, n - s) + 1)
            a[s:s + l] = a[s:s + l][::-1]

        cur_score = _estimate_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        it += 1

    return best

    deadline = start_time + budget_s - 0.1   # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # Return a plain Python list of floats (non‑negative, any length)
    return [float(x) for x in best_seq]
