"""
Local hill‑climbing optimizer for the self‑convolution bound.

The algorithm works with a non‑negative vector `a` whose ℓ₁‑norm is fixed to
√(2·n).  Starting from a good initial construction (the global variable
`height_sequence_1` when present, otherwise a few random starts) it performs
many tiny “mass‑transfer” moves: a small amount ε is taken from position i
and added to position j.  The effect on the convolution a∗a can be updated
in O(n) time by adjusting only the two affected length‑n slices and a few
individual entries, so a large number of proposals can be evaluated within
the 1000 s budget.

A simulated‑annealing acceptance rule (always accept improvements, sometimes
accept degradations early on) helps escape local minima.  When the search
stagnates a random perturbation (mix of the current best solution and a fresh
exponential vector) is applied and the process continues.

The best sequence found is returned as a plain Python list.
"""

import time
import math
import random
import numpy as np

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all entries equal
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (height) vector that minimises
    `evaluate_sequence`.  The routine runs a fast hill‑climbing local search
    based on pairwise mass transfers and simulated‑annealing acceptance.
    The best sequence encountered is returned.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------ #
    # Problem size ------------------------------------------------------ #
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base = np.asarray(height_sequence_1, dtype=float)
    else:
        # a modest size makes each iteration cheap while still offering enough freedom
        n = 1200
        base = np.full(n, 1.0, dtype=float)

    target_sum = math.sqrt(2.0 * n)          # ℓ₁‑norm constraint  ∑ a = √(2n)

    # Normalise the base vector
    base = base / base.sum() * target_sum

    # ------------------------------------------------------------------ #
    # Assemble a few starters and keep the best one as the initial iterate #
    starters = []

    # 1) reuse the known construction (if any)
    starters.append(base.copy())

    # 2) uniform vector
    starters.append(np.full(n, target_sum / n, dtype=float))

    # 3) random exponential vector
    rand_exp = np.random.exponential(scale=1.0, size=n)
    rand_exp = rand_exp / rand_exp.sum() * target_sum
    starters.append(rand_exp)

    # quick random search (uses only a tiny fraction of the total budget)
    while time.time() < start_time + 0.02 * total_time:
        cand = np.random.exponential(1.0, size=n)
        cand = cand / cand.sum() * target_sum
        starters.append(cand)

    best_seq = None
    best_score = np.inf
    for cand in starters:
        conv = np.convolve(cand, cand)
        score = float(conv.max())  # because ∑a = √(2n) ⇒ evaluate_sequence = max(conv)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    # ------------------------------------------------------------------ #
    # Hill‑climbing parameters ------------------------------------------ #
    a = best_seq.copy()                     # current iterate
    conv = np.convolve(a, a)                # current self‑convolution
    cur_max = float(conv.max())
    best_score = cur_max
    best_seq = a.copy()

    L = 2 * n - 1                           # length of the convolution
    eps_factor = 0.20                       # initial relative step size
    min_eps_factor = 0.001
    decay_rate = 0.9999                     # slow decay of eps_factor
    stagnation_counter = 0
    stagnation_limit = 5000
    iteration = 0

    # simulated‑annealing temperature schedule
    T_start = cur_max * 0.02                 # start temperature (2 % of the current value)
    T_end = 1e-7

    # ------------------------------------------------------------------ #
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time          # 0 → 1 over the run
        temperature = T_start * (1.0 - frac) + T_end * frac

        # ---- choose a move -------------------------------------------------
        # pick a source index with non‑zero mass
        i = random.randrange(n)
        if a[i] <= 1e-12:
            continue
        # pick a distinct target index
        j = random.randrange(n - 1)
        if j >= i:
            j += 1

        # amount to transfer (never exceeds a[i])
        max_move = eps_factor * a[i]
        if max_move <= 0.0:
            continue
        eps = random.random() * max_move

        # ---- store the part of conv that will change ----------------------
        slice_i = slice(i, i + n)            # a[i] contributes to conv[i … i+n‑1]
        slice_j = slice(j, j + n)

        old_i = conv[slice_i].copy()
        old_j = conv[slice_j].copy()

        # ---- apply the incremental update ----------------------------------
        # contribution of the linear term  -2·eps·a on slice i, +2·eps·a on slice j
        conv[slice_i] -= 2.0 * eps * a
        conv[slice_j] += 2.0 * eps * a

        # quadratic correction (eps²) – these positions are already inside the
        # modified slices, so restoring the slices later will undo them as well.
        conv[2 * i] += eps * eps
        conv[2 * j] += eps * eps
        conv[i + j] -= 2.0 * eps * eps

        new_max = float(conv.max())

        # ---- acceptance test (simulated annealing) -----------------------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            # possibly accept a worsening move early on
            if temperature > 0.0:
                prob = math.exp(-(new_max - cur_max) / temperature)
                if random.random() < prob:
                    accept = True

        if accept:
            # keep the modification
            a[i] -= eps
            a[j] += eps

            # numerical safety – clip tiny negatives that may appear due to floating error
            if a[i] < 0.0:
                a[i] = 0.0
            cur_max = new_max
            stagnation_counter = 0

            # keep record of the overall best solution
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[{iteration:6d}] New best max(conv) = {best_score:.6f} "
                      f"(t={elapsed:.1f}s, eps_factor={eps_factor:.5f})")

            # gradually shrink the step size
            eps_factor = max(eps_factor * decay_rate, min_eps_factor)

        else:
            # revert the convolution update (the vector a stays unchanged)
            conv[slice_i] = old_i
            conv[slice_j] = old_j
            stagnation_counter += 1

            # if we have been stuck for a while, inject a random perturbation
            if stagnation_counter >= stagnation_limit:
                # mixture of the current best and fresh noise
                noise = np.random.exponential(scale=1.0, size=n)
                noise = noise / noise.sum() * target_sum
                mix = random.random()
                a = mix * best_seq + (1.0 - mix) * noise
                a = np.maximum(a, 0.0)
                a = _simplex_projection(a, target_sum)
                conv = np.convolve(a, a)
                cur_max = float(conv.max())
                stagnation_counter = 0
                eps_factor = 0.20   # reset step size

        iteration += 1

        # occasional full recomputation to avoid accumulated drift
        if iteration % 10000 == 0:
            conv = np.convolve(a, a)
            cur_max = float(conv.max())

    # ------------------------------------------------------------------ #
    # Return the best sequence found (as a plain list) ------------------- #
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, target_sum / n)
    return best_seq.tolist()
