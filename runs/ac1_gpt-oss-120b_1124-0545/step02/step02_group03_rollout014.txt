def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = kwargs.get('n', 2000)

    # Initialize a uniform distribution.
    a = np.full(n, 1.0/n)

    # Helper for building convolution matrix A
    from scipy.sparse import diags

    def build_conv_matrix(a_vec):
        # returns scipy.sparse.csr_matrix shape (2n-1, n)
        # For offsets 0..n-1, each diagonal of length n contains a_i.
        # Use repeated a_i on diagonal.
        data = [np.full(n, a_vec[i]) for i in range(n)]
        offsets = np.arange(n)
        A = diags(data, offsets, shape=(2*n-1, n), format='csr')
        return A

    # compute conv using FFT for speed to get max.
    sizeFFT = 1 << (2*n-2).bit_length()
    # define conv compute
    def max_conv(a_vec):
        fa = np.fft.rfft(a_vec, n=sizeFFT)
        conv_full = np.fft.irfft(fa * fa, n=sizeFFT)[:2*n-1]
        return conv_full.max()

    # main loop: alternate between LP step and subgradient steps.
    best_score = float('inf')
    best_seq = a.copy()
    it = 0
    # We will perform at most some iteration count; each iteration:
    while time.time() < deadline:
        # compute M
        M = max_conv(a)
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{it}] new best = {best_score:.6f}")

        # Choose whether to perform LP (deterministic) or subgradient.
        # For example, perform LP every 10 iterations.
        if it % 10 == 0:
            # LP solve
            A = build_conv_matrix(a)
            # c for minimization = -ones to maximize sum(b)
            c = -np.ones(n)
            ub = np.full(2*n-1, M)
            res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs', options={'presolve': True})
            if not res.success:
                print(f"LP failed at iteration {it}: {res.message}")
                # fallback to small random perturbation
                a = _random_transfer(a, 0.01)
                it += 1
                continue
            b = res.x
            sum_b = b.sum()
            if sum_b > 0:
                # scale b to unit sum
                g = b / sum_b
                # convex combination update
                t = 0.5
                a = (1 - t) * a + t * g
                # renormalise to sum 1 (should already be 1)
                a = a / a.sum()
        else:
            # small random transfer move to explore
            a = _random_transfer(a, 0.02)

        it += 1

    # after loop return best_seq
    total = time.time() - start
    print(f"[done] best score={best_score:.6f}, time={total:.1f}s, it={it}")
    return best_seq.tolist()

"""
Hybrid Frank‑Wolfe / sub‑gradient optimiser for the max‑autocorrelation
objective.

The objective (with the sum of the coefficients normalised to 1) is

    obj(a) = 2·n·max(conv(a,a))

where conv is the linear convolution of the vector with itself.
The algorithm works in two phases:

1.  A short projected sub‑gradient descent (with occasional random
    “mass‑transfer” moves) quickly drives the solution away from the
    uniform distribution and finds a reasonably low‑peak point.

2.  A Frank‑Wolfe style iteration solves the linear programme

        maximise   Σ b_j
        subject to  T(a)·b ≤ M,   b ≥ 0

    where T(a) is the (2n‑1)×n convolution matrix built from the
    current iterate a and M = max(conv(a,a)).  The optimiser
    scales the LP solution to have unit‑sum and moves a towards it by a
    fixed mixing factor (t≈0.5).  Repeating this step monotonically lowers
    the peak M.

The whole procedure respects the 1000 s budget, uses only NumPy/SciPy
and returns the best sequence found (as a plain list of Python floats).
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level helpers (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of two real vectors via FFT of length *size*."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _max_convolution(a: np.ndarray, fft_size: int) -> np.ndarray:
    """Full linear convolution a*a, returned as a dense array."""
    return _conv_fft(a, a, fft_size)


def _subgradient_step(a: np.ndarray,
                      conv_full: np.ndarray,
                      mask: np.ndarray,
                      size2: int,
                      step: float) -> np.ndarray:
    """
    Perform a projected sub‑gradient step for max(conv(a,a)).
    *conv_full* is a*a, *mask* marks (approximately) the maximal entries.
    """
    n = a.shape[0]
    # reversed a
    a_rev = a[::-1]
    # convolution a_rev * mask (length 3n‑2)
    full = _conv_fft(a_rev, mask, size2)
    # extract the central n entries – the true sub‑gradient components
    g = full[n - 1: n - 1 + n]
    # sub‑gradient of the max entry
    d = -2.0 * g
    # make it a feasible direction (Σd = 0)
    d -= d.mean()
    # take step and re‑project onto the probability simplex
    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:            # fallback – uniform distribution
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Move a small random amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise to keep Σa = 1
    a_new /= a_new.sum()
    return a_new

# Main entry point required by the evaluation harness

    """

    start = time.time()

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – Frank‑Wolfe style LP refinement
    # ------------------------------------------------------------------
    t_mix = 0.5                     # mixing factor (1‑t)·a + t·g
    lp_iter = 0
    max_lp_iters = 40              # enough to see a clear trend

        # Linear programme: maximise Σb  ⇔  minimise −Σb
        c = -np.ones(n)
        ub = np.full(2 * n - 1, M)
        res = opt.linprog(c, A_ub=A, b_ub=ub,

    print(f"[done] best_score={final_score:.6f}, time={total:.1f}s, "
          f"sg_iters={it}, lp_iters={lp_iter}")

    # Return as a plain Python list of floats
    return best_seq.tolist()
