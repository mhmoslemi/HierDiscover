"""
Propose a low‑value step function (non‑negative sequence).

The algorithm is a smoothed‐max gradient descent on the self‑convolution.
We iteratively:

1. Compute the full convolution c = a * a (FFT, O(n log n)).
2. Form a soft‑max weight vector w that concentrates on the largest
   convolution entries (temperature μ, gradually increased).
3. The gradient of the smoothed max is 2·conv(a_rev, w).  We take a
   momentum step opposite to this gradient.
4. The iterate is projected back onto the probability simplex
   (non‑negative, sum = 1) using an Euclidean simplex projection.
5. Occasionally a tiny random “mass‑transfer” move is applied to escape
   plateaus.  If no improvement is seen for a while we restart from a
   random distribution.

The method improves upon the plain sub‑gradient approach and usually
finds a sequence with score well below 1.55 within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of real vectors u and v using FFT of given length."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    # keep only the true linear convolution part
    return w[: len(u) + len(v) - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Move a small fraction of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return _project_simplex(a_new)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that minimizes
        2·n·max(conv(a,a)) / (sum(a)**2)

    Parameters
    ----------
    seed : int
        Random seed.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Optional ``n`` (length of the sequence, default 6000).

    Returns
    -------
    list[float]
        The best sequence found (probability vector, sum≈1).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # initialise a starting distribution
    # --------------------------------------------------------------
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None:
        a = np.asarray(seq0, dtype=float)
        if a.size != n:
            # linear interpolation to required length

            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(x_new, x_old, a)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.random.rand(n)
        a = _project_simplex(a)
    else:
        a = np.random.rand(n)
        a = _project_simplex(a)

    # --------------------------------------------------------------
    size_conv = _next_pow2(2 * n - 1)   # for a * a
    size_grad = _next_pow2(3 * n - 2)   # for a_rev * weight vector

    # --------------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------------
    eta0 = 0.8               # base step size
    beta = 0.9               # momentum coefficient
    mu = 0.5                 # initial soft‑max temperature
    mu_max = 20.0
    mu_inc = 1.001           # multiplicative increase per iteration
    rand_prob = 0.03         # probability of random mass‑transfer move
    rand_scale = 0.02        # scale of random transfer
    stagnation_time = 30.0   # seconds without improvement → restart
    last_improve = start
    it = 0

    conv_full = _conv_fft(a, a, size_conv)

    # --------------------------------------------------------------
    while time.time() < deadline:

        conv_full = _conv_fft(a, a, size_conv)

            best_seq = a.copy()
            last_improve = time.time()

        if np.random.rand() < rand_prob:
            # cheap random mass‑transfer (exploration)
            a = _random_transfer(a, rand_scale)
        else:
            # ----- smoothed‑max gradient step -----
            # soft‑max weights on convolution entries
            # shift weights so the maximum is zero to avoid overflow
            max_c = max_b
            exp_weights = np.exp(mu * (conv_full - max_c))
            w = exp_weights / exp_weights.sum()          # shape (2n‑1,)

            # gradient = 2 * (a_rev * w)   (full conv length 3n‑2)
            a_rev = a[::-1]
            grad_full = _conv_fft(a_rev, w, size_grad)
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # central n components

            # momentum update
            momentum = beta * momentum + (1.0 - beta) * grad

            # step size (diminishing)
            step = eta0 / np.sqrt(it + 1.0)

            # projected gradient descent step (move opposite to gradient)
            a = a - step * momentum
            a = _project_simplex(a)

            # slowly raise the temperature μ
            if mu < mu_max:
                mu = min(mu_max, mu * mu_inc)

        # 4) stagnation restart
        if time.time() - last_improve > stagnation_time:
            a = np.random.rand(n)
            a = _project_simplex(a)
            momentum.fill(0.0)
            mu = 0.5
            print(f"[{it:7d}] restart after stagnation")

        it += 1

    # --------------------------------------------------------------
    # final report
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    # Ensure the returned list respects the evaluation function's safety checks
    final_seq = best_seq.astype(float).tolist()
    return final_seq
