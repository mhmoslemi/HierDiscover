"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Analytic grid search** – a small parametric family
   a_i ∝ ((i+δ)*(n‑1‑i+δ))^{-p} is scanned over a grid of
   (p,δ).  The best grid point is used as the start of the
   optimisation.

2. **Smooth‑max gradient descent** – we minimise a smooth
   approximation of the max‑convolution peak:
   f(a)=log‑sum‑exp(α·conv(a,a)).
   Its gradient is
       ∇f_j = 2·∑_k w_k a_{k‑j},   w_k = softmax(α·conv_k).
   An exponentiated‑gradient step keeps the vector on the
   probability simplex.  The smoothing parameter α is increased
   gradually (continuation) while the step size is decreased.

3. **Exploration & restart** – occasional random mass transfers
   (to escape tiny local minima) and a restart with jitter when
   stagnation is detected.

The routine runs for the allotted time (default 1000 s) and
returns the best sequence found (a plain list of floats).  All
numbers are non‑negative and sum to 1, which is sufficient because
the evaluation function is scale‑invariant.
"""

def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale‑invariant) for a non‑negative array a."""
    s = a.sum()
    if s < 1e-12:  # should never happen
        return np.inf
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2.0 * len(a) * max_b / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def generate_grid_candidates(n: int) -> list[np.ndarray]:
    """Scan the (p,δ) grid of product‑type shapes and return the best few."""
    i = np.arange(n, dtype=np.float64)
    # use (i+δ)*(n‑1‑i+δ) as the denominator

    deltas = np.linspace(0.0, 2.0, 21)       # offset δ
    candidates = []

    best_arr = None

    for p in ps:

            # avoid division by zero (only possible if delta=0 and i=0 or i=n-1)

    scores = [compute_score(c) for c in candidates]
    top_idx = np.argsort(scores)[:10]
