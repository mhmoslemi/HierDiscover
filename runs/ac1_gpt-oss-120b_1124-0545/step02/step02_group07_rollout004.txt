"""
Propose a step‑function that minimises the bound

    score = 2·n·max(a∗a) / (∑a)²

The algorithm is a *smooth‑max* Adam optimiser:

* The sequence is always kept normalised (∑a = 1) and non‑negative.
* The (non‑smooth) objective L(a)=maxₖ(a∗a)ₖ is approximated by the
  soft‑max

        L_α(a) = (1/α)·log( Σₖ exp(α·(a∗a)ₖ) )

  for a temperature‑parameter α that slowly grows from a small value
  (α≈5) to a larger one (α≈45).  This gives a differentiable surrogate
  that concentrates on the active convolution index while still
  providing a useful gradient for all components.
* Gradients are obtained analytically:
        ∂L_α/∂aᵢ = 2· Σₖ wₖ·a_{k−i}
  where wₖ are the soft‑max weights.
  The sum is a convolution of the weight‑vector w (length 2·n−1) with
  the reversed a; both convolutions are performed by FFT for speed.
* Adam (β₁=0.9, β₂=0.999) updates the *log* of the coefficients
  (xᵢ=log aᵢ) which automatically respects positivity.
* The optimiser runs until the time budget (≈1000 s) expires or a
  target score (≤ 1.5) is reached.  The best exact score seen
  (computed with the official `evaluate_sequence`) is returned.

The optimiser starts from a near‑geometric decay, whose decay‑parameter
c is selected by a quick grid‑search.  If the global variable
`height_sequence_1` exists it is used as a warm‑start instead.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using a power‑of‑two FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)
    return conv[: 2 * n - 1]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weights wₖ = exp(α·convₖ)/Σ exp(α·conv)  in a
    numerically stable way.
    """
    max_c = conv.max()
    # subtract the max to avoid overflow
    exp_vals = np.exp(alpha * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    return w


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of L_α(a) w.r.t. a.
    gᵢ = 2· Σₖ wₖ·a_{k−i}.
    Implemented as a convolution of w (length 2n−1) with a_rev.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    # length needed for linear convolution
    N = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    fft_w = np.fft.rfft(w, n=N)
    fft_a_rev = np.fft.rfft(a_rev, n=N)
    conv_full = np.fft.irfft(fft_w * fft_a_rev, n=N)
    # indices n‑1 … n‑1+(n‑1) contain the required n entries
    start = n - 1
    g = 2.0 * conv_full[start:start + n]
    return g


def _initial_exponential(n: int, rng: np.random.Generator):
    """
    Scan a modest grid of decay constants c and return the exponential
    shape with the lowest *approximate* score 2·n·max(a∗a).
    """
    best_a = None
    best_score = np.inf
    best_c = None
    # coarse grid, cheap enough to evaluate many candidates
    for c in np.linspace(0.5, 5.0, 46):
        r = np.exp(-c / n)
        # geometric series – normalise to sum = 1
        a = (1 - r) * (r ** np.arange(n, dtype=float))
        a = a / a.sum()
        conv = _conv_fft(a)
        approx = 2.0 * n * conv.max()
        if approx < best_score:
            best_score = approx
            best_a = a
            best_c = c
    # a tiny random perturbation to break exact symmetries
    best_a += 1e-12 * rng.random(n)
    best_a /= best_a.sum()
    return best_a, best_score, best_c


# ----------------------------------------------------------------------
# Core optimiser (Adam on the smooth‑max objective)
# ----------------------------------------------------------------------


def _adam_optimize(start_a: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   target_score: float = 1.5,
                   alpha_start: float = 5.0,
                   alpha_end: float = 45.0,
                   eta_start: float = 0.2,
                   eta_end: float = 0.001) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best *exact* sequence discovered before ``deadline``.
    """
    a = start_a.copy()
    n = a.shape[0]

    # log‑space representation – guarantees positivity
    x = np.log(a + 1e-30)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # bookkeeping
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    it = 0
    total_time = deadline - (deadline - 1000.0)   # 1000 s (used only for progress)

    while time.time() < deadline:
        elapsed = time.time() - (deadline - 1000.0)
        progress = elapsed / total_time

        # ---- schedules -------------------------------------------------
        alpha = alpha_start + progress * (alpha_end - alpha_start)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # ---- forward pass (objective) ----------------------------------
        conv = _conv_fft(a)
        approx_score = 2.0 * n * conv.max()      # exact because Σa = 1
        # occasional call to the official evaluator (cheap when improved)
        if approx_score < best_score * 0.995:    # a small margin to avoid noise
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(f"[{elapsed:.1f}s] New best exact score={best_score:.6f}")
                if best_score <= target_score:
                    break

        # ---- gradient of the smooth‑max --------------------------------
        w = _softmax_weights(conv, alpha)
        g = _gradient_softmax(a, w)

        # ---- Adam update in log‑space ----------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---- back‑to the simplex ---------------------------------------
        a = np.exp(x)
        s = a.sum()
        if s == 0.0:                     # protect against numerical underflow
            a = best_a.copy()
            x = np.log(a + 1e-30)
            continue
        a /= s

        # ---- exploration noise (helps to escape plateaus) ------------
        if rng.random() < 0.01:
            noise = rng.normal(scale=1e-3, size=n)
            a += noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            x = np.log(a + 1e-30)

    return best_a
