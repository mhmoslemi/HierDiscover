"""
Optimisation of the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The routine consists of three phases:

1. **Initialisation** – Load a known good construction (global
   ``height_sequence_1``) or fall back to the classic arcsine shape.
   The sequence is normalised to a probability simplex.

2. **Smooth‑max projected gradient descent** – Replace the non‑smooth
   ``max`` with an entropy‑smoothed approximation
   ``α·log∑exp(conv/α)``.  The gradient of this surrogate can be written
   as a convolution, so each step costs only two FFT‑type convolutions.
   After a gradient move we project back onto the simplex.  The
   smoothing parameter ``α`` and the step size are annealed linearly.

3. **Linear‑programming “mass‑pumping”** – With the current maximum
   convolution value ``M`` we solve the LP

        maximise   1ᵀb
        subject to  a∗b ≤ M   (component‑wise)
                    b ≥ 0

   The LP finds a direction that can carry *more* mass without raising
   the peak.  Mixing the current vector with the LP solution (with a
   small mixing factor ``τ``) strictly reduces the peak after a few
   repetitions.  A few final gradient steps polish the result.

The whole search respects a 1000 s budget.  The best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative probability vector.
    The vector is assumed to sum to 1; the function returns
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # convolution (zero‑padded)
    c = np.convolve(a, a)
    max_c = c.max()
    return float(2.0 * n * max_c)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def build_toeplitz_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n convolution matrix T such that
        T @ b = a ∗ b .
    Row k (0‑based) contains the shifted copy of a that contributes to
    the k‑th entry of the convolution.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    idx = np.arange(rows)[:, None] - np.arange(n)[None, :]
    T = np.where((idx >= 0) & (idx < n), a[idx], 0.0)
    return T


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``2·n·max(conv(a,a))/(sum(a)**2)``.  The algorithm combines a
    smooth‑max projected gradient descent with an LP‑based “mass pumping”
    phase.  The best sequence found before the time budget expires is
    returned.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the external harness
    deadline = start + budget_s - 0.5

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1 – initialise from a good known construction
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, list) and len(init_seq) == n:
        a = np.array(init_seq, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        print("[init] loaded height_sequence_1")
    else:
        # arcsine / inverse‑sqrt product – a classic strong start
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a = a / a.sum()
        print("[init] built arcsine start")

    best = a.copy()
    best_score = compute_score(best)
    print(f"[init] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – smooth‑max projected gradient descent (first ~30 % of the budget)
    # ------------------------------------------------------------------
    # schedule parameters
    grad_time_limit = deadline - 30.0          # keep ~30 s for the LP phase
    init_alpha = 5e-2      # smoothing at start
    final_alpha = 5e-6    # final smoothing (close to true max)
    init_step = 5e-1
    final_step = 1e-4

    while time.time() < grad_time_limit:
        elapsed = time.time() - start
        prog = (elapsed) / (budget_s - 30.0)               # 0 → 1

        alpha = init_alpha * (1.0 - prog) + final_alpha * prog
        step = init_step * (1.0 - prog) + final_step * prog

        # ----- smooth‑max gradient -------------------------------------------------
        c = np.convolve(a, a)                # full convolution (2n‑1 entries)
        w = np.exp(c / alpha)
        w /= w.sum()                         # soft‑max weights, sum = 1

        a_rev = a[::-1]
        # gradient = 2·(w ∗ a_rev) restricted to the central n entries
        g_full = np.convolve(w, a_rev)       # length 3n‑2
        grad = 2.0 * g_full[n - 1:n - 1 + n]  # shape (n,)

        # ----- projected step -------------------------------------------------------
        cand = a - step * grad
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)

        cand_score = compute_score(cand)

        # keep the global best
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            print(f"[grad] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # simple Metropolis‑style acceptance (occasional uphill moves)
        if cand_score < compute_score(a) or np.random.rand() < 0.01:
            a = cand

        # occasional random mass‑swap to help escape tiny traps
        if np.random.rand() < 0.02:
            a = random_move(a, 0.05)

    # ------------------------------------------------------------------
    # 3 – LP “mass‑pumping” phase (use remaining time)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:  # pragma: no cover
        print("[lp] SciPy not available – skipping LP phase")
        # fall back to a few extra gradient steps and finish
        while time.time() < deadline:
            # a very cheap copy of the gradient step with tiny step size
            c = np.convolve(a, a)
            w = np.exp(c / final_alpha)
            w /= w.sum()
            grad = 2.0 * np.convolve(w, a[::-1])[n - 1:n - 1 + n]
            a = project_simplex(a - final_step * grad)
            best_score = min(best_score, compute_score(a))
        total = time.time() - start
        print(f"[done] time={total:.1f}s, best score={best_score:.6f}")
        return best.tolist()

    # parameters for the LP loop
    tau_candidates = (0.5, 0.3, 0.15, 0.07, 0.03)  # decreasing mixing factors
    last_improve = time.time()

    while time.time() < deadline:
        # current peak M
        conv_a = np.convolve(a, a)
        M = conv_a.max()

        # ----- build the convolution matrix (a ∗ b ≤ M) -------------------------
        T = build_toeplitz_matrix(a)           # shape (2n‑1, n)

        # ----- linear programme: maximise sum(b) subject to a∗b ≤ M -------------
        lp_res = linprog(c=-np.ones(n),
                         A_ub=T,
                         b_ub=np.full(2 * n - 1, M),
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not lp_res.success:
            print("[lp] LP failed – exiting LP phase")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b == 0:
            print("[lp] LP returned zero mass – exiting")
            break

        g = b / sum_b                         # normalised LP solution

        improved = False
        for tau in tau_candidates:
            cand = (1.0 - tau) * a + tau * g
            cand = project_simplex(cand)
            cand_score = compute_score(cand)

            if cand_score < best_score - 1e-9:
                best_score = cand_score
                best = cand.copy()
                a = cand
                improved = True
                last_improve = time.time()
                elapsed = last_improve - start
                print(f"[lp] new best = {best_score:.6f} (τ={tau:.3f}, t={elapsed:.1f}s)")
                break

        if not improved:
            # no improvement – try a very small mixing step to still move a little
            tau = 0.01
            a = project_simplex((1.0 - tau) * a + tau * g)

        # occasional random jitter to avoid stagnation
        if np.random.rand() < 0.1:
            a = random_move(a, 0.01)

        # restart if we have been stuck for >20 s
        if time.time() - last_improve > 20.0:
            jitter = np.random.rand(n) * 1e-4
            a = project_simplex(best + jitter)
            last_improve = time.time()
            print("[lp] restart after stagnation")

    # ------------------------------------------------------------------
    # 4 – final polishing (a few cheap gradient steps)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        c = np.convolve(best, best)
        w = np.exp(c / final_alpha)
        w /= w.sum()
        grad = 2.0 * np.convolve(w, best[::-1])[n - 1:n - 1 + n]
        cand = best - final_step * grad
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            print(f"[final] new best = {best_score:.6f}")
        else:
            break

    total = time.time() - start
    print(f"[done] iterations completed, best score = {best_score:.6f}, time = {total:.1f}s")
    return best.tolist()
