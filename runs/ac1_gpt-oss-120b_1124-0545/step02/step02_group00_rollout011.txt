"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Analytic seeding** – scan a few simple parametric families
   (product‑power, sum‑power, uniform, and the known best run) and
   pick the best normalised vector (∑a_i = 1).

2. **Smoothed projected gradient descent** – replace the non‑smooth
   “max‑convolution’’ by a log‑sum‑exp smoothing.
   For the current vector a we compute
        w_k ∝ exp(μ·conv_k)   (μ starts small and grows),
   then the gradient of the smooth objective is
        g_i = 2·∑_k w_k·a_{k−i}.
   After a step a←a−η·g we project back onto the probability simplex.
   Temperature‑controlled Metropolis acceptance allows occasional uphill steps.

3. **Fine‑tuning** – a short run of classic sub‑gradient steps that
   target the single maximal convolution entry, with a very small step size.

The routine respects a 1000 s budget, logs progress, and returns the best
sequence found (as a plain list of floats).  It works for any length *n*
(the default is 2000) and uses only NumPy – no external solvers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale‑invariant) for a non‑negative array a."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = np.convolve(a, a)          # linear convolution (zero‑padded)
    max_b = b.max()
    # When we keep ∑a_i = 1 the denominator is 1, and the objective is 2·n·max_b.
    return float(2.0 * len(a) * max_b / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from a random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def smooth_gradient_move(a: np.ndarray, step: float, mu: float) -> np.ndarray:
    """
    One projected gradient step on the smoothed objective
        f_μ(a) = (1/μ)·log(∑_k exp(μ·conv_k(a))).

    Returns the projected new vector.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length L = 2n-1
    L = conv.shape[0]                            # = 2n-1
    # stable softmax over conv entries
    max_c = conv.max()
    w = np.exp(mu * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient g_i = 2 * sum_k w_k * a_{k-i}
    # We compute the cross‑correlation via a reversed convolution.
    w_rev = w[::-1]
    # conv_rev[t] = sum_k w_rev[k] * a[t-k] = sum_k w_{L-1-k} * a[t-k]
    conv_rev = np.convolve(w_rev, a, mode='full')   # length L + n - 1 = 3n-2
    # Desired slice: conv_rev[(L-1)-i] for i = 0..n-1
    idx = (L - 1) - np.arange(n, dtype=int)          # (2n-2) - i
    g = 2.0 * conv_rev[idx]

    a_new = a - step * g
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_simplex(a_new)
    return a_new


def max_subgradient_move(a: np.ndarray, step: float) -> np.ndarray:
    """
    Classic sub‑gradient step that follows the (single) maximal convolution entry.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    g = np.zeros_like(a)
    # sub‑gradient: g_i = 2 * a_{k_max-i} for the supported indices
    g[idx] = 2.0 * a[k_max - idx]
    a_new = a - step * g
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_simplex(a_new)
    return a_new


def initialise_candidates(n: int) -> list[np.ndarray]:
    """
    Produce several analytically motivated candidates (all normalised to sum=1).
    """
    cand = []

    # uniform
    cand.append(np.ones(n) / n)

    # product‑power family: a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}
    i = np.arange(n, dtype=float)
    for p in np.linspace(0.30, 0.70, 9):          # includes the classic 0.5
        prod = (i + 0.5) * (n - i - 0.5)
        a = prod ** (-p)
        cand.append(a / a.sum())

    # sum‑power family: a_i ∝ (i+1)^{-p} + (n-i)^{-p}
    for p in (0.3, 0.5, 0.7, 1.0):
        a = (i + 1) ** (-p) + (n - i) ** (-p)
        cand.append(a / a.sum())

    # pure arcsine (p=0.5, no 0.5 offset)
    prod0 = (i + 1) * (n - i)
    cand.append((prod0 ** (-0.5)) / (prod0 ** (-0.5)).sum())

    # try to import the global best if it exists (height_sequence_1)
    # and reshape / truncate it to length n
    global_vars = globals()
    if "height_sequence_1" in global_vars:
        hs = np.asarray(global_vars["height_sequence_1"], dtype=float)
        if hs.size >= n:
            a = hs[:n].copy()
        else:
            # pad with tiny mass if the stored vector is shorter
            a = np.zeros(n, dtype=float)
            a[:hs.size] = hs
        a = np.maximum(a, 0.0)
        a = project_simplex(a)
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Main entry point (required by the evaluation harness)
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation value.
    The search runs for up to *budget_s* seconds (default 1000) and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1 – Analytic seeding
    # ------------------------------------------------------------------
    candidates = initialise_candidates(n)
    best_seq = None
    best_score = np.inf
    print("[init] evaluating analytic candidates")
    for idx, a in enumerate(candidates):
        sc = compute_score(a)
        print(f"  cand {idx:02d}  score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    print(f"[init] best seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – Smoothed projected gradient descent
    # ------------------------------------------------------------------
    a = best_seq.copy()
    iteration = 0
    last_improve = start_time

    # annealing / schedule parameters
    init_temp = 1.0
    final_temp = 0.001
    init_step = 0.4
    final_step = 5e-5
    mu_start = 0.1          # start smoothing factor
    mu_end = 12.0           # end smoothing factor (focuses on max)
    random_prob = 0.12      # probability of a cheap random move

    while time.time() < deadline:
        elapsed = time.time() - start_time
        prog = elapsed / (budget_s - 10.0)          # in [0,1] (smooth)
        prog = min(max(prog, 0.0), 1.0)

        # schedules (linear)
        temp = init_temp * (1.0 - prog) + final_temp * prog
        step = init_step * (1.0 - prog) + final_step * prog
        mu = mu_start * (1.0 - prog) + mu_end * prog

        # decide move type
        if np.random.rand() < random_prob:
            cand = random_move(a, step * 0.5)
        else:
            cand = smooth_gradient_move(a, step, mu)

        cand_score = compute_score(cand)

        # Metropolis‑style acceptance
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            # probability to accept a worse move
            prob = np.exp(-(cand_score - best_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f}  (t={elapsed:.1f}s)")

        # occasional restart if stagnating
        if iteration % 30000 == 0 and (time.time() - last_improve) > 30.0:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            print(f"[{iteration:07d}] restart due to stagnation")

        iteration += 1

    # ------------------------------------------------------------------
    # 3 – Fine‑tuning with pure max‑subgradient
    # ------------------------------------------------------------------
    # run a short deterministic refinement (no randomness) using the exact
    # sub‑gradient of the max‑convolution.  This often squeezes a few
    # extra hundredths out of the objective.
    print("[fine‑tune] starting max‑subgradient refinement")
    fine_iters = 2000
    fine_step_start = 4e-2
    fine_step_end = 1e-5
    for it in range(fine_iters):
        prog = it / fine_iters
        step = fine_step_start * (1.0 - prog) + fine_step_end * prog
        cand = max_subgradient_move(best_seq, step)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[fine {it:04d}] improved = {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    # Convert to plain Python list of floats for the harness
    return best_seq.tolist()
