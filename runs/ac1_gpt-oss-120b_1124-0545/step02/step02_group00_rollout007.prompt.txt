def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.

    n = v.shape[0]
    u = np.sort(v)[::-1]                # sorted descending

    rho_arr = u - cssv / (np.arange(n) + 1.0)

        theta = 0.0
