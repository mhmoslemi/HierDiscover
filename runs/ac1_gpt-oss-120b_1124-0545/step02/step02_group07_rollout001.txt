def _project_onto_simplex(v):
    # from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

   The gradient is obtained by a single FFT‑convolution.
   Using an Adam optimiser we perform many projected‑gradient steps,
   gradually increasing β (from 0.2 to ≈30) so that we first follow a
   very smooth landscape and then hone in on the true max.

   After each step the vector is projected onto the probability simplex
   (non‑negative, Σ a_i = 1) – this automatically respects all security
   checks in `evaluate_sequence`.

3. **Random‑restart polishing** – after the main run we repeatedly
   perturb the current best vector with a tiny Gaussian noise,
   re‑run a short Adam optimisation and keep any improvement.

All the work respects the global time budget (≈1000 s).  The best
sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
from math import log, exp, sqrt

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0 , sum x_i = 1}
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns array of length 2·len(a)‑1 (the exact convolution).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)
    return conv[: 2 * n - 1]


def _conv_with_weight(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g = 2 * (a_rev ⊛ w)  where a_rev = a[::-1] and ⊛ denotes
    linear convolution.  The returned vector has length len(a) and
    corresponds exactly to the gradient of the smoothed max‑convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # convolution length needed : len(a_rev) + len(w) – 1 = n + (2n‑1) – 1 = 3n‑2
    N = 1 << ((3 * n - 2).bit_length())
    fft_rev = np.fft.rfft(a_rev, n=N)
    fft_w = np.fft.rfft(w, n=N)
    full = np.fft.irfft(fft_rev * fft_w, n=N)
    # the gradient entries are the slice [n‑1 : n‑1+n]
    return 2.0 * full[n - 1 : n - 1 + n]


def _smooth_max_gradient(a: np.ndarray, beta: float) -> tuple:
    """
    For a normalised vector a and smoothing parameter beta,
    return (conv, grad) where
        conv = a*a  (full convolution, length 2n‑1)
        grad = ∇ S_β(a)  (same shape as a)
    """
    conv = _conv_fft(a)                         # exact auto‑convolution
    max_val = conv.max()
    # stabilised soft‑max weights
    exps = np.exp(beta * (conv - max_val))
    w = exps / exps.sum()
    grad = _conv_with_weight(a, w)               # efficient via FFT
    return conv, grad


def _approx_score(a: np.ndarray) -> float:
    """
    Fast homogeneous upper bound: 2·n·max(a*a)  (sum(a)=1 assumed).
    """
    return float(2 * a.shape[0] * _conv_fft(a).max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient list that minimises
    `evaluate_sequence`.  The function respects a ~1000 s time budget
    and returns the best sequence encountered.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    n = 8192                     # length that gave the previous best
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None and len(hs1) == n:
        seq = np.asarray(hs1, dtype=float)
        seq = seq / seq.sum()
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from "
                  f"height_sequence_1 → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic scan of analytic families (more fine‑grained)
    # ------------------------------------------------------------------
    # 1) Geometric (near‑unit exponential) with a dense grid on the
    #    decay constant c.
    if best_seq is None:
        # start from a tiny random vector if we have nothing yet
        a0 = rng.random(n)
        a0 = a0 / a0.sum()
        best_seq = a0.tolist()
        best_score = evaluate_sequence(best_seq)

    # scan many c‑values (≈150 points) – cheap approximate filter first
    c_grid = np.linspace(0.1, 6.0, 150)
    for c in c_grid:
        if time.time() >= deadline:
            break
        r = np.exp(-c / n)                     # geometric ratio
        a = np.power(r, np.arange(n, dtype=float))
        a = a / a.sum()
        approx = _approx_score(a)
        # only call the expensive evaluator if the bound is promising
        if approx < best_score * 1.05:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.tolist()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"Geo c={c:5.2f} → score={best_score:.6f}")
                if best_score <= 1.503:
                    print("Target reached during Phase 1.")
                    break

    # 2) A few beta‑like shapes (a_i = (i+1)^α·(n‑i)^β)
    exponents = [(0.5, 0.5), (0.5, 1.0), (1.0, 0.5), (1.0, 1.0), (2.0, 2.0)]
    for alpha, beta in exponents:
        if time.time() >= deadline:
            break
        i = np.arange(1, n + 1, dtype=float)
        a = (i ** alpha) * ((n - i + 1) ** beta)
        a = a / a.sum()
        approx = _approx_score(a)
        if approx < best_score * 1.05:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.tolist()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"Beta α={alpha} β={beta} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max (Adam) refinement
    # ------------------------------------------------------------------
    a = np.asarray(best_seq, dtype=float)
    a = a / a.sum()                     # enforce Σa=1  (scale‑invariant)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    # schedule of smoothing parameters (geometric progression)
    beta_vals = np.geomspace(0.2, 30.0, num=18)

    # helper to decide when to ask the official evaluator
    def _maybe_update(t_iter):
        # check roughly every 50 iterations or when a noticeable drop occurs
        return (t_iter % 50 == 0) or (t_iter == 1)

    for smooth_beta in beta_vals:
        # inner optimisation loop for the current β
        for _ in range(800):                     # enough steps per β
            if time.time() >= deadline:
                break

            # compute smoothed convolution and its gradient
            conv, grad = _smooth_max_gradient(a, smooth_beta)

            # Adam update (minimise the smoothed max)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.02 * m_hat / (np.sqrt(v_hat) + eps)   # learning‑rate 0.02

            a_new = a - step
            a = _project_onto_simplex(a_new)               # enforce Σ=1, a≥0

            # occasional exact check
            if _maybe_update(t):
                approx = _approx_score(a)
                if approx < best_score:
                    exact = evaluate_sequence(a.tolist())
                    if exact < best_score:
                        best_score = exact
                        best_seq = a.tolist()
                        print(f"[{time.time() - start_time:.1f}s] "
                              f"New best (smooth β={smooth_beta:.2f}) "
                              f"score={best_score:.6f}")
                        if best_score <= 1.503:
                            print("Target achieved – exiting refinement.")
                            break
        if time.time() >= deadline or best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 3 – random‑restart polishing (if time permits)
    # ------------------------------------------------------------------
    attempts = 0
    while time.time() < deadline and attempts < 12:
        attempts += 1
        # small Gaussian perturbation of the currently best vector
        noise = rng.normal(scale=0.02, size=n)
        a_pert = best_seq + noise
        a_pert = np.maximum(a_pert, 0.0)               # keep non‑negative
        a_pert = a_pert / a_pert.sum()

        # short Adam run starting from the perturbed point
        m = np.zeros_like(a_pert)
        v = np.zeros_like(a_pert)
        t = 0
        for inner in range(300):
            if time.time() >= deadline:
                break
            conv, grad = _smooth_max_gradient(a_pert, smooth_beta=10.0)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.02 * m_hat / (np.sqrt(v_hat) + eps)
            a_pert = _project_onto_simplex(a_pert - step)

        # evaluate the polished candidate
        approx = _approx_score(a_pert)
        if approx < best_score:
            exact = evaluate_sequence(a_pert.tolist())
            if exact < best_score:
                best_score = exact
                best_seq = a_pert.tolist()
                print(f"[{time.time() - start_time:.1f}s] Polished restart #{attempts} "
                      f"→ score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq

    print(f"[{time.time() - start_time:.1f}s] Finished – best score={best_score:.6f}")
    # Ensure the returned list follows the API contract (list of floats)
    return [float(x) for x in best_seq]
```
