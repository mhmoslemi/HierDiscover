Implementation of proj_simplex:

def proj_simplex(v):

    v = np.asarray(v)

    if v.ndim != 1: raise Value error.

    n = len(v)

    u = np.sort(v)[::-1]  # descending

    cssv = np.cumsum(u) - 1

    ind = np.arange(1, n+1)

    cond = u - cssv / ind > 0

    rho = ind[cond][-1] - 1   # zero-index

    theta = cssv[cond][-1] / (rho + 1)

    w = np.maximum(v - theta, 0)

    return w

"""
Adam‑based simplex projection on the step‑function autocorrelation peak.

The algorithm keeps the coefficient vector on the probability simplex
(∑a = 1, a_i ≥ 0) and uses an Adam optimiser to descend the L∞‑norm
of its autocorrelation.  At each iteration the full autocorrelation
is obtained by an FFT, the k largest entries are selected (k≈5) and a
sub‑gradient is built from the corresponding shifts of the current
vector.  After a gradient step the vector is projected back onto the
simplex.  Small random “mass‑transfer’’ moves and a restart heuristic
avoid stagnation.  The best sequence seen during the time budget is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must stay top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT of length ``size`` (≥ len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _max_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return the full convolution a*a using a pre‑computed FFT size."""
    return _conv_fft(a, a, size)


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small random amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise onto the simplex (projection is cheap because sum stays 1)
    a_new /= a_new.sum()
    return a_new


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, ∑x_i = 1}.
    """
    # borrow from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Ensure exact sum = 1 (might be off by numerical noise)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


def _gradient_from_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute the (sub‑)gradient of ``max(conv(a,a))`` using the active
    indices where the autocorrelation is (almost) maximal.
    The gradient is the sum of the individual gradients of the selected
    convolution entries.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    # offset = index - (n-1)  (zero‑lag corresponds to offset 0)
    offs = active_idx - (n - 1)

        if d >= 0:

            grad[d:] += a[: n - d]
        else:
            k = -d
            grad[k:] += a[: n - k]
            grad[: n - k] += a[k:]
    return grad

"""
Adam‑based simplex optimisation for minimizing the ℓ∞‑norm of a step
function’s autocorrelation.

The vector a (length n) is kept on the probability simplex
(∑a = 1, a_i ≥ 0).  In each iteration we:

1. compute the full convolution a∗a with an FFT,
2. pick the `active_k` largest convolution entries (they are the
   constraints that are currently binding),
3. build a sub‑gradient by adding the two shifted copies of a that
   correspond to each active offset,
4. perform an Adam update with a linearly decaying learning rate,
5. project the result back onto the simplex.

Occasional random “mass‑transfer’’ moves and a restart after a few
seconds without improvement help escape plateaus.  The best sequence
seen during the time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT of length ``size`` (must be ≥ len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _max_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a∗a (full convolution) using a pre‑computed FFT size."""
    return _conv_fft(a, a, size)


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, ∑x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                   # zero‑based index of the last positive entry
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Random exploration move
# ----------------------------------------------------------------------


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small random amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise (still on the simplex because we only moved mass)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Gradient from the active autocorrelation entries
# ----------------------------------------------------------------------


def _gradient_from_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute the (sub‑)gradient of max(conv(a, a)) using the selected
    indices (the “active’’ set).  For each active offset d = i-(n‑1) we add
    the two shifted copies of ``a`` that appear in the derivative of the
    corresponding convolution entry.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    # offsets with respect to the centre of the convolution
    offsets = active_idx - (n - 1)
    for d in offsets:
        if d >= 0:
            # a_{j+d} term
            grad[: n - d] += a[d:]
            # a_{j-d} term (symmetry)
            grad[d:] += a[: n - d]
        else:
            k = -d
            grad[k:] += a[: n - k]          # a_{j-d}
            grad[: n - k] += a[k:]          # a_{j+d}
    return grad
