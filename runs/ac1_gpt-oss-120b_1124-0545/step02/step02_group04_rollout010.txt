"""
Projected gradient descent on a smoothed version of the maximal
self‑convolution entry.

The objective to minimise is
    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

which is homogeneous – scaling of the vector does not change the
value.  We normalise the vector to the conventional ℓ1‑norm
√(2 n) and iteratively descent the smooth max of the convolution.
The smooth max is approximated by a soft‑max with a temperature that
is increased during the run (annealing).  The gradient of the
smooth‑max can be written in closed form as a convolution, which we
compute with an FFT.  After each gradient step the vector is projected
onto the probability simplex (non‑negative and prescribed sum) using
the O(n log n) algorithm of Wang & Carreira‑Perpiñán.

A simple momentum term and occasional random perturbations help the
search escape shallow local minima.  The routine runs for at most the
given time budget (default 1000 s) and returns the best step‑function
found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector v onto the simplex {x >= 0, sum(x)=s}.
    Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – should happen only when s=0
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation function defined in the problem statement.

    The algorithm uses projected gradient descent on a smoothed
    maximum of the self‑convolution, with an annealing schedule for
    the smoothness parameter and a simple momentum term.
    """
    # ------------------- initialisation ---------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # start from a known construction if available, otherwise random
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2500
        a = np.random.exponential(scale=1.0, size=n)

    n = a.size
    target_sum = math.sqrt(2.0 * n)               # ℓ₁‑norm we keep throughout
    a = np.maximum(a, 0.0)
    a = a / a.sum() * target_sum                  # project onto the simplex

    # FFT length: must accommodate linear convolution of length 3·n‑2
    nfft = 1 << ((3 * n - 2).bit_length())

    # ------------------- optimisation parameters -----------------------
    # soft‑max temperature schedule
    alpha_start = 1.0
    alpha_end = 30.0

    # step‑size schedule (initial step ≈ 0.1·target_sum/n, decays to 0)
    step0 = 0.1 * target_sum / n

    # momentum coefficient
    mu = 0.9
    velocity = np.zeros_like(a)

    # patience for random perturbations
    last_improve_iter = 0
    perturb_every = 8000            # iterations
    max_no_imp = 15000              # if no improvement for this many iters, perturb

    # ------------------- bookkeeping ------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin
    iteration = 0
    best_a = a.copy()
    # compute initial objective (using the true evaluation function form)
    conv_full = np.convolve(a, a)
    max_conv = conv_full.max()
    best_score = 2.0 * n * max_conv / (a.sum() ** 2)

    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = (elapsed) / (deadline - start_time)

        # ---- annealing of the smooth‑max parameter --------------------
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)

        # ---- FFT of the current vector --------------------------------
        a_fft = np.fft.rfft(a, n=nfft)

        # ---- full self‑convolution (linear, first 2n‑1 entries) -------
        conv = np.fft.irfft(a_fft * a_fft, n=nfft)[: 2 * n - 1]

        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        # ---- keep the best seen so far --------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve_iter = iteration
            print(f"[{iteration:7d}] New best score: {best_score:.6f} (elapsed {elapsed:.1f}s)")

        # ---- smooth‑max weights (soft‑max over the convolution) -------
        # shift by the maximum to avoid overflow
        shifted = conv - max_conv
        weights = np.exp(alpha * shifted)
        w_sum = weights.sum()
        if w_sum == 0 or np.isnan(w_sum) or np.isinf(w_sum):
            # fallback to the hard max (single active entry)
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            w = weights / w_sum

        # ---- gradient of the smooth‑max -------------------------------
        # g = 2 * (w_rev * a)  where * denotes linear convolution
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=nfft)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=nfft)  # circular conv
        # linear part needed for gradient: first (2n‑1)+(n‑1)=3n‑2 samples
        conv_rev = grad_full[: (2 * n - 1) + n - 1]       # length 3n‑2

        L = 2 * n - 1
        # gradient_i = 2 * conv_rev[L - n + i] (reversed order)
        grad = 2.0 * conv_rev[L - n : L][::-1]

        # ---- step size (decaying) ------------------------------------
        step = step0 * (1.0 - frac)                     # linear decay to zero

        # ---- momentum update -----------------------------------------
        velocity = mu * velocity - step * grad
        a_candidate = a + velocity

        # ---- projection onto the simplex (keeps sum = target_sum) -----
        a_candidate = _simplex_projection(a_candidate, target_sum)

        # ---- occasionally inject a small random perturbation ----------
        if iteration - last_improve_iter > max_no_imp:
            a_candidate = best_a + np.random.normal(scale=target_sum * 0.001,
                                                    size=n)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _simplex_projection(a_candidate, target_sum)
            velocity[:] = 0.0
            last_improve_iter = iteration

        if iteration % perturb_every == 0 and iteration > 0:
            # gentle jitter to avoid stagnation
            noise = np.random.normal(scale=target_sum * 0.0005, size=n)
            a_candidate = np.maximum(a_candidate + noise, 0.0)
            a_candidate = _simplex_projection(a_candidate, target_sum)

        a = a_candidate
        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence found – as a plain Python list
    return best_a.tolist()
