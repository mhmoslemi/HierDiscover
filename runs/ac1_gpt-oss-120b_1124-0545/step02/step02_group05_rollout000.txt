def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w >= 0 , ∑ w = s }."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s

    if not np.any(cond):
        # fallback – all entries equal
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step‑function with minimal max self‑conv."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # problem size & target ℓ₁‑norm
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2500                             # reasonable default
    target_sum = math.sqrt(2.0 * n)          # ∑a = √(2·n) ⇒ objective = max(conv)

    # ------------------------------------------------------------------
    # helper: evaluate the (scaled) max‑convolution of a vector a
    def _max_conv(a_vec: np.ndarray) -> float:
        conv = np.convolve(a_vec, a_vec)
        return float(conv.max())

    # ------------------------------------------------------------------
    # gradient of the smoothed max (log‑sum‑exp) surrogate
    def _grad_smooth(a_vec: np.ndarray, eps: float):
        conv = np.convolve(a_vec, a_vec)                     # length 2n‑1
        if eps < 1e-8:                                        # hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # stable soft‑max
            conv_shift = conv - conv.max()
            w_raw = np.exp(conv_shift / eps)
            w_sum = w_raw.sum()
            if not np.isfinite(w_sum) or w_sum == 0.0:
                # numerical fallback → hard max
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = w_raw / w_sum
        # sigma_j = Σ_{i=j}^{j+n‑1} w_i   (j = 0 … n‑1)
        cumsum_w = np.cumsum(w)
        sigma = cumsum_w[n-1:2*n-1] - np.concatenate(([0.0], cumsum_w[:n-1]))
        # first term = (w * rev(a))_segment
        a_rev = a_vec[::-1]
        conv_wa = np.convolve(w, a_rev)
        first = conv_wa[(n-1):(n-1+n)]
        grad = first + a_vec * sigma
        return grad, conv.max()

    # ------------------------------------------------------------------
    # initialise best solution
    uniform = np.full(n, target_sum / n)
    rand_exp = np.random.exponential(scale=1.0, size=n)
    rand_exp = rand_exp / rand_exp.sum() * target_sum
    starters = [uniform, rand_exp]
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        base = base / base.sum() * target_sum
        starters.append(base)

    best_seq = None
    best_score = np.inf
    for cand in starters:
        cand = _simplex_projection(cand, target_sum)
        sc = _max_conv(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # optimisation parameters
    eps_start, eps_end = 0.5, 1e-6            # smoothing schedule
    lr_start = target_sum * 0.001            # Adam learning‑rate start
    lr_end   = target_sum * 0.00005          #   … end
    beta1, beta2 = 0.9, 0.999                # Adam moments
    hard_eps_thresh = 1e-5                   # switch to hard max when eps is tiny
    perturb_interval = 400                   # steps between random mixes
    max_inner_steps = 2000                   # steps before we optionally restart

    # Adam state (shared across restarts, re‑initialised on restart)
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    # global iteration counter (used for schedules)
    global_step = 0

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # spare 5 s for clean‑up

    # ------------------------------------------------------------------
    # main optimisation loop
    a = best_seq.copy()                     # current iterate
    last_improve_step = 0

    while time.time() < deadline:
        # schedule based on elapsed fraction of total budget
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)   # 0 → 1

        # smoothing and learning‑rate schedules (exponential decay)
        eps = eps_start * ((eps_end / eps_start) ** frac)
        lr  = lr_start  * ((lr_end  / lr_start ) ** frac)

        # gradient (smooth or hard, depending on eps)
        grad, cur_max = _grad_smooth(a, eps)
        # Adam update
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _simplex_projection(a_new, target_sum)

        a = a_new
        global_step += 1

        # record best
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            last_improve_step = global_step
            print(f"[{global_step:7d}] New best max(conv) = {best_score:.6f} "
                  f"(t={elapsed:.1f}s, eps={eps:.2e})")

        # early exit if target already reached
        if best_score <= 1.503:
            print("Target reached – terminating early.")
            break

        # ------------------------------------------------------------------
        # occasional perturbations to escape plateaus
        if global_step % perturb_interval == 0:
            # blend best solution with a fresh random Dirichlet sample
            rnd = np.random.exponential(scale=1.0, size=n)
            rnd = rnd / rnd.sum() * target_sum
            mix = np.random.rand()
            a_pert = mix * best_seq + (1.0 - mix) * rnd
            a_pert = _simplex_projection(a_pert, target_sum)

            pert_max = _max_conv(a_pert)
            if pert_max < best_score:
                best_score = pert_max
                best_seq = a_pert.copy()
                last_improve_step = global_step
                print(f"[{global_step:7d}] New best after perturbation: {best_score:.6f}")

            # replace current iterate if perturbation improves it
            if pert_max < cur_max:
                a = a_pert.copy()
                # reset Adam moments for the new point
                m.fill(0.0)
                v.fill(0.0)
                t_adam = 0

        # ------------------------------------------------------------------
        # restart if stagnation observed
        if global_step - last_improve_step > max_inner_steps:
            # re‑initialise from mixture of best and noise
            noise = np.random.exponential(scale=1.0, size=n)
            noise = noise / noise.sum() * target_sum
            a = 0.5 * best_seq + 0.5 * noise
            a = _simplex_projection(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            last_improve_step = global_step
            print(f"[{global_step:7d}] Stagnation – restart performed.")

    # ------------------------------------------------------------------
    # Final local hill‑climbing on the best sequence found
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_local = best_seq.copy()
        cur_best = best_score
        n_moves = 2000                     # number of pairwise moves
        move_step = target_sum * 0.001     # typical δ
        for mv in range(n_moves):
            if time.time() >= deadline:
                break
            # pick a source with non‑zero mass
            src_idxs = np.where(a_local > 1e-12)[0]
            if src_idxs.size == 0:
                break
            i = int(np.random.choice(src_idxs))
            j = np.random.randint(0, n)
            if i == j:
                continue
            delta = min(a_local[i], move_step * np.random.rand())
            if delta <= 0.0:
                continue
            a_candidate = a_local.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # (sum unchanged, non‑negativity already guaranteed)
            cand_max = _max_conv(a_candidate)
            if cand_max < cur_best:
                cur_best = cand_max
                a_local = a_candidate
                best_seq = a_candidate.copy()
                best_score = cur_best
                print(f"[HC {global_step+mv:7d}] Improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (plain list)
    return best_seq.tolist()
