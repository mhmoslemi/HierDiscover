"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution.  The algorithm consists of:

1.  Build a diverse collection of seed vectors for a chosen length.
    Families include geometric, power‑law, combined geometric‑power,
    symmetric geometric, sine, Hann, Beta‑distribution shapes
    (including the arcsine‑type U‑shaped distribution).

2.  Quick scan of the seeds with the fast FFT‑based evaluator to pick
    the most promising few (by actual objective value).

3.  For each of the promising seeds run a few thousand iterations of a
    soft‑max Adam optimiser.  The loss is a log‑sum‑exp smooth
    approximation of the max‑convolution; the smoothing parameter grows
    geometrically, turning the loss into the true max.

4.  Starting from the best vector found so far run a hard‑max
    sub‑gradient descent (projected onto the probability simplex) for
    a few thousand additional steps.  The gradient of the max‑entry is
    computed analytically in O(n).

All steps respect the supplied time budget (default 1000 s).  The best
sequence seen during the entire run is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _geom(n: int, r: float) -> np.ndarray:
    """Geometric progression (r^i) normalised to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric * power‑law."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (mirrored around centre)."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _hann_window(n: int) -> np.ndarray:
    """Standard Hann window (cosine‑squared)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta distribution on {0,…,n‑1}.
    Uses mid‑point sampling to avoid infinities at the ends.
    """
    # points in (0,1) avoiding exact 0/1
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # compute pdf ∝ x^{α‑1} (1‑x)^{β‑1}
    a = np.where(
        (x > 0) & (x < 1),
        np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x)),
        0.0,
    )
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _arcsine_shape(n: int) -> np.ndarray:
    """
    The continuous arcsine density (α=β=0.5) discretised on n points.
    Values are ∝ 1 / sqrt(x (1‑x)), with x the mid‑point.
    """
    return _beta_shape(n, 0.5, 0.5)

def _adam_optimize(a0: np.ndarray, n: int, deadline: float,
                   max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best vector (sum = 1) discovered and its objective.
    """
    v = np.log(a0 + 1e-30)      # logits
    m = np.zeros_like(v)       # first moment
    v2 = np.zeros_like(v)      # second moment
    t = 0

    # Adam hyper‑parameters
    lr0 = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    # smoothing schedule for the log‑sum‑exp approximation
    beta_start = 2.0
    beta_mul = 1.015

    best_seq = None
    best_val = float("inf")
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                      # stays on the simplex
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[adam] n={n:5d} it={it:4d} new best={cur_val:.6f}")
        else:
            no_improve += 1
            if no_improve >= 1200:           # early stop if stagnating
                break

        # --- smoothed max (log‑sum‑exp) ---
        beta = beta_start * (beta_mul ** it)
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # --- gradient of the smoothed objective w.r.t a ---
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length ≈ 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]            # central n entries

        # --- back‑propagate through the softmax ---
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # --- Adam update on logits ---
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _hard_max_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max_{k} (a * a)_k at a specific index k.
    Implements the exact derivative of the selected convolution entry.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    # iterate over all pairs (i, j) with i + j = k
    for i in range(i_min, i_max + 1):
        j = k - i
        grad[i] += a[j]
        grad[j] += a[i]
    return grad


def _subgradient_optimize(a0: np.ndarray, n: int, deadline: float,
                         max_iters: int = 2000, step0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses the hard max (no smoothing) and projects onto the simplex after each step.
    """
    a = a0.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * conv.max()
    t = 0
    no_improve = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        k = int(conv.argmax())
        grad = _hard_max_gradient(a, k)          # sub‑gradient of cur_max
        eta = step0 / np.sqrt(t)                 # diminishing step size
        a_next = a - eta * grad
        a = _project_simplex(a_next)

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[subgrad] n={n:5d} it={t:4d} new best={cur_val:.6f}")
        else:
            no_improve += 1
            if no_improve >= 500:               # stagnation guard
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # lengths to try (larger lengths give a finer bound but are slower)
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        # ------------------------------------------------------------------
        # Build a collection of diverse seed vectors for this length
        # ------------------------------------------------------------------
        seeds = []

        # Geometric families
        for c in np.linspace(0.5, 5.0, 9):
            r = 1.0 - c / n
            if r > 0.0:
                seeds.append(_geom(n, r))

        # Pure power‑law families
        for p in np.linspace(0.0, 2.5, 7):
            seeds.append(_power(n, p))

        # Combined geometric × power
        for c in np.linspace(0.5, 4.0, 5):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 5):
                seeds.append(_geom_power(n, r, p))

        # Symmetric geometric
        for c in np.linspace(0.5, 4.5, 5):
            r = 1.0 - c / n
            if r > 0.0:
                seeds.append(_sym_geom(n, r))

        # Classical windows
        seeds.append(_sine_window(n))
        seeds.append(_hann_window(n))

        # Beta‑distribution shapes (including arcsine)
        beta_params = [
            (0.5, 0.5),   # arcsine – U‑shaped
            (0.8, 0.8),
            (1.2, 1.2),   # near‑uniform bell
            (0.5, 2.0),
            (2.0, 0.5),
            (1.5, 1.5),
        ]
        for a, b in beta_params:
            seeds.append(_beta_shape(n, a, b))

        # A few random initial points (logits are Gaussian)
        for _ in range(6):
            v_rand = np.random.randn(n)
            seeds.append(_softmax(v_rand))

        # If a previous construction exists, reuse it (potentially resized)
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                arr = np.asarray(prev, dtype=np.float64)
                if arr.shape == (n,):
                    seeds.append(arr / arr.sum())
                else:
                    # truncate / pad to the required length
                    adj = np.zeros(n, dtype=np.float64)
                    m = min(len(arr), n)
                    adj[:m] = arr[:m]
                    adj = adj / adj.sum()
                    seeds.append(adj)
            except Exception:
                pass

        # ------------------------------------------------------------------
        # Quick scan of all seeds – keep the best few for intensive optimisation
        # ------------------------------------------------------------------
        scored_seeds = []
        for a in seeds:
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            scored_seeds.append((a, cur_val))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] n={n} new global best={cur_val:.6f}")

        # Sort by the quick objective and keep the most promising ~15 seeds
        scored_seeds.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in scored_seeds[:15]]

        # ------------------------------------------------------------------
        # Run Adam optimisation on the selected seeds (time‑aware)
        # ------------------------------------------------------------------
        for idx, a0 in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(a0, n, deadline, max_iters=2000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall] n={n} seed={idx} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Final refinement via hard‑max sub‑gradient descent (if time permits)
        # ------------------------------------------------------------------
        if global_best_seq is not None and time.time() < deadline:
            seq, val = _subgradient_optimize(global_best_seq, n, deadline,
                                            max_iters=2000, step0=0.6)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[final] n={n} sub‑grad improved global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform distribution (should never happen)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] using uniform distribution")

    # Return as plain list; the outer evaluation will apply the built‑in security checks.
    return global_best_seq.tolist()
