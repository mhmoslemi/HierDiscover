"""
A fast Adam‑based optimizer with a temperature‑annealed soft‑max
approximation of the “max‑convolution’’ functional.

The objective to minimise is

    J(a) = 2·n·max(a⋆a) / (∑a)² ,

which is invariant under scaling, so the search stays on the
probability simplex (∑a = 1, aᵢ ≥ 0).

The algorithm works in two phases:

1.  A warm‑up phase with a small softening parameter κ (large
    temperature) that yields a smooth approximation of the max.
    A regularised Adam step is taken and the parameters are projected
    back onto the simplex.

2.  κ is increased linearly with the elapsed time (annealing).  For
    large κ the weight vector approaches the binary mask of the
    active constraints, so the update becomes a conventional
    sub‑gradient step.  Adam’s momentum helps to escape shallow
    plateaux that defeat plain sub‑gradient descent.

Random “mass‑transfer’’ moves and occasional restarts are added to
avoid getting trapped in local minima.

The whole routine respects the 1000 s budget, prints progress, and
returns the best sequence discovered (as a plain Python list of
floats)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution of `u` and `v` using an FFT of length `size`.
    The returned array has length ``len(u) + len(v) - 1``.
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _max_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Convenience wrapper for a⋆a."""
    return _conv_fft(a, a, size)


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Move a small amount of mass from a random index `i`
    to another random index `j`.  The vector is re‑normalised.
    """
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    if a[i] == 0.0:
        return a
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def _project_simplex(x: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s == 0.0:
        return np.ones_like(x) / x.size
    return x / s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.

    The optimiser uses an Adam update on a temperature‑annealed
    soft‑max approximation of ``max(a⋆a)``.  Random mass‑transfer
    perturbations and time‑based restarts improve exploration.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the harness
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # initialise a starting distribution
    # --------------------------------------------------------------
    try:
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            seq0 = np.asarray(seq0, dtype=float)
            if seq0.size != n:
                # linear interpolation to the required length
                x_old = np.linspace(0, 1, seq0.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, seq0)
                a = np.clip(a, 0.0, None)
            else:
                a = np.clip(seq0, 0.0, None)
        else:
            raise KeyError
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)

    # --------------------------------------------------------------
    # pre‑compute FFT lengths (powers of two) – they stay constant
    # --------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a⋆a
    size2 = _next_pow2(3 * n - 2)       # for a_rev⋆weights

    # --------------------------------------------------------------
    # Adam state
    # --------------------------------------------------------------
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base = 0.12                    # base learning rate (will be scaled)
    t_iter = 0

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    rand_prob = 0.09                  # probability of a random transfer move
    rand_scale = 0.04                 # relative size of the random transfer
    stagnation_restart = 55.0         # seconds without improvement → restart
    last_improve = start

    # temperature (soft‑max) schedule
    kappa_start = 5.0                 # low temperature – highly smooth
    kappa_end = 25000.0               # high temperature – close to hard max
    total_budget = budget_s

    # bookkeeping of the best solution found
    conv_full = _max_convolution(a, size1)
    max_b = conv_full.max()
    best_score = 2.0 * n * max_b
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        t_iter += 1

        # --- 1. convolution & current score ---------------------------------
        conv_full = _max_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{t_iter:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

            # early exit if we already beat the target
            if best_score < 1.503:
                print("[target reached]")
                break

        # --- 2. occasional random move ------------------------------------
        if np.random.rand() < rand_prob:
            a = _random_transfer(a, rand_scale)
            # reset Adam moments (the perturbation invalidates them)
            m.fill(0.0)
            v.fill(0.0)
            continue

        # --- 3. compute annealed soft‑max weights -------------------------
        elapsed = time.time() - start
        kappa = kappa_start + (kappa_end - kappa_start) * (elapsed / total_budget)
        kappa = min(kappa, kappa_end)

        # diff = max_b - conv ≥ 0  (numerical safety)
        diff = max_b - conv_full
        diff = np.maximum(diff, 0.0)

        # soft‑max (weights sum to 1).  For very large kappa this becomes the mask.
        w = np.exp(-kappa * diff)
        w_sum = w.sum()
        if w_sum == 0.0 or kappa > 1e6:      # fallback to a hard mask
            mask = (conv_full >= max_b - 1e-12).astype(float)
            w = mask
            w_sum = w.sum()
        w /= w_sum

        # --- 4. gradient of the smoothed objective ------------------------
        a_rev = a[::-1]
        full = _conv_fft(a_rev, w, size2)          # length 3n‑2
        grad = 2.0 * full[n - 1 : n - 1 + n]       # central block = ∂F/∂a

        # --- 5. Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr_base * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # --- 6. stagnation restart ----------------------------------------
        if time.time() - last_improve > stagnation_restart:
            a = np.random.rand(n)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{t_iter:7d}] restart due to stagnation")

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={t_iter}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
