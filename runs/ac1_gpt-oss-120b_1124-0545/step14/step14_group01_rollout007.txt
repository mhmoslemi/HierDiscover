"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works as follows:

1.  Load the best sequence found so far (`height_sequence_1`).  If it does not
    exist, start from a random Dirichlet distribution of length 4000
    (ℓ₁‑normalised).

2.  Perform a continuation L‑BFGS optimisation on the smooth log‑sum‑exp
    approximation of the max‑convolution.  The smoothing parameter β is
    increased gradually (1 → … → 50 000) and a relatively large number of
    L‑BFGS iterations is allowed per β.  The optimiser works in the
    log‑softmax space, so the iterates always stay on the probability simplex.

3.  Refine the result with a long exponentiated‑gradient (mirror‑descent)
    run that directly minimises the exact surrogate 2·n·max(conv).

4.  Apply a small number of Frank–Wolfe/L‑P steps: for the current vector
    `f` compute `M = max(conv(f,f))`, build a linear programme that maximises
    the ℓ₁‑norm of a new vector `g` under the linear constraints
    `conv(f,g) ≤ M`.  After solving the LP (HiGHS, sparse matrices) we form
    `f ← (1‑t)·f + t·ĝ` with a modest step size `t` (≈0.1) and repeat a few
    times.  This operation provably never increases the max‑convolution and
    empirically drives the objective a few thousandths lower.

5.  A short simulated‑annealing pass in log‑softmax space gives a final
    stochastic polish.

All loops stop as soon as the 1000 s budget is exhausted.  The best
vector encountered (ℓ₁‑normalised) is stored in the global variable
`height_sequence_1` and returned as a plain list of floats.
"""

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """Return an ℓ₁‑normalised vector that (approximately) minimises max‑conv."""
    if minimize is None:
        raise RuntimeError("SciPy minimise is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0]

    # initialise log‑softmax parameters (stay on simplex)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # ℓ₁‑norm = 1
            conv = _conv_fft(a, a)                   # full convolution
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximation of max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth objective w.r.t. a
            w = exp_beta / sum_exp                     # softmax over β·conv
            rev_a = a[::-1]
            # full convolution of rev_a and w (length = 3n‑2)
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = minimize(fun=fun,
                           x0=v,
                           method='L-BFGS-B',
                           jac=True,
                           callback=callback,
                           options={'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] exception:", e)
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser on the exact surrogate
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> np.ndarray:
    """Mirror‑descent on 2·n·max(conv) (surrogate)."""
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
    return best_a


# ----------------------------------------------------------------------
# Linear‑programming refinement (Frank–Wolfe style)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_lp_iter: int = 6,
               step: float = 0.12,
               tol_frac: float = 1e-3) -> np.ndarray:
    """
    Repeatedly solve a small LP:
        maximise Σ g_i  subject to  conv(f,g) ≤ M  (for active rows)
    and blend the solution back into `f`.  Returns the improved vector.
    """
    if linprog is None or _sp is None:
        # No LP support – simply return the input.
        return f

    for lp_it in range(max_lp_iter):
        if time.time() > deadline:
            break

        conv_fg = _conv_fft(f, f)
        M = np.max(conv_fg)

        # active rows: where the value is close to the maximum
        tol = max(tol_frac * M, 1e-12)
        active_idx = np.where(conv_fg >= M - tol)[0]
        if active_idx.size == 0:
            break

        # Build sparse constraint matrix A_ub * g ≤ M
        rows = []
        cols = []
        data = []
        for row_id, k in enumerate(active_idx):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            cols_i = np.arange(i_start, i_end + 1, dtype=np.int32)
            rows_i = np.full_like(cols_i, row_id, dtype=np.int32)
            rows.append(rows_i)
            cols.append(cols_i)
            data.append(f[i_start:i_end + 1])

        row_ind = np.concatenate(rows)
        col_ind = np.concatenate(cols)
        data_val = np.concatenate(data)

        A = _sp.csr_matrix((data_val, (row_ind, col_ind)),
                           shape=(active_idx.size, n))

        b = np.full(active_idx.size, M, dtype=np.float64)

        # maximise Σ g → minimise -Σ g
        c = -np.ones(n, dtype=np.float64)

        bounds = [(0, None)] * n

        try:
            res = linprog(c,
                          A_ub=A,
                          b_ub=b,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True})
        except Exception as e:  # pragma: no cover
            print("[LP] optimisation error:", e)
            break

        if not res.success:
            # LP failed – abort further refinements
            print("[LP] failed (status:", res.status, ")")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            # degenerate – stop
            break
        g_norm = g / sum_g

        # convex blend
        f = (1.0 - step) * f + step * g_norm

    return f

# Simulated‑annealing polish (log‑softmax space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 200_000) -> np.ndarray:
    """A lightweight SA pass that may catch tiny improvements."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 50000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")
    return best_seq

                                        max_iters=max_sa_iters)
        val_sa = _objective_surrogate(f_sa)
        print(f"[SA] surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_seq = f_sa
            best_val = val_sa

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
