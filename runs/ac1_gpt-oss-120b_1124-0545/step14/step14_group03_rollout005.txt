"""
Propose a low‑peak autocorrelation (self‑convolution) step function.

The algorithm consists of four stages:

1. **Rich seed pool** – deterministic analytic families (uniform,
   triangular, beta, inverse‑sqrt, windows, random Dirichlet /
   exponential, low‑frequency cosine mixtures) and many random mixtures.
   All seeds are ℓ₁‑normalised (∑a = 1) so the objective is scale‑invariant.

2. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp surrogate
   of the max‑convolution.  A β‑schedule ending at 1e5 is used and a
   generous per‑β iteration budget.  The optimiser works on the logits
   of the soft‑max (the probability simplex) and updates `v` directly.

3. **Exponentiated‑gradient (EG) refinement** – a mirror‑descent run on
   the exact (non‑smooth) objective.  The sub‑gradient comes from the
   current maximal convolution index.  Step‑size ηₜ = η₀ / (t+1)⁰·⁶
   (η₀=1.0) gives a long, slowly‑decaying descent.

4. **Permutation & polishing** – the best vector is randomly permuted
   (values are kept, positions are shuffled) many times; any improvement
   is kept.  Finally a short high‑β L‑BFGS polishing is performed.

The routine respects the supplied `budget_s` (≈ 1000 s) and aborts
gracefully before the deadline.  The best sequence found is stored in
the global name ``height_sequence_1`` and returned as a plain list of
floats.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ------------------------------------------------------------
# Low‑level numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    out = np.fft.irfft(fx * fy, n=N)[:L]
    return out


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on {1,…,n} (discretised)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt(n: int) -> np.ndarray:
    """Inverse‑sqrt shape a_i ∝ 1/√(i·(n‑i+1)), i=1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, ℓ₁‑renormalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ------------------------------------------------------------
# Surrogate objective (smooth log‑sum‑exp) used by L‑BFGS
# ------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Function and gradient for the smooth max‑convolution surrogate."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)              # full autocorrelation (self‑conv)
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_term)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp               # soft weighting of each lag
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # gradient w.r.t. a

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)          # chain rule through softmax
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        bet = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, bet)

        # tiny callback just to obey deadline
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={bet}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] caught exception {e} (beta={bet})")
            break

    return _softmax(v)


# ------------------------------------------------------------
# Exact max‑convolution objective (no surrogate)
# ------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (active‑set = single max index)."""
    n = len(a)
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))                # index of the current max
    g = np.zeros_like(a)
    # contribution of ordered pairs (i, j) with i+j = k
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_range = np.arange(i_min, i_max + 1, dtype=int)
    j_range = k - i_range
    g[i_range] = 2.0 * a[j_range]          # (i,j) and (j,i) counted
    return g


def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 2_000_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the exact objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_maxconv(a)
        eta = eta0 / ((it + 1) ** 0.6)      # slower decay than 1/√t
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 250_000 == 0:
            cur = _objective_exact(a)
            print(f"[EG] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val

# Seed generation (including random mixtures and permutations)
# ------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_inverse_sqrt(n))

    # sine windows
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))

    # Gaussian family
    for sigma in np.linspace(0.08, 0.45, 8):
        seeds.append(_gaussian(n, sigma))

    # Power‑law
    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power(n, p))

    # Beta family (wide grid, including values < 0.5)
    a_vals = np.concatenate([np.linspace(0.1, 0.5, 5),
                             np.linspace(0.6, 3.0, 6)])
    b_vals = a_vals.copy()
    for a in a_vals:
        for b in b_vals:
            seeds.append(_beta_shape(n, a, b))

    # Low‑frequency curves
    for nf in (2, 3, 5, 7):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Mixtures of two random seeds
    base = len(seeds)
    for _ in range(250):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Permutation of the best previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())
        # also add a few random permutations of it
        for _ in range(12):
            perm = rng.permutation(n)
            seeds.append(previous[perm] / previous.sum())

    # final sanitising
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

                     500.0, 1000.0, 2000.0, 5000.0,

        val_opt = surrogate(seq_opt)

            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long exponentiated‑gradient refinement on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(5_000_000, remaining * 0.35 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_exp_iters,
                                                      eta0=1.0)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[EG] improved exact objective to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – random permutations of the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        perm_iters = 200
        print("\n--- Random permutation polishing ---")
        for _ in range(perm_iters):
            if time.time() > deadline:
                break
            perm = rng.permutation(n)
            cand = global_best_seq[perm]
            val = surrogate(cand)
            if val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = val
                print(f"[Perm] new best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final high‑β L‑BFGS polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(500,
                             int(min(3000, remaining * 0.04 * 1e5)))
        print("\n--- Final high‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[50000.0, 100000.0],
                                           max_iter_per_beta=per_beta_iters)
        val_final = surrogate(seq_final)
        print(f"[Final‑L‑BFGS] surrogate = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (only for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
