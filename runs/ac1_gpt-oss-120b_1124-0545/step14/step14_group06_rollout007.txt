"""
propose_candidate
-----------------
Search for a non‑negative step‑function that minimises the
``evaluate_sequence`` objective.

Algorithm
~~~~~~~~~
* Build a rich pool of ℓ1‑normalised seeds – analytic windows
  (exponential, power, Gaussian, logistic, beta, Tukey), random
  Dirichlet draws, and a very fine sweep of the exponential decay
  rate and power exponent.
* Pick the best few seeds according to the exact surrogate
  ``2·n·max(conv(a,a))`` (identical to ``evaluate_sequence`` when the
  sum of ``a`` is 1).
* Perform a cheap local search on the exponential‑decay and power
  parameters (randomised log‑scale perturbations) to tighten the best
  seed.
* Run a long exponentiated‑gradient (mirror‑descent on the simplex)
  optimisation starting from the refined seed.  The update uses the
  sub‑gradient of the active convolution entry and a diminishing step
  size ``η₀/√t``.  The loop stops when the global deadline is reached.
* Return the best sequence found and store it in the global variable
  ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (equals evaluate_sequence when sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    """Geometric decay, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(
                np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size – √t schedule
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # simplex projection (cheap because x stays non‑negative)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Local parameter search (tiny stochastic hill‑climb)
# ----------------------------------------------------------------------
def _local_search_exponential(n: int, init_rate: float,
                             rng: np.random.Generator,
                             deadline: float,
                             max_trials: int = 2000) -> tuple[np.ndarray, float]:
    """Random walk on the decay rate (log‑scale) to improve the exponential seed."""
    best_rate = init_rate
    best_seq = _exponential_decay(n, best_rate)
    best_val = _objective_surrogate(best_seq)

    for _ in range(max_trials):
        if time.time() > deadline:
            break
        # propose a new rate by Gaussian perturbation in log‑space
        log_r = math.log(best_rate)
        new_log_r = log_r + rng.normal(scale=0.005)   # 0.5 %‑ish changes
        new_rate = math.exp(new_log_r)
        new_rate = max(0.9, min(0.999999, new_rate))
        cand = _exponential_decay(n, new_rate)
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_rate = new_rate
            best_seq = cand
    return best_seq, best_val


def _local_search_power(n: int, init_p: float,
                        rng: np.random.Generator,
                        deadline: float,
                        max_trials: int = 2000) -> tuple[np.ndarray, float]:
    """Random walk on the power exponent."""
    best_p = init_p
    best_seq = _power(n, best_p)
    best_val = _objective_surrogate(best_seq)

    for _ in range(max_trials):
        if time.time() > deadline:
            break
        new_p = best_p * (1.0 + rng.uniform(-0.1, 0.1))
        new_p = max(0.05, min(10.0, new_p))
        cand = _power(n, new_p)
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_p = new_p
            best_seq = cand
    return best_seq, best_val

    print("\nTop surrogate values after seeding:")

        print(f"  #{i+1:2d}: {v:.6f}")

    # Optional local fine‑tuning of exponential / power parameters
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # focus on the best exponential seed (if present)
    exp_candidates = [s for s in top_seeds if np.isclose(s[0], s[0])]  # placeholder
    # (we simply re‑use the best seed if its shape is exponential‑like)
    # A cheap heuristic: try local search around the best exponential rate
    # that we know from the sweep (the one giving the smallest value).
    # Extract the rate that yielded the best exponential seed.
    # The sweep used eps = 1‑rate, so recover it:
    best_rate = None
    best_power = None
    for eps in np.linspace(0.00015, 0.0012, 120):
        rate = 1.0 - eps
        cand = _exponential_decay(n, rate)
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val, best_seq, best_rate = val, cand, rate
    # If a good exponential was found, run a small stochastic refinement
    if best_rate is not None:
        print("[Local] refining exponential rate")
        best_seq, best_val = _local_search_exponential(
            n, best_rate, rng, deadline, max_trials=2500)

    # Try a local refinement of the best power‑law seed
    # Find the best power exponent from the sweep:
    best_p = None
    for p in np.linspace(0.2, 3.5, 150):
        cand = _power(n, p)
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val, best_seq, best_p = val, cand, p
    if best_p is not None:
        print("[Local] refining power exponent")
        best_seq, best_val = _local_search_power(
            n, best_p, rng, deadline, max_trials=2500)

    # ------------------------------------------------------------------
    # Stage 1 – heavy exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1900))
        print(f"\n[EG] running up to {max_iters:,} iterations")
        best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                               max_iters=max_iters,
                                               eta0=0.23)
        print(f"[EG] surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store the result globally for reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This script explores the exponential‑decay and power‑law families with a very fine parameter sweep, refines the most promising candidate by stochastic local search on its defining parameters, and then applies a long exponentiated‑gradient mirror descent on the simplex to push the solution to a (near) local optimum.  All steps respect the 1000 s deadline, and the final vector is returned as a plain Python list and saved in `height_sequence_1`.
