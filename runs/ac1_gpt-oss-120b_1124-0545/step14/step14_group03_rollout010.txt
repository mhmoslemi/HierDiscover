"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The idea is to *flatten* the self‑convolution (the distribution of i+j)
as much as possible.  The target distribution is the uniform one on
{0,…,2n‑2}.  We minimise the mean‑squared error between the current
convolution and this uniform target using Adam on the soft‑max logits.
Periodically we evaluate the true objective and keep the best vector.
Finally a short L‑BFGS run on a smooth log‑sum‑exp surrogate is used for
polishing.

The whole routine respects the 1000 s budget and returns a plain list
of floats that can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ------------------------------------------------------------
# Basic numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ------------------------------------------------------------
# Surrogate (log‑sum‑exp) utilities – used for final polishing
# ------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Function and gradient for the log‑sum‑exp surrogate of max(conv)."""
    a = _softmax(v)
    conv = _conv_fft(a, a)                     # length 2n‑1
    shift = np.max(conv)                       # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp                     # soft‑weights
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    L‑BFGS‑B on the smooth surrogate.  Variable = logits of the probability vector.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_objective(v_local, n, float(beta))

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta}")
            break
        except Exception as e:          # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta})")
            break

    return _softmax(v)


# ------------------------------------------------------------
# Adam optimisation of the MSE to a uniform convolution
# ------------------------------------------------------------
def _adam_flatten(start_vec: np.ndarray,
                  n: int,
                  deadline: float,
                  target: np.ndarray,
                  max_iters: int = 150_000,
                  lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam on the logits of the probability vector.
    Loss = mean((conv(a,a) - target)**2).
    Returns the best vector found and its exact objective value.
    """
    # initialise
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    tgt_len = target.shape[0]

    best_a = _softmax(v)
    best_obj = 2.0 * n * np.max(_conv_fft(best_a, best_a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)                # length 2n‑1
        diff = conv - target
        loss = np.mean(diff * diff)

        # gradient w.r.t. a (MSE part)
        w = (2.0 / tgt_len) * diff           # (∂loss/∂conv)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = full[n - 1:n - 1 + n]       # slice to length n

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v * grad_v)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        v = v - lr0 * m_hat / (np.sqrt(v_hat_corr) + eps)

        # occasional bookkeeping
        if it % 5_000 == 0:
            cur_obj = 2.0 * n * np.max(_conv_fft(_softmax(v), _softmax(v)))
            print(f"[Adam] it={it:,} loss={loss:.2e} obj={cur_obj:.6f}")
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = _softmax(v).copy()

    # final check
    final_a = _softmax(v)
    final_obj = 2.0 * n * np.max(_conv_fft(final_a, final_a))
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = final_a.copy()

    return best_a, best_obj


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size -------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # uniform target for the convolution --------------------------------
    tgt_len = 2 * n - 1
    uniform_target = np.full(tgt_len, 1.0 / tgt_len, dtype=np.float64)

    # ------------------------------------------------------------------
    # Seed pool ---------------------------------------------------------
    seeds = [
        _uniform(n),
        _gaussian(n, sigma_frac=0.15),
        _gaussian(n, sigma_frac=0.30),
        _beta_shape(n, alpha=2.0, beta=2.0),
        _random_dirichlet(n, conc=0.5, rng=rng),
        _random_exponential(n, rng=rng)
    ]

    # keep previous solution if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,) and prev_arr.sum() > 0:
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # Helper to evaluate the true objective (surrogate = exact here)
    def _eval_obj(a_vec: np.ndarray) -> float:
        conv = _conv_fft(a_vec, a_vec)
        return float(2.0 * n * np.max(conv))

    # ------------------------------------------------------------------
    # Adam flattening on each seed
    max_total_iters = 150_000
    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Adam optimisation started from seed #{idx + 1} ---")
        seq, val = _adam_flatten(seed_vec, n, deadline,
                                 uniform_target,
                                 max_iters=max_total_iters,
                                 lr0=0.5)
        print(f"[Seed {idx + 1}] best objective = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Update] new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with L‑BFGS on the smooth max‑conv surrogate
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(500, int(min(2000, remaining * 0.08 * 1e5)))
        print("\n--- Final L‑BFGS polishing (high β) ---")
        polished = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                          beta_schedule=[2000.0, 5000.0, 10000.0],
                                          max_iter_per_beta=per_beta)
        polished_val = _eval_obj(polished)
        print(f"[Polish] objective = {polished_val:.6f}")
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished.copy()
            print(f"[Update] improved after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (purely for reporting)
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible warm‑start in the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
