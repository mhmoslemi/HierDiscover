def _adam_optimize_extended(start, n, deadline,
                           max_iters=1_500_000,
                           init_lr=0.04,
                           p_start=1.0,
                           p_factor=1.5,
                           p_step=3000,
                           p_max=300.0):
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    p = p_start
    while time.time() < deadline and t < max_iters:
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # softmax weights
        shifted = conv - max_conv
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)  # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv == max_conv).astype(float)
            w_sum = w.sum()
        w /= w_sum

        # gradient
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)
        grad = 2.0 * grad_full[n-1:n-1+n]

        # Adam update
        t += 1
        m = 0.9 * m + 0.1 * grad  # could use beta1, beta2 param; for simplicity use fixed
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = _uniform(n)

        # increase p schedule
        if t % p_step == 0:
            p = min(p * p_factor, p_max)

    final_val = _objective_surrogate(best_a)
    return best_a, final_val

"""
Propose a step‐function that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

with a_i ≥ 0.

The algorithm builds a very diverse pool of ℓ₁‑normalised seeds
(uniform, analytic windows, random Dirichlet/exponential draws,
low‑frequency cosine mixtures, and dense parameter sweeps), keeps the
best few according to the exact surrogate 2·n·max(conv), and then
polishes them through several optimisation stages:

1. **Fast EG (exponentiated‑gradient) refinements** on the top seeds.
2. **Adam on a smooth soft‑max surrogate** with a slowly increasing
   smoothing parameter p up to a very large value (≫300) – this
   approximates the true max‑convolution gradient very closely.
3. **Sub‑gradient EG on the exact active set** (all indices whose
   convolution value is within a tiny tolerance of the maximum).
4. **Simulated‑annealing polishing** for large‑scale moves.
5. **Final long EG polishing** to squeeze the last bits.

All stages respect the global deadline (default 1000 s).  The best
vector found is stored in the global variable ``height_sequence_1`` for
future reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search (sum(seq) = 1 is enforced):
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection – richer and deeper sweeps
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a large, diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families (same as before)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.25, 0.3):
        seeds.append(_gaussian(n, sigma))
    for rate in (0.998, 0.999, 0.9995, 0.9998):
        seeds.append(_exponential_decay(n, rate))
    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        seeds.append(_power(n, p))
    for alpha in (0.5, 0.8, 1.0, 1.2):
        for beta in (1.0, 1.5, 2.0, 3.0, 5.0):
            seeds.append(_beta_shape(n, alpha, beta))
    for scale in (5.0, 10.0, 20.0, 40.0):
        seeds.append(_logistic_window(n, scale))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # ------------------------------------------------------------------
    # Fine sweeps – extended (denser) parameter grids
    # ------------------------------------------------------------------
    # exponential rate
    for rate in np.linspace(0.9985, 0.9999, 40):
        seeds.append(_exponential_decay(n, rate))

    # power exponent (including negatives)
    for p in np.linspace(-2.0, -0.1, 22):
        seeds.append(_power(n, p))
    for p in np.linspace(0.2, 3.0, 40):
        seeds.append(_power(n, p))

    # sine gamma (0.2 – 0.9) – flatter windows
    for gamma in np.linspace(0.2, 0.9, 15):
        seeds.append(_sine_window(n, gamma))

    # logistic scale sweep
    for scale in np.linspace(2.0, 80.0, 25):
        seeds.append(_logistic_window(n, scale))

    # beta parameter fine grid
    for alpha in np.linspace(0.2, 2.0, 15):
        for beta in np.linspace(0.5, 5.0, 15):
            seeds.append(_beta_shape(n, float(alpha), float(beta)))

    # low‑frequency curves with more components
    for nf in (5, 6, 7):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform (to encourage a tiny flat component)
    for mix_alpha in (0.1, 0.3, 0.5, 0.7):
        curve = _lowfreq_curve(n, 3, rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # extra Dirichlet draws (very sparse and very dense regimes)
    for conc in np.geomspace(0.01, 2.0, 10):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ------------------------------------------------------------------
    # Previous solution (if any)
    # ------------------------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final safety – ensure all seeds are ℓ₁‑normalised
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient at the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # multiplicative update (clip exponent for safety)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _adam_optimize_extended(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_lr: float = 0.04,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    p_start: float = 1.0,
    p_factor: float = 1.5,
    p_step: int = 3000,
    p_max: float = 300.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth surrogate
        F_p(a) = (1/p)·log Σ_k exp(p·conv_k(a))

    The smoothing parameter p is multiplied by `p_factor` every `p_step`
    iterations, capped at `p_max`.  A large `p_max` (e.g. 2000) makes
    the surrogate a very tight approximation of the true max.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p_start

    while time.time() < deadline and t < max_iters:
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # soft‑max weights
        shifted = conv - max_conv               # ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)     # safety
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                       # under‑flow fallback
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the smooth surrogate
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)        # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        # increase smoothness gradually
        if t % p_step == 0:
            p = min(p * p_factor, p_max)

    final_val = _objective_surrogate(best_a)
    return best_a, final_val


def _subgradient_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    epsilon: float = 1e-12,
    eta0: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    EG on the exact sub‑gradient of the max‑convolution.
    All indices whose convolution value lies within `epsilon` of the
    maximum are treated as active and contribute equally to the gradient.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        max_conv = conv.max()

        # active set (absolute tolerance)
        tol = epsilon * max(1.0, max_conv)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        a_rev = x[::-1]
        grad_full = _conv_fft(w, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        eta = eta0 / math.sqrt(it)
        delta = -eta * grad
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # simplex projection
        s = x.sum()
        if s > 0.0:
            x = x / s
        else:
            x = _uniform(n)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        # two families of moves
        if rng.random() < 0.55:
            # reshuffle a small random block
            k = rng.integers(4, 13)
            idx = rng.choice(n, size=k, replace=False)
            tot = cur[idx].sum()
            if tot <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Assemble a rich set of seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # Evaluate surrogate for all seeds and keep the best few.
    # ------------------------------------------------------------------
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, v) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – quick EG refinement on the very best seeds.
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick EG refinement (selected seeds) ---")
        for idx, seed_vec in enumerate(best_seeds[:10]):
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            # allocate iterations proportionally to remaining budget
            max_it = int(min(120_000, remaining * 1500))
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=max_it,
                                          eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation with a high smoothing parameter.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam optimisation (large p) ---")
        remaining = deadline - time.time()
        max_adam_iters = int(min(1_200_000, remaining * 1600))
        best_seq, best_val = _adam_optimize_extended(
            best_seq,
            n,
            deadline,
            max_iters=max_adam_iters,
            init_lr=0.04,
            p_start=1.0,
            p_factor=1.5,
            p_step=2500,
            p_max=2000.0,
        )
        print(f"[Adam] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – sub‑gradient (exact max) EG refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient EG refinement ---")
        remaining = deadline - time.time()
        max_subgrad_iters = int(min(500_000, remaining * 1800))
        best_seq, best_val = _subgradient_optimize(
            best_seq, n, deadline,
            max_iters=max_subgrad_iters,
            epsilon=1e-12,
            eta0=0.22,
        )
        print(f"[Sub‑grad EG] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_sa_iters = int(min(400_000, remaining * 1900))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_sa_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final long EG polishing (very cheap, helps descend).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_final_it = int(min(300_000, remaining * 2100))
        if max_final_it > 0:
            best_seq, best_val = _expgrad_optimize(
                best_seq, n, deadline,
                max_iters=max_final_it,
                eta0=0.18,
            )
            print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
