"""
Hybrid optimizer for the non‑negative step‑function minimisation problem.

The algorithm proceeds in four phases:

0️⃣  **Parameter‑mix search** – Randomly mix a few analytically‑defined
    shapes (uniform, sine‑window, Gaussian, power‑law, beta, Tukey,
    exponential‑decay).  Each mixture is ℓ₁‑normalised and evaluated
    with the exact surrogate 2·n·max(conv(a,a)).  The best few candidates
    are retained.

1️⃣  **Local refinement** – For the current best vector we run a short
    sub‑gradient descent on the exact objective and a longer exponential‑
    gradient (mirror‑descent) polish.  Both respect the probability
    simplex.

2️⃣  **LP‑based polishing** – The vector is down‑sampled to a modest size
    (≈1000).  For this coarse representation we solve the linear program

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M      for all k
                    b_j ≥ 0,

    where M = max(f*f).  The solution b ₀ is re‑normalised to unit ℓ₁‑norm,
    then mixed with the current f via a convex combination with a
    decreasing step size.  A handful of such LP steps dramatically
    reduces the max‑convolution bound.

3️⃣  **Final polishing** – The up‑sampled result is again polished with the
    sub‑gradient and exponential‑gradient stages.

All stages respect a global time budget (default 1000 s) and fall back
to the best sequence discovered so far, guaranteeing a valid answer.
"""

import time
import math
import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑p = 1):
        2·n·max(conv(p,p)).
    This is exactly the value returned by ``evaluate_sequence`` after the
    security clamping (because the sum is 1 and the values are already in range).
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers (sub‑gradient & exponential‑gradient)
# ----------------------------------------------------------------------
def _subgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Additive sub‑gradient descent on the exact objective."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 20_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow/underflow
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val

# Linear‑programming polishing (coarse scale)
# ----------------------------------------------------------------------
def _lp_polish(
    f_coarse: np.ndarray,
    deadline: float,
    max_iters: int = 20,
    step0: float = 0.3,
) -> np.ndarray:
    """

    Returns the polished coarse vector (still ℓ₁‑normalised).
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        print("[LP] cvxpy not available – skipping LP polishing.")
        return f_coarse

    m = f_coarse.shape[0]
    f = f_coarse.copy()
    t = step0
    it = 0

    # Pre‑build the Toeplitz‑like convolution matrix (sparse)
    offsets = np.arange(m)
    data = [np.full(m, f[i]) for i in range(m)]
    A = sp.diags(data, offsets, shape=(2 * m - 1, m), format='csc')

    while time.time() < deadline and it < max_iters:
        M = np.max(_conv_fft(f, f))

        # Re‑build matrix with the *current* f (A depends on f)
        data = [np.full(m, f[i]) for i in range(m)]
        A = sp.diags(data, offsets, shape=(2 * m - 1, m), format='csc')

        b = cp.Variable(m)
        constraints = [A @ b <= M, b >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=2000, verbose=False)
        except Exception:  # pragma: no cover
            print("[LP] solver failed at iteration", it)
            break

        if b.value is None:
            print("[LP] no solution at iteration", it)
            break

        b_val = np.maximum(b.value, 0.0)
        sum_b = b_val.sum()
        if sum_b <= 0:
            break
        g = b_val / sum_b

        # convex combination
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        # keep if we improve
        old_val = _objective_surrogate(f)
        new_val = _objective_surrogate(f_new)
        if new_val < old_val:
            f = f_new
            print(f"[LP] iteration {it+1:2d} improved surrogate to {new_val:.6f}")
        else:
            # if no improvement, shrink step size
            t = max(0.01, t * 0.7)

        it += 1
        # slowly decay step size as well
        t = max(0.01, t * 0.97)

    return f

def _mix_random_shapes(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random mixture of a few analytic shapes.
    The number of components is 2 or 3, weights are Dirichlet‑distributed.
    """
    shape_fns = [
        lambda: _uniform(n),
        lambda: _sine_window(n, gamma=rng.uniform(0.5, 4.0)),
        lambda: _gaussian(n, sigma_frac=rng.uniform(0.15, 0.45)),
        lambda: _power(n, p=rng.uniform(0.3, 3.0)),
        lambda: _beta_shape(n,
                            alpha=rng.uniform(0.5, 4.0),
                            beta=rng.uniform(0.5, 4.0)),
        lambda: _tukey_window(n, alpha=rng.uniform(0.1, 0.9)),
        lambda: _exponential_decay(n, rate=rng.uniform(0.995, 0.9999)),
    ]
    k = rng.integers(2, 4)                     # 2 or 3 components
    idxs = rng.choice(len(shape_fns), size=k, replace=False)
    weights = rng.dirichlet(np.ones(k))
    mix = np.zeros(n, dtype=np.float64)
    for w, i in zip(weights, idxs):
        mix += w * shape_fns[i]()
    mix = mix / mix.sum()
    return mix

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # Length of the sequence – 4000 gives a good trade‑off between accuracy
    # and speed for FFT‑based convolutions.
    n = 4000
    print(f"\n--- Sequence length: n = {n} ---")

    # ------------------------------------------------------------------
    # 0️⃣  Seed generation via random mixtures and analytic families
    # ------------------------------------------------------------------
    print("\n--- Seed generation (mixture search) ---")
    best_seq = None
    best_val = np.inf

    # always keep the previous solution if it exists
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and previous:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n:
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 0:
                arr = arr / s
                best_seq = arr.copy()
                best_val = _objective_surrogate(best_seq)
                print(f"[Prev] starting surrogate = {best_val:.6f}")

    # deterministic analytic families (helpful baselines)
    deterministic = [
        _uniform,
        _triangular,
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _gaussian(n, sigma_frac=0.25),
        lambda n: _power(n, p=1.0),
        lambda n: _beta_shape(n, alpha=2.0, beta=2.0),
        lambda n: _tukey_window(n, alpha=0.5),
        lambda n: _exponential_decay(n, rate=0.9995),
    ]
    for fn in deterministic:
        cand = fn(n)
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Det] new best surrogate = {best_val:.6f}")

    # Random mixture search – keep looping until time runs out or we have
    # collected a modest number of candidates.
    mix_trials = 0
    while time.time() < deadline - 0.5 and mix_trials < 2500:
        cand = _mix_random_shapes(n, rng)
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Mix] improved surrogate to {best_val:.6f} (trial {mix_trials})")
        mix_trials += 1

    print(f"\nBest surrogate after seed phase: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Local refinement (sub‑gradient + exponential‑gradient)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Sub‑gradient refinement ---")
        remaining = deadline - time.time()
        max_it = int(min(200_000, remaining * 300))
        best_seq, best_val = _subgrad_optimize(best_seq, n, deadline, max_iters=max_it, eta0=0.7)
        print(f"[SubGrad] surrogate = {best_val:.6f}")

    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        max_it = int(min(800_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(best_seq, n, deadline, max_iters=max_it, eta0=0.25)
        print(f"[ExpGrad] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  LP‑based polishing on a coarse grid
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- LP polishing (coarse) ---")
        # Down‑sample to a manageable size (m ≈ 1000)
        m = 1000
        if n % m != 0:
            # fallback to linear interpolation
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, m)
            f_coarse = np.interp(x_new, x_old, best_seq)
        else:
            block = n // m
            f_coarse = best_seq.reshape(m, block).mean(axis=1)
        f_coarse = np.maximum(f_coarse, 0.0)
        f_coarse = f_coarse / f_coarse.sum()

        f_coarse = _lp_polish(f_coarse, deadline, max_iters=30, step0=0.3)

        # Upsample back to full resolution (piecewise‑constant)
        if n % m == 0:
            best_seq = np.repeat(f_coarse, n // m)
        else:
            x_old = np.linspace(0, 1, m)
            x_new = np.linspace(0, 1, n)
            best_seq = np.interp(x_new, x_old, f_coarse)

        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_val = _objective_surrogate(best_seq)
        print(f"[LP] surrogate after up‑sampling = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final polish (sub‑gradient + exponential‑gradient) – short pass
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final short sub‑gradient pass ---")
        remaining = deadline - time.time()
        max_it = int(min(80_000, remaining * 300))
        best_seq, best_val = _subgrad_optimize(best_seq, n, deadline, max_iters=max_it, eta0=0.5)
        print(f"[Final SubGrad] surrogate = {best_val:.6f}")

    if best_seq is not None and time.time() < deadline:
        print("\n--- Final short exponential‑gradient pass ---")
        remaining = deadline - time.time()
        max_it = int(min(200_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(best_seq, n, deadline, max_iters=max_it, eta0=0.2)
        print(f"[Final ExpGrad] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
