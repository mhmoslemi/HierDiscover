def _lbfgs_optimize(start, n, deadline, p_target=300, max_iter=200):

    a0 = start.copy()
    # Ensure sum=1
    a0 = a0 / a0.sum()
    # define b param
    b0 = np.log(a0 + 1e-30)  # log of a -> initial b

    # clamp p increasing schedule maybe not needed; we can set p_target.

    # objective wrapper:
    def fun_and_grad(b):

        maxb = np.max(b)
        expb = np.exp(b - maxb)  # stable
        s = expb.sum()
        a = expb / s  # softmax
        conv = _conv_fft(a, a)

        p = p_target

        maxc = np.max(p * conv)
        expc = np.exp(p * conv - maxc)

"""
Improved search for a non‑negative step function minimising

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

The algorithm proceeds in four stages:

1. **Rich seed pool** – deterministic analytic windows, a dense sweep of
   parameterised families (exponential, power, beta, logistic, tukey,
   plateau+exponential, double‑exponential) and many random draws
   (Dirichlet, exponential, low‑frequency cosine curves).  All candidates are
   ℓ₁‑normalised and scored with the exact surrogate
   `2·n·max(conv(a,a))`.

2. **Fast mirror‑descent (exponential‑gradient) refinement** of the best
   seeds, using a sub‑gradient of the active convolution shift.

3. **Smooth optimisation** – Adam on the smooth soft‑max surrogate
   `F_p(a) = (1/p)·log Σ_k exp(p·conv_k(a))` with a schedule that raises the
   smoothing parameter `p` up to ≈800, followed by an optional L‑BFGS‑B
   optimisation (SciPy) on the same surrogate but with an explicit soft‑max
   parametrisation `a = softmax(b)`.

4. **Final polishing** – a long exponential‑gradient run and a short
   simulated‑annealing (SA) phase.

Throughout the run time budget (default 1000 s) is monitored and each stage
receives a share of the remaining time.  The best sequence found is stored in
the global variable `height_sequence_1` and returned as a plain Python list.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (2·n·max(conv(seq,seq))) for ℓ₁‑normalised seq."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _plateau_exponential(n: int, plateau_frac: float, rate: float) -> np.ndarray:
    """Plateau (constant) of length plateau_frac·n followed by exponential decay."""
    plateau_len = max(1, int(n * plateau_frac))
    a = np.zeros(n, dtype=np.float64)
    a[:plateau_len] = 1.0
    tail_len = n - plateau_len
    if tail_len > 0:
        i = np.arange(tail_len, dtype=np.float64)
        a[plateau_len:] = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate1: float, rate2: float, mix: float) -> np.ndarray:
    """Convex combination of two decreasing exponentials."""
    i = np.arange(n, dtype=np.float64)
    a = mix * (rate1 ** i) + (1.0 - mix) * (rate2 ** i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_extended_seeds(n: int, rng: np.random.Generator,
                           previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a large diversified set of ℓ₁‑normalised seeds."""
    seeds = []

    # deterministic families (same as before)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.25, 0.3):
        seeds.append(_gaussian(n, sigma))
    for rate in (0.998, 0.999, 0.9995, 0.9998):
        seeds.append(_exponential_decay(n, rate))
    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        seeds.append(_power(n, p))
    for alpha in (0.5, 0.8, 1.0, 1.2):
        for beta in (1.0, 1.5, 2.0, 3.0, 5.0):
            seeds.append(_beta_shape(n, alpha, beta))
    for scale in (5.0, 10.0, 20.0, 40.0):
        seeds.append(_logistic_window(n, scale))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # new families ------------------------------------------------------
    # plateau + exponential
    for frac in (0.05, 0.1, 0.2, 0.35):
        for rate in (0.9995, 0.9998, 0.9999):
            seeds.append(_plateau_exponential(n, frac, rate))

    # double exponential mixtures
    for r1 in (0.998, 0.999, 0.9995):
        for r2 in (0.9995, 0.9998, 0.9999):
            for mix in (0.2, 0.5, 0.8):
                seeds.append(_double_exponential(n, r1, r2, mix))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves (more frequencies)
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # fine sweeps of key parameters ------------------------------------
    # exponential rate sweep (very fine)
    for rate in np.linspace(0.9985, 0.99995, 60):
        seeds.append(_exponential_decay(n, float(rate)))
    # power exponent sweep (fine)
    for p in np.linspace(0.2, 3.5, 60):
        seeds.append(_power(n, float(p)))
    # logistic scale sweep
    for scale in np.linspace(4.0, 60.0, 40):
        seeds.append(_logistic_window(n, float(scale)))
    # beta parameter sweep (dense)
    for alpha in np.linspace(0.3, 2.0, 12):
        for beta in np.linspace(0.5, 5.0, 12):
            seeds.append(_beta_shape(n, float(alpha), float(beta)))

    # mixtures with uniform to bias towards flatness
    for mix_alpha in (0.1, 0.3, 0.6):
        for nf in (2, 3):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # include the previously best solution if supplied
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final cleanup – ensure ℓ₁‑normalisation and non‑negativity
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # compute full convolution to locate the active shift
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size – √t schedule
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # safety for exp
        x = x * np.exp(delta)

        # project back to simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.04,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 1.0, p_factor: float = 1.6,
                          p_step: int = 2500,
                          max_total_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        F_p(a) = (1/p)·log Σ_k exp(p·conv_k(a))
    with a gentle increase of the smoothing parameter p.
    """
    a = start.copy()
    best_a = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        # current convolution and true objective
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # compute soft‑max weights for the current p
        shifted = conv - max_conv               # ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)     # safety against overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                       # underflow fallback
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the smooth surrogate
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)        # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        # increase smoothing parameter p
        if t % p_step == 0:
            p = min(p * p_factor, 800.0)  # cap higher for better approximation

    final_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))
    return best_a, final_val


def _lbfgs_optimize(start: np.ndarray, n: int, deadline: float,
                    p_target: float = 300.0, max_iter: int = 200) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B optimisation on the smooth surrogate using a soft‑max
    parametrisation a = softmax(b).  Falls back gracefully if SciPy is unavailable.
    """
    try:
        import scipy.optimize
    except Exception:
        # SciPy not present – skip this stage
        return start, _objective_surrogate(start)

    # initialise b from a = start (already ℓ₁‑normalised)
    eps = 1e-30
    b0 = np.log(start + eps)

    def fun_and_grad(b):
        # soft‑max mapping
        maxb = np.max(b)
        expb = np.exp(b - maxb)
        a = expb / expb.sum()

        conv = _conv_fft(a, a)

        # soft‑max over convex combination of exponentials
        pc = p_target * conv
        maxc = np.max(pc)
        expc = np.exp(pc - maxc)
        w = expc / expc.sum()
        # smooth max‐value (log‑sum‑exp) divided by p
        val = (np.log(expc.sum()) + maxc) / p_target

        # gradient wrt a
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)
        grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

        # chain rule to get gradient wrt b
        dot = np.dot(a, grad_a)
        grad_b = a * (grad_a - dot)
        return float(val), grad_b.astype(np.float64)

    options = {"maxiter": max_iter, "disp": False}
    res = scipy.optimize.minimize(fun_and_grad, b0, method="L-BFGS-B", jac=True,
                                  options=options)

    b_opt = res.x
    maxb = np.max(b_opt)
    expb = np.exp(b_opt - maxb)
    a_opt = expb / expb.sum()
    return a_opt, _objective_surrogate(a_opt)


def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA with two move families – reshuffle a block or a tiny transfer."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99992, min_temp)

        if rng.random() < 0.55:
            # block reshuffle (size 4‑12)
            k = rng.integers(4, 13)
            idx = rng.choice(n, size=k, replace=False)
            tot = cur[idx].sum()
            if tot <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two entries
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.06)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] iteration {it:7d} – best surrogate {best_val:.6f}")

    return best, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries strike a good balance.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Assemble a very diverse pool of seeds.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_extended_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # Quick surrogate evaluation and keep the most promising seeds.
    # ------------------------------------------------------------------
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, v) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – short EG refinement on a few elite seeds.
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick EG refinement (few seeds) ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # allocate a modest budget per seed, respecting remaining time
            max_it = int(min(120_000, remaining * 2000))
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=max_it, eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on a smooth surrogate (larger p).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam smooth optimisation (higher p) ---")
        remaining = deadline - time.time()
        max_adam_iters = int(min(1_800_000, remaining * 2100))
        best_seq, best_val = _adam_smooth_optimize(best_seq, n, deadline,
                                                   init_lr=0.04,
                                                   p0=1.0, p_factor=1.6,
                                                   p_step=2500,
                                                   max_total_iters=max_adam_iters)
        print(f"[Adam] best surrogate after smoothing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional L‑BFGS‑B refinement on the same smooth objective.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS‑B refinement (if SciPy available) ---")
        remaining = deadline - time.time()
        # We limit the number of LBFGS iterations so that we stay within budget.
        max_lbfgs_iters = 150
        try:
            cand, val = _lbfgs_optimize(best_seq, n, deadline,
                                        p_target=500.0, max_iter=max_lbfgs_iters)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[L‑BFGS] improved surrogate to {best_val:.6f}")
        except Exception as e:
            # In case SciPy is not present or any error occurs, continue.
            print("[L‑BFGS] skipped due to exception:", e)

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 2200))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_iters)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final long EG polishing (helps escape plateaus).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(500_000, remaining * 2300))
        if max_iters > 0:
            best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                                   max_iters=max_iters,
                                                   eta0=0.22)
            print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation with the provided evaluate_sequence().
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
