def _grad_descent_optimize(start, n, deadline, max_iters, lr_schedule function):

    x = start.copy()
    best_x = x.copy()
    best_val = surrogate(x)

    for t in range(1, max_iters+1):
        if time.time() > deadline: break

        conv = _conv_fft(x, x)
        k = np.argmax(conv)   # index of max conv
        # compute gradient
        # g[i] = 2 * x[k - i] if indices in bounds else 0
        # compute using arrays: shift = k
        g = np.zeros_like(x)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size schedule: lr = lr0 / sqrt(t)
        lr = 0.5 / math.sqrt(t)  # hyperparameter may adjust
        x = x - lr * g

        # projection onto simplex
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val

"""
Step‑function optimizer for ``evaluate_sequence``.

The objective for a non‑negative vector a (sum a_i = 1) is
    obj(a) = 2·n·max_k (a * a)[k]     (conv is the full linear convolution).
Uniform a gives obj = 2, but better (lower) values exist.
We treat the problem as a min‑max quadratic program and apply a hierarchy
of cheap stochastic / deterministic refinements:

1. **Seed generation** – a large, diversified pool of normalized vectors:
   uniform, triangular, Gaussian, Tukey, Beta, power‑law, Dirichlet
   draws, sparse supports and mixtures.

2. **Additive sub‑gradient descent** – iteratively moves mass away from the
   shift ``k*`` where the convolution attains its maximum.
   Gradient at iteration t is  g_i = 2·a_{k*‑i} (zero outside the range);
   after a step we project onto the probability simplex.

3. **Multiplicative (exponential‑gradient) polishing** – a mirror‑descent
   update that often yields a finer reduction of the max‑convolution peak.

4. **Simulated‑annealing fine‑tuning** – random reshuffles of a small
   subset, pairwise mass transfers and occasional swaps, with an exponential
   cooling schedule.

All stages respect the global ``budget_s`` (≈1000 s) deadline.
The best sequence found is stored in the global name ``height_sequence_1``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Utility helpers (no closures, no lambdas)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v_i ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(p,p)) for a probability vector (∑p = 1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(
                np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _power_law(n: int, exponent: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-exponent}  (monotonically decreasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(i, exponent)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()

# Optimisers
# ----------------------------------------------------------------------
def _grad_descent_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    lr0: float = 0.5,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent on the exact max‑convolution.
    Gradient of max_k (a*a)[k] at the active shift k* is
        g_i = 2·a_{k*‑i}   (zero outside the allowed index range).
    After every step we project onto the simplex.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # compute full convolution and locate the maximum index
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # gradient = 2 * shifted copy of x
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # learning rate schedule η_t = lr0 / sqrt(t)
        lr = lr0 / math.sqrt(it)
        x = x - lr * g

        # projection onto the simplex (and positivity)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[Grad] iter={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.30,
    sparsity_thr: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with KL (multiplicative) updates.
    After each update values < sparsity_thr are zeroed to encourage sparsity.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow in exp()
        x = x * np.exp(delta)

        if sparsity_thr > 0.0:
            mask = x < sparsity_thr
            if np.any(mask):
                x[mask] = 0.0

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.2,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Standard SA with three move types and exponential cooling."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.99993, min_temp)   # slower cooling than the previous script

        move_type = rng.integers(3)

        if move_type == 0:   # reshuffle a random subset
            k_sub = rng.integers(6, 14)          # 6‑13 positions
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc

        elif move_type == 1:   # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.03)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        else:                  # swap two entries (helps escape plateaus)
            i, j = rng.choice(n, size=2, replace=False)
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # keep feasibility (non‑negative) and normalise
        s = cand.sum()
        if s <= 0.0:
            continue
        cand = cand / s

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse pool of normalised candidate vectors."""
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.15, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    for a, b in ((2.0, 2.0), (0.8, 0.8), (0.5, 1.5), (1.5, 0.5)):
        seeds.append(_beta_shape(n, a, b))
    for alpha in (0.3, 0.6, 0.8):
        seeds.append(_tukey_window(n, alpha))
    for exp in (0.4, 0.7, 1.0, 1.3):
        seeds.append(_power_law(n, exp))

    # random Dirichlet draws (very sparse and fairly dense)
    for conc in (0.02, 0.05, 0.2, 0.5):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse supports of various sizes
    for m in (150, 250, 350, 450):
        for _ in range(3):
            seeds.append(_random_sparse_support(n, m, rng))

    # mixtures of uniform and a sparse support
    for mix_alpha in (0.2, 0.4, 0.6):
        for m in (200, 300):
            sparse = _random_sparse_support(n, m, rng)
            mixed = _mix_two(_uniform(n), sparse, mix_alpha)
            seeds.append(mixed)

    # Low‑frequency random cosine‑sum curves
    for nf in (2, 3, 4):
        for _ in range(3):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                seeds.append(a / s)

    # previous best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # final clean‑up – assure proper normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

def propose_candidate(seed: int = 271828, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function respects a hard time limit (≈budget_s s)
    and stores the best ever discovered vector in the global variable
    ``height_sequence_1`` for possible warm‑starts in later runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Length of the step function – 4000 is a good compromise.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising a step function of length {n} ===")

    # ------------------------------------------------------------------
    # Stage 0 – seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # evaluate surrogate for each seed and keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k], 1):
        print(f"  #{i:02d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – additive sub‑gradient descent (multiple starts)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (multiple starts) ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            max_iters = int(min(180_000, remaining * 250))   # heuristic
            cand, val = _grad_descent_optimize(seed_vec, n, deadline, lr0=0.6,
                                               max_iters=max_iters)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Grad] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – sparse exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1800))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline,
            max_iters=max_iters,
            eta0=0.30,
            sparsity_thr=1e-9,
        )
        print(f"[ExpGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing refinement ---")
        remaining = deadline - time.time()
        max_iters = int(min(900_000, remaining * 2100))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters, init_temp_factor=0.2
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional short greedy sweep (very cheap moves) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final greedy sweep (if time permits) ---")
        rng = np.random.default_rng(seed + 12345)
        cur = best_seq.copy()
        cur_val = best_val
        it = 0
        while time.time() < deadline and it < 300_000:
            it += 1
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.015)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = cand / cand.sum()
            new_val = _objective_surrogate(cand)
            if new_val < cur_val:
                cur, cur_val = cand, new_val
                if new_val < best_val:
                    best_val, best_seq = new_val, cand.copy()
                    print(f"[Greedy] it={it:7d} new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
