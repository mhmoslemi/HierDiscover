def propose_candidate(seed=123, budget_s=1000, **kwargs):
    # set random seed for reproducibility
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 4000

    # step 0: gather seeds
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(previous, dtype=np.float64)

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    # evaluate surrogate for each; store as list of tuples
    scores = [(seed_vec, _objective_surrogate(seed_vec)) for seed_vec in seeds]
    scores.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in scores[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"Initial best surrogate: {best_val:.6f}")

    # step 1: random convex combos of top seeds
    if time.time() < deadline:
        print("\n--- Random convex combinations of top seeds ---")
        M = len(top_seeds)
        seeds_mat = np.vstack(top_seeds)  # shape (M,n)
        combos = 10000
        for i in range(combos):
            if time.time() >= deadline: break
            # sample Dirichlet weights, concentration parameter medium
            conc = 0.5  # moderate sparsity
            w = rng.dirichlet(np.full(M, conc))
            mixed = w @ seeds_mat
            # ensure normalised sum=1
            mixed = mixed / mixed.sum()
            val = _objective_surrogate(mixed)
            if val < best_val:
                best_val, best_seq = val, mixed.copy()
                # print occasional improvement
                if i % 500 == 0:
                    print(f"[Combo] iter {i}, improved to {best_val:.6f}")

    # step 2: subgradient projected descent
    if time.time() < deadline:
        print("\n--- Subgradient projected descent ---")
        # choose hyperparameters
        eta0 = 0.5
        max_iters = 50000
        best_seq_sub = best_seq.copy()
        best_val_sub = best_val
        a = best_seq.copy()
        for it in range(1, max_iters+1):
            if time.time() >= deadline:
                break
            conv = _conv_fft(a, a)
            max_c = conv.max()
            val = 2.0 * n * max_c
            # track best
            if val < best_val_sub:
                best_val_sub, best_seq_sub = val, a.copy()
                print(f"[Subgrad] iter {it}, new best {best_val_sub:.6f}")
            # compute subgradient for active shift(s)
            active = np.where(conv >= max_c - 1e-12)[0]
            # if multiple, set weight = 1/len(active)
            weight = 1.0 / len(active)
            g = np.zeros_like(a)
            for k in active:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end+1)
                rev = k - idx
                g[idx] += weight * 2.0 * a[rev]
            # step size
            eta = eta0 / math.sqrt(it)
            a = a - eta * g
            # project onto simplex
            a = _proj_simplex(a)
        # after loop, adopt best
        if best_val_sub < best_val:
            best_val, best_seq = best_val_sub, best_seq_sub
            print(f"[Subgrad] final best = {best_val:.6f}")

    # step 3: optional Adam on logits using subgradient of exact max (or smoothed)
    if time.time() < deadline:
        print("\n--- Adam optimizer on logits ---")
        # transform to logits (log a)
        eps = 1e-15
        y = np.log(np.maximum(best_seq, eps))
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        beta1 = 0.9
        beta2 = 0.999
        lr0 = 0.2  # base learning rate
        max_iters = 20000
        best_seq_adam = best_seq.copy()
        best_val_adam = best_val
        for t in range(1, max_iters+1):
            if time.time() >= deadline:
                break
            a = _softmax(y)
            conv = _conv_fft(a, a)
            max_c = conv.max()
            val = 2.0 * n * max_c
            if val < best_val_adam:
                best_val_adam, best_seq_adam = val, a.copy()
                print(f"[Adam] iter {t}, new best {best_val_adam:.6f}")
            # active shifts
            active = np.where(conv >= max_c - 1e-12)[0]
            w = 1.0 / len(active)
            g = np.zeros_like(a)
            for k in active:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end+1)
                rev = k - idx
                g[idx] += w * 2.0 * a[rev]  # subgradient g for a
            # gradient wrt y
            dot = np.dot(a, g)
            grad_y = a * (g - dot)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 * (np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t))
            y = y - lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        # after loop
        if best_val_adam < best_val:
            best_val, best_seq = best_val_adam, best_seq_adam
            print(f"[Adam] final best = {best_val:.6f}")

    # step 4: final L-BFGS with high smoothing p=500 maybe 1000, with many iter
    if time.time() < deadline:
        print("\n--- Final L-BFGS refinement (smooth max) ---")
        # start from best_seq
        y0 = np.log(np.maximum(best_seq, 1e-15))
        # Define p schedule finishing with large p
        p_schedule = [5, 15, 45, 135, 405, 1215]
        a_curr = best_seq.copy()
        y_curr = y0.copy()
        for p in p_schedule:
            if time.time() >= deadline: break
            max_iter = 500  # more iterations
            # call scipy minimize
            res = _opt.minimize(_lbfgsb_func, y_curr,
                                args=(n, float(p)),
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter, 'disp': False})
            y_curr = res.x
            a_curr = _softmax(y_curr)
            val = _objective_surrogate(a_curr)  # exact surrogate (sum=1)
            print(f"[L‑BFGS] p={p}, surrogate={val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_curr.copy()
                print(f"    ** new global best {best_val:.6f}")
            # optional random perturb restart between p steps
            if time.time() < deadline:
                # small perturbation
                y_curr = y_curr + rng.normal(scale=0.1, size=n)
        # after schedule, assign best_seq
        best_seq = best_seq  # already set

    # final evaluation
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Generate a diverse pool of ℓ₁‑normalised seed vectors using analytic windows
   and random Dirichlet draws (plus the previous solution, if any).
2. Keep the best few seeds according to the exact surrogate
   `2·n·max(conv(a,a))` (the evaluate_sequence value when the sum = 1).
3. Random convex combinations of the top seeds (Dirichlet‑weighted mixes)
   are sampled – this often “flattens’’ the auto‑correlation and gives an
   immediate improvement.
4. A short additive sub‑gradient descent on the exact max‑convolution
   (projected onto the simplex) is run from the best candidate.
5. Adam optimisation on the logits (`y = log a`) using the exact sub‑gradient
   of the max‑convolution further refines the solution.
6. Finally a high‑precision L‑BFGS‑B optimisation of a smooth log‑sum‑exp
   surrogate with a very large smoothing parameter (p≈5000) and several
   random restarts is performed.  This step approximates the true non‑smooth
   objective while benefitting from quasi‑Newton acceleration.
7. The best vector found is evaluated with ``evaluate_sequence`` and stored
   globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = y.max()
    e = np.exp(y - m)
    return e / e.sum()


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (sum(seq)=1) → 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0, 5.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.3, 0.4, 0.6):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0, 5.0, 10.0):
        for b in (0.5, 1.0, 2.0, 3.0, 5.0, 10.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform (adds slight flat bias)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # keep previous solution (if any) as a seed
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and np.sum(arr) > 0:
            seeds.append(arr / np.sum(arr))

    # final sanity normalisation
    out = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        out.append(v / s if s > 0 else _uniform(n))
    return out

# Smooth surrogate for L‑BFGS‑B (log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """Smooth surrogate and its gradient w.r.t. logits y."""
    a = _softmax(y)                     # a lies on the simplex
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = conv.max()

    shifted = conv - max_c               # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)  # safety
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                    # fallback – exact max
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)     # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a: np.ndarray, n: int,
                     deadline: float, p: float = 20.0,
                     max_iter: int = 250) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smooth surrogate and return exact surrogate value."""
    y0 = np.log(np.maximum(start_a, 1e-15))
    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, float(p)),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})
    a_opt = _softmax(res.x)
    val = _objective_surrogate(a_opt)
    return a_opt, val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (projected onto simplex)
# ----------------------------------------------------------------------
def _subgrad_proj_optimize(start: np.ndarray, n: int,
                           deadline: float,
                           max_iters: int = 20000,
                           eta0: float = 0.2) -> tuple[np.ndarray, float]:
    """Additive sub‑gradient method with Euclidean projection onto the simplex."""
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a, a)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val, best_a = val, a.copy()

        # active shifts (tolerance to catch ties)
        active = np.where(conv >= max_c - 1e-12)[0]
        weight = 1.0 / len(active)
        g = np.zeros_like(a)

        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] += weight * 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)

    return best_a, best_val

    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    best_a = _softmax(y)
    best_val = _objective_surrogate(best_a)

    for t in range(1, max_iters + 1):

            best_val, best_a = val, a.copy()

        # active shifts
        active = np.where(conv >= max_c - 1e-12)[0]
        weight = 1.0 / len(active)
        g = np.zeros_like(a)

        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] += weight * 2.0 * a[rev]

        dot = np.dot(a, g)
        grad_y = a * (g - dot)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = lr0 * (np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t))
        y = y - lr * m_hat / (np.sqrt(v_hat) + 1e-12)

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence as a list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n--- Problem size -------------------------------------------------")
    print(f"Length of the sequence: n = {n}")

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    print("\n--- Seed generation ------------------------------------------------")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # --------------------------------------------------------------
    # Evaluate seeds and keep the best few
    # --------------------------------------------------------------
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"\nTop {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\nInitial best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Random convex combinations of the top seeds
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random convex combinations of top seeds ----------------------")
        M = len(top_seeds)
        mat = np.vstack(top_seeds)               # shape (M, n)
        combos = 20000
        for i in range(combos):
            if time.time() >= deadline:
                break
            # concentration parameter – favouring sparse mixes
            conc = rng.uniform(0.2, 1.0)
            w = rng.dirichlet(np.full(M, conc))
            mixed = w @ mat
            mixed = np.maximum(mixed, 0.0)
            s = mixed.sum()
            if s > 0:
                mixed /= s
            else:
                mixed = _uniform(n)
            val = _objective_surrogate(mixed)
            if val < best_val:
                best_val, best_seq = val, mixed.copy()
                print(f"[Mix] iteration {i:5d} → improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient descent (projected) – short but may escape local minima
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Projected sub‑gradient descent --------------------------------")
        sub_a, sub_val = _subgrad_proj_optimize(best_seq, n, deadline,
                                                max_iters=15000, eta0=0.15)
        if sub_val < best_val:
            best_val, best_seq = sub_val, sub_a.copy()
            print(f"[Sub‑grad] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Adam optimisation on logits (exact max‑convolution gradient)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam optimisation (logits) -----------------------------------")
        adam_a, adam_val = _adam_optimize(best_seq, n, deadline,
                                          max_iters=15000, lr0=0.1)
        if adam_val < best_val:
            best_val, best_seq = adam_val, adam_a.copy()
            print(f"[Adam] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final high‑precision L‑BFGS‑B refinements (large p, several restarts)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS‑B refinements (high smoothing) ------------------")
        p_schedule = [5, 15, 45, 135, 405, 1215, 3645, 10935]   # geometric growth
        y_curr = np.log(np.maximum(best_seq, 1e-15))

        for p in p_schedule:
            if time.time() >= deadline:
                break
            max_iter = 800
            a_opt, val = _lbfgsb_optimize(_softmax(y_curr), n, deadline,
                                          p=float(p), max_iter=max_iter)
            print(f"[L‑BFGS] p={p:6.0f} → surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.copy()
                y_curr = np.log(np.maximum(best_seq, 1e-15))
                print(f"    ** new global best {best_val:.6f}")

            # small random perturbation before next p‑stage
            if time.time() < deadline:
                y_curr = y_curr + rng.normal(scale=0.05, size=n)

        # a few random restarts at the final (largest) p
        final_p = p_schedule[-1]
        for r in range(3):
            if time.time() >= deadline:
                break
            y_start = np.log(np.maximum(best_seq, 1e-15)) + rng.normal(scale=0.1, size=n)
            a_opt, val = _lbfgsb_optimize(_softmax(y_start), n, deadline,
                                          p=float(final_p), max_iter=800)
            print(f"[Restart] p={final_p} (run {r+1}) → surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.copy()
                print(f"    ** new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
