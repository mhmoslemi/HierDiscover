"""
A fast heuristic for the non‑negative step–function minimisation problem.

The algorithm proceeds in four stages

1. **Seed pool** – deterministic families (uniform, triangular,
   sine‑window, etc.) together with systematic exponential / power‑law
   families.  All vectors are ℓ₁‑normalised (sum = 1) so the objective
   is scale‑invariant.
2. **Smooth surrogate optimisation** – L‑BFGS on a log‑sum‑exp
   approximation of the max‑convolution.  An extended β‑schedule
   (β up to 10 000) and a generous iteration budget drive the solution
   towards a local optimum of the true objective.
3. **Fast local refinement** – a specialised hill‑climbing step that
   moves a small amount of mass between two coordinates and updates the
   full convolution by cheap O(n) slice operations (exact for the
   quadratic form).  This explores the neighbourhood of the L‑BFGS
   solution without the overhead of recomputing the FFT each move.
4. **Final evaluation** – the best vector is stored globally as
   ``height_sequence_1`` and its true value is printed using the supplied
   ``evaluate_sequence`` function.

The whole search respects the time budget (≈ 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – produces a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), ℓ₁‑normalised."""
    if lam <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shape(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + offset) ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Surrogate objective (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Function and gradient for the log‑sum‑exp surrogate."""
    a = _softmax(v)                     # probability vector
    conv = _conv_fft(a, a)               # full autocorrelation
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp               # soft weights
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The optimisation variable are the logits of the softmax.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    v = np.log(start_vec + 1e-30)  # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val}")
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Fast exact local search (mass transfer between two coordinates)
# ----------------------------------------------------------------------
def _local_search_optimize(a_start: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing on the exact objective.
    A small amount of mass ε is moved from index i to index j.
    The full convolution is updated in O(n) using slice arithmetic,
    which is exact up to the ignored ε² terms; a periodic recompute
    restores perfect consistency.
    """
    a = a_start.copy()
    conv = _conv_fft(a, a)                 # current full convolution
    best_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    rng = np.random.default_rng()
    it = 0
    conv_len = len(conv)

    # helper to keep the convolution exact every few thousand steps
    RECOMPUTE_EVERY = 5000

    while time.time() < deadline and it < max_iters:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        ai = a[i]
        if ai <= 0.0:
            continue
        # transfer a fraction of the mass at i to j
        eps = ai * rng.random() * 0.5   # at most half of a[i]
        if eps <= 0.0:
            continue

        aj = a[j]                       # old value at j (needed for revert)

        # --- apply the exact update to the convolution -------------
        # conv[k] ↦ conv[k] – 2·eps·a[k‑i]  (for all k where k‑i in range)
        conv[i:i + n] -= 2.0 * eps * a
        # conv[k] ↦ conv[k] + 2·eps·a[k‑j]
        conv[j:j + n] += 2.0 * eps * a
        # ε² corrections (self‑terms and the cross‑term i+j)
        conv[2 * i] += eps * eps
        conv[2 * j] += eps * eps
        conv[i + j] -= 2.0 * eps * eps

        # --- update the probability vector -------------------------
        a[i] = ai - eps
        a[j] = aj + eps

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        else:
            # ---- revert -------------------------------------------------
            a[i] = ai
            a[j] = aj
            # undo the convolution update (using the restored a)
            conv[i:i + n] += 2.0 * eps * a
            conv[j:j + n] -= 2.0 * eps * a
            conv[2 * i] -= eps * eps
            conv[2 * j] -= eps * eps
            conv[i + j] += 2.0 * eps * eps
            # no improvement – keep the old best

        it += 1
        # occasional full recompute to avoid accumulated rounding errors
        if it % RECOMPUTE_EVERY == 0:
            conv = _conv_fft(a, a)

    return best_a, best_val
