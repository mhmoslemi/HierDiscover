"""
Hybrid L‑BFGS + exponential‑gradient optimizer for the step‑function bound.

The algorithm works in the logit space:
    a_i = softmax(z_i)  (∑ a_i = 1, a_i ≥ 0)

For a given smoothing parameter α we minimise the smooth surrogate
    L_α(z) = (1/α)·log Σ_k exp(α·(a*a)[k])
which converges to max(a*a) as α → ∞.
The gradient of L_α with respect to the logits is obtained analytically
using FFTs (the same formula that the reference code uses for the Adam
surrogate).  A short schedule of increasing α values together with a
few L‑BFGS runs drives the solution close to a local optimum.  Finally a
few hundred exponential‑gradient (multiplicative‑weights) steps polish
the result.  The whole pipeline respects the 1000 s time budget and
stores the best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import fmin_l_bfgs_b
    _SCIPY_OK = True
except Exception:                     # pragma: no cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {w ≥ 0, sum w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑p = 1):
        2·n·max(conv(p,p))
    This matches the value returned by ``evaluate_sequence`` for a
    ℓ₁‑normalised sequence.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector that sums to 1)."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Shape families (used for seeding)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a diverse pool of ℓ₁‑normalised seed vectors.
    """
    seeds: list[np.ndarray] = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.18))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_gaussian(n, 0.35))
    for p in (0.5, 1.0, 2.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0):
        for b in (0.5, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # previous solution
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # safety check (ensure positivity and ℓ₁‑normalisation)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Log‑sum‑exp surrogate and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
def _surrogate_and_grad(z: np.ndarray, alpha: float, n: int):
    """
    Compute the smoothed surrogate L_α(z) and its gradient w.r.t. the
    logits z (where a = softmax(z)).  The implementation follows the
    reference solution but works directly with the logits, thereby
    avoiding any explicit projection onto the simplex.
    """
    # a = softmax(z)
    a = _softmax(z)

    # length needed for convolutions (a length n, softmax weights length 2n‑1)
    N = _next_pow2(3 * n)

    # FFT of a
    fx = np.fft.rfft(a, n=N)

    # full convolution a*a
    conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

    # ---- smooth max (log‑sum‑exp) ----
    max_c = np.max(conv_full)
    # stability: subtract max before exponentiation
    exp_term = np.exp(alpha * (conv_full - max_c))
    sum_exp = np.sum(exp_term)
    # surrogate value (equals max_c + log(sum_exp)/alpha)
    loss = max_c + np.log(sum_exp) / alpha

    # probability weights for the surrogate
    w = exp_term / sum_exp

    # gradient w.r.t. a:
    #   ∂L/∂a = 2 * (a * w)   where * denotes linear conv
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(np.conj(fx) * fw, n=N)[:n]
    grad_a = 2.0 * corr

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)

    return loss, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimiser in logit space
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_logit: np.ndarray,
                    n: int,
                    alpha: float,
                    max_iter: int,
                    deadline: float):
    """
    Run L‑BFGS on the surrogate with a given smoothing parameter α.
    Returns the probability vector a (ℓ₁‑normalised) and the final surrogate
    value.
    """
    if not _SCIPY_OK:                    # pragma: no cover
        return _softmax(start_logit), float('inf')

    # objective wrapper required by scipy (return loss, gradient)
    def fun(z):
        loss, grad = _surrogate_and_grad(z, alpha, n)
        return loss, grad

    # time‑guarded callback – abort if we run out of budget
    def callback(xk):
        if time.time() > deadline:
            raise RuntimeError("deadline reached")

    try:
        x_opt, f_opt, _ = fmin_l_bfgs_b(fun,
                                        start_logit,
                                        maxiter=max_iter,
                                        iprint=-1,
                                        callback=callback)
    except RuntimeError:                 # deadline hit
        x_opt = start_logit.copy()
        f_opt = np.nan

    a_opt = _softmax(x_opt)
    return a_opt, float(f_opt)


# ----------------------------------------------------------------------
# Exponential‑gradient (mirror descent) polish – same as in the reference
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 0.22) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val

    top_k = 6

    best_seq = best_seeds[0].copy()

        true_val = None
