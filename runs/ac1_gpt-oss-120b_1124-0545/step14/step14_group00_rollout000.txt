"""propose_candidate
-------------------

Search for a non‑negative step function (list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where conv is a full linear convolution.  The algorithm combines

* a rich pool of analytic “window” families,
* up‑sampling of the best known solution to larger problem sizes,
* a continuation L‑BFGS optimiser on a log‑sum‑exp smooth approximation of
  the max‑convolution,
* a long mirror‑descent (exponentiated‑gradient) phase that works on the
  exact max,
* a final short simulated‑annealing refinement.

Multiple problem sizes (2000, 4000, 8000) are tried; the best sequence
found across all sizes is returned.  The code is time‑aware (1000 s
budget) and works with or without SciPy installed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS‑B)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact max for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """For a vector with sum=1 return 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _sin_power(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float) -> np.ndarray:
    """Run L‑BFGS on the smoothed max‑convolution objective."""
    if _opt is None:
        # SciPy not available – fall back to the start vector unchanged.
        return start_vec.copy()

    # aggressive β schedule – the last value gives an excellent approximation
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 1000.0, 2000.0, 5000.0,
                     10000.0, 20000.0, 50000.0]

    # work in log‑softmax space to stay on the simplex automatically
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            mx = np.max(conv)

            # log‑sum‑exp smoothing of the maximum
            shift = mx
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice of length n

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # tiny wrapper to abort early when time runs out
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': 2000,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000) -> np.ndarray:
    """Fine‑tune the exact max‑convolution using exponentiated gradient."""
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (only indices that participate)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size → diminishing 1/√t
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # optional progress output
        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:7d} surrogate={best_val:.6f}")

    return best_a

# Simulated‑annealing refinement (logit + softmax)

def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> np.ndarray:
    """Short SA in log‑probability space to escape shallow local minima."""
    rng = np.random.default_rng(12345)
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-7          # temperature schedule
        sigma = 0.25 * (1.0 - progress) + 1e-5      # proposal strength

        prop_logits = logits + sigma * rng.normal(size=n)
        prop_seq = _softmax(prop_logits)
        prop_val = _objective_surrogate(prop_seq)

        if prop_val < cur_val or math.exp(-(prop_val - cur_val) / T) > rng.random():
            logits = prop_logits
            cur_seq = prop_seq
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq

# Upsampling / downsampling utilities

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(new_x, old_x, seq)
    s = interp.sum()
    if s == 0.0:
        return _uniform(new_n)
    return interp / s

        top_k = 8
        top_seeds = [vec for vec, _, _ in scored[:top_k]]
        print(f"  Generated {len(seeds)} seeds, retaining top {top_k}")

        # --------------------------------------------------------------
        # 3. Continuation L‑BFGS on each top seed
        # --------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            # allocate a modest slice of the remaining budget to this seed
            max_iters_per_beta = max(800, int(remaining * 0.03))
            print(f"\n  L‑BFGS run #{idx+1}/{len(top_seeds)} "
                  f"(max_iter_per_beta≈{max_iters_per_beta})")
            seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline)
            val_opt = _objective_surrogate(seq_opt)
            print(f"    L‑BFGS surrogate = {val_opt:.6f}")

            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"    *** New global best surrogate = {global_best_val:.6f}")

            # ----------------------------------------------------------
            # 4. Mirror‑descent polishing (exact max)
            # ----------------------------------------------------------
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_exp_iters = max(50_000, int(remaining * 0.04))
                seq_exp = _exp_gradient_opt(global_best_seq, n, deadline,
                                            max_iters=max_exp_iters)
                val_exp = _objective_surrogate(seq_exp)
                if val_exp < global_best_val:
                    global_best_val = val_exp
                    global_best_seq = seq_exp.copy()
                    print(f"    ExpGrad improved surrogate to {global_best_val:.6f}")

        # --------------------------------------------------------------
        # 5. Final simulated‑annealing push (short)
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_sa_iters = max(30_000, int(remaining * 0.02))
            print(f"\n  Simulated‑annealing (≈{max_sa_iters} iters) for n={n}")
            seq_sa = _simulated_annealing(global_best_seq, n, deadline,
                                          max_iters=max_sa_iters)
            val_sa = _objective_surrogate(seq_sa)
            if val_sa < global_best_val:
                global_best_val = val_sa
                global_best_seq = seq_sa.copy()
                print(f"    SA improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    if global_best_seq is None:
        raise RuntimeError("Search failed to produce a candidate.")

    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for the next round (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
