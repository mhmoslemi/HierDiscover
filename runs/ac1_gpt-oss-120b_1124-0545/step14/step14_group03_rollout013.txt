"""
Improved search for a non‑negative step function minimizing the
evaluation function

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm builds on the previous solution but makes three
independent enhancements:

1. **Larger problem size** – we use n = 8000 (instead of 4000).  The
   theoretical lower bound 2·n/(2n−1) ≈ 1 becomes tighter, giving the
   optimizer more freedom to push the objective under the old record
   ≈ 1.5085.

2. **More aggressive L‑BFGS continuation** – the β‑schedule now reaches
   5·10⁵ and the per‑β iteration budget is increased (up to several
   thousand).  This yields a much tighter approximation of the true
   max‑convolution while still staying inside the 1000 s time budget.

3. **Repeated exponentiated‑gradient (EG) refinement** – after each
   L‑BFGS run we launch a long EG optimisation on the *exact* objective.
   The EG phase is executed repeatedly with fresh random perturbations
   of the current best vector; this helps escape shallow local minima
   and often improves the bound by a few thousandths.

All steps respect the security checks of ``evaluate_sequence``:
the vectors are ℓ₁‑normalised (sum = 1) before being handed to the
evaluator, so the objective is exactly ``2·n·max(conv)``.
The routine aborts gracefully before the deadline and stores the best
found sequence in the global name ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0 (NumPy)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, ℓ₁‑renormalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (smooth max) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the log‑sum‑exp surrogate.
    The optimisation variable is the log‑softmax `v`.
    """
    a = _softmax(v)                     # probability vector
    conv = _conv_fft(a, a)               # full linear convolution
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp               # soft weights for each lag
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # Jacobian of softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective via L‑BFGS‑B.
    ``start_vec`` must be ℓ₁‑normalised.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 1e5,
                         5e5]

    v = np.log(start_vec + 1e-30)               # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val}")
            break
        except Exception as e:                 # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exact objective and sub‑gradient (max(conv))
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t a.
    If several lags attain the maximum we average their contributions.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = np.max(conv)
    # tolerance to collect all “active” lags (helps stability)
    eps = 1e-12 * max_val
    active = np.where(conv >= max_val - eps)[0]

    g = np.zeros_like(a)
    for idx in active:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] += 2.0 * a[rev_idx]
    # average over the active set
    if len(active) > 0:
        g /= len(active)
    return g


def _logits_gradient(a: np.ndarray, grad_a: np.ndarray) -> np.ndarray:
    """
    Convert a gradient w.r.t. a into a gradient w.r.t. logits.
    a = softmax(v)
    """
    dot = np.dot(grad_a, a)
    return a * (grad_a - dot)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 1_500_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns the best vector seen and its exact objective value.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)               # sub‑gradient of max(conv)
        eta = eta0 / np.sqrt(it)                  # 1/√t schedule
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)                       # back to simplex

        # occasional check
        if it % 200_000 == 0:
            cur_val = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur_val:.6f}")
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

    # final check
    cur_val = _objective_exact(a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()

    return best_a, best_val

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian family (various bandwidths)
    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 6.0, 13):
        seeds.append(_power(n, p))

    # Beta family
    for a in np.linspace(0.5, 3.0, 5):
        for b in np.linspace(0.5, 3.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))

    # Random Dirichlet draws (both sparse and dense)
    for conc in (0.02, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponentials
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency sinusoidal curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of already generated seeds
    mix_cnt = 300
    base_idx = np.arange(len(seeds))
    for _ in range(mix_cnt):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Clean‑up (non‑negative, normalised)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

    def surrogate(v):
        return 2.0 * n * np.max(_conv_fft(v, v))

    evaluated = [(vec, surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 7
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values among seeds (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # Continuation L‑BFGS on smooth surrogate
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    # an aggressive β‑schedule
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 2000.0, 5000.0,
                  1e4, 2e4, 5e4, 1e5, 5e5]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all L‑BFGS seeds.")
            break

        remaining = deadline - time.time()
        # allocate a per‑β iteration budget proportional to remaining time
        per_beta_iters = max(800, int(min(6000,
                                          remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={per_beta_iters}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=per_beta_iters)
        val_opt = surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Repeated exponentiated‑gradient refinement (exact objective)
    # --------------------------------------------------------------
    # We will run a few EG phases with fresh random perturbations.
    exp_phases = 3
    for phase in range(exp_phases):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate up to a few hundred thousand iterations per phase
        max_exp_iters = int(min(800_000,
                                remaining * 0.25 * 1e5))
        if max_exp_iters <= 0:
            break
        print(f"\n--- EG refinement phase {phase+1} (max_iters={max_exp_iters}) ---")
        # add a tiny random perturbation to escape possible stagnation
        eps = 1e-4
        pert = global_best_seq + eps * rng.normal(size=n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()
        seq_exp, val_exp = _exp_gradient_optimize(pert,
                                                  n,
                                                  deadline,
                                                  max_iters=max_exp_iters,
                                                  eta0=1.0)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved exact objective to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final polishing L‑BFGS (high β) – a short high‑precision run
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(500, int(min(2500,
                                          remaining * 0.06 * 1e5)))
        print("\n--- Final high‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[5e5, 1e6],
                                           max_iter_per_beta=per_beta_iters)
        val_final = surrogate(seq_final)
        print(f"[Final‑L‑BFGS] surrogate = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after polishing = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator (logging only)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                              # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
