"""
Propose a better step‑function (list of non‑negative heights) for the
challenge objective

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in three stages:

1. **Initialisation** – start from the previously found best sequence (if
   available) or from a uniform distribution, normalised to L₁‑norm = 1.

2. **Frank‑Wolfe‑style LP refinement** – for the current sequence `f` we
   compute `M = max(conv(f,f))`.  Holding `f` fixed we solve a linear
   programme:

        maximise   Σ g_i
        subject to (f * g)[k] ≤ M   for all k
                    g_i ≥ 0

   The solution `g` puts mass where `f` is small while respecting the
   convolution bound.  After a simple scaling to keep the L₁‑norm fixed we
   perform a convex combination  `f ← (1‑t)·f + t·g`.  Repeating this step
   drives the maximal autocorrelation down.

3. **Smooth polishing** – the final `f` is refined with a smooth
   log‑sum‑exp surrogate of the max, optimised by L‑BFGS‑B on the
   log‑softmax (simplex) parametrisation, followed by a few
   exponentiated‑gradient iterations.

All work respects the 1000‑second time budget and stores the best sequence
found in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector, sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised vector: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# L‑BFGS optimisation of a smooth max‑convolution surrogate
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Optimise a log‑sum‑exp smooth approximation of max(conv)."""
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on the simplex
            conv = _conv_fft(a, a)            # full linear convolution
            max_c = np.max(conv)
            shift = max_c                     # numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximation of max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # weights = softmax(β·conv)
            w = exp_beta / sum_exp

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

            # Jacobian of softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,

    """Mirror‑descent on the exact max‑convolution (projected via softmax)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming (Frank‑Wolfe style) refinement
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_iter: int = 30,
               step: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Perform up to ``max_iter`` Frank‑Wolfe style LP updates.
    Returns the best sequence seen and its surrogate value.
    """
    best_f = f.copy()
    best_val = _objective_surrogate(best_f)
    target_sum = best_f.sum()

    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        # current bound
        conv_f = _conv_fft(f, f)
        M = float(conv_f.max())

        # Build A_ub where (f * g)[k] = (A_ub @ g)[k] ≤ M
        # A_ub is a (2n‑1) × n matrix with constant diagonals = f_i
        data = [np.full(n, fi) for fi in f]   # one array per diagonal
        offsets = np.arange(n)                # offsets 0 … n‑1
        A_ub = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

        # Linear program: maximise Σ g_i  (i.e. minimise -Σ g_i)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        lp_res = _opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                              bounds=bounds, method='highs',
                              options={'presolve': True})
        if not lp_res.success:
            print(f"[LP] iteration {it}: solver failed ({lp_res.message})")
            break

        g = lp_res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # Rescale g to the original L₁‑norm
        g = g * (target_sum / sum_g)

        # Convex combination
        f_new = (1.0 - step) * f + step * g
        # Re‑normalise (tiny numerical drift)
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()

        # Evaluate surrogate
        val_new = _objective_surrogate(f_new)
        if val_new < best_val:
            best_val = val_new
            best_f = f_new.copy()
            print(f"[LP] iter={it:2d} improved surrogate {best_val:.6f}")

        # Prepare for next iteration
        f = f_new
        # slowly decay the step size to stabilise
        step = max(0.02, step * 0.95)

    return best_f, best_val
