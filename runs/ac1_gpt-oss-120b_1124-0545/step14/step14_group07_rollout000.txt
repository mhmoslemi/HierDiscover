"""
Propose a non‑negative step function (a probability vector) that minimizes
the ``evaluate_sequence`` objective.

Algorithm
---------
1. **Parameter choice** – use a relatively large length (n≈6000) to give the
   optimiser many degrees of freedom while keeping FFT‑based convolutions cheap.
2. **Seed generation** – a very diverse pool:
   * deterministic analytic windows (uniform, triangular, sine, Gaussian,
     power, beta, Tukey, exponential)
   * many random Dirichlet draws with a wide range of concentration parameters
   * random low‑frequency cosine curves and mixtures with the uniform vector.
   * the previous best solution (if any) is kept as a seed.
3. **Surrogate evaluation** – the inexpensive quantity
   `2·n·max(conv(a,a))` (the exact objective for a normalised vector) is used
   for ranking and for all inner‑loop optimisation.
4. **Local optimisation** – for the best few seeds a multi‑stage
   smoothing‑max L‑BFGS‑B is performed.
   The smoothing parameter `p` is increased geometrically
   (`5, 15, 45, 135, 405, 1215`).  At each stage a short L‑BFGS run
   (`max_iter≈600`) is executed together with a handful of random restarts
   (Gaussian noise on the logits).  This aggressively reduces the max‑convolution.
5. **Fine‑grained polishing** – a long exponentiated‑gradient (mirror‑descent)
   pass (up to a few hundred thousand iterations) with a small step size
   finishes the optimisation.
6. **Final evaluation** – the true `evaluate_sequence` value is computed for the
   best candidate and stored globally as ``height_sequence_1``.

The whole procedure respects a 1000 s wall‑clock budget and prints intermediate
progress information.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(vec: np.ndarray) -> float:
    """Return 2·n·max(conv(vec,vec)) (sum(vec) must be 1)."""
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed collection – very diverse
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a large, diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.1, 0.15, 0.2, 0.3, 0.4, 0.6):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9995))

    for p in (0.3, 0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    for a in (0.3, 0.5, 1.0, 2.0):
        for b in (0.3, 0.5, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))

    for alpha in (0.2, 0.4, 0.6, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws – wide range of concentrations
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves / mixtures
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(curve)
            # mix slightly with uniform to add a flat bias
            for mix_alpha in (0.1, 0.3, 0.5):
                seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # randomised sine / Gaussian parameters
    for _ in range(8):
        gamma = rng.uniform(0.5, 5.0)
        seeds.append(_sine_window(n, gamma))
        sigma = rng.uniform(0.1, 0.5)
        seeds.append(_gaussian(n, sigma))

    # keep previous solution as a seed
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final normalisation (guard against numerical hiccups)
    out = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        out.append(v / s if s > 0 else _uniform(n))
    return out


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int,
                      deadline: float,
                      max_iters: int = 250_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent using the sub‑gradient of the active max‑conv."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size schedule √t
        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)       # guard against overflow
        x = x * np.exp(delta)

        # project back onto the simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val, best_x = cur_val, x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient (logits parametrisation)
# ----------------------------------------------------------------------
def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """Smooth surrogate and its gradient wrt logits y."""
    a = _softmax(y)                     # a lies on the simplex
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = conv.max()

    shifted = conv - max_c               # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)  # safety for float64
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                    # fallback – use hard max mask
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)     # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)         # chain rule through soft‑max

    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a: np.ndarray,
                     n: int,
                     deadline: float,
                     p: float = 20.0,
                     max_iter: int = 250) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the smooth surrogate with logits parametrisation.
    Returns the best vector found and its *exact* surrogate value.
    """
    y0 = np.log(np.maximum(start_a, 1e-15))
    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})
    a_opt = _softmax(res.x)
    val = _objective_surrogate(a_opt)
    return a_opt, val


# ----------------------------------------------------------------------
# Multi‑stage L‑BFGS refinement (with random restarts)
# ----------------------------------------------------------------------
def _refine_sequence(start_a: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a geometric schedule of smoothed‑max L‑BFGS optimisations,
    augmenting each stage with a few random‑restart runs.
    """
    best_a = start_a.copy()
    best_val = _objective_surrogate(best_a)

    p_schedule = [5.0, 15.0, 45.0, 135.0, 405.0, 1215.0]

    for p in p_schedule:
        if time.time() >= deadline:
            break
        # main optimisation run
        cand_a, cand_val = _lbfgsb_optimize(best_a, n, deadline, p=p, max_iter=600)
        if cand_val < best_val:
            best_val, best_a = cand_val, cand_a
            print(f"   [L‑BFGS] p={p:5.1f} ⇒ improved to {best_val:.6f}")

        # a few stochastic restarts at the same smoothing level
        for r in range(3):
            if time.time() >= deadline:
                break
            # perturb logits with Gaussian noise
            y_noise = np.log(np.maximum(best_a, 1e-15)) + rng.normal(scale=0.3, size=n)
            pert_a = _softmax(y_noise)
            cand_a, cand_val = _lbfgsb_optimize(pert_a, n, deadline, p=p, max_iter=600)
            if cand_val < best_val:
                best_val, best_a = cand_val, cand_a
                print(f"   [Restart] p={p:5.1f} ⇒ improved to {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best sequence as a plain Python list.
    """
    # ------------------------------------------------------------------
    # Initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # keep a tiny safety margin

    # ------------------------------------------------------------------
    # Choice of sequence length – a compromise between expressiveness and speed
    # ------------------------------------------------------------------
    n = 6000
    print(f"\n--- Problem size -------------------------------------------------")
    print(f"Length of the sequence : n = {n}")

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    print("\n--- Seed generation ------------------------------------------------")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # ------------------------------------------------------------------
    # Rank seeds using the cheap surrogate
    # ------------------------------------------------------------------
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Initialise global best
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – intensive local optimisation on the best seeds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Multi‑stage L‑BFGS refinement ---------------------------------")
        for idx, seed_vec in enumerate(best_seeds):
            if time.time() >= deadline:
                break
            cand_seq, cand_val = _refine_sequence(seed_vec, n, deadline, rng)
            if cand_val < best_val:
                best_val, best_seq = cand_val, cand_seq
                print(f"[Refine] seed #{idx+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – final long exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing -------------------------------------------")
        remaining = deadline - time.time()
        # Rough budget: 2000 updates per second on a typical VM
        max_iters = int(min(600_000, remaining * 2000))
        print(f"Polishing with up to {max_iters} EG iterations.")
        cand_seq, cand_val = _expgrad_optimize(best_seq, n, deadline,
                                               max_iters=max_iters,
                                               eta0=0.12)
        if cand_val < best_val:
            best_val, best_seq = cand_val, cand_seq
            print(f"[Polish EG] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (slow but exact)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store the result globally for possible warm‑starts in later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
