def _lp_update(a, n, M, active_idx, t):
    N = len(a)
    m = len(active_idx)
    # Build A_ub shape (m,N)
    rows = []
    for k in active_idx:
        row = np.zeros(N, dtype=np.float64)
        # This row corresponds to sum_j a_{k-j} * b_j <= M
        # For j in [max(0, k-(N-1)), min(k,N-1)]
        start_j = max(0, k-(N-1))
        end_j = min(k, N-1)
        # j range inclusive
        for j in range(start_j, end_j+1):
            row[j] = a[k - j]
        rows.append(row)
    A_ub = np.vstack(rows) if rows else None
    # Objective maximize sum(b) => minimize -sum(b)
    c = -np.ones(N, dtype=np.float64)
    # Bound: b >= 0
    bounds = [(0, None)] * N
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback: return a unchanged
        return a
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return a
    # Scale to sum = 1
    b = b / sum_b
    # Mix
    new_a = (1 - t) * a + t * b
    # Renormalize to sum 1 (in case numeric)
    new_a = new_a / new_a.sum()
    return new_a

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 4000
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)

    # evaluate surrogate and get top seeds
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec,_ in scored[:top_k]]
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"Init best surrogate {best_val:.6f}")

    # quick EG on top seeds
    for i, seed_vec in enumerate(best_seeds[:3]):
        if time.time() > deadline: break
        cand, val = _expgrad_optimize(seed_vec, n, deadline, max_iters=50000, eta0=0.3)
        if val < best_val:
            best_val, best_seq = val, cand
            print(f"[EG] seed #{i+1} improved to {best_val:.6f}")

    # LP refinement loop
    rel_tol = 1e-4
    t = 0.2
    lp_iter = 0
    while time.time() < deadline:
        conv = _conv_fft(best_seq, best_seq)
        M = conv.max()
        # active indices
        active_idx = np.where(conv >= M * (1 - rel_tol))[0]
        # limit to at most 8 active indices
        if active_idx.shape[0] > 8:
            active_idx = active_idx[:8]  # take first few
        new_seq = _lp_update(best_seq, n, M, active_idx, t)
        new_val = _objective_surrogate(new_seq)
        if new_val < best_val - 1e-8:
            best_val = new_val
            best_seq = new_seq
            print(f"[LP] iter {lp_iter} new best {best_val:.6f}")
        # decay t
        t = max(0.01, t * 0.995)
        lp_iter += 1
        # stop after many iterations without improvement?
        if lp_iter > 2000:
            break

    # final EG polishing
    if time.time() < deadline:
        best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                                max_iters=200000, eta0=0.2)

    # evaluate true objective
    true_val = evaluate_sequence(best_seq.tolist())
    print(f"\n=== Final result ===\nSurrogate {best_val:.6f}, true {true_val:.6f}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Propose candidate step function minimizing evaluate_sequence.

Pipeline:
- Generate a large diverse set of ℓ₁‑normalised seeds (analytic windows,
  beta families including highly U‑shaped, DPSS, Kaiser, Blackman, …).
- Keep best few seeds according to the exact surrogate 2·n·max(conv).
- Apply a short multiplicative‑gradient (EG) refinement on top seeds.
- Perform an iterative linear‑program (LP) refinement.  For the current
  sequence a we solve a linear program that maximises the ℓ₁‑norm of a
  partner vector b under the active convolution constraints
  a∗b ≤ max(a∗a).  Mixing a with the normalised b (convex combination)
  provably reduces the worst‑case convolution value.
- Finish with a longer EG polishing.
All stages respect the global deadline (default 1000 s).  The best
sequence found is returned and stored in the global variable
`height_sequence_1`.
"""

import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – surrogate when sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    a = np.blackman(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    a = np.hamming(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    a = np.hanning(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, NW: float = 3.0) -> np.ndarray:
    if _sig_windows is None:
        return _uniform(n)
    a = _sig_windows.dpss(n, NW, sym=False)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))

    # DPSS (Slepian) windows with several time‑bandwidth products
    for NW in (2.0, 3.0, 4.0):

    # ------------------------------------------------------------------

    for rate in np.linspace(0.9985, 0.9999, 40):

def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _lp_update(a: np.ndarray, n: int, M: float,
               active_idx: np.ndarray, mix: float) -> np.ndarray:
    """One LP‑based polishing step.

    Solves:
        maximise   sum(b)
        subject to   a⋆b ≤ M  (for each k in active_idx)
                     b ≥ 0
    The solution b is then ℓ₁‑normalised and mixed with a.
    """
    # Build constraint matrix for the active indices
    if active_idx.size == 0:
        return a  # nothing to do

    rows = []
    for k in active_idx:
        row = np.zeros(n, dtype=np.float64)
        # b_j contributes a_{k-j} if indices are in range
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # fill row[j] = a[k-j]
        # vectorised slice assignment
        idx = np.arange(j_start, j_end + 1, dtype=int)
        row[idx] = a[k - idx]
        rows.append(row)

    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)  # maximise sum(b)  → minimise -sum(b)
    bounds = [(0.0, None)] * n

    # Solve the LP – the problem is tiny (few constraints) so HiGHS is fast.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # If the LP fails, fall back to the current vector.
        return a

    b = res.x
    s = b.sum()
    if s <= 0.0:
        return a
    b = b / s  # enforce ℓ₁‑normalisation

    new_a = (1.0 - mix) * a + mix * b
    # Re‑normalise to alleviate numerical drift
    new_a = new_a / new_a.sum()
    return new_a

Pipeline
--------
1. Generate a large, diverse set of ℓ₁‑normalised seeds:
   - classic windows (uniform, triangular, sine, Gaussian, exponential, …)
   - many β‑shapes, including very U‑shaped variants (α,β < 1)
   - Kaiser, Blackman, Hamming, Hann, DPSS (Slepian) windows
   - random Dirichlet / exponential draws and low‑frequency cosine curves
   - mixtures with the uniform window and the previous best solution.
2. Keep the best few seeds according to the exact surrogate
   ``2·n·max(conv(a,a))`` (sum(a)=1).
3. Run a short exponential‑gradient (multiplicative mirror‑descent) refinement

4. Perform many cheap LP‑based polishing steps.

       maximise   sum(b)

                                 of ``a`` with itself attains its maximum)

   The solution ``b`` is ℓ₁‑normalised and mixed with ``a``.
   Repeating this step provably drives the worst‑case convolution value down.
5. End with a longer EG polishing.
All stages respect the global deadline (default 1000 s).  The best sequence

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate equal to 2·n·max(conv(seq,seq)) when sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    a = np.blackman(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    a = np.hamming(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    a = np.hanning(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, NW: float = 3.0) -> np.ndarray:
    if _sig_windows is None:
        return _uniform(n)
    a = _sig_windows.dpss(n, NW, sym=False)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

    for alpha in (0.1, 0.2, 0.3, 0.5, 0.8, 1.0):

    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_hann_window(n))

    # DPSS (Slepian) windows
    for NW in (2.0, 3.0, 4.0):

    # Low‑frequency random curves
    for nf in (2, 3, 4):

    # Fine sweeps (cheap surrogate)
    for rate in np.linspace(0.9985, 0.9999, 40):

def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _lp_update(a: np.ndarray, n: int, M: float,
               active_idx: np.ndarray, mix: float) -> np.ndarray:
    """One LP‑based polishing step.

    Solve:
        maximise   sum(b)
        subject to a ∗ b ≤ M   for each k in active_idx
                  b ≥ 0
    Then mix the normalized solution with ``a``.
    """
    if active_idx.size == 0:
        return a

    rows = []
    for k in active_idx:
        row = np.zeros(n, dtype=np.float64)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        idx = np.arange(j_start, j_end + 1, dtype=int)
        row[idx] = a[k - idx]
        rows.append(row)

    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)          # maximise sum(b)  → minimise -sum(b)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a

    b = res.x
    s = b.sum()
    if s <= 0.0:
        return a
    b = b / s                                   # ℓ₁‑normalise

    new_a = (1.0 - mix) * a + mix * b
    new_a = new_a / new_a.sum()
    return new_a
