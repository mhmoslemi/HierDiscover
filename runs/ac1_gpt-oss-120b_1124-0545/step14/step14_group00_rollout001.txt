"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.  The search works as
follows:

1. **Rich seed generation** – diverse analytical families (uniform,
   beta‑shape, power‑law, Gaussian, random Dirichlet, sparse k‑point
   uniform) are created and ℓ₁‑normalised.

2. **Coarse ranking** – each seed is evaluated with the exact
   surrogate 2·n·max(conv) (the denominator is 1 after normalisation)
   and the best few (≈5) are kept.

3. **Long exponentiated‑gradient (mirror‑descent) optimisation** – each
   of the top seeds is refined in the log‑softmax space using many
   sub‑gradient steps (≈ 10⁵–10⁶ iterations depending on remaining time).
   The update is ``log_a ← log_a – η·g`` with η≈0.5/√t and ``a = softmax(log_a)``.
   This directly minimises the true max‑convolution objective.

4. **Sub‑gradient polishing** – the current global best is further
   polished by projected sub‑gradient descent (still on the simplex).

5. **Simulated‑annealing final push** – a short SA in logit space
   explores the neighbourhood of the polished solution.

All steps are time‑aware and stop before the 1000 s wall‑clock budget.
The best sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform distribution on a random subset of size k."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to keep simplex constraint automatically
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full convolution to locate the active (max) index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # classic step‑size schedule η_t = 0.5 / sqrt(t)
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best sequence seen and its surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]   # same as surrogate

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple SA in the log‑softmax (logit) space.
    """
    rng = np.random.default_rng(12345)
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-7
        sigma = 0.20 * (1.0 - progress) + 1e-5

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep the size used in the reference solution.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Retrieve previous solution if it exists.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation.
    # ------------------------------------------------------------------
    seeds = []

    # baseline uniform
    seeds.append(_uniform(n))

    # previous best (if any)
    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape grid (coarse)
    for a in np.linspace(0.2, 5.0, 9):
        for b in np.linspace(0.2, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # beta‑shape random draws (log‑uniform parameters)
    for _ in range(120):
        a = 10 ** rng.uniform(-2, 2)   # between 0.01 and 100
        b = 10 ** rng.uniform(-2, 2)
        seeds.append(_beta_shape(n, a, b))

    # power‑law grid + random
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for _ in range(50):
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

    # Gaussian grid + random sigma
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for _ in range(30):
        sigma = rng.uniform(0.05, 0.55)
        seeds.append(_gaussian(n, sigma))

    # Sparse k‑point uniform (k = 10,20,50,100)
    for k in (10, 20, 50, 100):
        seeds.append(_random_k_uniform(n, k, rng))

    # random Dirichlet draws with several concentration values
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # Normalise everything (just in case) and drop degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total <= 0.0:
            continue
        normed_seeds.append(s / total)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Evaluate all seeds (surrogate) and keep the best few.
    # ------------------------------------------------------------------
    scored = []
    for vec in normed_seeds:
        val = _objective_surrogate(vec)
        scored.append((vec, val))
    scored.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate among raw seeds = {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Long exponentiated‑gradient optimisation on the top seeds.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all EG runs.")
            break

        # decide how many iterations we can afford for this seed
        remaining = max(0.0, deadline - time.time())
        # empirical 3000 iterations per second (very rough)
        est_iters = int(min(600_000, remaining * 3000))
        if est_iters < 20_000:
            est_iters = int(min(200_000, remaining * 3000))
        if est_iters <= 0:
            break

        print(f"\n--- Exponentiated‑gradient run #{idx+1} (max_iters≈{est_iters}) ---")
        seq_opt, val_opt = _exp_gradient_opt(seed_vec, n, deadline, max_iters=est_iters)

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient polishing of the current best.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(800_000, remaining * 0.5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n, deadline, max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing refinement (short).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(300_000, remaining * 0.3))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline, max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
