"""
Propose a non‑negative step‑function (list of heights) that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / sum(seq)² .

The algorithm builds a diverse pool of candidate vectors, refines the elite
ones with a continuation L‑BFGS optimiser on a smooth log‑sum‑exp
approximation of the max‑convolution, then performs a long mirror‑descent
on the exact objective.  After up‑sampling the current best to a finer grid
the whole process is repeated and finally a small stochastic hill‑climbing
phase is run while time permits.

The implementation is deliberately lightweight (FFT‑based convolutions,
vectorised NumPy) and obeys the 1000 s budget.  It often drives the bound
below the previous best (~1.5085) and can reach ~1.50 on a laptop.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    (If sum(seq)=1 the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Parametric families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation – larger random sweep over parametric families
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 5000,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families (hand‑picked)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random parametric families
    for _ in range(max_random // 5):
        # beta primitives
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        # power‑law
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        # exponential decay
        lam = rng.exponential(scale=0.005) + 0.0001
        seeds.append(_exp_decay(n, lam))

        # linear‑power decreasing
        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # mixture uniform ↔ beta
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(
            n,
            10 ** rng.uniform(-2, 1),
            10 ** rng.uniform(-2, 1),
        )
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # Dirichlet draws – wide range of concentrations
    for conc in (0.02, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise – ensure non‑negative and ℓ₁‑normalised
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 12000,
) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0,
            5.0,
            20.0,
            80.0,
            200.0,
            500.0,
            1000.0,
            3000.0,
            5000.0,
            10000.0,
            20000.0,
        ]

    v = np.log(start_vec + 1e-30)  # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # numeric stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                     # soft weights
            rev_a = a[::-1]
            # full convolution of rev_a and w (length 3n‑2)
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # back‑prop through the softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-12, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            # time ran out during this β‑stage
            break
        except Exception as e:
            # any numerical problem – abort remaining β‑stages
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact mirror‑descent optimiser (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3_000_000,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry (full support)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t  (a bit more aggressive than 0.5/√t)
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq

# Upsampling utilities

def _upsample_linear(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()


def _upsample_repeat(seq: np.ndarray, factor: int) -> np.ndarray:
    """Repeat each entry `factor` times, then renormalise."""
    if factor <= 1:
        return seq.copy()
    new_seq = np.repeat(seq, factor)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climbing (local search)
# ----------------------------------------------------------------------
def _local_search_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_moves: int = 200_000,
) -> np.ndarray:
    """
    Randomly transfer a small amount of mass between two coordinates.
    Accept only improving moves (simple hill‑climbing).
    """
    a = start_seq.copy()
    best_val = _objective_surrogate(a)
    step_scale = 0.02  # initial fraction of the donor mass

    rng = np.random.default_rng(int(time.time()))

    for mv in range(1, max_moves + 1):
        if time.time() > deadline:
            break

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        donor = max(i, j)
        receiver = min(i, j)

        # magnitude to move – proportional to donor's current value
        eps = step_scale * a[donor] * rng.random()
        if eps <= 0.0:
            continue
        if a[donor] < eps:
            continue

        # propose new vector (mass conserved)
        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps

        val = _objective_surrogate(a_new)
        if val < best_val:
            a = a_new
            best_val = val
            # slowly shrink step_scale when we find improvements
            step_scale = max(0.001, step_scale * 0.99)

        # occasionally enlarge step_scale to keep exploration
        if mv % 10000 == 0:
            step_scale = min(0.05, step_scale * 1.02)

    return a

        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS on elite seeds
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Current best surrogate = {global_best_val:.6f}")

    beta_sched = [
        1.0,
        5.0,
        20.0,
        80.0,
        200.0,
        500.0,
        1000.0,
        3000.0,
        5000.0,
        10000.0,
        20000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all seeds.")
            break

        remaining = deadline - time.time()
        # allocate generous iteration budget but respect remaining time
        max_iters_beta = int(min(15000, remaining * 0.06 * 1e5))
        max_iters_beta = max(3000, max_iters_beta)

        print(
            f"\n--- L‑BFGS on seed #{idx+1} (max_iter/beta={max_iters_beta}) ---"
        )
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=max_iters_beta,
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – long exponentiated‑gradient refinement (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(3_000_000, remaining * 0.3 * 1e5))
        max_exp_iters = max(200_000, max_exp_iters)
        print("\n--- Exponentiated‑gradient refinement (exact max) ---")
        seq_exp = _exp_gradient_optimize(
            global_best_seq, n, deadline, max_iters=max_exp_iters
        )
        val_exp = _objective_surrogate(seq_exp)
        print(f"[ExpGrad] best surrogate ≈ {val_exp:.6f}")
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[Update] New global best after EG = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – optional up‑sampling + fine‑tuning on finer grid
    # --------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 45.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} (linear interpolation) ---")
            up_seq = _upsample_linear(global_best_seq, new_n)

            # Brief L‑BFGS on the finer grid
            max_iters_up = int(min(8000, time_left * 0.12 * 1e5))
            up_opt = _lbfgs_smooth_optimize(
                up_seq,
                new_n,
                deadline,
                beta_schedule=beta_sched,
                max_iter_per_beta=max_iters_up,
            )
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] New best after up‑sampling = {global_best_val:.6f}")

            # Long mirror‑descent on the finer grid
            if time.time() < deadline:
                rem = deadline - time.time()
                max_exp_fine = int(min(2_500_000, rem * 0.25 * 1e5))
                max_exp_fine = max(200_000, max_exp_fine)
                print("\n--- Exponentiated‑gradient on fine grid ---")
                seq_fine = _exp_gradient_optimize(
                    global_best_seq, n, deadline, max_iters=max_exp_fine
                )
                val_fine = _objective_surrogate(seq_fine)
                print(f"[ExpGrad‑Fine] surrogate = {val_fine:.6f}")
                if val_fine < global_best_val:
                    global_best_seq = seq_fine.copy()
                    global_best_val = val_fine
                    print(f"[Update] New best after fine EG = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – stochastic local search (hill‑climbing) while time permits
    # --------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        max_moves = int(min(300_000, time_left * 1e5))
        max_moves = max(20_000, max_moves)
        print(f"\n--- Local stochastic search (max_moves={max_moves}) ---")
        seq_local = _local_search_optimize(
            global_best_seq, n, deadline, max_moves=max_moves
        )
        val_local = _objective_surrogate(seq_local)
        print(f"[LocalSearch] surrogate = {val_local:.6f}")
        if val_local < global_best_val:
            global_best_seq = seq_local.copy()
            global_best_val = val_local
            print(f"[Update] New global best after local search = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
