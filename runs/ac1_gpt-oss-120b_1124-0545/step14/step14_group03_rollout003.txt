"""
Propose a step‐function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

using a multi‑stage scheme:

1. Build a diverse seed pool (deterministic windows, random Dirichlet /
   exponential draws, and random mixtures).  All seeds are ℓ₁‑normalised
   (sum = 1) so the objective is scale‑invariant.

2. Continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The β‑schedule is extended up to 1e5 and a generous per‑β iteration budget
   is used.

3. Exponentiated‑gradient (mirror descent) on the exact max‑convolution
   objective with a decaying step‑size ηₜ = η₀/√t (η₀ = 1.0).  Millions of
   EG steps are performed while time permits.

4. A cheap random hill‑climbing phase that swaps a small amount of mass
   between two coordinates; accepted only if the exact objective improves.

5. Final high‑β L‑BFGS polishing.

The whole routine respects a 1000 s budget, updates the global
``height_sequence_1`` variable and returns the best sequence found as a
plain ``list[float]``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ------------------------------------------------------------
# Low‑level numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# Analytic shape families (return ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0 (NumPy)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a*(1‑α) + b*α, ℓ₁‑renormalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ------------------------------------------------------------
# Seed generation
# ------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_gaussian(n, 0.12))
    seeds.append(_gaussian(n, 0.08))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    for a in np.linspace(0.5, 3.0, 7):
        for b in np.linspace(0.5, 3.0, 7):
            seeds.append(_beta_shape(n, a, b))

    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of already‑generated seeds
    num_mix = 150
    base_idx = np.arange(len(seeds))
    for _ in range(num_mix):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        mix = _mix_shapes(seeds[i], seeds[j], alpha)
        seeds.append(mix)

    # add previous solution if available
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final clean‑up (remove tiny negatives, renormalise)
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ------------------------------------------------------------
# Exact objective (max‑convolution) – sum is assumed to be 1
# ------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) with sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (single active lag)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    # gradient_i = 2 * a_{idx-i} for all i that contribute
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad) where obj = 2·n·log‑sum‑exp_β(conv(a,a))
    and grad is w.r.t. logits v (a = softmax(v)).
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)                     # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # soft weights for each lag
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length 2·n‑1
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule: d obj / d logits = J_softmax · grad_a
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule=None,
    max_iter_per_beta: int = 2000,
) -> np.ndarray:
    """
    Optimize the smoothed max‑convolution objective using L‑BFGS‑B.
    The variable is the logits v of the probability vector a = softmax(v).
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0,
            50_000.0, 100_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(local_v):
            return _smooth_objective(local_v, n, beta_val)

        def callback(local_v):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time out during beta={beta_val}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (EG) on the exact max‑convolution objective.
    Returns (best_vector, best_value).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur_val = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur_val:.6f}")
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

    # final check
    cur_val = _objective_exact(a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()

    return best_a, best_val


# ------------------------------------------------------------
# Random hill‑climbing perturbation
# ------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill climbing: redistribute a small fraction of mass
    from one coordinate to another if it improves the exact objective.
    """
    cur_seq = a.copy()
    cur_val = _objective_exact(cur_seq)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            it += 1
            continue
        eps_frac = rng.random() * 0.08   # up to 8 % of the mass at i
        eps = eps_frac * cur_seq[i]
        new_seq = cur_seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps

        new_val = _objective_exact(new_seq)
        if new_val < cur_val:
            cur_seq = new_seq
            cur_val = new_val
            print(f"[HillClimb] improved to {cur_val:.6f} (iter {it})")
        it += 1

    return cur_seq, cur_val


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(
    seed: int = 123,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a tighter theoretical bound.
    # ------------------------------------------------------------------
    n = 8000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Gather seeds (including previous best if present)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Quick ranking of seeds using the exact objective (sum = 1)
    # ------------------------------------------------------------------
    seed_vals = [(vec, _objective_exact(vec)) for vec in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in seed_vals[:top_k]]

    print(f"Best seed objectives (top {top_k}):")
    for i, (_, val) in enumerate(seed_vals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_exact(global_best_seq)
    print(f"\n[Init] objective = {global_best_val:.6f}")

    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0,
        5_000.0, 10_000.0, 20_000.0,
        50_000.0, 100_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS seeds.")
            break

        remaining = deadline - time.time()
        # allocate a per‑β iteration budget (bounded)
        per_beta_iters = max(500, int(min(4000,
                                          remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={per_beta_iters}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta_iters,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient (exact) refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(5_000_000,
                                remaining * 0.25 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp, val_exp = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_exp_iters,
                eta0=1.0,
            )
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved objective to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – cheap random hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random hill‑climbing phase ---")
        seq_hc, val_hc = _hill_climb(
            global_best_seq,
            n,
            deadline,
            max_iters=300_000,
        )
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] improved objective to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final high‑β L‑BFGS polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(500, int(min(3000,
                                          remaining * 0.06 * 1e5)))
        print("\n--- Final L‑BFGS polishing (high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[50_000.0, 100_000.0],
            max_iter_per_beta=per_beta_iters,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
