```python
"""
Optimise step function heights to minimise `evaluate_sequence`.

The algorithm:
1. Initialise a non‑negative vector `f` (length n, Σ f = 1).  If a previous
   successful construction exists, it is down‑sampled; otherwise a Dirichlet
   seed is used.
2. Iteratively solve a linear programme

       minimise   t
       subject to  (f * g)[k] ≤ t   for all k
                   Σ g = 1,   g ≥ 0

   The LP returns a new distribution `g` and the improved bound `t`.
   We replace `f ← (f + g)/2` (average) and repeat while the bound
   decreases.  Each LP reduces the maximum convolution value,
   i.e. the quantity `max(p*p)` that appears in the official objective.
   The LP is solved with `scipy.optimize.linprog` (highs).

3. After the LP phase, a short soft‑max Adam optimiser (smooth surrogate
   of the max) refines the result.

4. The final candidate is evaluated with the official `evaluate_sequence`
   and stored in the global variable `height_sequence_1`.

The whole search respects the supplied time budget (default 1000 s) and
returns a list of heights ready for submission.
"""

import time
import math
import numpy as np

# optional SciPy import – required for the LP stage
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic convolution / simplex utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_maxconv(p: np.ndarray) -> float:
    """Return max(p * p) for a non‑negative vector (no normalisation)."""
    conv = _conv_fft(p, p)
    return float(conv.max())


def _objective_surrogate(p: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(p,p)) (the value returned by evaluate_sequence
    when Σp = 1)."""
    n = len(p)
    return 2.0 * n * _objective_maxconv(p)


# ----------------------------------------------------------------------
# Linear‑programming step (minimise the maximal convolution)
# ----------------------------------------------------------------------
def _lp_minimize_max(p: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Solve the LP:

        minimise   t
        subject to (p * g)[k] ≤ t  for every shift k
                   Σ g = 1, g ≥ 0

    Returns (g, t) where g is the optimiser (∑g = 1) and t = max(p * g).
    """
    if not _SCIPY_AVAILABLE:  # pragma: no cover
        raise RuntimeError("SciPy not available – LP stage cannot run.")

    n = len(p)
    L = 2 * n - 1

    # Build dense constraint matrix A where A[k,i] = p[i] if i contributes to shift k
    A = np.zeros((L, n), dtype=np.float64)
    for i, pi in enumerate(p):
        # rows i … i+n-1 correspond to the shift positions where p[i] participates
        A[i:i + n, i] = pi

    # Upper‑bound constraints: A @ g - t ≤ 0   →   [A | -1] @ [g; t] ≤ 0
    A_ub = np.hstack([A, -np.ones((L, 1), dtype=np.float64)])
    b_ub = np.zeros(L, dtype=np.float64)

    # Equality: Σ g = 1
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([1.0], dtype=np.float64)

    # Objective: minimise t
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    bounds = [(0.0, None)] * (n + 1)   # g_i ≥ 0, t ≥ 0

    # Solve with the HiGHS interior‑point method (fast for dense matrices)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:  # pragma: no cover
        raise RuntimeError(f"LP failed: {res.message}")

    g = res.x[:n]
    t = res.x[-1]
    # numerical safety – enforce simplex
    g = _proj_simplex(g)
    return g, float(t)


# ----------------------------------------------------------------------
# Soft‑max Adam optimiser (smooth approximation of the max)
# ----------------------------------------------------------------------
def _adam_softmax_optimize(start: np.ndarray,
                           deadline: float,
                           max_iters: int = 200_000,
                           alpha_start: float = 20.0,
                           alpha_end: float = 10000.0,
                           lr: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate

        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]

        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:

            best_x = x.copy()

        if t % 30000 == 0:

"""
Optimise step‑function heights to minimise ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Initialisation** – start from the previous best construction (if
   available) down‑sampled to a manageable length, otherwise use a
   random Dirichlet vector.  The vector is projected onto the probability
   simplex (non‑negative, sum = 1).

2. **Linear‑programming reduction** – repeatedly solve

       minimise   t
       subject to (f * g)[k] ≤ t   (for every shift k)
                  Σ g = 1,   g ≥ 0

   The LP returns a new distribution ``g`` and a strictly smaller bound
   ``t`` whenever possible.  We replace the current vector by the average
   ``(f+g)/2`` (still a simplex point) and repeat while the bound improves.
   The LP is solved with ``scipy.optimize.linprog`` (HiGHS) on a dense
   constraint matrix; a dozen iterations are typically enough.

3. **Soft‑max Adam polishing** – a smooth approximation of the max
   (log‑sum‑exp) is minimised with Adam, starting from the LP‑refined
   vector.  The temperature grows from 20 to 12000, giving a fine‑grained
   final adjustment.

The final vector is evaluated with ``evaluate_sequence`` and stored in the
global name ``height_sequence_1`` for reuse by later runs.

The whole search respects the supplied ``budget_s`` (default 1000 s) and
returns a list of non‑negative heights ready for submission.
"""

import time
import math
import numpy as np

# Optional SciPy import – required for the LP stage
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic convolution / simplex utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_maxconv(p: np.ndarray) -> float:
    """Return max(p * p) for a non‑negative vector."""
    conv = _conv_fft(p, p)
    return float(conv.max())


def _objective_surrogate(p: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(p,p)) – equal to evaluate_sequence for Σp=1."""
    n = len(p)
    return 2.0 * n * _objective_maxconv(p)

# Linear‑programming phase: minimise the maximal convolution bound
# ----------------------------------------------------------------------
def _lp_minimize_max(p: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Solve the LP

        minimise   t
        subject to (p * g)[k] ≤ t   for every shift k
                   Σ g = 1,   g ≥ 0

    Returns (g, t) where g is a probability vector (Σg=1) and
    t = max(p * g) is the optimal bound.
    """
    if not _SCIPY_AVAILABLE:  # pragma: no cover
        raise RuntimeError("SciPy not available – LP stage cannot be executed.")

    n = len(p)
    L = 2 * n - 1

    # Build the dense matrix A where A[k,i] = p[i] if i contributes to shift k
    A = np.zeros((L, n), dtype=np.float64)
    for i, pi in enumerate(p):
        A[i:i + n, i] = pi    # rows where p[i] participates

    # Upper‑bound constraints: A @ g - t ≤ 0
    A_ub = np.hstack([A, -np.ones((L, 1), dtype=np.float64)])
    b_ub = np.zeros(L, dtype=np.float64)

    # Equality Σ g = 1
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64),
                      np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([1.0], dtype=np.float64)

    # Objective: minimise t
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    bounds = [(0.0, None)] * (n + 1)   # g_i ≥ 0, t ≥ 0

    # Solve with HiGHS (fast for dense problems)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success:  # pragma: no cover
        raise RuntimeError(f"LP failed: {res.message}")

    g = res.x[:n]
    t = float(res.x[-1])

    # Numerical clean‑up – enforce the simplex exactly
    g = _proj_simplex(g)
    return g, t
