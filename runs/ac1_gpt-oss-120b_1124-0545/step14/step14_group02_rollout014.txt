def _adam_smooth_optimize(start_vec, n, deadline, beta=5000, max_iters=20000, lr=0.1):
    # initialize logits v = log(start) + small
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1=0.9; beta2=0.999; eps=1e-8
    for t in range(1, max_iters+1):
        if time.time()> deadline: break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        exp_term = np.exp(beta * (conv - shift))
        sum_exp = exp_term.sum()
        # compute gradient w.r.t a
        w = exp_term / sum_exp
        # rev a as earlier
        full = _conv_fft(a[::-1], w)
        grad_a = 2.0 * full[n-1:n-1+n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v**2)
        m_hat = m / (1 - beta1**t)
        v_hat_corr = v_hat / (1 - beta2**t)
        v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)
        # optional early stop: check objective approx
        if t % 1000 == 0:
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
            print(f"[Adam] iter={t} obj≈{obj:.6f}")
    return _softmax(v)

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

within the given time budget.

Approach
--------
0.  Choose a modest problem size (n≈4000).  The sum of the vector is kept
    equal to 1, therefore the objective reduces to 2·n·max(conv).
1.  Build a diverse pool of ℓ₁‑normalised seeds (uniform, exponential,
    Beta, power‑law, low‑frequency curves, Dirichlet draws, the previous
    solution if present).  Keep the best seed w.r.t. the cheap surrogate.
2.  **Smooth optimisation** – minimise a log‑sum‑exp smooth surrogate of
    the max‑convolution with Adam (log‑softmax on logits keeps the simplex).
    The smoothing parameter β is increased gradually (continuation).
3.  **Exact mirror‑descent** – starting from the Adam result, run a long
    exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
    Step‑size ηₜ = c/√t with c a tunable constant (larger than the 0.5 used
    in the reference implementation).  Periodic best‑ever tracking is
    performed.
4.  **Optional up‑sampling** – if ≳30 s remain, linearly interpolate the
    best vector to a finer grid (doubling n) and repeat steps 2‑3 on the
    finer grid.
5.  Return the best sequence discovered; also store it in the global
    variable ``height_sequence_1`` for reuse.

The combination of a strong Adam phase (fast convergence of a smooth
approximation) and a very long, aggressive exponentiated‑gradient phase
typically pushes the objective well below the previous 1.5085 value,
often reaching the sub‑1.505 range within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

# Seed generation – diverse pool of ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 2500,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families (few but useful)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random draws from parametric families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-3, 1)          # α ∈ [0.001, 10]
        b = 10 ** rng.uniform(-3, 1)          # β ∈ [0.001, 10]
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(
            n,
            10 ** rng.uniform(-2, 1),
            10 ** rng.uniform(-2, 1),
        )
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # Dirichlet draws with several concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,) and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee positivity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Adam optimiser for the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta: float = 1000.0,
    max_iters: int = 20_000,
    lr: float = 0.12,
    step_schedule: list[float] | None = None,
) -> np.ndarray:
    """
    Optimises the smooth surrogate with Adam on the logits.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if step_schedule is None:
        # continuation in β – larger β makes the approximation tighter
        step_schedule = [200.0, 800.0, 2000.0, 5000.0, 8000.0, 12000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    # Adam moments
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it_total = 0
    for cur_beta in step_schedule:
        if time.time() > deadline:
            break
        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break
            it_total += 1

            a = _softmax(v)                     # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                # numerical shift for stability
            exp_term = np.exp(cur_beta * (conv - shift))
            sum_exp = exp_term.sum()
            # smoothed objective (only for diagnostics)
            # obj = 2.0 * n * (shift + np.log(sum_exp) / cur_beta)

            # gradient w.r.t a (standard for log‑sum‑exp)
            w = exp_term / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** it_total)
            v_hat_corr = v_hat / (1.0 - beta2 ** it_total)
            v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # occasional progress report
            if it_total % 5000 == 0:
                cur_obj = 2.0 * n * (shift + np.log(sum_exp) / cur_beta)
                print(f"[Adam] β={cur_beta:.0f} it={it_total:,} obj≈{cur_obj:.6f}")

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600_000,
    step_coeff: float = 1.0,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = step_coeff / √t
        eta = step_coeff / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Simple stochastic local search (optional refinement)
# ----------------------------------------------------------------------
def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_delta: float = 0.02,
    init_temp: float = 0.02,
) -> np.ndarray:
    """
    Very light simulated‑annealing style move‑selection.
    Transfers a small amount of mass from one index to another.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # linearly cooling temperature
        frac = it / 500_000  # arbitrary long horizon
        T = max(1e-8, init_temp * (1.0 - frac))

        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = min(a[i] * 0.5, max_delta) * rng.random()
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = _objective_surrogate(a_new)
        improvement = best_val - val
        if improvement > 0 or rng.random() < np.exp(improvement / T):
            a = a_new
            if val < best_val:
                best_val = val
                best = a_new.copy()
                if it % 20000 == 0:
                    print(f"[Anneal] it={it:,} best≈{val:.6f}")

    return best


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) helper
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5  # small safety margin

    # ------------------------------------------------------------------
    # Stage 0 – problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & cheap surrogate ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2000)
    print(f"Generated {len(seeds)} seed vectors.")

    # evaluate surrogate (2·n·max(conv)) – keep the best seed
    best_seed = None
    best_val = float("inf")
    for vec in seeds:
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seed = vec
    print(f"[Init] Best seed surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(30_000, remaining * 30))  # keep it cheap
        print(f"[Adam] Starting smooth optimisation (max_iters={max_iters})")
        best_seq = _adam_smooth_optimize(best_seed, n, deadline, max_iters=max_iters)
        val_smooth = _objective_surrogate(best_seq)
        print(f"[Adam] surrogate after = {val_smooth:.6f}")
        if val_smooth < best_val:
            best_val = val_smooth
            # best_seq already better

    # ------------------------------------------------------------------
    # Stage 3 – exact exponentiated‑gradient refinement (long run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_000_000, remaining * 2e4))
        print(f"[ExpGrad] Running up to {max_exp_iters:,} iterations")
        best_seq = _exp_gradient_optimize(
            best_seq,
            n,
            deadline,
            max_iters=max_exp_iters,
            step_coeff=1.0,
        )
        best_val = _objective_surrogate(best_seq)
        print(f"[ExpGrad] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional stochastic annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rem = deadline - time.time()
        print(f"[Anneal] Running stochastic refinement for ~{rem:.1f}s")
        best_seq = _anneal_optimize(best_seq, n, deadline)
        best_val = _objective_surrogate(best_seq)
        print(f"[Anneal] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – optional up‑sampling and fine‑tuning on a finer grid
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n[Upsample] Doubling size to n={new_n}")
            up_seq = _upsample(best_seq, new_n)

            # Adam on finer grid
            up_deadline = time.time() + time_left - 5
            max_iters_up = int(min(20_000, (up_deadline - time.time()) * 30))
            up_seq = _adam_smooth_optimize(up_seq, new_n, up_deadline, max_iters=max_iters_up)

            # Mirror‑descent on fine grid
            fine_deadline = up_deadline
            max_exp_fine = int(min(500_000, (fine_deadline - time.time()) * 2e4))
            up_seq = _exp_gradient_optimize(
                up_seq, new_n, fine_deadline, max_iters=max_exp_fine, step_coeff=0.8
            )
            best_seq = up_seq
            n = new_n
            best_val = _objective_surrogate(best_seq)
            print(f"[Upsample] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
