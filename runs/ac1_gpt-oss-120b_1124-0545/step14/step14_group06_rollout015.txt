"""
Propose a step‑function minimizing

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

The implementation combines three ideas:

1. A rich pool of analytic windows (uniform, triangular, sine, Gaussian,
   exponential, power, beta, Tukey, Kaiser, Hamming, Blackman, Welch …)
   together with random Dirichlet draws.
2. A small number of LP refinements.  For a fixed vector ``f`` the constraints
   ``conv(f,g) ≤ M`` are linear in ``g``.  We maximise ``sum(g)`` under those
   constraints (SciPy highs).  Mixing the resulting ``g`` with ``f`` (convex
   combination) provably does not increase the objective.
3. A final long mirror‑descent (exponential‑gradient) polishing on the exact
   max‑convolution surrogate.

All stages respect the global deadline (default 1000 s).  The best sequence
found is stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate used during the search (exact when Σ seq = 1):
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window (beta >= 0)."""
    try:
        from scipy.signal import kaiser
    except Exception:  # pragma: no cover
        return _uniform(n)
    a = kaiser(n, beta, sym=False).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    try:
        from scipy.signal import hamming
    except Exception:  # pragma: no cover
        return _uniform(n)
    a = hamming(n, sym=False).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    try:
        from scipy.signal import blackman
    except Exception:  # pragma: no cover
        return _uniform(n)
    a = blackman(n, sym=False).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _welch_window(n: int) -> np.ndarray:
    """Welch (parabolic) window."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - ((i - (n - 1) / 2.0) / ((n - 1) / 2.0)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (including fine sweeps)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidates."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.25, 0.3):
        seeds.append(_gaussian(n, sigma))
    for rate in (0.998, 0.999, 0.9995, 0.9998, 0.9999):
        seeds.append(_exponential_decay(n, rate))

    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        seeds.append(_power(n, p))

    # beta shapes – dense sweep
    for alpha in (0.5, 0.8, 1.0, 1.2):
        for beta in (1.0, 1.5, 2.0, 3.0, 5.0):
            seeds.append(_beta_shape(n, alpha, beta))

    # logistic – several scales
    for scale in (5.0, 10.0, 20.0, 40.0):
        seeds.append(_logistic_window(n, scale))

    # Tukey windows
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # Additional windows from SciPy
    for beta in (5.0, 8.0, 12.0):
        seeds.append(_kaiser_window(n, beta))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_welch_window(n))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Fine parameter sweeps for exponential and power families
    for rate in np.linspace(0.995, 0.99995, 60):
        seeds.append(_exponential_decay(n, rate))
    for p in np.linspace(0.1, 3.0, 60):
        seeds.append(_power(n, p))

    # Include the previous solution if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final safety – ensure all seeds are ℓ₁‑normalised
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size – √t schedule
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # project onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, n: int, M: float) -> np.ndarray | None:
    """
    Solve:
        maximise   sum(g)
        subject to   conv(f, g) ≤ M  (for all shifts)
                    g ≥ 0
    Returns the optimal g (unnormalised) or None on failure.
    """
    # Build sparse matrix A where A[k, j] = f_{k-j} if 0 ≤ k-j < n else 0.
    # Using the identity: row = i + j, col = j, data = f_i
    i = np.arange(n, dtype=np.int64)
    j = np.arange(n, dtype=np.int64)

    row = np.repeat(i, n) + np.tile(j, n)          # shape (n*n,)
    col = np.tile(j, n)                           # shape (n*n,)
    data = np.repeat(f, n)                        # each f_i repeated n times

    A = scipy.sparse.coo_matrix((data, (row, col)), shape=(2 * n - 1, n)).tocsr()

    c = -np.ones(n, dtype=np.float64)             # maximise sum(g)  -> minimise -sum(g)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        print("[LP] optimisation failed:", res.message)
        return None


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length choice – 2000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # evaluate surrogate for all seeds, keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, v) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # Quick EG refinement on the best seed
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick EG warm‑up ---")
        remaining = deadline - time.time()
        max_iters = int(min(150_000, remaining * 1800))
        best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                               max_iters=max_iters,
                                               eta0=0.30)
        print(f"[Warm‑up EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Linear‑programming refinement loop
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP refinement ---")
        lp_iters = 5
        mix_t = 0.3   # convex‑combination weight
        while lp_iters > 0 and time.time() < deadline:
            conv = _conv_fft(best_seq, best_seq)
            M = np.max(conv)

            g = _lp_refine(best_seq, n, M)
            if g is None:
                break

            # normalise g to the simplex
            if g.sum() > 0:
                g = g / g.sum()
            else:
                g = _uniform(n)

            # convex combination (preserves sum = 1)
            cand = (1.0 - mix_t) * best_seq + mix_t * g
            cand = _proj_simplex(cand)

            cand_val = _objective_surrogate(cand)
            if cand_val < best_val:
                best_val, best_seq = cand_val, cand.copy()
                print(f"[LP] iteration improved to {best_val:.6f}")
            else:
                print(f"[LP] iteration no improvement (val={cand_val:.6f})")
            lp_iters -= 1

    # ------------------------------------------------------------------
    # Final long EG polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 2100))
        best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                               max_iters=max_iters,
                                               eta0=0.22)
        print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
