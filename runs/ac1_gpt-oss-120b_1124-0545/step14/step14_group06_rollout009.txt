"""
Propose the best non‑negative step function (length 4000) that minimizes

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))² .

The algorithm explores a rich family of analytic windows (exponential,
power, Gaussian, logistic, beta, Kaiser, Tukey) together with many random
Dirichlet draws.  For each shape we sweep its parameters densely (or
sample them randomly) and keep the best candidate according to the exact
surrogate 2·n·max(conv).  After a global search a short mirror‑descent
(EG) run refines the best vector.  The whole pipeline respects the
global deadline (default 1000 s) and finally returns the best sequence.
"""

import time
import math
import numpy as np
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """
    Exact surrogate for the objective (sum(seq) == 1):
        2·n·max(conv(seq,seq))
    """
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0) -> np.ndarray:
    shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window, β ≥ 0."""
    if beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (n - 1) / 2.0
    t = (i - a) / a                     # range [-1,1]
    a = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) optimiser on the simplex
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 300_000, eta0: float = 0.25) -> np.ndarray:
    """Multiplicative (KL‑mirror) descent on the probability simplex."""
    x = start.copy()
    best = x.copy()
    best_val = _objective_surrogate(x, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]          # ∂(x*x)_k / ∂x_i = 2·x_{k-i}

        eta = eta0 / math.sqrt(it)    # √t schedule
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)          # multiplicative update

        # re‑project onto the simplex
        s = x.sum()
        if s > 0.0:
            x = x / s
        else:
            x = _uniform(n)

        cur_val = _objective_surrogate(x, n)
        if cur_val < best_val:
            best_val = cur_val
            best = x.copy()
    return best

    polish, and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # Helper to keep track of the best candidate
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    best_rate = None
    best_power = None
    best_sigma = None
    best_scale = None
    best_alpha = None
    best_beta = None
    best_kaiser = None

    def _try(seq, val, tag=None):
        nonlocal best_seq, best_val
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            # store tag‑specific parameters for later fine‑tuning
            return True
        return False

    # ------------------------------------------------------------------
    # 1) deterministic families
    # ------------------------------------------------------------------
    # uniform (baseline)
    _try(_uniform(n), _objective_surrogate(_uniform(n), n), 'uniform')

    # ------------------------------------------------------------------
    # 2) Exponential decay – a dense sweep plus a refined local zoom
    # ------------------------------------------------------------------
    rates = np.linspace(0.998, 0.99995, 2000)
    for r in rates:
        if time.time() > deadline:
            break
        seq = _exponential_decay(n, r)
        if _try(seq, _objective_surrogate(seq, n), 'exp'):
            best_rate = r
    # refined neighbourhood around the current best rate (if any)
    if best_rate is not None:
        lo = max(0.0, best_rate - 5e-5)
        hi = min(0.999999, best_rate + 5e-5)
        fine = np.linspace(lo, hi, 2000)
        for r in fine:
            if time.time() > deadline:
                break
            seq = _exponential_decay(n, r)
            if _try(seq, _objective_surrogate(seq, n), 'exp_fine'):
                best_rate = r

    # ------------------------------------------------------------------
    # 3) Power‑law shape
    # ------------------------------------------------------------------
    powers = np.linspace(0.1, 5.0, 2000)
    for p in powers:
        if time.time() > deadline:
            break
        seq = _power(n, p)
        if _try(seq, _objective_surrogate(seq, n), 'pow'):
            best_power = p
    # fine‑tune the best power if we have one
    if best_power is not None:
        lo = max(0.0, best_power - 0.05)
        hi = best_power + 0.05
        fine = np.linspace(lo, hi, 2000)
        for p in fine:
            if time.time() > deadline:
                break
            seq = _power(n, p)
            if _try(seq, _objective_surrogate(seq, n), 'pow_fine'):
                best_power = p

    # ------------------------------------------------------------------
    # 4) Gaussian windows (σ as a fraction of n)
    # ------------------------------------------------------------------
    sigmas = np.linspace(0.05, 0.35, 1500)
    for s in sigmas:
        if time.time() > deadline:
            break
        seq = _gaussian(n, s)
        if _try(seq, _objective_surrogate(seq, n), 'gauss'):
            best_sigma = s
    if best_sigma is not None:
        lo = max(0.0, best_sigma - 0.02)
        hi = best_sigma + 0.02
        fine = np.linspace(lo, hi, 1500)
        for s in fine:
            if time.time() > deadline:
                break
            seq = _gaussian(n, s)
            if _try(seq, _objective_surrogate(seq, n), 'gauss_fine'):
                best_sigma = s

    # ------------------------------------------------------------------
    # 5) Logistic windows (scale parameter)
    # ------------------------------------------------------------------
    scales = np.linspace(4.0, 45.0, 1500)
    for sc in scales:
        if time.time() > deadline:
            break
        seq = _logistic_window(n, sc)
        if _try(seq, _objective_surrogate(seq, n), 'log'):
            best_scale = sc
    if best_scale is not None:
        lo = max(0.1, best_scale - 1.0)
        hi = best_scale + 1.0
        fine = np.linspace(lo, hi, 1500)
        for sc in fine:
            if time.time() > deadline:
                break
            seq = _logistic_window(n, sc)
            if _try(seq, _objective_surrogate(seq, n), 'log_fine'):
                best_scale = sc

    # ------------------------------------------------------------------
    # 6) Beta‑shape – random sampling in log‑space
    # ------------------------------------------------------------------
    for _ in range(3000):
        if time.time() > deadline:
            break
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        seq = _beta_shape(n, alpha, beta)
        if _try(seq, _objective_surrogate(seq, n), 'beta'):
            best_alpha, best_beta = alpha, beta

    # ------------------------------------------------------------------
    # 7) Kaiser windows (β‑parameter)
    # ------------------------------------------------------------------
    betas = np.linspace(0.5, 12.0, 1500)
    for b in betas:
        if time.time() > deadline:
            break
        seq = _kaiser_window(n, b)
        if _try(seq, _objective_surrogate(seq, n), 'kaiser'):
            best_kaiser = b
    if best_kaiser is not None:
        lo = max(0.0, best_kaiser - 0.3)
        hi = best_kaiser + 0.3
        fine = np.linspace(lo, hi, 1500)
        for b in fine:
            if time.time() > deadline:
                break
            seq = _kaiser_window(n, b)
            if _try(seq, _objective_surrogate(seq, n), 'kaiser_fine'):
                best_kaiser = b

    # ------------------------------------------------------------------
    # 8) Random Dirichlet / exponential draws
    # ------------------------------------------------------------------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(300):
            if time.time() > deadline:
                break
            seq = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            _try(seq, _objective_surrogate(seq, n), 'dir')
    for _ in range(500):
        if time.time() > deadline:
            break
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()
        _try(a, _objective_surrogate(a, n), 'exp_rand')

    # ------------------------------------------------------------------
    # 9) If we have a previous solution, use it as a seed
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seq = arr / arr.sum()
            _try(seq, _objective_surrogate(seq, n), 'prev')

    print(f"After global sweep: best surrogate ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # 10) Mirror‑descent (multiplicative EG) polishing on the best seed
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # decide a generous iteration budget proportional to remaining time
        it_budget = int(min(500_000, remaining * 2000))
        best_seq = _expgrad_optimize(best_seq, n, deadline,
                                     max_iters=it_budget, eta0=0.22)
        best_val = _objective_surrogate(best_seq, n)
        print(f"[EG polish] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (exact – includes the safety clamps)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("evaluate_sequence raised:", exc)
        true_val = None

    print(f"\n=== Result ===\nSurrogate ≈ {best_val:.6f}, true = {true_val:.6f}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
