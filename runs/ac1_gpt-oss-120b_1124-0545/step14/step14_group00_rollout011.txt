def solve_lp_max_sum(a, M, deadline):
    n = a.shape[0]
    L = 2 * n - 1
    # Build matrix A
    diags = [np.full(n, coeff, dtype=np.float64) for coeff in a]  # each diag of length n
    offsets = np.arange(n)  # 0..n-1
    A = sp.diags(diags, offsets, shape=(L, n), format='csr')
    # solve LP
    c = -np.ones(n, dtype=np.float64)  # maximize sum(b)
    b_ub = np.full(L, M, dtype=np.float64)
    bounds = [(0, None)] * n
    # Use SciPy linprog with method='highs'
    res = scipy.optimize.linprog(c=c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                 method='highs',
                                 options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
    if res.success:
        return res.x
    else:
        return None

"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a deterministic LP‑based improvement loop:

1. Start from the previously best sequence (if any) or a random Dirichlet
   seed, normalise its ℓ₁‑norm to √(2 n).
2. Repeatedly
      – compute M = max(conv(a,a))
      – build the full linear‑convolution matrix of a (sparse Toeplitz)
      – solve the linear programme
            maximise   1ᵀ b
            subject to  a * b ≤ M   (component‑wise) ,   b ≥ 0
        which yields a vector b with a larger ℓ₁‑norm than a.
      – rescale b back to the original ℓ₁‑norm (g = (∥a∥₁/∥b∥₁)·b)
        and replace a ← (1‑t) a + t g (t = 0.5).  The ℓ₁‑norm stays fixed
        while the max‑convolution strictly decreases.
   The loop stops when no improvement is possible or the time budget
   expires.
3. After the LP phase a short polishing with the exact objective is
   performed (exponentiated‑gradient) and the final sequence is
   re‑normalised to ℓ₁‑norm 1 (scaling does not change the objective).

The routine respects the 1000 s budget, prints progress, stores the best
sequence in the global variable ``height_sequence_1`` and returns it as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# SciPy imports (required for the linear programme)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception as _e:      # pragma: no cover
    _opt = None
    _sp = None

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n·log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


# ----------------------------------------------------------------------
# Build the Toeplitz convolution matrix A such that (A @ b) = a * b
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> _sp.csr_matrix:
    """
    Returns a sparse matrix A (shape (2n‑1, n)) with
        (A @ b) == np.convolve(a, b, mode='full')
    for any vector b of length n.
    """
    n = a.shape[0]
    L = 2 * n - 1
    # each diagonal i (0 ≤ i < n) contains the constant a[i]
    # and has exactly n entries
    data = [np.full(n, coeff, dtype=np.float64) for coeff in a]
    offsets = np.arange(n, dtype=int)          # 0,1,…,n‑1
    A = _sp.diags(data, offsets, shape=(L, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Linear‑programming step: maximise ℓ₁‑norm of b under convolution bound
# ----------------------------------------------------------------------
def _lp_max_sum(a: np.ndarray, M: float, deadline: float):
    """
    Solve
        maximise   1ᵀ b
        subject to  A @ b ≤ M·1 ,  b ≥ 0
    where A is the convolution matrix of a.
    Returns the optimal b (as a dense np.ndarray) or None on failure.
    """
    if _opt is None or _sp is None:          # pragma: no cover
        raise RuntimeError("SciPy is required for the LP step.")
    n = a.shape[0]
    L = 2 * n - 1

    A = _build_conv_matrix(a)

    c = -np.ones(n, dtype=np.float64)        # maximise sum(b)  → minimise -sum(b)
    b_ub = np.full(L, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # No explicit time limit for the solver; we simply stop if the outer
    # loop runs out of wall‑clock time.
    try:
        res = _opt.linprog(
            c=c, A_ub=A, b_ub=b_ub, bounds=bounds,
            method='highs',
            options={'presolve': True, 'disp': False}
        )
    finally:
        # free the large matrix ASAP
        del A

    if res.success:
        return res.x
    else:
        print(f"[LP] solver failed: {res.message}")
        return None


# ----------------------------------------------------------------------
# Simple exponentiated‑gradient (mirror‑descent) polishing on the exact
# objective – useful for a final fine‑tune.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000):
    """Mirror‑descent on the exact max‑convolution, returns (seq, val)."""
    n = start_seq.size
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (as in the reference code)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val
