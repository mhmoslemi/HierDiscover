def _adam_softmax_optimize(start, n, deadline, max_iters=300_000, lr=0.12, alpha_start=5.0, alpha_end=20000.0):
   x = start.copy()
   best_x = x.copy()
   best_val = _objective_surrogate(x)
   m = np.zeros_like(x)
   v = np.zeros_like(x)
   beta1=0.9; beta2=0.999; eps=1e-8

   N = _next_pow2(3*n -2)  # length for convolution in both directions

   for t in range(1, max_iters+1):
       if time.time() > deadline: break

       # compute fft of x
       fx = np.fft.rfft(x, n=N)
       conv_full = np.fft.irfft(fx * fx, n=N)[:2*n-1]
       # compute softmax weights
       alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iters)  # linear schedule
       max_c = np.max(conv_full)
       exp_vals = np.exp(alpha * (conv_full - max_c))
       soft = exp_vals / np.sum(exp_vals)

       # gradient: corr(p, soft)
       fs = np.fft.rfft(soft, n=N)
       corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
       grad = 2.0 * corr

       # Adam update
       m = beta1 * m + (1 - beta1) * grad
       v = beta2 * v + (1 - beta2) * (grad ** 2)
       m_hat = m / (1 - beta1 ** t)
       v_hat = v / (1 - beta2 ** t)
       step = lr * m_hat / (np.sqrt(v_hat) + eps)
       x = x - step
       
       # projection to simplex
       x = _proj_simplex(x)

       # evaluate surrogate (maybe every 1000 steps)
       cur_val = _objective_surrogate(x)
       if cur_val < best_val:
           best_val = cur_val
           best_x = x.copy()
           # maybe output progress
   return best_x, best_val

def _expgrad_optimize(start, n, deadline, max_iters, eta0=0.2):
   x = start.copy()

       cur_val = _objective_surrogate(x)
       if cur_val < best_val:
           best_val = cur_val

"""
Propose a step‑function (non‑negative heights) that beats the previous
1.508549 bound.  The method works in four stages:

0.  Build a very diverse pool of “seed’’ vectors using classic window
    functions (Hann, Hamming, Blackman, Tukey, Kaiser, DPSS, …),
    analytic families (Gaussian, Beta, sine‑power), random Dirichlet draws
    and random mixes of the above.  All vectors are ℓ₁‑normalised
    (∑a_i = 1).

1.  Pick the few best seeds (according to the exact objective
    `2·n·max(conv)`, which for a normalised vector equals the official
    `evaluate_sequence` value) and polish each of them with a high‑precision
    Adam optimiser on a smooth surrogate of the max (log‑sum‑exp with an
    annealed temperature).  The optimisation runs for as many iterations
    as the time budget permits.

2.  Refine the current best vector with a sparse exponential‑gradient
    (mirror‑descent) step that directly follows the sub‑gradient of the
    active convolution entry.

3.  Finish with a simulated‑annealing pass and a cheap hill‑climbing
    local search, both respecting the simplex constraint.

All stages respect the global time budget (default 1000 s).  The final
vector is stored in ``height_sequence_1`` for reuse by later runs.
"""

import time
import math
import numpy as np
from scipy import signal  # windows: kaiser, dpss


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (returns full 2·n‑1 length)."""
    n = len(a) + len(b) - 1
    N = _next_pow2(n)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:n]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact surrogate for the official objective when ∑p = 1:
        2·n·max(conv(p,p))
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic window families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (
                1
                + np.cos(
                    np.pi
                    * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)
                )
            )
    return a / a.sum()


def _sine_power_window(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** power
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    a = signal.windows.kaiser(n, beta, sym=True)
    return a / a.sum()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    a = signal.windows.dpss(n, nw, Kmax=1, sym=True)[0]
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    alpha_start: float = 5.0,
    alpha_end: float = 20_000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth log‑sum‑exp surrogate of the ℓ∞‑norm."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(2 * n - 1)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full convolution via FFT
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        # temperature schedule (linear in iteration)
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iters)

        # soft‑max of conv (log‑sum‑exp)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # gradient = 2·corr(x, soft)
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # keep on simplex
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.25,
    sparsity_thr: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Mirrored (exponential) gradient using the active convolution shift."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        if sparsity_thr > 0.0:
            x[x < sparsity_thr] = 0.0

        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _simulated_annealing_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.5:
            # reshuffle a small random subset
            k_sub = rng.integers(5, 16)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        cand_val = _objective_surrogate(cand)
        delta = cand_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
    return best, best_val


def _hill_climb_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Very cheap greedy local search – move a tiny amount of mass."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        if cur[i] <= 1e-14:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        eps = rng.random() * min(cur[i], 0.05)  # move up to 5 % of i‑mass
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_val = _objective_surrogate(cand)
        if cand_val < cur_val:
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm builds a large pool of analytic and
    random seeds, keeps the best few, refines them with Adam, exponential‑
    gradient, simulated‑annealing and a final hill‑climb.  The best
    sequence (as a plain Python list) is stored in the global name
    `height_sequence_1`.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0        # tiny safety margin

    # ------------------------------------------------------------------
    # Length (4000–6000 is a good trade‑off).  Larger n gives a slightly
    # tighter theoretical lower bound but also costs a bit more.
    # ------------------------------------------------------------------
    n = 5000
    print(f"\n=== Step‑function optimisation, length n={n} ===")

    # ------------------------------------------------------------------
    # 0.  Generate a very diverse seed pool
    # ------------------------------------------------------------------
    print("\n--- Generating seeds ---")
    seeds = []

    # deterministic analytic windows
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hann(n))
    seeds.append(_hamming(n))
    seeds.append(_blackman(n))
    seeds.append(_gaussian(n, 0.15))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_gaussian(n, 0.35))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_beta_shape(n, 1.0, 3.0))
    seeds.append(_beta_shape(n, 3.0, 1.0))
    seeds.append(_tukey_window(n, 0.2))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_tukey_window(n, 0.8))
    seeds.append(_sine_power_window(n, 0.5))
    seeds.append(_sine_power_window(n, 1.0))
    seeds.append(_sine_power_window(n, 2.0))
    seeds.append(_kaiser_window(n, 2.0))
    seeds.append(_kaiser_window(n, 5.0))
    seeds.append(_kaiser_window(n, 8.0))
    seeds.append(_dpss_window(n, 2.5))
    seeds.append(_dpss_window(n, 3.0))
    seeds.append(_dpss_window(n, 4.0))

    # random Dirichlet draws with several concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse supports (exactly m equal non‑zero entries)
    for m in (150, 250, 350):
        for _ in range(4):
            idx = rng.choice(n, size=m, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[idx] = 1.0 / m
            seeds.append(a)

    # mixtures of two random members of the pool
    for _ in range(100):
        a, b = rng.choice(seeds, size=2, replace=False)
        alpha_mix = rng.random()
        mixed = _mix_two(a, b, alpha_mix)
        seeds.append(mixed)

    # previous best (if available)
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) == n:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # normalise all seeds (protect against zero‑sum vectors)
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        clean_seeds.append(v / s if s > 0.0 else _uniform(n))

    print(f"Created {len(clean_seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Evaluate all seeds (exact surrogate) and keep the best few
    # ------------------------------------------------------------------
    print("\n--- Evaluating seeds ---")
    scored = [(s, _objective_surrogate(s)) for s in clean_seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 7
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 1. Adam‑soft‑max optimisation (multiple starts)
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Adam optimisation (multiple starts) ---")
        for idx, seed_vec in enumerate(top_seeds[:3]):
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            # give each start roughly the same remaining budget
            max_iters = int(min(350_000, remaining * 1500))
            cand, val = _adam_softmax_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=max_iters,
                alpha_start=5.0,
                alpha_end=20000.0,
                lr=0.12,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Sparse exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 1300))
        best_seq, best_val = _expgrad_optimize(

            deadline,
            max_iters=max_iters,
            eta0=0.25,
            sparsity_thr=1e-12,

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing fine‑tuning ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing_optimize(
            best_seq,
            n,
            deadline,
            max_iters=max_iters,
        )
        print(f"[SA] surrogate after annealing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Tiny hill‑climbing moves (greedy local search)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final hill‑climbing refinement ---")
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 2000))
        best_seq, best_val = _hill_climb_optimize(
            best_seq,
            n,
            deadline,
            max_iters=max_iters,
        )
        print(f"[HillClimb] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
            f"official evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
