"""
Optimisation of a non‑negative step‑function under the official
``evaluate_sequence`` objective.

The method consists of:

1. **Rich seed library** – analytic windows (uniform, triangular,
   Gaussian, Tukey, Kaiser, power‑law), Dirichlet draws, random sparse
   supports and a few low‑frequency “wiggly’’ curves.  All seeds are
   ℓ₁‑normalised.

2. **Soft‑max Adam polishing** – a smooth surrogate
   ``L_α(p)= (1/α)·log Σ_k exp(α·(p*p)[k])`` is minimised.
   The temperature α is increased up to 20000 and the optimiser is run
   for many iterations on the best seeds.

3. **Multiplicative (exponential‑gradient) descent** – a true
   sub‑gradient step on the active convolution index
   ``k* = argmax_k (p*p)[k]`` with an AdaGrad‑style learning‑rate.

4. **Aggressive simulated‑annealing** – random reshuffles of
   moderately sized subsets and large mass‑transfer moves; the cooling
   schedule is very slow to allow thorough exploration.

All stages respect the global time budget (≈1000 s).  The best sequence
found (as a plain Python list) is stored in the global name
``height_sequence_1`` and returned.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(N log N))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(p,p)) for a probability vector p (∑p = 1).
    Used in all optimisation stages – cheap (FFT) and exact up to
    numerical noise.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        # pure cosine (Hann)
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Numpy’s Kaiser window (always non‑negative)."""
    a = np.kaiser(n, beta)
    return a / a.sum()


def _power_law(n: int, exp: float) -> np.ndarray:
    """Weights ∝ (i+1)^exp, normalized.  Handles negative exponents."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** exp
    if np.any(np.isinf(a)) or np.any(np.isnan(a)):
        # fallback to uniform if overflow occurs
        return _uniform(n)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Exactly `m` equal non‑zero entries."""
    if m <= 0:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves (non‑negative)."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()

def _adam_softmax_optimize(start: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 150_000,
                           alpha_start: float = 20.0,
                           alpha_end: float = 20_000.0,
                           lr: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
    L_α(p) = (1/α)·log Σₖ exp(α·(p·p)[k]).
    α grows linearly from ``alpha_start`` to ``alpha_end``.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Full convolution via FFT (real)
        fx = rfft(x, n=N)
        conv_full = irfft(fx * fx, n=N)[: 2 * n - 1]

        # Soft‑max of the convolution
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # Gradient of the smooth surrogate = 2·corr(p, soft)
        fs = rfft(soft, n=N)
        corr = irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # Projection onto the simplex (ensures positivity & sum=1)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 30_000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _expgrad_optimize(start: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 600_000,
                      eta0: float = 0.5,
                      sparsity_thr: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) on the true max‑conv objective.
    Uses a diminishing step size η_t = η₀/√t (AdaGrad‑like).
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift (ordered pairs)
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)      # avoid overflow in exp()
        x = x * np.exp(delta)

        if sparsity_thr > 0.0:
            mask = x < sparsity_thr
            if np.any(mask):
                x[mask] = 0.0

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(start: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-15) -> tuple[np.ndarray, float]:
    """
    Aggressive simulated‑annealing with two move families:
        • reshuffle a random subset (size 5‑30)
        • transfer a variable amount between two random positions
    The schedule cools slowly (×0.99996 per iteration) to allow
    extensive exploration.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99996, min_temp)

        move_type = rng.random()
        if move_type < 0.55:
            # reshuffle a random subset (size 5‑30)
            k_sub = rng.integers(5, 31)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # transfer a possibly large fraction between two positions
            i, j = rng.choice(n, size=2, replace=False)
            # move up to 80 % of the source mass
            eps = rng.random() * min(cur[i], 0.8)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generation (large diversified pool)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.15))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_gaussian(n, 0.35))

    # Tukey alphas
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, float(alpha)))

    # Kaiser betas
    for beta in np.arange(0.0, 13.0, 2.0):
        seeds.append(_kaiser_window(n, float(beta)))

    # Power‑law shapes (both decreasing & increasing)
    for exp in (-3.0, -2.0, -1.0, -0.5, 0.5, 1.0, 2.0, 3.0):
        seeds.append(_power_law(n, float(exp)))

    # Beta‑shape sweeps
    for a in (0.5, 1.0, 2.0, 3.0, 5.0):
        for b in (0.5, 1.0, 2.0, 3.0, 5.0):
            seeds.append(_beta_shape(n, float(a), float(b)))

    # Random sparse supports (various cardinalities)
    for m in (60, 120, 200, 300, 500):
        for _ in range(3):
            seeds.append(_random_sparse_support(n, m, rng))

    # Random Dirichlet draws (different concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, float(conc), rng))

    # Low‑frequency random curves (to add some structure)
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform and a sparse support (different mixing weights)
    for mix_alpha in (0.2, 0.4, 0.6):
        for m in (150, 250, 350):
            sparse = _random_sparse_support(n, m, rng)
            mixed = _mix_two(_uniform(n), sparse, float(mix_alpha))
            seeds.append(mixed)

    # previous best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # final cleanup – guaranteed proper normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search (≤ budget_s seconds) for a step function that minimises the
    official ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # a small safety margin

    # ------------------------------------------------------------------
    # Length of the step function – 4000 gives a good compromise.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n--- Length n={n} ---")

    # ------------------------------------------------------------------
    # Stage 0 – diverse seed pool
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Created {len(seeds)} seeds.")

    # cheap surrogate evaluation (FFT based)
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = min(30, len(scored))
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – soft‑max Adam polishing (multiple starts)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple starts) ---")
        # allocate roughly half of the remaining budget to Adam
        adam_deadline = time.time() + (deadline - time.time()) * 0.5
        for idx, seed_vec in enumerate(best_seeds[:8]):   # up to 8 starts
            if time.time() >= adam_deadline:
                break
            # allow a generous number of iterations but respect remaining time
            it_budget = int(min(200_000, (adam_deadline - time.time()) * 130))
            cand, val = _adam_softmax_optimize(
                seed_vec, n, adam_deadline,
                max_iters=it_budget,
                alpha_start=20.0,
                alpha_end=20_000.0,
                lr=0.12,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – sparse exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sparse Exponential‑Gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline,
            max_iters=max_iters,
            eta0=0.5,
            sparsity_thr=1e-9,
        )
        print(f"[ExpGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(1_200_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional very short local hill‑climbing (final few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final hill‑climbing tweak ---")
        cur = best_seq.copy()
        cur_val = _objective_surrogate(cur)
        step = 0
        while time.time() < deadline:
            # pick two indices, move a random fraction (up to 30 %) from i → j
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.3)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps
            new_val = _objective_surrogate(cand)
            if new_val < cur_val:
                cur = cand
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cand.copy()
            step += 1
            if step % 200_000 == 0:
                print(f"[Hill] steps={step} best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
