"""
Improved step‑function search.

The algorithm proceeds as follows (all work respects the time budget
passed in ``budget_s``).

1. **Fast numeric utilities** – FFT based full convolution and a cheap
   objective evaluator ``_objective`` that returns the exact value of
   ``2·n·max(conv(a,a))`` for an ℓ₁‑normalised vector ``a``.

2. **Parametric families** – a handful of one‑parameter families
   (power‑law, exponential, Gaussian, sine‑window) are optimised with
   ``scipy.optimize.minimize_scalar`` (bounded golden‑section search).
   Each family yields a candidate vector; the best of them is kept.

3. **Mixture with the uniform vector** – the current best vector is
   linearly mixed with the uniform distribution; the mixing weight is
   optimised (again by ``minimize_scalar``).  This often reduces the
   peak of the convolution without destroying the already good shape.

4. **Exponentiated‑gradient (mirror‑descent) refinement** – starting
   from the mixture, a long EG run minimises the *exact* max‑convolution
   objective.  The step‑size follows ``ηₜ = η₀ / √t`` with a slightly
   larger ``η₀=1.5`` than the previous baseline.  The loop runs as many
   iterations as time permits, tracking the best vector seen.

5. **Final polishing** – a short L‑BFGS run on the smooth log‑sum‑exp
   surrogate with a very high β‑schedule (up to 2·10⁴) gives a final
   local refinement.

The best sequence found is stored in the global name ``height_sequence_1``
and returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax; yields a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum(a)=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Returns ∂/∂a of the *active* shift (the one attaining the maximum).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    # g_i = 2 * a_{idx-i} for indices that lie inside the vector
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Families of analytic shapes (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = lam ** i
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth max‑convolution surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Log‑sum‑exp surrogate of max(conv); returns value and gradient w.r.t. logits."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)               # full autocorrelation
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp               # soft weights of each lag
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]  # gradient w.r.t. a

    # chain rule: d obj / d logits = Jacobian(softmax)·grad_a
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule,
                           max_iter_per_beta: int = 500) -> np.ndarray:
    """
    Optimise the smooth surrogate using L‑BFGS‑B on the log‑softmax
    (logits) variables.
    """
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val}")
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns the best vector seen and its exact objective value.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # sub‑gradient of the active shift
        g = _subgradient_maxconv(a)

        # step‑size ηₜ = η₀ / √t
        eta = eta0 / np.sqrt(it)

        # EG update in log‑space
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        # occasional bookkeeping
        if it % 100_000 == 0:
            cur_val = _objective(a, n)
            print(f"[ExpGrad] it={it:,} obj={cur_val:.6f}")
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

    # final check
    cur_val = _objective(a, n)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------

    n = 4000
    print(f"\n=== Starting enhanced search (n={n}) ===")

    # Helper to evaluate a candidate and update the global best
    # ------------------------------------------------------------------
    best_seq = _uniform(n)   # starter

    print(f"[Init] uniform baseline = {best_val:.6f}")

    def try_candidate(vec: np.ndarray, label: str = ""):
        nonlocal best_seq, best_val
        val = _objective(vec, n)
        if val < best_val - 1e-12:

            best_val = val

    # ------------------------------------------------------------------
    # 1️⃣  Parametric families – coarse optimisation
    # ------------------------------------------------------------------
    # Power‑law exponent
    def obj_pow(p):
        return _objective(_power_law(n, p), n)

    res_pow = _opt.minimize_scalar(obj_pow, bounds=(0.0, 5.0), method='bounded',
                                   options={'xatol': 1e-3, 'maxiter': 40})
    if res_pow.success:
        try_candidate(_power_law(n, res_pow.x), f"power(p={res_pow.x:.4f})")

    # Exponential decay (λ close to 1)
    def obj_exp(lam):
        # keep λ in (0.90, 0.9999)
        return _objective(_exponential(n, lam), n)

    res_exp = _opt.minimize_scalar(obj_exp, bounds=(0.90, 0.9999), method='bounded',
                                   options={'xatol': 5e-5, 'maxiter': 40})
    if res_exp.success:
        try_candidate(_exponential(n, res_exp.x), f"exp(λ={res_exp.x:.6f})")

    def obj_gauss(s):
        return _objective(_gaussian(n, s), n)

    res_gauss = _opt.minimize_scalar(obj_gauss, bounds=(0.05, 0.45), method='bounded',
                                     options={'xatol': 1e-3, 'maxiter': 30})
    if res_gauss.success:
        try_candidate(_gaussian(n, res_gauss.x), f"gauss(σ={res_gauss.x:.3f})")

    # Sine window (γ parameter)
    def obj_sine(g):
        return _objective(_sine_window(n, g), n)

    res_sine = _opt.minimize_scalar(obj_sine, bounds=(0.3, 3.0), method='bounded',
                                    options={'xatol': 2e-3, 'maxiter': 30})
    if res_sine.success:
        try_candidate(_sine_window(n, res_sine.x), f"sine(γ={res_sine.x:.3f})")

    try_candidate(_triangular(n), "triangular")

        vec /= vec.sum()
        try_candidate(vec, "expo_draw")

    # ------------------------------------------------------------------
    # 3️⃣  Linear mixtures of the best candidates found so far
    # ------------------------------------------------------------------

    for _ in range(150):

        alpha = rng.random()
        mixed = (1.0 - alpha) * a + alpha * b
        mixed /= mixed.sum()
        try_candidate(mixed, "mix")

    # ------------------------------------------------------------------
    uniform = _uniform(n)

        # both operands already sum to 1 → mix also sums to 1
        return _objective(mix, n)

    res_alpha = _opt.minimize_scalar(obj_alpha, bounds=(0.0, 1.0), method='bounded',
                                     options={'xatol': 5e-4, 'maxiter': 30})
    if res_alpha.success:
        mix = (1.0 - res_alpha.x) * best_seq + res_alpha.x * uniform
        try_candidate(mix, f"mix_uniform(α={res_alpha.x:.3f})")

    # ------------------------------------------------------------------
    # 5️⃣  Exponentiated‑gradient refinement on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Rough estimate: each iteration ≈ 0.001 s → allocate a generous budget
        max_iters = int(min(4_000_000, remaining * 800))
        print(f"\n--- Exponentiated‑gradient refinement (max_iters={max_iters}) ---")
        seq_exp, val_exp = _exp_gradient_optimize(best_seq, n, deadline,
                                                  max_iters=max_iters,
                                                  eta0=1.5)
        if val_exp < best_val:
            best_seq = seq_exp.copy()
            best_val = val_exp
            print(f"[ExpGrad] improved exact objective to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final polishing with high‑β L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(300, int(min(1500, remaining * 70)))  # generous but bounded
        beta_sched = [2000.0, 5000.0, 10_000.0, 20_000.0]
        print("\n--- Final L‑BFGS polishing (high β) ---")
        seq_final = _lbfgs_smooth_optimize(best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=beta_sched,
                                           max_iter_per_beta=per_beta_iters)
        val_final = _objective(seq_final, n)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < best_val:
            best_seq = seq_final.copy()
            best_val = val_final
            print(f"[Update] New best after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible warm‑start in a future run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
