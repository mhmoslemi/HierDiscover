"""
Search for a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where conv is the full linear convolution.  The algorithm combines

* a fast L‑BFGS optimisation of a log‑sum‑exp surrogate of the
  max‑convolution (aggressive β‑schedule up to 1e5),
* exponentiated‑gradient (mirror descent) on the exact max‑convolution,
* a cheap Frank‑Wolfe‑style improvement via a linear programme that
  maximises the ℓ₁‑mass of a second vector under the current max‑convolution
  bound (using only the most active constraints), and
* a final sub‑gradient polishing step.

All stages respect a global wall‑clock deadline (default 1000 s).  The
best sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain ``list`` of floats.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# SciPy is optional but required for the optimisation steps.
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = None
    _sp = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return 2.0 * n * max_c / (sum_a ** 2)


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate for minimisation when the vector is ℓ₁‑normalised."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        # Very aggressive schedule – extended to β = 1e5
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0,
                         5_000.0, 10_000.0, 20_000.0,
                         50_000.0, 100_000.0]

    # Initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # log‑sum‑exp smoothing of the max
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t. a
            w = exp_beta / sum_exp                     # softmax weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # stop early if we run out of time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # Sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑program improvement (Frank‑Wolfe style)
# ----------------------------------------------------------------------
def _lp_improve_step(a: np.ndarray,
                     n: int,
                     M: float,
                     active_idx: np.ndarray,
                     deadline: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀb   subject to   a * b ≤ M   (active rows only)
    and b ≥ 0.
    Returns the ℓ₁‑normalised vector b or None on failure.
    """
    if _opt is None or _sp is None:
        return None

    # Build sparse constraint matrix for the selected rows.
    rows = []
    cols = []
    data = []
    for row_id, k in enumerate(active_idx):
        col_start = max(0, k - n + 1)
        col_end = min(k, n - 1)
        col_range = np.arange(col_start, col_end + 1, dtype=np.int32)
        offsets = k - col_range                     # indices into a
        coeffs = a[offsets]                         # f_i for this row
        rows.append(np.full_like(col_range, row_id, dtype=np.int32))
        cols.append(col_range)
        data.append(coeffs)

    row_idx = np.concatenate(rows)
    col_idx = np.concatenate(cols)
    data_arr = np.concatenate(data)

    A = _sp.csr_matrix((data_arr, (row_idx, col_idx)), shape=(len(active_idx), n))
    b_ub = np.full(len(active_idx), M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # maximise sum(b)  → minimise -sum(b)

    # Time limit for the LP – keep a small safety margin.
    time_left = deadline - time.time()
    options = {'presolve': True}
    if time_left > 0:
        options['time_limit'] = max(0.0, time_left)

    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b_ub,
                           bounds=(0, None),
                           method='highs',
                           options=options)
    except Exception as e:  # pragma: no cover
        print(f"[LP] solve exception: {e}")
        return None

    if not res.success:
        # LP failed (infeasible/unbounded)
        return None

    b = res.x
    if b.sum() == 0.0:
        return None
    b = b / b.sum()                                 # ℓ₁‑normalise
    return b

            print(f"[LP] iter {lp_iters+1:2d} improved surrogate = {best_val:.6f}")
        else:
            print(f"[LP] iter {lp_iters+1:2d} no improvement")
        lp_iters += 1

    # ------------------------------------------------------------------
    # 4. Final sub‑gradient polishing (if any time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub_iters = int(min(600_000, remaining * 1e5))
        if max_sub_iters > 0:
            seq_sub, val_sub = _subgradient_opt(best_seq, n, deadline,
                                                max_iters=max_sub_iters)
            if val_sub < best_val:
                best_seq, best_val = seq_sub, val_sub
                print(f"[SubGrad] final improvement, surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
