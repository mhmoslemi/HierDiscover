"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.
The search combines three ideas:

1. **Fast parametric seeds** – a handful of analytic families (uniform,
   beta, power‑law, Gaussian) and the previous best solution (if any).
   All seeds are ℓ¹‑normalised.

2. **Continuation L‑BFGS** – optimisation of a smooth log‑sum‑exp
   surrogate for the max‑convolution (aggressive β‑schedule up to 2·10⁴).

3. **Alternating linear‑programming descent** – given a current shape *f*,
   solve a linear program that maximises the ℓ¹‑norm of a partner *g*
   under the linear constraints ``conv(f,g) ≤ M`` where ``M = max(conv(f,f))``.
   After normalising *g* to sum=1 we replace *f* by *g* (or a convex
   combination) if the true objective improves.  Repeating this step
   quickly drives the max‑convolution towards its lower theoretical bound.

The routine respects a 1000 s wall‑clock budget, logs progress and finally
stores the best sequence in ``height_sequence_1`` (global variable).  The
implementation works with a moderate problem size ``n = 2000`` to keep the
LP tractable while still allowing a very good objective value.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s > 0.0 else np.full_like(v, 1.0 / len(v))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (scale‑invariant)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for a simplex‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ¹‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Optimises the smoothed max‑convolution objective using L‑BFGS‑B."""
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 20000.0]

    v = np.log(start_vec + 1e-30)          # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = minimize(fun=fun,
                           x0=v,
                           method='L-BFGS-B',
                           jac=True,
                           callback=callback,
                           options={'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:               # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Exponentiated‑gradient (mirror descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> np.ndarray:
    """Mirror‑descent on the exact objective (∑=1 simplex)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx_i
        g[idx_i] = 2.0 * a[rev]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq

# Linear‑programming helper (alternating descent)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build a sparse matrix A where (A @ g)[k] = conv(f, g)[k].
    Shape of A: (2*len(f)-1, len(f)).
    """
    n = f.shape[0]
    # rows = i + j  (i is column index, j is index inside f)
    i = np.arange(n, dtype=np.int64)               # shape (n,)
    j = np.arange(n, dtype=np.int64)               # shape (n,)
    rows = (i[:, None] + j[None, :]).ravel()       # length n*n
    cols = np.repeat(i, n)
    data = np.tile(f, n)
    A = sparse.csr_matrix((data, (rows, cols)),
                          shape=(2 * n - 1, n))
    return A


def _lp_max_sum_given_M(f: np.ndarray,
                        M: float,
                        deadline: float) -> np.ndarray | None:
    """
    Solve   max  sum(g)
    s.t.    conv(f, g) ≤ M   (element‑wise)
            g ≥ 0
    Returns the optimal *g* (already normalised to ℓ¹=1) or None on failure.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)
    c = -np.ones(n)                     # maximize sum(g)  →  minimize -sum(g)
    b = np.full(A.shape[0], M, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # Solver options – we ask for a modest tolerance to keep it fast
    opts = {'presolve': True, 'time_limit': max(1, int(deadline - time.time()))}
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options=opts)
    except Exception:                  # pragma: no cover
        return None

    if not res.success:
        return None

    g0 = res.x
    total = g0.sum()
    if total <= 1e-12:
        return None
    return g0 / total                     # normalise to ℓ¹ = 1


def _alternating_lp_descent(init_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 10) -> np.ndarray:
    """
    Repeatedly solve the LP described above and replace the current
    sequence with the LP solution whenever it improves the exact objective.
    """
    f = init_seq.copy()
    best_val = _objective_surrogate(f)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        M = np.max(_conv_fft(f, f))
        g = _lp_max_sum_given_M(f, M, deadline)
        if g is None:
            break

        val_g = _objective_surrogate(g)

        if val_g + 1e-9 < best_val:
            f = g
            best_val = val_g
            print(f"[LP‑Descent] iteration {it+1}: improved to {best_val:.6f}")
            continue

        # try a convex blend (t=0.5) if pure replacement does not improve
        t = 0.5
        blend = (1 - t) * f + t * g
        blend /= blend.sum()
        val_blend = _objective_surrogate(blend)

        if val_blend + 1e-9 < best_val:
            f = blend
            best_val = val_blend
            print(f"[LP‑Descent] iteration {it+1}: blended improvement to {best_val:.6f}")
            continue

        # no improvement – stop
        break

    return f

    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(600_000, remaining * 0.07 * 1e5))
        print("\n--- Exponentiated‑gradient polishing ---")
        best_seq = _exp_gradient_opt(best_seq, n, deadline, max_iters=max_exp)
        best_val = _objective_surrogate(best_seq)
        print(f"[ExpGrad] final surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Evaluate with the official evaluator and store globally.
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
