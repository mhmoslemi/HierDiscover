"""
Propose a non‑negative step function that minimises
``evaluate_sequence`` within a 1000 s budget.

Algorithm
---------

1. **Broad analytical grid** – generate a large, diverse
   collection of ℓ₁‑normalised candidates from many analytic families:
   uniform, linear/quadratic decays, beta, power, Gaussian,
   sine‑window, Tukey, exponential, random Dirichlet draws, and
   low‑frequency cosine sums.

2. **Fast surrogate ranking** – evaluate every candidate with the
   exact surrogate 2·n·max(conv(a,a)) (the same as the official
   objective when Σa=1) using an FFT‑based convolution.  Keep the
   best ≈20 vectors.

3. **Exponentiated‑gradient (EG) polishing** – for each of the top
   vectors run a cheap mirror‑descent optimiser on the true max‑
   convolution (sub‑gradient at the active shift).  The EG step
   respects the simplex and aggressively reduces the peak.

4. **L‑BFGS‑B on a smooth log‑sum‑exp max surrogate** – start from the
   best EG result and minimise a smoothed version of the max‑convolution
   ``Fₚ(a) = (1/p)·log Σₖ exp(p·convₖ(a))`` using the logits
   parameterisation ``a = softmax(y)``.  The smoothing parameter ``p``
   is increased geometrically (5 → 15 → 45 → 135 → 400 → 1200) and a
   short random‑restart is performed for each ``p``.

5. **Final EG polish** – a short EG run from the best L‑BFGS vector.

All work is bounded by the deadline; the best sequence found is
stored in the global ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = y.max()
    e = np.exp(y - m)
    return e / e.sum()


def _evaluate_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (2·n·max(conv)) for a sequence whose sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decrease(n: int) -> np.ndarray:
    """a_i ∝ (n - i)."""
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _quadratic_decrease(n: int) -> np.ndarray:
    """a_i ∝ (n - i)²."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** 2
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 200_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent (KL‑projection) on the exact max‑convolution."""
    x = start.copy()
    best_x = x.copy()
    best_val = _evaluate_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for shift k (as in the reference implementation)
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # stepsize schedule η_t = η0 / sqrt(t)
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)     # avoid overflow
        x = x * np.exp(delta)

        # projection onto the simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _evaluate_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """
    Returns (value, gradient) of the smooth surrogate
    Fₚ(a) = (1/p)·log Σₖ exp(p·convₖ(a))  with a = softmax(y).
    """
    a = _softmax(y)                     # a lies on the simplex
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = conv.max()

    shifted = conv - max_c               # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)  # safety
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                    # fallback
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)     # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    # chain rule for soft‑max
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a: np.ndarray, n: int, deadline: float,
                     p: float = 20.0, max_iter: int = 300) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the smoothed max‑convolution.
    Returns the best vector found and its *exact* surrogate value.
    """
    y0 = np.log(np.maximum(start_a, 1e-15))
    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})
    a_opt = _softmax(res.x)
    max_conv = _conv_fft(a_opt, a_opt).max()
    val = 2.0 * n * max_conv
    return a_opt, val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns the best sequence as a list of Python floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting optimisation, n = {n} ===")

    # --------------------------------------------------------------
    # 1. Broad analytical grid
    # --------------------------------------------------------------
    candidates = []

    # deterministic families
    candidates.append(_uniform(n))
    candidates.append(_linear_decrease(n))
    candidates.append(_quadratic_decrease(n))
    candidates.append(_triangular(n))

    for gamma in np.linspace(0.5, 5.0, 10):
        candidates.append(_sine_window(n, gamma))

    for sigma in np.linspace(0.12, 0.45, 8):
        candidates.append(_gaussian(n, sigma))

    for p in np.linspace(0.3, 3.5, 10):
        candidates.append(_power(n, p))

    for a in np.arange(0.2, 5.01, 0.4):
        for b in np.arange(0.2, 5.01, 0.4):
            candidates.append(_beta_shape(n, a, b))

    for alpha in np.linspace(0.1, 0.9, 9):
        candidates.append(_tukey_window(n, alpha))

    for rate in np.linspace(0.99, 0.9999, 5):
        candidates.append(_exponential_decay(n, rate))

    # random draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            candidates.append(_random_dirichlet(n, conc, rng))

    for nf in (1, 2, 3, 4):
        for _ in range(2):
            candidates.append(_lowfreq_curve(n, nf, rng))

    # mixtures with the uniform baseline
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            candidates.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # keep previous solution (if any) as a seed
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            candidates.append(arr / arr.sum())

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation & keep the best few
    # --------------------------------------------------------------
    scored = []
    best_val = float('inf')
    best_seq = None

    for idx, vec in enumerate(candidates):
        if time.time() > deadline:
            break
        val = _evaluate_surrogate(vec)
        scored.append((val, vec))
        if val < best_val:
            best_val, best_seq = val, vec.copy()

    scored.sort(key=lambda x: x[0])
    top_k = min(20, len(scored))
    top_vectors = [v for _, v in scored[:top_k]]

    print(f"Initial grid best surrogate = {best_val:.6f} (kept top {top_k})")

    # --------------------------------------------------------------
    # 3. EG polishing on the promising seeds
    # --------------------------------------------------------------
    while time.time() < deadline:
        # pick a random seed from the elite set
        start_vec = top_vectors[rng.integers(len(top_vectors))]
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 2500))
        if max_iters < 5_000:
            break
        cand, val = _expgrad_optimize(start_vec, n, deadline,
                                      max_iters=max_iters, eta0=0.30)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[EG] new best surrogate = {best_val:.6f}")
            # maintain elite list
            top_vectors.append(cand)
            top_vectors = sorted(top_vectors,
                                 key=lambda v: _evaluate_surrogate(v))[:top_k]

    # --------------------------------------------------------------
    # 4. L‑BFGS‑B on the smooth max‑surrogate (increasing p)
    # --------------------------------------------------------------
    if time.time() < deadline:
        p_schedule = [5, 15, 45, 135, 400, 1200]
        cur_vec = best_seq.copy()
        for p in p_schedule:
            if time.time() >= deadline:
                break
            cand, val = _lbfgsb_optimize(cur_vec, n, deadline,
                                         p=p, max_iter=350)
            print(f"[L‑BFGS] p={p:5.0f} → surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"   ** improved global best to {best_val:.6f}")
            # one random restart per p
            if time.time() < deadline:
                y_noise = np.log(np.maximum(cur_vec, 1e-15)) + rng.normal(scale=0.15, size=n)
                pert = _softmax(y_noise)
                cand2, val2 = _lbfgsb_optimize(pert, n, deadline,
                                               p=p, max_iter=350)
                if val2 < best_val:
                    best_val, best_seq = val2, cand2.copy()
                    print(f"[Restart] p={p} → improved to {best_val:.6f}")
            cur_vec = best_seq  # feed‑forward

    # --------------------------------------------------------------
    # 5. Final tiny EG polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(80_000, remaining * 2000))
        if max_iters > 0:
            cand, val = _expgrad_optimize(best_seq, n, deadline,
                                          max_iters=max_iters, eta0=0.25)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Final EG] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Normalise and evaluate with the official function
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s == 0.0:
        best_seq = _uniform(n)
    else:
        best_seq = best_seq / s

    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
