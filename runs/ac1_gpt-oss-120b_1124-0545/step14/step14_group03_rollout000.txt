"""
Propose a near‑optimal non‑negative step‑function.

The search consists of four stages:

1. **Rich seed pool** – deterministic families (uniform, triangular,
   sine, beta, etc.) and random Dirichlet / exponential draws.  All
   vectors are ℓ₁‑normalised, i.e. they represent probability
   distributions (∑a_i = 1).

2. **Fast screening** – evaluate the exact objective
   `2·n·max(conv(a,a))` (sum‑normalised, so the objective is
   scale‑invariant) for every seed using an FFT based convolution.
   Keep the `top_k` most promising seeds.

3. **Continuation L‑BFGS** on a smooth log‑sum‑exp surrogate of the
   max‑convolution.  The β‑schedule is extended up to 5·10⁴ and a
   generous per‑β iteration budget is allocated.  This yields a high‑
   quality local optimum for each seed.

4. **Exponentiated‑gradient (mirror‑descent) refinement** on the
   *exact* max‑convolution objective.  The update `log a ← log a − η·g`
   uses the true sub‑gradient (all indices attaining the current
   maximum are taken into account).  The step size follows
   ηₜ = η₀ / √t with η₀ = 1.0 and runs for as many iterations as the
   remaining time permits.

The best vector found is stored in the global name ``height_sequence_1``
and returned as a plain list of floats.  The routine respects a total
budget (≈ 1000 s) and aborts gracefully before the deadline.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1} (ℓ₁‑normalised)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    t = np.arange(n, dtype=np.float64) / (n - 1)   # in [0,1]
    # add a tiny epsilon to avoid 0⁰ problems when alpha<1 or beta<1
    eps = 1e-12
    a = (np.maximum(t, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - t, eps) ** (beta - 1.0))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α followed by ℓ₁‑renormalisation."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum = 1)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def _subgradient_and_maxconv(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a and the current max value.
    If several lags attain the maximum (within a tiny tolerance) the
    gradient is averaged over all active lags.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = np.max(conv)
    # tolerance for “active set” (absolute)
    eps = 1e-14 * max_val
    active = np.where(conv >= max_val - eps)[0]

    g = np.zeros_like(a)
    for idx in active:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] += 2.0 * a[rev_idx]

    if len(active) > 0:
        g /= float(len(active))

    return g, max_val


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float) -> Tuple[float, np.ndarray]:
    """Function value and gradient w.r.t. logits `v`."""
    a = _softmax(v)                         # probability simplex
    conv = _conv_fft(a, a)                  # full autocorrelation
    shift = np.max(conv)                    # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # surrogate of 2·n·max(conv) ≈ 2·n·(shift + log_sum_exp/β)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # soft weighting of each lag
    w = exp_term / sum_exp
    # gradient wrt a : 2·conv(rev_a, w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule: d obj / d logits = Jacobian(softmax)·grad_a
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 800) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution surrogate using L‑BFGS‑B.
    The optimisation variable is the logits of a probability vector.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(local_v):
            return _smooth_objective(local_v, n, beta_val)

        def callback(local_v):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            # deadline reached inside L‑BFGS
            break
        except Exception as e:   # pragma: no cover
            # unexpected error – abort this run
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 1_000_000,
                           eta0: float = 1.0) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent (EG) on the *exact* max‑convolution objective.
    Returns the best vector seen and its exact objective value.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters:
        it += 1
        if time.time() > deadline:
            break

        # Sub‑gradient of the active lag(s)
        g, cur_max = _subgradient_and_maxconv(a)

        # Step‑size schedule
        eta = eta0 / math.sqrt(it)

        # EG update (in log‑space)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        # Opportunistic logging & best‑track
        if it % 200_000 == 0:
            cur_val = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur_val:.6f}")
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

    # Final check
    cur_val = _objective_exact(a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> List[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: List[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))

    # Beta family – we explore a fine grid down to 0.2 (U‑shaped)
    betas_alphas = [0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0]
    for a in betas_alphas:
        for b in betas_alphas:
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Mixtures of previously generated seeds (adds diversity)
    base_idx = np.arange(len(seeds))
    for _ in range(150):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        mix = _mix_shapes(seeds[i], seeds[j], alpha)
        seeds.append(mix)

    # Previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean up tiny negatives, normalise
    cleaned: List[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)

    return cleaned

                print(f"[ExpGrad] exact objective = {val_exp:.6f}")
                if val_exp < best_val:
                    best_seq = seq_exp.copy()
                    best_val = val_exp
                    print(f"[Update] New global best after EG = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final polishing – short L‑BFGS with very high β
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(400, int(min(2000,
                                          remaining * 0.06 * 1e5)))
        print("\n--- Final polishing L‑BFGS (β = 20000, 50000) ---")
        seq_final = _lbfgs_smooth_optimize(best_seq,
                                            n,
                                            deadline,
                                            beta_schedule=[20000.0, 50000.0],
                                            max_iter_per_beta=per_beta_iters)
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] surrogate = {val_final:.6f}")
        if val_final < best_val:
            best_seq = seq_final.copy()
            best_val = val_final
            print(f"[Update] New global best after final polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator (purely for logging)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible warm‑starts in future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
