def compute_max_conv(a):
    n = len(a)
    L = 2*n-1
    N = 1 << ((L-1).bit_length())
    # compute FFT of a zero-padded to N
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa*fa, n=N)[:L]
    max_conv = conv.max()
    return max_conv
def objective(a):
    s = a.sum()
    maxc = compute_max_conv(a)
    return 2.0 * len(a) * maxc / (s * s)
```

"""
Propose a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of four stages:

1. **Diverse seed pool** – analytic windows, random Dirichlet / exponential
   draws and many mixtures.  All vectors are ℓ₁‑normalised (sum = 1) so that
   the objective is scale‑invariant.

2. **Continuation L‑BFGS** – smooth log‑sum‑exp surrogate of the max‑convolution.
   A schedule of increasing β values drives the surrogate towards the true
   max.  The optimisation runs on the best few seeds.

3. **Permutation annealing** – the multiset of values is kept fixed while a
   simulated‑annealing on the index order is performed.  Swapping two positions
   can considerably lower the peak of the convolution without changing the
   ℓ₁‑norm.

4. **Final polishing** – a short high‑β L‑BFGS run refines the best permutation
   found.

The routine respects the 1000 s budget, prints intermediate bounds
and stores the best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using an FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, re‑normalised to ℓ₁‑norm = 1."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) for the smooth max‑convolution surrogate:

        f(a) = 2·n·(shift + log Σ exp(β·(conv‑shift))/β)

    where a = softmax(v).
    """
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)               # full autocorrelation
    shift = np.max(conv)                 # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # softmax weights of each lag
    w = exp_term / sum_exp
    # gradient wrt a : 2·conv(rev_a, w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3·n−2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The optimisation variable is the log‑softmax (logits) of the probability vector.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    # initialise logits from the start vector
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(local_v):
            return _smooth_objective(local_v, n, beta_val)

        def callback(local_v):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline during beta={beta_val}")
            break
        except Exception as exc:  # pragma: no cover
            print(f"[L‑BFGS] exception {exc} (beta={beta_val})")
            break

    return _softmax(v)

# Seed pool generation

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian family
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_gaussian(n, 0.12))
    seeds.append(_gaussian(n, 0.08))

    # Power‑law
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family
    for a in np.linspace(0.5, 3.0, 7):
        for b in np.linspace(0.5, 3.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse binary patterns (useful for permutation stage)
    for density in (0.02, 0.05, 0.10):
        k = max(2, int(density * n))
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=k, replace=False)
        mask[idx] = 1.0
        seeds.append(mask / mask.sum())

    # Mixtures of already‑generated seeds
    num_mix = 200
    base_idx = np.arange(len(seeds))
    for _ in range(num_mix):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanitisation (non‑negative, ℓ₁‑normalised)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

    conv = _conv_fft(a, a)

                             int(min(4000,
                                     (deadline - time.time()) * 0.07 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={per_beta_iters}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                          n,
                                          deadline,
                                          beta_schedule=beta_sched,
                                          max_iter_per_beta=per_beta_iters)
        val_opt = _surrogate_obj(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] New global best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – permutation annealing (keep multiset of values)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation annealing (order optimisation) ---")
        cur_seq = best_seq.copy()
        cur_obj = best_val
        best_seq_perm = cur_seq.copy()
        best_obj_perm = cur_obj

        # simple linear cooling schedule
        total_perm_time = deadline - time.time()
        while time.time() < deadline:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            new_seq = cur_seq.copy()
            new_seq[i], new_seq[j] = cur_seq[j], cur_seq[i]

            new_obj = _surrogate_obj(new_seq)
            delta = new_obj - cur_obj

            # accept if improvement; otherwise accept with a small probability
            if delta < 0:
                cur_seq = new_seq
                cur_obj = new_obj
                if new_obj < best_obj_perm:
                    best_seq_perm = new_seq.copy()
                    best_obj_perm = new_obj
                    elapsed = time.time() - start_time
                    print(f"[Perm] new best surrogate {best_obj_perm:.6f} at {elapsed:.1f}s")
            else:
                # temperature decreases linearly to 0
                remaining = deadline - time.time()
                T = (remaining / total_perm_time) * best_obj_perm
                if rng.random() < math.exp(-delta / (T + 1e-12)):
                    cur_seq = new_seq
                    cur_obj = new_obj

        best_seq = best_seq_perm
        best_val = best_obj_perm
        print(f"[Perm] best surrogate after annealing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final polishing L‑BFGS (high β)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing L‑BFGS (high β) ---")
        per_beta = max(500,
                       int(min(3000,
                               (deadline - time.time()) * 0.07 * 1e5)))
        seq_final = _lbfgs_smooth_optimize(best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[5000.0, 10000.0, 20000.0],
                                           max_iter_per_beta=per_beta)
        val_final = _surrogate_obj(seq_final)
        print(f"[Final‑L‑BFGS] surrogate = {val_final:.6f}")
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Update] New best after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
