def _subgrad_max_conv(a):
    n = len(a)
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))
    g = np.zeros_like(a)

"""
search for a non‑negative step function (probability vector) that
minimises ``evaluate_sequence``.

Approach
--------
1. **Problem size** – Use a power‑of‑two length (8192).  This gives a fine
   discretisation while keeping FFT‑based convolutions cheap.
2. **Seed pool** – Analytic windows (uniform, triangular, sine, Gaussian,
   power, beta, Tukey, exponential) together with random Dirichlet
   draws and low‑frequency random curves.  The previous best solution
   (``height_sequence_1``) is added as a seed.
3. **Smooth max optimisation** – minimise a log‑sum‑exp surrogate of the
   max‑convolution via L‑BFGS‑B on the simplex (parameterised by logits and
   a soft‑max).  The smoothing parameter *p* is increased geometrically
   (5 → 15 → 45 → 135 → 405 → 1215 → 3645).  For each *p* we run a few
   random restarts.
4. **Exact sub‑gradient polishing** – after the smooth phase we run a
   projected (Euclidean) sub‑gradient descent on the exact max‑convolution.
   The sub‑gradient for the active shift *k* is
   ``g_j = a_{k-j} + a_j`` (with out‑of‑range indices treated as 0).  The
   iterate is projected onto the probability simplex after every step.
5. **Evaluation** – the best vector found is finally fed to
   ``evaluate_sequence`` and stored globally as ``height_sequence_1``.

The whole routine respects the 1000 s deadline, checks the wall‑clock
after each major block, and prints intermediate surrogate values and
the final true objective.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = y.max()
    e = np.exp(y - m)
    return e / e.sum()


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate value = 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n, gamma=1.0):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n, sigma_frac):
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n, alpha, beta):
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n, alpha):
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n, rate=0.998):
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n, conc, rng):
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n, rng):
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n, num_freq, rng):
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a, b, alpha):
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(n, rng, previous):
    """Return a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0, 5.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.3, 0.4, 0.6):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a_ in (0.5, 1.0, 2.0, 3.0):
        for b_ in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a_, b_))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixture with uniform (adds a flat bias)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final sanity normalisation
    out = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        out.append(v / s if s > 0 else _uniform(n))
    return out


# ----------------------------------------------------------------------
# Smooth max surrogate (L‑BFGS‑B) utilities
# ----------------------------------------------------------------------
def _lbfgsb_func(y, n, p):
    """Smooth max surrogate and its gradient w.r.t. logits y."""
    a = _softmax(y)                         # a lies on the simplex
    conv = _conv_fft(a, a)                  # length 2n‑1
    max_c = conv.max()

    shifted = conv - max_c                  # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)     # safety
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                        # fallback if underflowed completely
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)         # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    # chain rule for soft‑max parametrisation
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    # smooth surrogate value: (1/p)·log Σ exp(p·cₖ)
    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a, n, deadline, p, max_iter=400):
    """
    L‑BFGS‑B optimiser on the smooth surrogate.
    Returns the best vector found and its *exact* surrogate value.
    """
    y0 = np.log(np.maximum(start_a, 1e-15))
    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})
    a_opt = _softmax(res.x)
    val = _objective_surrogate(a_opt)
    return a_opt, val


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient (projected gradient descent)
# ----------------------------------------------------------------------
def _subgrad_max_conv(a):
    """Return sub‑gradient of max(conv(a,a)) and the corresponding max value."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))
    max_val = conv[k]

    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    rev = k - idx

    # contribution a_{k-i}
    g[idx] += a[rev]
    # contribution a_i
    g[rev] += a[idx]

    return g, max_val


def _pgd_optimize(start, n, deadline,
                  max_iters=300_000, alpha0=0.2):
    """
    Projected (Euclidean) sub‑gradient descent on exact max‑convolution.
    Returns the best vector and its surrogate value.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g, _ = _subgrad_max_conv(x)

        # step size schedule (classic 1/√t)
        alpha = alpha0 / math.sqrt(it)

        # gradient step (move opposite to sub‑gradient)
        x = x - alpha * g

        # project back onto simplex
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

        # Very cheap early‑stop check – if the improvement in the last
        # 10 000 iterations is negligible we can quit early.
        if it % 10000 == 0 and best_val - cur_val < 1e-9:
            break

    return best_x, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose a size that is a power of two (FFT friendly).
    # --------------------------------------------------------------
    n = 8192
    print("\n--- Problem size -------------------------------------------------")
    print(f"Length of the sequence: n = {n}")

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    print("\n--- Seed generation ------------------------------------------------")
    prev = globals().get("height_sequence_1")
    previous = np.asarray(prev, dtype=np.float64) if isinstance(prev, (list, tuple, np.ndarray)) else None
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    # Quick screening with the exact surrogate
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # Initialise best solution
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – L‑BFGS‑B on the smooth max (geometric p schedule)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS‑B optimisation (smooth max) ---------------------------")
        p_schedule = [5.0, 15.0, 45.0, 135.0, 405.0, 1215.0, 3645.0]
        for p in p_schedule:
            if time.time() >= deadline:
                break
            # primary optimisation from current best
            cand, val = _lbfgsb_optimize(best_seq, n, deadline, p=p, max_iter=400)
            print(f"[L‑BFGS] p={p:7.1f} → surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"    ** improved global best to {best_val:.6f}")

            # a couple of random restarts for each p (if time permits)
            for r in range(2):
                if time.time() >= deadline:
                    break
                # add a modest Gaussian perturbation in log‑space
                y_noise = np.log(np.maximum(best_seq, 1e-15)) + rng.normal(scale=0.15, size=n)
                perturbed = _softmax(y_noise)
                cand2, val2 = _lbfgsb_optimize(perturbed, n, deadline, p=p, max_iter=300)
                if val2 < best_val:
                    best_val, best_seq = val2, cand2
                    print(f"[Restart] p={p:.1f} (run {r+1}) → improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – Exact sub‑gradient polishing (projected gradient descent)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exact sub‑gradient polishing (PGD) ---------------------------")
        remaining = deadline - time.time()
        # allocate a modest budget (e.g. 15 % of remaining time) to the PGD phase
        pgd_deadline = time.time() + remaining * 0.15
        cand, val = _pgd_optimize(best_seq, n, pgd_deadline,
                                  max_iters=250_000, alpha0=0.25)
        print(f"[PGD] surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, cand
            print(f"    ** new global best after PGD: {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official function
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
