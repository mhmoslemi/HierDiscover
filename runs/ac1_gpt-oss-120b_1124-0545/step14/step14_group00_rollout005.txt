"""
Propose a step‐function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm uses a low‑dimensional Fourier‑basis parametrisation
combined with a continuation L‑BFGS optimiser (smooth log‑sum‑exp
approximation of the max), a large set of analytically‑generated seeds,
and a final mirror‑descent polishing step.  The problem size is
increased to n=8000 (the theoretical lower bound for the objective is
≈1, so a larger n gives the optimiser more freedom).  All heavy work is
guarded by a deadline (budget_s‑2 s) so the function always returns a
valid sequence before the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS, linear programmes)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = None
    _sp = None

# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float = 8.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # modified Bessel (numpy)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

def _fourier_basis(n: int, K: int) -> np.ndarray:
    """
    Returns an (n, K) matrix whose columns are cosine basis vectors:
    phi_k[i] = cos(π * k * i/(n‑1)),   k = 1 … K
    (k=0 is omitted because it adds a constant to logits and is
    irrelevant after softmax.)
    """
    i = np.arange(n, dtype=np.float64)
    basis = np.empty((n, K), dtype=np.float64)
    for k in range(1, K + 1):
        basis[:, k - 1] = np.cos(np.pi * k * i / (n - 1))
    return basis


def _seq_from_logits(logits: np.ndarray) -> np.ndarray:
    """Convert raw logits to a normalised non‑negative vector."""
    return _softmax(logits)


def _seq_from_coeffs(coeffs: np.ndarray, basis: np.ndarray) -> np.ndarray:
    """Build sequence = softmax(basis @ coeffs)."""
    logits = basis @ coeffs
    return _seq_from_logits(logits)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iter_per_beta: int = 8000,
                           beta_schedule=None) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)          # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            smooth_max = shift + np.log(sum_exp) / beta_val
            obj = 2.0 * n * smooth_max

            # gradient wrt a (derived from the surrogate)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max (logits)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser (exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> np.ndarray:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)           # slightly larger step than before
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq

    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            vec = _random_dirichlet(n, conc, rng)
            add(vec, ('dirichlet', conc))

    # low‑frequency random curves
    for _ in range(4):
        vec = _lowfreq_curve(n, num_freq=3, rng=rng)
        add(vec, ('lowfreq',))

    # Fourier‑basis random coefficients (low‑dimensional)
    K = 12
    basis = _fourier_basis(n, K)
    for _ in range(5):
        coeffs = rng.normal(scale=0.5, size=K)
        vec = _seq_from_coeffs(coeffs, basis)
        add(vec, ('fourier',))

    # keep best 30 seeds
    pool.sort(key=lambda x: x[1])
    return [(vec, score) for vec, score, _ in pool[:30]]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # problem size – a larger n gives the optimiser more freedom
    n = 8000

    # ------------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------------
    print("\n--- Generating seed pool ---")
    seed_pool = _build_seed_pool(n, rng)

    # ------------------------------------------------------------------
    # 2. Keep a modest top‑k for deep optimisation
    # ------------------------------------------------------------------
    seed_pool.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in seed_pool[:top_k]]
    print(f"Collected {len(seed_pool)} seeds, keeping top {top_k} for optimisation.")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS on the best seeds (more iterations than before)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] current best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        # allocate a share of the remaining budget
        max_iter_beta = int(min(8000, remaining * 0.08 * 1e5))
        max_iter_beta = max(1200, max_iter_beta)

        print(f"\n--- L‑BFGS run #{idx + 1}/{len(top_seeds)} (max_iter/beta={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 4. Short exponentiated‑gradient polishing on the current best
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 3.0:
            max_exp = int(min(300_000, remaining * 0.07 * 1e5))
            seq_exp = _exp_gradient_opt(global_best_seq,
                                        n,
                                        deadline,
                                        max_iters=max_exp)
            val_exp = _objective_surrogate(seq_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final mirror‑descent polishing (longer run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_final = int(min(500_000, remaining * 0.07 * 1e5))
        if max_exp_final > 0:
            print("\n--- Final Exp‑Gradient polishing ---")
            seq_final = _exp_gradient_opt(global_best_seq,
                                          n,
                                          deadline,
                                          max_iters=max_exp_final)
            val_final = _objective_surrogate(seq_final)
            if val_final < global_best_val:
                global_best_seq = seq_final.copy()
                global_best_val = val_final
                print(f"[Final ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional symmetry enforcement (helps reduce numerical noise)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sym = (global_best_seq + global_best_seq[::-1]) * 0.5
        global_best_seq = sym / sym.sum()
        global_best_val = _objective_surrogate(global_best_seq)
        print(f"[Symmetry] enforced, surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
