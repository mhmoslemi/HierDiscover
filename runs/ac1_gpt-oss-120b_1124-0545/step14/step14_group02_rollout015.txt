def _local_perturb_optimize(start_seq, n, deadline, step_factor=0.1, batch=10):
   a = start_seq.copy()
   best_val = _objective_surrogate(a)
   while time.time() < deadline:
       # propose perturbation: randomly select subset of indices of size s (maybe 10)
       s = max(2, int(n * 0.001)) # ~4 indices
       idx = rng.choice(n, size=s, replace=False)
       # scale them randomly up/down with some factor
       perturb = rng.normal(scale=step_factor, size=s)
       new_a = a.copy()
       new_a[idx] = np.maximum(new_a[idx] + perturb, 0.0)
       # renormalize to sum to 1
       new_a = new_a / new_a.sum()
       val = _objective_surrogate(new_a)
       if val < best_val:
           a = new_a
           best_val = val
           # optionally reduce step_factor
   return a

Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence`` within the given time budget.

Algorithm (≈ 1000 s total):
0️⃣  Choose a problem size (n=4000 works well).  
1️⃣  Generate a **large, diverse seed pool** – deterministic families
    (uniform, triangular, sine‑windows, Gaussians, power‑law,
    exponential, double‑exponential, double‑power, beta‑like,
    Dirichlet draws, low‑frequency random curves).  All vectors are
    ℓ₁‑normalised.
2️⃣  Compute the cheap **surrogate** 2·n·max(conv) for every seed,
    keep the best *k* (k≈15).
3️⃣  For each elite seed run a **continuation L‑BFGS** on a smooth
    approximation of the max‑convolution (log‑sum‑exp).  The schedule
    of the smoothing parameter β is extended up to 20000, and the
    per‑β iteration budget is chosen dynamically from the remaining
    time.  The optimiser works on logits; the soft‑max keeps the
    iterate on the simplex.
4️⃣  Starting from the best vector run a **mirror‑descent (exponentiated‑gradient)**
    optimisation on the exact non‑smooth objective.  The step size
    ηₜ = 0.5 / √t and the number of iterations are again limited by
    the time left.
5️⃣  **Local stochastic search** – a simulated‑annealing style
    swapping+mass‑transfer moves on the current best vector.  Swaps
    change the ordering of the entries (important because the
    convolution depends on the ordering), while small mass‑transfer
    moves adjust the values.  Both are accepted if they improve the
    surrogate, otherwise with a temperature‑dependent probability.
    This phase often reduces the objective a little further.
6️⃣  If ≥30 s remain, **upsample** the current best (linear interpolation)

7️⃣  Return the best sequence discovered (plain list) and store it as

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30
    return a / a.sum()


def _double_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric U‑shape: a_i ∝ e^{-lam·i}+e^{-lam·(n-1-i)}."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return a / a.sum()


def _double_power(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)          # 1‑based for readability
    a = (i ** (-p)) + ((n - i + 1) ** (-p))
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):

    for nf in (2, 3, 4, 5):

# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 4000,
) -> np.ndarray:
    """Optimise a smooth surrogate of max‑convolution using L‑BFGS‑B."""
    if beta_schedule is None:
        beta_schedule = [
            1.0,
            5.0,
            20.0,
            80.0,
            200.0,
            500.0,
            1000.0,
            3000.0,
            5000.0,
            10000.0,
            20000.0,
        ]

    # initialise logits safely away from -inf
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # a cheap shift
            # smoothed max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (standard for soft‑max of max)
            w = exp_beta / sum_exp                     # weight vector
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-13, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            # time limit for this seed reached – move on
            break
        except Exception as e:
            # unexpected problem – abort remaining β‑steps for this seed
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)

# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
) -> np.ndarray:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active convolution location
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Local stochastic search (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _local_search(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Simulated‑annealing style hill‑climbing:
    * random swaps of two positions
    * random transfer of a small mass from i → j
    Both moves are accepted if they improve the surrogate,
    otherwise with a temperature‑dependent probability.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    start_time = time.time()
    # temperature schedule (linear decay)
    T_start = 5e-3
    T_end = 1e-6

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = T_start * (1.0 - frac) + T_end * frac

        # decide move type (50 % swap, 50 % mass transfer)
        if rng.random() < 0.5:
            # ----- swap two entries -------------------------------------------------
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur[i], cur[j] = cur[j], cur[i]          # perform swap
            new_val = _objective_surrogate(cur)
            accept = new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / T)
            if accept:
                cur_val = new_val
                if new_val < best_val:
                    best_val, best = new_val, cur.copy()
                    print(f"[Local] new best {best_val:.6f} (swap) after {it} iters")
            else:
                # revert swap
                cur[i], cur[j] = cur[j], cur[i]
        else:
            # ----- transfer a small amount of mass ---------------------------------
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] == 0.0:
                continue
            # transfer up to 10 % of source mass (scaled by a random factor)
            frac_transfer = 0.02 * rng.random()
            delta = frac_transfer * cur[i]
            cur[i] -= delta
            cur[j] += delta
            new_val = _objective_surrogate(cur)
            accept = new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / T)
            if accept:
                cur_val = new_val
                if new_val < best_val:
                    best_val, best = new_val, cur.copy()
                    print(f"[Local] new best {best_val:.6f} (transfer) after {it} iters")
            else:
                # revert transfer
                cur[i] += delta
                cur[j] -= delta

        it += 1
    return best


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin for final evaluation

    # ------------------------------------------------------------------
    # Stage 0 – problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & cheap surrogate ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=4000)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values (top {top_k} seeds):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS on elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    beta_sched = [
        1.0,
        5.0,
        20.0,
        80.0,
        200.0,
        500.0,
        1000.0,
        3000.0,
        5000.0,
        10000.0,
        20000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all elite seeds.")
            break

        # allocate iterations depending on remaining budget
        remaining = deadline - time.time()
        per_beta_iters = max(800, int(min(5000, remaining * 0.04 * 1e5)))
        print(f"\n--- L‑BFGS on elite seed #{idx+1} (max_iter/beta={per_beta_iters}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta_iters,
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – mirror‑descent (exact max) refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(3_000_000, remaining * 0.25 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_exp_iters,
            )
            val_exp = _objective_surrogate(seq_exp)
            if val_exp < global_best_val:
                global_best_seq, global_best_val = seq_exp.copy(), val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – local stochastic search (swap + mass‑transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Local stochastic (swap/transfer) search ---")
        seq_local = _local_search(global_best_seq, n, deadline, rng)
        val_local = _objective_surrogate(seq_local)
        if val_local < global_best_val:
            global_best_seq, global_best_val = seq_local.copy(), val_local
            print(f"[Local] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – optional up‑sampling and fine‑tuning on a finer grid
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample(global_best_seq, new_n)

            # repeat the smooth‑L‑BFGS on the finer grid
            up_beta_sched = beta_sched[:]
            up_seq_opt = _lbfgs_smooth_optimize(
                up_seq,
                new_n,
                deadline,
                beta_schedule=up_beta_sched,
                max_iter_per_beta=2000,
            )
            up_val = _objective_surrogate(up_seq_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq, global_best_val = up_seq_opt.copy(), up_val
                n = new_n
                print(f"[Update] new best after up‑sampling = {global_best_val:.6f}")

            # final mirror‑descent on fine grid
            if time.time() < deadline:
                rem = deadline - time.time()
                max_exp2 = int(min(800_000, rem * 0.12 * 1e5))
                print("\n--- Final exponentiated‑gradient on fine grid ---")
                seq_final = _exp_gradient_optimize(
                    global_best_seq,
                    n,
                    deadline,
                    max_iters=max_exp2,
                )
                val_final = _objective_surrogate(seq_final)
                if val_final < global_best_val:
                    global_best_seq, global_best_val = seq_final.copy(), val_final
                    print(f"[Final‑ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
