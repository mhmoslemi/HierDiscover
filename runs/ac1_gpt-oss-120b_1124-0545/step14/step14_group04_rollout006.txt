"""
Improved non‑negative step‑function search.

The algorithm consists of three phases:

1️⃣  Fast Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution.
    This quickly yields a good basin while keeping the entries on the probability simplex.

2️⃣  A Frank–Wolfe‑style refinement based on an exact linear programme.
    For the current vector f we solve
        maximise   Σ b_j
        subject to b_j ≥ 0  and   (f * b)[k] ≤ M   (∀k),
    where M = max_k (f * f)[k].  The LP gives a new extreme point g that does not enlarge
    the maximal convolution.  We update f ← (1‑γ)·f + γ·g (γ = 0.5 or a decreasing schedule)
    and repeat while time permits.

3️⃣  Short additive sub‑gradient polishing on the true objective
    (2·n·max(f*f)) with Euclidean projection onto the simplex.

All stages respect a global time budget (default 1000 s).  The best vector found
is returned as a plain Python list and stored in the global variable
``height_sequence_1`` for warm‑starting the next run.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(p: np.ndarray) -> float:
    """Exact objective for a vector with Σp = 1:
       2·n·max(conv(p,p)).
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on a smooth surrogate (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------
def _adam_softmax_optimize(start: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 200_000,
                           alpha_start: float = 20.0,
                           alpha_end: float = 2500.0,
                           lr: float = 0.12) -> tuple[np.ndarray, float]:
    """Adam on L(p) = (1/α)·log Σ exp(α·conv(p)[k])."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective(x)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(3 * n - 2)        # length needed for the FFTs

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- soft‑max over the full convolution -------------------------
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)                # soft‑max weights

        # gradient w.r.t. x (see comment in the statement)
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam update on the *unconstrained* variables
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # projection onto the simplex
        s = x.sum()
        if s <= 0.0:
            x = np.full_like(x, 1.0 / n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 20_000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Additive sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgrad_optimize(start: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 120_000,
                      eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(p)=2·n·max(conv(p,p))."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 30_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Frank–Wolfe‑style LP refinement
# ----------------------------------------------------------------------
def _lp_refinement(f: np.ndarray,
                   n: int,
                   deadline: float,
                   max_lp_iters: int = 12) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve:
        maximise Σ b_j
        s.t. b ≥ 0  and  (f * b)[k] ≤ M   (∀k),
    where M = max conv(f,f).
    The returned vector is a convex combination of the previous iterate
    and the LP extreme point (full replacement if it improves).
    """
    # Pre‑compute the static index pattern for the constraint matrix.
    i = np.arange(n, dtype=np.int32)
    row_idx = np.repeat(i, n) + np.tile(i, n)          # shape (n*n,)
    col_idx = np.tile(i, n)                           # shape (n*n,)
    nnz = n * n
    # allocate a sparse matrix with placeholder data – the data will be
    # overwritten on every iteration (no re‑allocation of the sparsity pattern).
    A_ub = sparse.csr_matrix((np.empty(nnz, dtype=np.float64),
                              (row_idx, col_idx)),
                             shape=(2 * n - 1, n))

    best_f = f.copy()
    best_val = _objective(best_f)

    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        # current maximal convolution value
        M = np.max(_conv_fft(best_f, best_f))

        # fill the data vector (each row i repeats the coefficient f_i n times)
        A_ub.data = np.repeat(best_f, n)

        # Linear programme: maximise Σ b_j  ⇔ minimise -Σ b_j
        c = -np.ones(n, dtype=np.float64)   # minimise -sum(b)
        b_ub = M * np.ones(2 * n - 1, dtype=np.float64)

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(1, int(deadline - time.time()))})
        if not res.success:
            print(f"[LP] iteration {lp_it} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b / sum_b                     # feasible point on the simplex

        # try a full replacement first – it often improves
        cand = g
        cand_val = _objective(cand)
        if cand_val < best_val:
            best_f, best_val = cand, cand_val
            print(f"[LP] it={lp_it:2d} full replace → {best_val:.6f}")
            continue

        # otherwise perform a Frank–Wolfe step with γ = 0.5 (or decreasing)
        gamma = 0.5
        cand = (1.0 - gamma) * best_f + gamma * g
        cand_val = _objective(cand)
        if cand_val < best_val:
            best_f, best_val = cand, cand_val
            print(f"[LP] it={lp_it:2d} FW step → {best_val:.6f}")
        else:
            # nothing improves – stop early
            print(f"[LP] it={lp_it:2d} no improvement, terminating LP phase")
            break

    return best_f, best_val
