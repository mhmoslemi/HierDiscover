def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = y.max()
    e = np.exp(y - m)
    return e / e.sum()


def _objective_surrogate(seq: np.ndarray) -> float:

def _uniform(n: int) -> np.ndarray:

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:

    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)

        return a / a.sum()

def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:

    s = a.sum()

# Exponentiated‑gradient optimiser (mirror descent on simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # stepsize (√t schedule)
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val

# L‑BFGS‑B optimisation of the smooth surrogate
# ----------------------------------------------------------------------
def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """Smooth surrogate and its gradient w.r.t. logits y."""
    a = _softmax(y)                     # a lies on the simplex
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = conv.max()

    shifted = conv - max_c               # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)  # safety
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                    # fallback to exact max locations
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)     # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    # chain rule: ∂F/∂y = diag(a)·(∂F/∂a – (a·∂F/∂a)·1)
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    # smooth surrogate value: (1/p)·log Σ exp(p·cₖ)
    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a: np.ndarray, n: int, deadline: float,
                     p: float = 20.0, max_iter: int = 250) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the smooth surrogate with logits parametrisation.
    Returns the best vector found and its *exact* surrogate value
    (2·n·max(conv)).
    """
    y0 = np.log(np.maximum(start_a, 1e-15))
    # SciPy's minimize will call our function with (y, n, p) via args
    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})

    y_opt = res.x
    a_opt = _softmax(y_opt)

    max_conv = _conv_fft(a_opt, a_opt).max()
    val = 2.0 * n * max_conv
    return a_opt, val
