def _gaussian_mixture(n, rng, num_components=2):
   a = np.zeros(n)

    """

    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = np.max(_conv_fft(a, a))
        # Build convolution matrix
        L = 2 * n - 1

            prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
        except Exception as e:
            print("[LP] solving error:", e)
            break
        if b.value is None:
            print("[LP] no solution")
            break
        b_opt = np.maximum(np.array(b.value).flatten(), 0.0)
        sum_b = b_opt.sum()

            # if no improvement, maybe break

"""
Improved step‑function search.

The algorithm works in several stages:
1.  Build a large pool of cheap seeds (analytic families, random Dirichlet,
    low‑frequency curves, Gaussian mixtures).  All vectors are ℓ₁‑normalised.
2.  Pick the best few seeds according to the cheap surrogate
    `2·n·max(conv(a,a))` (the denominator is 1 because of the normalisation).
3.  For each elite seed run a continuation L‑BFGS on a smooth log‑sum‑exp
    approximation of the max‑convolution (many iterations – cheap because the
    problem size is small, n≈1024).
4.  Refine the best seed with a large‑scale Adam optimiser that minimises the
    p‑norm of the convolution for a schedule of growing p (this smoothly
    approaches the true ∞‑norm).
5.  Apply a linear‑programming “Frank‑Wolfe‑style’’ improvement: maximise the
    ℓ₁‑norm of a candidate while keeping its convolution under the current
    maximum, then take a convex combination.  The LP is tiny (n≈1024) and can be
    solved repeatedly.
6.  Upsample the best vector to a finer grid (n≈4096) and repeat steps 3‑5.
7.  Finish with a short exponentiated‑gradient (mirror‑descent) pass and a
    very cheap stochastic hill‑climbing phase.
8.  Return the best sequence found (as a plain list) and store it in the
    global name ``height_sequence_1`` for possible reuse.
"""

# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (always returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _logsumexp(x: np.ndarray) -> float:
    """Numerically stable log‑sum‑exp."""
    m = np.max(x)
    return m + np.log(np.sum(np.exp(x - m)))


# ----------------------------------------------------------------------
# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _gaussian_mixture(n: int, rng: np.random.Generator,
                      components: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    for _ in range(components):
        mu = rng.uniform(0, n - 1)
        sigma = max(1.0, rng.uniform(0.01, 0.2) * n)
        w = rng.uniform(0.1, 1.0)
        i = np.arange(n, dtype=np.float64)
        a += w * np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float,
                     rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int,
                   rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 3000) -> list[np.ndarray]:
    """Diverse ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random parametric families
    for _ in range(max_random // 5):
        # beta‑like
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        # power‑law
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        # exponential decay
        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        # linear‑power decreasing
        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # mixture uniform–beta
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

        # Gaussian mixture
        seeds.append(_gaussian_mixture(n, rng, components=2))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Surrogate (scale‑invariant) objective
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    Since sum(seq)=1 the true objective equals this value.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (log‑sum‑exp smooth max)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray, n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimises a log‑sum‑exp smooth approximation of max‑convolution
    using L‑BFGS‑B.  Returns a ℓ₁‑normalised vector (soft‑max of logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 3000.0, 5000.0]

    v = np.log(start_vec + 1e-30)          # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # smooth max
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[LBFGS] time limit reached at beta={beta_val}")
            break
        except Exception as e:
            print(f"[LBFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser on p‑norm surrogate (smooth ∞‑norm)
# ----------------------------------------------------------------------
def _adam_max_approx(start_vec: np.ndarray, n: int,
                     deadline: float,
                     p_schedule=None,
                     max_total_iters: int = 300_000) -> np.ndarray:
    """
    Adam optimisation of ` (∑ conv^p)^{1/p}` which approximates max(conv).
    Returns a ℓ₁‑normalised vector.
    """
    if p_schedule is None:
        p_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = _softmax(np.log(start_vec + 1e-30))        # start on simplex
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    total_iters = 0
    for p in p_schedule:
        # allocate a modest budget per p (but respect the global deadline)
        while time.time() < deadline and total_iters < max_total_iters:
            t += 1
            total_iters += 1

            conv = _conv_fft(a, a)          # length 2n‑1
            log_conv = np.log(conv + 1e-30)

            # compute stable weights for the p‑norm gradient
            log_sum_p = _logsumexp(p * log_conv)
            log_w = (p - 1) * log_conv - ((p - 1) / p) * log_sum_p
            w = np.exp(log_w)                # shape 2n‑1, sums to 1

            # gradient w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update (on logits)
            m = beta1 * m + (1 - beta1) * grad_v
            v = beta2 * v + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)

            logits = np.log(a + 1e-30) - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _softmax(logits)

            # occasional evaluation
            if total_iters % 2000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
        if time.time() > deadline or total_iters >= max_total_iters:
            break

    return best_seq


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 800_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# LP refinement (Frank‑Wolfe style)
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray, n: int,
               deadline: float,
               max_iters: int = 8,
               t: float = 0.2) -> np.ndarray:
    """
    Repeatedly solve a linear program:
        maximise sum(b)  s.t.  conv(seq, b) ≤ M, b ≥ 0
    where M = max(conv(seq, seq)).
    Take a convex combination with mixing weight `t`.
    Returns the best sequence encountered.
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print("[LP] cvxpy not available, skipping LP refinement.")
        return seq

    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = np.max(_conv_fft(a, a))

        # Build the convolution matrix (dense, but n≈1024 → manageable)
        L = 2 * n - 1
        A = np.zeros((L, n), dtype=np.float64)
        idx = np.arange(n)
        for i in range(n):
            A[i + idx, idx] = a[i]

        b = cp.Variable(n)
        constraints = [A @ b <= M,
                       b >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=300, abstol=1e-8, reltol=1e-8)
        except Exception as e:
            print(f"[LP] solver failed: {e}")
            break

        if b.value is None:
            print("[LP] No solution returned.")
            break
        b_opt = np.maximum(np.asarray(b.value).flatten(), 0.0)
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            print("[LP] sum(b) is non‑positive.")
            break

        g = b_opt / sum_b                # normalise to ℓ₁‑norm 1
        a_new = (1 - t) * a + t * g

        val = _objective_surrogate(a_new)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            print(f"[LP] iter {it+1}: improved surrogate = {val:.6f}")
        else:
            print(f"[LP] iter {it+1}: no improvement")
            break

    return best_seq


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (tiny local search)
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                n: int,
                deadline: float,
                max_trials: int = 2000,
                step_std: float = 0.05) -> np.ndarray:
    """
    Very cheap multiplicative perturbations on two random coordinates.
    Accepts only improvements.
    """
    rng = np.random.default_rng()
    a = seq.copy()
    best_val = _objective_surrogate(a)

    for trial in range(max_trials):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.normal(scale=step_std)
        factor = np.exp(delta)          # multiplicative change
        a_new = a.copy()
        a_new[i] *= factor
        a_new[j] /= factor
        # stay in the simplex automatically (product of two factors = 1)
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        cur_val = _objective_surrogate(a_new)
        if cur_val < best_val:
            best_val = cur_val
            a = a_new
    return a


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    # ------------------------------------------------------------------
    # Stage 0 – problem sizes
    # ------------------------------------------------------------------
    n_small = 1024
    n_big = 4096
    print(f"\n=== Starting search (n_small={n_small}, n_big={n_big}) ===")

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & cheap surrogate ranking (small grid)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n_small, rng, prev_arr, max_random=3000)
    print(f"Generated {len(seeds)} seed vectors (small grid).")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (small grid):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS on elite seeds (small grid)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Small‑grid best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all L‑BFGS seeds.")
            break
        remaining = deadline - time.time()
        max_iters_beta = max(1500, int(min(6000, remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS on elite seed #{idx+1} (max_iter/beta={max_iters_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n_small,
                                         deadline,
                                         beta_schedule=[1,5,20,80,200,500,1000,3000,5000],
                                         max_iter_per_beta=max_iters_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Adam optimisation (smooth p‑norm) on the best small‑grid seed
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_adam_iters = int(min(300_000, remaining * 0.30 * 1e5))
        print(f"\n--- Adam optimisation (p‑norm) – up to {max_adam_iters} iters ---")
        seq_adam = _adam_max_approx(global_best_seq,
                                    n_small,
                                    deadline,
                                    max_total_iters=max_adam_iters)
        val_adam = _objective_surrogate(seq_adam)
        print(f"[Adam] surrogate = {val_adam:.6f}")
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Update] New best after Adam = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – LP refinement on the small grid
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP refinement (small grid) ---")
        seq_lp = _lp_refine(global_best_seq,
                            n_small,
                            deadline,
                            max_iters=8,
                            t=0.2)
        val_lp = _objective_surrogate(seq_lp)
        print(f"[LP] surrogate = {val_lp:.6f}")
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[Update] New best after LP = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Exponentiated‑gradient fine‑tuning (small grid)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(500_000, remaining * 0.20 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient (small grid) ---")
            seq_exp = _exp_gradient_optimize(global_best_seq,
                                             n_small,
                                             deadline,
                                             max_iters=max_exp_iters)
            val_exp = _objective_surrogate(seq_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – Upsampling to a finer grid
    # ------------------------------------------------------------------
    if time.time() < deadline and (deadline - time.time()) > 30.0:
        print(f"\n--- Upsampling to n_big={n_big} ---")
        up_seq = _upsample(global_best_seq, n_big)

        # L‑BFGS on big grid (light weight)
        remaining = deadline - time.time()
        max_iters_big = int(min(5000, remaining * 0.12 * 1e5))
        seq_big = _lbfgs_smooth_optimize(up_seq,
                                         n_big,
                                         deadline,
                                         beta_schedule=[1,5,20,80,200,500,1000],
                                         max_iter_per_beta=max_iters_big)
        val_big = _objective_surrogate(seq_big)
        print(f"[Upsample‑L‑BFGS] surrogate = {val_big:.6f}")

        if val_big < global_best_val:
            global_best_seq = seq_big.copy()
            global_best_val = val_big
            n = n_big
            print(f"[Update] New best after up‑sampling = {global_best_val:.6f}")
        else:
            n = n_big   # keep the larger n for later stages

        # Follow‑up exponentiated‑gradient on the fine grid
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp_big = int(min(600_000, remaining * 0.15 * 1e5))
            if max_exp_big > 0:
                print("\n--- Exponentiated‑gradient (fine grid) ---")
                seq_exp_big = _exp_gradient_optimize(global_best_seq,
                                                     n,
                                                     deadline,
                                                     max_iters=max_exp_big)
                val_exp_big = _objective_surrogate(seq_exp_big)
                if val_exp_big < global_best_val:
                    global_best_seq = seq_exp_big.copy()
                    global_best_val = val_exp_big
                    print(f"[ExpGrad‑big] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 7 – Short stochastic hill‑climbing (final polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_trials = int(min(3000, remaining * 300))   # a few thousand cheap trials
        print("\n--- Stochastic hill‑climbing polish ---")
        seq_final = _hill_climb(global_best_seq,
                                n,
                                deadline,
                                max_trials=max_trials,
                                step_std=0.07)
        val_final = _objective_surrogate(seq_final)
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
