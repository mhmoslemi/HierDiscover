"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

The optimiser works in three stages:

1. **Broad parameter sweep** – generate many ℓ₁‑normalised candidates from
   analytic families (uniform, triangular, exponential, power,
   Gaussian, beta, Tukey, sine) and random constructions
   (Dirichlet draws, low‑frequency cosine mixtures).  The exact
   surrogate ``2·n·max(conv(a,a))`` (which is the same as the official
   objective after normalisation) is evaluated for every candidate and the
   best few are kept.

2. **Sub‑gradient (Exponentiated‑Gradient) refinement** – starting from the
   top seeds we run a mirror‑descent on the simplex using the sub‑gradient
   of the active convolution shift.  The update is multiplicative
   (``x ← x·exp(−η·g)``) followed by a projection onto the probability
   simplex.  A decreasing step size (``η∝1/√t``) together with many
   iterations drives the max‑convolution down.

3. **Smooth‑max L‑BFGS‑B optimisation** – a smooth approximation of the
   ``max`` is minimised:

        Fₚ(a) = (1/p)·log Σₖ exp(p·convₖ(a))

   The gradient w.r.t. logits (soft‑max parametrisation) is supplied to
   SciPy's L‑BFGS‑B.  The smoothing parameter ``p`` is increased
   geometrically (5 → 15 → 45 → 135 → 400 → 1200); for each ``p`` a few
   random restarts (Gaussian noise on the logits) are performed.
   The best vector found is finally polished with a short EG pass.

The best vector is evaluated with the official ``evaluate_sequence`` and
stored in the global ``height_sequence_1`` variable for possible reuse.
The whole routine respects the supplied time budget (default 1000 s).
"""
import time
import math
import numpy as np
import scipy.optimize as _opt

# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate (scale‑invariant) = 2·n·max(conv(a,a))."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Exponentiated‑gradient optimiser (mirror descent on simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)        # keep exp() stable
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (logits parametrisation)
# ----------------------------------------------------------------------
def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """Smooth max surrogate and gradient w.r.t. logits ``y``."""
    a = _softmax(y)                     # a lives on the simplex
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = conv.max()

    shifted = conv - max_c               # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)  # avoid overflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                     # fallback – should rarely happen
        w = (conv == max_c).astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)     # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    # chain rule for logits (soft‑max)
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a: np.ndarray, n: int, deadline: float,
                     p: float = 20.0, max_iter: int = 250) -> tuple[np.ndarray, float]:
    """L‑BFGS‑B on the smooth surrogate (logits parametrisation)."""
    y0 = np.log(np.maximum(start_a, 1e-15))

    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})

    a_opt = _softmax(res.x)
    val = _objective_surrogate(a_opt)     # exact surrogate value
    return a_opt, val
