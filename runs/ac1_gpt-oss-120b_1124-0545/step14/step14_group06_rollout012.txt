"""
Improved search for a non‑negative step function that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

The implementation follows the same overall pipeline as the reference
code but adds a few new ingredients:

* a richer set of analytic “windows’’ – arcsine shape, more extreme beta
  parameters and mixtures of two exponentials;
* a dedicated 1‑D golden‑section search to optimise the exponential decay
  rate (the reference sweep stopped at 0.9999);
* a multi‑active exponential‑gradient (mirror‑descent) optimiser that
  aggregates sub‑gradients from all nearly‑active convolution entries
  (instead of only the single maximum);
* a higher smoothing schedule for the Adam optimiser (p grows up to 500);
* a longer final EG polishing phase.

All stages respect the global deadline (default 1000 s).  The best vector
found is returned as a plain list and stored in the global variable
``height_sequence_1`` for possible reuse.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate when Σ seq = 1:  2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, r1: float, r2: float,
                        alpha: float) -> np.ndarray:
    """
    Convex combination of two exponentials:
        a = (1‑α)·r1**i + α·r2**i   (ℓ₁‑normalised)
    """
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - alpha) * (r1 ** i) + alpha * (r2 ** i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """U‑shaped arcsine window: a_i ∝ 1/√(i·(n‑i))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n - i + 0.5))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

    sub‑gradient of *all* near‑active convolution entries.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        max_val = conv.max()
        # active set – all entries within 0.01 % of the maximum
        tol_abs = max_val * 1e-4 + 1e-14
        active = np.where(conv >= max_val - tol_abs)[0]

        g = np.zeros_like(x)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] += 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.04,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 1.0, p_target: float = 500.0,
                          p_step: int = 3000,
                          max_total_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate
          F_p(a) = 1/p·log Σ_k exp(p·conv_k(a))
    with a gradual increase of p up to ``p_target``.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # log‑sum‑exp weights
        shifted = conv - max_conv
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of smooth surrogate
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else _uniform(n)

        # increase smoothness gradually
        if t % p_step == 0:
            p = min(p * 1.5, p_target)

    final_val = _objective_surrogate(best_a)
    return best_a, final_val


def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Very lightweight SA, same as reference but with a slightly slower cooling."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99996, min_temp)

        if rng.random() < 0.55:
            # reshuffle a small random block
            k = rng.integers(4, 13)
            idx = rng.choice(n, size=k, replace=False)
            tot = cur[idx].sum()
            if tot <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                # occasional progress report
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative step function under the given
    time budget.  Returns a plain list of floats; the best found vector
    is also stored in the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # tiny safety margin

    # ------------------------------------------------------------------
    # Length – 5000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 5000
    print(f"\n--- Length set to n={n} ---")

    # ------------------------------------------------------------------
    # Assemble seed pool (including an analytically optimised exponential)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)

    # add the 1‑D exponential optimum (cheap, but often strong)
    exp_opt = _best_exponential_rate(n, deadline)
    if exp_opt is not None:
        seeds.append(exp_opt)

    print(f"Collected {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # Keep a handful of the most promising seeds (according to the exact surrogate)
    # ------------------------------------------------------------------
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, v) in enumerate(scored[:top_k]):
        print(f"  #{i + 1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – multi‑active EG refinement on the best seeds
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Multi‑active EG refinement (first pass) ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            max_it = int(min(120_000, remaining * 1800))
            cand, val = _expgrad_multi_optimize(seed_vec, n, deadline,
                                                max_iters=max_it,
                                                eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[EG‑multi] seed #{idx+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – high‑p Adam smoothing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam smoothing (p up to 500) ---")
        remaining = deadline - time.time()
        max_adam_iters = int(min(1_300_000, remaining * 1700))
        best_seq, best_val = _adam_smooth_optimize(best_seq, n, deadline,
                                                   init_lr=0.045,
                                                   p0=1.0,
                                                   p_target=500.0,
                                                   p_step=2500,
                                                   max_total_iters=max_adam_iters)
        print(f"[Adam] best surrogate after smoothing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1900))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_iters)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final long EG polishing (plain EG, many iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 2100))
        if max_iters > 0:
            best_seq, best_val = _expgrad_multi_optimize(best_seq, n, deadline,
                                                         max_iters=max_iters,
                                                         eta0=0.22)
            print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
