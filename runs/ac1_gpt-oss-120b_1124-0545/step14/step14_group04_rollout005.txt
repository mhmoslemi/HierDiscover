"""
Hybrid optimiser (v2) for the non‑negative step‑function problem.

Key ideas
---------
*   Enrich the seed pool with *reversed* (high‑later‑mass) versions of all
    analytic families and random draws – the optimal distribution is often
    highly asymmetric, concentrating weight near the end of the support.
*   Scan a finer grid of parameters for the analytic windows (β‑shapes,
    sine‑windows, Gaussian σ, Tukey α) and add a few explicit “increasing”
    power / exponential families.
*   Run a short sub‑gradient polishing from each of the top‑k seeds (instead
    of only from the single best seed) to escape early local minima.
*   Keep the later stages (long‑run exponential‑gradient, simulated‑annealing)
    exactly as before, but with a slightly more aggressive schedule.

The function respects the global time budget (default 1000 s) and always
returns the best sequence found so far, guaranteeing a finite answer.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(p,p)) for a ℓ₁‑normalised vector."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Helper to generate the *reversed* version of any seed (still ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _reversed(v: np.ndarray) -> np.ndarray:
    """Return the reversed version of v (still normalised)."""
    r = v[::-1].copy()
    s = r.sum()
    return r / s if s > 0 else r


# ----------------------------------------------------------------------
# Seed collection (now with many reversed / extra families)
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    families = []

    families.append(_uniform(n))
    families.append(_triangular(n))

    # sine windows – wider range of γ
    for gamma in (0.5, 1.0, 2.0, 3.0):
        families.append(_sine_window(n, gamma))

    # Gaussians – several σ fractions
    for sigma_frac in (0.12, 0.2, 0.3, 0.4):
        families.append(_gaussian(n, sigma_frac))

    # Exponential decays – two directions (normal & reversed)
    families.append(_exponential_decay(n, 0.9995))
    families.append(_reversed(_exponential_decay(n, 0.9995)))

    # Power laws – decreasing & increasing
    for p in (0.5, 1.0, 2.0, 3.0):
        families.append(_power(n, p))
        families.append(_reversed(_power(n, p)))

    # Beta shapes – dense grid, also reversed automatically later
    for a in (0.2, 0.3, 0.5, 0.8, 1.0, 2.0):
        for b in (0.2, 0.3, 0.5, 0.8, 1.0, 2.0):
            families.append(_beta_shape(n, a, b))

    # Tukey windows – several α
    for alpha in (0.1, 0.2, 0.5, 0.8):
        families.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (wide range of concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            families.append(_random_dirichlet(n, conc, rng))

    # Random exponentials
    for _ in range(6):
        families.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            families.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform – both directions
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))
            seeds.append(_mix_shapes(curve, _uniform(n), mix_alpha))

    # Explicit “increasing” families (just reversed versions of decreasing ones)
    # (already covered by the generic reversal step below).

    # Add all deterministic families to the pool
    seeds.extend(families)

    # previous solution, if any (include both orientations)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())
            seeds.append(_reversed(arr))

    # ------------------------------------------------------------------
    # Add reversed versions of everything (except already symmetric cases)
    # ------------------------------------------------------------------
    rev_seeds = [_reversed(v) for v in seeds]
    seeds.extend(rev_seeds)

    # final sanity‑check (ensure non‑negative and ℓ₁‑normalised)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        # clip possible negatives caused by numerical noise
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Optimisers (unchanged, slightly tweaked hyper‑parameters)
# ----------------------------------------------------------------------
def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow/underflow
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,d} best={best_val:.6f}")

    return best_x, best_val


def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    alpha_start: float = 30.0,
    alpha_end: float = 2200.0,
    lr: float = 0.13,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        L(p) = (1/α)·log Σₖ exp(α·(p*p)[k]).
    α grows linearly from ``alpha_start`` to ``alpha_end``.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # gradient of the soft‑max surrogate
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # project back onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 30_000 == 0:
                print(f"[Adam] it={t:,d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _subgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """Additive sub‑gradient descent on the exact objective."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 25_000 == 0:
                print(f"[SubGrad] it={it:,d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.08,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Tiny random transfers + subset reshuffles, accepting worsening moves."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99997, min_temp)

        if rng.random() < 0.55:
            # Reshuffle a random subset
            k_sub = rng.integers(2, 12)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # Tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 150_000 == 0:
                    print(f"[SA] it={it:,d} best={best_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Length of the sequence – chosen to keep FFTs cheap while still giving
    # enough degrees of freedom for a good asymmetry.
    n = 4000
    print(f"\n--- Sequence length: n = {n} ---")

    # ------------------------------------------------------------------
    # Stage 0 – seed generation (including reversed/asymmetric variants)
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds (including reversed versions).")

    # Score seeds with the exact surrogate (they are ℓ₁‑normalised)
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 15
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation (multiple starts)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Adam optimisation (multiple starts) ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # allocate a modest budget per start
            max_iters = int(min(100_000, remaining * 110))
            cand, val = _adam_softmax_optimize(
                seed_vec, n, deadline, max_iters=max_iters,
                alpha_start=30.0, alpha_end=2000.0, lr=0.13,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Quick sub‑gradient runs from the top few seeds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Quick sub‑gradient polishing from top seeds ---")
        for idx, seed_vec in enumerate(best_seeds[:4]):
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            max_iters = int(min(70_000, remaining * 120))
            cand, val = _subgrad_optimize(
                seed_vec, n, deadline, max_iters=max_iters, eta0=0.65
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[SubGrad] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Long‑run additive sub‑gradient descent (single best)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Additive sub‑gradient optimisation (deep) ---")
        remaining = deadline - time.time()
        max_iters = int(min(180_000, remaining * 300))
        best_seq, best_val = _subgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters, eta0=0.7
        )
        print(f"[SubGrad‑deep] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Exponential‑gradient polishing (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(900_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters, eta0=0.20
        )
        print(f"[ExpGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Simulated‑annealing final tweak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing final tweak ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
