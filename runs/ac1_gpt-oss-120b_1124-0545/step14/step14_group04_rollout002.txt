    i = np.arange(n)
    x = (i - (n-1)/2) / ((n-1)/2)
    a = np.sqrt(np.maximum(0, 1 - x**2))
    s = a.sum()
    return a/s if s>0 else _uniform(n)

    coordinates and projects back onto the probability simplex.  The fitness is
    the exact surrogate ``2·n·max(conv(p,p))`` (identical to the official
    ``evaluate_sequence`` after normalisation).  The loop runs until the time
    budget is exhausted.

3️⃣  **Polish** – a short additive sub‑gradient descent on the true objective
    (the `max` of the convolution) starting from the best individual found by
    the evolutionary stage.

The whole routine respects the 1000 s budget, stores the result in the global
``height_sequence_1`` variable for warm‑starts, and finally returns the list
of floats.  It strives to beat the previous bound of 1.5085 and targets the
desired ≤ 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑p = 1):
        2·n·max(conv(p,p)).
    This equals the value returned by ``evaluate_sequence`` after the
    security clamping (because the sum is 1 and the values are already in range).
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised.
    """
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _semicircle(n: int) -> np.ndarray:
    """Discrete semi‑circular window (sqrt(1‑x²) shape)."""
    i = np.arange(n, dtype=np.float64)
    x = (i - (n - 1) / 2.0) / ((n - 1) / 2.0)   # map to [-1,1]
    a = np.sqrt(np.maximum(0.0, 1.0 - x ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed generation (extended)
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.15))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_gaussian(n, 0.35))
    seeds.append(_exponential_decay(n, 0.9999))
    for p in (0.2, 0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    # beta shapes – a richer grid
    for a in (0.3, 0.5, 0.8, 1.0, 2.0):
        for b in (0.3, 0.5, 0.8, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.1, 0.2, 0.4, 0.6, 0.8):
        seeds.append(_tukey_window(n, alpha))
    seeds.append(_semicircle(n))

    # random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.15, 0.35, 0.5, 0.75):
        for nf in (2, 3, 5):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # sanity‑check
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Simple sub‑gradient optimiser (used for final polishing)
# ----------------------------------------------------------------------
def _subgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent on the exact objective
        f(p) = 2·n·max(conv(p,p)).
    Projected onto the probability simplex after each step.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

    return best_x, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Evolutionary + sub‑gradient search for a non‑negative step function
    minimising ``evaluate_sequence``.  The routine respects the given time
    budget and returns the best list of heights found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – use a power‑of‑two length for fast FFTs.
    # ------------------------------------------------------------------
    n = 4096                                   # 2ⁿ, fits comfortably in the budget
    print(f"\n[Info] sequence length = {n}")

    # ------------------------------------------------------------------
    # 0️⃣  Seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"[Seed] generated {len(seeds)} diverse seeds")

    # ------------------------------------------------------------------
    # 1️⃣  Initialise a small population (size ≈ 30)
    # ------------------------------------------------------------------
    pop_size = 30
    elite_sz = 6

    population = []
    scores = []
    for i in range(pop_size):
        if i < len(seeds):
            vec = seeds[i].copy()
        else:
            vec = _random_dirichlet(n, 1.0, rng)
        population.append(vec)
        scores.append(_objective_surrogate(vec))

    # ------------------------------------------------------------------
    # 2️⃣  Evolutionary loop (simple GA)
    # ------------------------------------------------------------------
    gen = 0
    last_print = 0.0
    while time.time() < deadline:
        # sort by fitness (lower is better)
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        best_score = scores[0]
        best_vec = population[0]

        # occasional progress report (every ~30 s)
        now = time.time()
        if now - last_print > 30.0:
            elapsed = now - start_time
            print(f"[Gen {gen}] elapsed={elapsed:.1f}s best={best_score:.6f}")
            last_print = now

        # elite preservation
        new_pop = population[:elite_sz]
        new_scores = scores[:elite_sz]

        # fill the rest of the generation
        while len(new_pop) < pop_size and time.time() < deadline:
            # parent selection – favour elites but keep some diversity
            p1 = population[rng.integers(elite_sz)]
            p2 = population[rng.integers(pop_size)]
            alpha = rng.random()
            child = alpha * p1 + (1.0 - alpha) * p2

            # mutation – random re‑allocation on a small subset
            k_sub = rng.integers(2, 9)                # size of the mutated subset
            idx = rng.choice(n, size=k_sub, replace=False)
            mass = child[idx].sum()
            if mass > 0.0:
                # take a modest fraction and reshuffle it
                epsilon = 0.02 * mass                 # ≈2 % of the subset mass
                child[idx] = np.maximum(child[idx] - epsilon / k_sub, 0.0)
                child[idx] += rng.dirichlet(np.full(k_sub, 1.0)) * epsilon

            # project onto simplex
            child = np.maximum(child, 0.0)
            s = child.sum()
            if s > 0.0:
                child = child / s
            else:
                child = _uniform(n)

            # evaluate
            val = _objective_surrogate(child)

            new_pop.append(child)
            new_scores.append(val)

        population = new_pop
        scores = new_scores
        gen += 1

    # ------------------------------------------------------------------
    # 3️⃣  Choose the best individual discovered
    # ------------------------------------------------------------------
    best_idx = int(np.argmin(scores))
    best_vec = population[best_idx]
    best_val = scores[best_idx]
    print(f"\n[Result] after EA best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing with additive sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # allocate a budget proportional to the remaining time
        max_iters = int(min(200_000, remaining * 300))
        best_vec, best_val = _subgrad_optimize(
            best_vec, n, deadline, max_iters=max_iters, eta0=0.7
        )
        print(f"[Polish] sub‑gradient best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation and storage
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for a possible warm‑start in the next run
    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
