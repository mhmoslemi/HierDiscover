"""
Propose a candidate step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The strategy is a richer, time‑aware search:

* 1️⃣  Generate a large pool of analytic and random seeds (uniform,
  triangular, sine‑window, Gaussian, Beta‑shapes, low‑frequency cosine
  series, Dirichlet draws, …).  All vectors are ℓ₁‑normalised.

* 2️⃣  Keep the best few seeds (default 10) and run a continuation
  L‑BFGS optimiser on a smooth log‑sum‑exp surrogate of the max‑convolution.
  The β‑schedule is aggressive (up to 1·10⁵) and each stage receives a
  modest number of L‑BFGS iterations.

* 3️⃣  Starting from the best L‑BFGS result, perform a long
  Adam‑type mirror‑descent on the *exact* max‑convolution (sub‑gradient
  of the active lag).  The Adam step‑size decays as 1/√t and the
  momentum parameters are the classic (β₁=0.9, β₂=0.999).  This stage
  accounts for the majority of the remaining time budget and typically
  refines the bound by a few ·10⁻³.

* 4️⃣  A final short simulated‑annealing run in logit space gives a
  last local‑search push.

All heavy loops are time‑checked regularly; the function exits
gracefully before the supplied wall‑clock budget expires and stores the
best sequence into the global variable ``height_sequence_1``.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (ℓ₁‑norm = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for an ℓ₁‑normalised vector (sum(seq)==1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random low‑frequency cosine series (non‑negative after clipping).
    """
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet with concentration ``conc``."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    # Initialise logits (log‑softmax parametrisation)
    eps = 1e-30
    v = np.log(start_vec + eps)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on the simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # log‑sum‑exp smoothing (numerically stable)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smoothed max (w.r.t. a)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # ∂/∂a of Σ w_i·conv_i

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # ∂obj/∂logits
            return obj, grad_v

        def callback(_):
            # interrupt if we have exceeded the overall deadline
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Adam optimiser on the *exact* max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_exact_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 2_000_000,
                    base_lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam momentum on the exact max‑convolution.
    Operates in the logit (softmax) space; returns the best vector found
    and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    logits = np.log(a + 1e-30)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # --- exact max‑convolution & its active set gradient ---
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active lag (the factor 2 already accounts
        # for the symmetry of convolution).
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(g, a)
        grad_logit = a * (g - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / (np.sqrt(it) + 1.0)      # decay a little with √t
        logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _softmax(logits)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report (never hurts)
        if it % 200_000 == 0:
            print(f"[Adam] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Simulated‑annealing refinement (logit + softmax)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Simple SA in logit space.  Uses a linear cooling schedule and a
    decreasing proposal width.
    """
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    rng = np.random.default_rng(12345)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-7
        sigma = 0.20 * (1.0 - progress) + 1e-5

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

def _add_candidate(pool, vec, tag):
    """Normalise ``vec`` to ℓ₁=1, compute surrogate, and store."""
    s = vec.sum()
    if s <= 0.0:

    vec = vec / s
    pool.append((vec, _objective_surrogate(vec), tag))

    # random Dirichlet draws (≈300 draws, varying concentration)
    conc_values = np.geomspace(0.02, 5.0, 15)
    for conc in conc_values:
        for _ in range(20):
            if time.time() > deadline - 2.0:
                break
            vec = _random_dirichlet(n, conc, rng)
            _add_candidate(pool, vec, ('dirichlet', conc))

    return pool


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # 0. Load previous best if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and np.isfinite(arr).all() and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None
    # ------------------------------------------------------------------
    # 1. Generate a large, diverse seed pool (time‑aware)
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    pool = _generate_seeds(n, rng, deadline)

    if prev_vec is not None:
        _add_candidate(pool, prev_vec, ('previous',))

    print(f"Collected {len(pool)} raw candidates")

    # ------------------------------------------------------------------
    # 2. Keep the best few (top‑k) for expensive optimisation
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[1])   # sort by surrogate (lower is better)
    top_k = 10
    top_seeds = [vec for vec, _, _ in pool[:top_k]]
    print(f"Retaining top {top_k} seeds for refinement")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS on each seed (aggressive β‑schedule)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before completing all L‑BFGS runs")
            break

        remaining = deadline - time.time()
        # allocate a modest share of remaining budget to this seed
        max_iter_beta = int(min(3000, remaining * 0.07 * 1e5))
        max_iter_beta = max(800, max_iter_beta)

        print(f"\n--- L‑BFGS run #{idx + 1} (max_iter/β={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=max_iter_beta)

        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # --------------------------------------------------------------
        # 4. Adam‑type exact max‑convolution polishing
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            # roughly allocate remaining budget to Adam (up to ~1e6 iters)
            max_adam_iters = int(min(1_000_000, remaining * 0.15 * 1e5))
            max_adam_iters = max(200_000, max_adam_iters)
            print(f"[Adam] polishing with up to {max_adam_iters:,} iterations")
            seq_adam, val_adam = _adam_exact_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_adam_iters)
            if val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing push (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        remaining = deadline - time.time()
        max_sa = int(min(500_000, remaining * 0.08 * 1e5))
        print("\n--- Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                              n,
                                              deadline,
                                              max_iters=max_sa)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. True evaluation with official ``evaluate_sequence``
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
