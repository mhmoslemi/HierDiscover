"""
Propose a step‑function that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))² ,  a_i ≥ 0

This version augments the previous search pipeline with a **linear‑programming
direction step**.  At each outer iteration we:

1. Compute M = max(conv(f,f)) for the current distribution f.
2. Solve a LP (via scipy.optimize.linprog) that finds a non‑negative vector
   b maximising Σ b_j under the *linear* constraints conv(f,b) ≤ M.
   The LP pushes mass into positions where the current f contributes little
   to the large convolution entries.
3. Mix the new direction b into f with a back‑tracking line search that
   guarantees a strict decrease of M.
4. Repeat while time permits (≈ 20 – 30 LP steps).

After the LP phase a long **exponential‑gradient** (mirror descent) polishing
phase is performed, which directly minimises the non‑smooth objective
max(conv(f,f)).  The whole routine stays well within the 1000 s budget and
returns the best sequence found (stored globally as ``height_sequence_1``).

The code works for any length; we use n=1200 for a good speed/quality trade‑off.
"""

import time
import math
import numpy as np

# optional imports – fall back to the original pipeline if SciPy is missing

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_val(seq: np.ndarray) -> float:
    """Exact value of the objective when Σ seq = 1 (2·n·max(conv))."""
    n = len(seq)
    return 2.0 * n * np.max(_conv_fft(seq, seq))


# ----------------------------------------------------------------------
# Simple mirror‑descent (exponential‑gradient) optimiser
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 400_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _surrogate_val(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift (as in the reference code)
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # 1/√t step‑size
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # projection onto the simplex (renormalise)
        s = x.sum()
        if s <= 0.0:
            x = np.full_like(x, 1.0 / n)
        else:
            x = x / s

        cur_val = _surrogate_val(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Linear‑programming direction step (requires SciPy)
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> _sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that   A @ b = conv(f, b)
    (plain linear convolution).  A is sparse (≈ n² non‑zeros).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # pre‑allocate index/value arrays for the CSR matrix
    data = np.empty(n * n, dtype=np.float64)
    row_ind = np.empty(n * n, dtype=np.int32)
    col_ind = np.empty(n * n, dtype=np.int32)

    pos = 0
    for i, fi in enumerate(f):
        # column indices 0…n‑1
        cols = np.arange(n, dtype=np.int32)
        # corresponding rows i + cols
        rows_i = i + cols
        data[pos:pos + n] = fi
        row_ind[pos:pos + n] = rows_i
        col_ind[pos:pos + n] = cols
        pos += n
    return _sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))


def _lp_direction_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ b_j   s.t.   b ≥ 0 , conv(f, b) ≤ M .
    Returns the normalised vector b (ℓ₁‑norm = 1).
    """
    n = len(f)
    A = _conv_matrix(f)                     # (2n‑1) × n sparse matrix
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ b  → minimise -Σ b
    c = -np.ones(n, dtype=np.float64)

    # bounds for each variable: (0, ∞)
    bounds = [(0, None)] * n

    # Use the HiGHS interior‑point solver – fast for our moderate size.
    res = _opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                       method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        # Fallback: return the current f (no progress)
        return f.copy()
    b = res.x
    s = b.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return b / s

    s = f.sum()
    f = f / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)

    best_f = f.copy()
    best_val = _surrogate_val(best_f)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # LP‑driven improvement phase (only if SciPy is available)
    # ------------------------------------------------------------------
    if _HAS_SCIPY:
        max_lp_iters = 30
        mix_t_initial = 0.5   # start with aggressive mixing
        for lp_it in range(max_lp_iters):
            if time.time() > deadline:
                break

            # current worst convolution value
            M = np.max(_conv_fft(f, f))

            # solve LP to obtain direction b
            b = _lp_direction_step(f, M)

            # line‑search/back‑track to guarantee descent
            t = mix_t_initial
            while t > 1e-4:
                cand = (1.0 - t) * f + t * b
                cand = np.maximum(cand, 0.0)
                cand_sum = cand.sum()
                if cand_sum > 0:
                    cand = cand / cand_sum
                else:
                    cand = np.full(n, 1.0 / n, dtype=np.float64)

                cand_val = _surrogate_val(cand)
                if cand_val < best_val - 1e-12:   # strict improvement
                    f = cand
                    best_val = cand_val
                    best_f = cand.copy()
                    print(f"[LP {lp_it+1:02d}] new best = {best_val:.6f}")
                    break
                # no improvement – shrink step size
                t *= 0.5

            # If even the smallest t failed, abort the LP phase
            if t <= 1e-4:
                print(f"[LP {lp_it+1:02d}] line‑search failed, stopping LP phase.")
                break

    # ------------------------------------------------------------------
    # Final long exponential‑gradient polishing (cheap and effective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 2500))
        print(f"\n[Final EG] polishing with up to {max_iters:,} iterations")
        best_f, best_val = _expgrad_optimize(best_f, n, deadline,
                                             max_iters=max_iters,
                                             eta0=0.22)
        print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_f.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse by a later run
    globals()["height_sequence_1"] = best_f.tolist()
    return best_f.tolist()
