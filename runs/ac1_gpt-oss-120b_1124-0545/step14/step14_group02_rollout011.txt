Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence`` (i.e. 2·n·max(conv(a,a))/(sum(a)²)).
The algorithm consists of:

0.  Small deterministic seed pool + a large random sweep over several
    parametric families (uniform, triangular, sine‑window, Gaussian,
    Beta, exponential‑decay, power‑law, mixtures, low‑frequency random
    curves, Dirichlet draws).  All vectors are ℓ₁‑normalised.

1.  Cheap surrogate evaluation (2·n·max(conv)) to keep the best
    ``top_k`` seeds.

2.  Continuation L‑BFGS on the smooth log‑sum‑exp surrogate of the max
    convolution.  A schedule of increasing β (smoothness) drives the
    solution toward the true max‑convolution.  The optimisation lives on
    the probability simplex using a soft‑max over logits.

3.  Random‑restart polishing: perturb the logits of the current best
    solution with Gaussian noise and run a short L‑BFGS again.

4.  Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the
    true non‑smooth objective for a large number of iterations.

5.  A final L‑BFGS “polish” using the high‑β surrogate.

6.  Optional up‑sampling (double the grid size) and repeat steps 2‑5
    on the finer grid if time permits.

The whole search respects the given time budget (≈1000 s) and keeps
track of the best vector discovered.  The final vector is stored in the
global ``height_sequence_1`` variable for possible reuse.
"""

# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p  (increasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30  # avoid exact zero before normalisation
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)

    return a / a.sum()

# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule=None,
    max_iter_per_beta: int = 5000,
) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0,
            5.0,
            20.0,
            80.0,
            200.0,
            500.0,
            1000.0,
            3000.0,
            5000.0,
            10000.0,
            20000.0,
        ]

    v = np.log(start_vec + 1e-30)  # initial logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                  # numerical shift for stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-13, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            # time out inside the optimiser
            break
        except Exception as e:
            # any other failure – abort remaining β‑steps for this seed
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)

def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600_000,
) -> np.ndarray:

    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
    objective.  Returns the best vector seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t   (a slightly larger constant can be tried)
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq

# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)

        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()

                max_exp_fine = int(min(800_000, rem * 0.15 * 1e5))
                if max_exp_fine > 0:
                    print("\n--- Final exponentiated‑gradient on fine grid ---")
                    seq_fine = _exp_gradient_optimize(
                        global_best_seq, n, deadline, max_iters=max_exp_fine
                    )
                    val_fine = _objective_surrogate(seq_fine)
                    if val_fine < global_best_val:
                        global_best_seq = seq_fine.copy()
                        global_best_val = val_fine
                        print(f"[Fine‑ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
