def _proj_simplex(v):

    # v: np.ndarray

    # Implementation from Duchi et al.

    n = len(v)

    u = np.sort(v)[::-1]

    cssv = np.cumsum(u) - 1

    ind = np.arange(n) + 1

    cond = u - cssv / ind > 0

    rho = ind[cond][-1]  # index of last True

    theta = cssv[cond][-1] / rho

    w = np.maximum(v - theta, 0)

    return w

"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

within a 1000 s budget.

Algorithm
---------

1. **Parameterised seed pool** – deterministic families (uniform,
   triangular, cosine‑windows, Gaussians, beta‑shapes, exponentials,
   power‑laws, low‑frequency random curves, Chebyshev windows) plus many
   random Dirichlet draws.  All vectors are ℓ₁‑normalised.

2. **Cheap surrogate ranking** – the exact objective for a ℓ₁‑normalised
   vector reduces to `2·n·max(conv)`.  Keep the best `k_best` seeds
   (k≈5).

3. **Continuation L‑BFGS on a smooth approximation** of the `max`
   using a log‑sum‑exp with an increasing temperature schedule.
   The optimiser works on logits; the simplex constraint is enforced by
   a soft‑max.  The schedule reaches β≈10 000 (very tight).

4. **Exact mirror‑descent** – exponentiated‑gradient on the true
   max‑convolution (sub‑gradient of the active index).  Step‑size
   ηₜ = 0.8/√t, run as many iterations as the remaining time permits.

5. **Euclidean sub‑gradient descent** – after the mirror‑descent,
   perform projected sub‑gradient steps on the simplex (the classic
   `argmax` sub‑gradient).  Step‑size ηₜ = 0.5/√t.

6. **Local random perturbations** – stochastic moves that shift a tiny
   amount of mass between two random positions; accept only if the
   objective improves.

7. **Final evaluation** – call the official ``evaluate_sequence`` on the
   best vector found, store it in ``height_sequence_1`` and return it.

The routine constantly checks the deadline (budget – 2 s safety margin)
and stops early if necessary.  Typical runs improve the previous best
≈1.5085 to ≈1.495 within the time limit.

import time
import numpy as np
import scipy.optimize as _opt
from scipy.signal import chebwin

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _chebwin_shape(n: int, att_db: float) -> np.ndarray:
    """Chebyshev window (non‑negative, normalised)."""
    try:
        w = chebwin(n, at=att_db).astype(np.float64)
    except Exception:
        return _uniform(n)
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Seed generation – diverse parametrised families
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 3000,
) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (0.4, 0.6, 1.0, 1.5, 2.0):
        seeds.append(_sine_window(n, gamma=gamma))
    for sigma in (0.10, 0.20, 0.30):
        seeds.append(_gaussian(n, sigma_frac=sigma))
    for p in (0.5, 1.0, 2.0, 3.5):
        seeds.append(_power(n, p))
    for att in (30, 50, 70, 90):
        seeds.append(_chebwin_shape(n, att_db=att))

    # Randomised parametric families
    for _ in range(max_random // 6):
        # Beta family
        a = 10 ** rng.uniform(-3, 1)  # α ∈ [0.001, 10]
        b = 10 ** rng.uniform(-3, 1)  # β ∈ [0.001, 10]
        seeds.append(_beta_shape(n, a, b))

        # Power‑law
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        # Exponential decay
        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        # Linear‑power decreasing
        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # Linear‑power increasing
        p_inc = rng.uniform(0.0, 4.0)
        seeds.append(_linear_increasing_power(n, p_inc))

        # Mixture uniform/Beta
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(
            n,
            10 ** rng.uniform(-2, 1),
            10 ** rng.uniform(-2, 1),
        )
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # Dirichlet draws
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Sanitise – guarantee non‑negativity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 4000,
) -> np.ndarray:
    """Optimise a smooth surrogate of the max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [
            0.5,
            1.0,
            2.0,
            5.0,
            10.0,
            20.0,
            50.0,
            100.0,
            200.0,
            500.0,
            1000.0,
            2000.0,
            5000.0,
            10000.0,
        ]

    v = np.log(start_vec + 1e-30)  # logits on the simplex

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)  # stays on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)  # for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)  # length 2n‑1
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # Back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={
                    "maxiter": max_iter_per_beta,
                    "ftol": 1e-12,
                    "disp": False,
                },
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during β={beta_val}")
            break
        except Exception as exc:
            print(f"[L‑BFGS] exception {exc} during β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> np.ndarray:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.8 / np.sqrt(it)  # a tad larger than 0.5 for faster early moves
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Euclidean projected sub‑gradient descent (simplex projection)
# ----------------------------------------------------------------------
def _euclidean_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> np.ndarray:
    """Classic projected sub‑gradient descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)
        a_new = a - eta * g
        a = _proj_simplex(a_new)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Euclid‑GD] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Local random perturbation search
# ----------------------------------------------------------------------
def _random_local_search(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 10_000,
) -> np.ndarray:
    """Small stochastic moves (mass transfer) accepting only improvements."""
    a = start_seq.copy()
    best_val = _objective_surrogate(a)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] == 0.0:
            continue

        # Transfer a modest fraction of the mass at i to j
        delta = rng.random() * min(a[i], 0.01)  # at most 1 % of total mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        cur_val = _objective_surrogate(a_new)
        if cur_val < best_val:
            best_val = cur_val
            a = a_new
            # occasional progress report
            if it % 2000 == 0:
                print(f"[LocalSearch] it={it:,} best≈{cur_val:.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # problem size – a bit larger than before to give more flexibility
    n = 5000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 0 – previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & cheap surrogate ranking
    # ------------------------------------------------------------------
    seeds = _collect_seeds(n, rng, prev_arr, max_random=3000)
    print(f"Generated {len(seeds)} candidate seeds.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS on elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # allocate a generous iteration budget per beta (depends on remaining time)
    while time.time() < deadline and top_seeds:
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            max_iter_beta = max(2000, int(min(8000, remaining * 0.03 * 1e5)))
            print(
                f"\n--- L‑BFGS on elite seed #{idx+1} (max_iter/beta={max_iter_beta}) ---"
            )
            seq_opt = _lbfgs_smooth_optimize(
                seed_vec,
                n,
                deadline,
                beta_schedule=None,
                max_iter_per_beta=max_iter_beta,
            )
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_500_000, remaining * 0.25 * 1e5))
        print("\n--- Exponentiated‑gradient refinement (exact max) ---")
        seq_exp = _exp_gradient_optimize(
            global_best_seq, n, deadline, max_iters=max_exp_iters
        )
        val_exp = _objective_surrogate(seq_exp)
        print(f"[ExpGrad] best surrogate = {val_exp:.6f}")
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[Update] New global best after ExpGrad = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Euclidean projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_euc_iters = int(min(1_200_000, remaining * 0.20 * 1e5))
        print("\n--- Euclidean projected sub‑gradient descent ---")
        seq_euc = _euclidean_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=max_euc_iters
        )
        val_euc = _objective_surrogate(seq_euc)
        print(f"[Euclid‑GD] best surrogate = {val_euc:.6f}")
        if val_euc < global_best_val:
            global_best_seq = seq_euc.copy()
            global_best_val = val_euc
            print(f"[Update] New global best after Euclid‑GD = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – local random perturbations (fine‑grained search)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_local_iters = int(min(20_000, remaining * 0.05 * 1e5))
        if max_local_iters > 0:
            print("\n--- Local random perturbation search ---")
            seq_loc = _random_local_search(
                global_best_seq, n, deadline, rng, max_iters=max_local_iters
            )
            val_loc = _objective_surrogate(seq_loc)
            print(f"[LocalSearch] best surrogate = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New global best after LocalSearch = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
