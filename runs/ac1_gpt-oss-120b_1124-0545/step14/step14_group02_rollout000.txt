def adam_optimize(start_vec, n, deadline, temp_schedule, max_iters_per_temp):
   logs = np.log(start_vec + 1e-30)
   m = np.zeros_like(logs); v = np.zeros_like(logs)
   beta1=0.9; beta2=0.999; eps=1e-8
   t=0
   for T in temp_schedule:
       for _ in range(max_iters_per_temp):
           if time.time()>deadline: break outer loops
           t+=1
           a = softmax(logs)
           conv = conv_fft(a,a)
           shift = np.max(conv)
           exp_terms = np.exp((conv - shift)/T)   # sort of using temperature: T = 1/beta
           # Actually we want smoothed max = shift + T * log(sum(exp((conv - shift)/T)))
           sum_exp = exp_terms.sum()
           obj = 2.0 * n * (shift + T * np.log(sum_exp))
           
           # gradient w.r.t a
           w = exp_terms / sum_exp   # weights
           rev_a = a[::-1]
           full = conv_fft(rev_a, w)
           grad_a = 2.0 * full[n-1:n-1+n]   # same as earlier.
           
           # backprop through softmax: grad_v = a * (grad_a - np.dot(grad_a, a))
           grad_v = a * (grad_a - np.dot(grad_a, a))
           
           # Adam update logs
           m = beta1 * m + (1 - beta1) * grad_v
           v = beta2 * v + (1 - beta2) * (grad_v**2)
           m_hat = m / (1 - beta1**t)
           v_hat = v / (1 - beta2**t)
           logs -= 0.01 * m_hat / (np.sqrt(v_hat) + eps)  # learning rate 0.01 maybe
   return softmax(logs)

"""
Fast multi‑phase optimizer for the step‑function evaluation problem.

Algorithm
---------
* **Seed generation** – a rich pool of deterministic and random parametric
  families (uniform, triangular, sine windows, Gaussians, power‐laws,
  Beta‑shapes, exponential decay, Dirichlet draws, low‑frequency curves).
  All vectors are ℓ₁‑normalised and non‑negative.

* **Cheap surrogate ranking** – evaluate the surrogate objective
  `2·n·max(conv(a,a))` (the denominator is 1 because the vectors are ℓ₁‑normalised)
  and keep the best few seeds.

* **Smooth optimisation** – for each elite seed run an Adam optimiser on a
  *log‑sum‑exp* smooth approximation of the max‑convolution.  The temperature
  schedule `[5, 2, 1, 0.5, 0.2, 0.1, 0.05]` gradually tightens the surrogate.
  Adam’s adaptive learning‑rate gives reliable progress with a modest number
  of gradient evaluations.

* **Exact mirror‑descent** – starting from the best smoothed vector perform a
  long exponentiated‑gradient (mirror descent) on the exact max‑convolution.
  Step‑size `ηₜ = 0.5 / √t` (per‑iteration bias‑corrected gradient).

* **Hill‑climbing perturbations** – with the remaining budget repeatedly move a
  small amount of mass from one index to another, accept only improving
  moves, and slowly decay the perturbation size.  This stochastic local search
  can escape shallow basins left by the gradient steps.

* **Optional random restarts** – a few noisy copies of the current best vector
  are fed again through the Adam stage to explore nearby basins.

* **Final evaluation** – the best vector is evaluated with the official
  `evaluate_sequence` and stored in the global name `height_sequence_1`.

The whole pipeline runs under the given time budget (`budget_s`), constantly
checking the deadline and cutting short any phase when time runs out.
"""

import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, always returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 3000,
) -> list[np.ndarray]:

    seeds: list[np.ndarray] = []

    # deterministic families (always useful)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))
    seeds.append(_exp_decay(n, lam=0.001))
    seeds.append(_exp_decay(n, lam=0.005))

    # random parametric families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-3, 1)  # α ∈ [1e‑3,10]
        b = 10 ** rng.uniform(-3, 1)  # β ∈ [1e‑3,10]
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(
            n,
            10 ** rng.uniform(-2, 1),
            10 ** rng.uniform(-2, 1),
        )
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee positivity & ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned

    eps = 1e-8
    t = 0

    for T in temp_schedule:
        if time.time() > deadline:
            break
        # Gradients become sharper as T ↓; we keep a modest max_iter per temperature.
        for _ in range(max_iters_per_temp):
            if time.time() > deadline:
                return _softmax(v)

            t += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_val = np.max(conv)

            # we use the shifted log‑sum‑exp to avoid overflow
            exp_vals = np.exp((conv - max_val) / T)
            sum_exp = exp_vals.sum()
            w = exp_vals / sum_exp  # soft weights for the smoothed max

            # gradient w.r.t a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # back‑prop through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v2 / (1.0 - beta2 ** t)

            # learning rate – a small constant works well for this problem
            lr = 0.02
            v -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror descent) on the *exact* max‑convolution.
    The step‑size ηₜ = 0.5 / √t as in the reference implementation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress reports
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Small stochastic hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(
    seq: np.ndarray,
    n: int,
    deadline: float,
    steps: int = 5000,
    eps_start: float = 0.02,
    eps_end: float = 0.0005,
    rng: np.random.Generator | None = None,
) -> np.ndarray:
    """
    Simple random mass‑transfer local search.
    At each step move a tiny amount of probability mass from a random i to a random j.
    Accept only if the surrogate decreases.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = seq.copy()
    best_val = _objective_surrogate(a)

    for k in range(steps):
        if time.time() > deadline:
            break
        # linear schedule for epsilon
        eps = eps_start * (1 - k / steps) + eps_end * (k / steps)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        shift = min(eps, a[i])
        if shift <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= shift
        a_new[j] += shift
        # renormalise (numerical drift)
        a_new = a_new / a_new.sum()

        val_new = _objective_surrogate(a_new)
        if val_new < best_val:
            a = a_new
            best_val = val_new
            if (k + 1) % 1000 == 0:
                print(f"[HillClimb] step {k+1} improved to {best_val:.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 123,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found (list of floats).  The global variable
    `height_sequence_1` is updated to allow reuse in subsequent calls.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size (choose a size that gives a good trade‑off between quality
    # and speed)
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 1 – seed generation + cheap surrogate ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print("[Stage 1] Generating seeds …")
    seeds = _collect_seeds(n, rng, prev_arr, max_random=3500)
    print(f"  generated {len(seeds)} seeds")

    # cheap surrogate evaluation
    evals = []
    for vec in seeds:
        if time.time() > deadline:
            break
        evals.append((_objective_surrogate(vec), vec))
    evals.sort(key=lambda x: x[0])
    top_k = min(20, len(evals))
    top_seeds = [vec for _, vec in evals[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate from seeds = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam smooth optimisation on elite seeds
    # ------------------------------------------------------------------
    print("[Stage 2] Adam smoothing on elite seeds …")
    temp_sched = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    # allocate iterations per temperature based on remaining time
    remaining = deadline - time.time()
    iters_per_temp = max(800, int(min(2500, remaining * 4)))  # very rough heuristic

    for idx, seed_vec in enumerate(top_seeds[:10]):
        if time.time() > deadline:
            break
        print(f"  Adam optimisation for seed #{idx+1}")
        seq_opt = _adam_smooth_optimize(
            seed_vec,
            n,
            deadline,
            temp_schedule=temp_sched,
            max_iters_per_temp=iters_per_temp,
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"    surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"    ** new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2b – random noisy restarts (few)
    # ------------------------------------------------------------------
    print("[Stage 2b] Random noisy restarts")
    logits = np.log(best_seq + 1e-30)
    for r in range(5):
        if time.time() > deadline:
            break
        noisy_logits = logits + rng.normal(scale=0.15, size=n)
        noisy_start = _softmax(noisy_logits)
        seq_opt = _adam_smooth_optimize(
            noisy_start,
            n,
            deadline,
            temp_schedule=temp_sched,
            max_iters_per_temp=int(iters_per_temp * 0.8),
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"    restart #{r+1} surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"    ** improved via restart to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact mirror‑descent (exponentiated‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_000_000, remaining * 2000))  # ~2k iters per sec
        print(f"[Stage 3] Exponentiated‑gradient for up to {max_exp_iters:,} iters …")
        seq_md = _exp_gradient_optimize(best_seq, n, deadline, max_iters=max_exp_iters)
        val_md = _objective_surrogate(seq_md)
        print(f"    mirror‑descent surrogate = {val_md:.6f}")
        if val_md < best_val:
            best_seq = seq_md.copy()
            best_val = val_md
            print(f"    ** new best after mirror‑descent = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – stochastic hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        steps = int(min(8000, remaining * 8))  # about a few thousand steps
        print(f"[Stage 4] Hill‑climbing with {steps} steps …")
        seq_hc = _hill_climb(best_seq, n, deadline, steps=steps, rng=rng)
        val_hc = _objective_surrogate(seq_hc)
        print(f"    hill‑climb surrogate = {val_hc:.6f}")
        if val_hc < best_val:
            best_seq = seq_hc.copy()
            best_val = val_hc
            print(f"    ** improved after hill‑climb to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
