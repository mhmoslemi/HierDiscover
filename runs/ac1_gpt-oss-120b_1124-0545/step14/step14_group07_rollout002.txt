"""
Non‑negative step‑function optimiser.

The objective that we need to minimise is

    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the full linear convolution.
Because the evaluation routine clips values to [0,1000] the objective is
scale‑invariant, so we may restrict ourselves to probability vectors
(sum(a)=1).  In that case the objective simplifies to the surrogate

    S(a) = 2·n·max(conv(a,a))

which we can evaluate extremely fast with an FFT.

Algorithm
---------

1. **Very diverse seed pool.**  Apart from the deterministic windows used
   before we now sweep a large grid of parametric families (Beta,
   power‑law, Gaussian, sine‑window, Tukey) and keep the best few according
   to the exact surrogate S(a).  Random Dirichlet draws and a few low‑
   frequency curves are also added.

2. **Smooth‑max L‑BFGS‑B.**  Starting from the best seed we minimise the
   smooth approximation

        Fₚ(a) = (1/p)·log Σₖ exp(p·convₖ(a,a))

   using SciPy's L‑BFGS‑B on the logits y (a = softmax(y)).  The smoothing
   parameter p is increased geometrically (5 → 15 → 45 → 135 → 405 → …)
   and for each p we run a generous number of L‑BFGS iterations
   (≈800).  After each p we also try a short random‑restart.

3. **Fine‑grained exponentiated‑gradient polishing.**  The exact sub‑gradient
   of the (non‑smooth) surrogate is cheap to compute; we perform many
   multiplicative EG updates with a slowly decaying stepsize and keep the
   best vector seen.

4. **Final evaluation** with the official ``evaluate_sequence`` function.
   The best sequence is stored in the global name ``height_sequence_1`` for
   possible reuse by later runs.

The whole pipeline respects a 1000 s wall‑clock budget and prints a short
progress log.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate (sum(a)=1) : 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 150_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size decays like 1/√t
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        # clip to avoid under‑/overflow
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # Renormalise onto the simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of the smooth surrogate
# ----------------------------------------------------------------------
def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """Smooth surrogate and its gradient wrt logits y."""
    a = _softmax(y)                     # a lies on the simplex
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = conv.max()

    # soft‑max over the p‑scaled shifted values (numerically stable)
    shifted = conv - max_c               # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)  # safety against underflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                    # fallback when everything underflows
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    # gradient wrt a (see derivation in the analysis)
    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)     # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

    # chain‑rule to obtain gradient wrt logits (soft‑max Jacobian)
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    # smooth surrogate value: (1/p)·log Σₖ exp(p·cₖ)
    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a: np.ndarray, n: int, deadline: float,
                     p: float = 20.0, max_iter: int = 800) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the smooth surrogate with logits parametrisation.
    Returns the best vector and its *exact* surrogate value.
    """
    y0 = np.log(np.maximum(start_a, 1e-15))
    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})
    a_opt = _softmax(res.x)
    val = _objective_surrogate(a_opt)
    return a_opt, val


# ----------------------------------------------------------------------
# Seed generation (large, systematic parameter sweep)
# ----------------------------------------------------------------------
def _collect_rich_seeds(n: int, rng: np.random.Generator,
                        previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse seed pool (full sweep of several parametric families)."""
    seeds: list[np.ndarray] = []

    # deterministic families without parameters
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.9995))

    # --------------------------------------------------------------
    # Parameter sweeps – keep the grid moderately fine but not huge.
    # --------------------------------------------------------------
    # Beta shapes
    alphas = np.linspace(0.2, 4.8, 12)   # 12 values
    betas = np.linspace(0.2, 4.8, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Power‑law
    for p in np.linspace(0.1, 5.0, 20):
        seeds.append(_power(n, p))

    # Gaussian windows
    for sigma in np.linspace(0.05, 0.6, 12):
        seeds.append(_gaussian(n, sigma))

    # Sine windows
    for gamma in np.linspace(0.5, 5.0, 10):
        seeds.append(_sine_window(n, gamma))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # --------------------------------------------------------------
    # Randomised families
    # --------------------------------------------------------------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    for nf in (2, 3, 4):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform and low‑frequency curves (adds a mild flat bias)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Keep the previous solution if supplied
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # Final normalisation (guard against accidental zeros)
    out = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        out.append(v / s if s > 0 else _uniform(n))
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for final evaluation/printing
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a problem size.
    # ------------------------------------------------------------------
    n = 4000                     # a good trade‑off between expressive power & speed
    print(f"\n=== Problem size ===\nLength of the sequence: n = {n}")

    # ------------------------------------------------------------------
    # Stage 0 – generate a very diverse seed pool.
    # ------------------------------------------------------------------
    print("\n=== Seed generation ===")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_rich_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    # Evaluate surrogate for every seed – keep the top‑k only.
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – quick EG refinement on the best few seeds.
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n=== Quick EG refinement ===")
        for idx, seed_vec in enumerate(best_seeds[:5]):      # at most five seeds
            if time.time() >= deadline:
                break
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=180_000, eta0=0.28)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[EG] seed #{idx+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS‑B with a progressive smoothing schedule.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== L‑BFGS‑B optimisation (smooth max) ===")
        # geometric schedule, ending at a fairly large p
        p_schedule = [5.0, 15.0, 45.0, 135.0, 405.0, 1215.0, 3645.0]
        for p in p_schedule:
            if time.time() >= deadline:
                break
            # run a relatively long L‑BFGS optimisation
            cand, val = _lbfgsb_optimize(best_seq, n, deadline,
                                         p=p, max_iter=800)
            print(f"[L‑BFGS] p={p:6.1f} → surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"   ** new global best = {best_val:.6f}")

            # one modest random restart per p value
            if time.time() < deadline:
                noise = rng.normal(scale=0.25, size=n)
                y_noise = np.log(np.maximum(best_seq, 1e-15)) + noise
                perturbed = _softmax(y_noise)
                cand2, val2 = _lbfgsb_optimize(perturbed, n, deadline,
                                               p=p, max_iter=600)
                if val2 < best_val:
                    best_val, best_seq = val2, cand2
                    print(f"[Restart] p={p:.1f} → improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final exponentiated‑gradient polishing (lots of cheap iters).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Final EG polishing ===")
        remaining = deadline - time.time()
        # Rough estimate: 2000 updates per second on a typical CPU
        max_iters = int(min(300_000, remaining * 2000))
        if max_iters > 0:
            cand, val = _expgrad_optimize(best_seq, n, deadline,
                                          max_iters=max_iters, eta0=0.25)
            print(f"[Final EG] surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand

    # ------------------------------------------------------------------
    # Official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store globally for possible reuse in later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
