"""
Propose a non‑negative step‑function (a probability vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

within the allocated time budget (≈1000 s).

Algorithm
---------

1. **Seed generation** – a rich pool of ℓ₁‑normalised vectors coming from
   deterministic families (uniform, triangular, Gaussian, beta, power‑law,
   windows) and a large random sweep over parametric mixtures
   (mixture of Gaussians, mixture of Betas, Dirichlet draws, low‑frequency
   random curves).  The best few seeds are kept.

2. **Continuation L‑BFGS** – each elite seed is refined by minimising a
   smooth surrogate of the max‑convolution (log‑sum‑exp) with a schedule
   of increasing smoothness parameters (β).  The optimisation works on the
   logits of a soft‑max parametrisation, guaranteeing positivity and
   ℓ₁‑normalisation.

3. **Adam on the exact non‑smooth objective** – starting from the best L‑BFGS
   result we run a long exponentiated‑gradient (mirror‑descent) loop,
   but the update is performed with the Adam optimiser on the logits.
   The sub‑gradient is obtained from the active convolution index.

4. **Simulated annealing (SA)** – a temperature‑controlled random walk
   over the logits (Gaussian perturbations) is performed to escape possible
   local minima.  The SA schedule is very aggressive at the beginning and
   cools down logarithmically.

5. **Optional up‑sampling** – if a generous amount of time (>30 s) is left,
   the current best vector is linearly interpolated to a finer grid
   (doubling `n`) and steps 2‑4 are repeated on the finer resolution.
   Finer grids allow a lower continuous‑limit objective.

6. **Return** – the best sequence found (as a plain list) is stored in the
   global variable ``height_sequence_1`` and returned.

The whole routine constantly checks the remaining time and aborts
gracefully, guaranteeing that a valid sequence is always produced before
the time budget expires.

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like polynomial shape."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p  (increasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _gaussian_mixture(n: int, rng: np.random.Generator,
                      ncomp: int = 3) -> np.ndarray:
    """Random mixture of a few Gaussian bumps."""
    weights = rng.dirichlet(np.ones(ncomp))
    components = []
    i = np.arange(n, dtype=np.float64)
    for _ in range(ncomp):
        # random centre anywhere in the grid
        centre = rng.uniform(0, n - 1)
        sigma_frac = rng.uniform(0.02, 0.12)   # relatively narrow
        sigma = sigma_frac * n
        comp = np.exp(-((i - centre) ** 2) / (2.0 * sigma ** 2))
        components.append(comp)
    mixed = sum(w * c for w, c in zip(weights, components))
    mixed = np.maximum(mixed, 0.0)
    if mixed.sum() == 0.0:
        return _uniform(n)
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Seed generation – diverse parametrised families
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 5000) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random parametric families
    for _ in range(max_random // 6):
        # random Beta
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        # random power‑law
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        # random exponential decay
        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        # random linear‑power (decreasing)
        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # mixture of Beta and Uniform
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

        # Gaussian mixture (3 components)
        seeds.append(_gaussian_mixture(n, rng, ncomp=3))

    # Dirichlet draws with various concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise – enforce non‑negativity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray, n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 5000.0]

    v = np.log(start_vec + 1e-30)          # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_loc):
            a = _softmax(v_loc)
            conv = _conv_fft(a, a)

            # log‑sum‑exp approximation of max
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_loc):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit hit (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser on the exact (non‑smooth) max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray, n: int,
                   deadline: float,
                   max_iters: int = 400_000,
                   lr: float = 0.30,
                   beta1: float = 0.90,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimiser on logits for the exact objective.
    Works on the ℓ₁‑simplex via soft‑max.
    """
    v = np.log(start_seq + 1e-30)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    best_a = start_seq.copy()
    best_val = _objective_surrogate(best_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        a = _softmax(v)

        # compute active convolution index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # exact sub‑gradient of max(conv) at active index k
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[k] = 1.0
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # back‑prop to logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam updates (bias‑corrected)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_sq = beta2 * v_sq + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v_sq / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        v = v - step

        # periodic evaluation (once per 2 000 iters)
        if t % 2000 == 0:
            cur_val = _objective_surrogate(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_a = _softmax(v).copy()
    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (random walk on logits)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray, n: int,
                        deadline: float,
                        max_iters: int = 200_000,
                        init_sigma: float = 0.12,
                        final_sigma: float = 0.001,
                        init_temp: float = 5e-3,
                        final_temp: float = 1e-6,
                        rng: np.random.Generator = None) -> np.ndarray:
    """
    Simple SA on the logits.  The temperature and proposal scale decay
    logarithmically.
    """
    if rng is None:
        rng = np.random.default_rng()

    v_cur = np.log(start_seq + 1e-30)
    best_v = v_cur.copy()
    best_a = _softmax(v_cur)
    best_val = _objective_surrogate(best_a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # logarithmic schedules
        frac = it / max_iters
        sigma = init_sigma * (final_sigma / init_sigma) ** frac
        temp = init_temp * (final_temp / init_temp) ** frac

        v_new = v_cur + sigma * rng.normal(size=n)
        a_new = _softmax(v_new)
        val_new = _objective_surrogate(a_new)

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            best_v = v_new.copy()
            v_cur = v_new
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(val_new - best_val) / max(temp, 1e-12)):
                v_cur = v_new

        # occasional progress report
        if it % 10000 == 0:
            print(f"[SA] it={it:,}, best≈{best_val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of Python ``float``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin to guarantee we can always call ``evaluate_sequence`` at the end
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Stage 0 – problem size (initial grid)
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search – n={n} ===")

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & cheap surrogate ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=5000)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values (top {top_k} seeds):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS on elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Current best surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 3000.0, 5000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before processing all elite seeds.")
            break
        remaining = deadline - time.time()
        # allocate a modest part of the remaining budget
        max_iter_beta = max(1500, int(min(8000, remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS on elite seed #{idx + 1} (max_iter/beta={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Adam optimisation on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_adam_iters = int(min(400_000, remaining * 0.25 * 1e5))
        if max_adam_iters > 0:
            print("\n--- Adam optimisation on exact max (sub‑gradient) ---")
            seq_adam = _adam_optimize(global_best_seq,
                                      n,
                                      deadline,
                                      max_iters=max_adam_iters,
                                      lr=0.30)
            val_adam = _objective_surrogate(seq_adam)
            if val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa_iters = int(min(200_000, remaining * 0.25 * 1e5))
        if max_sa_iters > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa = _simulated_annealing(global_best_seq,
                                          n,
                                          deadline,
                                          max_iters=max_sa_iters,
                                          rng=rng)
            val_sa = _objective_surrogate(seq_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – optional up‑sampling to a finer grid
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 35.0:
            new_n = n * 2
            print(f"\n=== Upsampling: n={new_n} ===")
            up_seq = _upsample(global_best_seq, new_n)

            # quick L‑BFGS on the finer grid (few iterations, just to get a good start)
            up_seq_opt = _lbfgs_smooth_optimize(up_seq,
                                                new_n,
                                                deadline,
                                                beta_schedule=beta_sched,
                                                max_iter_per_beta=2500)

            # Adam on the finer grid
            remaining = deadline - time.time()
            max_adam_fine = int(min(150_000, remaining * 0.20 * 1e5))
            if max_adam_fine > 0:
                seq_fine_adam = _adam_optimize(up_seq_opt,
                                               new_n,
                                               deadline,
                                               max_iters=max_adam_fine,
                                               lr=0.25)
                up_seq_opt = seq_fine_adam

            # final SA on fine grid (very short)
            remaining = deadline - time.time()
            max_sa_fine = int(min(80_000, remaining * 0.15 * 1e5))
            if max_sa_fine > 0:
                seq_fine_sa = _simulated_annealing(up_seq_opt,
                                                   new_n,
                                                   deadline,
                                                   max_iters=max_sa_fine,
                                                   rng=rng)
                up_seq_opt = seq_fine_sa

            val_up = _objective_surrogate(up_seq_opt)
            print(f"[Upsample] final surrogate = {val_up:.6f}")

            if val_up < global_best_val:
                global_best_seq = up_seq_opt.copy()
                global_best_val = val_up
                n = new_n
                print(f"[Update] new best after up‑sampling = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
