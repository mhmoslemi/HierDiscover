"""
Propose a step‑function (non‑negative heights) that minimizes the official
``evaluate_sequence`` objective.

The algorithm is a heavier‑duty version of the previous approach:

*   Use a longer sequence (n=8000).  The FFT size grows only modestly,
    so each iteration is still cheap.
*   Generate a much richer pool of seeds (analytic windows, many
    Dirichlet draws, random mixtures of a few Gaussians, sparse supports,
    and the previous best if available).
*   Run a soft‑max Adam optimiser on the smooth surrogate
    ``L_α(p)= (1/α)·log∑exp(α·(p∗p)[k])`` with a *slow* temperature schedule
    (α from 10 up to 8000) and many more iterations than before.
*   Refine the best candidate with an exponential‑gradient (mirror‑descent)
    update that works directly on the true ``max(p∗p)`` sub‑gradient.
*   Finish with a long simulated‑annealing pass that reshuffles mass on
    random subsets and occasionally transfers a tiny amount between two
    positions.
*   Throughout the search the true objective (via ``evaluate_sequence``)
    is evaluated whenever a new surrogate best is found – if the real
    value drops below the target ``1.503`` the search stops early.

All operations respect the global time budget (default 1000 s).  The final
sequence is stored in the global name ``height_sequence_1`` for possible
reuse in later runs.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Should not happen, but fall‑back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """2·n·max(conv(p,p)) – the smooth surrogate used for optimisation."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-9) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Vector with exactly `m` non‑zero equal entries."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_multi_gaussian(n: int, n_peaks: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few (different) Gaussian windows, normalised."""
    a = np.zeros(n, dtype=np.float64)
    for _ in range(n_peaks):
        mu = rng.integers(0, n)
        sigma = rng.random() * (n / 5.0) + (n / 30.0)   # avoid too narrow
        i = np.arange(n, dtype=np.float64)
        a += np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s

# Optimisers
# ----------------------------------------------------------------------
def _adam_softmax_optimize(start: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           alpha_start: float = 10.0,
                           alpha_end: float = 8000.0,
                           lr: float = 0.10) -> tuple[np.ndarray, float]:
    """Adam on the smooth log‑sum‑exp surrogate."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    N = _next_pow2(3 * n - 2)            # size sufficient for both FFTs

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # FFT of current point
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        # annealed temperature
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        # stable log‑sum‑exp
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # gradient of smooth surrogate = 2·corr(x, soft)
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # projection onto simplex (ensures non‑negativity and sum=1)
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            # occasional progress report
            if t % 50000 == 0 or time.time() > deadline - 1.0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _expgrad_optimize(start: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 800_000,
                      eta0: float = 0.30,
                      sparsity_thr: float = 1e-9) -> tuple[np.ndarray, float]:
    """Mirror‑descent (multiplicative) using the true max‑convolution sub‑gradient."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # learning rate that decays like 1/√t
        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow
        x = x * np.exp(delta)

        # optional sparsity projection
        if sparsity_thr > 0.0:
            mask = x < sparsity_thr
            if np.any(mask):
                x[mask] = 0.0

        # simplex projection (renormalise)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 200_000 == 0 or time.time() > deadline - 1.0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(start: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_000_000,
                         init_temp_factor: float = 0.08,
                         min_temp: float = 1e-13) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.99994, min_temp)

        if rng.random() < 0.55:
            # reshuffle a random subset (size 6‑20)
            k_sub = rng.integers(6, 21)
            idx = rng.choice(n, size=k_sub, replace=False)
            tot = cur[idx].sum()
            if tot <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * tot
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.03)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 250_000 == 0 or time.time() > deadline - 1.0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.15, 0.22, 0.30, 0.38):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_beta_shape(n, 2.0, 3.0))
    seeds.append(_tukey_window(n, 0.45))

    # Dirichlet draws (different concentrations)
    for conc in (0.02, 0.05, 0.12, 0.25):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse equal‑weight supports
    for m in (150, 250, 350, 500):
        for _ in range(3):
            seeds.append(_random_sparse_support(n, m, rng))

    # Low‑frequency random cosine series
    for nf in (2, 3, 4):
        for _ in range(5):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            if a.sum() > 0.0:
                seeds.append(a / a.sum())

    # Random multi‑Gaussian mixtures
    for peaks in (2, 3, 4):
        for _ in range(6):
            seeds.append(_random_multi_gaussian(n, peaks, rng))

    # Mixture of uniform and sparse support
    for mix_alpha in (0.25, 0.40):
        for m in (300, 450):
            sparse = _random_sparse_support(n, m, rng)
            mixed = _mix_two(_uniform(n), sparse, mix_alpha)
            seeds.append(mixed)

    # Previous best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # final clean‑up – ensure proper normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search – within ``budget_s`` seconds – for a non‑negative step function
    that minimises the official ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Length of the step function – 8000 offers a better asymptotic bound.
    # ------------------------------------------------------------------
    n = 8000
    print(f"\n--- Length n={n} ---")

    # ------------------------------------------------------------------
    # Stage 0 – generate a large and very diverse seed library
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Created {len(seeds)} seeds.")

    # Evaluate surrogate and keep the top performing ones
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – soft‑max Adam optimisation (multiple starts)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple starts) ---")
        # number of Adam starts limited by remaining time
        for idx, seed_vec in enumerate(best_seeds[:7]):   # a few promising starts
            if time.time() >= deadline:
                break
            rem = deadline - time.time()
            # allocate roughly 0.2 s per 1000 iterations => generous budget
            max_iters = int(min(300_000, rem * 3000))
            cand, val = _adam_softmax_optimize(
                seed_vec, n, deadline,
                max_iters=max_iters,
                alpha_start=10.0,
                alpha_end=8000.0,
                lr=0.10,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

            # early stop if we already beat the target bound
            if best_val < 1.503 * 2 * n / (n):   # equivalent of objective < 1.503
                break

    # ------------------------------------------------------------------
    # Stage 2 – sparse exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sparse Exponential‑Gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(900_000, remaining * 2000))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline,
            max_iters=max_iters,
            eta0=0.30,
            sparsity_thr=1e-9,
        )
        print(f"[ExpGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(1_200_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
