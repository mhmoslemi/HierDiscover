"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.  The approach is a
light‑weight evolutionary / gradient search:

1.  **Rich seeding** – uniform, the previous best (if any), a broad set of
    analytic windows (beta, power‑law, Gaussian, sine, Tukey) and several
    random Dirichlet draws.  All vectors are ℓ₁‑normalised.

2.  **Coarse selection** – compute the cheap surrogate
    ``2·n·max(conv)`` for every seed and keep the best *K* (≈20).

3.  **Adam optimisation on a smoothed max** – each kept seed is refined
    by minimising the log‑sum‑exp surrogate of the max‑convolution.
    The optimisation works in the log‑softmax (logit) space, uses an
    Adam‑style update and a multi‑stage schedule of the smoothing
    parameter β (1 → 5 → … → 5·10⁴).  At each β‑stage a few hundred Adam
    steps are performed; the best iterate (lowest surrogate) is retained.

4.  **Final polishing** – a short exponentiated‑gradient / sub‑gradient
    pass on the exact max‑convolution for the current global best.

5.  **Evaluation** – run the official ``evaluate_sequence`` on the final
    candidate, store it in the global variable ``height_sequence_1`` and
    return the list of floats.

The whole routine respects a wall‑clock budget (default 1000 s) and
periodically checks the deadline.  No external files or network access
are used.  The implementation relies only on NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, softmax, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (ℓ₁‑norm = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective (surrogate = exact when ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(seq,seq)).  ``seq`` is assumed to sum to 1."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc`` – already ℓ₁‑normalised."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   steps_per_beta: int = 800) -> np.ndarray:
    """
    Adam optimisation in log‑softmax (logit) space.
    Returns the ℓ₁‑normalised vector with the smallest surrogate value
    encountered during the run.
    """
    # initialise logits
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v_acc = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq, n)

    for beta in beta_schedule:
        beta = float(beta)
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_seq

            a = _softmax(v)                     # current probability vector
            conv = _conv_fft(a, a)               # full convolution
            shift = np.max(conv)                 # max for numerical stability

            # log‑sum‑exp smoothing
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # (the objective itself is not needed for the gradient)
            w = exp_beta / sum_exp                # softmax weights over shifts

            # gradient w.r.t. a (see derivation in the original code)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # slice the central part

            # chain rule through the softmax parametrisation
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_acc = beta2 * v_acc + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v_acc / (1.0 - beta2 ** t)
            lr = 0.2 / (math.sqrt(t) + 1.0)      # simple decreasing schedule
            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional evaluation of the surrogate
            if (t % 100) == 0:
                seq = _softmax(v)
                val = _objective_surrogate(seq, n)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()

    return best_seq


# ----------------------------------------------------------------------
# Mirror‑descent (exp‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (final push)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (same pattern as in EG)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = _uniform(len(a))

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a tiny safety margin

    # --------------------------------------------------------------
    # Problem size – keep the same dimensionality as the previous runs
    # --------------------------------------------------------------

    # --------------------------------------------------------------
    # 0.   Pull the previous best (if any)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1.   Rich seed generation
    # --------------------------------------------------------------
    seeds = []

    # uniform baseline
    seeds.append(_uniform(n))

    # previous best, if available

        seeds.append(prev_arr)

    # analytic families with a modest sweep of parameters
    for gamma in np.linspace(0.5, 5.0, 7):
        seeds.append(_sine_window(n, gamma))

    for sigma in np.linspace(0.05, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    for alpha in np.linspace(0.2, 5.0, 7):
        for beta in np.linspace(0.2, 5.0, 7):
            seeds.append(_beta_shape(n, alpha, beta))

    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))

    # random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # a few random beta shapes sampled log‑uniformly (to explore extremes)
    for _ in range(30):
        if time.time() > deadline:
            break
        a = 10 ** rng.uniform(-2.0, 1.0)   # α ∈ [0.01, 10]
        b = 10 ** rng.uniform(-2.0, 1.0)   # β ∈ [0.01, 10]
        seeds.append(_beta_shape(n, a, b))

    # --------------------------------------------------------------
    # 2.   Coarse evaluation and selection of promising seeds
    # --------------------------------------------------------------
    scored = []
    for vec in seeds:
        if time.time() > deadline:

        val = _objective_surrogate(vec, n)
        scored.append((vec, val))

    scored.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"\n[Seed selection] evaluated {len(scored)} candidates, "
          f"keeping top {len(top_seeds)} for optimisation.")

    # --------------------------------------------------------------
    # 3.   Adam‑based smoothing optimisation for each top seed
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq, n)
    print(f"[Init] best surrogate = {global_best_val:.6f}")

    beta_schedule = [1, 5, 20, 80, 200, 500, 1000,

        remaining = deadline - time.time()
        remaining_per_seed = remaining / max(1, (len(top_seeds) - idx))
        seed_deadline = time.time() + remaining_per_seed - 0.1

        seq_opt = _adam_optimize(seed_vec,

            global_best_val = val_pg

        seq_pg, val_pg = _subgradient_opt(global_best_seq,

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")

        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
