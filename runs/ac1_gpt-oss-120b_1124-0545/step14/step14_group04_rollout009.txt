All steps respect a global 1000 s budget and the best sequence found is
returned (and stored as ``height_sequence_1`` for the next run).
"""

import time
import math
import numpy as np
from scipy.optimize import differential_evolution
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑p = 1):
        2·n·max(conv(p,p)).
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase (ℓ₁‑normalised)."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (including a very wide random sweep)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0, 5.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.15))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_gaussian(n, 0.35))
    for rate in (0.999, 0.9995, 0.9998):
        seeds.append(_exponential_decay(n, rate))
    for p in (0.3, 0.5, 0.8, 1.2, 2.0):
        seeds.append(_power(n, p))
    for a in (0.3, 0.5, 1.0, 2.0):
        for b in (0.3, 0.5, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # random Beta with tiny parameters (log‑uniform search space)
    for _ in range(30):
        log_a = rng.uniform(-2.5, 0.5)   # 10**(-2.5) .. 10**0.5  ≈ 0.003..3
        log_b = rng.uniform(-2.5, 0.5)
        seeds.append(_beta_shape(n, 10 ** log_a, 10 ** log_b))

    # random Dirichlet draws (tiny concentration)
    for conc in (0.01, 0.005, 0.001):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves (more frequencies)
    for nf in (5, 6, 7):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform (to diversify)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (3, 5):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # sparse spike patterns (followed by a tiny Gaussian blur)
    for spikes in (30, 60, 100):
        for _ in range(3):
            vec = np.zeros(n, dtype=np.float64)
            idx = rng.choice(n, size=spikes, replace=False)
            vec[idx] = 1.0
            vec /= vec.sum()
            # light blur (σ≈1 % of length)
            sigma = max(1, int(0.01 * n))
            kernel = np.exp(-np.arange(-sigma, sigma + 1) ** 2 / (2 * (sigma / 2) ** 2))
            kernel /= kernel.sum()
            vec = np.convolve(vec, kernel, mode='same')
            vec = np.maximum(vec, 0.0)
            vec /= vec.sum()
            seeds.append(vec)

    # previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final sanity check
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisers (same as the previous best solver, but with tuned defaults)
# ----------------------------------------------------------------------
def _adam_softmax_optimize(start, n, deadline, max_iters=200_000,
                           alpha_start=20.0, alpha_end=3000.0, lr=0.15):
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # gradient of the log‑sum‑exp surrogate
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val, best_x = cur_val, x.copy()
            if t % 25_000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _subgrad_optimize(start, n, deadline, max_iters=150_000, eta0=0.6):
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g = np.zeros_like(x)
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val, best_x = cur_val, x.copy()
            if it % 30_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _expgrad_optimize(start, n, deadline, max_iters=400_000, eta0=0.35):
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # protect from overflow
        x = x * np.exp(delta)
        s = x.sum()
        x = x / s if s > 0 else _uniform(n)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val, best_x = cur_val, x.copy()
            if it % 70_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(start, n, deadline, max_iters=800_000,
                         init_temp_factor=0.09, min_temp=1e-12):
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99996, min_temp)

        if rng.random() < 0.55:
            # reshuffle a random subset
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total > 0.0:
                new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
                cand = cur.copy()
                cand[idx] = new_alloc
            else:
                continue
        else:
            # tiny transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur, cur_val = cand, new_val
            if new_val < best_val:
                best_val, best = new_val, cand.copy()
                if it % 120_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Differential‑evolution on the 2‑parameter Beta family (quick global search)
# ----------------------------------------------------------------------
def _de_optimize_beta(n: int, deadline: float):
    """Runs a cheap DE on (α,β) in log‑space and returns the best β‑shape."""
    def obj(log_params):
        if time.time() > deadline:
            # return a large penalty – the optimiser will abandon the trial
            return 1e9
        a = 10 ** log_params[0]
        b = 10 ** log_params[1]
        vec = _beta_shape(n, a, b)
        return _objective_surrogate(vec)

    bounds = [(-2.5, 0.5), (-2.5, 0.5)]          # α,β ∈ [10⁻²·⁵ , 10⁰·⁵] ≈ [0.003, 3]
    result = differential_evolution(
        obj,
        bounds,
        maxiter=30,
        popsize=12,
        polish=True,
        seed=42,
        updating='deferred',
        workers=1,
        tol=1e-6,
        disp=False
    )
    best_vec = _beta_shape(n, 10 ** result.x[0], 10 ** result.x[1])
    return best_vec, result.fun

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    # ------------------------------------------------------------------
    # Choose a comfortable length – large enough for a rich shape, but
    # still cheap enough for many FFTs.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n--- Sequence length: n = {n} ---")

    # ------------------------------------------------------------------
    # Seed generation – very diverse pool
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds (including analytic families, random β‑shapes, spikes…)")

    # Score seeds with the exact surrogate (already ℓ₁‑normalised)
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"\nTop {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Quick global Beta‑shape optimisation (DE)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Differential‑evolution on Beta family ---")
        vec_beta, val_beta = _de_optimize_beta(n, deadline)
        if val_beta < scored[0][1]:
            best_seeds[0] = vec_beta
            print(f"DE found a beta shape with surrogate {val_beta:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation pipeline – multi‑seed, multi‑stage
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    for seed_idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimising seed #{seed_idx+1} ---")
        # Allocate a modest slice of the remaining budget to each seed
        remaining = deadline - time.time()
        if remaining < 60.0:          # not enough time for a full run
            break

        # Stage 1: Adam (smooth surrogate)
        max_iters_adam = int(min(120_000, remaining * 110))
        cand_adam, val_adam = _adam_softmax_optimize(
            seed_vec, n, deadline, max_iters=max_iters_adam,
            alpha_start=20.0, alpha_end=2500.0, lr=0.12
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, cand_adam.copy()
            print(f"[Adam] improved to {best_val:.6f}")

        if time.time() > deadline:
            break

        # Stage 2: Sub‑gradient descent on the exact max‑conv objective
        max_iters_sub = int(min(130_000, (deadline - time.time()) * 180))
        cand_sub, val_sub = _subgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters_sub, eta0=0.7
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, cand_sub.copy()
            print(f"[SubGrad] improved to {best_val:.6f}")

        if time.time() > deadline:
            break

        # Stage 3: Exponential‑gradient (mirror descent) polishing
        max_iters_exp = int(min(300_000, (deadline - time.time()) * 250))
        cand_exp, val_exp = _expgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters_exp, eta0=0.33
        )
        if val_exp < best_val:
            best_val, best_seq = val_exp, cand_exp.copy()
            print(f"[ExpGrad] improved to {best_val:.6f}")

        if time.time() > deadline:
            break

        # Stage 4: Simulated‑annealing fine‑tune
        max_iters_sa = int(min(600_000, (deadline - time.time()) * 400))
        cand_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters_sa
        )
        if val_sa < best_val:
            best_val, best_seq = val_sa, cand_sa.copy()
            print(f"[SA] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final long‑run Adam polish on the absolute best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long‑run Adam polish ---")
        remaining = deadline - time.time()
        max_iters_final = int(min(300_000, remaining * 180))
        best_seq, best_val = _adam_softmax_optimize(
            best_seq, n, deadline, max_iters=max_iters_final,
            alpha_start=30.0, alpha_end=5000.0, lr=0.10
        )
        print(f"[Final Adam] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
