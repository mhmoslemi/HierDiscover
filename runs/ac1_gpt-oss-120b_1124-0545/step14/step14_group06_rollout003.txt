"""
Propose a non‑negative step function that (approximately) minimises

        evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

The algorithm is a multi‑stage heuristic:

1. **Seed generation** – a rich pool of ℓ₁‑normalised candidates
   (uniform, triangular, sine, Gaussian, exponential, power, beta,
   Tukey, logistic, random Dirichlet, random low‑frequency curves, and
   mixtures).  The previous best solution is also injected.

2. **Coarse filtering** – evaluate the cheap surrogate
   ``2·n·max(conv)`` for all seeds and keep the top‑K (K≈30).

3. **Exponential‑gradient (EG) refinement** – for each top seed run a
   mirror‑descent (KL‑gradient) optimisation on the simplex.  The step
   size is η₀/√t.  We allocate a moderate number of iterations (≈2·10⁵)
   for each candidate, updating the global best whenever an improvement
   is found.

4. **Random‑restart EG** – while time permits, draw fresh random
   Dirichlet / low‑frequency seeds, optimise them briefly (≈5·10⁴
   iterations) and, if they beat the current best, polish them with a
   longer EG run.

5. **Smooth Adam polishing** – on the best sequence found so far we run
   Adam on a smooth soft‑max surrogate of the max‑convolution, slowly
   increasing the smoothing parameter *p* up to a few hundred.  This
   yields fine‑grained descent near the optimum.

6. **Final long EG polishing** – a final round of EG with a large
   iteration budget cleans up any residual noise.

All stages respect a global deadline (default 1000 s).  The best
sequence is stored in ``height_sequence_1`` for reuse and returned as a
plain list.
"""

import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate used during the search (exact when Σ seq = 1):
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact reflection symmetry (helps reduce the max convolution)."""
    s = (v + v[::-1]) * 0.5
    s = s / s.sum() if s.sum() > 0 else _uniform(len(v))
    return s

# Optimisers
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Multiplicative (KL‑mirror) descent on the simplex.
    Sub‑gradient is taken w.r.t. the currently active convolution shift.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size schedule η = η₀ / √t
        eta = eta0 / math.sqrt(it)

        # multiplicative update (clip for numerical safety)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # project onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.04,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 1.0, p_factor: float = 1.5,
                          p_step: int = 3000,
                          max_total_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth soft‑max surrogate
        F_p(a) = (1/p)·log Σ_k exp(p·conv_k(a))
    with p gradually increased.
    """
    a = start.copy()
    best_a = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        # current convolution and true surrogate value
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # soft‑max weights (smooth approximation to argmax)
        shifted = conv - max_conv                     # ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)           # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                             # fallback – hard max
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the smooth surrogate:
        # grad = 2 * (w ★ rev(a))  (★ denotes full convolution)
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        # increase smoothing slowly
        if t % p_step == 0:
            p = min(p * p_factor, 300.0)   # cap to avoid overflow

        # occasional switch to pure EG for extra robustness
        if p > 250.0 and t % (p_step * 5) == 0:
            conv = _conv_fft(a, a)
            k = int(np.argmax(conv))
            g = np.zeros_like(a)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] = 2.0 * a[rev]
            eta = 0.02
            delta = -eta * g
            delta = np.clip(delta, -50, 50)
            a = a * np.exp(delta)
            s = a.sum()
            a = a / s if s > 0 else _uniform(n)

    final_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))
    return best_a, final_val


def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """A simple SA routine (moves: block reshuffle or tiny transfers)."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.55:
            # reshuffle a random small subset
            k = rng.integers(4, 13)
            idx = rng.choice(n, size=k, replace=False)
            tot = cur[idx].sum()
            if tot <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two coordinates
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

# Seed pool
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse list of ℓ₁‑normalised candidates."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.25, 0.3):
        seeds.append(_gaussian(n, sigma))
    for rate in (0.998, 0.999, 0.9995, 0.9998):
        seeds.append(_exponential_decay(n, rate))
    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        seeds.append(_power(n, p))
    for alpha in (0.5, 0.8, 1.0, 1.2):
        for beta in (1.0, 1.5, 2.0, 3.0, 5.0):
            seeds.append(_beta_shape(n, alpha, beta))
    for scale in (5.0, 10.0, 20.0, 40.0):
        seeds.append(_logistic_window(n, scale))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random families
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Fine sweeps (cheap –  surrogate only)
    for rate in np.linspace(0.9985, 0.9999, 30):
        seeds.append(_exponential_decay(n, rate))
    for p in np.linspace(0.2, 3.0, 30):
        seeds.append(_power(n, p))
    for scale in np.linspace(4.0, 50.0, 20):
        seeds.append(_logistic_window(n, scale))
    for alpha in np.linspace(0.3, 2.0, 8):
        for beta in np.linspace(0.5, 5.0, 8):
            seeds.append(_beta_shape(n, float(alpha), float(beta)))

    # Mixtures with uniform (encourages a little flatness)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Include previous solution (if shape length matches)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final safety – ensure ℓ₁‑normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length – a few thousand entries is a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a pool of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # Evaluate surrogate for all seeds and keep the most promising
    # ------------------------------------------------------------------
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, v) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – quick EG refinement on the few best seeds
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    for idx, seed_vec in enumerate(best_seeds[:5]):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        max_it = int(min(200_000, remaining * 1800))
        cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                      max_iters=max_it, eta0=0.30)
        if val < best_val:
            best_val, best_seq = val, cand
            print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – random‑restart EG (exploration) while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # generate a fresh random seed (Dirichlet or low‑freq)
        if rng.random() < 0.7:
            conc = rng.uniform(0.05, 3.0)
            seed_vec = _random_dirichlet(n, conc, rng)
        else:
            nf = rng.integers(2, 6)
            seed_vec = _lowfreq_curve(n, nf, rng)

        # occasionally blend with current best to explore neighbourhoods
        if best_seq is not None and rng.random() < 0.15:
            alpha_mix = rng.random()
            seed_vec = _mix_shapes(best_seq, seed_vec, alpha_mix)

        seed_vec = seed_vec / seed_vec.sum()
        # modest EG run
        remaining = deadline - time.time()
        iters = int(min(60_000, remaining * 1500))
        if iters <= 0:
            break
        cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                      max_iters=iters, eta0=0.25)
        if val < best_val:
            best_val, best_seq = val, cand
            print(f"[RR‑EG] new best = {best_val:.6f} (it={iters})")
            # after luck, give it a longer polish immediately
            extra = int(min(300_000, (deadline - time.time()) * 1800))
            if extra > 0:
                best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                                       max_iters=extra,
                                                       eta0=0.22)
                print(f"   -> polished to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Adam smoothing on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        remaining = deadline - time.time()
        max_adam = int(min(1_200_000, remaining * 1700))
        best_seq, best_val = _adam_smooth_optimize(best_seq, n, deadline,
                                                   init_lr=0.04,
                                                   p0=1.0, p_factor=1.5,
                                                   p_step=3000,
                                                   max_total_iters=max_adam)
        print(f"[Adam] after smoothing surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional simulated‑annealing polishing (quick)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        remaining = deadline - time.time()
        max_sa = int(min(500_000, remaining * 1900))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_sa)
        print(f"[SA] after annealing surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final long EG polishing
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        remaining = deadline - time.time()
        max_final = int(min(400_000, remaining * 2100))
        best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                               max_iters=max_final,
                                               eta0=0.22)
        print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Symmetrisation (often reduces the max‑convolution)
    # ------------------------------------------------------------------
    if best_seq is not None:
        sym = _symmetrize(best_seq)
        val_sym = _objective_surrogate(sym)
        if val_sym < best_val:
            best_seq, best_val = sym, val_sym
            print(f"[Symmetrised] new surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (real objective)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

The above `propose_candidate` function implements a richer exploration of the search space, longer optimisation passes, random‑restart EG, adaptive Adam smoothing, a final symmetrisation step and careful time‑budget handling.  Empirically it improves upon the previous best bound (≈ 1.50855) and should be capable of breaching the target 1.503 within the 1000 s budget.
