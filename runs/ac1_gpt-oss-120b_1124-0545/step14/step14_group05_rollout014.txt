"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The search combines a very diverse seed pool,
soft‑max Adam polishing, an additive sub‑gradient descent on the true
max‑convolution, and a final simulated‑annealing fine‑tuning.  New
analytic families (inverse‑power, exponential‑decay, sorted Dirichlet)
are added to explore strongly skewed shapes that tend to lower the
maximum of the self‑convolution.  The routine respects a total time
budget (default 1000 s) and stores the best sequence in the global name
``height_sequence_1``.
"""

import time

# Helper utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(p,p)) for a probability vector (∑p=1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Vector with exactly `m` non‑zero equal entries."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _inv_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _exp_decay(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    if not (0.0 < r < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    return a / a.sum()


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw sorted in decreasing order (strongly skewed)."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]
    return a / a.sum()


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()

def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha_start: float = 20.0,
    alpha_end: float = 20000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth log‑sum‑exp surrogate."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)

        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # grad = 2 * corr(p, soft)
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # project onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 25000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _subgrad_descent_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_lr: float = 0.5,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent on the true max‑convolution.
    Each step moves opposite to the gradient of the active shift
    and projects back onto the simplex.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    lr = init_lr

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # compute sub‑gradient of conv[k] w.r.t. x
        grad = np.zeros_like(x)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            grad[i] += x[j]
            if i != j:
                grad[j] += x[i]

        # gradient descent step (note: we want to *reduce* the max)
        x = x - lr * grad
        # project onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        lr *= decay  # slowly shrink step size

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.25,
    sparsity_thr: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent (multiplicative) on the surrogate."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        if sparsity_thr > 0.0:
            mask = x < sparsity_thr
            if np.any(mask):
                x[mask] = 0.0

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 150_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_000_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.5:
            # reshuffle a random subset (size 5‑20)
            k_sub = rng.integers(5, 21)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of normalised candidate vectors."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.15))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_tukey_window(n, 0.5))

    # new families
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_inv_power(n, p))
    for r in (0.999, 0.9995, 0.9998):
        seeds.append(_exp_decay(n, r))

    # random Dirichlet draws (different concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sorted Dirichlet (strongly decreasing)
    for conc in (0.01, 0.05, 0.1):
        for _ in range(3):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # exact‑size sparse supports
    for m in (150, 250, 350, 450):
        for _ in range(2):
            seeds.append(_random_sparse_support(n, m, rng))

    # low‑frequency random curves
    for nf in (2, 3):
        for _ in range(4):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                seeds.append(a / s)

    # mixtures of uniform and a sparse support
    for mix_alpha in (0.3, 0.5):
        for m in (250, 350):
            sparse = _random_sparse_support(n, m, rng)
            mixed = _mix_two(_uniform(n), sparse, mix_alpha)
            seeds.append(mixed)

    # previous best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # final clean‑up (ensure proper normalisation)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

            cand, val = _adam_softmax_optimize(
                seed_vec, n, deadline,
                max_iters=max_iters,
                alpha_start=20.0,
                alpha_end=20000.0,
                lr=0.12,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – additive sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Additive sub‑gradient descent ---")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 300))
        best_seq, best_val = _subgrad_descent_optimize(
            best_seq, n, deadline, max_iters=max_iters, init_lr=0.5, decay=0.9996
        )
        print(f"[SubGrad] best surrogate after descent = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – sparse exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sparse Exponential‑Gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline,
            max_iters=max_iters,
            eta0=0.25,
            sparsity_thr=1e-8,
        )
        print(f"[Sparse EG] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(1_000_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
