def _peak_shave_optimize(start, n, deadline):
    cur = start.copy()
    best = cur.copy()
    best_val = _objective_surrogate(cur)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))
        # find a pair (i, j) that contributes most to conv[max_idx]
        # i ranges max(0, max_idx-(n-1)) to min(n-1, max_idx)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        # compute contributions c_i = cur[i] * cur[max_idx - i]
        idxs = np.arange(i_start, i_end + 1)
        rev = max_idx - idxs
        contrib = cur[idxs] * cur[rev]
        # pick i with largest contrib
        i_max = idxs[np.argmax(contrib)]
        # propose to transfer mass from i_max or rev_max to a random other pos
        # choose direction (reduce from both?).
        # Reduce cur[i_max] and cur[rev_max] by epsilon each, distribute to random j,k.
        eps = 0.001 * min(cur[i_max], cur[rev_max])
        if eps <= 0: 
            # if zero skip
            continue
        # reduce both
        cand = cur.copy()
        cand[i_max] -= eps
        cand[rev] -= eps # rev = max_idx - i_max; careful rev may equal i_max
        # ensure nonnegative
        if cand[i_max] < 0 or cand[rev] < 0: continue
        # distribute total 2*eps to random positions (maybe same).
        # pick two random indices different from i_max, rev
        j1, j2 = rng.choice(n, size=2, replace=False)
        cand[j1] += eps
        cand[j2] += eps
        # project onto simplex (maybe adjust)
        total = cand.sum()
        cand = cand / total  # renorm helps keep sum=1
        new_val = _objective_surrogate(cand)
        # accept if improves (or with prob.)
        if new_val < best_val:
            best_val = new_val
            best = cand.copy()
        if new_val < _objective_surrogate(cur) or rng.random() < 0.05: # maybe accept sometimes
            cur = cand
    return best, best_val

"""
Propose a non‑negative step function (list of heights) that lowers the official
evaluate_sequence objective. The search consists of:

1.  A very diverse seed pool – analytical windows, power‑law,
    exponential / linear ramps, beta families, many Dirichlet draws
    (including sorted draws) and random low‑frequency sinusoids.
2.  A short soft‑max Adam polishing of the top seeds (smooth approximation
    of the max via log‑sum‑exp).
3.  An exponential‑gradient (mirror‑descent) stage that directly follows
    the active‑set sub‑gradient of the current maximum convolution entry.
4.  A simulated‑annealing fine‑tuner that reshuffles mass in random subsets.
5.  A dedicated “peak‑shave’’ local search that repeatedly reduces the
    contribution of the most offending convolution shift.
All stages respect the global deadline (budget_s seconds) and the best
vector is stored in the global name ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple simplex projection (non‑negative, unit‑sum)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact official objective for sum == 1)
# ----------------------------------------------------------------------
def _objective_surrogate(p: np.ndarray) -> float:
    """Return 2·n·max(conv(p,p)) for a probability vector (∑p = 1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic / parametric families (all return unit‑sum vectors of length n)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _linear_increasing(n: int) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _linear_decreasing(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey (cosine‑taper) window, α∈[0,1]."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_dirichlet_sorted(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw then sort descending (highly asymmetric)."""
    a = _random_dirichlet(n, conc, rng)
    a = np.sort(a)[::-1]           # descending
    # re‑normalise (sorting does not change sum but keep for safety)
    return a / a.sum()


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Exactly `m` non‑zero equal entries (still sum to 1)."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()

def _adam_softmax_optimize(
    start: np.ndarray,
    deadline: float,
    max_iters: int = 80_000,
    alpha_start: float = 20.0,
    alpha_end: float = 5000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Adam on the smoothed surrogate  L = (1/α)·log Σ exp(α·conv)."""
    n = len(start)
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full auto‑convolution via FFT (real)
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        # schedule α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)

        # log‑sum‑exp smooth max
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # gradient of smooth surrogate = 2·corr(x, soft)
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # projection onto probability simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.25,
    sparsity_thr: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Multiplicative mirror‑descent on the true max‑convolution objective."""
    n = len(start)
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift (only the overlapping indices)
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        # clamp to avoid overflow in exp
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        if sparsity_thr > 0.0:
            mask = x < sparsity_thr
            if np.any(mask):
                x[mask] = 0.0

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    deadline: float,
    max_iters: int = 600_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Classic simulated‑annealing with two simple move types."""
    rng = np.random.default_rng()
    n = len(start)
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.5:
            # reshuffle a random subset (5‑15 indices)
            k_sub = rng.integers(5, 16)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between a random pair
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
    return best, best_val


def _peak_shave_optimize(
    start: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    eps_frac: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Targeted local search: repeatedly identify the shift with the highest
    convolution value, locate the pair (i, j) that contributes most to this
    peak and move a tiny amount of mass from that pair to two random sites.
    This “shaves’’ the peak while preserving the total sum.
    """
    rng = np.random.default_rng()
    n = len(start)
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline and max_iters > 0:
        conv = _conv_fft(cur, cur)
        k = int(np.argmax(conv))               # shift with biggest overload
        # overlapping index range for this shift
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        contrib = cur[idx] * cur[rev]          # pairwise contributions
        # the most harmful pair
        pair_i = idx[np.argmax(contrib)]
        pair_j = rev[np.argmax(contrib)]

        # amount to move – a small fraction of the smaller member of the pair
        eps = eps_frac * min(cur[pair_i], cur[pair_j])
        if eps <= 0.0:
            # if one of them vanished, fall back to a random tiny move
            i, j = rng.choice(n, size=2, replace=False)
            eps = eps_frac * cur[i]
            if eps <= 0.0:
                max_iters -= 1
                continue
            cand = cur.copy()
            cand[i] = max(0.0, cand[i] - eps)
            cand[j] += eps
        else:
            # remove eps from both members and redistribute 2*eps elsewhere
            cand = cur.copy()
            cand[pair_i] -= eps
            cand[pair_j] -= eps
            # choose two (possibly identical) random destinations
            d1, d2 = rng.choice(n, size=2, replace=False)
            cand[d1] += eps
            cand[d2] += eps

        # numerical safety – clip negatives and re‑normalise
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s <= 0.0:
            cand = _uniform(n)
        else:
            cand = cand / s

        new_val = _objective_surrogate(cand)
        # accept if improvement; otherwise keep current (no temperature here)
        if new_val < cur_val:
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
        max_iters -= 1

    return best, best_val

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a rich and diverse library of normalised candidate vectors."""
    seeds = []

    # Basic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_increasing(n))
    seeds.append(_linear_decreasing(n))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))

    # Power‑law and exponential families
    for p in (0.5, 1.0, 1.5, 2.0):
        seeds.append(_power_law(n, p))
    for lam in (0.0005, 0.001, 0.002, 0.005):
        seeds.append(_exponential_decay(n, lam))

    # Beta distribution grid (asymmetric shapes too)
    for a in (0.2, 0.5, 1.0, 2.0, 5.0):
        for b in (0.2, 0.5, 1.0, 2.0, 5.0):
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws – both raw and sorted (highly asymmetric)
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
            seeds.append(_random_dirichlet_sorted(n, conc, rng))

    # Exact‑size sparse supports (various cardinalities)
    for m in (150, 250, 400):
        for _ in range(2):
            seeds.append(_random_sparse_support(n, m, rng))

    # Low‑frequency random Fourier constructions
    for nf in (2, 3, 4):
        for _ in range(3):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            if a.sum() > 0.0:
                seeds.append(a / a.sum())

    # Mixtures of uniform and a sparse support
    for mix_alpha in (0.3, 0.5):
        for m in (250, 350):
            sparse = _random_sparse_support(n, m, rng)
            seeds.append(_mix_two(_uniform(n), sparse, mix_alpha))

    # Previous best (if provided)
    if previous is not None and previous.size == n:
        if previous.sum() > 0.0:
            seeds.append(previous / previous.sum())

    # Final clean‑up – enforce proper normalisation and positivity
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            clean.append(_uniform(n))
        else:
            clean.append(v / s)
    return clean


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the official
    `evaluate_sequence` objective.  A hierarchical pipeline (seed generation,
    Adam polishing, exponential‑gradient, simulated‑annealing, peak‑shave)
    is executed under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # 2 s safety margin

    # ------------------------------------------------------------------
    # Length of the step function – a power‑of‑two size speeds up FFTs.
    # ------------------------------------------------------------------
    n = 4096
    print(f"\n=== Starting search: n={n} (budget {budget_s}s) ===")

    # ------------------------------------------------------------------
    # Stage 0 – build a heterogeneous library of seeds
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # Evaluate surrogate and retain the most promising ones
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k], 1):
        print(f"  #{i:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – soft‑max Adam (multiple restarts on the best seeds)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple starts) ---")
        for idx, seed_vec in enumerate(best_seeds[:8]):   # up to 8 restarts
            if time.time() >= deadline:
                break
            # allocate a slice of remaining time to this run
            remaining = deadline - time.time()
            # aim for ~70 k iterations per run if time permits
            max_iters = min(70_000, int(remaining * 120))
            cand, val = _adam_softmax_optimize(
                seed_vec, deadline, max_iters=max_iters,
                alpha_start=20.0,
                alpha_end=5000.0,
                lr=0.12,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – sparse exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(700_000, remaining * 1000))
        best_seq, best_val = _expgrad_optimize(
            best_seq, deadline, max_iters=max_iters,
            eta0=0.25,
            sparsity_thr=1e-8,
        )
        print(f"[ExpGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 800))
        best_seq, best_val = _simulated_annealing(
            best_seq, deadline, max_iters=max_iters
        )
        print(f"[SA] surrogate after SA = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – peak‑shave targeted local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Peak‑shave local optimisation ---")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 600))
        best_seq, best_val = _peak_shave_optimize(
            best_seq, deadline, max_iters=max_iters, eps_frac=0.001
        )
        print(f"[PeakShave] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose the best sequence for potential reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
