"""
Non‑negative step‑function optimizer (≈ 1 k s budget).

Strategy
--------
* A moderate problem size (n = 5 000) gives a finer discretisation of the
  continuous optimal shape while keeping FFT costs low.
* A large, diverse seed pool is built from analytic windows, Dirichlet /
  exponential draws and random mixtures (including the previous best).
* For each promising seed we run a continuation L‑BFGS on a log‑sum‑exp
  smooth approximation of the ‖a∗a‖∞ objective.  The β‑schedule is
  extended far beyond the previous implementation (up to 5·10⁴) and the
  per‑β iteration budget is increased.
* The best vector from L‑BFGS is refined by a long exponentiated‑gradient
  (mirror‑descent) run on the *exact* max‑convolution objective.  The
  initial step‑size η₀ is larger (2.0) than before, which speeds early
  descent, and the step‑size decays as 1/√t.
* After EG we apply a few random perturbations + short L‑BFGS re‑optimisations
  (a simple “shake‑and‑polish” phase) to escape possible local minima.
* A final high‑β L‑BFGS polishing pass yields the returned sequence.
* All vectors are kept ℓ₁‑normalised (sum = 1); the evaluator is scale‑invariant,
  and the algorithm respects the non‑negative / finite‑value checks.

The routine prints timing and objective information and stores the best
vector in the global name ``height_sequence_1``.
"""

import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all vectors are ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Modified Bessel function I0 from NumPy
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential‐decay window a_i ∝ exp(-lam·|i‑center|)."""
    if lam <= 0.0:
        return _uniform(n)
    c = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * np.abs(i - c))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture (1‑α)·a + α·b, ℓ₁‑renormalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (log‑sum‑exp smoothing) and its gradient
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) for the smoothed max‐convolution.
    Variable v are logits (softmax → probability vector a).
    """
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)               # full linear autocorrelation
    shift = np.max(conv)                 # numerical stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # smoothed max = shift + log(sum_exp)/beta
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # soft weights for each lag
    w = exp_term / sum_exp               # length = 2n‑1
    # gradient w.r.t a : 2·conv(rev_a, w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # Chain rule: Jacobian(softmax)·grad_a
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    L‑BFGS‑B on the smooth surrogate.
    Variable is the logits v = log a (a = softmax(v)).
    """
    if beta_schedule is None:
        # extended schedule – higher β allows a tighter approximation to max
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val}")
            break
        except Exception as e:               # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact max‑convolution objective and sub‑gradient
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) (sum(a)=1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a (active‑set = single maximal lag).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    # g_i = 2 * a_{idx-i} for all i where the pair contributes
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g


def _logits_gradient(a: np.ndarray, grad_a: np.ndarray) -> np.ndarray:
    """Convert ∂f/∂a into ∂f/∂logits (softmax Jacobian)."""
    dot = np.dot(grad_a, a)
    return a * (grad_a - dot)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 2_000_000,
                          eta0: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Uses the classic exponentiated‑gradient update in log‑space.
    Returns the best vector seen and its objective value.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # sub‑gradient of the currently active lag
        g = _subgradient_maxconv(a)

        # step size ηₜ = η₀ / √t
        eta = eta0 / math.sqrt(it)

        # EG update in log‑space
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        # bookkeeping every 100 k iterations
        if it % 100_000 == 0:
            cur = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation (deterministic families + random draws + mixtures)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))
    for a in np.linspace(0.5, 3.0, 7):
        for b in np.linspace(0.5, 3.0, 7):
            seeds.append(_beta_shape(n, a, b))
    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))
    for lam in (0.001, 0.003, 0.01, 0.03, 0.1):
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet / exponential draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of existing seeds
    base_idx = np.arange(len(seeds))
    for _ in range(150):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Include previous solution if available
    if previous is not None and previous.size == n:
        s = previous / previous.sum()
        seeds.append(s)

    # Clean any numerical negatives and re‑normalise
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (plain list of floats).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Problem size – a compromise between discretisation quality and speed
    # ------------------------------------------------------------------
    n = 5000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Seed pool & quick surrogate ranking
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # surrogate = 2·n·max(conv) (since sum=1)
    def surrogate(v):
        return 2.0 * n * np.max(_conv_fft(v, v))

    evaluated = [(vec, surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 7
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on smooth surrogate
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 2000.0, 5000.0,
                  10000.0, 20000.0, 50000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all L‑BFGS seeds.")
            break

        remaining = deadline - time.time()
        per_beta_iters = max(1200, int(min(8000,
                                          remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={per_beta_iters}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=per_beta_iters)
        val_opt = surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient refinement on exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_500_000,
                                remaining * 0.25 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_exp_iters,
                                                      eta0=2.0)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved exact objective to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – shake‑and‑polish: random perturbations + short L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Shake‑and‑Polish phase (random perturbations) ---")
        attempts = 5
        for att in range(attempts):
            if time.time() > deadline:
                break
            # multiplicative noise (log‑normal perturbation)
            noise = rng.normal(loc=0.0, scale=0.02, size=n)   # 2 % log‑noise
            pert = global_best_seq * np.exp(noise)
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            # short L‑BFGS (low‑β only)
            per_beta_iters = 1500
            seq2 = _lbfgs_smooth_optimize(pert,
                                          n,
                                          deadline,
                                          beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0],
                                          max_iter_per_beta=per_beta_iters)
            val2 = surrogate(seq2)
            print(f"[Shake {att+1}] surrogate = {val2:.6f}")
            if val2 < global_best_val:
                global_best_seq = seq2.copy()
                global_best_val = val2
                print(f"  -> improvement! new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final high‑β L‑BFGS polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(800, int(min(4000,
                                          remaining * 0.07 * 1e5)))
        print("\n--- Final polishing L‑BFGS (high β) ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[20000.0, 50000.0],
                                           max_iter_per_beta=per_beta_iters)
        val_final = surrogate(seq_final)
        print(f"[Final‑L‑BFGS] surrogate = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation expands the β‑schedule, uses a larger step‑size for the exponentiated‑gradient phase, adds an extra “shake‑and‑polish” stage with random perturbations, and employs a slightly larger problem size (n = 5000) to reduce discretisation error. All work respects the 1000‑second budget and returns the best non‑negative height sequence found.
