        if val < best_val:
            best_val = val
            best_seq = s

    print(f"Initial best surrogate {best_val}")

    # main loop: perform random perturbation + L-BFGS/smoothing
    while time.time() < deadline:
        # pick random seed from seeds or also perturb current best
        if rng.random() < 0.3:
            # fresh seed random Dirichlet
            start_vec = _random_dirichlet(n, conc= rng.uniform(0.05,2.0), rng=rng)
        else:
            # perturb best
            logit = np.log(best_seq + 1e-30)
            sigma = 0.1 * (deadline - time.time()) / (deadline - start_time)  # decaying
            pert = logit + sigma * rng.normal(size=n)
            start_vec = _softmax(pert)

        # run L-BFGS with schedule
        seq_opt = _lbfgs_smooth_optimize(start_vec, n, deadline,
                                         beta_schedule=[1,5,20,80,200,500,1000,2000,5000,10000,20000],
                                         max_iter_per_beta=2000)
        val_opt = _objective_surrogate(seq_opt)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt
            print(f"New best surrogate {best_val}")

        # short exponentiated gradient polish
        seq_exp, val_exp = _exp_gradient_opt(best_seq, n, deadline, max_iters=50_000)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = seq_exp
            print(f"After expgrad best {best_val}")

        # break if time near end
        if deadline - time.time() < 5:
            break

    # final evaluation
    true_val = evaluate_sequence(best_seq.tolist())
    print(f"Final with evaluate_sequence = {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

Improved search for a non‑negative step‑function that minimises

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm proceeds in three stages while respecting a 1000 s time budget.

1. **Seed generation**
   • A rich pool of ℓ₁‑normalised vectors is assembled from analytic families,
     Dirichlet draws of several concentrations and many shuffled variants
     (alternating order, modular step permutations, random permutations).
   • The cheap surrogate 2·n·max(conv) (exact for ℓ₁‑normalised vectors) ranks the
     seeds; the best *top_k* seeds are kept for heavy optimisation.

2. **Continuation L‑BFGS + exponentiated‑gradient polishing**
   • Each top seed is refined with an L‑BFGS‑B optimiser on a smooth
     log‑sum‑exp approximation of the max‑convolution.
   • A short exponentiated‑gradient (mirror‑descent with softmax) polish
     follows each L‑BFGS run.

3. **Long sub‑gradient descent and simulated annealing**
   • Using the current global best, a projected sub‑gradient descent runs for a
     large number of iterations, lowering the true max‑convolution.
   • Finally a simulated‑annealing walk in log‑softmax space explores the
     neighbourhood further.

All phases continuously check the deadline; the best sequence found is stored
in the global variable ``height_sequence_1`` and returned as a plain list of
floats.  The final value is obtained by calling the official ``evaluate_sequence``.

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by the evaluator (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from the left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


def _shuffle_by_step(sorted_desc: np.ndarray, step: int) -> np.ndarray:
    """
    Distribute the sorted values over the index set using a modular step.
    ``step`` should be coprime with n to obtain a permutation.
    """
    n = len(sorted_desc)
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val in sorted_desc:
        out[pos] = val
        pos = (pos + step) % n
    return out

def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse collection of ℓ₁‑normalised candidate vectors.
    """
    seeds: list[np.ndarray] = []

    # --- Analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.18, 0.25):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # --- Random Dirichlet draws + shuffles ---------------------------------
    conc_list = (0.05, 0.2, 0.5, 1.0, 2.0)
    step_list = [3, 7, 11, 13, 17, 19, 23, 29, 31, 37]      # coprime to 4000
    for conc in conc_list:
        for _ in range(4):
            raw = _random_dirichlet(n, conc, rng)
            # raw itself (already normalised)
            seeds.append(raw)

            # descending order (helps spreading large values)
            desc = np.sort(raw)[::-1]
            seeds.append(desc)

            # alternating order (large–small–large–small)
            seeds.append(_alternating_order_from_vec(raw))

            # modular step permutations (5 different steps)
            for step in step_list[:5]:
                seeds.append(_shuffle_by_step(desc, step))

            # a random permutation (extra diversity)
            perm = rng.permutation(n)
            seeds.append(raw[perm])

    # --- Previous solution -------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly (protect against NaNs)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on the simplex
            conv = _conv_fft(a, a)                # full conv
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                # soft‑weights for gradient
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Exponentiated‑gradient (mirror descent) optimiser (smooth surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Short exponentiated‑gradient polish on the exact max‑conv surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv) for ℓ₁‑normalised vectors."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]      # surrogate = exact for ℓ₁‑normed a

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Basic initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size – 4000 is a good trade‑off between granularity and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Seed pool construction
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # cheap surrogate (exact for ℓ₁‑normed vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12                       # keep a modest number for heavy optimisation
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS + ExpGrad polishing on the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # β‑schedule (aggressive) – same for each seed
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 1000.0, 2000.0, 5000.0,
                     10000.0, 20000.0, 50000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing all L‑BFGS runs.")
            break

        # allocate a slice of the remaining time to this seed
        remaining = deadline - time.time()
        # be generous but never exceed a few thousand iterations per β
        max_iter_beta = int(min(5000, remaining * 0.06 * 1e5))
        max_iter_beta = max(1500, max_iter_beta)

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/β = {max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # --------------------------------------------------------------
        # 3a. Short exponentiated‑gradient polishing
        # --------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(400_000, remaining * 0.05 * 1e5))
            max_exp = max(150_000, max_exp)
            seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(1_200_000, remaining * 0.05 * 1e5))
        if max_sub > 0:
            print("\n--- Long sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing refinement (last phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(800_000, remaining * 0.04 * 1e5))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing_opt(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
