"""
Hybrid search for the *minimum* of the official ``evaluate_sequence`` objective.

Main ideas
-----------
1. **Rich seed pool** – generate many ℓ₁‑normalised candidates:
   * classic analytic windows (uniform, triangular, sine, Tukey, …)
   * β‑shapes with a dense grid of α,β (including values < 0.5)
   * edge‑exponential, parabolic, low‑frequency random curves
   * Dirichlet / exponential random draws
   * mixtures of the above (including the previous best solution).

2. **Fast surrogate** – the objective after ℓ₁‑normalisation is
   ``2·n·max(conv(a,a))``.  Using an FFT based convolution is cheap
   (O(n log n)) and gives the exact value that ``evaluate_sequence`` will
   return (no clipping occurs for the generated vectors).

3. **Two‑phase local optimisation**
   * **Projected sub‑gradient descent** on the *exact* surrogate, with an
     adaptive step size and a simple back‑tracking line‑search.  The
     update direction is the true sub‑gradient of the active convolution
     entry.
   * **Mirror‑descent (exponential‑gradient)** as a polish, keeping the
     probability‑simplex constraints via a KL‑projection.
   * Finally a short **simulated‑annealing** sweep (tiny mass transfers
     and subset re‑draws) to escape possible plateaux.

4. **Time aware** – the global budget (default 1000 s) is split among the
   stages; each stage checks the deadline and falls back to the best
   solution found so far.

The routine returns a plain ``list[float]`` (the step heights) that is
ready for ``evaluate_sequence``.  The best vector is also stored in the
global ``height_sequence_1`` variable for warm‑starts in later runs.
"""

import time

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, ∑w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑p = 1):
        2·n·max(conv(p,p))
    (identical to ``evaluate_sequence`` after the security clamping).
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Peak in the centre (classic tent)."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete symmetric (or asymmetric) Beta‑type shape."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _parabolic(n: int) -> np.ndarray:
    """Concave parabola, zero at the edges, peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - ((i - c) / c) ** 2
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _edge_exponential(n: int, gamma: float = 1.0) -> np.ndarray:
    """Symmetric exponential decay away from the edges."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-gamma * d)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()

def _collect_seeds(

    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    # dense Beta‑shape grid (including α,β < 0.5)
    alphas = np.concatenate([np.linspace(0.2, 0.9, 8),
                             np.linspace(1.0, 3.0, 9)])
    betas  = alphas
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # Tukey windows
    for alpha in (0.1, 0.25, 0.4, 0.6):
        seeds.append(_tukey_window(n, alpha))

    # Edge‑exponential family – robust for low maximum correlation
    for gamma in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        seeds.append(_edge_exponential(n, gamma))

    # random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform – provides mild perturbations
    for mix_alpha in (0.1, 0.3, 0.5):
        for _ in range(4):
            curve = _lowfreq_curve(n, 3, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # final sanity‑check
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

def _subgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact surrogate.
    Simple back‑tracking line search (geometric factors) is added to
    increase robustness near a plateau.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift (see original code)
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size – classic diminishing schedule
        base_eta = eta0 / math.sqrt(it)

        # try a few geometric variants and keep the best improvement
        best_local = x
        best_local_val = best_val
        for factor in (0.5, 1.0, 1.5, 2.0):
            cand = _proj_simplex(x - base_eta * factor * g)
            val = _objective_surrogate(cand)
            if val < best_local_val:
                best_local_val = val
                best_local = cand

        x = best_local
        cur_val = best_local_val

        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

        # occasional progress report
        if it % 50_000 == 0:
            print(f"[SubGrad] it={it:7d}  best={best_val:.6f}")

    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL (multiplicative) updates."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # prevent overflow/underflow
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:7d}  best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-13,
) -> tuple[np.ndarray, float]:
    """Tiny random transfers + subset reshuffles, accepting worsening moves."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-13)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99996, min_temp)

        if rng.random() < 0.55:
            # reshuffle a random subset
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total > 0.0:
                new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
                cand = cur.copy()
                cand[idx] = new_alloc
            else:
                continue
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.08)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d}  best={best_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Sequence length – a reasonable compromise between resolution and speed.
    n = 4000
    print(f"\n=== Optimisation run (n = {n}) ===")

    # ------------------------------------------------------------------
    # Stage 0 – seed generation
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    # Quick surrogate scoring
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"\nTop {top_k} surrogate scores after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Projected sub‑gradient descent (aggressive)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Sub‑gradient optimisation (first pass) ---")
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 300))
        best_seq, best_val = _subgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters, eta0=0.8
        )
        print(f"[SubGrad] best after first pass = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Second‑pass sub‑gradient from several good seeds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient optimisation (multiple starts) ---")
        # try a few additional seeds if time permits
        for idx, seed_vec in enumerate(best_seeds[1:5]):   # up to 4 extra starts
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            iters = int(min(150_000, remaining * 250))
            cand, val = _subgrad_optimize(seed_vec, n, deadline, max_iters=iters, eta0=0.6)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[SubGrad] seed #{idx+2} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Exponential‑gradient polishing (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(best_seq, n, deadline, max_iters=max_iters)
        print(f"[ExpGrad] best after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing final tweak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing fine‑tune ---")
        remaining = deadline - time.time()
        max_iters = int(min(500_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline, max_iters=max_iters)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for warm‑start in the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
