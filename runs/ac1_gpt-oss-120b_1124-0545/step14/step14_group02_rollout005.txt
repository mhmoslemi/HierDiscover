def _adam_smooth_optimize(start_vec, n, deadline, max_iters, beta=5000):
    logit = np.log(start_vec + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    best_vec = start_vec
    best_val = _objective_surrogate(best_vec)
    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        a = _softmax(logit)
        conv = _conv_fft(a, a)
        shift = np.max(conv)  # numeric shift
        # compute smoothed weights with beta
        exp_beta = np.exp(beta * (conv - shift))
        w = exp_beta / np.sum(exp_beta)
        # compute grad wrt a

def _collect_seeds(n, rng, previous, max_random=2500):

    # deterministic families
    seeds.append(_uniform(n))

    seeds.append(_sine_window(n, gamma=1))

    seeds.append(_gaussian(n, sigma_frac=0.15))

    # random parametric families

        # Beta shape

        seeds.append(_beta_shape(n, a, b))
        # Power law
        p = rng.uniform(0.0,5.0)
        seeds.append(_power(n,p))
        # Exponential decay

        seeds.append(_exp_decay(n, lam))
        # Linear power

        seeds.append(_linear_power(n, p_lin))
        # Parabolic

        p_par = rng.uniform(0.5,10.0)
        seeds.append(_parabolic(n, w_frac, p_par))
        # Block uniform
        block_frac = rng.uniform(0.1,0.5)

        # Mixture uniform + beta random shape
        mix_alpha = rng.uniform(0,1)
        b_shape = _beta_shape(n,
                              10 ** rng.uniform(-2,1),
                              10 ** rng.uniform(-2,1))
        seeds.append(_mix_shapes(_uniform(n), b_shape, mix_alpha))

    # Dirichlet draws
    for conc in (0.05,0.2,0.5,1.0,2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Low freq curves
    for nf in (2,3,4,5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution
    if previous is not None:
        seeds.append(previous / previous.sum())

    # final sanitization
    cleaned = []

def _adam_optimize(start_seq, n, deadline, max_iters=None):
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_vec = start_seq.copy()
    best_val = _objective_surrogate(best_vec)
    t = 0
    while True:
        if time.time() >= deadline:
            break
        if max_iters is not None and t >= max_iters:
            break
        t += 1
        a = _softmax(logit)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        beta_max = 5000.0
        beta = 1.0 + (beta_max - 1.0) * min(t / 200000.0, 1.0)
        exp_beta = np.exp(beta * (conv - shift))
        w = exp_beta / np.sum(exp_beta)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        dot = np.dot(grad_a, a)
        grad_logit = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_logit
        v = beta2 * v + (1 - beta2) * (grad_logit ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.5 / (np.sqrt(t) + 1.0)
        logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            a_cur = _softmax(logit)
            cur_val = _objective_surrogate(a_cur)
            if cur_val < best_val:
                best_val = cur_val
                best_vec = a_cur.copy()
                print(f"[Adam] iter {t:,} new best surrogate {best_val:.6f}")

    return best_vec

"""
Optimise a non‑negative step‑function (a probability vector) to minimise

    2·n·max(conv(a, a)) / (sum(a)²)

using a two‑phase approach:
1.  A smooth Adam optimisation on a log‑sum‑exp approximation of the max‑convolution.
    The smoothing parameter β is increased gradually so that early steps act on a
    smooth landscape while later steps focus on the true maximum.
2.  A long mirror‑descent (exponentiated‑gradient) refinement on the exact
    max‑convolution.

The algorithm works on several randomly generated seeds (uniform, triangular,
sine‑windows, Gaussians, power‑laws, Beta‑like shapes, exponential decay,
parabolic blocks, low‑frequency random curves, Dirichlet draws, …).  The best
seed is optimised, then up‑sampled to a finer grid and refined again.
All work is stopped before the given time budget, and the best vector found is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                    # should never happen, but guard anyway
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq, seq)) for a ℓ₁‑normalised vector.
    If ∑seq = 1 the surrogate equals the true objective.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _parabolic(n: int, w_frac: float, p: float) -> np.ndarray:
    """Parabolic block: a_i ∝ (1 - ((i-c)/(w/2))^2)^p on a central window."""
    w = max(1, int(round(w_frac * n)))
    half = w // 2
    c = (n - 1) // 2
    a = np.zeros(n, dtype=np.float64)
    lo = max(0, c - half)
    hi = min(n, c + half + 1)
    x = np.arange(lo, hi, dtype=np.float64) - c
    radius = w / 2.0
    inner = 1.0 - (x / radius) ** 2
    inner = np.maximum(inner, 0.0) ** p
    a[lo:hi] = inner
    return a / a.sum()


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0 / block_len
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation – a diverse pool of normalised vectors
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 2500,
) -> list[np.ndarray]:
    """Return a large, diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families (always useful)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # Randomised parametric families
    for _ in range(max_random // 10):
        # beta‑like
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        # power‑law
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        # exponential decay
        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        # linear‑power decreasing
        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # parabolic block
        w_frac = rng.uniform(0.2, 1.0)
        p_par = rng.uniform(0.5, 10.0)
        seeds.append(_parabolic(n, w_frac, p_par))

        # block‑uniform
        block_frac = rng.uniform(0.1, 0.5)
        seeds.append(_block_uniform(n, block_frac, rng))

        # mixture of uniform and a beta shape
        mix_alpha = rng.uniform(0.0, 1.0)
        b_sh = _beta_shape(
            n,
            10 ** rng.uniform(-2, 1),
            10 ** rng.uniform(-2, 1),
        )
        seeds.append(_mix_shapes(_uniform(n), b_sh, mix_alpha))

    # Dirichlet draws with various concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Preserve the previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Final sanitisation – guarantee non‑negativity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Adam optimiser on a smooth log‑sum‑exp approximation of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> np.ndarray:
    """
    Adam optimiser on a smoothed version of   max(conv(a,a)).
    Returns the best ℓ₁‑normalised vector seen before ``deadline``.
    """
    # initialise logits (log‑softmax parameters)
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_vec = start_seq.copy()
    best_val = _objective_surrogate(best_vec)

    t = 0
    while True:
        if time.time() >= deadline:
            break
        if max_iters is not None and t >= max_iters:
            break
        t += 1

        # current probability vector
        a = _softmax(logit)

        # convolution and smooth max weights
        conv = _conv_fft(a, a)                     # length 2n‑1
        shift = np.max(conv)                       # numeric shift for stability
        # smoothing parameter β grows linearly to β_max
        beta_max = 5000.0
        beta = 1.0 + (beta_max - 1.0) * min(t / 200_000.0, 1.0)

        exp_beta = np.exp(beta * (conv - shift))   # all ≤0 ⇒ safe
        w = exp_beta / np.sum(exp_beta)            # soft‑max weights (sum to 1)

        # gradient w.r.t. a  (see derivation in the description)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]      # central n entries

        dot = np.dot(grad_a, a)
        grad_logit = a * (grad_a - dot)            # back‑prop through softmax

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.5 / (np.sqrt(t) + 1.0)              # simple decaying step‑size
        logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report and best‑track
        if t % 5000 == 0:
            cur = _softmax(logit)
            cur_val = _objective_surrogate(cur)
            if cur_val < best_val:
                best_val = cur_val
                best_vec = cur.copy()
                print(f"[Adam] iter {t:,} new best surrogate {best_val:.6f}")

    return best_vec


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) on the true max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600_000,
) -> np.ndarray:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses a sub‑gradient of the active convolution entry.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # active sub‑gradient (only the indices that form the maximal entry)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 20000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Mirror] it={it:,} new best surrogate {best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid (preserves total mass)."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()
