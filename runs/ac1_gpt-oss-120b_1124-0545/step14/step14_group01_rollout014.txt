def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised) – used for seed generation
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a large list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families (the same set used previously)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_gaussian(n, 0.20))
    for sigma in np.linspace(0.10, 0.45, 9):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws – a much larger set than before
    conc_set = (0.05, 0.10, 0.20, 0.40, 0.80, 1.5, 3.0)
    for conc in conc_set:
        for _ in range(12):            # 12 draws per concentration
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Previous solution (if size matches)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise (defensive)
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Momentum exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_momentum(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 500_000,
                           eta0: float = 0.5,
                           beta: float = 0.9,
                           active_k: int = 5,
                           jitter_every: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Uses a few of the largest convolution entries (active_k) to form the
    sub‑gradient and adds exponential momentum.  Occasionally adds a tiny
    random jitter to the logits to escape flat regions.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current convolution (full linear conv)
        conv = _conv_fft(a, a)
        max_val = float(np.max(conv))

        # pick the active_k largest indices (indices of the current max)
        if active_k >= len(conv):
            active_idxs = np.arange(len(conv))
        else:
            # argpartition is O(N)
            active_idxs = np.argpartition(conv, -active_k)[-active_k:]

        # sub‑gradient for each active entry
        g = np.zeros_like(a)
        for k in active_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        g /= float(len(active_idxs))

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # learning‑rate schedule ηₜ = η₀ / √t
        eta = eta0 / math.sqrt(it)
        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional jitter to avoid stagnation
        if jitter_every and (it % jitter_every == 0):
            rng = np.random.default_rng()
            log_a += 0.01 * rng.normal(size=n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Refine a vector with L‑BFGS on a smooth max‑convolution surrogate."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    # Parameterisation in log‑softmax space
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on the simplex
            conv = _conv_fft(a, a)                # linear convolution
            max_c = float(np.max(conv))
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = float(exp_beta.sum())
            # smooth approximation of max (log‑sum‑exp)
            obj = 2.0 * n * (shift + math.log(sum_exp) / beta_val)

            # gradient of the smooth max w.r.t. a
            w = exp_beta / sum_exp                # softmax‑like weights (sum=1)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Main entry point required by the challenge harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the challenge
    evaluation function.  The routine respects the time budget (≈1000 s)
    and returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin for the final call to the official evaluator
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size – a compromise between expressiveness and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # cheap surrogate for ℓ₁‑normalised vectors
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"\nBest surrogate values among the first {top_k} seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Momentum exponentiated‑gradient polishing (most work is done here)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    # allocate a fraction of the remaining time for the gradient stage
    remaining_total = deadline - time.time()
    per_seed_budget = remaining_total / max(1, top_k)

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all seeds.")
            break

        # give each seed its own deadline
        seed_deadline = time.time() + per_seed_budget
        max_iters = 500_000          # a generous limit; the loop stops earlier if time runs out

        print(f"\n--- Exponentiated‑gradient (momentum) on seed #{idx + 1} ---")
        seq_opt, val_opt = _exp_gradient_momentum(seed_vec,
                                                  n,
                                                  seed_deadline,
                                                  max_iters=max_iters,
                                                  eta0=0.5,
                                                  beta=0.9,
                                                  active_k=5,
                                                  jitter_every=200_000)
        print(f"[ExpGrad] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional brief L‑BFGS polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # a modest L‑BFGS run – keep it cheap
        max_iter_per_beta = int(min(3000, remaining * 15))  # very rough estimate
        max_iter_per_beta = max(800, max_iter_per_beta)
        print("\n--- Short L‑BFGS polishing ---")
        try:
            seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq,
                                               n,
                                               deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1000.0],
                                               max_iter_per_beta=max_iter_per_beta)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
            print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
            if val_lbfgs < global_best_val:
                global_best_seq = seq_lbfgs.copy()
                global_best_val = val_lbfgs
                print(f"[Update] after L‑BFGS best surrogate = {global_best_val:.6f}")
        except Exception as exc:  # pragma: no cover
            print("L‑BFGS polishing failed:", exc)

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
