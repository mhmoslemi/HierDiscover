"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.  The optimisation proceeds in
three phases:

1. **Rich seeding** – a wide collection of analytic families (uniform,
   triangular, beta, power‑law, Gaussian, Tukey, windows), a few random
   Dirichlet draws and the previous best solution (if any).  All seeds are
   ℓ₁‑normalised.

2. **Continuation L‑BFGS** – smooths the ℓ∞‑objective by a log‑sum‑exp
   surrogate and runs L‑BFGS‑B with an aggressive β‑schedule that reaches
   10⁶, allowing the optimiser to approach the exact max‑convolution.

3. **Fine polishing** – a large‑scale Adam‑style mirror‑descent on the
   exact max‑convolution (sub‑gradient w.r.t. the soft‑max logits) for as
   many iterations as the remaining time budget permits.  The best vector
   seen during polishing is returned.

The routine respects the 1000 s wall‑clock budget, prints progress, stores
the best sequence globally as ``height_sequence_1`` and finally returns it
as a plain list of floats.
"""

import time
import math
import numpy as np

# Optional SciPy import for L‑BFGS‑B
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate used during optimisation:
    for ℓ₁‑normalised vectors it equals 2·n·max(conv).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

# Continuation L‑BFGS on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimises the log‑sum‑exp surrogate of the max‑convolution using
    L‑BFGS‑B.  Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser required for L‑BFGS.")

    if beta_schedule is None:
        # Very aggressive schedule reaching β = 1e6
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0,
                         5_000.0, 10_000.0, 20_000.0,
                         50_000.0, 100_000.0, 500_000.0, 1_000_000.0]

    # Initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            # a lives on the simplex
            a = _softmax(v_local)

            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # log‑sum‑exp smoothing
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp          # shape (2n-1,)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2, focus on centre part
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] Exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a

# Adam‑style mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_maxconv_opt(start_vec: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the exact max‑convolution (sub‑gradient of the max).
    Returns the best vector and its surrogate value.
    """
    a = start_vec.copy()
    logits = np.log(a + 1e-30)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # current probabilities
        a = _softmax(logits)

        # exact max‑convolution
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the max entry w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t. logits (mirror‑descent)
        dot = np.dot(g, a)
        grad_logit = a * (g - dot)

        # Adam step (learning‑rate decays like 1/√t)
        lr = 0.12 / math.sqrt(t)
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate surrogate occasionally
        if t % 5000 == 0 or time.time() > deadline:
            a = _softmax(logits)
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] iter={t:8d} best surrogate={best_val:.6f}")

    return best_seq, best_val

    seeds.append((_tukey_window(n, alpha=0.8), 'tukey0.8'))
    seeds.append((_gaussian(n, sigma_frac=0.15), 'gauss0.15'))
    seeds.append((_gaussian(n, sigma_frac=0.30), 'gauss0.30'))
    seeds.append((_power(n, p=0.5), 'power0.5'))
    seeds.append((_power(n, p=1.0), 'power1.0'))

    # beta‑shape grid (coarse)
    alphas = np.linspace(0.3, 4.0, 7)
    betas = np.linspace(0.3, 4.0, 7)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            vec = _beta_shape(n, a, b)
            seeds.append((vec, f'beta_{a:.2f}_{b:.2f}'))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            seeds.append((vec, f'dirichlet_{conc}'))

    # low‑frequency random curves
    for freq in (2, 3, 4):
        if time.time() > deadline:
            break
        vec = _lowfreq_curve(n, num_freq=freq, rng=rng)
        seeds.append((vec, f'lowfreq_{freq}'))

    # previous best, if any
    if prev_arr is not None:
        seeds.append((prev_arr, 'previous'))

    # ------------------------------------------------------------------
    # 2. Evaluate and keep the most promising seeds
    # ------------------------------------------------------------------
    print("\n--- Evaluating seeds ---")
    seed_vals = []
    for vec, name in seeds:
        if time.time() > deadline:
            break
        val = _eval_seed(vec)
        seed_vals.append((val, vec, name))

    seed_vals.sort(key=lambda x: x[0])
    top_k = 10
    top_seeds = [vec for _, vec, _ in seed_vals[:top_k]]
    print(f"Collected {len(seed_vals)} seeds, retaining top {top_k}.")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS on the best seeds
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break

        print(f"\n--- L‑BFGS run #{idx+1}/{len(top_seeds)} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                          n,
                                          deadline,
                                          max_iter_per_beta=4000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam‑style mirror‑descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate up to a few hundred thousand Adam steps (fast enough)
        max_adam_iters = int(min(1_200_000, remaining * 500.0))
        print("\n--- Adam mirror‑descent polishing ---")
        seq_adam, val_adam = _adam_maxconv_opt(best_seq,
                                               n,
                                               deadline,
                                               max_iters=max_adam_iters)
        if val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"[Adam] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
