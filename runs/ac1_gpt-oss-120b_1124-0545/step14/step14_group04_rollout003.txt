"""
Propose a step‑function with minimal `evaluate_sequence` value.

Algorithm
---------
1. **Seed generation** – deterministic windows (uniform, triangular,
   sine, Gaussian, power, beta, Tukey), random Dirichlet draws,
   low‑frequency cosine curves and, when available, a DPSS window.
   Parameters of the analytic families are sampled uniformly in
   sensible ranges, producing a diverse pool of ℓ₁‑normalised candidates.

2. **Coarse selection** – evaluate every seed with the exact surrogate
   `2·n·max(conv(p,p))` (fast via FFT) and keep the best K seeds.

3. **Smooth optimisation** – run a few short Adam optimisers on the
   log‑sum‑exp surrogate of the max‑convolution, starting from the top
   seeds.  The surrogate becomes tighter as the temperature parameter
   α grows, allowing the optimiser to find a good basin.

4. **Exact sub‑gradient polishing** – perform additive sub‑gradient
   descent on the true objective (projected onto the probability simplex).

5. **Hill‑climbing perturbations** – repeatedly pick a small random subset
   of coordinates, re‑distribute their total mass using a Dirichlet draw,
   and accept the change if the surrogate improves.

6. **Exponential‑gradient polishing** – a final mirror‑descent run (KL
   projection) to squeeze out any remaining slack.

7. **Evaluation** – compute the official `evaluate_sequence` value,
   store the sequence globally for the next round and return it.

All stages respect a global time budget (default 1000 s).  The code
requires only `numpy` and, optionally, `scipy.signal` (for DPSS windows).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑p = 1):
        2·n·max(conv(p,p)).
    This matches ``evaluate_sequence`` after the security clamping.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sum a few cosine components with random amplitude/phase;
    ℓ₁‑normalised.
    """
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Optimisers
# ----------------------------------------------------------------------
def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha_start: float = 20.0,
    alpha_end: float = 2000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        L(p) = (1/α)·log Σ exp[α·(p*p)[k]].
    α grows linearly from ``alpha_start`` to ``alpha_end``.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        # log‑sum‑exp trick for stability
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # gradient of the surrogate
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # project onto simplex (ensures non‑negativity + ℓ₁‑norm 1)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 20_000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _subgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent on the exact objective
        f(a) = 2·n·max(conv(a,a)),
    with Euclidean projection onto the probability simplex.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 25_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with KL divergence (multiplicative updates).
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow/underflow
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Helper: hill‑climbing with random subset re‑allocation
# ----------------------------------------------------------------------
def _hill_climb(
    current: np.ndarray,
    cur_val: float,
    n: int,
    deadline: float,
    steps: int,
) -> tuple[np.ndarray, float]:
    """Tiny random redistributions on a random subset of coordinates."""
    rng = np.random.default_rng()
    cur = current.copy()
    val = cur_val
    for it in range(steps):
        if time.time() > deadline:
            break
        # pick a subset size between 5 and 20
        k_sub = rng.integers(5, 21)
        idx = rng.choice(n, size=k_sub, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.ones(k_sub)) * total
        cand = cur.copy()
        cand[idx] = new_alloc
        new_val = _objective_surrogate(cand)
        if new_val < val:
            cur = cand
            val = new_val
    return cur, val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official `evaluate_sequence` objective within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 points yields a power‑of‑two length for FFTs.
    # --------------------------------------------------------------
    n = 4096
    print(f"\n--- Sequence length: n = {n} ---")

    # --------------------------------------------------------------
    # Stage 0 – rich seed pool
    # --------------------------------------------------------------
    print("\n--- Seed generation ---")
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_gaussian(n, sigma_frac=0.2))
    seeds.append(_gaussian(n, sigma_frac=0.35))
    seeds.append(_power(n, p=1.0))
    seeds.append(_power(n, p=2.0))
    seeds.append(_beta_shape(n, alpha=0.5, beta=0.5))
    seeds.append(_tukey_window(n, alpha=0.5))

    # randomised analytic families
    for _ in range(25):
        seeds.append(_sine_window(n, gamma=rng.uniform(0.3, 5.0)))
    for _ in range(25):
        seeds.append(_gaussian(n, sigma_frac=rng.uniform(0.08, 0.45)))
    for _ in range(25):
        seeds.append(_power(n, p=rng.uniform(0.2, 4.5)))
    for _ in range(20):
        a = rng.uniform(0.2, 4.0)
        b = rng.uniform(0.2, 4.0)
        seeds.append(_beta_shape(n, alpha=a, beta=b))
    for _ in range(20):
        seeds.append(_tukey_window(n, alpha=rng.uniform(0.0, 1.0)))

    # DPSS windows (if SciPy is present)
    try:
        from scipy.signal import windows
        for _ in range(12):
            NW = rng.uniform(2.0, 6.0)
            seeds.append(windows.dpss(n, NW, sym=True).astype(np.float64))
    except Exception:
        pass

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency cosine mixtures
    for nf in (2, 3, 4):
        for _ in range(8):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with a uniform baseline
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous best (if available)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # sanitise: non‑negative and ℓ₁‑normalised
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            clean.append(v / s)

    # --------------------------------------------------------------
    # Stage 1 – evaluate and keep the top performing seeds
    # --------------------------------------------------------------
    print(f"Generated {len(clean)} seeds.")
    scored = [(vec, _objective_surrogate(vec)) for vec in clean]
    scored.sort(key=lambda x: x[1])
    K = 20
    top_seeds = [vec for vec, _ in scored[:K]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – short Adam runs from the very best seeds
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam optimisation (multiple starts) ---")
        for idx, seed_vec in enumerate(top_seeds[:8]):
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            max_iters = int(min(180_000, remaining * 140))  # rough budget
            cand, val = _adam_softmax_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=max_iters,
                alpha_start=15.0,
                alpha_end=2500.0,
                lr=0.10,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – additive sub‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 280))
        best_seq, best_val = _subgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters, eta0=0.65
        )
        print(f"[SubGrad] best surrogate after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – stochastic hill‑climb refinements
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Hill‑climb refinements ---")
        climbs = 4
        for i in range(climbs):
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            steps = int(min(300_000, remaining * 450))
            best_seq, best_val = _hill_climb(best_seq, best_val, n, deadline, steps)
            print(f"[HillClimb] iteration {i+1} best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – final exponential‑gradient polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final exponential‑gradient polish ---")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 1400))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters, eta0=0.22
        )
        print(f"[ExpGrad] final surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
