def _lp_refine_step(a, t=0.2, max_active=200):
    n = len(a)
    conv = _conv_fft(a, a)
    M = conv.max()
    # collect active shift indices: top max_active values
    idx_sorted = np.argsort(-conv)[:max_active]  # descending
    A = np.zeros((len(idx_sorted), n), dtype=np.float64)
    for row_i, k in enumerate(idx_sorted):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i index in f (a), column j = i_range? Let's compute columns j = i_min..i_max.
        # For each i in [i_start, i_end], j = k - i.
        # So we have: A[row_i, j] = a[i].
        # We can vectorize: i_arr = np.arange(i_start, i_end+1); j_arr = k - i_arr
        # A[row_i, j_arr] = a[i_arr]
        i_arr = np.arange(i_start, i_end+1, dtype=int)
        j_arr = k - i_arr
        A[row_i, j_arr] = a[i_arr]
    # Right-hand side vector: M * ones
    b = np.full(len(idx_sorted), M, dtype=np.float64)

    # Solve LP: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)  # minimize c^T x = -sum g => maximize sum
    # bounds (0, None)
    bounds = [(0, None)] * n

    # Use linprog
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        return a  # no change
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a
    # normalize g
    g = g / sum_g
    # mix
    new_a = (1 - t) * a + t * g
    # normalize to sum 1
    new_a = new_a / new_a.sum()
    return new_a

"""Optimization for the convolution‑max objective.

This version builds on the previous multi‑stage pipeline but adds a
linear‑programming refinement step.  Given a current non‑negative vector
`a` (ℓ₁‑normalised) we solve the LP

    max   Σ b_j
    s.t.  (a * b)_k ≤ M  for a subset of shifts k,
          b_j ≥ 0

where `M` is the current maximal self‑convolution value.
The LP pushes mass to indices where `a_j` is smallest, which tends to
lower the maximal convolution when the two vectors are mixed.
We repeatedly mix the obtained `b` with the current vector (convex
combination) and continue a short EG/Adam polishing.
The whole routine respects the global 1000 s deadline.

The main entry point is :func:`propose_candidate`, which returns the best
list of heights found and stores it in ``height_sequence_1``.
"""
import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective when Σ seq = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (including fine sweeps)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidates."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.25, 0.3):
        seeds.append(_gaussian(n, sigma))
    for rate in (0.998, 0.999, 0.9995, 0.9998):
        seeds.append(_exponential_decay(n, rate))

    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        seeds.append(_power(n, p))

    for alpha in (0.5, 0.8, 1.0, 1.2):
        for beta in (1.0, 1.5, 2.0, 3.0, 5.0):
            seeds.append(_beta_shape(n, alpha, beta))

    for scale in (5.0, 10.0, 20.0, 40.0):
        seeds.append(_logistic_window(n, scale))

    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    for nf in (2, 3):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # fine sweeps – cheap surrogate only
    for rate in np.linspace(0.9985, 0.9999, 30):
        seeds.append(_exponential_decay(n, rate))

    for p in np.linspace(0.2, 3.0, 30):
        seeds.append(_power(n, p))

    for scale in np.linspace(4.0, 50.0, 20):
        seeds.append(_logistic_window(n, scale))

    for alpha in np.linspace(0.3, 2.0, 8):
        for beta in np.linspace(0.5, 5.0, 8):
            seeds.append(_beta_shape(n, float(alpha), float(beta)))

    # include previous if present
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final sanity – ensure ℓ₁‑normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 200_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)
        # project onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.04,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 1.0, p_factor: float = 1.5,
                          p_step: int = 2000,
                          max_total_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        F_p(a) = (1/p)·log Σ_k exp(p·conv_k(a))
    with a gentle increase of the smoothing parameter p.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        shifted = conv - max_conv               # ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)     # safety
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)        # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        if t % p_step == 0:
            p = min(p * p_factor, 300.0)

    final_val = _objective_surrogate(best_a)
    return best_a, final_val


def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.55:
            k = rng.integers(4, 13)
            idx = rng.choice(n, size=k, replace=False)
            tot = cur[idx].sum()
            if tot <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
    return best, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_refine_step(a: np.ndarray, t: float = 0.2, max_active: int = 200) -> np.ndarray:
    """
    One LP‑based refinement step.

    Parameters
    ----------
    a : np.ndarray
        Current ℓ₁‑normalised vector (sum == 1, non‑negative).
    t : float
        Mixing factor for the convex combination with the LP solution.
    max_active : int
        Number of highest‑convolution shifts to keep as constraints.
        Using a subset makes the LP cheap while still targeting the
        bottleneck positions.

    Returns
    -------
    np.ndarray
        New vector after mixing (renormalised to sum = 1).  If the LP
        fails or produces a zero‑vector, the original ``a`` is returned.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    M = conv.max()
    # select active shift indices – either top values or all if very few
    if max_active >= conv.size:
        active = np.arange(conv.size, dtype=int)
    else:
        active = np.argpartition(-conv, max_active - 1)[:max_active]

    # Build dense constraint matrix (len(active) × n)
    m = active.shape[0]
    A = np.zeros((m, n), dtype=np.float64)

    for row, k in enumerate(active):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_arr = np.arange(i_start, i_end + 1, dtype=int)
        j_arr = k - i_arr
        # place a_i values
        A[row, j_arr] = a[i_arr]

    b = np.full(m, M, dtype=np.float64)

    # Linear program: maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n
    # Using HiGHS via scipy.optimize.linprog (fast for medium‑size LPs)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})

    if not res.success or res.x is None:
        # fallback – keep the original vector
        return a

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0 or np.isnan(sum_g):
        return a

    # Normalise g to a probability vector
    g = g / sum_g

    # Convex combination and renormalisation
    new_a = (1.0 - t) * a + t * g
    s = new_a.sum()
    if s <= 0.0:
        return a
    new_a = new_a / s
    return new_a

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    The routine builds a diverse seed pool, applies aggressive
    exponential‑gradient and Adam polishing, then performs a handful of
    linear‑programming refinements before a final simulated‑annealing /
    EG polish.
    The best sequence is stored in the global ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # length of the sequence – a compromise between expressive power and speed
    n = 4000
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # evaluate all seeds (cheap surrogate) and keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, v) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # initialise from the best seed
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – aggressive EG refinement (more iterations, larger eta)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Aggressive EG refinement ---")
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline,
            max_iters=300_000,
            eta0=0.35
        )
        print(f"[EG] surrogate after aggressive refinement = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam smooth optimisation (larger max_iters)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam smooth optimisation ---")
        best_seq, best_val = _adam_smooth_optimize(
            best_seq, n, deadline,
            init_lr=0.05,
            p0=1.0,
            p_factor=1.5,
            p_step=1500,
            max_total_iters=1_800_000
        )
        print(f"[Adam] surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – linear‑programming refinement loop
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP refinement loop ---")
        lp_steps = 0
        while time.time() < deadline and lp_steps < 20:
            # try a larger mixing factor first
            new_seq = _lp_refine_step(best_seq, t=0.25, max_active=250)
            new_val = _objective_surrogate(new_seq)
            if new_val < best_val:
                best_seq, best_val = new_seq, new_val
                print(f"[LP] step {lp_steps+1:2d} improved to {best_val:.6f}")
                lp_steps += 1
                continue
            # if no improvement, try a smaller step
            new_seq = _lp_refine_step(best_seq, t=0.10, max_active=250)
            new_val = _objective_surrogate(new_seq)
            if new_val < best_val:
                best_seq, best_val = new_seq, new_val
                print(f"[LP-small] step {lp_steps+1:2d} improved to {best_val:.6f}")
                lp_steps += 1
                continue
            # otherwise stop the LP loop
            break
        print(f"LP loop finished after {lp_steps} successful steps; current surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing (shorter, but still useful)

"""Optimization for the convolution‑max objective.

This version builds on the earlier multi‑stage pipeline but adds a
linear‑programming refinement step.  Given the current normalised vector
`a` we solve

    max   Σ b_j
    s.t.  (a * b)_k ≤ M    for a selected set of shifts k,
          b_j ≥ 0

where `M = max(conv(a, a))`.  The LP concentrates mass on indices where
`a` is smallest, which tends to lower the maximal convolution after a
convex combination.  The LP step is repeated a few times, followed by
standard EG/Adam polishing and a final simulated‑annealing pass.
The routine respects the 1000 s deadline and stores the best sequence in
the global variable ``height_sequence_1``.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective when Σ seq = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds(n: int, rng: np.random.Generator,

    """Return a diverse list of ℓ₁‑normalised candidates."""

    # deterministic families

    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):

    for alpha in (0.5, 0.8, 1.0, 1.2):

def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 200_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.04,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 1.0, p_factor: float = 1.5,
                          p_step: int = 2000,
                          max_total_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        F_p(a) = (1/p)·log Σ_k exp(p·conv_k(a))
    with a gentle increase of the smoothing parameter p.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        shifted = conv - max_conv
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0.0 else _uniform(n)

        if t % p_step == 0:
            p = min(p * p_factor, 300.0)

    final_val = _objective_surrogate(best_a)
    return best_a, final_val


def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.55:
            k = rng.integers(4, 13)
            idx = rng.choice(n, size=k, replace=False)
            tot = cur[idx].sum()
            if tot <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
    return best, best_val

# Linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_refine_step(a: np.ndarray, t: float = 0.2, max_active: int = 200) -> np.ndarray:
    """
    Single LP‑based refinement.

    Parameters
    ----------
    a : np.ndarray
        Current ℓ₁‑normalised vector (sum == 1, non‑negative).
    t : float
        Mixing factor for the convex combination with the LP solution.
    max_active : int
        Number of top convolution shifts used as constraints.

    Returns
    -------
    np.ndarray
        Updated vector (renormalised to sum = 1).  If the LP fails,
        the original `a` is returned unchanged.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    M = conv.max()

    # choose active shift indices (largest values)
    if max_active >= conv.size:
        active = np.arange(conv.size, dtype=int)
    else:
        active = np.argpartition(-conv, max_active - 1)[:max_active]

    m = active.shape[0]
    A = np.zeros((m, n), dtype=np.float64)

    for row, k in enumerate(active):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_arr = np.arange(i_start, i_end + 1, dtype=int)
        j_arr = k - i_arr
        A[row, j_arr] = a[i_arr]

    b = np.full(m, M, dtype=np.float64)

    # maximise sum(g)  ->  minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})

    if not res.success or res.x is None:
        return a

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0 or np.isnan(sum_g):
        return a

    g = g / sum_g
    new_a = (1.0 - t) * a + t * g
    s = new_a.sum()
    if s <= 0.0:
        return a
    new_a = new_a / s
    return new_a
